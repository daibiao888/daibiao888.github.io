𖼳𖽡 𖼣𖽔𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏
𖽃𖽡𖾐 𖼸𖽙𖾐 𖽐𖼞𖽻𖾏，𖼞𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏.
87
𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑.
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼮𖽑𖽠𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼑𖽡𖾏 𖽃𖽔 𖼛𖽻𖾏，
𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐.
3 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼘𖽪！
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
4 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼖𖽔𖾑 𖼦𖽔𖾐 𖼀𖽳 (𖼖𖽔𖾑 𖼦𖽔𖾐 𖼀𖽳 𖼟𖽪𖾏 𖼣𖽹 𖼮𖽡 ) 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝,
𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑.
𖼑𖽔𖾏 𖼘𖽪！ 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖽐𖼎𖽙 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑.
5 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼳𖽡 𖼣𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏:
“𖼷𖽷𖾑 𖼐𖽡𖾑 𖼷𖽷𖾑 𖼉𖽺𖾏 𖼖𖽔𖾏 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 .”
6 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼡𖽑𖽳𖾏 𖼋𖽠𖾏: “𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑.” （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
7 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏, 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑.”
