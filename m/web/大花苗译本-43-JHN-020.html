（𖼊𖽑𖽹𖾏 28：1-8；𖼞𖽑𖽙𖾐 16：1-8；𖼖𖽪𖾏 24：1-12）
20
𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑，𖼄𖽙 𖼊𖽔𖾏 𖼖𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏，𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼺𖽘𖾏 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏，𖼐𖽱𖾏 𖼁𖽙𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏， 2 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼏𖽘 𖼋𖽔𖾏 𖼚𖽔𖾏𖼳𖽡𖾑 𖼄𖽞 · 𖼀𖽡𖾐 𖼊𖽳𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖽐𖼚𖽙 𖼮𖽑𖽳𖾑 𖼮𖽵𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼀𖽡𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼮𖽡𖽝𖾏．” 3 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼅𖽻𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 ． 4 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼏𖽘，𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑 𖼏𖽘 𖼇𖽺𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼊𖽳，𖼺𖽪𖾐 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏， 5 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙， 𖼁𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼺𖽷𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼁𖽺𖾏 𖼅𖽻𖾏． 6 𖼳𖽡𖾑 𖼄𖽞 · 𖼀𖽡𖾐 𖼊𖽳 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻，𖼎𖽺𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏，𖼎𖽺𖾏 𖼁𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐． 7 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐，𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼺𖽘𖾐 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐． 8 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏，𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖽃𖽡𖾑，𖼎𖽺𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏，𖼐𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖽐𖼮𖽜 𖼋𖽘𖾏． 9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼀𖽻𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏，𖼟𖽪𖾏：“ 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼯𖽡𖽝𖾏 𖼗𖽙𖾏 ” 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏． 10 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 𖼙𖽠𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽔𖾏
（𖼊𖽑𖽹𖾏 28：9-10；𖼞𖽑𖽙𖾐 16：9-11）
11 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼦𖽡𖾑 𖽐𖼷𖽘𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼐𖽡𖽝𖾑．𖼐𖽱𖾏 𖼊𖽑𖽻𖾏 𖼯𖽪𖾏 𖼐𖽡𖽝𖾑 𖼊𖽑𖽻𖾏 𖼯𖽪𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼚𖽙 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏． 12 𖼎𖽺𖾏 𖼁𖽙𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼚𖽠𖾐，𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼑𖽜𖾏，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽡𖽙𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼺𖽘𖾑 𖽃𖽡𖾑，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽡𖽙𖾐 𖼊𖽠𖾏 𖼺𖽘𖾑 𖽃𖽡𖾑． 13 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔 𖼁𖽙𖾏， 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽝𖾑？” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ：“ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽳𖾑 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼙𖽠𖾏，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼮𖽡𖽝𖾏．” 14 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙， 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑． 15 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔 𖼁𖽙𖾏， 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽝𖾑 ？𖼟𖽡 𖽐𖼎𖽑𖽦𖾑 𖼡𖽔 𖼋𖽱𖾏？” 𖽃𖽔 𖼁𖽙𖾏 𖼸𖽙𖾏 𖼞𖽑𖽡𖽝𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼻𖽙𖾑 𖼉𖽘𖾏 ，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪！𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽑𖽳𖾑 𖼐𖽱𖾏 𖼙𖽠𖾏，𖼷𖽑𖽜𖾑 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼮𖽑𖽳𖾑 𖼐𖽱𖾏 𖼗𖽙𖾏．” 16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏．” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖼯𖽡𖽙𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼖𖽔𖾑 𖼀𖽙𖾑 𖼐𖽡！” （ “ 𖼖𖽔𖾑 𖼀𖽙𖾑 𖼐𖽡”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼗𖽪𖾏 𖼟𖽪𖾏 “ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 ” ）． 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼞𖽪𖾑 𖼊𖽡𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏．𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡，𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏， 𖼟𖽪𖾏 𖼅𖽡 𖼉𖽺𖾏，𖼚𖽔𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑．” 18 𖼄𖽙 𖼊𖽔𖾏 𖼖𖽔𖾑𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏，𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼋𖽘𖾏．” 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏，𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼑𖽔𖾏
（𖼊𖽑𖽹𖾏 28：16-20；𖼞𖽑𖽙𖾐 16：14-18；𖼖𖽪𖾏 24：36-49）
19 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑，𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏， 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽺𖾏 𖼡𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼮𖽡𖽝𖾏．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽔𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪！” 20 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼺𖽘𖾑 𖼖𖽡𖾏 𖼊𖽘𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏．𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼎𖽺𖾏 𖼙𖽔． 21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪！𖼉𖽺𖾏 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼅𖽡 𖼖𖽡𖾏 𖼯𖽤𖾏．” 22 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼷𖽑𖽔𖾐 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼀𖽘𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ，𖼐𖽑𖽡𖾏： “𖼅𖽡 𖼷𖽹𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾！ 23 𖼅𖽡 𖼯𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼾𖽔𖾏，𖼡𖽔 𖼋𖽱𖾏 𖼾𖽔𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏，𖼅𖽡 𖼷𖽷𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼾𖽔𖾏 𖼮𖽡𖽝𖾏，𖼡𖽔 𖼋𖽱𖾏 𖼾𖽔𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏．”
24 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼊𖽙𖾑 𖼄𖽔𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏“ 𖽃𖽔𖾑 𖽐𖼷𖽹𖾐”．𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽔𖾏，𖼐𖽱𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐． 25 𖼸𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ：“𖼀𖽡𖾐 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼋𖽘𖾏．”𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼏𖽻 𖼺𖽙𖾑 𖼖𖽑𖽻𖾏 𖽐𖼮𖽦𖾏，𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼙𖽡 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼺𖽙𖾑 𖼖𖽑𖽻𖾏 𖽐𖼮𖽦𖾏，𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼙𖽡 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼺𖽘𖾑 𖼖𖽡𖾏 𖼊𖽘𖾑 ，𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽜．”
26 𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼾𖽡𖾏 𖼐𖽑𖽪𖾐 ，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏， 𖼊𖽙𖾑 𖼄𖽔𖾐 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐．𖽃𖽔 𖼛𖽻𖾏 𖼡𖽙𖾏 𖼮𖽡𖽝𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽔𖾏 ，𖼳𖽠𖾑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽑𖽡𖾏：“ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪！” 27 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽙𖾑 𖼄𖽔𖾐 𖼐𖽑𖽡𖾏： “𖼳𖽤𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑，𖼎𖽺𖾏 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏，𖼎𖽺𖾏 𖼳𖽤𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼙𖽡 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡𖾏 𖼊𖽘𖾑．𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼷𖽑𖽠𖾑 𖽐𖼊𖽱𖾏 ，𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜．” 28 𖼊𖽙𖾑 𖼄𖽔𖾐 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ！” 29 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “𖼊𖽙𖾑 𖼄𖽔𖾐，𖼟𖽡 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼟𖽡 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖽐𖼮𖽜；𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼎𖽺𖾏 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏．”
30 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖽃𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼊𖽑𖽡𖽝𖾐，𖼦𖽡𖾐 𖼺𖽻𖾏 𖽐𖼮𖽡𖾏 𖼛𖽔𖾏 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑． 31 𖼈𖽡𖽝𖾏 𖼺𖽻𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼮𖽡𖽝𖾏， 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽐𖼮𖽜 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪，𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐， 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼋𖽘𖾏 ，𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼻𖽡𖽝𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑．
