𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 16：8 - 22）
105
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏！
2 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 , 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 ,
𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑！
3 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼗𖽙𖾏 𖼷𖽻𖾑！
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑！
4 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 .
5-6 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏,
𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼺𖽹𖾑 𖽃𖽡𖾑 𖼘𖽪！
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑,
𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏.
7 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏.
8 𖼐𖽱𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏,
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽑𖽡𖽝𖾐 𖼷𖽑𖽷𖾏.
9 𖼟𖽪𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽃𖽔𖾏,
𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼏𖽵 .
10 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼏𖽵 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏,
𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼏𖽵 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏，
11 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 ,
𖽃𖽔𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖽐𖼀𖽠 .”
 
12 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼮𖽡𖽻𖾐,
𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼇𖽜𖾐 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼊𖽔𖾐.
13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑,
𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖽃𖽡𖾑.
14 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾐, 𖼖𖽔𖾏 𖼡𖽑𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏，
15 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽪𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐.”
 
16 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑,
𖼮𖽷𖾐 𖼊𖽪𖾏 𖼺𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽜𖾑 𖼑𖽻𖾏，
17 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏,
𖼽𖽙 𖼺𖽳 𖼊𖽻𖾏 𖼋𖽺 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑.
18 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼸𖽡𖽝𖾏 𖼊𖽑𖽡𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐,
𖼐𖽱𖾏 𖼦𖽡𖾑 𖼚𖽘𖾐 𖼊𖽻𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽻𖾏 𖼡𖽑𖽹𖾐 𖼮𖽡𖽝𖾏.
19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼮𖽡𖽺𖾏 𖼐𖽱𖾏,
𖼸𖽙 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼞𖽑𖽔𖾏 𖽃𖽔𖾏 𖼸𖽪𖾏 𖼋𖽘𖾏.
20 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼚𖽠𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏,
𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾑 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏，
21 𖼺𖽹𖾑 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼾𖽡𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑 ,
𖼞𖽳𖾑 𖼯𖽵𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏，
22 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼐𖽱𖾏 𖼉𖽻 𖼷𖽻𖾏,
𖼯𖽡𖽙𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼗𖽻.
 
23 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼣𖽹 𖼮𖽡 𖼋𖽘𖾏,
𖼽𖽔𖾐 𖼞𖽙 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼦𖽗𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.
24 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑，
25 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽙𖾑 𖼅𖽻𖾏 𖼳𖽪𖾐 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼎𖽑𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑.
26 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑
𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽹𖾑，
27 𖼐𖽡𖽙𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼺𖽻𖾏,
𖼐𖽡𖽙𖾐 𖼦𖽗𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼊𖽑𖽡𖽝𖾐.
28 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼐𖽱𖾏 𖼗𖽪𖾏.

36 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐,

37 𖼐𖽱𖾏 𖼨𖽑𖽳𖾏 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾐、𖼑𖽡𖽝𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,

38 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝 ,

39 𖼐𖽱𖾏 𖼊𖽑𖽔𖾑 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼗𖽙𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏,

40 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽻 𖽃𖽱𖾏 𖼽𖽘𖾏 𖼋𖽔𖾏,

41 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽙𖾏 𖽐𖼀𖽑𖽔𖾐 𖼺𖽘𖾏, 𖽃𖽻𖾐 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏,

42 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 ，
