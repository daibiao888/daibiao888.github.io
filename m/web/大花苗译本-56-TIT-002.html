6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏，𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼻𖽙𖾑 𖼾𖽨𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏． 7 𖼟𖽡 𖼾𖽨𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽔𖾑 𖼳𖽷𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼻𖽻𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽻； 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼊𖽑𖽱𖾏 𖼺𖽡𖽝𖾐 𖼚𖽠𖾐、 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳， 8 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽻𖾏，𖼟𖽙𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼖𖽔𖾏 𖼾𖽱， 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐，𖼦𖽡𖾐 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼐𖽑𖽡𖾏 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼾𖽙，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏．
9 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖼽𖽔𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼮𖽵𖾑 ，𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖽃𖽺𖾐 𖼙𖽔，𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼊𖽱𖾑 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏． 10 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽡𖽝 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐，𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖼟𖽙𖾏 𖼀𖽡𖾐 𖼮𖽨𖾏 𖼮𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏．
