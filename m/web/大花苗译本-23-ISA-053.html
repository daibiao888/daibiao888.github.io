𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼮𖽜 𖼖𖽺𖾑?
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑?
2 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼘𖽑𖽙𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼅𖽔𖾏,
𖼺𖽱𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼘𖽑𖽘𖾑 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡𖾏．
𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼼𖽘𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖽐𖼎𖽻,
𖼟𖽙𖾏 𖼀𖽡𖾐 𖼑𖽔𖾏 𖼐𖽱𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼼𖽘𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼮𖽳𖾐 𖼉𖽳 𖼐𖽱𖾏.
3 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏,
𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼟𖽱𖾏 𖼎𖽻𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐．
𖼐𖽱𖾏 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼀𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼐𖽱𖾏.
 
4 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼀𖽡𖾐 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼞𖽱𖾑 𖼎𖽷𖾏 𖼀𖽡𖾐 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖼑𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏,
𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼊𖽻𖾏 𖼯𖽡𖽙𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖼋𖽘 .
5 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏,
𖼑𖽜 𖼀𖽡𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐.
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐, 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖽐𖼚𖽠𖾏 𖽐𖼊𖽻𖾏, 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏.
6 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼺𖽱𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖾏 𖼁𖽜𖾏 𖼙𖽠𖾏,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽔𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
 
7 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏,
𖼐𖽡𖽙𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼑𖽻𖾏 𖽃𖽦𖾐 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏,
𖼐𖽱𖾏 𖼺𖽱𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼮𖽤𖾐 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽔𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽻𖾑 𖼚𖽻𖾐 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
8 𖼑𖽜 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖽐𖼎𖽙 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼡𖽔 𖼋𖽱𖾏 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼊𖽻𖾏 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼺𖽘𖾏,
𖼐𖽱𖾏 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏, 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼯𖽡𖽙 .
9 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 ,
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼈𖽡𖽝𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽱𖾏,
𖼋𖽔 𖼙𖽠𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼅𖽔𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽱𖾏.
 
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 (“𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖽃𖽺𖾐 𖼙𖽔”) 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼟𖽱𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔．
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽻𖾏 𖼾𖽔𖾏．
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼽𖽔𖾑 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏．
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏.
11 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐,
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏．
𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼯𖽜𖾏 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.
12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽔𖾏 𖽐𖼀𖽠𖾏,
𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼇𖽺𖾐 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏．
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼊𖽑𖽠𖾐 𖼟𖽙𖾏 𖼢𖽻 𖼙𖽠𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼾𖽔𖾏,
𖼎𖽺𖾏 𖼘𖽑𖽪𖾐 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
