𖼊𖽪𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 . 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽑𖽘𖾑 𖼢𖽔𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏. 7 𖽃𖽔𖾏 𖼐𖽡𖾑 ，𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑
