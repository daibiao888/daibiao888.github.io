𖼽𖽨 𖼊𖽔𖾏𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏
17
𖼽𖽨 𖼊𖽔𖾏𖼑𖽡𖾏 𖼾𖽔𖾏, 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽔𖾑 𖽐𖼚𖽘𖾏 𖼖𖽑𖽻𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼮𖽡𖽾𖾑 𖽐𖼞𖽘𖾑 𖽐𖼮𖽡𖽙𖾏 𖼮𖽡𖽝𖾏, 𖼸𖽻 𖼞𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽻𖾐 𖼋𖽪 𖼞𖽪𖾐 𖼮𖽡𖽝𖾏. 2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏, 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖽐𖼎𖽙 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐, 𖼐𖽡𖽙𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾐 𖼄𖽡𖾐 𖼺𖽡𖽝𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪. 3 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 , 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑, 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐, 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑. 4 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼾𖽨𖾏 𖼾𖽔𖾏, 𖼟𖽡 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼺𖽘𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑, 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼯𖽡, 𖼯𖽡 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼞𖽑𖽡𖽠𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 (“𖼛𖽘𖾏 𖼼𖽙𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 ”),
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖽃𖽡𖾑,
𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼋𖽘𖾏！
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽔𖾏 𖼑𖽡𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾐 𖼞𖽑𖽜𖾏,
𖼦𖽡𖾐 𖼁𖽙𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏.
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 ,
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼮𖽡𖽳𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐.
7 “𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼞𖽑𖽡𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼏𖽔𖾏,
𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
8 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽻𖾐,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽔𖾐 𖼟𖽱𖾐 𖼗𖽔 𖽃𖽔𖾑 𖼯𖽤𖾐,
𖼳𖽙𖾑 𖼉𖽠𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏,𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼮𖽑𖽹𖾏,
𖽃𖽔 𖽐𖼚𖽻𖾏 𖼖𖽔𖾏 𖽐𖼷𖽔𖾐 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼯𖽡𖽙𖾏,
𖼐𖽡𖽙𖾐 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽳𖾏,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽘.
𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼗𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼮𖽵𖾑
9 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼉𖽻𖾏 𖼮𖽘𖾑,
𖼗𖽱𖾏 𖼸𖽙 𖼡𖽘𖾐,
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖼋𖽘 𖼖𖽺𖾑?
10 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐,𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽐𖼀𖽱𖾏,
𖼽𖽔𖾑 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏.”
11 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼺𖽱𖾏 𖼡𖽹𖾐 𖼞𖽙𖾏 𖼦𖽡𖾐 𖼚𖽙𖾐 𖼁𖽔 𖼡𖽳𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖽐𖼊𖽡,
𖼸𖽙 𖼚𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏,𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏,
𖼸𖽙 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼐𖽱𖾏 𖼊𖽪𖾏 𖼖𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼺𖽘𖾏.
12 𖼀𖽡𖾐 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏,
𖼎𖽑𖽻𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖼊𖽪𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐.
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼘𖽪 !
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼇𖽳𖾑 𖼟𖽡 𖼺𖽘𖾏,𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼞𖽪𖾑 𖼙𖽠𖾏,
𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽘𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼋𖽡𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼺𖽘𖾏.”
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽑𖽙𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏,
𖼮𖽨𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏.
𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏
15 𖼑𖽔𖾏 𖼘𖽪!𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
