𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑
（𖼞𖽑𖽙𖾐 2：1-12；𖼖𖽪𖾏 5：17-26）
9
𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏，𖼋𖽡𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑，𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼙𖽻． 2 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽱𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑，𖼀𖽱𖾏 𖼎𖽻𖾏 𖼊𖽔 𖼸𖽘𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑 𖼐𖽑𖽡𖾏：“ 𖼷𖽔𖾑 𖼊𖽪𖾐，𖼮𖽙𖾏 𖼺𖽡𖽝𖾐！𖼟𖽡 𖼾𖽔𖾏 𖼯𖽔𖾏 𖼋𖽘．” 3 𖼊𖽑𖽻𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 ： “ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ．” 4 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽔𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼘𖽪？ 5 𖼼𖽘𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏 𖼖𖽺𖾑？ 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼾𖽔𖾏 𖼯𖽔𖾏 𖼋𖽘 ’， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽑𖽡𖾏：‘ 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼅𖽻𖾏 𖼀𖽔𖾏 ’． 6 𖼈𖽡𖽝𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽔𖾏 𖼾𖽔𖾏．” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑 𖼐𖽑𖽡𖾏：“𖼳𖽠𖾑 𖼗𖽙𖾏！ 𖼮𖽑𖽳𖾑 𖼟𖽡 𖼊𖽔𖾑 𖼸𖽘𖾏 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼦𖽡𖾑．” 7 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏， 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏． 8 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏 𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏，𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾏 𖼄𖽔𖾐 𖼊𖽑𖽹𖾏
（𖼞𖽑𖽙𖾐 2：13-17；𖼖𖽪𖾏 5：27-32）
9 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼮𖽑𖽡𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏， 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖼊𖽑𖽹𖾏，𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐．” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏， 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐．
10 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼾𖽔𖾏 （“𖼊𖽪𖾐 𖼾𖽔𖾏” 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑）𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼋𖽔𖾏，𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏． 11 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏 ： “ 𖼅𖽡 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼖𖽺𖾑？” 12 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽪𖾐 𖼞𖽑𖽙𖾏 𖽐𖼷𖽘𖾏 ， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽪𖾐 𖼞𖽑𖽙𖾏 𖽐𖼷𖽘𖾏． 13 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼐𖽑𖽡𖾏：‘ 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐，𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼙𖽔 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏．’ 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ，𖼅𖽡 𖼅𖽻𖾏 𖽐𖼊𖽱𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼗𖽙𖾏， 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖼟𖽪𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑
（𖼞𖽑𖽙𖾐 2：18-22；𖼖𖽪𖾏 5：33-39）
14 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼽𖽙 𖼦𖽗 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖽐𖼎𖽙 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑，𖼟𖽡 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼘𖽪？ ” 15 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 ： “ 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼡𖽑𖽔𖾏 𖼺𖽷𖾐，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏 𖼘𖽪？ 𖼈𖽡𖽝𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼸𖽙 𖼋𖽘𖾏， 𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽳𖾏 𖼙𖽠𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏，𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼋𖽘𖾏． 16 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼮𖽑𖽦𖾐 𖽐𖼷𖽷𖾑 𖼮𖽑𖽙𖾏 𖼡𖽪𖾐，𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾑 𖼮𖽑𖽙𖾏 𖼡𖽪𖾐 𖽐𖼚𖽪𖾐 𖼮𖽑𖽙𖾏 𖽐𖼚𖽔 𖼺𖽘𖾏，𖽐𖼚𖽔 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾐 𖼻𖽻𖾏． 17 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐， 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼘𖽑𖽱𖾏 𖼡𖽪𖾐，𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼘𖽑𖽱𖾏 𖼡𖽪𖾐 𖽐𖼮𖽑𖽷𖾏 𖼙𖽠𖾏，𖼮𖽡𖽠𖾑 𖼎𖽺𖾏 𖼮𖽑𖽻𖾏 𖼙𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼘𖽑𖽱𖾏 𖼖𖽔𖾏 𖼖𖽔𖾑 𖽐𖼚𖽔 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼺𖽘𖾏． 𖼮𖽔 𖼅𖽔𖾏 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖼽𖽔𖾑 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼘𖽑𖽱𖾏 𖼮𖽑𖽦𖾐，𖽃𖽔𖾐 𖼼𖽘𖾐 𖼞𖽑𖽡𖽝𖾐 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼄𖽙𖾐 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑
（𖼞𖽑𖽙𖾐 5：21-43；𖼖𖽪𖾏 8：40-56）
18 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼑𖽔𖾏 𖼘𖽪！𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼋𖽔𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼋𖽔 𖼙𖽠𖾏，𖼮𖽑𖽙 𖼟𖽡 𖼋𖽔𖾏 𖼮𖽙𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼀𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽡𖽝𖾏 𖼗𖽙．” 19 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽻．
20 𖼊𖽑𖽻𖾏 𖼐𖽡𖾑 ，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 ，𖼄𖽙𖾐 𖼄𖽙𖾐 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖼊𖽻𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏，𖼋𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼘𖽑𖽘𖾐 𖼳𖽷𖾏． 21 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼯𖽔 𖼽𖽔𖾑 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏．” 22 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙， 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏，𖼟𖽡 𖼮𖽙𖾏 𖼺𖽡𖽝𖾐！𖼟𖽡 𖽐𖼮𖽜 𖼟𖽙𖾏 𖼟𖽡 𖼻𖽻𖾏 𖼋𖽘𖾏．” 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 ，𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏．
23 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼸𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏，𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼷𖽑𖽙𖾑 𖼢𖽜𖾏 𖼳𖽜𖾑，𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏， 24 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ： “𖼋𖽠𖾏 𖼙𖽠𖾏！𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏，𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 𖼊𖽔𖾐．” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼚𖽙𖾏 𖼐𖽱𖾏． 25 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽠𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽔𖾏 𖼻𖽻𖾑 𖼙𖽠𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏，𖼊𖽔𖾑 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏，𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼋𖽘𖾏． 26 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽙𖾏 𖽐𖼷𖽷𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜．
𖼞𖽑𖽙𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽔𖾐 𖼄𖽱𖾐
27 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏，𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼸𖽙𖾏 𖼐𖽱𖾏， 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 ： “𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏， 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼀𖽡𖾐 𖼘𖽪！” 28 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏，𖼐𖽔𖾐 𖼄𖽱𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐． 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽐𖼮𖽜 𖼞𖽪𖾑 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼘𖽪𖾏 ？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ： “ 𖼮𖽵𖾑 𖼘𖽪 ！𖼀𖽡𖾐 𖽐𖼮𖽜 𖼯𖽡𖽙𖾏．” 29 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔，𖼐𖽑𖽡𖾏：“𖼸𖽙𖾏 𖼅𖽡 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏．” 30 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏 ．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏：“𖼅𖽡 𖼽𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐 ，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐．” 31 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 ， 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜．
