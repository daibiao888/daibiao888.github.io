𖼖𖽑𖽘𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼘𖽘𖾑𖽑 𖼾𖽨

𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽪𖾐 𖼘𖽘𖾑𖽑 𖼾𖽨 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽦𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝 , 𖼽𖽔𖾑 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 .”
