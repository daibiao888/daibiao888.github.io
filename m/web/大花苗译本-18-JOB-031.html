“𖼞𖽪𖾑 𖽐𖼎𖽙 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 ,

𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼊𖽻𖾏 , 𖼟𖽪𖾏 𖼋𖽪 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑 ?
3 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 ,
𖼳𖽻𖾏 𖽐𖼊𖽻𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽃𖽡𖾑 𖼖𖽺𖾑 ?
4 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼮𖽔𖾑 𖼾𖽱 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽠𖾏 𖼐𖽜𖾑 𖼖𖽺𖾑 ?
 
5 𖼞𖽪𖾑 𖼮𖽔𖾑 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 ,
𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼮𖽔𖾑 𖼖𖽠𖾑 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 ,
6 𖼯𖽡𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼸𖽱𖾏 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼗𖽙𖾏 𖼙𖽪𖾏 𖼞𖽪𖾑 ,
𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
7 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼮𖽔𖾑 𖼇𖽳𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 ,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽔𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼅𖽻𖾏 𖽃𖽔𖾏 ,
𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼘𖽙𖾏 𖼎𖽻𖾏 ,
8 𖼯𖽡𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼯𖽡𖽙 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 , 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼚𖽙𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 .
 
9 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽑𖽵𖾑 , 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼳𖽠𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 ,
𖽐𖼚𖽺𖾏 𖼎𖽻𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼡𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 ,
10 𖼯𖽡𖾏 𖼎𖽺𖾏 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽤𖾐 𖼑𖽜 𖼗𖽔 𖼢𖽪𖾏 𖼡𖽜𖾑 ,
𖼗𖽔 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏 .
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 ,
𖼟𖽪𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 .
12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 , 𖼸𖽙 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼷𖽑𖽪𖾐 𖼺𖽘𖾏 ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼚𖽙𖾐 𖼺𖽘𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼯𖽡𖽙 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 .
 
13 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 ,
𖼞𖽪𖾑 𖼮𖽔𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 ,
14 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑 ?
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽹𖾐 𖼑𖽪 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼊𖽡𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑 ?
15 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼞𖽪𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖽃𖽡𖾑 , 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼐𖽱𖾏 𖼊𖽑𖽡𖽝𖾐 𖼘𖽪 ？
𖼷𖽷𖾏 𖼐𖽱𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼘𖽪 ?
 
16 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼊𖽻𖾏 ,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼟𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼛𖽙𖾏 𖼺𖽘𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 ,
17 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼑𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 ,
𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼑𖽻𖾏 ,
18 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 , 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼘𖽑𖽙𖾐 , 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 ,
𖼞𖽪𖾑 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 , 𖼞𖽪𖾑 𖽐𖼀𖽘𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼋𖽘𖾏( “𖽐𖼀𖽘𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏”).
19 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 𖼋𖽔 𖼙𖽠𖾏 ,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼈𖽳𖾑 ,
20 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼚𖽻𖾐 𖼾𖽘𖾐 𖼊𖽻𖾏 𖼳𖽙𖾑 ,
𖼯𖽡𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼦𖽡𖾐 𖼑𖽜 𖼞𖽪𖾑 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ,
21 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼁𖽙𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 ,
𖼎𖽺𖾏 𖼮𖽘𖾏 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 ,
22 𖼯𖽡𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼀𖽡 𖼢𖽙𖾏 𖼚𖽑𖽘𖾏 𖼙𖽠𖾏 ,
𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖽃𖽔𖾏 𖼽𖽘𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 .
23 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽙𖾏 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 ,
𖼑𖽜 𖼐𖽱𖾏 𖼚𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 , 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 .
