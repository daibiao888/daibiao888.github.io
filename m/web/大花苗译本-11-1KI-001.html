𖼽𖽔𖾏 𖼊𖽙𖾑 𖼐𖽡 𖼽𖽔𖾐 𖼦𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐
5 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼦𖽔𖾐 𖼮𖽡 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼊𖽙𖾑 𖼐𖽡 𖼽𖽔𖾐 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐 𖼾𖽨𖾏, 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.” 𖼎𖽺𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐、 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏, 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏. 6 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼦𖽡𖾐 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼐𖽱𖾏, 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼟𖽡 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑?” 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽐𖼎𖽻, 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖽃𖽔𖾑 𖼡𖽜𖾐. 7 𖼽𖽔𖾏 𖼊𖽙𖾑 𖼐𖽡 𖼽𖽔𖾐 𖽐𖼎𖽙 𖼳𖽡𖾐 𖼖𖽪𖾐 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼽𖽔 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼦𖽡𖾏 𖼊𖽳𖾏, 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏, 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏. 8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼺𖽔𖾐 𖼊𖽪、 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽡𖾐 𖼐𖽔 𖼽𖽔𖾐、 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽔 𖼊𖽗𖾑、 𖼳𖽷𖾏 𖼄𖽺𖾐、 𖼖𖽡𖾏 𖼽𖽡𖾑， 𖼊𖽑𖽡𖽝𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼋𖽻, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼽𖽔𖾏 𖼊𖽙𖾑 𖼐𖽡 𖼽𖽔𖾐. 9 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐, 𖼽𖽔𖾏 𖼊𖽙𖾑 𖼐𖽡 𖼽𖽔𖾐 𖼐𖽡𖽙𖾐 𖼽𖽡𖽾𖾐 · 𖼘𖽙 𖼮𖽡𖽝 𖽃𖽔𖾑 𖽐𖼊𖽪, 𖼺𖽙𖾐 𖼳𖽡𖾑 𖼖𖽡𖽝 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼊𖽔𖾏 𖼑𖽨𖾏、 𖼾𖽘𖾏、 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼛𖽙, 𖼮𖽘𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏, 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏. 10 𖼈𖽡𖽝𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽔 𖼊𖽗𖾑, 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽔 𖼽𖽔𖾐， 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼋𖽻, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞， 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼮𖽘𖾏.
