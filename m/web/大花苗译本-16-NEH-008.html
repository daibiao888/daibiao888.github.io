9 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼐𖽡 𖼳𖽡𖾑 𖼄𖽡𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏.” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖽐𖼊𖽠𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼞𖽑𖽔𖾏 𖼐𖽡𖽝𖾑 𖼺𖽘𖾏 𖼐𖽔𖾑. 10 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼅𖽻𖾏 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼛𖽙 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖼦𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼽𖽔𖾑 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏. 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽳𖾏, 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏（“𖼛𖽘𖾏 𖼼𖽙𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑”）.” 11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏, 𖼐𖽑𖽡𖾏 : “𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽳𖾏.” 12 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏, 𖼎𖽺𖾏 𖼇𖽺𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼙𖽔 𖼚𖽙𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑.
