𖼷𖽻𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
𖼞𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽐𖼞𖽻𖾏,𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 “ 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔𖾏 𖼡𖽜𖾏 ” 𖼳𖽘𖾐.

𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼟𖽪𖾏 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼀𖽘𖾏.
2 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏,
𖼇𖽻𖾑 𖼎𖽻𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼸𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼳𖽡𖽠𖾏，
3 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏,
𖼇𖽻𖾑 𖼎𖽻𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼀𖽻𖾏 𖼐𖽔𖾑 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏,
𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
4 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼁𖽘 𖼛𖽡𖾏, 𖼖𖽪𖾑 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼐𖽡𖾑 𖼇𖽺𖾐 𖽐𖼚𖽱𖾏, 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏.
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼖𖽪𖾑 𖼙𖽻 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙,
𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑.
6 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏, 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽠𖾏 𖼺𖽘𖾐, 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼾𖽘𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
7 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐,
𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔！ （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
8 𖼅𖽡 𖼋𖽔𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖼑𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏.
9 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼺𖽻𖾏,
𖼐𖽱𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼊𖽪𖾑 𖼐𖽑𖽜𖾑 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼄𖽑𖽪𖾑, 𖼞𖽑𖽡𖽠𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏.
10 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏, 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏.
11 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐,
𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔！ （𖼳𖽡𖾏 𖼖𖽔𖾑）
