25
𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏 . 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼽𖽙 𖼽𖽦𖾑 𖼊𖽗𖾏 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 . 2 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 . 3 𖼐𖽱𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼎𖽺𖾏 𖼊𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏 , 4 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼊𖽠𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 , 𖼐𖽑𖽡𖾏 : “ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼊𖽪𖾐 𖼊𖽔𖾏 𖼉𖽺𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼉𖽺𖾏 𖼊𖽔𖾏 𖼊𖽪𖾐 , 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼾𖽔𖾏 𖼋𖽔 𖼊𖽔𖾐 .”
𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼈𖽡𖽝𖾏 𖼖𖽑𖽘𖾑 𖼁𖽠𖾏
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 14：7）
5 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼦𖽡 𖼸𖽱 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼸𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 , 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼎𖽺𖾏 𖼾𖽱 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼎𖽑𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽡𖽻𖾏 𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼄𖽑𖽪𖾑 𖽐𖼎𖽙 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 , 𖼏𖽙 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼉𖽻𖾏 . 6 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 𖼑𖽡𖽝𖾏, 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼉𖽻𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 . 7 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ! 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼏𖽙 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼐𖽡𖽙𖾐 . 8 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏, 𖼯𖽡𖾏 𖼊𖽑𖽱𖾏 𖼼𖽙 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 ! 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼟𖽡 𖼢𖽻 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽠𖾏 𖽐𖼀𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 , 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼢𖽻 𖼙𖽠𖾏 .” 9 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : “𖽃𖽡𖾐 𖼀𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼏𖽙 𖽃𖽡𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑 ?” 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼸𖽠𖾏 𖼄𖽔𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽡𖽻𖾐 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 .” 10 𖽃𖽔𖾏 𖼐𖽡𖾑 , 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼇𖽺𖾐 𖼏𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 . 𖽃𖽔𖾏 𖼐𖽡𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖼮𖽑𖽡𖾏 𖽐𖼷𖽔𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 .
11 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼮𖽙𖾏 𖼚𖽔𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖽐𖼮𖽳𖾑 , 𖼊𖽔𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 . 12 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽳𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼷𖽔𖾏, 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽔𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏, 𖽐𖼮𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏. 13 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼟𖽙𖾏 𖼎𖽙𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼎𖽙 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑 , 𖽐𖼊𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 , 𖼎𖽑𖽻𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼀𖽳 · 𖼦𖽙 𖼖𖽪𖽝 , 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼷𖽷𖾐 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .
14 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼊𖽔𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 , 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼁𖽠𖾏 𖼗𖽙𖾏 , 𖼮𖽘𖾏 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼮𖽵𖾑 𖼳𖽺 , 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 . 15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 , 𖼷𖽷𖾐 𖼟𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 , 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼖𖽺𖾑 ?” 16 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼡𖽔 𖼋𖽱𖾏 𖼮𖽘𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼸𖽠𖾐 𖼖𖽺𖾑 ? 𖼡𖽙𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 ! 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽐𖼎𖽑𖽦𖾑 𖽐𖼊𖽻𖾏 𖼖𖽺𖾑 ?” 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼎𖽺𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 , 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼑𖽻 𖼞𖽪𖾑 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐, 𖼞𖽪𖾑 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼟𖽡.”

17 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽻 𖼷𖽻𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 , 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽪𖾏 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖼽𖽙 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼋𖽔𖾏 ! 𖼀𖽡𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 .” 18 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼐𖽑𖽡𖾏 : “𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼑𖽡𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 , 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼑𖽡𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 , 𖼐𖽑𖽡𖾏 : 𖼄𖽔𖾐 𖼟𖽡 𖽐𖼷𖽑𖽹𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼐𖽤𖾐 . 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼮𖽑𖽡𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 , 𖼯𖽪𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼺𖽘𖾏 . 19 𖼟𖽡 𖼐𖽑𖽡𖾏 : 𖼑𖽔𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 . 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏. 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖾏 𖼞𖽳𖾑 𖼋𖽘𖾏 , 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 , 𖼟𖽙𖾏 𖼾𖽨𖾏 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽪𖽙 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼖𖽺𖾑 ?”
20 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻 𖼸𖽙𖾏 , 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼻𖽻𖾏 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼮𖽵𖾑 𖼳𖽺 . 21 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼀𖽳 · 𖼳𖽷𖾏 𖼄𖽳 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼦𖽡𖾏 𖼁𖽙𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 . 22 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼦𖽡𖾐 𖼳𖽠𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 . 23 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾏 , 𖼐𖽡𖽙𖾐 𖼀𖽳 · 𖼳𖽷𖾏 𖼄𖽳 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼽𖽙 𖼦𖽔𖾐 𖼺𖽷𖾑 (“𖼽𖽙 𖼦𖽔𖾐 𖼺𖽷𖾑” 𖼟𖽪𖾏 “𖼽𖽔𖾏 𖼦𖽔𖾐 𖼳𖽡𖽝𖾏 ” ) 𖼐𖽱𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 , 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 . 𖼎𖽺𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 , 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽃𖽔 𖼛𖽻𖾏 𖼸𖽙 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼞𖽪𖾑 𖼞𖽪𖾐 𖽃𖽔 𖼛𖽻𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 24 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐 𖽐𖼎𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖽃𖽳 𖼀𖽡𖽝 · 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼡𖽑𖽙𖾑 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽃𖽡𖾑 , 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 , 𖼞𖽑𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼯𖽡𖽙𖾏 𖼙𖽠𖾏 , 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼸𖽺𖾏 𖼮𖽡𖽝𖾏 ,𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 .
25 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼋𖽔 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 . 26 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏 ,𖼎𖽑𖽻𖾏 𖼁𖽱𖾏 𖼸𖽙 𖼺𖽻𖾏 , 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏《 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏》 𖼮𖽡𖽝𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ? 27 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽜𖾑 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼖𖽔𖾑 𖼮𖽡 , 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼖𖽔𖾑 𖼮𖽡 , 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 . 28 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼑𖽜𖾏 𖼮𖽹𖾑 𖼐𖽱𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼺𖽷𖾐 𖼗𖽙 , 𖼊𖽱𖾏 𖼎𖽻𖾏𖼽𖽨 𖼊𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 .
