𖼊𖽡𖾐 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐．
𖼅𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？

𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏 𖽃𖽡𖾑,

3 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼑𖽨𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,

𖼷𖽑𖽷𖾑 𖼵𖽪𖾐 𖼳𖽤𖾏, 𖼷𖽻𖾑 𖼖𖽔 𖼊𖽺𖾐 𖼁𖽠𖾏 𖽃𖽡𖾑, 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼺𖽹𖾑 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑,
𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐.
4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼺𖽹𖾑 𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼟𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽹𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐,
𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻 ,
𖼎𖽙𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼺𖽹𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽡𖾑.”
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼁𖽡𖽝
5 𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖽃𖽡𖾑 𖼘𖽪 ! 𖼽𖽔𖾑 𖼑𖽻 𖼐𖽱𖾏 𖼗𖽪𖾏 :
“𖼅𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏, 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼅𖽡 ,
𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼖𖽠𖾑 𖼅𖽡 𖽃𖽡𖾑, 𖼙𖽔 𖼐𖽑𖽡𖾏 :
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .
𖼟𖽙𖾏 𖼀𖽡𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼅𖽡 𖼙𖽔 𖼚𖽙𖾏.
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
6 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽐𖼚𖽙 𖼙𖽻 ,
𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 ,

7 𖼳𖽡 𖼣𖽔𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽑𖽡𖽝𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼑𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼺𖽷𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼋𖽘 ,
𖼦𖽡𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽷𖾐 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼊𖽪𖾐 𖼋𖽘 .
 
8 𖼊𖽡𖾐 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼊𖽔𖾐 𖼅𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ?
𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼊𖽔𖾐 𖼅𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼖𖽺𖾑?
𖼈𖽡𖽝𖾏 𖼳𖽡 𖼣𖽔𖾑 𖽃𖽡𖾐 𖽐𖼀𖽱𖾏 𖼄𖽙𖾐 𖼷𖽹𖾏 𖼖𖽑𖽘𖾑 𖼾𖽨 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼋𖽘𖾏,
𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼙𖽔 𖼐𖽙𖾑𖽑 ？
𖼡𖽔 𖼋𖽱𖾏 𖼙𖽔 𖼁𖽙𖾏 𖼖𖽺𖾑?
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽃𖽻𖾐 𖼁𖽺𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼖𖽺𖾑?
𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐,
𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼡𖽙𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼖𖽺𖾑?
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏
10 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽡𖾑 𖼘𖽪 !
𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 ,

11 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏 ,
𖼟𖽙𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,

12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: “ 𖼑𖽔𖾏 𖼘𖽪 !

𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼐𖽱𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏．
