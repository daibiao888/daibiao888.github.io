𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏, 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏. 𖼯𖽡𖽹𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏. 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽔𖾑 𖽃𖽳𖾐 𖼙𖽠𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 ,𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 , 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽳𖾏 𖼟𖽡. 3 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐, 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏. 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼏𖽵 𖼎𖽻𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽃𖽡𖾑. 4 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼺𖽱𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏； 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏. 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏, 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏、𖼡𖽘𖾑𖽑 𖼢𖽔𖾏、𖼼𖽘𖾏 𖼻𖽡𖽝𖾑.” 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽡𖾑 𖼖𖽔𖾑 𖽃𖽳𖾐.
𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼖𖽔𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐
7 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪 𖼸𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐, 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑.” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽤𖾐.” 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 “𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖼊𖽔𖾏 𖼞𖽪𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏.” 8 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼗𖽡𖾏. 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 , 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼇𖽜𖾏 𖼦𖽡𖾐 𖼡𖽑𖽻𖾐. 9 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼨𖽑𖽪𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽤𖾐 𖼯𖽡𖽙 , 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽔𖾑 𖼞𖽱𖾑 ?” 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼋𖽔 .” 10 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼘𖽪? 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽔𖾑𖽑 𖽐𖼮𖽷𖾑 𖼣𖽔𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽤𖾐 𖼀𖽱𖾏, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽙 𖼙𖽠𖾏.” 11 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼘𖽙𖾏 𖼏𖽻 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼯𖽡𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔.”
12 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼡𖽻𖾐, 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼀𖽔𖾏 𖽐𖼀𖽠, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏. 13 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏, 𖼐𖽑𖽪𖾐 𖼅𖽔𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽑𖽪𖾐, 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐. 14 𖼐𖽱𖾏 𖼅𖽔𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖽐𖼀𖽙𖾏 𖼑𖽨𖾐, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼐𖽱𖾏. 15 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼸𖽻𖾏 𖼡𖽑𖽜𖾑 𖽐𖼮𖽡𖽠𖾏 𖼮𖽡𖽝𖾏, 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽔𖾏 𖼷𖽑𖽠𖾏 𖼺𖽘𖾏, 𖼷𖽑𖽪𖾐 𖼀𖽪𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼮𖽡𖽝𖾏. 16 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼇𖽳𖾑 𖼀𖽡𖾐 𖼺𖽘𖾏 𖼅𖽻𖾏 𖼙𖽠𖾏 𖼦𖽡𖾑! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼀𖽡𖾐.”
17 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽔𖾑 𖽃𖽳𖾐 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 18 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽐𖼮𖽡𖽠𖾏, 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼋𖽔 𖼙𖽠𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼷𖽑𖽪𖾐 𖼺𖽘𖾏, 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽐𖼮𖽡𖽠𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼨𖽑𖽪𖾏, 𖼨𖽑𖽪𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏. 19 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽘𖾑 𖽐𖼮𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼷𖽠𖾑𖽑. 20 𖼮𖽡𖾑 𖼖𖽔𖾑 𖽃𖽳𖾐 𖼊𖽪𖾐 𖼾𖽨 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼊𖽪𖾐 𖼾𖽨 𖼯𖽦𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏, 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖽃𖽻𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼁𖽡𖽝.” 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼣𖽹 𖼺𖽳 (“𖼣𖽹 𖼺𖽳”𖼟𖽪𖾏 “𖼦𖽡𖾏 𖽐𖼊𖽜𖾏”), 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏. 21 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼎𖽺𖾏 𖽐𖼮𖽡𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽜 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖾑 𖽐𖼷𖽷𖾏 𖼳𖽡𖾑 𖼊𖽑𖽡 𖼐𖽔 (“𖼳𖽡𖾑 𖼊𖽑𖽡 𖼐𖽔”𖼟𖽪𖾏 “𖼞𖽡𖾑 𖼀𖽔𖾑”). 22 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽐𖼮𖽡𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼑𖽜 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼊𖽡𖾏 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖾑 𖽐𖼷𖽷𖾏 𖼖𖽡𖾏 𖼦𖽙 𖼀𖽳 (“𖼖𖽡𖾏 𖼦𖽙 𖼀𖽳”𖼟𖽪𖾏 “𖼇𖽘𖾑”). 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑, 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏.”
23 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑 𖼙𖽠𖾏. 24 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐, 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐, 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼘𖽘𖾑𖽑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.” 25 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘, 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐； 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼮𖽡𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐.
