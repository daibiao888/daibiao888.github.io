𖼑𖽜 𖼳𖽡 𖼣𖽔𖾑 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏
9
“𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼟𖽪𖾏 𖽃𖽻𖾐,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐,
𖼞𖽪𖾑 𖼻𖽻𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 (“𖼖𖽡𖾑 𖼉𖽪𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏”) 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏, 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏.
2 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼅𖽔𖾏 𖽐𖼞𖽔𖾏 𖼡𖽑𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼺𖽘𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼯𖽡𖽙𖾏,
𖼟𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖽃𖽡𖾑.
3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽻𖾑 𖽃𖽔𖾑 𖽐𖼚𖽺𖾏 𖼮𖽡𖽝𖾏, 𖼺𖽱𖾏 𖼐𖽜𖾑𖽑 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼦𖽡𖾐 𖼾𖽙 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼚𖽪𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖽐𖼚𖽪𖾏,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼞𖽪𖾑．”
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
4 “𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼼𖽷𖾏 𖼼𖽙 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏,
𖼼𖽷𖾏 𖼼𖽙 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽔𖾑 𖼐𖽡𖽝𖾏 𖼮𖽑𖽵𖾑 𖼼𖽷𖾏 𖼼𖽙 ,
𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽔𖾐 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼾𖽻 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼎𖽑𖽻𖾏 𖼼𖽙 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.
6 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑．”
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:
“𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽙𖾑 𖽐𖼮𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 , 𖼞𖽪𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼾𖽔𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑?
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼈𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏, 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏,
𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏,
𖼈𖽡𖽝𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏.
9 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑?”
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼊𖽻𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
10 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼐𖽡𖽝𖾑 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼑𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽐𖼎𖽘𖾏 𖽐𖼡𖽜𖾐 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖽻𖾏 𖼎𖽻𖾏,
𖼯𖽦𖾏 𖼳𖽘𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼢𖽔𖾏,
𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏, 𖼞𖽑𖽔𖾏 𖼀𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏.
11 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼚𖽻𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐, 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼙𖽻 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏, 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏.
12 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳, 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑? 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼙𖽔 𖽐𖼎𖽙 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽡𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏? 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏, 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖽻𖾏 𖼎𖽻𖾏 𖼖𖽺𖾑? 13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽔𖾏. 14 𖼯𖽔 𖼸𖽙𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼼𖽔𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼊𖽔𖾐, 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼡𖽑𖽔𖾐 𖽃𖽡𖾑, 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽔𖾑 𖼖𖽡 𖼙𖽠𖾏. 15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑： 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼄𖽡𖾏 𖼇𖽪𖾑 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼑𖽻𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼦𖽻𖾏. 16 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼙𖽠𖾏, 𖼟𖽪𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼖𖽠𖾑 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
 
17 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏,
𖼨𖽑𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽐𖼡𖽔 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼋𖽔𖾏,
𖼎𖽺𖾏 𖼷𖽷𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽐𖼡𖽔 𖼐𖽡𖽝𖾑 𖽃𖽡𖾑 𖼋𖽔𖾏.
18 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾐 𖼺𖽡𖽝𖾐,
𖼟𖽙𖾏 𖼀𖽡𖾐 𖼞𖽔𖾏 𖼅𖽔 𖽐𖼊𖽔𖾏 𖽐𖼊𖽔𖾏,
𖼟𖽙𖾏 𖼀𖽡𖾐 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖼅𖽔 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼗𖽙𖾏 .
19 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽙𖾑𖽑 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼳𖽡 𖼣𖽔𖾑 , 𖼐𖽑𖽡𖾏:
𖼀𖽡𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡 𖼋𖽙 𖼗𖽠𖾏 𖼋𖽘𖾏！
𖼀𖽡𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐！
𖼀𖽡𖾐 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼀𖽡𖾐 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏.
 
20 𖽃𖽔 𖼁𖽙𖾏 𖼅𖽡 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏,
𖼷𖽹𖾏 𖼳𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼅𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼡𖽑𖽔𖾐 𖼼𖽷𖾏 𖼼𖽙 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐．
21 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏,
𖼁𖽺𖾏 𖼚𖽔𖾏 𖼀𖽡𖾐 𖽐𖼚𖽙 𖼟𖽻 ,
𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏,
𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐.
22 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖽐𖼎𖽘𖾏 𖼯𖽢𖾏 𖼺𖽱𖾏 𖼮𖽑𖽡𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼷𖽻𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼇𖽳𖾑 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽻𖾏 𖼞𖽑𖽡𖽠𖾏.
