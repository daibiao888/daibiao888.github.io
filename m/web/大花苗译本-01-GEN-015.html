15
𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏！ 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼁𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖼟𖽡.” 2 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ！𖼟𖽡 𖼄𖽔𖾐 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪? 𖼸𖽙 𖼀𖽘𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽡𖾑 𖼳𖽡𖽝𖾏.”
3 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽔𖾏， 𖼟𖽡 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼄𖽔𖾐 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏.” 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽙𖾏； 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽙𖾏.” 5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏, 𖼾𖽱 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐, 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼸𖽠𖾏 𖼾𖽱 𖼋𖽘 𖼋𖽻𖾏.” 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.” 6 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖽐𖼮𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳.
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼙𖽔 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼋𖽠𖾏 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖽃𖽡𖾑 𖽂𖽪 𖽃𖽳𖾐, 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏.” 8 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼘𖽪?” 9 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼮𖽤𖾐 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽨𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏, 𖽃𖽡𖾐 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏, 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏, 𖽃𖽡𖾐 𖼋𖽪 𖽐𖼡𖽔𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑, 𖽃𖽡𖾐 𖼋𖽪 𖼣𖽔𖾑 𖼣𖽔𖾏 𖽐𖼡𖽔𖾐 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏.” 10 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼅𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼋𖽔𖾏, 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼀𖽑𖽔𖾏 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐, 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼊𖽜𖾐 𖼮𖽡𖽝𖾏, 𖼯𖽔 𖼅𖽔𖾏 𖽐𖼡𖽔𖾐 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼀𖽑𖽔𖾏. 11 𖼅𖽔𖾏 𖼚𖽘𖾑 𖼽𖽘𖾏 𖼋𖽔𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼋𖽡𖾏 𖽐𖼡𖽹𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽘𖾏 𖼙𖽠𖾏.
12 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 𖼷𖽷𖾏 𖼖𖽘𖾑 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼊𖽻𖾏 𖼀𖽱𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼙𖽠𖾏, 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼅𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑. 13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏, 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼗𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐； 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽃𖽦𖾐 𖼊𖽻𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏. 14 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏. 15 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏, 𖼋𖽡𖽝𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽪 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼚𖽔𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽱𖾏. 16 𖼸𖽙 𖼚𖽔𖾏 𖼚𖽻𖾐 𖼷𖽑𖽷𖾏 𖼷𖽑𖽷𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽷𖾐 𖼗𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏 𖼺𖽷𖾐.”
17 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼮𖽡𖽝𖾏, 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼋𖽡𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 𖽐𖼡𖽹𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐. 18 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽔𖾑 𖼟𖽱𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽨𖾏 𖼇𖽔 𖼖𖽔𖾑 𖼊𖽡𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏, 19 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼐𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼮𖽡𖾑 𖼐𖽡 𖼳𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼮𖽢 𖼄𖽙𖾑 𖼐𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐、 20 𖼦𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼀𖽡𖾐 𖼖𖽡𖾏 𖼳𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼖𖽡𖾏 𖼇𖽔 𖼽𖽡𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐、 21 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼞𖽳 𖼮𖽢𖾑 𖼺𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.”
