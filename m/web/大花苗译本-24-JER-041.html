4 𖼐𖽱𖾏 𖼐𖽑𖽙𖾏 𖼺𖽘𖾏 𖼮𖽡𖾑 𖼊𖽔𖾏 𖼖𖽡𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐. 5 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐, 𖼅𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖽐𖼎𖽙 𖼳𖽷𖾏 𖼘𖽙 𖼊𖽑𖽡𖽝𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼋𖽔𖾏, 𖼮𖽑𖽪𖾑 𖼚𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼺𖽘𖾏, 𖽐𖼮𖽑𖽷𖾏 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏, 𖽐𖼷𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑, 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼊𖽑𖽡𖽝𖾐 𖼵𖽪𖾐 𖼳𖽤𖾏, 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 6 𖼐𖽡 𖼊𖽑𖽔𖾏 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼎𖽑𖽻𖾏 𖼄𖽡𖾐 𖼺𖽷𖾑 𖼀𖽔𖾑𖼋𖽠𖾏 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼐𖽡𖽝𖾑 𖼷𖽷𖾐 𖼖𖽘𖾑. 𖽐𖼮𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽘, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “ 𖼮𖽑𖽙 𖼅𖽡 𖼋𖽔𖾏 𖼑𖽔𖾏 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼞𖽗𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼮𖽡𖾑 𖼊𖽔𖾏 𖼖𖽡𖾏 .” 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑, 𖼐𖽡 𖼊𖽑𖽔𖾏 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾑, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏, 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼺𖽘𖾏. 8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽔𖾏 𖼀𖽡𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼅𖽙𖾏、𖽃𖽔𖾑 𖼖𖽪𖾐 𖼳𖽻𖾐、𖼛𖽙𖾏、𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖼈𖽺𖾏 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼮𖽡𖽝𖾏 𖼺𖽷𖾐.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼮𖽳 𖼋𖽡𖾏 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾏 𖼎𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
