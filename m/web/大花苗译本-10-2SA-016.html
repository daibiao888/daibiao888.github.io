𖼊𖽔𖾏 𖽂𖽺𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼅𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐, 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳𖾏 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼳𖽡𖾐 𖼀𖽔𖾑 𖼮𖽤𖾐 𖽃𖽔𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏, 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼮𖽹𖾑 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐, 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔𖾏 𖼅𖽔 𖼑𖽜𖾏, 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼮𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼷𖽷𖾑 𖼑𖽡𖾏 𖼳𖽡𖽝𖾐, 𖽃𖽡𖾐 𖼘𖽑𖽱𖾏 𖼮𖽡𖽠𖾑 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏. 2 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪 𖼳𖽡𖾐 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?” 𖼳𖽡𖾐 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽜𖾏 𖼖𖽔𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽜𖾏, 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼑𖽻𖾏, 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼦𖽻𖾏.” 3 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?” 𖼳𖽡𖾐 𖼀𖽔𖾑 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼺𖽷𖾐， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼑𖽡𖾏 𖼞𖽪𖽙 𖼎𖽻𖾏 𖼞𖽪𖾑.” 4 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼳𖽡𖾐 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽙𖾏 𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳𖾏 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼋𖽘𖾏.” 𖼳𖽡𖾐 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏.”

15 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐, 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼊𖽙𖾑 𖼇𖽪 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏. 16 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼽𖽺 𖼗𖽔 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼮𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽪𖾏 𖼳𖽺𖾑 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼉𖽻𖾏 𖼳𖽡𖽻𖾏! 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼉𖽻𖾏 𖼳𖽡𖽻𖾏!” 17 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼑𖽪 𖼦𖽪𖾏 𖼳𖽺𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼽𖽺 𖼗𖽔 𖼐𖽔𖾑 𖼖𖽔𖾏? 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼽𖽺 𖼗𖽔 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏 𖼖𖽺𖾑?” 18 𖼦𖽪𖾏 𖼳𖽺𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼾𖽙, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽹𖾑 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐. 19 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑? 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼘𖽪？ 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡 𖼉𖽺𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡.” 20 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼊𖽙𖾑 𖼇𖽪 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑, 𖼀𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼻𖽻𖾏?” 21 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼊𖽙𖾑 𖼇𖽪 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼣𖽔𖾏 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼟𖽙𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼐𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽱𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏, 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼳𖽪𖾐 𖼟𖽡, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼊𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼋𖽘𖾏.” 22 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽐𖼞𖽔𖾏 𖼏𖽪𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐, 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼸𖽻𖾏 𖼣𖽔𖾏 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖽃𖽡𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏. 23 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼊𖽙𖾑 𖼇𖽪 𖼡𖽑𖽙𖾑 𖼋𖽠𖾏, 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐, 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼊𖽙𖾑 𖼇𖽪 𖼑𖽜𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽜 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼋𖽠𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐.
