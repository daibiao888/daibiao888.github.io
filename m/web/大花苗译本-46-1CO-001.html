4 𖼞𖽪𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽜 𖼅𖽡 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ，𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼊𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼚𖽙 𖼺𖽘𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡． 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼚𖽙 𖼊𖽻𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏，𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 、𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼀𖽻𖾐 𖼟𖽳， 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼮𖽡𖾑 𖼊𖽪 𖽃𖽔𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐， 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼊𖽻𖾏，𖼋𖽙𖾏 𖼸𖽙 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏． 8 𖼐𖽱𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼸𖽙 𖼡𖽘𖾐， 𖼟𖽙𖾏 𖼅𖽡 𖼋𖽙𖾏 𖼸𖽙 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏，𖼦𖽡𖾐 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼖𖽔𖾏 𖼾𖽱 𖼅𖽡． 9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼐𖽱𖾏 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼅𖽡， 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐—𖼀𖽡𖾐 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏．
𖼀𖽻𖾐 𖼘𖽙 𖼞𖽑𖽻𖾑 𖼸𖽻𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐
