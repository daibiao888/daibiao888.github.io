𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽱𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 . 2 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼖𖽑𖽘𖾑 𖼋𖽘𖾏 𖼺𖽘𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 , 𖼎𖽺𖾏 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼑𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ， 3 𖼎𖽺𖾏 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼞𖽳𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔 𖼾𖽠 , 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 、 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼷𖽷𖾑 𖼄𖽡𖾏 𖼑𖽨𖾏 𖼳𖽡𖽝𖾐 、 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼳𖽡𖽝𖾐 . 4 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽷𖾐 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏、 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 、 𖽐𖼮𖽢𖾏 𖼙𖽔 、 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 5 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 ， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡 𖼯𖽡𖾏 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 、 𖼽𖽔𖾐 𖼳𖽡𖽫𖾑 、 𖼳𖽷𖾏 𖼄𖽡𖾐 𖼖𖽔𖾑 𖼄𖽙 、 𖼽𖽦𖾑 𖼳𖽡𖽝 、 𖼄𖽔𖾐 𖼊𖽑𖽔𖾑 𖼊𖽑𖽡 𖼽𖽔𖾐 、 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔 、 𖼀𖽡𖾐 𖼐𖽔 𖼽𖽔𖾐 、 𖽃𖽳 𖼀𖽡𖽝 · 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 、 𖼽𖽦𖾑 𖼖𖽡𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼺𖽙𖾑 、 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏， 𖼮𖽔 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖽐𖼊𖽻𖾏 𖽐𖼎𖽔𖾏 𖽐𖼀𖽔𖾏, 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼎𖽙𖾑 . 6 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼀𖽡𖾐 𖼐𖽔 𖼽𖽔𖾐 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼦𖽔𖾐 𖼳𖽡 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼦𖽡𖾐 𖼳𖽙𖾏 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏 .
𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫
（𖽐𖼞𖽻𖾏 105：1 - 5；96：1 -13；106：1，47- 48）
7 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼟𖽻𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 , 𖼯𖽡𖽙𖾏 𖽐𖼞𖽻𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 :
8 “𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 ,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 ！
9 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 、 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 ,
𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 .
10 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼗𖽙𖾏 𖼷𖽻𖾑 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 .
11 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 ,
𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 .
12-13 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 ,
𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼺𖽹𖾑 𖼘𖽪 ！
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 ,
𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 .
 
14 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 .
15 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏 ,
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾏 𖼇𖽺𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽑𖽡𖽝𖾐 𖼷𖽑𖽷𖾏 .
16 𖼟𖽪𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾑 𖼦𖽔𖾐 𖽃𖽔𖾏 ,
𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼏𖽵 .
17 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼏𖽵 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽑𖽘𖾑 𖼢𖽔𖾏 ,
𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼏𖽵 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏，
18 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡,
𖽃𖽔𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖽐𖼀𖽠 .
 
19 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼯𖽠 ,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼊𖽔𖾐 .
20 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 𖼳𖽔𖾏 𖽐𖼮𖽡 𖼚𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽡𖾑 ,
𖼎𖽑𖽻𖾏 𖼞𖽪𖽙 𖼐𖽡𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 .
21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ,
𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾏 𖼡𖽑𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽡，
22 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽪𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼅𖽡 .

24 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 ,

25 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 , 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 ,
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼮𖽵𖾑 𖼳𖽺 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 .
26 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 ,
𖼯𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽷𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼊𖽪𖾏 .
27 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼚𖽔𖾐 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 ,
𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼎𖽙 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑．
 
28 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼘𖽪 ! 𖼅𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 ,
𖼞𖽑𖽔𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 !
29 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 ,
𖼯𖽡𖽙𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 ,
𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼐𖽑𖽘𖾑 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
30 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 ,
𖼐𖽱𖾏 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 .

“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽘𖾏 !”
