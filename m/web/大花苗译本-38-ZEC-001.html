𖼊𖽔𖾏 𖼖𖽡𖾏 𖽂𖽪𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼾𖽡𖾏 𖼖𖽑𖽡𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼽𖽡𖾏 𖼊𖽙𖾑 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 , 𖼐𖽑𖽡𖾏: 2 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝. 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽔𖾏 𖼅𖽡, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 4 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼾𖽻 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐, 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: ‘𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑： 𖼅𖽡 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼅𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.’ 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 5 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑? 𖼸𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑 𖼸𖽠𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼯𖽡𖽝𖾏 𖽐𖼎𖽙𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏? 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽃𖽡𖾑, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐, 𖼐𖽑𖽡𖾏: ‘𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼸𖽙𖾏 𖼀𖽡𖾐 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼯𖽡𖽙𖾏 𖼀𖽡𖾐, 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼋𖽘𖾏. ’ ”
𖼁𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖽃𖽡𖾑 𖼼𖽘𖾏 𖼼𖽔𖾏
7 𖼊𖽔𖾏 𖼖𖽡𖾏 𖽂𖽪𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏, 𖼟𖽪𖾏 𖼳𖽡𖾏 𖼀𖽔𖾏 𖼊𖽑𖽳 𖼖𖽑𖽡𖾏, 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼚𖽻𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼽𖽡𖾏 𖼊𖽙𖾑 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 , 𖼐𖽑𖽡𖾏: 8 “𖼞𖽪𖾑 𖼋𖽘 𖼄𖽑𖽙𖾏 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽜𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼖𖽡𖽝𖾐, 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼦𖽘𖾑 𖽃𖽡𖾑 𖼞𖽙𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼣𖽑𖽪𖾐 𖼉𖽜 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐. 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼑𖽜𖾏 𖼖𖽡𖽝𖾐, 𖼑𖽜𖾏 𖼖𖽡𖽝𖾐 𖼚𖽪𖾐 𖽐𖼎𖽙 𖼑𖽜𖾏 𖼚𖽠𖾐.” 9 𖼞𖽪𖾑 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼘𖽪! 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼟𖽡 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏.” 10 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽙𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼣𖽑𖽪𖾐 𖼉𖽜 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐, 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑.” 11 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼣𖽑𖽪𖾐 𖼉𖽜 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏, 𖼁𖽙𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 .”
