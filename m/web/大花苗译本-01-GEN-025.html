25
𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼎𖽺𖾏 𖼷𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐, 𖽐𖼷𖽷𖾏 𖼮𖽡𖾑 𖼊𖽑𖽪𖾐 𖼖𖽔𖾑 ． 2 𖼮𖽡𖾑 𖼊𖽑𖽪𖾐 𖼖𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼳𖽡𖽾𖾑 𖼖𖽔、𖼽𖽙 𖼳𖽗𖾑、𖼄𖽡𖾐 𖼊𖽗𖾏、𖼄𖽡𖾐 𖼊𖽡𖽝𖾏、𖼽𖽡𖾑 𖼳𖽷𖾑 𖼀𖽔𖾑、𖽐𖼎𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏. 3 𖼽𖽙 𖼳𖽗𖾑 𖼅𖽔𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼊𖽗𖾏. 𖼊𖽡𖾐 𖼊𖽗𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽪𖾑 𖼖𖽡𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖼖𖽡𖾏 𖼊𖽪𖾑 𖼳𖽷𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽐𖼎𖽙 𖼖𖽡𖾏 𖽂𖽪𖾑 𖼄𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐. 4 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼇𖽔、𖼽𖽡𖾑 𖼇𖽪、𖼦𖽔𖾐 𖼐𖽙、𖼽𖽔𖾏 𖼀𖽡𖾐 𖼊𖽔𖾏、𖼽𖽡𖾑 𖼖𖽳 𖼊𖽔𖾏. 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽑𖽪𖾐 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏. 5 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐. 6 𖽃𖽡𖾐 𖼈𖽡𖽝𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼣𖽔𖾏 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖼳𖽻𖾏 𖽃𖽡𖾑, 𖼷𖽪𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼺𖽷𖾐, 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽜𖾑 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽔𖾐, 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼙𖽠𖾏.
𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐
7 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏. 8 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏, 𖼊𖽪𖾏 𖼀𖽘𖾏 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 (“𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 ” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 ” ) 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 9 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐、𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼄𖽳 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼖𖽪𖾑 𖽐𖼚𖽙 𖼡𖽑𖽙𖾑． 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼄𖽔𖾏 𖼖𖽡𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏, 𖼦𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽙𖾐 𖼳𖽢 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼇𖽪 𖼖𖽪𖽝 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐， 10 𖼟𖽪𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽐𖼎𖽙 𖼦𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏. 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼖𖽔𖾏 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 11 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼋𖽔 𖼺𖽘𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽔𖾐. 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼐𖽡𖽙𖾐 𖼊𖽡𖾏 𖼀𖽡𖾐 𖽃𖽳𖾐· 𖼖𖽔𖾑 𖼦𖽹𖾐·𖼖𖽹.
𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 1：28 - 31）

24 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼸𖽙 𖼋𖽘𖾏, 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼾𖽙 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐. 25 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼳𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐, 𖽃𖽡𖾐 𖼮𖽡𖾏 𖼙𖽱 𖼅𖽔𖾏 𖼚𖽻𖾐, 𖼺𖽱𖾏 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 ( “𖼽𖽡𖾑 𖼺𖽻𖾐” 𖼟𖽪𖾏 “𖼅𖽔𖾏 𖼚𖽻𖾐 ”). 26 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑 , 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼊𖽔𖾑 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖽃𖽔 𖼙𖽻𖾏 𖼊𖽠𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 ( “𖼽𖽔𖾐 𖼞𖽙” 𖼟𖽪𖾏 “𖼊𖽔𖾑 𖼏𖽔𖾏 ” ). 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖼳𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏. 27 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼘𖽑𖽙𖾐 𖼟𖽡𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 , 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼘𖽑𖽻𖾐 𖽐𖼡𖽔𖾏 𖽐𖼊𖽻𖾏 𖽐𖼡𖽹𖾏, 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑； 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 , 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼚𖽙 . 28 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽐𖼮𖽑𖽠𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼑𖽻𖾏 𖼐𖽱𖾏 𖽐𖼡𖽹𖾏 𖼻𖽻𖾏 𖼸𖽔；𖼈𖽡𖽝𖾏 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖼯𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖼽𖽔𖾐 𖼞𖽙.

29 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐, 𖼽𖽔𖾐 𖼞𖽙 𖼎𖽑𖽻𖾐 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼋𖽻𖾏, 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼺𖽷𖾐 𖼗𖽙 𖼚𖽑𖽔𖾏 𖼋𖽘𖾏. 30 𖼽𖽡𖾑 𖼺𖽻𖾐 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼚𖽑𖽔𖾏 𖼋𖽘𖾏, 𖼮𖽑𖽙 𖼟𖽡 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼞𖽔𖾏 𖼋𖽻𖾏 𖼖𖽡𖽝𖾐 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽻𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼎𖽺𖾏 𖽐𖼷𖽷𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 (“𖼽𖽡𖾐 𖼊𖽪𖽾𖾑” 𖼟𖽪𖾏 “𖼖𖽡𖽝𖾐 ” ). 31 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼺𖽪𖾐 𖼄𖽔𖾐 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼙𖽻 𖼋𖽪𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼞𖽪𖾑.” 32 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽑𖽹𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔 𖼋𖽘𖾏, 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼙𖽻 𖼋𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏?” 33 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼀𖽘𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼏𖽵 𖼗𖽪𖾏!” 𖼽𖽡𖾑 𖼺𖽻𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼏𖽵 𖼗𖽪𖾏, 𖼄𖽔𖾐 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼙𖽻 𖼋𖽪𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙. 34 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾐 𖼞𖽙 𖼄𖽔𖾐 𖼳𖽡𖽝𖾐 𖽐𖼎𖽙 𖼞𖽔𖾏 𖼋𖽻𖾏 𖼖𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐, 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽻𖾏 𖼋𖽘𖾏 𖼦𖽻𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽔𖾏 𖽐𖼮𖽠𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼙𖽻 𖼋𖽪𖾏.
