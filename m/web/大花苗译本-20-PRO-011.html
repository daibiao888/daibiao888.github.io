11
𖼸𖽱𖾏 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐,
𖽃𖽔𖾑 𖼖𖽪𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 .
2 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏, 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼖𖽔𖾏 𖼋𖽔𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼾𖽨𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼑𖽡𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼺𖽘𖾏.
4 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼮𖽔 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏.
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖼑𖽡𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼎𖽺𖾏 𖼋𖽡𖽝𖾏 𖼎𖽺𖾏 𖽐𖼞𖽳𖾏,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼢𖽻 .
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼾𖽨𖾏,
𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼦𖽠𖾑 𖼉𖽳 𖽐𖼚𖽙 .
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽂𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼞𖽑𖽔𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏；
𖼊𖽪𖾐 𖼾𖽔𖾏 𖼑𖽡𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔,
𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼗𖽙𖾏 𖼘𖽑𖽪𖾑 𖼐𖽱𖾏 .
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼼𖽷𖾏 𖼼𖽙 .
𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼊𖽻𖾏 𖼮𖽨𖾏.
10 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏, 𖽃𖽡𖾐 𖼙𖽻 𖼀𖽡𖾏 𖼙𖽡𖽝𖾏 𖼙𖽔 𖼚𖽙𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏, 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖽃𖽺𖾐 𖼙𖽔 .
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼖𖽪𖾑 𖼙𖽻 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐；
𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼊𖽑𖽠𖾐 𖽐𖼷𖽜𖾑.
12 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼼𖽷𖾏 𖼼𖽙 , 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 ；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼗𖽪𖾏.
13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼐𖽑𖽡𖾏 𖼺𖽹𖾑 𖼗𖽪𖾏, 𖼮𖽑𖽻𖾏 𖼙𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼸𖽠𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼑𖽪𖾏 .
14 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽺𖾏 𖼦𖽡 𖼋𖽙；
𖼊𖽪𖾐 𖼸𖽠𖾏 𖼮𖽡𖽻𖾐, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏.
15 𖼑𖽜 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻𖾏 𖽃𖽦𖾐,
𖼳𖽪𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽔𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾑 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
16 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 ,
𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼊𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐.
17 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼾𖽨𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼦𖽺𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
18 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 , 𖼊𖽻𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼟𖽪𖾏 𖽐𖼊𖽘𖾐 𖼦𖽡𖾏 𖽐𖼚𖽙𖾏,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑, 𖼊𖽻𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼋𖽡𖽝 .
19 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑；
𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏.
20 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐, 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 .
21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏.
22 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼮𖽷𖾑 ,
𖼺𖽱𖾏 𖼁𖽻 𖼞𖽪𖾐 𖼎𖽻𖾏 𖼎𖽻𖾏 𖽐𖼀𖽔𖾏 𖽃𖽔𖾑 𖽐𖼀𖽱.
23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐.
24 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼙𖽳𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏；
