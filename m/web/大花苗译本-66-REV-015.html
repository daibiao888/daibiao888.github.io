15
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑，𖼟𖽪𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼳𖽤𖾏 𖼼𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼋𖽘𖾏 𖽃𖽡𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽤𖾏 𖼼𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽠𖾏 𖼟𖽙𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏．
2 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼺𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼊𖽡𖾏 𖽃𖽻𖾏， 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼳𖽡𖽠𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏；𖼞𖽪𖾑 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼊𖽪𖾑 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼑𖽡𖾏 𖼁𖽠𖾏 𖼼𖽘𖾏，𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼯𖽠，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼊𖽡𖾏 𖽃𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 ， 𖼯𖽡𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾑 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽷𖾑 𖼺𖽙𖾑． 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽜𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼎𖽙 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏， 𖼐𖽑𖽡𖾏：
“𖼮𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼘𖽪！
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑！
𖼉𖽻𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏！
4 𖼮𖽵𖾑 𖼘𖽪 ！𖼡𖽔 𖼋𖽱𖾏 𖼞𖽔𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽡，
𖼦𖽡𖾐 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼖𖽺𖾑？
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽔 𖼅𖽔𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑，
𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼋𖽔𖾏 ，
𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳， 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏．”
5 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽜𖾐 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏． 6 𖼳𖽤𖾏 𖼗𖽜𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼳𖽤𖾏 𖼼𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔，𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼟𖽡𖾏 𖼮𖽡𖾏 𖼮𖽑𖽦𖾑，𖼖𖽪𖾑 𖼚𖽔𖾑 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 𖼞𖽪𖾐． 7 𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪， 𖼞𖽑𖽡𖽠𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼞𖽑𖽙𖾐 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼄𖽔𖾐 𖼎𖽻𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽃𖽡𖾑． 8 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐， 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏． 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 ，𖼽𖽔𖾑 𖼋𖽙𖾏 𖼸𖽙 𖼳𖽤𖾏 𖼼𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼞𖽳𖾑 𖼯𖽵𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏．
