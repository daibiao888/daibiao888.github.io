14
𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼗𖽻 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐. 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 3 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽠𖾏 𖼷𖽹𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽘 , 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼛𖽠𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼾𖽔𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼞𖽪𖾑 𖼖𖽺𖾑? 4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼄𖽔𖾐 𖼁𖽠𖾏 𖼷𖽹𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼾𖽔𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼈𖽡𖽝𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽐𖼊𖽜, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏, 5 𖼻𖽻𖾏 𖼦𖽔𖾏 𖼎𖽙𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼁𖽠𖾏 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐.
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏, 𖼐𖽑𖽡𖾏: 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙！ 𖼇𖽳𖾑 𖼅𖽡 𖼋𖽡𖾏 𖼁𖽠𖾏 𖼺𖽘𖾏 , 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽑𖽪𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐. 7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼇𖽺𖾐 𖼡𖽔𖾏, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽠𖾏 𖼷𖽹𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼾𖽔𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼈𖽡𖽝𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼾𖽨𖾏 𖼽𖽔𖾑 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏. 8 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼷𖽜𖾑 𖼚𖽑𖽪𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼮𖽡𖽠𖾏, 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽙𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏. 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 9 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼮𖽑𖽵𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼯𖽔𖾏 𖼷𖽷𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼮𖽑𖽵𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏. 10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏. 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽜 𖼑𖽪 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐, 11 𖼞𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼇𖽳𖾑 𖼞𖽪𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼏𖽻 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼾𖽨𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
