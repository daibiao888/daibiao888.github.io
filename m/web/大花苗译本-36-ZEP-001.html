𖼳𖽡𖾑 𖼇𖽔𖾑 𖼽𖽔𖾐 𖽐𖼊𖽠𖾑
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏
1
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼚𖽻𖾐 𖼷𖽑𖽷𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼘𖽔𖾑𖽑、 𖼮𖽡𖾑 𖼊𖽔𖾏 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑、 𖼞𖽪𖾐 𖼳𖽷 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼇𖽔𖾑 𖼽𖽔𖾐 .
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼖𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾐 𖼊𖽡𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 .
3 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼯𖽦𖾏 ,
𖽐𖼎𖽙 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏, 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑,
𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽠𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐,
𖼞𖽪𖾑 𖼖𖽔𖾑 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏.”
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
4 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏
𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ,
𖼎𖽺𖾏 𖼖𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼀𖽔𖾑 𖼖𖽡 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑,
𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽠𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏,
5 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖼎𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼼𖽘𖾏 𖼼𖽔𖾏 𖽃𖽡𖾑,
𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼏𖽵 𖼗𖽪𖾏 ,
𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼄𖽔𖾐 𖼖𖽳 𖼞𖽑𖽗𖾑 𖼏𖽵 𖼗𖽪𖾏 𖽃𖽡𖾑,
6 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼺𖽷𖾐 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾑,
𖼦𖽡𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼐𖽱𖾏 𖽃𖽡𖾑.
 
7 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼋𖽘𖾏,
𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖼋𖽔𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏.
8 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼨𖽑𖽢𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐,
𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽙𖾏 𖽃𖽡𖾑.
9 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼚𖽑𖽡𖽝𖾏 𖼮𖽑𖽡𖽻𖾏 𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏,
𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖽐𖼎𖽙 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼗𖽙𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑.
 
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼎𖽑𖽻𖾏 𖽐𖼀𖽳𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼖𖽔𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽔𖾐 𖼚𖽻𖾏 𖼚𖽻𖾏 𖽃𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖽃𖽔𖾑 𖼯𖽢𖾏 𖼟𖽡𖾏 𖼖𖽔𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏,
𖼎𖽑𖽻𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽙𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼼𖽠𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
11 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐𖼄𖽔𖾐 𖼞𖽳 𖼊𖽑𖽡 𖼳𖽷𖾑 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼛𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏 !
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼙𖽪𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏.
12 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖽐𖼮𖽡 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼎𖽻𖾏 𖼡𖽔𖾑 𖼮𖽡𖽠𖾑 𖽃𖽡𖾑.
𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏:
‘ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .’
13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,𖽃𖽺𖾐 𖼋𖽡𖽝 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏,𖽃𖽺𖾐 𖼋𖽡𖽝 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖾐 𖼡𖽔𖾏 𖼁𖽙𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽙 ,
𖼯𖽡𖽙 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑.
 
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼨𖽑𖽻𖾏 𖼊𖽡𖾐,
𖼨𖽑𖽻𖾏 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼇𖽺𖾏,
𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼑𖽡𖾏 𖼳𖽘𖾐,
𖼊𖽪𖾐 𖼋𖽻𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 .
15 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑,
𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏,
𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼚𖽪𖾐 𖼖𖽡𖾐 𖽐𖼮𖽡𖽝𖾐,
16 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏,
𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼏𖽔𖾏 𖽐𖼎𖽙 𖼙𖽻 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑.
 
17 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼟𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽘𖾑, 𖼖𖽔𖾑 𖽐𖼮𖽢𖾑𖽑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼺𖽱𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏, 𖼖𖽔𖾑 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼺𖽱𖾏 𖼮𖽑𖽡𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏.
18 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏, 𖼽𖽔𖾑 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼖𖽻𖾐 𖼺𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖼖𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
