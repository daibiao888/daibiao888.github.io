𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏, 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼺𖽔𖾐 𖼊𖽔 𖼖𖽔𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐. 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼊𖽔 𖼐𖽑𖽡𖾏:“𖼺𖽔𖾐 𖼊𖽔 𖼘𖽪 ! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖼟𖽡, 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼺𖽹𖾑 𖼞𖽑𖽡𖽠𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼟𖽡 , 𖼊𖽔𖾑 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖽃𖽡𖾐 𖼋𖽪 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?” 3 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏, 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐. 4 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏.” 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽔𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼐𖽑𖽘𖾑.” 5 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼊𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼊𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏,𖼎𖽺𖾏 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽘𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 .
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: 7 “𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑： 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑, 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑, 𖼟𖽡 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏, 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽻𖾏 𖼗𖽙𖾏 . 8 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼘𖽪 ! 𖼟𖽡 𖽐𖼎𖽙 𖼽𖽺 𖼗𖽔 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻． ( 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏 ). 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼑𖽡𖾏 “𖼊𖽪𖾑 𖼟𖽻𖾏” 𖼘𖽑𖽘𖾑 𖼋𖽠𖾏 𖼋𖽔𖾏. 9 𖼑𖽔𖾏 𖼘𖽪 ！ 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔 . 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼾𖽨𖾏 𖼸𖽻 𖼞𖽳𖾏 𖼮𖽡𖽙𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼊𖽔𖾐, 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 10 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼼𖽻𖾏 𖼎𖽻𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖽃𖽡𖾑 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
