𖼄𖽔𖾐 𖼞𖽑𖽙𖾐 𖼄𖽻𖾏 𖼻𖽻𖾏
𖼽𖽙 𖼦𖽗 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏
（𖼊𖽑𖽹𖾏 3：1-12；𖼖𖽪𖾏 3：1-18；𖼽𖽙 1：19-28）
1
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐（𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖽐𖼊𖽠𖾑 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐” 𖽃𖽡𖾐 𖼗𖽙𖾏 𖼐𖽡𖾑 ）𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪， 𖼐𖽱𖾏 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼁𖽱𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏．
2 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 ：
“ 𖼑𖽔𖾏 𖼘𖽪， 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼊𖽑𖽔𖾏，
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏；
3 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：
‘ 𖼽𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽵𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏，
𖼎𖽺𖾏 𖼨𖽑𖽢𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏．’ ”
4 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑， 𖼅𖽔𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗 𖼋𖽔𖾏， 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 ， 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼊𖽻𖾏 𖽐𖼷𖽔𖾑， 𖼟𖽙𖾏 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏． 5 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼯𖽜𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏， 𖼐𖽡𖽙𖾐 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼊𖽻𖾏 𖽐𖼷𖽔𖾑 𖼋𖽘𖾏． 6 𖼽𖽙 𖼦𖽗 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼚𖽻𖾐 𖼘𖽙 𖼊𖽑𖽙𖾐 𖽐𖼊𖽙𖾏， 𖼳𖽦𖾏 𖼖𖽑𖽔𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏， 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼷𖽷𖾑 𖼷𖽷 𖽐𖼎𖽙 𖼞𖽔𖾏 𖼄𖽪𖾑 𖼉𖽺𖾏 𖼻𖽻𖾑． 7 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏：“ 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑， 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼚𖽠𖾏 𖼖𖽑𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏． 8 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖽐𖼷𖽔𖾑 𖼅𖽡 𖼊𖽔𖾐， 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖽐𖼷𖽔𖾑 𖼅𖽡．”
𖼽𖽙 𖼦𖽗 𖽐𖼷𖽔𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 3：13-17；𖼖𖽪𖾏 3：21-22）
9 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼙𖽻 𖼋𖽔𖾏，𖼐𖽡𖽙𖾐 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼊𖽻𖾏 𖼽𖽙 𖼦𖽗 𖽐𖼷𖽔𖾑 𖼐𖽱𖾏． 10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖽃𖽡𖾐 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏， 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼡𖽑𖽜𖾐 𖼋𖽘， 𖼅𖽔𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼺𖽱𖾏 𖽐𖼡𖽔𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ． 11 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼋𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏， 𖼞𖽪𖾑 𖼙𖽔 𖼟𖽡 𖼊𖽔𖾐 𖼋𖽡𖽝．”
（𖼊𖽑𖽹𖾏 4：1-11；𖼖𖽪𖾏 4：1-13）
12 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏． 13 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼏𖽻 𖼺𖽔𖾐 𖼊𖽔 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑， 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼐𖽡𖽙𖾐， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏．
（𖼊𖽑𖽹𖾏 4：12-17；𖼖𖽪𖾏 4：14-15）
14 𖼽𖽙 𖼦𖽗 𖼊𖽻𖾏 𖼄𖽔𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽻𖾏 𖼻𖽻𖾏， 15 𖼐𖽑𖽡𖾏：“𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼀𖽪𖾑 𖼋𖽘𖾏， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏， 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 𖽐𖼮𖽜 𖼄𖽻𖾏 𖼻𖽻𖾏．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼚𖽻𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏
（𖼊𖽑𖽹𖾏 4：18-22；𖼖𖽪𖾏 5：1-11）
16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼊𖽙𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 𖼅𖽻𖾏， 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼳𖽡𖾑 𖼄𖽞， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼣𖽔𖾑 𖼊𖽳 𖼖𖽡𖽝 𖼗𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼎𖽻𖾏 𖼳𖽡𖽠𖾑； 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼾𖽠𖾏 𖽐𖼀𖽳𖾏． 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼾𖽠𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼀𖽳𖾏 ．” 18 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 ，𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏． 19 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽻𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼏𖽔𖾏 𖼞𖽡𖾑 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏， 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼊𖽑𖽹𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼽𖽙 𖼦𖽗 𖼐𖽡𖽙𖾐 𖽐𖼞𖽙𖾏 𖽐𖼷𖽷𖾑 𖼸𖽙𖾏， 20 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼋𖽔𖾏， 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼇𖽳𖾑 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼊𖽑𖽹𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏 𖼎𖽻𖾏 𖽐𖼞𖽙𖾏 𖼮𖽡𖽝𖾏， 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐
（𖼖𖽪𖾏 4：31-37）
21 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾 𖽐𖼚𖽙 𖼙𖽻， 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐． 22 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏， 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏． 23 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑， 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑， 24 𖼐𖽱𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖼀𖽡𖾐 𖽐𖼎𖽙𖾏 𖼟𖽡 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼐𖽔𖾑？ 𖼟𖽡 𖼋𖽔𖾏 𖼖𖽻𖾐 𖼀𖽡𖾐 𖼘𖽪𖾏 ？ 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏， 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 ．” 25 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “ 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏， 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏！ ” 26 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼊𖽑𖽡 𖼗𖽻𖾏 𖼋𖽘， 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏， 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏． 27 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏， 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼑𖽪 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：“ 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖽃𖽔𖾑 𖼳𖽷𖾏 𖼐𖽔𖾑 ？ 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 ！ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏， 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏！” 28 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽺𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼊𖽙𖾏 𖽐𖼷𖽷𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼳𖽡𖾑 𖼄𖽞 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽹𖾏
（𖼊𖽑𖽹𖾏 8：14-17；𖼖𖽪𖾏 4：38-41）
29 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽂𖽪𖾏 𖼋𖽠𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼗𖽙𖾏， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼊𖽑𖽡𖽝𖾐 𖼽𖽙 𖼦𖽗 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼄𖽞 𖼊𖽑𖽡𖽝𖾐 𖼣𖽔𖾑 𖼊𖽳 𖼖𖽡𖽝𖽐𖼞𖽔𖾏． 30 𖼳𖽡𖾑 𖼄𖽞 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽹𖾏 𖼄𖽙𖾐 𖼄𖽙𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼞𖽪𖾐 𖼘𖽑𖽳𖾐 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑． 31 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼁𖽙𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽘𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏，𖼄𖽙𖾐 𖼟𖽪𖾏 𖼞𖽪𖾐 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏， 𖼖𖽔𖾑 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
32 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 ， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽳𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼄𖽙𖾐， 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏， 𖼋𖽔𖾏 𖼸𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 33 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏． 34 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐，𖼎𖽺𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼙𖽠𖾏， 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼐𖽔𖾑．
（𖼖𖽪𖾏 4：42-44）
35 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑， 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼳𖽠𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏， 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏． 36 𖼳𖽡𖾑 𖼄𖽞 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖽃𖽡𖾑，𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑， 37 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡．” 38 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼐𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏， 𖼟𖽪𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼐𖽡𖾑， 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖽃𖽡𖾐 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼋𖽔𖾏 𖼯𖽡𖽙．” 39 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼚𖽔𖾏𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱， 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏、 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏
（𖼊𖽑𖽹𖾏 8：1-4；𖼖𖽪𖾏 5：12-16）
40 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏 𖼋𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏， 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼮𖽔𖾑 𖼯𖽪𖾏， 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏．” 41 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏， 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼀𖽔𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼯𖽪𖾏 𖼯𖽡𖽙， 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏！” 42 𖼄𖽙𖾐 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏，𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏． 43 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼅𖽻𖾏， 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： 44 “ 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏，𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 ， 𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏， 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾏 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑， 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏．” 45 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏 ，𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼄𖽔𖾐 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾐 𖼻𖽻𖾏 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼚𖽔𖾏 𖼙𖽻，𖼯𖽔 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼊𖽔𖾐． 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑．
