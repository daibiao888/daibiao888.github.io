𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏， 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏.
2 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏,
𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼐𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑 𖼀𖽪𖾑 𖼋𖽘𖾏,
𖼐𖽱𖾏 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼐𖽱𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼾𖽔𖾏,
𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
 
3 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽡𖾑，
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽔𖾏 𖼨𖽑𖽢𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 .
4 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽘𖾑 𖼊𖽙𖾐 𖼽𖽔𖾑 𖼷𖽑𖽪𖾐 𖼟𖽪𖾏 𖼀𖽪𖾑 ,
𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼳𖽻𖾐 𖼽𖽔𖾑 𖼊𖽻𖾑𖽑 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏,
𖼺𖽡𖽝𖾐 𖼺𖽡𖽝𖾐 𖼢𖽜 𖼢𖽜 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 ,
𖼀𖽡𖾐 𖽐𖼞𖽡𖾐 𖼀𖽡𖾐 𖽐𖼞𖽱𖾐 𖼽𖽔𖾑 𖼚𖽠𖾏 𖽃𖽔𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏.
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼑𖽔𖾏 𖼁𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
 
6 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼄𖽔！”
𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?”
𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽐𖼡𖽜𖾐,
𖼐𖽱𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼑𖽡𖾏 𖼁𖽘𖾏.
7 𖽐𖼡𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡, 𖼁𖽘𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼀𖽑𖽙𖾐 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼀𖽘𖾏 𖼷𖽑𖽔𖾐 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑.
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼾𖽙 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽐𖼡𖽜𖾐.
8 𖽐𖼡𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡, 𖼁𖽘𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼉𖽜 ,
𖼮𖽔 𖼅𖽔𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼽𖽔𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏.”
𖼮𖽵𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏
9 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼻𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼘𖽪！
𖼟𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼻𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！
𖼟𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏,
𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽻 𖼐𖽑𖽡𖾏:
“𖼑𖽔𖾏 𖼘𖽪！ 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．”
10 𖼑𖽔𖾏 𖼘𖽪 ! 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏,
𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.
𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽱𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
11 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼊𖽪𖾑 𖼾𖽨 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐,
𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼦𖽡 𖼸𖽱 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼁𖽔 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐,
𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼊𖽪𖾑 𖼾𖽨 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖽃𖽡𖾑.
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖽐𖼮𖽔𖾏 𖼋𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
12 𖼡𖽔 𖼋𖽱𖾏 𖼙𖽔 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼋𖽡 𖽐𖼮𖽔𖾏 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 ,
𖼎𖽑𖽻𖾏 𖼙𖽘𖾑 𖼙𖽘𖾏 𖽃𖽔𖾏 𖼚𖽙𖾏 𖽐𖼮𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼎𖽑𖽻𖾏 𖼳𖽜𖾐 𖼊𖽱𖾏 𖼙𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 ,
𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼞𖽡𖾏 𖼙𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐,
𖼎𖽑𖽻𖾏 𖼸𖽱𖾏 𖼞𖽡𖾏 𖼙𖽪𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼖𖽺𖾑?
13 𖼡𖽔 𖼋𖽱𖾏 𖼙𖽔 𖽐𖼮𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖽾 ,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼸𖽠𖾐 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼖𖽺𖾑?
14 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏, 𖼡𖽔 𖼋𖽱𖾏 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 ,
𖼡𖽔 𖼋𖽱𖾏 𖼄𖽔𖾐 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 ,

15 𖼑𖽔𖾏 𖼘𖽪 , 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼊𖽑𖽻𖾐 𖽃𖽻𖾐 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼀𖽡𖾑 𖼙𖽱 ,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼸𖽱𖾏 𖼞𖽡𖾏 𖼑𖽡𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏,
𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐, 𖼺𖽱𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐.
16 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼑𖽡𖾏 𖽐𖼊𖽻𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼚𖽻𖾑 ,
𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 .
17 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼅𖽔𖾏,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼗𖽙𖾏 𖼦𖽡𖾐 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼟𖽡𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼋𖽡𖽝 .
 
18 𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼗𖽙𖾏 𖽐𖼮𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼯𖽡𖽙𖾏 𖼼𖽘𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡 𖽐𖼮𖽔 𖼖𖽺𖾑?
19 𖼊𖽪𖾑 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼀𖽔𖾐 𖼘𖽡𖾑𖽑 ,
𖼖𖽡𖾏 𖼞𖽱𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏,
𖼑𖽜 𖼐𖽱𖾏 𖼘𖽡𖾑𖽑 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼑𖽡𖽝𖾏.
20 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼋𖽡𖾏 𖼀𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼐𖽡𖾑 ,
𖼯𖽡𖾏 𖼺𖽹𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼗𖽱𖾐 ,
𖼑𖽜 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼸𖽠𖾏,
𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼼𖽙 .
 
21 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼀𖽻𖾐 𖼘𖽪?
𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼘𖽪?
𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼋𖽳 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼊𖽡𖾐 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏 ,
𖼅𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?
22 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼺𖽱𖾏 𖼷𖽷 𖼟𖽻𖾏．
𖼐𖽱𖾏 𖼊𖽑𖽔𖾑 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐,
𖼊𖽑𖽔𖾑 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐.
23 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽙𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏,
𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑.
 
24 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼯𖽡𖽙 𖼮𖽡𖽝𖾏,
𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼀𖽳𖾐 𖼮𖽡𖽝𖾏,
𖽃𖽔𖾑 𖼯𖽤𖾐 𖼖𖽔𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼗𖽔 𖼎𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽙 ,
𖼐𖽱𖾏 𖽃𖽡𖾐 𖼷𖽑𖽔𖾐 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑,𖼎𖽺𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼋𖽘𖾏,
𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽔𖾐 𖼙𖽠𖾏, 𖼺𖽱𖾏 𖽃𖽔𖾏 𖽐𖼊𖽻𖾏 𖼡𖽻𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.
25 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼗𖽙𖾏 𖽐𖼮𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 ,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖼺𖽱𖾏 𖼖𖽺𖾑?”
26 𖼅𖽡 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐,
𖼑𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼷𖽷𖾏 𖼋𖽡𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼐𖽡𖾑 ,
𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖽃𖽡𖾐 𖼼𖽘𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼮𖽔𖾑𖽑 𖼯𖽠 .
𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
27 𖼽𖽔𖾐 𖼞𖽙 𖼘𖽪! 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑？
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪! 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏,
𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼨𖽑𖽢𖾐？”
28 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼀𖽻𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼋𖽡𖽝𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐, 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼯𖽻 𖼚𖽑𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽻 𖼼𖽙 ,
𖼐𖽱𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖽐𖼞𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽐𖼮𖽔𖾏 .
29 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏, 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽻 𖼼𖽙， 𖼐𖽱𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏.
30 𖼾𖽙 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼯𖽻 𖼛𖽘𖾏 𖼯𖽻 𖼼𖽙,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼢𖽻 𖼙𖽠𖾏.
31 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏．
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼚𖽘𖾑 𖽐𖼊𖽑𖽔𖾑 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼽𖽘𖾏 𖼉𖽳𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽘 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼯𖽻 𖼼𖽙 ,
𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏.
