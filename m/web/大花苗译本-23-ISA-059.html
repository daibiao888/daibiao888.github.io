5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽔 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼡𖽳𖾏,

𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼡𖽳𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔．
𖼋𖽡𖾏 𖼡𖽳𖾏 𖼐𖽡𖾑 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼯𖽪𖾏, 𖼽𖽔𖾑 𖼚𖽑𖽘𖾏 𖼋𖽠𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖽐𖼚𖽪𖾏.

𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼊𖽔𖾐 𖼋𖽡𖽝.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼏𖽘 𖼇𖽺𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽺𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,

8 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖽝𖾏 𖼋𖽪, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐,

𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐.
𖼮𖽵𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏

𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽑𖽪𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼽𖽨𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,
𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏.
17 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼐𖽑𖽘𖾑 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼷𖽷𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼮𖽱𖾑 𖼖𖽱𖾐 𖼋𖽻𖾏 𖽐𖼊𖽻𖾏 𖼮𖽡𖽝𖾏 ,
𖼞𖽑𖽡𖽠𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽑𖽘𖾑 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖼳𖽙𖾑 𖼐𖽑𖽘𖾑 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼙𖽠𖾏 .
18 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏,
𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑,
𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑,
𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏.
19 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑,𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏,
