38 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 𖼛𖽙𖾏 ,
𖼦𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 , 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑 ?
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼀𖽘𖾏 𖼅𖽡 ,
𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼅𖽡 !
 
39 𖼅𖽡 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼀𖽻𖾐 , 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖽃𖽡𖾑 ,
𖼋𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ,
𖼞𖽪𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔 , 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 ,
𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽨𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
40 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽑𖽡𖾏 ,
𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼏𖽵 𖼗𖽪𖾏 :
41 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼨𖽑𖽙𖾑 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖾏 ,
𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 ,

42 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼈𖽙𖾏 𖼦𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 𖼦𖽻𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼢𖽻 𖼮𖽡𖽠𖾑 𖼙𖽠𖾏 ,
𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼗𖽠𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑 .
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼽𖽔𖾑 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 , 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏 .
 
43 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼘𖽪 ！ 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼙𖽔 𖼨𖽑𖽪𖾏 ,

48 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 49 𖼟𖽡 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼀𖽔𖾑 𖼖𖽡𖽾 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽡𖾑 𖼐𖽡 𖼀𖽙𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼐𖽡𖽙𖾐 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 , 𖼑𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽡𖾑 . 50 𖼟𖽡 𖼽𖽔𖾑 𖼋𖽔 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽃𖽡𖾑 , 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼗𖽠𖾏 (“𖽃𖽔𖾑 𖼾𖽠 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏”) , 𖼺𖽱𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 , 𖼋𖽔 𖼎𖽻𖾏 𖼦𖽙 𖽃𖽳𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 . 51 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼳𖽱 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 , 𖼮𖽢𖾑 𖼊𖽡𖾐 𖼺𖽷𖾑 𖽃𖽡𖾑 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼀𖽔𖾑 𖽃𖽻𖾐 , 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑 . 52 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼑𖽔𖾏 𖼁𖽙𖾏 , 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 .
