𖽐𖼷𖽘𖾏 𖼙𖽠𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼖𖽻𖾐
5
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼘𖽪！ 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼑𖽜 𖼅𖽡 𖽃𖽔𖾏 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼐𖽡𖾑.
2 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 ( “𖼖𖽡𖾑 𖼉𖽪𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑”) 𖼊𖽻𖾏 𖼢𖽻, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 ,
𖼐𖽱𖾏 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽘𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑.
3 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼙𖽻 , 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼏𖽙 𖼅𖽻𖾏,
𖼮𖽔𖾑 𖼳𖽙𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏, 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏, 𖼮𖽔𖾑 𖼳𖽙𖾏 𖼟𖽻𖾏 𖼗𖽜𖾏 .”
𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼮𖽵𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼅𖽡 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏.
5 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼚𖽔𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽡 𖼮𖽢 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽑𖽡𖽻𖾏 𖼚𖽔𖾏 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽡 𖼮𖽢 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏,
𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏.”
 
6 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽙 𖼺𖽳 𖼾𖽡𖾏 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼯𖽡 𖼗𖽙𖾏 ,
𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾏 𖼟𖽪𖾏 𖼋𖽔 .
7 𖼅𖽡 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼄𖽡𖾏 𖼇𖽪𖾑 𖽃𖽡𖾑,
𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐．
8 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼄𖽻𖾐、 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼳𖽺𖾑,
𖼟𖽙𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼁𖽙𖾏 𖼞𖽡𖾑,
𖼟𖽙𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼟𖽙𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼗𖽙𖾏 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑，𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
9 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑.
10 𖼅𖽡 𖼳𖽪𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽡𖾑,
𖼳𖽪𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽡𖾑.
11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏,
𖼼𖽠𖾏 𖼀𖽑𖽡𖽝𖾏 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 ,
𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼙𖽔 𖼸𖽻 𖼯𖽡𖽙𖾏 𖼗𖽙 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼈𖽡𖽝𖾏,
𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 ,
𖼯𖽡𖽙 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼈𖽡𖽝𖾏,
𖼦𖽡𖾐 𖼊𖽻𖾏 𖼦𖽻𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼮𖽡𖽠𖾑 .
12 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼅𖽡 𖼮𖽑𖽡𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐,
𖼅𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼅𖽡 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 ,
𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏.
13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼐𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽑𖽷𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
 
14 𖼅𖽡 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼯𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏.
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏, 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐.
15 𖼽𖽔𖾑 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖽐𖼮𖽑𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 .
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏
16 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼐𖽡𖽝𖾑,
𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼯𖽢𖾏 𖼟𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏:
‘ 𖼳𖽻𖾏 𖽐𖼚𖽔! 𖼳𖽻𖾏 𖽐𖼚𖽔!’
𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼋𖽔𖾏 𖼐𖽡𖽝𖾑,
𖼨𖽑𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐.
17 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
18 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！
𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽺𖾑?
𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 .
19 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽐𖼮𖽷𖾐 𖼚𖽹𖾏,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼀𖽔𖾐 𖼖𖽪𖾑 𖼮𖽑𖽤 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼋𖽙𖾏.
20 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼖𖽺𖾑?
𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼚𖽪𖾐 𖼖𖽡𖾐 𖽐𖼮𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼖𖽺𖾑?”
𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝 𖼮𖽵𖾑 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏
21 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼳𖽪𖾐 𖼅𖽡 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼅𖽡 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽹𖾑.
22 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼈𖽡𖽝𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼯𖽦𖾏 𖼛𖽙 𖼟𖽪𖾏 𖼅𖽡 𖼯𖽡𖽙𖾏 𖼗𖽙 𖽃𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 .
23 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼡𖽔𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽻 𖼅𖽡 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏.
24 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼦𖽡 𖽐𖼚𖽙 𖼀𖽡𖾐 𖼘𖽙𖾐 ,
𖼟𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽐𖼚𖽱𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏.
25 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼘𖽪！ 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼄𖽔𖾐 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼘𖽪? 26 𖼅𖽡 𖼞𖽱𖾑 𖼅𖽡 𖼮𖽵𖾑 𖼳𖽺 “𖼺𖽔𖾐 𖼞𖽪𖾐 ” 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼮𖽵𖾑 𖼳𖽺 “ 𖼮𖽢𖾑 𖽂𖽺𖾑 ” 𖼁𖽠𖾏 𖼼𖽘𖾏, 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽜 𖼾𖽨𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑. 27 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼙𖽠𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
