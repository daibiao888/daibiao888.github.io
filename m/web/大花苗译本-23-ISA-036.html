𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖼚𖽻𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼐𖽔 𖼮𖽡𖾑 𖼖𖽡𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽻 𖼟𖽪𖾏 𖼏𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼙𖽻. 2 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼖𖽔𖾑 𖼮𖽡 𖼷𖽷𖾐 𖼖𖽔𖾑 𖼀𖽳 𖼳𖽔𖾑 𖼮𖽡𖾑 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼏𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼸𖽷 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼟𖽱𖾏 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪, 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽻𖾏 𖽐𖼊𖽻𖾐 𖽃𖽡𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐. 3 𖼳𖽡𖾑 𖼖𖽳 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏, 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽃𖽔𖾏 𖽐𖼊𖽠𖾑 𖼳𖽳𖾐 𖼀𖽳 𖼐𖽔 𖼊𖽑𖽡𖽝𖾐 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼽𖽔𖾏 𖼟𖽪𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼑𖽪𖾏, 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼖𖽔𖾑 𖼀𖽳 𖼳𖽔𖾑 𖼮𖽡𖾑 .
𖼖𖽔𖾑 𖼀𖽳 𖼳𖽔𖾑 𖼮𖽡𖾑 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏
4 𖼖𖽔𖾑 𖼀𖽳 𖼳𖽔𖾑 𖼮𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼻𖽡𖽝𖾏 𖼘𖽪𖾏? 5 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽐𖼎𖽙 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐 𖼯𖽡𖽙 , 𖼟𖽡 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏, 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖼖𖽺𖾑? 6 𖼑𖽔𖾏 𖼘𖽪！ 𖼣𖽹 𖼮𖽡 𖼟𖽪𖾏 𖼟𖽡 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽔𖾑 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼖𖽙𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼐𖽡𖾑 , 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽙𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏. 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏. 7 𖼟𖽡 𖼮𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼀𖽡𖾐 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖼐𖽡𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼘𖽪?
8 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏, 𖼞𖽪𖾑 𖼄𖽔𖾐 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏 𖼑𖽜𖾏 𖼎𖽻𖾏 𖼟𖽡, 𖼑𖽔𖾏 𖼟𖽡 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼷𖽻𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏. 9 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙, 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽔𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖽃𖽡𖾑 𖼖𖽺𖾑? 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽐𖼎𖽙 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼺𖽷𖾐 𖼖𖽔𖾏? 10 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 , 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ? 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑.”
11 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼮𖽡𖾏 、 𖼳𖽳𖾐 𖼀𖽳 𖼐𖽔 、 𖼽𖽙 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼀𖽳 𖼳𖽔𖾑 𖼮𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽜𖾑 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼀𖽻𖾐 𖼯𖽡𖽙 ,𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏: 𖽐𖼮𖽑𖽹𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼐𖽙𖾑𖽑.”
12 𖼖𖽔𖾑 𖼀𖽳 𖼳𖽔𖾑 𖼮𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼋𖽔𖾏, 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼦𖽡𖾏 𖼇𖽔𖾑, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽡𖾐 𖼸𖽻 𖼑𖽻𖾏 𖼾𖽨𖾏 𖼡𖽔𖾑 𖼦𖽻𖾏 𖼾𖽨𖾏 𖼉𖽜𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?”
13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼖𖽔𖾑 𖼀𖽳 𖼳𖽔𖾑 𖼮𖽡𖾑 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏, 𖼯𖽡𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼗𖽪𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖼗𖽪𖾏 . 14 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼮𖽑𖽵𖾑 𖼋𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼅𖽡 . 15 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐, 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
16 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼗𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏． 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼚𖽻𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑, 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼾𖽨𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 , 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐, 𖼦𖽻𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼑𖽡𖾏 𖽃𖽻𖾐. 17 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼨𖽑𖽳𖾏 𖼅𖽡 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐, 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖽐𖼎𖽙 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 .
18 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼮𖽑𖽵𖾑 𖼡𖽑𖽔𖾐 𖼅𖽡 𖼐𖽑𖽡𖾏: 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐.𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼘𖽪?
19 𖼦𖽔𖾐 𖼄𖽔𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼀𖽔 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑? 𖼳𖽡𖾑 𖼇𖽔 𖽂𖽫𖾐 𖼽𖽡𖽾𖾑 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑? 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼮𖽨𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼘𖽪𖾏? 20 𖼋𖽡𖾏 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼙𖽔 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼟𖽙𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼘𖽪? 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?”
21 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏,𖽃𖽡𖾐 𖼗𖽙𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 .”
22 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼳𖽡𖾑 𖼖𖽳 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏, 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽃𖽔𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼳𖽳𖾐 𖼀𖽳 𖼐𖽔 𖼊𖽑𖽡𖽝𖾐 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼽𖽔𖾏 𖼟𖽪𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼑𖽪𖾏,𖼊𖽪𖾏 𖽐𖼮𖽑𖽷𖾏 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏, 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼄𖽔𖾐 𖼖𖽔𖾑 𖼀𖽳 𖼳𖽔𖾑 𖼮𖽡𖾑 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏.
