1
𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖼖𖽪𖾑 𖼚𖽻𖾐 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼀𖽜𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼀𖽔𖾑 𖼖𖽪𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏,𖼊𖽻𖾏 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏. 2 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼚𖽻𖾐 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼀𖽜𖾐, 3 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼮𖽢𖾑 𖼀𖽔𖾑 𖼖𖽪𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼀𖽪𖾏 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽡𖾑 𖼳𖽡𖾑 𖼮𖽡𖽝 𖼐𖽑𖽡𖾏. 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 (“𖼷𖽑𖽹𖾏 𖼋𖽡𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼛𖽘𖾏 𖼼𖽙𖾏”) 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
