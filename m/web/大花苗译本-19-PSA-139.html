7 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼟𖽡 𖼖𖽡𖽾?

10 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 ,
𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 .
11 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼞𖽪𖾑 ,

12 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼁𖽙𖾏,
𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼯𖽡𖽙 ,

13 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽐𖼀𖽱𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏,

14 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼷𖽷𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼷𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏,

15 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼁𖽙𖾏 𖼊𖽻𖾏 𖼷𖽷𖾏, 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼊𖽻𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 ,

16 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼺𖽷𖾐,
𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼷𖽙𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏,
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼏𖽵 𖼮𖽡𖽝𖾏, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼟𖽱𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 ，

20 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽱𖾑 𖼊𖽠𖾏 𖼟𖽡 ,

𖽐𖼮𖽔𖾏 𖼮𖽡𖽺𖾏 𖼞𖽪𖾑, 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏，
24 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏,
