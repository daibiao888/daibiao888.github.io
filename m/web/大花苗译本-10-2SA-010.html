𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼦𖽔𖾐 𖼐𖽺𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 19：1 - 19）
10
𖽃𖽔𖾑 𖼡𖽜𖾐 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼦𖽔𖾐 𖼐𖽺𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 2 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼦𖽔𖾐 𖼐𖽺𖾏 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼐𖽔 𖼳𖽢 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼦𖽔𖾐 𖼐𖽺𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼅𖽻𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼐𖽱𖾏. 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖽐𖼚𖽙. 3 𖼈𖽡𖽝𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼦𖽔𖾐 𖼐𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼟𖽡, 𖼟𖽡 𖽐𖼊𖽱𖾏 𖼋𖽡𖽝𖾏 𖼐𖽱𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼐𖽔𖾑 𖼖𖽔𖾏? 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼋𖽔𖾏, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼢𖽻 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼖𖽺𖾑?” 4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽔𖾐 𖼐𖽺𖾏 𖼄𖽔𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖼮𖽡𖽝𖾏, 𖼮𖽑𖽪𖾑 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼚𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼮𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼽𖽘𖾑 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼺𖽘𖾏, 𖼟𖽙𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼽𖽘𖾑 𖼀𖽡𖾐 𖼊𖽡𖾐, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏. 5 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼸𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑, 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼚𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼊𖽡𖾑 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼎𖽺𖾏 𖼗𖽙𖾏.”
𖽐𖼊𖽻𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽑𖽡𖽝𖾐 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐
6 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼳𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽳 · 𖼖𖽡𖾏 𖼦𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖽐𖼎𖽙 𖼺𖽙𖾐 𖼀𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼏𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽠𖾏 𖽃𖽔𖾐 𖼉𖽻𖾏， 𖼊𖽑𖽡𖽝𖾐 𖼄𖽔𖾐 𖼮𖽢𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐, 𖼊𖽑𖽙 𖼀𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐. 7 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼽𖽙 𖼽𖽔 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏. 8 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏， 𖼋𖽠𖾏 𖼏𖽙 𖼎𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼮𖽡𖽝𖾏, 𖼺𖽙𖾐 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼦𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼊𖽑𖽙 𖼀𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼄𖽔𖾐 𖼮𖽢𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼋𖽠𖾏 𖼏𖽙 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏.
𖼽𖽔𖾏 𖼖𖽔 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽞 𖽃𖽔𖾐 𖼷𖽑𖽻𖾏 𖼏𖽘 𖼀𖽜𖾑
9 𖼽𖽙 𖼽𖽔 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼋𖽠𖾏 𖼏𖽙 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼏𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼋𖽠𖾏 𖼏𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽑𖽜, 𖼋𖽠𖾏 𖼏𖽙 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐. 10 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼺𖽹𖾑 𖼳𖽙𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼳𖽺𖾑, 𖼋𖽠𖾏 𖼏𖽙 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐. 11 𖼽𖽙 𖼽𖽔 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼳𖽺𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼋𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑, 𖼟𖽡 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑. 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼋𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽡, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼋𖽔𖾏 𖽐𖼀𖽘𖾏 𖼟𖽡. 12 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼚𖽔𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏, 𖼑𖽜 𖼾𖽨𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼀𖽡𖾏 𖼮𖽨𖾑, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏.” 13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽙 𖼽𖽔 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖽐𖼊𖽻𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽙 𖼽𖽔 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼳𖽠𖾑 𖼏𖽘 𖼀𖽜𖾑. 14 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽘 𖼀𖽜𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼳𖽺𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼳𖽠𖾑 𖼏𖽘 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻． 𖼽𖽙 𖼽𖽔 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏, 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏.
