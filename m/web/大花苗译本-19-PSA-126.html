𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽳𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼳𖽡 𖼣𖽔𖾑 𖽃𖽡𖾑,
𖼀𖽡𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾐.
2 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼀𖽡𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏,
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽘！”
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽜 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽘 ,
𖼀𖽡𖾐 𖼎𖽺𖾏 𖼙𖽔 𖼚𖽙𖾏.
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼎𖽙𖾑 𖼗𖽙𖾏 ,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼎𖽺𖾏 𖽐𖼚𖽱𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼀𖽳𖾐 𖼮𖽘𖾑 ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼙𖽔 𖼚𖽙𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏！
6 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼯𖽡𖽙𖾏 𖼮𖽘𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼀𖽳𖾐 𖽃𖽡𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼙𖽔 𖼚𖽙𖾏 𖼯𖽡𖽙𖾏 𖼡𖽔𖾑 𖼷𖽻𖾏 𖼡𖽻𖾐 𖼺𖽷𖾐 𖼗𖽙！
