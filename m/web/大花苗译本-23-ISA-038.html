（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 20：1 - 11；𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 32：24 - 26）
38
𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔 , 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏,𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼋𖽔 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼯𖽡𖽝𖾏 𖼋𖽘 .”
2 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽑𖽪𖾏 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼮𖽑𖽤 , 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 3 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼑𖽪𖾏, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏.” 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽡𖽝𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏.
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: 5 “𖼟𖽡 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼋𖽘𖾏, 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼞𖽔𖾏 𖼅𖽔 𖼋𖽘𖾏．𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼟𖽻𖾏 𖼀𖽜𖾑 𖼳𖽡𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏. 6 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 .
7 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼺𖽪𖾐 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 . 8 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽑𖽹𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏，𖼟𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼟𖽻𖾏 𖼏𖽔𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼊𖽑𖽹𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼡𖽜𖾐 𖼙𖽠𖾏 𖼟𖽻𖾏 𖼏𖽔𖾏.
𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼅𖽳𖾏
9 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼐𖽑𖽡𖾏:
10 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏,
𖼞𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔 𖼛𖽻𖾏,
𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼟𖽱𖾏 𖼋𖽘𖾏.
11 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏.
12 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼊𖽻𖾏 𖼮𖽑𖽹𖾏 𖼙𖽠𖾏, 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏,
𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼾𖽨 𖼾𖽘𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼞𖽪𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼮𖽡𖽝𖾏,
𖼺𖽱𖾏 𖼊𖽪𖾑 𖼊𖽑𖽡𖾏 𖽐𖼊𖽻𖾐 𖼑𖽡𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐．
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼊𖽑𖽡𖾏 𖽐𖼊𖽻𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼷𖽑𖽳𖾏 𖼊𖽪𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏,
𖼎𖽑𖽻𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼸𖽙 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
13 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼾𖽨𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏,
𖼐𖽱𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏,
𖼎𖽑𖽻𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼸𖽙 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
 
14 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼐𖽻 𖼙𖽡𖽝𖾏 𖼞𖽡𖾏 𖼘𖽪𖾏，
𖼺𖽱𖾏 𖼚𖽘𖾑 𖼑𖽻𖾏 𖽐𖼀𖽳 𖼢𖽔𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖼢𖽔𖾏,
𖼞𖽪𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖽃𖽔𖾑 𖼅𖽔 𖼎𖽺𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏，
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾏.
15 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼏𖽵 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼎𖽺𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏,
𖼑𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽡𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 .
 
16 𖼮𖽵𖾑 𖼘𖽪！ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏, 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼞𖽳𖾑．
𖼞𖽪𖾑 𖼖𖽡𖽾 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼞𖽳𖾑.
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏.
17 𖼑𖽔𖾏 𖼘𖽪！𖼞𖽪𖾑 𖼊𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪,
𖼟𖽡 𖼑𖽜 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼎𖽺𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼺𖽘𖾏.
18 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡 ,
𖼛𖽙𖾏 𖼋𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 .
19 𖼮𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼸𖽠𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡,
𖼺𖽱𖾏 𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡 𖽃𖽡𖾐 𖼼𖽘𖾐．
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼉𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 .
 
20 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼯𖽪𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 ,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏
𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 ,
𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽷𖾑 𖼺𖽙𖾑 𖽐𖼎𖽷𖾑 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖽐𖼞𖽻𖾏.”
 
21 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼷𖽷𖾑，𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼑𖽡𖾏 𖼳𖽡𖽝𖾐 𖼷𖽷𖾑, 𖼘𖽙𖾏 𖼎𖽻𖾏 𖼷𖽑𖽘𖾐 𖼀𖽡𖾏 𖼮𖽡𖽙𖾏, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏.” 22 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼑𖽪 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?”
