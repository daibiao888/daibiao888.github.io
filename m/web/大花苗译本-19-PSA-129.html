𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏:
“𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽡𖾑 𖼗𖽙 , 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼯𖽪 𖼯𖽪𖾏 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑，
2 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽡𖾑 𖼗𖽙, 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼯𖽪 𖼯𖽪𖾏 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑 ,
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼳𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏.
3 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾑 𖼖𖽡𖾑 𖼉𖽻𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼊𖽡𖾑 𖼢𖽻𖾏,
𖼸𖽙𖾏 𖼟𖽱𖾏 𖼖𖽡𖾏 𖼉𖽻𖾏 𖽐𖼊𖽡𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 .
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼐𖽱𖾏 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖼺𖽘𖾏.”
5 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼳𖽡 𖼣𖽔𖾑 ,
𖼖𖽔𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏！
6 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 ,
𖼦𖽡𖾐 𖼑𖽡𖽝 𖼘𖽑𖽙𖾐 𖼊𖽡𖽝𖾑 𖼺𖽷𖾐 𖼎𖽺𖾏 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏，
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 𖽃𖽡𖾐 𖼋𖽡 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽻𖾏 𖼡𖽑𖽹𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 𖼁𖽔 .
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼅𖽡！
𖼀𖽡𖾐 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏, 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡！”
