6 𖼅𖽡 𖼎𖽺𖾏 𖼑𖽪 𖼑𖽔𖾏, 𖽃𖽔𖾑 𖼾𖽠 𖼸𖽠𖾏 𖼾𖽨 𖼊𖽪𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏?

𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 ,

𖼽𖽔𖾐 𖼞𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼟𖽡𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 ,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽹𖾏.
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐, 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼟𖽡 ,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼟𖽡 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖽃𖽡𖾑 𖼖𖽻𖾐 𖼟𖽙𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖼺𖽘𖾏.
