𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
2 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 , 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼾𖽡 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 ,
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑.”
3 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼟𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽪𖾐 𖼮𖽢𖾐 𖼐𖽻 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏
𖽐𖼎𖽙 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽦𖾑.
4 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼚𖽻𖾐 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼟𖽡 ,
𖼟𖽡 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、𖼟𖽪𖾏 𖼳𖽻𖾐.
5 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽪𖾑 𖼈𖽙𖾏 𖼟𖽪𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼽𖽘𖾏 𖼋𖽔𖾏，
6 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼦𖽡𖾏 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼋𖽘 𖼐𖽑𖽪𖾐 𖼋𖽔𖾏 𖼖𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐.
 
7 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼢𖽻 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽡𖾏 𖼊𖽘𖾑,
𖼡𖽔𖾑 𖼉𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼢𖽻 𖼎𖽻𖾏 𖼟𖽡 𖼺𖽘𖾑 𖼺𖽷𖾏,
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼟𖽡 .
8 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏 ,
𖼁𖽙𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏.
 
9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐，
10 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼗𖽙𖾏 𖼸𖽙 𖼟𖽡 ,
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼦𖽡𖾐 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 .
 
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐,
𖼐𖽡𖽙𖾐 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼅𖽻𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 .
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼮𖽑𖽳𖾑 𖼟𖽡 𖼮𖽡𖽝𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼊𖽱𖾑𖽑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐.
13 𖼟𖽡 𖼽𖽔𖾑 𖼯𖽪𖾏 𖼎𖽻𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼐𖽘𖾐 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼯𖽪𖾏 𖼖𖽻𖾐 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
 
14 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐.
15 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽐𖼚𖽙 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .
16 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 ,
𖼄𖽔𖾐 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏.”
