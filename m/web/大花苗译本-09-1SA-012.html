𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽐𖼊𖽜 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼯𖽪𖾏 𖼋𖽘𖾏, 𖼑𖽜 𖼅𖽡 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽘𖾏. 2 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼅𖽡. 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼞𖽪𖾑 𖼗𖽻𖾏 𖼟𖽡𖾏 𖼚𖽠𖾐 𖼇𖽻𖾏 𖼋𖽠𖾏 𖼋𖽘𖾏, 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑． 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏, 𖼖𖽔𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼅𖽡. 3 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏. 𖼞𖽪𖾑 𖼙𖽔 𖼦𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼑𖽨𖾏? 𖼦𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑? 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏? 𖼯𖽡𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏? 𖼎𖽑𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼖𖽺𖾑? 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽻𖾏.” 4 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾑 𖼙𖽔 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼀𖽡𖾐，𖼦𖽡𖾐 𖼙𖽔 𖼯𖽡𖽙𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼎𖽑𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼊𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏.” 5 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏.”
