𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼊𖽻𖾏 𖼢𖽻 𖼞𖽡𖾑 𖼀𖽔𖾑
𖼽𖽔𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽷𖾑 𖼺𖽙𖾑.
76
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐,
𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 , 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
2 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼅𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 ,
𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐.
3 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼊𖽪𖾑 𖼐𖽑𖽜𖾑 𖼑𖽡𖾏 𖼺𖽪𖾐 𖼈𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏,
𖽐𖼎𖽙 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐, 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼎𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
4 𖼟𖽡 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼚𖽔𖾐,
𖼇𖽔𖾑 𖼊𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼡𖽹𖾏 𖼻𖽻𖾏 𖼸𖽔 .
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼘𖽑𖽙𖾐 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼀𖽱𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼋𖽻𖾏 𖼋𖽡𖾏.
6 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖽂𖽪𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾏 𖼡𖽑𖽔𖾐,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼀𖽑𖽪𖾐 𖼼𖽙𖾐 𖽃𖽡𖾑、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼞𖽑𖽔𖾏 𖼀𖽱𖾏 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏 𖼙𖽠𖾏.
 
7 𖼟𖽡，𖼮𖽔 𖼅𖽔𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏！
𖼟𖽡 𖽂𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼖𖽺𖾑?
8-9 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐,
𖼽𖽔𖾑 𖼮𖽨𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 .
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼊𖽡𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼎𖽺𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 . （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
10 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼟𖽙𖾏 𖼟𖽡 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔,
𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼺𖽷𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏.
11 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 , 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽡𖾑 .
12 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼑𖽡𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏,
𖼐𖽱𖾏 𖼢𖽳 𖼚𖽔𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏.
