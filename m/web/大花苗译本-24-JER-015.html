𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “ 𖼅𖽔𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼘𖽑𖽪𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼦𖽡𖾑！ 2 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪 𖼟𖽡 𖼐𖽑𖽡𖾏, 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ：
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼋𖽔 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏, 𖼽𖽔𖾑 𖽐𖼮𖽷𖾐 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏.
3 𖼞𖽪𖾑 𖼏𖽵 𖼚𖽻𖾐 𖼼𖽘𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼟𖽪𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏、𖼚𖽡𖾑 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔、𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏, 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏.”𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 4 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽔𖾑 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼟𖽪𖾏 𖼾𖽱 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑.
5 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪! 𖼡𖽔 𖼋𖽱𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽡 𖼖𖽺𖾑?
𖼡𖽔 𖼋𖽱𖾏 𖼑𖽜 𖼟𖽡 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼖𖽺𖾑?
𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖼑𖽪 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽺𖾑?
6 𖼟𖽡 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏,
𖼎𖽙𖾑 𖼡𖽜𖾐 𖼑𖽡𖾏 𖼙𖽠𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼟𖽡 , 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼟𖽡 ,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽡 𖼋𖽘𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
7 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 ,
𖼎𖽑𖽻𖾏 𖼈𖽘𖾐 𖼡𖽻𖾏 𖼷𖽻𖾑 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼺𖽘𖾏,
𖼞𖽪𖾑 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐,
𖼞𖽪𖾑 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼋𖽘 𖼐𖽑𖽪𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻,
𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐,
𖼟𖽙𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑.
9 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼾𖽨 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐, 𖼯𖽻 𖼛𖽘𖾏 𖼯𖽻 𖼼𖽙,
𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼯𖽡𖽙𖾏 𖼺𖽷𖾐, 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏,
𖼐𖽱𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼚𖽑𖽪𖾏.
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼺𖽷𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
