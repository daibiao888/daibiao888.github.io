𖼯𖽡𖽹𖾏 𖼐𖽡𖾑 ， 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼉𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽪𖾐 𖼦𖽡𖾏 𖼐𖽹𖾏．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽪𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏，𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏． 2 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼮𖽡𖽝𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐． 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 ，𖼽𖽔𖾑 𖼢𖽳 𖼙𖽪 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽙𖾑，𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖽐𖼷𖽑𖽷𖾏 𖼎𖽻𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏，𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖼏𖽪𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏． ”
（𖼊𖽑𖽹𖾏 10：28-31）
4 “𖼞𖽪𖾑 𖼽𖽺 𖼗𖽔， 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 ，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 5 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽳𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏：𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼊𖽘𖾏，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼗𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽻𖾏 𖽐𖼷𖽷𖾏 𖼞𖽡 𖼑𖽡𖽻 𖽃𖽡𖾑． 𖼾𖽙 𖼯𖽡𖽙𖾏 ， 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 6 𖼀𖽜𖾐 𖼋𖽪 𖼑𖽻 𖼀𖽡𖾏 𖼮𖽑𖽡𖽙𖾑， 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼋𖽺 𖽃𖽔𖾐 𖼇𖽺𖾏 𖼑𖽡𖽝𖾏 𖼘𖽪！𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼦𖽡𖾐 𖼸𖽠𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼋𖽪． 7 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼊𖽪𖾏 𖼙𖽔 𖼾𖽱 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽘𖾏．𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏， 𖼅𖽡 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼑𖽻 𖼀𖽡𖾏 𖼮𖽑𖽡𖽙𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐．”

8 “𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼯𖽜𖾏 𖼞𖽪𖾑，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽜𖾏 𖼐𖽱𖾏． 9 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼐𖽱𖾏． 10 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐，𖼾𖽔𖾏 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼋𖽻，𖼮𖽔 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼳𖽟𖾏 𖼖𖽡𖽾， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼾𖽔𖾏． 11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱，𖼚𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏，𖼚𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼅𖽡 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼊𖽳𖾏 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼽𖽔𖾑 𖼊𖽡𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 ， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏． 12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽙 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼳𖽟𖾏 𖼖𖽡𖽾 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏．”
