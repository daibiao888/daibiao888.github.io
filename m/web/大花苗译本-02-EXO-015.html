15
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾏 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 :
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 ,
𖼞𖽑𖽡𖽠𖾏 𖼑𖽜𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 .
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 , 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 ,
𖼎𖽺𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 ,
𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 ,
𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼐𖽱𖾏 .
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 ,
𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
 
4 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐、𖼏𖽙 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽑𖽡𖽠𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏 ,
𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽹𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼖𖽔𖾏 𖼦𖽡 𖼋𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼖𖽡𖽝𖾐 𖼙𖽠𖾏.
5 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼷𖽑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼦𖽡 𖼋𖽙 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐 .
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 ,
𖼢𖽳 𖼋𖽠𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼇𖽜𖾏 𖼇𖽜𖾏 𖽐𖼮𖽠𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼋𖽪 𖼺𖽘𖾏 .
7 𖼟𖽡 𖼢𖽳 𖼋𖽠𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼚𖽔𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 , 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼸𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 ,
𖼟𖽡 𖼢𖽳 𖼋𖽠𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 , 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼷𖽑𖽷𖾑 𖽃𖽔𖾑 𖽐𖼊𖽻𖾏 𖽐𖼡𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.
8 𖼟𖽡 𖼮𖽙𖾏 𖼀𖽘𖾏 𖼟𖽪𖾏 𖼗𖽙𖾏 𖽃𖽔𖾑 𖽐𖼀𖽱 , 𖽃𖽻𖾐 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼚𖽻𖾏 ,
𖽃𖽻𖾐 𖽐𖼚𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 ,
𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐 𖼞𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 .
9 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏：𖼞𖽪𖾑 𖼽𖽔𖾑 𖼖𖽠𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼯𖽢 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼇𖽺𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼷𖽹𖾏 𖼋𖽡𖾏 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏.
10 𖼟𖽡 𖼟𖽙𖾏 𖼟𖽡 𖼮𖽢𖾏 𖽂𖽪𖾏 𖼷𖽑𖽔𖾐 , 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽪𖾐 𖼺𖽘𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼷𖽑𖽔𖾏 𖼦𖽡 𖼋𖽙 𖼚𖽔𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 .
 
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 , 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼺𖽱𖾏 𖼟𖽡 ? 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼺𖽱𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼚𖽙 𖽃𖽺𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 !
𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽐𖼮𖽑𖽹𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 , 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 .
12 𖼟𖽡 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 ,
𖼊𖽡𖾐 𖼎𖽺𖾏 𖽐𖼡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 .
 
13 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼨𖽑𖽳𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼅𖽔𖾏 𖼎𖽙𖾑 𖽃𖽡𖾑 ,
𖼟𖽡 𖼎𖽑𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 .
14 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 ,
𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾑 𖼮𖽡𖽝𖾏𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 .
15 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑 𖽐𖼮𖽑𖽹𖾏 𖼏𖽘𖾏 ,
𖼄𖽙𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼊𖽻𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾑 𖼮𖽡𖽝𖾏 ,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼐𖽔 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼞𖽑𖽔𖾏 𖼾𖽘𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏 .
16 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼑𖽜 𖼟𖽡 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼼𖽙 ,
𖼋𖽙𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏,
𖼋𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏.
17 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽳𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 , 𖼯𖽡𖽙 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 .
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽜 𖼾𖽨𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 ,
𖼮𖽵𖾑 𖼘𖽪 ! 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 .
18 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏 .”
𖼄𖽡𖾐 𖼖𖽡𖾏 𖼣𖽔𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫

“𖼅𖽡 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 ,
𖼄𖽔𖾐 𖼑𖽜𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 .”
𖼄𖽔𖾐 𖼖𖽔𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽦𖾐
22 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼨𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼖𖽡𖽝𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽪𖾑 𖽃𖽳𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 , 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 , 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽃𖽻𖾐 . 23 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽔𖾐 𖼖𖽔𖾑 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼦𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽻𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽻𖾐 𖽃𖽦𖾐 𖼐𖽔𖾑 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖼖𖽔𖾑 (“𖼄𖽔𖾐 𖼖𖽔𖾑”𖼟𖽪𖾏 “𖽃𖽦𖾐”) . 24 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏𖼄𖽙𖾑 𖼳𖽡𖾑 , 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖼖𖽔𖾑 𖼦𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑 ?” 25 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 , 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽃𖽻𖾐 , 𖽃𖽻𖾐 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼋𖽘𖾏 .
