2
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 2 “𖼟𖽡 𖼅𖽻𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼐𖽑𖽡𖾏:
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼑𖽡𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏,
𖼐𖽤𖾐 𖼉𖽻𖾏 𖼮𖽑𖽦𖾐 𖼑𖽡𖾏 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏,
𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼮𖽡𖽠𖾏 𖼗𖽙𖾏 𖼯𖽡𖽙 𖼡𖽻𖾐 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼞𖽪𖾑,
𖼞𖽪𖾑 𖽐𖼮𖽡𖽙𖾏 𖽐𖼊𖽪𖾐 𖼘𖽙𖾏 .
3 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼺𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖽃𖽔𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼳𖽦𖾑,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼐𖽱𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏,
𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
4 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏. 5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:
“𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼐𖽔𖾑,
𖼎𖽺𖾏 𖼡𖽔𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐, 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼾𖽨𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼙𖽠𖾏 𖼖𖽺𖾑?
6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼑𖽪:
𖼷𖽷𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏,
𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏、
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼯𖽷𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖽐𖼎𖽙 𖼡𖽑𖽙𖾑 𖼊𖽙𖾐、
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖽐𖼎𖽙 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏、
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖽻𖾏 𖽐𖼎𖽙 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼘𖽪?
7 𖼞𖽪𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼛𖽙 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼟𖽙𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼷𖽷𖾑 𖽐𖼎𖽙 𖼻𖽻𖾏 𖼡𖽜𖾑,
𖼈𖽡𖽝𖾏 𖼅𖽡 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼞𖽪𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 𖼺𖽘𖾏.
8 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼦𖽡𖾐 𖼑𖽪, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱 𖼖𖽺𖾑?
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼞𖽪𖾑, 𖼉𖽻 𖼷𖽻𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑,
𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼀𖽔𖾑 𖼖𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏,
𖼸𖽙𖾏 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.”
 
9 𖼞𖽪𖾑 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
10 𖼅𖽡 𖼋𖽡𖾏 𖼚𖽔𖾏 𖼺𖽹𖾏 𖼀𖽑𖽪𖾐 𖼖𖽪𖾏 𖼺𖽷𖾑 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼅𖽻𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏,
𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼊𖽔, 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏,
𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼙𖽔 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏.
11 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼞𖽪𖽙 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼘𖽪𖾑𖽑 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼺𖽘𖾏 𖼖𖽺𖾑?
𖼋𖽡𖾏 𖽃𖽡𖾑 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼋𖽡𖽝,
𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖼘𖽪𖾑𖽑 𖼺𖽘𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
12 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼘𖽪! 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼏𖽘𖾏 𖼀𖽡𖾏 𖼮𖽨𖾑,
𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼊𖽔𖾑 𖼋𖽡𖽝, 𖼟𖽡𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑!
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
13 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼟𖽪𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼐𖽡𖾑,
𖼑𖽜 𖼾𖽨𖾏 𖼸𖽻 𖼋𖽠𖾏 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽻𖾐,
𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐.”
𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼾𖽨𖾏 𖼊𖽻𖾏
14 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼡𖽑𖽜𖾑 𖼘𖽪𖾏?
𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏 𖼘𖽑𖽘𖾑 𖼾𖽨𖾏 𖼘𖽪𖾏?
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏?
15 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏，𖼮𖽘𖾏 𖼺𖽘𖾐 𖼢𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼖𖽪𖾑 𖼙𖽻 𖼖𖽔𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐.
16 𖼐𖽙 𖼇𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽔 𖼀𖽡𖾐 𖼐𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼖𖽔𖾑 𖽐𖼊𖽻𖾏 𖽐𖼮𖽻 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏.

21 𖼞𖽪𖾑 𖼯𖽡𖽙 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.

34 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽘𖾐 𖼮𖽑𖽙𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑,

𖼅𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼈𖽡𖽝𖾏,

𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼳𖽹𖾐 𖼑𖽪 𖼟𖽡,

36 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏, 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼖𖽺𖾑?
