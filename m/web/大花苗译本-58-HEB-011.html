𖽐𖼮𖽜 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖽃𖽔𖾑 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝，𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼁𖽙𖾏 𖼺𖽷𖾐 𖼑𖽡𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝． 2 𖽃𖽔 𖼊𖽑𖽻𖾏 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼅𖽔𖾏 𖽐𖼮𖽜 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏． 3 𖼀𖽡𖾐 𖼅𖽔𖾏 𖽐𖼮𖽜 ，𖼎𖽺𖾏 𖼀𖽻𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏．𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼷𖽷𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏．
4 𖼽𖽔𖾏 𖼀𖽳 𖼅𖽔𖾏 𖽐𖼮𖽜𖾏， 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽹𖾑 𖼽𖽡𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏， 𖼐𖽱𖾏 𖼅𖽔𖾏 𖽐𖼮𖽜𖾏 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼐𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽑𖽘𖾑． 𖼽𖽔𖾏 𖼀𖽳 𖼋𖽔 𖼀𖽙𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽐𖼮𖽜 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼯𖽡𖽙 𖼺𖽷𖾐． 5 𖼽𖽡𖾑 𖼐𖽙 𖼅𖽔𖾏 𖽐𖼮𖽜𖾏， 𖼊𖽻𖾏 𖼷𖽹𖾏 𖼙𖽠𖾏， 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼋𖽔，𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼷𖽹𖾏 𖼙𖽠𖾏．𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼊𖽻𖾏 𖼷𖽹𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏， 𖼐𖽱𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼐𖽑𖽡𖾏． 6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼮𖽜， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽱𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼎𖽺𖾏 𖽐𖼮𖽜 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏． 7 𖼐𖽙 𖼽𖽔𖾏 𖼅𖽔𖾏 𖽐𖼮𖽜𖾏，𖼎𖽺𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼁𖽙𖾏 𖼺𖽷𖾐，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼷𖽱𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽙𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏．𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼊𖽡𖾏 𖼷𖽑𖽷𖾏 𖽃𖽡𖾑 𖼾𖽔𖾏，𖼾𖽨𖾏 𖼎𖽺𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼮𖽜 𖼗𖽙𖾏．
