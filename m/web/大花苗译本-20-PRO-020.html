20
𖼮𖽡𖽠𖾑 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼗𖽔 , 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽐𖼊𖽑𖽡𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼑𖽜 𖼮𖽡𖽠𖾑 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏, 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
2 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼦𖽺𖾏 𖼾𖽨𖾏 𖼳𖽦𖾑.
3 𖽐𖼷𖽜𖾏 𖼺𖽘𖾏 𖼑𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏, 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 ,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖽐𖼡𖽔 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
4 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽜 𖼑𖽜 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼐𖽙𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖼯𖽡𖽙 𖼡𖽻𖾐,
𖼸𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼡𖽻𖾐, 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼋𖽪 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏.
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼊𖽙𖾐,
𖼮𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼸𖽠𖾏 𖼮𖽤𖾐 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖽐𖼮𖽷𖾐 𖼖𖽺𖾑？
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏！
8 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼳𖽹𖾐 𖼑𖽔𖾏（“𖼳𖽹𖾐 𖼑𖽔𖾏 ” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖼷𖽻𖾑 𖼮𖽑𖽻𖾏 ”） 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
9 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 :“ 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽘 ,
𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏 𖼖𖽺𖾑?”
10 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼦𖽡 𖼋𖽳, 𖼳𖽜𖾐 𖼊𖽱𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼦𖽡 𖼋𖽳,
𖽃𖽔𖾐 𖼮𖽘𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐.
11 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼙𖽻 .
12 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼑𖽻, 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼑𖽔𖾏 ,
𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾏 𖼯𖽡𖽙𖾏.
13 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽠𖾑 𖼀𖽱𖾏 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏,
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔, 𖼟𖽡 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏 .
14 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼻𖽻𖾏！𖼦𖽡𖾐 𖼻𖽻𖾏！”
𖼈𖽡𖽝𖾏 𖼅𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑.
15 𖼅𖽔𖾏 𖼞𖽪𖾐 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼈𖽡𖽝𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳, 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼋𖽡𖾏 𖼐𖽡𖾑 .
16 𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼑𖽡𖽻𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾑, 𖼯𖽡𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼮𖽑𖽙𖾏,
𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼡𖽑𖽠𖾑, 𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏.
17 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽻𖾏 𖼗𖽙𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼺𖽱𖾏 𖽃𖽺𖾐 𖼡𖽘𖾐 𖼄𖽪𖾑,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙, 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽔 𖼗𖽔𖾏 .
18 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 ；
𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
19 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼐𖽑𖽡𖾏 𖼺𖽹𖾑 𖼗𖽪𖾏, 𖼮𖽑𖽻𖾏 𖼙𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽻𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔 .
20 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏, 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 ,
𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼚𖽪𖾐 𖼖𖽡𖾐 𖽐𖼮𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
21 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼁𖽱 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑,
𖼸𖽙 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
22 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽪𖾏 𖼁𖽻𖾏 𖽐𖼚𖽪𖾏.”
𖼽𖽔𖾑 𖼋𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼟𖽡 .
23 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼦𖽡 𖼋𖽳 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐；
𖼸𖽱𖾏 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼻𖽻𖾏.
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽱𖾏 𖼷𖽑𖽔𖾏 𖼏𖽔𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽡𖾏 𖼮𖽡𖽝𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼖𖽺𖾑?

𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖽐𖼊𖽱𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 , 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼮𖽙𖾏 𖼾𖽨𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏.
26 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽑𖽻𖾏 𖼈𖽘𖾐 𖼷𖽻𖾑 𖼮𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼘𖽙𖾐 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏.
27 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐,
𖼳𖽹𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽐𖼀𖽱𖾏.
28 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼐𖽱𖾏 𖼞𖽪𖽙 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 .
29 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 ,
𖼚𖽻𖾐 𖼇𖽻𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼷𖽑𖽡𖽝𖾏 𖼗𖽡𖽝𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .
30 𖽐𖼚𖽠𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖽐𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖼸𖽠𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽐𖼀𖽱𖾏.
