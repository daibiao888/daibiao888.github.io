18
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼄𖽔𖾏 𖼖𖽡𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼳𖽙𖾑, 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽃𖽔 𖼛𖽻𖾏. 2 𖼮𖽘𖾏 𖽃𖽔𖾐 𖼅𖽔 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏. 𖼐𖽱𖾏 𖽂𖽪𖾏 𖼁𖽙𖾏, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼏𖽘 𖼅𖽻𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑， 3 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪! 𖼅𖽡 𖼋𖽔𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼟𖽡 𖼡𖽑𖽜𖾑. 4 𖼯𖽪𖾏 𖼞𖽪𖾑 𖼘𖽡𖾑𖽑 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖽃𖽻𖾐 𖼗𖽙 𖼎𖽻𖾏 𖼅𖽡 𖽐𖼷𖽔𖾑 𖼊𖽠𖾏, 𖼐𖽡𖽙𖾐 𖽐𖼊𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼳𖽙𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏. 5 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼳𖽡𖽝𖾐 𖼗𖽙, 𖼅𖽡 𖼑𖽻𖾏 𖼺𖽘𖾏 𖼟𖽪𖾏 𖼊𖽑𖽡𖽝𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏. 𖼅𖽡 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼚𖽔𖾏 𖼅𖽡 𖼡𖽑𖽜𖾑 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼋𖽘𖾏, 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏.” 6 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼁𖽙𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼞𖽑𖽡𖽠𖾏 𖼷𖽷𖾐 𖼳𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼄𖽑𖽻𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼳𖽡𖽝𖾐.” 7 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼎𖽺𖾏 𖼏𖽘 𖼚𖽔𖾏 𖽐𖼀𖽙𖾏 𖼑𖽨𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼮𖽤𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼗𖽙𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼡𖽑𖽜𖾑, 𖼡𖽑𖽜𖾑 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏. 8 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼄𖽡𖾏 𖼳𖽡𖽝𖾐 𖽐𖼎𖽙 𖼄𖽡𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏, 𖼊𖽜𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼾𖽨𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼋𖽘.
𖽐𖼊𖽔𖾏 𖼏𖽵 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼊𖽪𖾐
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼐𖽤𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐.” 10 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼸𖽙 𖼦𖽡𖾏 𖼑𖽡𖽻𖾏 𖼐𖽔𖾑 𖼯𖽡𖽹𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼐𖽤𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐.” 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼐𖽡𖽙𖾐 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼖𖽔𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑. 11 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏， 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼡𖽔𖾏 𖼖𖽑𖽡𖾏 𖼖𖽔𖾏 𖼊𖽪𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏. 12 𖼺𖽔𖾐 𖼖𖽔𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼚𖽙𖾏, 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼯𖽻 𖼛𖽘𖾏 𖼯𖽻 𖼼𖽙 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏, 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼖𖽔𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼻𖽻𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼘𖽪?” 13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼺𖽔𖾐 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏 𖼺𖽷𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏? 14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽔𖾑 𖼘𖽪? 𖼸𖽙 𖼦𖽡𖾏 𖼑𖽡𖽻𖾏 𖼐𖽔𖾑 𖼯𖽡𖽹𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼺𖽔𖾐 𖼖𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐.” 15 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼯𖽜𖾏, 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼚𖽙𖾏.” 𖼷𖽷𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼾𖽙, 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼯𖽡𖽙𖾏.”
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐、𖽃𖽳 𖼄𖽙 𖼖𖽔𖾑
