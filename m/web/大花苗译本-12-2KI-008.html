7 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳, 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽡𖽝 · 𖼦𖽔𖾐 𖼊𖽔 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼷𖽷𖾏 𖼖𖽘𖾑． 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼸𖽙 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏.” 8 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼯𖽡𖽙𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐, 𖽐𖼮𖽢𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼄𖽙𖾐 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼡𖽑𖽔𖾑 𖼸𖽠𖾏 𖼻𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼻𖽻𖾏?” 9 𖽃𖽔𖾏 𖼐𖽡𖾑, 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 𖼎𖽑𖽻𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼋𖽪𖾏 𖼘𖽙 𖼊𖽑𖽙𖾐, 𖼮𖽹𖾑 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽃𖽔𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏, 𖼅𖽻𖾏 𖼁𖽙𖾏𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑. 𖼸𖽙 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽡𖽝 · 𖼦𖽔𖾐 𖼊𖽔 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡, 𖼐𖽱𖾏 𖼑𖽪 𖼐𖽑𖽡𖾏: ‘𖼞𖽪𖾑 𖼊𖽪𖾑 𖼄𖽙𖾐 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼡𖽑𖽔𖾑 𖼸𖽠𖾏 𖼻𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼻𖽻𖾏?’” 10 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖽐𖼎𖽙 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏, 𖼊𖽪𖾑 𖼄𖽙𖾐 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼻𖽻𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔.” 11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼐𖽡𖽝𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏. 12 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼐𖽔𖾑 𖼐𖽡𖽝𖾑？” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖽃𖽦𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼺𖽘𖾏, 𖽐𖼮𖽠𖾏 𖼋𖽔 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏, 𖼀𖽑𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑．” 13 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼡𖽑𖽜𖾑 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼚𖽡𖾑 𖼊𖽔𖾐, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖾑 𖼖𖽺𖾑?” 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏.” 14 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 𖼇𖽳𖾑 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑, 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑． 𖼮𖽵𖾑 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏?” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼻𖽻𖾏.” 15 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐, 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 𖼞𖽑𖽡𖽠𖾏 𖼀𖽑𖽙𖾑 𖼡𖽔𖾏 𖼘𖽑𖽱𖾏 𖼚𖽻𖾏 𖼎𖽻𖾏 𖽃𖽻𖾐, 𖼀𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏． 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐.

16 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼖𖽔 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖽐𖼮𖽠𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼖𖽔 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 17 𖼽𖽙 𖼖𖽔 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏. 18 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼾𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽙𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏. 19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽜 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽔𖾑, 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼽𖽨 𖼊𖽔𖾏 , 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼏𖽵, 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼺𖽘𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏. 20 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼖𖽔 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑, 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼽𖽨 𖼊𖽔𖾏, 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼾𖽨𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 21 𖼽𖽙 𖼖𖽔 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼺𖽔𖾐 𖼽𖽡, 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖽐𖼊𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼐𖽱𖾏. 𖼽𖽨 𖼊𖽔𖾏 𖼏𖽙 𖼎𖽺𖾏 𖼏𖽘 𖼀𖽜𖾑, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 22 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼽𖽨 𖼊𖽔𖾏, 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏. 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼖𖽡 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖽐𖼷𖽜𖾑. 23 𖼽𖽙 𖼖𖽔 𖼑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏, 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏《 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏》 𖼮𖽡𖽝𖾏. 24 𖼽𖽙 𖼖𖽔 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏, 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏. 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼳𖽡𖽝𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼳𖽡𖽝𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 22：1 - 6）
25 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼖𖽔𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼖𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼳𖽡𖽝𖾏 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 26 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼳𖽡𖽝𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏. 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐, 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖽐𖼷𖽑𖽹𖾏. 27 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼳𖽡𖽝𖾏 𖼾𖽻 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼾𖽡𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾐 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼾𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼈𖽻𖾑. 28 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼖𖽔 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏𖼖𖽔𖾑 𖼄𖽙 𖼙𖽠𖾏, 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼽𖽙 𖼖𖽔 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏， 29 𖼽𖽙 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽷𖾐 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝, 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖼄𖽙 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏，𖼏𖽻 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏 𖽃𖽡𖾑. 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼖𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼳𖽡𖽝𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼖𖽔 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏 𖼐𖽔𖾑, 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏.
