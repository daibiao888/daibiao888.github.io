6 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑, 𖼦𖽡𖾏 𖼸𖽠𖾏 𖼮𖽑𖽘𖾑 𖽐𖼊𖽪𖾏, 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼚𖽪𖾐 𖼖𖽡𖾐 𖽐𖼮𖽡𖽝𖾐. 7 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽻𖾐 𖼯𖽡𖽙, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 .
8 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼚𖽱𖾏 𖼗𖽙𖾏, 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖽐𖼚𖽱𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑, 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖽐𖼚𖽱𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑, 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 .
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
