7 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑 ？𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽑𖽦𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼊𖽻𖾏， 𖼮𖽔 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼺𖽹𖾑 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼋𖽘𖾏．𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼼𖽔𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝． 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 ：
“𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔，
𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 ，
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽙𖾑𖽑 𖼗𖽪𖾏 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑 ．”
9 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏：
“𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽱𖾏 𖼞𖽡𖾏 𖼏𖽻𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏，𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼘𖽑𖽻𖾑，
𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽠𖾏，𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
10 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼇𖽔 𖼙𖽻 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑，
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽻𖾑 𖼚𖽔𖾑 ．”
𖼞𖽑𖽻𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽻𖾏 𖼮𖽨𖾏
