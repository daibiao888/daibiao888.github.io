𖼺𖽻𖾐 𖼘𖽙 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐
（𖼺𖽔𖾐 𖽃𖽡𖾐 31：1 - 13）
10
𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼏𖽘 𖼀𖽜𖾑， 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡 𖼀𖽙𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼏𖽻 𖼊𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 . 2 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽠𖾑 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼅𖽡 , 𖼎𖽺𖾏 𖼊𖽔𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 、 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽔 𖼊𖽔 、 𖼄𖽳 𖼮𖽡𖾑 · 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼺𖽘𖾏 . 3 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽃𖽺𖾐 𖼷𖽹𖾑 , 𖼺𖽻𖾐 𖼘𖽙 𖼊𖽻𖾏 𖼊𖽪𖾐 𖼀𖽙𖾑 𖼐𖽜𖾑𖽑 𖼖𖽠𖾑 𖼯𖽢 , 𖼀𖽙𖾑 𖼏𖽻 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽑𖽤𖾑 . 4 𖼺𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖽐𖼡𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼗𖽙𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼐𖽑𖽙𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼺𖽘𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖼐𖽡𖾑 𖼋𖽔𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑 .” 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 , 𖽃𖽔𖾏 𖼐𖽡𖾑 , 𖼺𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖼾𖽨𖾏 𖽐𖼀𖽪𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏 . 5 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑 𖼁𖽙𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏 , 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼎𖽺𖾏 𖼾𖽨𖾏 𖽐𖼀𖽪𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏 . 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏 , 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼙𖽱 𖽃𖽡𖾐 𖼸𖽻 𖼞𖽑𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏 . 7 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏, 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼏𖽘 𖼀𖽜𖾑 , 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼖𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏 . 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼺𖽘𖾏 𖼏𖽘 𖼀𖽜𖾑 , 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 .
8 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 , 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼚𖽠𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑 𖼮𖽑𖽙𖾏 ,𖼑𖽔𖾏 𖼁𖽙𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼋𖽔 𖼎𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡 𖼀𖽙𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 , 9 𖼎𖽺𖾏 𖼚𖽠𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼏𖽙 𖼸𖽻𖾏 , 𖼎𖽺𖾏 𖼮𖽷𖾐 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 , 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏 𖽐𖼮𖽡𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 . 10 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽱𖾏 𖼏𖽙 𖼸𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼞𖽔𖾏 𖼁𖽠 𖽐𖼚𖽙， 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖽐𖼮𖽦𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 𖼊𖽔𖾏 𖼞𖽪𖽝𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 . 11 𖼮𖽡𖾑 𖼖𖽡𖽝 · 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙， 12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼋𖽻 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼅𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼽𖽔𖾐 𖼀𖽡𖾐 , 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼑𖽡𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 , 𖼎𖽺𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 .
13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼺𖽻𖾐 𖼘𖽙 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 . 𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐 . 14 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼏𖽻 𖼊𖽔𖾏 , 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼞𖽪𖽙 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽠𖾏.
