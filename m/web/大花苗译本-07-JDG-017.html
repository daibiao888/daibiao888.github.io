𖼄𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼁𖽠𖾏
17
𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼄𖽡𖾐 𖼮𖽢𖾑. 2 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖽝𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼙𖽠𖾏, 𖼟𖽡 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏, 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼑𖽔𖾏 𖼘𖽪 ! 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼙𖽠𖾏.” 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐 𖼘𖽪 ! 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡.” 3 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖽝𖾏 𖼐𖽡𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐． 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼐𖽡𖾑 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼸𖽻 𖼞𖽳𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼁𖽠𖾏, 𖼘𖽑𖽡𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼁𖽠𖾏． 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡.” 4 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼄𖽔𖾐 𖼑𖽡𖽝𖾏 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖽝𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼑𖽡𖽝𖾏 𖼸𖽻 𖼞𖽳𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼁𖽠𖾏,𖼘𖽑𖽡𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼁𖽠𖾏, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼖𖽪𖾑 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏. 5 𖼷𖽷𖾑 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼊𖽜𖾐 𖼁𖽠𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖽐𖼎𖽙 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼋𖽢𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼁𖽠𖾏, 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏. 6 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏.
7 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑. 8 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖼙𖽻, 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐. 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏. 9 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏?” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖼋𖽔𖾏, 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐.” 10 𖼄𖽡𖾐 𖼮𖽢𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑， 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏, 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏． 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼄𖽔𖾐 𖼟𖽻𖾏 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖽝𖾏, 𖽃𖽡𖾐 𖼷𖽔𖾏 𖼞𖽡𖾑 𖼛𖽔𖾏 𖼮𖽑𖽙𖾏 𖽐𖼎𖽙 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡.” 𖼷𖽷𖾑 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏. 11 𖼷𖽷𖾑 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼯𖽪𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼑𖽔𖾏 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼊𖽪𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐. 12 𖼄𖽡𖾐 𖼮𖽢𖾑𖼟𖽙𖾏 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼄𖽡𖾐 𖼮𖽢𖾑 𖽐𖼞𖽔𖾏. 13 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: “𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏.”
