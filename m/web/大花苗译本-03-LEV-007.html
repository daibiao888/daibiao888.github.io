7
“𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽻𖾏 𖼾𖽔𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼋𖽡𖾏 𖼀𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼐𖽡𖾑． 2 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼊𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼊𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽻𖾏 𖼾𖽔𖾏 , 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼦𖽡 𖽐𖼊𖽔 𖽐𖼮𖽑𖽘𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽻𖾏 𖼯𖽪𖾏 . 3 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼯𖽦𖾏 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼛𖽙𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 , 𖼟𖽪𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼷𖽘𖾏 𖼟𖽪𖾏 𖼛𖽙 𖼊𖽑𖽡𖽝𖾐 𖼛𖽙𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽐𖼀𖽱𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖽃𖽔𖾏 𖼐𖽑𖽨𖾑 , 4 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼏𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼏𖽻𖾏 𖼛𖽙𖾏 , 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼖𖽪𖾑 𖼚𖽔𖾑 𖽃𖽡𖾑 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼛𖽙𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼛𖽙𖾏 𖼀𖽙𖾏 𖽐𖼎𖽙 𖼏𖽻𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽙𖾏 . 5 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽻𖾏 𖼾𖽔𖾏 . 6 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼾𖽠 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼀𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼐𖽡𖾑 , 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 . 7 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼯𖽡𖾏 , 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽻𖾏 𖼾𖽔𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 , 𖽃𖽔𖾐 𖼼𖽘𖾐 𖼞𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽜𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 . 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼁𖽻𖾏 𖼾𖽔𖾏 𖼗𖽙𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼋𖽡𖾏 𖽐𖼡𖽹𖾏 𖼐𖽡𖾑 . 8 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽜 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽑𖽘𖾑 , 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼐𖽡𖾑 𖼊𖽪𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏 . 9 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼷𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼷𖽙𖾏 , 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽑𖽻𖾏、𖼀𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾑 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 . 10 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼛𖽙𖾏 , 𖼟𖽪𖾏 𖽐𖼡𖽑𖽔𖾐 , 𖼽𖽔𖾑 𖼺𖽙𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 , 𖼇𖽺𖾐 𖼟𖽪𖾏 𖼦𖽡 𖼋𖽳 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜.
