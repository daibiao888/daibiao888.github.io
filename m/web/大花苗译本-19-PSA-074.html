𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼺𖽹𖾑 𖼑𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
𖼽𖽔𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏.
74
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼇𖽳𖾑 𖼀𖽡𖾐 𖼺𖽘𖾏 𖼖𖽺𖾑?
𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽡𖾐 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐 𖼖𖽺𖾑?
2 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼳𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏.
3 𖼮𖽑𖽙 𖼟𖽡 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽃𖽡𖾑,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑.
 
4 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽙𖾏 𖼾𖽨𖾏 𖼀𖽑𖽳𖾑 𖼮𖽡𖽝𖾏 𖽃𖽔𖾏 𖼺𖽻𖾏.
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼊𖽻𖾏 𖽐𖼊𖽙𖾑 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽡𖾏 𖽐𖼊𖽻𖾏.
6 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼡𖽔𖾑 𖼊𖽻𖾏, 𖼊𖽪𖾑 𖼮𖽑𖽪𖽺 𖽐𖼊𖽪𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑,
𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼸𖽙 𖼀𖽡𖾑 𖼊𖽡𖾐.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽔𖾑 𖼟𖽪𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏.”
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼄𖽔𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏.
 
9 𖼀𖽡𖾐 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼾𖽨𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐,
𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼺𖽷𖾐.
10 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼽𖽔𖾑 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?
11 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽐𖼞𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼙𖽠𖾏,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏.
 
12 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏.
13 𖼟𖽡 𖼙𖽔 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼇𖽺𖾐 𖼺𖽘𖾏,
𖼄𖽔𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏.
14 𖼟𖽡 𖽐𖼊𖽻𖾏 𖽐𖼮𖽻 𖼖𖽡𖾏 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾐 (“𖼖𖽡𖾏 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾐 ” 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼗𖽪𖾏 ，𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 “𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐” ) 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 ,
𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏.
15 𖼟𖽡 𖼙𖽔 𖼡𖽑𖽜𖾐 𖽐𖼀𖽑𖽔𖾐 𖼮𖽑𖽡𖽠𖾏 𖼷𖽠𖾑𖽑 𖽃𖽻𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐,
𖼟𖽡 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼚𖽱𖾏 𖼡𖽑𖽔𖾑 𖼙𖽠𖾏.
16 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼺𖽙𖾏 𖼟𖽡, 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼺𖽙𖾏 𖼟𖽡,
𖼁𖽙𖾏 𖼞𖽡𖾑 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏.
17 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖽐𖼚𖽜𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏,
𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼏𖽵 𖼮𖽡𖽝𖾏.
 
18 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏,
𖼷𖽑𖽜𖾑 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑.
19 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼟𖽡 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼎𖽻𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼟𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼑𖽡𖾏 𖼳𖽦𖾑.
 
20 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖽝𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼞𖽑𖽔𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐.
21 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼺𖽷𖾐 𖼙𖽠𖾏,

22 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼑𖽜 𖼟𖽡 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼞𖽡𖾑！
