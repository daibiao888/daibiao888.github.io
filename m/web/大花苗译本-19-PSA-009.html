𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 “𖼄𖽙𖾏 𖼖𖽔𖾑 𖼀𖽡𖽝𖾏”𖼳𖽘𖾐.
9
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑.
2 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖽐𖼷𖽷𖾏！
 
3 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼑𖽡𖾏 𖼙𖽠𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽂𖽪𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼎𖽺𖾏 𖼢𖽻 𖼛𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏.
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑, 𖼑𖽜 𖼞𖽪𖾑 𖼚𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽡𖽝𖾏 𖼋𖽘𖾏,
𖼟𖽡 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖼸𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏.
 
5 𖼟𖽡 𖼖𖽔𖾏 𖼉𖽻𖾏 𖼞𖽪𖽙, 𖼟𖽡 𖼖𖽻𖾐 𖼊𖽪𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐,
𖼟𖽡 𖼇𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏.
6 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼸𖽙 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
𖼟𖽡 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏 𖼞𖽑𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏.
 
7 𖼯𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖼐𖽱𖾏 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑.
8 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼨𖽑𖽢𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏.
 
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖽃𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑,
𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑.
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼖𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡.
 
11 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖽃𖽡𖾑,
𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏.
12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼾𖽱 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖼦𖽡𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼑𖽡𖾏 𖼐𖽡𖽝𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
 
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼮𖽑𖽳𖾑 𖼞𖽪𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼯𖽡𖽙𖾏,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑,
𖼑𖽔𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑,
14 𖼻𖽻𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼙𖽻 𖽃𖽔 𖼛𖽻𖾏( “𖼙𖽻” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖽐𖼷𖽑𖽹𖾏 ” ) 𖼑𖽜 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼚𖽙𖾏.
 
15 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼮𖽡𖽠𖾏 𖼮𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼊𖽻𖾏 𖼦𖽡𖾏 𖼚𖽱𖾏 𖽐𖼊𖽠𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼚𖽺𖾏 𖼊𖽑𖽔𖾑 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼋𖽘𖾏.
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼾𖽨𖾏 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼙𖽔 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼋𖽘𖾏,
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼊𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼦𖽡𖾏 𖼚𖽱𖾏 𖽐𖼊𖽠𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏. （𖼳𖽷𖾏 𖼮𖽢𖾑 𖼽𖽙𖽾𖾑、𖼳𖽡𖾏 𖼖𖽔𖾑）
 
17 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐, 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏.
18 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼑𖽡𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼚𖽔𖾏 𖼯𖽢𖾏.
 
19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼳𖽠𖾏！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐.
20 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾐. （𖼳𖽡𖾏 𖼖𖽔𖾑）
