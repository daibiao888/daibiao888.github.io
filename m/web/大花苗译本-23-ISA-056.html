𖼽𖽔𖾑 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏
56
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼅𖽡 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼊𖽡𖾏 𖼈𖽳𖾏,
𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏.
2 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏,
𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.”
 
3 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼇𖽺𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 .”
𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 .”
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏,
𖼺𖽹𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼙𖽔 ,
𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
5 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 ,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼋𖽜𖾏 𖽐𖼚𖽙 𖼅𖽔𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏, 𖼅𖽔𖾏 𖽐𖼷𖽷𖾏,
𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏．
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽘𖾏 𖼄𖽔𖾐 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼇𖽳𖾏 𖼺𖽘𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼉𖽻𖾏 𖼳𖽡𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖽃𖽡𖾑,
𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏, 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏,
𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑, 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼦𖽡𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏,
𖼎𖽺𖾏 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖽃𖽡𖾑,
7 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑．
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.”
8 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:
“𖼐𖽡𖽙𖾐 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖾑 𖼉𖽺𖾏 𖼢𖽻𖾏,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽙𖾑 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏.”
𖼖𖽔𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐
9 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼘𖽪 ! 𖼋𖽔𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼦𖽡𖾑！
𖼡𖽔𖾏 𖼻𖽻𖾏 𖽐𖼚𖽙 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼘𖽪 ! 𖼋𖽔𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏！
10 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 ,
𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼚𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽜 𖼅𖽜 ,
𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽠𖾏 𖼺𖽘𖾐, 𖼮𖽔𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 , 𖼢𖽻 𖼀𖽱𖾏, 𖽐𖼮𖽑𖽠𖾏 𖼀𖽱𖾏 𖼊𖽔𖾐，
11 𖼋𖽡𖾏 𖼚𖽡𖾑 𖼐𖽡𖾑 𖼦𖽠𖾑 𖼑𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏, 𖼋𖽪 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐,
𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼙𖽠𖾏,
𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽔𖾏 𖼺𖽘𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼾𖽨𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖽠𖾑, 𖼀𖽡𖾐 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐 𖼟𖽪𖾏 𖼇𖽔𖾑,
𖼀𖽡 𖼟𖽡 𖽐𖼎𖽙 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐,
𖼀𖽡𖾐 𖼽𖽔𖾑 𖼦𖽻𖾏 𖼟𖽙𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼇𖽔𖾑 𖼷𖽻𖾏.”
