𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼢𖽳 𖼋𖽠𖾏 𖼮𖽡𖾏 𖼗𖽙𖾏 .

𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏！
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼷𖽻𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑? 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
 
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼟𖽡𖾏 𖼦𖽡𖾏 𖼊𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏,

5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼟𖽡𖾏 𖼦𖽺𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽙𖾏.
6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾏 𖼋𖽔 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐, 𖼎𖽺𖾏 𖼊𖽔𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏,
𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼀𖽻𖾐.”
 
8 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼘𖽪！
𖼅𖽡 𖼽𖽔𖾑 𖽐𖼊𖽱𖾏 𖽃𖽡𖾐 𖽐𖼊𖽱𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼘𖽪！
𖼅𖽡 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼖𖽺𖾑?
9 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖽃𖽡𖾑, 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼐𖽑𖽙𖾑 𖼘𖽪?
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽡𖾑 , 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼘𖽪?
10 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼯𖽵𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 ,
𖼋𖽡𖽝𖾏 𖼾𖽙 𖼐𖽱𖾏 𖼦𖽡𖾐 𖽐𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽺𖾑?
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 𖼯𖽡𖽙𖾏.
 
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏、
𖼎𖽑𖽻𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
13 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 ,
𖼋𖽙𖾏 𖼸𖽙 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽡𖾑 𖽐𖼮𖽡𖽠𖾏 𖼊𖽡𖽝𖾑.
14 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏.
15 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽔𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 .
 
16 𖼡𖽔 𖼋𖽱𖾏 𖼯𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽡𖾑 ?
𖼡𖽔 𖼋𖽱𖾏 𖼯𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 ?
17 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽐𖼷𖽙𖾑 𖽐𖼷𖽙𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏.
18 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏, 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐 𖼛𖽔𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 .
19 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽡𖽻𖾐 𖼊𖽳𖾏 𖼮𖽡𖽻𖾐 𖽐𖼊𖽱𖾏,
𖼟𖽡 𖼎𖽳𖾏 𖽐𖼚𖽡𖾏 𖼞𖽪𖾑, 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
 
20 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽑𖽘𖾑 𖼢𖽔𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳, 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼖𖽺𖾑?
21 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡 𖼸𖽱 𖽐𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼋𖽔 .
22 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔,
𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
23 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.
