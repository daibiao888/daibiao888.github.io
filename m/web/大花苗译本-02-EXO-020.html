𖼟𖽻𖾏 𖼼𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
（𖼎𖽺𖾏 5：1 - 21）
20
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 , 𖼐𖽑𖽡𖾏 :
2 “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 .
3 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
4 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 , 𖼺𖽱𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏、𖼀𖽡𖾑 𖼊𖽡𖾐、𖼼𖽘𖾏 𖼚𖽘𖾐 𖽐𖼚𖽡𖽝 𖼊𖽡𖾐、𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 . 5 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼋𖽡𖾏 𖼼𖽘𖾏 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 . 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼾𖽱 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 , 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼸𖽙 𖼊𖽪𖾐 , 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾐 𖼚𖽻𖾐 𖼷𖽑𖽷𖾏 , 6 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 , 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽑𖽡𖽝𖾐 𖼷𖽑𖽷𖾏 .
7 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼨𖽑𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼨𖽑𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 .
8 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 , 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 9 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖽃𖽔𖾏 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 . 10 𖼈𖽡𖽝𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 . 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏、𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏、𖼯𖽦𖾏、𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼡𖽑𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 , 11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖽐𖼚𖽙 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏、𖼊𖽡𖾐、𖼳𖽡𖽠𖾑 , 𖽐𖼎𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼉𖽻𖾏 𖼮𖽘𖾑 , 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼳𖽙𖾏 𖼼𖽙 , 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 , 𖼏𖽵 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .
12 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 , 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 .
13 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 .
14 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 .
15 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼺𖽘𖾐 𖽃𖽔𖾏 𖼸𖽜 .
16 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙 .
17 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽠𖾑 𖼉𖽳 𖼼𖽷𖾏 𖼼𖽙 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 , 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐、𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏、𖼑𖽨𖾏、𖼑𖽜𖾏 𖼖𖽔𖾑、𖼊𖽑𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 .”
𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏
18 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽙𖾑 𖼺𖽙𖾐 𖼢𖽔𖾐、𖼁𖽙𖾏 𖼺𖽙𖾐 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼯𖽦𖾏、𖽐𖼎𖽙 𖼞𖽪𖾐 𖼑𖽡𖾏 𖼳𖽘𖾐、𖼊𖽑𖽡𖽝𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼋𖽠𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 , 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼳𖽠𖾑 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 . 19 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼷𖽑𖽜𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼀𖽡𖾐 𖼋𖽔 𖼺𖽘𖾏 .” 20 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽐𖼮𖽔𖾏 𖼅𖽡 , 𖼟𖽙𖾏 𖼅𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 , 𖼦𖽡𖾐 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏 .” 21 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼳𖽠𖾑 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼯𖽔 𖼅𖽔𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏 .
𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘
