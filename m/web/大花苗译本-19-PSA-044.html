2 𖼟𖽡 𖼙𖽔 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐,

3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼊𖽻𖾏 𖼳𖽠𖾏, 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏,
𖼟𖽡 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽱𖾏 𖼸𖽻𖾏.
 
4 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽙𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼊𖽻𖾏 𖼳𖽠𖾏.
5 𖼀𖽡𖾐 𖼻𖽡𖽝𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽑𖽠𖾐 𖽐𖼷𖽜𖾑 𖼀𖽡𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖼻𖽡𖽝𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼽𖽔𖾑 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾐 𖽃𖽡𖾑.
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼻𖽡𖽝𖾏 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼐𖽑𖽜𖾑,
𖼞𖽪𖾑 𖼊𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼳𖽠𖾏.
7 𖼯𖽔 𖼅𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
8 𖼀𖽡𖾐 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡 𖽐𖼷𖽷𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
9 𖼈𖽡𖽝𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼟𖽡 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼀𖽡𖾐 𖼺𖽘𖾏, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑,
𖼦𖽡𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼸𖽻𖾏 𖼏𖽙 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏.
10 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙𖾏 𖼙𖽠𖾏,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼀𖽡𖾐 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼦𖽔𖾏.
11 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼺𖽱𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼋𖽘𖾏 𖽃𖽡𖾑,
𖼄𖽔𖾐 𖼀𖽡𖾐 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼞𖽪𖽙.
12 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖽐𖼡𖽜𖾏 𖼳𖽷𖾐 𖼋𖽺 𖼺𖽘𖾏,
𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼟𖽙𖾏 𖼟𖽡 𖼅𖽔𖾏 𖼊𖽻𖾏.
 
13 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼞𖽪𖽙 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑,
𖼊𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽙𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼀𖽡𖾐.
14 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼛𖽙𖾏,
𖼟𖽙𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼇𖽜𖾏 𖼇𖽜𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐.
15 𖼞𖽪𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼞𖽪𖾑 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼄𖽔𖾐 𖼞𖽪𖾑 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼮𖽡𖽝𖾏.
16 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼀𖽡𖾐 𖽃𖽡𖾑 𖼳𖽘𖾐 𖽃𖽔𖾐 𖼡𖽜𖾏,
𖼎𖽺𖾏 𖼑𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼀𖽡𖾐 𖼐𖽔𖾑.
 
17 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼞𖽑𖽔𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼟𖽡 𖼺𖽘𖾏,

18 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐,
𖼀𖽡𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼅𖽻𖾏 𖽐𖼮𖽠𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑.
19 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐, 𖼸𖽺𖾏 𖼀𖽡𖾐 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏,
𖼎𖽑𖽻𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼀𖽡𖾐.
 
20 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼮𖽘𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺，
21 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼘𖽪?
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏.
22 𖼀𖽡𖾐 𖼑𖽜 𖼟𖽡 𖼐𖽔𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏,
