𖼀𖽻𖾐 𖼘𖽙 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽡𖾑 ， 𖼐𖽑𖽡𖾏：“ 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼯𖽡𖽙𖾏 𖼺𖽡𖽝𖾐 𖼚𖽠𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼺𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 ．” 2 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾏 𖼐𖽔 𖼐𖽡 𖼽𖽔𖾏 ， 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽃𖽡𖾐 𖼺𖽘𖾐 𖽃𖽡𖾑 𖽐𖼀𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏． 3 𖼀𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼋𖽜𖾏 𖼟𖽪𖾏 𖼘𖽙𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼇𖽜𖾐 𖼐𖽡𖾑，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼟𖽡！ 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽢𖾐 𖼑𖽪 𖼞𖽪𖾑， 𖼟𖽡 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽐𖼷𖽜𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼺𖽘𖾏，𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 𖼘𖽪𖾏？” 4 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼞𖽔𖾐 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼘𖽪𖾏？” 5 𖼀𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏：“ 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：‘𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 ．’”
6 𖼀𖽻𖾐 𖼘𖽙 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱，𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼊𖽪𖾑 𖼞𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐，𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏：“ 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡， 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼟𖽪𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏，𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪．” 7 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏， 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼊𖽪𖾑 𖼞𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽺𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼸𖽻 ． 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼊𖽪𖾑 𖼞𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙，𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼖𖽡𖽾，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙． 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏．𖼅𖽔𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖽃𖽡𖾐 𖼸𖽻 𖽃𖽡𖾑 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼐𖽑𖽡𖾏： “𖼀𖽡𖾐 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼖𖽡𖽾 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼙𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏， 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？” 10 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽠𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 ， 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼀𖽻𖾐 𖼘𖽙 𖼚𖽔𖾏 𖽐𖼮𖽑𖽷𖾏 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼘𖽑𖽻𖾑 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏， 𖼎𖽑𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ，𖼨𖽑𖽳𖾏 𖼚𖽔𖾏 𖼏𖽙 𖼖𖽪𖾑 𖼾𖽡 𖼙𖽠𖾏．

𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼐𖽑𖽙𖾏 𖼀𖽻𖾐 𖼘𖽙
