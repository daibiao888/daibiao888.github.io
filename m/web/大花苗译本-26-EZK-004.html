“𖼟𖽡, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼑𖽡 𖼋𖽺𖾏, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖽐𖼚𖽘𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽻 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼮𖽡𖽝𖾏, 2 𖼎𖽺𖾏 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 , 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼙𖽻, 𖼮𖽪𖾐 𖽃𖽔𖾏 𖼚𖽻𖾏 𖽃𖽘𖾑 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼯𖽢 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖽐𖼊𖽻𖾏 𖼙𖽻 , 𖼮𖽪𖾐 𖼾𖽡 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼊𖽜𖾐 𖼮𖽡𖽝𖾏 𖼡𖽔𖾏 𖼮𖽤𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽱𖾑𖽑 𖼙𖽻， 3 𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼋𖽺𖾏 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽡𖽝𖾏, 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼖𖽑𖽻𖾏. 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼽𖽔𖾑 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 , 𖼟𖽙𖾏 𖼙𖽻 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 , 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽻𖾏 𖼙𖽻𖾏 𖽐𖼊𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑, 𖼟𖽙𖾏 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏.
4 𖼟𖽡 𖼽𖽔𖾑 𖼀𖽱𖾏 𖼊𖽡𖾑 𖽐𖼮𖽹𖾏 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏, 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽱𖾏 𖼊𖽡𖾑 𖽐𖼮𖽹𖾏 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 𖽃𖽡𖾑, 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏. 5 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏, 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼑𖽡𖾏 𖼾𖽔𖾏. 6 𖼀𖽪𖾑 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼀𖽱𖾏 𖼊𖽡𖾑 𖽐𖼮𖽹𖾏 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼾𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏. 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐, 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽙𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏. 7 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾏, 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼽𖽔𖾑 𖼢𖽳 𖼟𖽡𖾏 𖼋𖽠𖾏 𖼙𖽪 , 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖽐𖼊𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 . 8 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼖𖽑𖽔𖾏 𖼡𖽑𖽹𖾐 𖽐𖼊𖽠𖾏 𖼟𖽡 , 𖼟𖽙𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼦𖽡𖾐 𖼷𖽷𖾐, 𖼋𖽙𖾏 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼙𖽻 𖽃𖽡𖾑 𖼀𖽪𖾑 𖼺𖽘𖾏.
9 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼳𖽻𖾐、 𖼅𖽙、 𖼋𖽻𖾏、𖼋𖽻𖾏 𖼖𖽡𖽝𖾐、 𖼺𖽻𖾑、 𖼅𖽙 𖽐𖼷𖽑𖽷𖾐, 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼮𖽡𖽝𖾏, 𖼯𖽡𖽙𖾏 𖼗𖽙𖾏 𖼑𖽜 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼳𖽡𖽝𖾐, 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽱𖾏 𖼊𖽡𖾑 𖽐𖼮𖽹𖾏 , 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼳𖽡𖽝𖾐 𖼐𖽡𖾑 . 10 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽻𖾏 , 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼑𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 , 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼑𖽻𖾏 . 11 𖼟𖽡 𖼦𖽻𖾏 𖽃𖽻𖾐 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏, 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼦𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡 𖼚𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠, 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼦𖽻𖾏. 12 𖼟𖽡 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼳𖽡𖽝𖾐 𖼐𖽡𖾑 𖼺𖽱𖾏 𖼑𖽻𖾏 𖼅𖽙 𖼳𖽡𖽝𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖾑 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼚𖽻𖾑 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼷𖽑𖽡𖽝𖾐.” 13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑, 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼑𖽻𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏.” 14 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖽃𖽺𖾐！ 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼦𖽡𖾐 𖼙𖽔 𖼊𖽻𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏, 𖼎𖽑𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼸𖽙 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼋𖽔 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽐𖼮𖽑𖽷𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑, 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.” 15 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼑𖽨𖾏 𖼮𖽑𖽡𖾑 𖼗𖽙𖾏 𖼘𖽪𖾑𖽑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼳𖽡𖽝𖾐 𖼷𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑.” 16 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏, 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏. 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼳𖽡𖽝𖾐 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼀𖽠 𖼑𖽻𖾏, 𖼑𖽻𖾏 𖼈𖽡𖽝𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏, 𖼦𖽻𖾏 𖽃𖽻𖾐 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽐𖼮𖽔𖾏 𖼮𖽡𖽝𖾏 𖼦𖽻𖾏, 𖼦𖽻𖾏 𖼟𖽡𖾏 𖼏𖽘𖾏 𖽐𖼎𖽘. 17 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼡𖽻𖾐, 𖼮𖽔𖾑𖽑 𖼯𖽠 𖽃𖽻𖾐, 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼑𖽜 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.”
