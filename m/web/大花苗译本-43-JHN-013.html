𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼁𖽡𖽝，𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙 𖼡𖽘𖾐． 2 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼄𖽑𖽙𖾏， 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼄𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐𖼳𖽡𖾑 𖼄𖽞 𖼊𖽪𖾐 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐． 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼉𖽺𖾏 𖼄𖽔𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼀𖽻𖾐 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼗𖽙𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 4 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼄𖽑𖽙𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏， 𖼚𖽠𖾏 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏，𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼾𖽨𖾏 𖼚𖽔𖾑， 5 𖼎𖽺𖾏 𖼘𖽑𖽡𖾑 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼋𖽺𖾏， 𖽐𖼷𖽔𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽠𖾏，𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼚𖽔𖾑 𖼳𖽙𖾏 𖼟𖽪𖾏 𖽐𖼡𖽑𖽔𖾐． 6 𖼐𖽱𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼳𖽡𖾑 𖼄𖽞 · 𖼀𖽡𖾐 𖼊𖽳，𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼟𖽡 𖼽𖽔𖾑 𖽐𖼷𖽔𖾑 𖼞𖽪𖾑 𖼊𖽠𖾏 𖼘𖽪𖾏？” 7 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏，𖼟𖽡 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐，𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼟𖽡 𖼖𖽔𖾑 𖼀𖽻𖾐 𖼯𖽡𖽙．” 8 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽔𖾑 𖼞𖽪𖾑 𖼊𖽠𖾏！” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽐𖼷𖽔𖾑 𖼟𖽡，𖼟𖽡𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼀𖽠．” 9 𖼳𖽡𖾑 𖼄𖽞 ·𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪 ！𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼷𖽔𖾑 𖼞𖽪𖾑 𖼊𖽠𖾏 𖼊𖽔𖾐， 𖼞𖽪𖾑 𖼋𖽡𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼖𖽔𖾏 𖽐𖼷𖽔𖾑 𖼊𖽑𖽡𖽝𖾐．” 10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼙𖽔 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼯𖽔 𖼽𖽔𖾑 𖽐𖼷𖽔𖾑 𖼊𖽠𖾏 ，𖽃𖽔𖾏 𖼮𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽺𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏．𖼅𖽡 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼯𖽡𖽙𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏．” 11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽺 𖼐𖽱𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼦𖽡𖾐 𖼾𖽙 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏．”
12 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼷𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽠𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏，𖼎𖽺𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？” 13 𖼅𖽡 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏、𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽵𖾑，𖼅𖽡 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼾𖽙 𖼯𖽡𖽙𖾏． 14 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏，𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼷𖽔𖾑 𖼅𖽡 𖼊𖽠𖾏，𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽔𖾑 𖽐𖼷𖽔𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽠𖾏． 15 𖼞𖽪𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼼𖽘𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼾𖽻，𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏． 16 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼡𖽑𖽜𖾑 𖼦𖽡𖾐 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏 𖼮𖽵𖾑，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼦𖽡𖾐 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼋𖽔𖾏． 17 𖼅𖽡 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏，𖼅𖽡 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏． 18 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜， 𖼞𖽪𖾑 𖼙𖽔 𖼺𖽹𖾑 𖼡𖽔 𖼋𖽱𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼯𖽡𖽙.𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑：‘𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼑𖽻𖾏 𖼳𖽡𖽝𖾐，𖼮𖽑𖽳𖾑 𖼡𖽘𖾐 𖽃𖽔 𖼙𖽻𖾏 𖼊𖽠𖾏 𖼋𖽔𖾏 𖼞𖽪𖾑 ．’ 19 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼑𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽃𖽔𖾏 𖼺𖽷𖾐，𖼞𖽪𖾑 𖼺𖽪𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏 𖼅𖽡 𖼎𖽺𖾏 𖽐𖼮𖽜 𖼞𖽪𖾑 𖼟𖽪𖾏 ‘𖼷𖽷𖾑 𖽃𖽡𖾑 ’𖼯𖽡𖽙𖾏 ． 20 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽷𖾐，𖼟𖽪𖾏 𖼷𖽹𖾏 𖼞𖽪𖾑；𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼞𖽪𖾑，𖼟𖽪𖾏 𖼷𖽹𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼗𖽙𖾏．”
𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 26：20-25；𖼞𖽑𖽙𖾐 14：17-21；𖼖𖽪𖾏 22：21-23）
21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏，𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏，𖼎𖽺𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽔𖾑 𖼋𖽺 𖼞𖽪𖾑．” 22 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐，𖼷𖽠𖾑𖽑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽡𖾏． 23 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐， 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽑𖽠𖾏，𖼢𖽹𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐． 24 𖼳𖽡𖾑 𖼄𖽞 · 𖼀𖽡𖾐 𖼊𖽳 𖼎𖽺𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏，𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽡𖾏． 25 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼢𖽹𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？” 26 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼚𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼳𖽡𖽝𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼚𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼳𖽡𖽝𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽡𖾑 𖼄𖽞 𖼊𖽪𖾐 𖼽𖽨 𖼊𖽔𖾏． 27 𖼐𖽱𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼳𖽡𖽝𖾐 𖼐𖽡𖾑 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼺𖽔𖾐 𖼊𖽔 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼁𖽻 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽔𖾑 𖽃𖽔𖾏 ，𖼞𖽑𖽜 𖼑𖽜 𖽃𖽔𖾏 𖼦𖽡𖾑！” 28 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽡𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐， 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑． 29 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼯𖽡𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖼑𖽡𖽝𖾏， 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏：𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼅𖽻𖾏 𖼅𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏．’ 30 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼳𖽡𖽝𖾐，𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏． 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼋𖽘．
𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑
31 𖼽𖽨 𖼊𖽔𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼋𖽘𖾏，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏． 32 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼊𖽻𖾏 𖼋𖽘𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽱𖾏． 33 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼅𖽡 𖼘𖽪！𖼞𖽪𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼋𖽘𖾏．𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼮𖽡𖽻𖾏．𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼐𖽔𖾑 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡． 34 𖼞𖽪𖾑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼎𖽻𖾏 𖼅𖽡，𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼅𖽡． 35 𖼅𖽡 𖼮𖽔𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼅𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏．”
36 𖼳𖽡𖾑 𖼄𖽞 · 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏？” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏，𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏，𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏．” 37 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼸𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏？ 𖼞𖽪𖾑 𖼐𖽑𖽡𖽻𖾏 𖼑𖽜 𖼟𖽡 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑．” 38 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼐𖽑𖽡𖽻𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼗𖽔𖾏 𖼟𖽡 𖼳𖽦𖾑？ 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡，𖼡𖽹𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼡𖽔𖾏 𖼺𖽷𖾐，𖼟𖽡 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼟𖽻𖾐 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼞𖽪𖾑．”
