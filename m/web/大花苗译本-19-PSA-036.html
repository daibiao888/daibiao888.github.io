𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏，

2 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏, 𖼾𖽨𖾏 𖼑𖽔𖾏 𖼾𖽨𖾏 𖼻𖽻𖾏,
𖼸𖽙𖾏 𖼞𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼳𖽪𖾐.
3 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼗𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽐𖼎𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐,
𖼐𖽱𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽐𖼎𖽙 𖼻𖽻𖾏 𖼑𖽪𖾏 𖼊𖽪𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
4 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼸𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽱𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽔𖾐,
𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾐, 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
 
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼺𖽡𖽝𖾐 𖼸𖽷𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏.
6 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼺𖽱𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐,
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼺𖽱𖾏 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼊𖽙𖾐.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼯𖽦𖾏， 𖼟𖽡 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 .
7 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖽐𖼚𖽙 𖼊𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼡𖽘𖾐 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏,
𖼟𖽡 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽻𖾏 𖼟𖽡 𖼗𖽙𖾏 𖼳𖽦𖾑 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐.
9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏,
𖼐𖽡𖽙𖾐 𖼟𖽡 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑.
 
10 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼺𖽘𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡,
