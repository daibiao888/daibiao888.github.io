𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱， 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏． 2 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼋𖽡𖽝𖾏 𖼖𖽔𖾑 𖼞𖽑𖽙𖾏 𖼦𖽡𖾐 𖼞𖽑𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼐𖽱𖾏． 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 ， 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐．” 4 𖼎𖽺𖾏 𖼑𖽪 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏， 𖼮𖽨𖾏 𖼳𖽦𖾑 𖼐𖽑𖽙𖾏 𖼳𖽦𖾑， 𖼼𖽘𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼢𖽻 𖼖𖽺𖾑？” 𖼐𖽱𖾏 𖼸𖽻 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼋𖽠𖾏 𖼺𖽘𖾐． 5 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼞𖽑𖽡 𖼙𖽻，𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽷𖾑 𖼐𖽡𖽙𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏： “ 𖼳𖽤𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏．” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼳𖽤𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐． 6 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏， 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼖𖽡 𖽃𖽡𖾐 𖼸𖽻 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖼊𖽳𖾏， 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽑𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑．
7 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪𖾏 𖼳𖽡𖽠𖾑， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼋𖽔𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏． 8 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏， 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏、𖼽𖽨 𖼊𖽑𖽹𖾏、𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐、 𖼽𖽡𖾐 𖼊𖽑𖽪𖾐 𖼄𖽹𖾐、𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙、 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏， 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼣𖽔𖾏 𖽐𖼞𖽙𖾏 ， 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽹𖾏 𖼐𖽱𖾏 𖼐𖽔𖾑． 10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽔𖾑， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼐𖽹𖾏 𖼋𖽔𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 ． 11 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐， 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐．” 12 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽹𖾑 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏
（𖼊𖽑𖽹𖾏 10：1-4；𖼖𖽪𖾏 6：12-16）
13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐， 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼏𖽻 𖼡𖽔 𖼋𖽱𖾏， 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 14 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼺𖽹𖾑 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼽𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏， 15 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐． 16 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏： 𖼳𖽡𖾑 𖼄𖽞，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽳， 17 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼊𖽑𖽹𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼦𖽗， 𖼎𖽺𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼀𖽔𖾏 𖼐𖽡 𖼮𖽑𖽡， 𖼟𖽪𖾏 “𖼺𖽙𖾐 𖼊𖽪𖾐 ” ， 18 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼣𖽔𖾑 𖼊𖽳 𖼖𖽡𖽝 、 𖼇𖽺𖾑 𖼖𖽡、 𖼀𖽔𖾑 𖼊𖽙𖾑 𖼘𖽙 𖼄𖽹𖾐、 𖼄𖽔𖾐 𖼊𖽑𖽹𖾏 、 𖼊𖽙𖾑 𖼄𖽔𖾐、 𖼽𖽔𖾏 𖼖𖽳 𖼇𖽺𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙 、 𖼊𖽔 𖼊𖽑𖽹𖾏 、 𖼇𖽞𖾏 𖼵𖽪𖽺𖾏 𖼊𖽗𖾐 𖼳𖽡𖾑 𖼄𖽞， 19 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽨 𖼊𖽔𖾏， 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐
（𖼊𖽑𖽹𖾏 12：22-23；𖼖𖽪𖾏 11：14-23；12：10）
20 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 ． 21 𖼐𖽱𖾏 𖼑𖽜𖾏 𖼮𖽘𖾏 𖼐𖽙𖾑𖽑 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾑 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼩𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏． 22 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼡𖽜𖾏 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖽝 𖼳𖽡𖾑 𖼀𖽑𖽪 𖼯𖽵𖾏 𖼐𖽱𖾏．” 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏： “𖼐𖽱𖾏 𖼻𖽡𖽝𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐．” 23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼺𖽔𖾐 𖼊𖽔 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼸𖽠𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼺𖽔𖾐 𖼊𖽔 𖼖𖽺𖾑？ 24 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼞𖽪𖽙 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏， 𖼖𖽪𖾑 𖼞𖽪𖽙 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏． 25 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏， 𖼾𖽡𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏． 26 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼊𖽔 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼾𖽨𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼦𖽡𖾐 𖼏𖽔𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼮𖽘𖾑． 27 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼺𖽪𖾐 𖼄𖽔𖾐 𖼷𖽷𖾑 𖼊𖽪𖾐 𖼋𖽻 𖽃𖽡𖾑 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼮𖽡𖽝𖾏， 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐． 28 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 ， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏． 29 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽷𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏．” 30 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼯𖽵𖾏 𖼐𖽔𖾑．
（𖼊𖽑𖽹𖾏 12：46-50；𖼖𖽪𖾏 8：19-21）
31 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼋𖽔𖾏 𖼋𖽘𖾏，𖼳𖽠𖾑 𖽐𖼎𖽙𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑， 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 ． 32 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼐𖽡𖽙𖾐 𖼟𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “ 𖼑𖽔𖾏 𖼘𖽪！ 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡．” 33 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 ？ 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼖𖽺𖾑？ ” 34 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑， 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼑𖽔𖾏， 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏． 35 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽡𖾑， 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼞𖽱𖾑 𖼻𖽷𖾑， 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼯𖽡𖽙．”
