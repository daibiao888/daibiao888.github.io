𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼎𖽙𖾑 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽡𖾐
14
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙 , 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 , 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏． 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏, 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏. 2 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽳𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼸𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼦𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑.
𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏
3 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼳𖽙𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏、 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏, 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼑𖽪𖾏 𖼐𖽑𖽤𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼊𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽃𖽡𖾑, 4 𖼟𖽡 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 𖼐𖽡𖾑 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏:
“𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔?
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏?
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏, 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼡𖽠𖾏,
6 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼯𖽪 𖼯𖽪𖾏 𖽐𖼊𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐,
𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽳𖾏 𖽃𖽡𖾑.
7 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼳𖽙𖾏, 𖼊𖽻𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 ,
𖽃𖽔𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏.
8 𖼊𖽑𖽪𖾑 𖽐𖼎𖽙 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏, 𖼐𖽑𖽡𖾏:
𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼢𖽻 𖼀𖽱𖾏 𖼙𖽠𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖽐𖼊𖽙𖾑 𖼀𖽡𖾐 𖼋𖽘 .
9 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼑𖽜 𖼟𖽡 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 ,
𖼙𖽔 𖼷𖽹𖾏 𖼟𖽡 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 ,
𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖽃𖽡𖾑 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽜 𖼟𖽡 𖼏𖽘𖾏 𖼀𖽡𖾏 𖼮𖽨𖾑 ,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑, 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐.
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖼖𖽔𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼯𖽻 𖼼𖽙 𖼺𖽱𖾏 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼋𖽘𖾏 𖼖𖽔𖾏?
𖼟𖽡 𖼖𖽔𖾏 𖼚𖽑𖽡𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
11 𖼟𖽡 𖼚𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼢𖽜𖾏 𖽐𖼎𖽔 𖼑𖽡𖾏 𖼳𖽘𖾐, 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏．
𖼟𖽡 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼀𖽔𖾏 𖼀𖽡𖾑 𖼞𖽘𖾐, 𖼟𖽡 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼟𖽪𖾏 𖼈𖽳𖾑 𖼞𖽘𖾑 𖼞𖽘𖾏.
 
12 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖾏、 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼘𖽪 !
𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽻𖾐 𖼗𖽙？
𖼟𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 , 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼊𖽻𖾏 𖽐𖼊𖽙𖾑 𖼢𖽻 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼙𖽠𖾏 ？
13 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼐𖽑𖽡𖾏:
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏.
14 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼸𖽷𖾐.
15 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼸𖽙 𖼚𖽔𖾏 𖼡𖽘𖾐 𖼡𖽑𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐.
16 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏 𖼟𖽡,
𖼑𖽔𖾏 𖼟𖽡 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖼐𖽑𖽡𖾏:
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙 ,
𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏,
17 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼟𖽙𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼼𖽠𖾏 𖼢𖽻 𖼙𖽠𖾏, 𖼦𖽡𖾐 𖼚𖽠𖾏 𖼮𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏,

𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼗𖽙𖾏, 𖼦𖽔𖾏 𖼳𖽪𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,

22 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,

𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
23 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽪𖾑 𖼯𖽻 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏,
𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽻𖾐 𖼞𖽡𖾏 𖼷𖽑𖽷𖾑,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽡𖾑 𖼎𖽑𖽔𖾐 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔 𖼮𖽑𖽡𖾐 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼐𖽱𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼽𖽔𖾏 𖼳𖽪
24 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼏𖽵 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏:
𖼞𖽪𖾑 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼖𖽡𖾏 𖼯𖽤𖾏 ,
𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑，
25 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖼖𖽻𖾐 𖼽𖽔𖾏 𖼳𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼯𖽪𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐,

27 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼖𖽺𖾑?
𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽡𖾏 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼞𖽻𖾏 𖼙𖽠𖾏 𖼖𖽺𖾑?
𖽐𖼊𖽔𖾏 𖼏𖽵 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑
28 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼋𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐 𖼐𖽡𖾑 :

30 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏,
