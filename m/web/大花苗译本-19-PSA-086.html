2 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐.

3 𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑 ,

4 𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼟𖽡 .
5 𖼮𖽵𖾑 𖼘𖽪！ 𖼟𖽡 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐, 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼯𖽪𖾏 𖼯𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼎𖽑𖽻𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼷𖽻𖾏 𖼯𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡 .
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏,
𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼊𖽜 𖼽𖽔𖾑 .
7 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽡𖾐 𖼞𖽪𖾑 .
 
8 𖼮𖽵𖾑 𖼘𖽪！ 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼮𖽵𖾑 𖼳𖽺 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼋𖽳𖾏 𖼟𖽡 ,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼋𖽳𖾏.
9 𖼮𖽵𖾑 𖼘𖽪！ 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼟𖽡 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏.
10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑,
𖼯𖽔 𖼅𖽔𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼟𖽡 𖼞𖽘𖾐 𖼞𖽡𖾏 𖽃𖽔𖾏,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏！
12 𖼮𖽵𖾑, 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.

14 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 ,

17 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼢𖽳 𖼋𖽠𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 ,
