7
𖼽𖽦𖾑 𖼘𖽪𖾏 ‧ 𖼀𖽔𖾑 𖼖𖽡 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏, 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼐𖽡𖽙𖾐 𖼦𖽔𖾐 𖼖𖽡 𖼮𖽑𖽡𖽠𖾏 𖼷𖽠𖾑𖽑 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼮𖽪𖾐 𖼾𖽡, 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽪𖾐 𖼾𖽡 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏, 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐.
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼮𖽡𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼄𖽔𖾐 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽪𖾏 𖼀𖽡𖾐 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐. 3 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏: 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼚𖽔𖾐 𖽃𖽡𖾑, 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 .” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏, 𖼮𖽔𖾑 𖼛𖽡𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼊𖽔𖾐.
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼯𖽡𖽙 𖼺𖽷𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽻𖾐, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼑𖽜𖾏 𖼟𖽡 𖽐𖼮𖽔𖾏 𖼮𖽡𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽻𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽻𖾏, 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽑𖽡𖾏: 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽻𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽻𖾏.” 5 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽻𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖽐𖼚𖽺𖾏 𖼦𖽻𖾏 𖽃𖽻𖾐 𖼺𖽱𖾏 𖼚𖽡𖾑 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼦𖽻𖾏 𖽃𖽻𖾐 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐.” 6 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐 𖼦𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 , 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼦𖽻𖾏 𖽃𖽻𖾐. 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐 𖼦𖽻𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼐𖽡𖾑 𖼮𖽨𖾏 𖼅𖽡, 𖼄𖽔𖾐 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼸𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼞𖽔𖾏.” 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑, 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼟𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐, 𖼮𖽔𖾑 𖼷𖽷𖾐 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼮𖽡𖽝𖾏 𖼊𖽔𖾐, 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽪𖾑 𖼾𖽡 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡 𖼯𖽡𖾏 𖼖𖽪𖾑 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏.
