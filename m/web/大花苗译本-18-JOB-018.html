18
𖼳𖽪𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾐 𖼖𖽳 𖼊𖽔 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 :
2 “𖼟𖽡 𖽐𖼎𖽑𖽦𖾑 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼘𖽪 ?
𖼟𖽡 𖽐𖼊𖽱𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 , 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 .
3 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼯𖽦𖾏 ?
𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼖𖽺𖾑 ?
4 𖼟𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽑𖽡𖽠𖾏 𖼾𖽨𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 ,
𖼋𖽡𖽝𖾏 𖼾𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼑𖽜 𖼟𖽡 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏、 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼽𖽔𖾑 𖼮𖽑𖽹𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼖𖽔𖾏 ?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 𖼺𖽘𖾏 ,
𖼐𖽱𖾏 𖼊𖽪𖾑 𖼀𖽡𖾏 𖼗𖽺𖾏 𖼋𖽠𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼮𖽡𖾏 ．
6 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 ,
𖼐𖽱𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾐 𖼇𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼋𖽔 𖼺𖽘𖾏 .
7 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼅𖽔𖾏 𖼼𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼁𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼐𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼊𖽻𖾏 , 𖼽𖽔𖾑 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 𖼐𖽱𖾏 .
8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼨𖽑𖽳𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 ,
𖼅𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽱𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 .
9 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽔𖾏 𖼽𖽔𖾑 𖼊𖽔𖾑 𖼐𖽱𖾏 𖼡𖽘𖾐 𖽃𖽔 𖼙𖽻𖾏 𖼊𖽠𖾏 𖼮𖽡𖽝𖾏 ,
𖼖𖽪𖾑 𖼸𖽙𖾏 𖼽𖽔𖾑 𖼞𖽑𖽱𖾐 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏 .
10 𖼖𖽪𖾑 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽔𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼈𖽺𖾏 𖼎𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼮𖽡𖽝𖾏 ,
𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼈𖽺𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 .
11 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼑𖽡𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽹𖾏 ,
𖼎𖽺𖾏 𖼖𖽠𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼊𖽠𖾏 𖼐𖽜𖾑 .
12 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼯𖽻 𖼼𖽙 ,
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼺𖽘𖾑 𖼖𖽡𖾏 𖼊𖽘𖾑 𖽐𖼷𖽜𖾏 𖼋𖽙𖾏 𖼐𖽱𖾏 .
13 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 ,
𖼛𖽙𖾏 𖼋𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼽𖽔𖾑 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼀𖽘𖾐 𖼊𖽠𖾏 𖼋𖽡𖾏 .
14 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼚𖽙𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼑𖽡 𖽐𖼮𖽑𖽹𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 .
15 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼁𖽡𖽝 , 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼚𖽙 ,
𖼖𖽨 𖼦𖽫 𖼽𖽔𖾑 𖼨𖽑𖽢𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 .
16 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼽𖽔𖾑 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 ,
𖽃𖽔𖾑 𖼳𖽔𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽑𖽘𖾐 𖼺𖽘𖾏 .
17 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 ,
𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 .
18 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼚𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽙 ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 .
19 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼊𖽪𖾐 𖼘𖽑𖽔𖾏 ,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 .
20 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼑𖽜 𖼐𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 ,
𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼊𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 .
21 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑.”
