𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼑𖽡𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑
1
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐：
2 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 ,
𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐；
3 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳、𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑、𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳；
4 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼚𖽠𖾐 𖽐𖼮𖽑𖽦𖾐,
𖼟𖽙𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑,
5 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏, 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖽐𖼞𖽡𖽝𖾏,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑；
6 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖽐𖼎𖽙 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏,
𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖼋𖽻 𖼟𖽪𖾏 𖼷𖽠𖾑𖽑.
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑
7 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏.
8 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼽𖽔𖾑 𖼑𖽻 𖼟𖽡 𖼉𖽺𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏.
9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼑𖽡𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖽃𖽔𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼞𖽪𖾐.
10 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼟𖽡 ,
𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏.
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼅𖽻 ,
𖼀𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑,
𖼀𖽡𖾑 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼼𖽻𖾏 𖽐𖼚𖽺𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏.
12 𖼀𖽡𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽝𖾏 𖼙𖽠𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼡𖽑𖽙𖾑, 𖼏𖽻 𖼀𖽡𖾐 𖼞𖽑𖽔𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏；
13 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,
𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼀𖽡𖾐 𖽐𖼞𖽔𖾏．
14 𖼟𖽡 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽺𖾐,
𖼀𖽡𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼊𖽔𖾐 𖼐𖽑𖽘𖾐.”
15 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽙𖾏 𖽃𖽡𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼽𖽔𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ；
16 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼏𖽘 𖼇𖽺𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽺𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑.
17 𖼐𖽡𖽙𖾐 𖼐𖽻 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼊𖽑𖽔𖾑 𖼸𖽙𖾏,
𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙.
18 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽐𖼚𖽺𖾏 𖼮𖽡𖽝𖾏, 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽔𖾑 𖽐𖼚𖽱𖾏 𖼾𖽨𖾏 𖽐𖼮𖽑𖽘𖾑,
𖼼𖽻𖾏 𖽐𖼚𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼦𖽺𖾏 𖼾𖽨𖾏 𖼳𖽦𖾑.
19 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼦𖽠𖾑 𖼉𖽳 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 , 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏；
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽠𖾑 𖼉𖽳 𖼐𖽡𖾑, 𖼽𖽔𖾑 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑.
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳
20 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏,
𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏；
21 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻, 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽐𖼚𖽙 𖼙𖽻 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏，
22 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽐𖼮𖽑𖽠𖾏 𖼑𖽪𖾏 𖼡𖽔 𖼏𖽔,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼳𖽪𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
23 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼐𖽔𖾑 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼖𖽡𖽾 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼅𖽡 ,
𖼄𖽔𖾐 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼅𖽡 .
24 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏, 𖼅𖽡 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻 𖼸𖽙𖾏；
𖼞𖽪𖾑 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽳𖾑 .
25 𖼅𖽡 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐,
𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏；
26 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼅𖽡 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼚𖽙𖾏；
𖼑𖽪𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼷𖽠𖾏 𖼅𖽡 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼋𖽔𖾏 𖼊𖽔𖾐 𖼡𖽔𖾏 𖼚𖽙𖾏.
27 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡 , 𖼺𖽱𖾏 𖼮𖽢𖾏 𖼘𖽑𖽙𖾐,
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼗𖽙𖾏 𖼸𖽙 𖼺𖽱𖾏 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 .
28 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐；
𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼞𖽪𖾑 ；
29 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽪𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖼦𖽡𖾐 𖼺𖽹𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫；
30 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐,
𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏；
31 𖽃𖽔𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽷𖾏,
𖼢𖽻 𖼎𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼊𖽻𖾏.
32 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 𖼇𖽳𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏, 𖼽𖽔𖾑 𖼾𖽨𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼾𖽨𖾏 𖼳𖽦𖾑；
33 𖼮𖽔 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐,
𖼊𖽻𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .”
