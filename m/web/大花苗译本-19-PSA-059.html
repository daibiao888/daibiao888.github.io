𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼮𖽵𖾑 𖼮𖽨𖾏 𖼟𖽙𖾏 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑
𖼺𖽻𖾐 𖼘𖽙 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼟𖽻𖾏 𖼻𖽙𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼞𖽔𖾏, 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼐𖽡𖾑, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑” 𖼳𖽘𖾐.
（𖼺𖽔𖾐 𖽃𖽡𖾐 19 ：11 - 17）
59
𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖼮𖽙𖾏 𖼞𖽪𖾑 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 .
2 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼮𖽨𖾏 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑！
 
3 𖼑𖽔𖾏 𖼘𖽪𖾏 ! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑．
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼑𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.
4 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼈𖽡𖽝𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽘 𖼋𖽔𖾏, 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 ,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏, 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑！
5 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼯𖽷𖾏 𖼳𖽠𖾑, 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼉𖽻𖾏 𖼞𖽪𖽙！
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏！ （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼎𖽙𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐,
𖼺𖽱𖾏 𖼊𖽪𖾑 𖼚𖽡𖾑 𖼟𖽡𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏, 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼖𖽪𖾑 𖼙𖽻 .
7 𖼑𖽔𖾏 𖼘𖽪！ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼊𖽙𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏:
“𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽙𖾑 𖼖𖽺𖾑?”
 
8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼚𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼟𖽡 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼉𖽻𖾏 𖼞𖽪𖽙 .
9 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽡,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼷𖽜𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐．
10 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽹𖾏 𖼞𖽪𖾑 ,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏.
 
11 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏.
𖼮𖽵𖾑 𖼘𖽪！ 𖼟𖽡 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼙𖽠𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽻𖾐 𖼢𖽜 .
12 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼾𖽔𖾏 𖽐𖼎𖽙 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼗𖽪𖾏,
𖼊𖽑𖽡𖽝𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖽐𖼎𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼮𖽡𖽝𖾏.
13 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏, 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼸𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼺𖽻𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
14 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼎𖽙𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐,
𖼺𖽱𖾏 𖼊𖽪𖾑 𖼚𖽡𖾑 𖼟𖽡𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏, 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼖𖽪𖾑 𖼙𖽻 .
15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏 𖽐𖼎𖽑𖽦𖾑 𖼡𖽜𖾑 𖼑𖽻𖾏 ,
𖼮𖽔𖾑 𖼑𖽻𖾏 𖼦𖽡𖾐 𖼮𖽻𖾏 𖼎𖽺𖾏 𖼟𖽡𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏.
 
16 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼷𖽜𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 ， 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔.
17 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 .
