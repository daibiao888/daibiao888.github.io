𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼦𖽡𖾐 𖼞𖽳𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔 𖼾𖽠 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼏𖽵 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼐𖽔 𖼳𖽡𖾏 𖽃𖽳𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐( “𖼐𖽔 𖼳𖽡𖾏 𖽃𖽳𖾐” 𖼟𖽪𖾏 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼮𖽵𖾑 𖼁𖽡𖽝” ) , 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼸𖽪𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 3 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼮𖽡𖽠𖾑 𖼸𖽺𖾏 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐 𖼟𖽪𖾏 𖼚𖽡𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽻𖾏 𖼞𖽔𖾏 𖼦𖽡𖾐 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽡𖽠𖾑 𖼸𖽺𖾏 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐 𖽃𖽔𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽦𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼞𖽡 . 4 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼀𖽜𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼸𖽪𖾏 , 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽔𖾏 , 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏、𖼀𖽡𖾑 𖼊𖽠𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 .
5 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼏𖽵 𖼀𖽜𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼸𖽪𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼮𖽑𖽪𖾑 𖼮𖽑𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 , 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼊𖽪𖾑 𖽐𖼷𖽡𖽝𖾏 𖽐𖼊𖽡𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 . 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼊𖽘𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼸𖽪𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾑 . 6 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽜𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼸𖽪𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼋𖽔 . 7 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼞𖽱𖾏 𖼻𖽷𖾑 𖼋𖽔 𖼙𖽠𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽙𖾏 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽜𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼸𖽪𖾏 , 𖼎𖽙𖾑 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼐𖽱𖾏 𖼊𖽪𖾑 𖼺𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 . 8 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽜𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼸𖽪𖾏 , 𖼟𖽪𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .
