10
𖼽𖽔𖾏 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼐𖽔 𖼊𖽔 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼦𖽪𖾏 , 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼯𖽡𖽙𖾏 𖼾𖽨𖾏 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼷𖽙𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 , 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 , 𖼎𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼳𖽤𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽑𖽷𖾑 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑 . 2 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 .
3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽑𖽡𖾏 : “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼐𖽑𖽡𖾏 : ‘ 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖼢𖽳 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .’ ” 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 .
4 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼨𖽑𖽪𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼾𖽡𖾏 𖽂𖽪𖾑 𖼳𖽡𖽫𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼄𖽡𖾐 𖼳𖽔𖾑 𖼖𖽡𖾏、𖼽𖽡𖾑 𖼖𖽡𖾏 𖼺𖽔𖾐 𖼇𖽔𖾐 𖼋𖽔𖾏 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼋𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼮𖽑𖽳𖾑 𖼅𖽡 𖼑𖽜𖾏 𖼮𖽘𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼞𖽱𖾑 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼙𖽠𖾏.” 5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 , 𖼮𖽑𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼞𖽱𖾑 𖼋𖽠𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖼾𖽡 𖼙𖽠𖾏 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼯𖽡𖽙𖾏 . 6 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 、𖼽𖽡𖾑 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽑𖽙𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼁𖽪𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽔 , 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 . 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼯𖽡 𖽃𖽡𖾑 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 . 7 𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼋𖽠𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽔 𖼛𖽻𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽔 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼚𖽜𖾑 𖼐𖽡𖽙𖾐 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 .” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏 .
