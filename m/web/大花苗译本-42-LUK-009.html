9
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏，𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼎𖽙 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ，𖼟𖽙𖾏 𖼯𖽵𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽡𖾑 𖼚𖽘𖾐，𖼞𖽑𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐， 2 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼞𖽘𖾐 𖼞𖽡𖾏，𖼞𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐， 3 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼋𖽪 𖼡𖽔𖾏 𖼳𖽷𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽜，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖽐𖼎𖽙 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼮𖽑𖽙𖾏 𖼄𖽡𖾐 𖼋𖽱𖾐． 4 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼡𖽔 𖼾𖽡𖾏 𖼋𖽱，𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 ，𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼅𖽻𖾏． 5 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼅𖽡，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼇𖽳𖾑 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑，𖼽𖽔𖾑 𖽐𖼮𖽑𖽡𖽙𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼾𖽙．” 6 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼼𖽙𖾏 𖼡𖽑𖽔𖾐 𖼄𖽻𖾏 𖼻𖽻𖾏，𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼞𖽑𖽙𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐．
7 𖼳𖽡𖾑 𖼖𖽡 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜，𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼊𖽱𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽙𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“𖼟𖽪𖾏 𖼽𖽙 𖼦𖽗 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙．” 8 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏．” 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼖𖽡𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼋𖽘．” 9 𖼳𖽡𖾑 𖼖𖽡 𖼐𖽑𖽡𖾏：“𖼽𖽙 𖼦𖽗 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽷𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏，𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽺𖾑？𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽙𖾑 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑？𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏．”
𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐 𖼮𖽡𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖼮𖽻𖾏
（𖼊𖽑𖽹𖾏 14：13-21；𖼞𖽑𖽙𖾐 6：30-44；𖼽𖽙 6：1-14）
10 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼺𖽷𖾐 𖼗𖽙，𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼺𖽘𖾏，𖼚𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽳 𖼺𖽹 𖼊𖽔𖾏． 11 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼋𖽘，𖼎𖽺𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽻，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼷𖽹𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼡𖽑𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼏𖽻 𖼞𖽑𖽙𖾏 𖽃𖽡𖾑．
12 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼖𖽔𖾑 𖽐𖼷𖽜𖾑 𖼋𖽘𖾏，𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽙𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽔𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼼𖽙𖾏 𖽐𖼎𖽑𖽦𖾑 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼀𖽱𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏．” 13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏！” 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖼯𖽔 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐，𖽃𖽔𖾐 𖼋𖽪 𖽐𖼀𖽳𖾏 𖼊𖽔𖾐，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏 𖼑𖽜 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 ． ” 14 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑， 𖽃𖽔𖾑 𖼾𖽠 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐．𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾏 𖽃𖽡𖾐 𖼼𖽘𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏， 𖽃𖽡𖾐 𖼼𖽘𖾏 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．” 15 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏，𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏． 16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼀𖽜𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖽃𖽔𖾐 𖼋𖽪 𖽐𖼀𖽳𖾏 𖽃𖽡𖾑，𖼑𖽔𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔，𖽐𖼀𖽹𖾏 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 ，𖼊𖽜𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐． 17 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏，𖼎𖽺𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼈𖽘𖾐．

𖼀𖽡𖾐 𖼊𖽳 𖼯𖽜𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪
（𖼊𖽑𖽹𖾏 16：13-19；𖼞𖽑𖽙𖾐 8：27-29）
18 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？” 19 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙．” 20 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？” 𖼀𖽡𖾐 𖼊𖽳 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽙𖾏 ．”
（𖼊𖽑𖽹𖾏 16：20-28；𖼞𖽑𖽙𖾐 8：30-9：1）
21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 22 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼊𖽻𖾏 𖼊𖽪𖾐 𖼗𖽻、 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙．”
𖼀𖽡𖾐𖼊𖽳 𖼯𖽜𖾏𖼽𖽦𖾑𖼺𖽪𖾑𖼟𖽪𖾏𖼮𖽡𖾑𖼊𖽪
23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼾𖽨𖾏， 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼏𖽷𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐． 24 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑． 25 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜，𖼈𖽡𖽝𖾏 𖼛𖽙𖾏 𖼾𖽨𖾏 𖼺𖽘𖾏，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼄𖽔𖾐 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼁𖽻𖾏 𖼺𖽘𖾏， 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑？ 26 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼚𖽙，𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑，𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏． 27 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽑𖽡𖽝𖾏 𖼋𖽔 𖽐𖼷𖽑𖽡𖽝𖾐 𖼺𖽷𖾐，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙．”
