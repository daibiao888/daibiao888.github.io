𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
9
𖼐𖽱𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼖𖽪𖾑 𖼙𖽻 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏.” 2 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖽃𖽔 𖼸𖽷 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖽃𖽡𖾑 𖼋𖽔𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙, 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖽐𖼊𖽠𖾑 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼚𖽪𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽺𖾏 𖼋𖽔𖾏, 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖼋𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 .
3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼦𖽡𖾏 𖼇𖽔𖾑, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽑𖽹𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏. 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙, 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼅𖽔𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼚𖽪𖾐 𖽃𖽡𖾑 𖼋𖽔𖾏. 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽻 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑, 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽜 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽡𖽝𖾑𖽑.” 5 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼐𖽑𖽙𖾑 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼀𖽜𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼙𖽻 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏. 𖼅𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽻𖾑 𖼅𖽳𖾏. 6 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 、 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐, 𖽐𖼎𖽙 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏、 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏, 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼳𖽠𖾑 𖼁𖽱𖾏 𖼞𖽑𖽔𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽑𖽡𖽝𖾑 𖼅𖽔𖾏 𖼺𖽻𖾏 𖽃𖽡𖾑 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽡𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽪𖾐 𖼊𖽔𖾏 𖼊𖽪𖾐 𖼗𖽻 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏. 7 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 , 𖼟𖽙𖾏 𖼊𖽡𖾑 𖼯𖽡𖽳𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽔𖾏. 𖼅𖽡 𖼋𖽠𖾏 𖼅𖽻𖾏 !” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏. 8 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖽃𖽡𖾑, 𖼮𖽔𖾑 𖼷𖽷𖾐 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽡𖽝𖾏 𖼊𖽔𖾐, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼚𖽑𖽪𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼐𖽑𖽡𖾏: “𖽃𖽺𖾐! 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼄𖽔𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼖𖽺𖾑?”
9 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼾𖽡𖾏 𖼑𖽡𖾏 𖼾𖽔𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑, 𖼙𖽻 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼀𖽡𖾐. 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐. 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏.”
11 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 , 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼅𖽔𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼚𖽪𖾐 𖽃𖽡𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼸𖽙𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏.”
