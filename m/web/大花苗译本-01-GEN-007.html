7
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽙 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳. 2 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜, 𖼟𖽡 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼡𖽔𖾏 𖼷𖽷𖾑 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝； 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼟𖽡 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼡𖽔𖾏 𖼷𖽷𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼡𖽔𖾑 𖼑𖽡𖽝； 3 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼡𖽔𖾏 𖼘𖽻𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝, 𖼟𖽙𖾏 𖼷𖽷𖾐 𖼮𖽘𖾑 𖼯𖽡𖽝𖾏, 𖼯𖽡𖽝𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑； 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼑𖽘 𖼗𖽙𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼄𖽑𖽙𖾏, 𖼄𖽔𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼷𖽷𖾏 𖼞𖽑𖽔𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏.” 5 𖼐𖽙 𖼽𖽔𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼋𖽘𖾏.
6 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑, 𖼐𖽙 𖼽𖽔𖾏 𖼀𖽪𖾑 𖼚𖽻𖾏 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏. 7 𖼐𖽙 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐、𖼐𖽱𖾏 𖼐𖽤𖾐、𖼊𖽪𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏, 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐. 8 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼊𖽑𖽡𖽝𖾐 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼐𖽻 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽡𖾑 𖼞𖽘𖾐, 9 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏, 𖼅𖽔𖾏 𖼡𖽔𖾏 𖼷𖽷𖾑 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝, 𖼚𖽔𖾏 𖼐𖽙 𖼽𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽙 𖼽𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐. 10 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐.
