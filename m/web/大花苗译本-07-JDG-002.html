𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼎𖽑𖽻𖾏 𖼮𖽡 𖼮𖽢 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼀𖽙𖾑 𖼮𖽡𖽾𖾑, 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏， 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼏𖽵 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖽃𖽡𖾑． 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼙𖽔 𖽐𖼎𖽙 𖼅𖽡 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏. 2 𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑? 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼅𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼅𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼅𖽡 𖼖𖽪𖾑 𖼸𖽙𖾏.” 4 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑. 5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼀𖽙𖾑 𖼮𖽡𖽾𖾑 ( “𖼀𖽙𖾑 𖼮𖽡𖽾𖾑” 𖼟𖽪𖾏“ 𖼐𖽡𖽝𖾑”). 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽱𖾐 𖼊𖽡𖾑 𖼢𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
