𖼨𖽑𖽹𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏

2 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼨𖽑𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼗𖽻 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 , 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 , 𖼎𖽑𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼺𖽘𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 . 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼳𖽤𖾏 𖼖𖽑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼞𖽑𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼚𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 . 4 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼗𖽻 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 , 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽱𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼮𖽡𖽝𖾏 . 5 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼞𖽱𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 , 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 . 6 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼖𖽑𖽘𖾑 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖽐𖼎𖽙 𖼑𖽨𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼾𖽱 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 . 7 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼞𖽱𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽤𖾑 𖼉𖽺𖾏 𖽐𖼚𖽙, 𖼟𖽪𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼑𖽡𖾏 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 . 8 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖽐𖼊𖽑𖽔𖾑 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽐𖼎𖽙 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏 . 9 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 𖽃𖽡𖾑 𖽃𖽺𖾐 𖽐𖼊𖽡𖾑 , 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼖𖽔𖾏 𖼁𖽙𖾏 𖼋𖽻𖾏 , 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼺𖽷𖾐 . 10 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽐𖼚𖽙 𖼯𖽔 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼊𖽔𖾐 , 𖼟𖽪𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏𖼣𖽹 𖼮𖽡 𖼗𖽙𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 ,𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐 𖼦𖽙 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼮𖽡𖽝𖾏 . 𖼋𖽠𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 .
11 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 , 𖼖𖽔𖾏 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼾𖽨𖾏 , 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 . 12 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 、 𖼳𖽡𖾑 𖼄𖽔𖾏 、 𖼽𖽦𖾑 𖼊𖽪𖾏 𖼊𖽪𖽝𖾏， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐、 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 , 𖼊𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 , 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 , 𖽐𖼊𖽻𖾏 𖽐𖼎𖽔𖾏 、 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼺𖽙𖾑 、 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 ， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏 . 13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖽃𖽡𖾑 ,𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏 、 𖽐𖼊𖽻𖾏 𖽐𖼎𖽔𖾏 、 𖼯𖽡𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼢𖽜𖾏 𖽐𖼎𖽔 , 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 :
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐 ,
𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽷𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 .”
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼅𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 . 14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 .
