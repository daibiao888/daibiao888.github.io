𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡， 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽃𖽔𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽑𖽤𖾑． 2 𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖼙𖽔 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 ，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏，𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏， 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜． 3 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼖𖽑𖽔𖾏 𖼺𖽻𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼑𖽜𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 ，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽻 𖼸𖽙𖾏 ， 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼮𖽡𖾏 𖼙𖽱 𖼙𖽱． 4 𖼑𖽔𖾏 𖼘𖽪！𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 ， 𖼎𖽺𖾏 𖼏𖽻 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼷𖽑𖽔𖾐 𖽐𖼊𖽑𖽱𖾑 𖼈𖽡𖽝𖾏，𖼮𖽔 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼣𖽔𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼚𖽔𖾑 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼊𖽔𖾐， 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼸𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽐𖼞𖽙𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐． 5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏，𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 ．
