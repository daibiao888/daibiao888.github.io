24
𖼑𖽔𖾏 𖼘𖽪! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑, 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼄𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏.
2 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏,
𖼡𖽑𖽜𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼮𖽵𖾑 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏,
𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖽃𖽔 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽺 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼷𖽹𖾏 𖼎𖽻𖾏 𖼗𖽔 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼗𖽔 𖼷𖽹𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖽐𖼮𖽷𖾏 𖼳𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖽐𖼮𖽷𖾏 𖼋𖽠𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏.
3 𖼊𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼞𖽑𖽔𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑, 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼗𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
4 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏 𖼖𖽻𖾏 𖼡𖽑𖽔𖾑 𖼋𖽘𖾏,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼗𖽱𖾏 𖽐𖼚𖽔𖾏 𖼖𖽻𖾏 𖼡𖽑𖽔𖾑 𖼋𖽘𖾏,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖼦𖽡 𖼋𖽙 𖼙𖽠𖾏.
5 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼺𖽘𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑,
𖼗𖽔𖾏 𖼺𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏, 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖾐 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑, 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏.
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼯𖽠 .
7 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏, 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼺𖽘𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏.
8 𖽐𖼊𖽻𖾏 𖽐𖼎𖽔𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼖𖽔𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼖𖽔𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏.
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼸𖽺𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐.
10 𖼙𖽻 𖼟𖽪𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼷𖽑𖽔𖾏 𖼾𖽡𖾏 𖼡𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼮𖽡𖽝𖾏, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏.
11 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽡𖽠𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼞𖽑𖽔𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏.
12 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽔 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼎𖽑𖽔𖾑 𖼨𖽑𖽪𖾑 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
13 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼽𖽔𖾑 𖼺𖽱𖾏 𖼞𖽗𖾐 𖼖𖽗 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼛𖽠𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼺𖽘𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽡𖾏 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼳𖽙𖾏 𖼀𖽡𖾑 𖼯𖽠 .
 
14 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼗𖽙 .
15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏.
16 𖼀𖽡𖾐 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼺𖽻𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏, 𖼐𖽑𖽡𖾏:
“𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳！”
𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏！ 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏！
𖼞𖽪𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！ 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖽂𖽪𖾏 𖽂𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.”
 
17 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！
𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏、 𖼡𖽑𖽙𖾑 𖼘𖽻𖾑𖽑、 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼟𖽡 .

𖼊𖽡𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼋𖽘𖾏.
19 𖼊𖽡𖾐 𖼞𖽑𖽔𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑, 𖼋𖽠 𖽐𖼀𖽑𖽔𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼋𖽘𖾏.
20 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖼐𖽡𖾑 𖼢𖽻 𖼎𖽻𖾏 𖼉𖽺𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼢𖽻 𖼮𖽡𖽠𖾑,
𖼎𖽺𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼅𖽻𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼗𖽙𖾏, 𖼺𖽱𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 ,
𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼸𖽺𖾏 𖼐𖽑𖽤𖾑,

𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
21 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐,
𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏.
22 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱, 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖽐𖼚𖽙 ,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼞𖽱𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖽐𖼚𖽙 ,
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼾𖽱 𖼾𖽔𖾏(“𖼖𖽠𖾑 𖼾𖽱 𖼾𖽔𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖼞𖽻𖾏 𖼮𖽳𖾐”).
23 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
