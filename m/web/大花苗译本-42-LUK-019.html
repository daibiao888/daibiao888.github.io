28 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼎𖽺𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏，𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽠𖾏． 29 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼇𖽺𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼀𖽳 𖼇𖽔 𖼮𖽑𖽡 𖽐𖼎𖽙 𖼀𖽳 𖼊𖽔𖾏 𖼐𖽡， 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼞𖽗𖾐 𖼖𖽗 𖼎𖽻𖾐， 𖼎𖽺𖾏 𖼷𖽷𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏： 30 “ 𖼅𖽡 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼼𖽙𖾏 𖼀𖽡 𖼋𖽡𖾏，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽡𖽝𖾏，𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼙𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽜𖾏 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐 𖼺𖽷𖾐，𖼚𖽠𖾏 𖼐𖽱𖾏 𖼮𖽤𖾐 𖼋𖽔𖾏． 31 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼐𖽱𖾏，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：‘ 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼏𖽻 𖼐𖽱𖾏 ．’ 32 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽷𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑． 33 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽠𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑， 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼚𖽠𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？” 34 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼽𖽔𖾑 𖼏𖽻 𖼐𖽱𖾏．” 35 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼚𖽠𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑， 𖼮𖽤𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 ，𖼎𖽺𖾏 𖼚𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼮𖽑𖽙𖾏 𖼀𖽔𖾏 𖼎𖽻𖾏 𖼦𖽡𖾐 𖼇𖽔𖾑，𖼎𖽺𖾏 𖼯𖽵𖾏 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼯𖽡𖽜𖾏 𖼮𖽡𖽝𖾏． 36 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽠𖾏 𖼾𖽨𖾏 𖼮𖽑𖽙𖾏 𖼊𖽑𖽔𖾑 𖼀𖽔𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑．

38 “𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼮𖽵𖾑 𖽐𖼷𖽷𖾏 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏，
