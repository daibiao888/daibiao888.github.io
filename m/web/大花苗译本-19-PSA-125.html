𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐,
𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙 .
2 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼎𖽑𖽻𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
 
3 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏, 𖼦𖽡𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼳𖽤𖾐 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏.
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽡𖾑 𖼟𖽙𖾏 𖼻𖽻𖾏.
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼚𖽔𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖽐𖼞𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐．
 
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝！
