7
𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼀𖽡 𖼟𖽡 𖼸𖽙 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼐𖽔𖾑, 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻, 𖽃𖽡𖾐 𖼳𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼮𖽔 𖼋𖽺 𖽃𖽡𖾐 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖽝𖾏 𖼊𖽔𖾐， 𖽃𖽔𖾐 𖼳𖽡𖾏 𖼽𖽔𖾏 𖼅𖽙 𖼮𖽔 𖼋𖽺 𖽃𖽡𖾐 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖽝𖾏 𖼋𖽔𖾐.” 2 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼮𖽔𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏, 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑.” 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼑𖽻𖾏.”
𖼽𖽔𖾏 𖼖𖽔 𖼏𖽙 𖼇𖽳𖾑 𖼖𖽪𖾑 𖼾𖽡 𖼏𖽘 𖼀𖽜𖾑
