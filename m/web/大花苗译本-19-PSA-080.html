𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼽𖽙 𖼺𖽳 𖼺𖽱𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻！
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼢𖽳 𖼋𖽠𖾏 𖼮𖽡𖾏 𖼗𖽙𖾏！
2 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 、𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 、𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼎𖽑𖽻𖾏 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽔𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐.
 
3 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 (“𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼎𖽺𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏”),
𖼟𖽙𖾏 𖼟𖽡 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏！
 
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
5 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼞𖽔𖾏 𖼅𖽔 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏,
𖼎𖽺𖾏 𖼮𖽡𖽻𖾐 𖽐𖼮𖽔𖾏 𖼋𖽠𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽻𖾏.
6 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖽙 𖼑𖽜 𖼀𖽡𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼀𖽡𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐.
 
7 𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 ,
𖼟𖽙𖾏 𖼟𖽡 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏！
 
8 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼮𖽑𖽹𖾏 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 ,
𖼖𖽠𖾑 𖼋𖽠𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖼄𖽔𖾐 𖼇𖽘𖾑 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼯𖽡𖽙 𖼮𖽡𖽝𖾏.
9 𖼟𖽡 𖼑𖽜 𖼇𖽘𖾑 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽡𖽝𖾏,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼎𖽺𖾏 𖼗𖽔 𖼟𖽪𖾏 𖼊𖽙𖾐, 𖼗𖽔 𖼀𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾏 𖼋𖽻𖾏.
10 𖼐𖽱𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐,
𖽃𖽔𖾑 𖼯𖽡 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐.
11 𖼐𖽱𖾏 𖼘𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼯𖽡, 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼘𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼄𖽑𖽘𖾐, 𖽐𖼊𖽡𖾑 𖼸𖽙 𖼖𖽪𖾑 𖽃𖽔𖾐 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
12 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼇𖽘𖾑 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼖𖽪𖾑 𖼺𖽙𖾑 𖼉𖽘𖾏,
𖼯𖽔𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 𖼚𖽡𖾏 𖼖𖽺𖾑?
13 𖽐𖼀𖽔𖾏 𖼻𖽻𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼋𖽔𖾏 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼺𖽘𖾏,
𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏.
 
14 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏, 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼐𖽡𖾑，
15 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼯𖽡𖽙
𖽐𖼎𖽙 𖽃𖽔𖾑 𖼯𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑.
16 𖼇𖽘𖾑 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼙𖽔 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑, 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽙𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼟𖽡 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼺𖽘𖾏.
17 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼺𖽘𖾑 𖼺𖽷𖾏,
𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑.
18 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽡𖾏 𖼡𖽜𖾐 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼟𖽡 ,
𖼷𖽑𖽜𖾑 𖼟𖽡 𖼮𖽨𖾏 𖼯𖽡𖽝𖾏 𖼀𖽡𖾐, 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏.
 
19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 ,
𖼟𖽙𖾏 𖼟𖽡 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏！
