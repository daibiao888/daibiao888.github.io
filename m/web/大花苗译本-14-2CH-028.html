𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 . 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 , 2 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 , 𖼎𖽺𖾏 𖼘𖽡𖾑𖽑 𖼀𖽔𖾑 𖼖𖽡 𖼁𖽠𖾏 𖼼𖽘𖾏 , 3 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼐𖽺𖾏 𖼷𖽷𖾐 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 , 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 , 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖽃𖽡𖾑 , 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖼳𖽪𖾐； 4 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 、 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 ,𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾐 𖼄𖽡𖾐 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 .
𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑

16 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 , 17 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 , 𖼦𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼙𖽠𖾏 . 18 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼳𖽪𖾏 𖼮𖽡𖽝𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼖𖽪𖾑 𖼙𖽻 , 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼀𖽳 · 𖼳𖽷𖾏 𖼄𖽳 、𖼽𖽔𖾏 𖼽𖽔𖾐 𖼖𖽪𖽝 、 𖼮𖽡𖾑 𖼊𖽡𖾐 𖼘𖽙 、 𖼺𖽙𖾑 𖼞𖽙𖾑 𖽐𖼎𖽙 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼺𖽙𖾑 𖼞𖽙𖾑 、 𖼊𖽑𖽡 𖼐𖽔 𖽐𖼎𖽙 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼊𖽑𖽡 𖼐𖽔 、 𖼮𖽡 𖼺𖽙𖾐 𖽐𖼎𖽙 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼮𖽡 𖼺𖽙𖾐 , 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 . 19 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽃𖽔𖾏 , 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 , 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾏 𖼢𖽜𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼺𖽘𖾏 . 20 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽡 𖼞𖽳 𖼖𖽔𖾑 · 𖼀𖽑𖽡𖾐 𖼐𖽡 𖼺𖽳 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 , 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼐𖽱𖾏 . 21 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼗𖽙𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼞𖽑𖽻𖾏 𖼋𖽪 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼋𖽘𖾏.
𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
22 𖼷𖽷𖾑 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 23 𖼐𖽱𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 , 𖼐𖽑𖽡𖾏 : “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽻𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 .” 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽃𖽡𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼺𖽘𖾏 . 24 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼄𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 , 𖽐𖼊𖽻𖾏 𖽐𖼮𖽻 𖼺𖽘𖾏 , 𖼎𖽺𖾏 𖼡𖽙𖾏 𖼮𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 . 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽙𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽔𖾑 𖼏𖽘 , 25 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 , 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 , 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 .
