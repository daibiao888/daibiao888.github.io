𖼮𖽵𖾑 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼘𖽪！
𖼞𖽪𖾑 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽡 .
2 𖼑𖽔𖾏 𖼘𖽪！ 𖼡𖽑𖽜𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼑𖽔𖾏 𖼮𖽵𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼑𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼀𖽡𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼀𖽡𖾐.
 
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼀𖽡𖾐, 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼀𖽡𖾐！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼸𖽙 𖼡𖽘𖾐.
4 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏、
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼸𖽙 𖼡𖽘𖾐.
