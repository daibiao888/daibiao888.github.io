𖼗𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐
22
𖼻𖽻𖾏 𖽐𖼷𖽷𖾏 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐.
2 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼦𖽡𖾐 𖽐𖼮𖽷𖾐,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏.
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼁𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼯𖽡𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼙𖽠𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼏𖽘 𖼅𖽻𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
4 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐 𖼢𖽜 𖽐𖼎𖽙 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼯𖽡𖾏 𖼊𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾐、𖽐𖼚𖽻 𖼮𖽡𖾏、𖼯𖽡𖽝𖾏 𖼳𖽦𖾑.
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 , 𖼅𖽔𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼸𖽙𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼟𖽙𖾏 𖼚𖽡𖾐.
6 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼅𖽻𖾏,
𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼗𖽻𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐.
7 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽷𖾏 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼮𖽙𖾏 𖽐𖼮𖽷𖾏 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑.
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼐𖽱𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏.
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏.
10 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼺𖽘𖾏,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖽐𖼎𖽙 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏.
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼅𖽳𖾏,
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔．
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 ,
𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 .
13 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽜 𖼐𖽑𖽡𖾏 : “ 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼅𖽔𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼏𖽻 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 .”
14 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽙𖾐,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼙𖽠𖾏.
15 𖼑𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾏 𖼚𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏,
𖼡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼸𖽠𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼑𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼟𖽪𖾏 𖼡𖽔𖾏 𖼚𖽡𖾐 𖼙𖽠𖾏.
16 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼯𖽵𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼟𖽙𖾏 𖼾𖽨𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼺𖽘𖾏.
17 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼗𖽪𖾏,
𖽐𖼊𖽱𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽳．
18 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏 ,
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼋𖽙𖾏 𖼏𖽔𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.
19 𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼟𖽡,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
20 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼺𖽷𖾐 𖼖𖽺𖾑?
21 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 ,
𖼟𖽙𖾏 𖼟𖽡 𖼻𖽻𖾏 𖼄𖽔𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼟𖽡 𖼋𖽔𖾏 𖽃𖽡𖾑.
 
22 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏, 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽻𖾏 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼸𖽹𖾏 𖼯𖽵𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐.
23 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼦𖽔𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑.
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼚𖽪𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 .
25 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼟𖽡 𖼾𖽻 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏,
