（𖼊𖽑𖽹𖾏 21：33-46；𖼖𖽪𖾏 20：9-19）
12
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏， 𖼎𖽺𖾏 𖼺𖽙𖾑 𖽃𖽡𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼉𖽘𖾏， 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽐𖼮𖽡𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼮𖽡𖽠𖾑， 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑， 𖼡𖽑𖽠𖾑 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽘𖾏 𖽃𖽔𖾏， 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼗𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽠𖾏． 2 𖼸𖽙 𖼯𖽡𖽹𖾏， 𖼐𖽱𖾏 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑 𖼚𖽔𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽘𖾏， 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽻𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐． 3 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽘𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏， 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏， 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏． 4 𖼉𖽘𖾏 𖼮𖽵𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 ． 5 𖼉𖽘𖾏 𖼮𖽵𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼅𖽻𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏． 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐， 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽻𖾏 𖼺𖽘𖾏， 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏． 6 𖼉𖽘𖾏 𖼮𖽵𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏， 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖽐𖼊𖽱𖾏 𖼐𖽑𖽡𖾏：‘ 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐．’ 7 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽘𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：‘ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏， 𖼋𖽔𖾏 𖼘𖽪！ 𖼀𖽡𖾐 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏， 𖽐𖼎𖽪𖾏 𖼡𖽜𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼁𖽡𖽝．’ 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏， 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏， 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼢𖽻𖾏 𖼉𖽘𖾏 𖼺𖽘𖾏． 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼖𖽪𖾑 𖼉𖽘𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？ 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． 10 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼟𖽡𖽠 𖼋𖽡𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑：
‘ 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏，
𖼎𖽺𖾏 𖽃𖽔𖾏 𖼞𖽪𖾐 𖽐𖼞𖽔𖾐 𖼊𖽠𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼮𖽡𖽙𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑．
11 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖽃𖽔𖾏，
𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑．’ ”
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼑𖽜 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏．
𖼊𖽪𖾏 𖼳𖽳𖾏 𖼎𖽻𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐
（𖼊𖽑𖽹𖾏 22：15-22；𖼖𖽪𖾏 20：19-26）
13 𖽃𖽔𖾑 𖼡𖽜𖾐 ， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼖𖽡 𖽃𖽡𖾐 𖼸𖽻 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼡𖽘𖾐 𖼟𖽡𖾏 𖼦𖽹𖾏 𖼐𖽱𖾏． 14 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏， 𖼀𖽡𖾐 𖼀𖽻𖾐 𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝， 𖼯𖽡𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡 𖼋𖽳， 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽑𖽪𖾏， 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝. 𖼊𖽪𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐 𖼋𖽡𖽝𖾏 𖼊𖽪𖾏 𖼢𖽻 𖼖𖽔𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼢𖽻 ？ 15 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼊𖽪𖾐？”𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖼮𖽑𖽵𖾑， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼞𖽪𖾑，𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏．” 16 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽔𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼼𖽘𖾏， 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼷𖽷𖾏？”𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽪𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐．” 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽹𖾑 𖼺𖽔𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．” 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏．
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙
（𖼊𖽑𖽹𖾏 22：23-33；𖼖𖽪𖾏 20：27-40）
18 𖼺𖽔𖾐 𖼊𖽪𖾑 𖼞𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽑𖽡𖾏， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏： 19 “𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏， 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼺𖽻𖾏 𖽐𖼊𖽠𖾑 𖼎𖽻𖾏 𖼀𖽡𖾐， 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼋𖽔 𖼙𖽠𖾏，𖼷𖽷𖾐 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼮𖽡𖽝𖾏， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏， 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼷𖽙𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖽃𖽔𖾏 𖼐𖽤𖾐， 𖼑𖽜 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏． 20 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏， 𖼷𖽷𖾑 𖼘𖽑𖽙𖾐 𖼷𖽙𖾏 𖼐𖽤𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼎𖽺𖾏 𖼋𖽔 𖼙𖽠𖾏． 21 𖼷𖽷𖾑 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼷𖽙𖾏 𖼐𖽱𖾏， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼋𖽔 𖼙𖽠𖾏， 𖼷𖽷𖾑 𖼏𖽷𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏． 22 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼮𖽡𖽝𖾏 ，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼖𖽔𖾑 𖼁𖽙𖾏 𖼖𖽔𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏． 23 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽤𖾐 ？ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼖𖽔𖾏 𖼙𖽔 𖼷𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐．” 24 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑， 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏． 𖼅𖽡 𖼎𖽺𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑． 25 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼷𖽙𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼡𖽔𖾏，𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐． 26 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏， 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼟𖽡𖽠 𖼄𖽙𖾑 𖼳𖽡𖾑 《𖽐𖼊𖽠𖾑 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 》𖼘𖽪𖾏？ 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏：‘ 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ’ 𖼘𖽪𖾏？ 27 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 𖼅𖽡 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼋𖽘．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐
（𖼊𖽑𖽹𖾏 22：34-40；𖼖𖽪𖾏 10：25-28）
28 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼐𖽑𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏， 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏， 𖼋𖽔𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼼𖽘𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼟𖽪𖾏 𖼼𖽘𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑？” 29 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼼𖽘𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 ，𖼐𖽑𖽡𖾏：‘ 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 ，𖼮𖽵𖾑 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐． 30 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜、𖽃𖽔𖾑 𖼛𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜、 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜、 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜， 𖽐𖼮𖽑𖽠𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．’ 31 𖽃𖽔𖾐 𖼼𖽘𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖽃𖽡𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏：‘ 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼾𖽨𖾏．’ 𖼋𖽠𖾏 𖼺𖽘𖾏 𖽃𖽔𖾐 𖼼𖽘𖾐 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽡𖾑 𖼋𖽘．” 32 𖼷𖽷𖾑 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“ 𖼻𖽻𖾏 𖼯𖽡𖽙𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 ， 𖼟𖽡 𖼐𖽑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐， 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽙， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 33 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜、 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜、 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏， 𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼺𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼾𖽨𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏．” 34 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼡𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼦𖽡𖾐 𖼚𖽡𖾐 𖼋𖽘．” 𖼋𖽳 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼺𖽡𖽝𖾐 𖼘𖽑𖽙𖾐 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽘𖾏．
𖼮𖽡𖾑 𖼊𖽪 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏
（𖼊𖽑𖽹𖾏 22：41-46；𖼖𖽪𖾏 20：41-44）
35 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏：𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 ？ 36 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼅𖽔𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼡𖽑𖽜𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
‘ 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏，
𖼟𖽡 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏，
