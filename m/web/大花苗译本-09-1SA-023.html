𖼺𖽻𖾐 𖼘𖽙 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏
7 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏: “𖼊𖽔𖾏 𖽂𖽺𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼽𖽡𖾑 𖼖𖽔𖾑.” 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼅𖽔𖾏 𖼎𖽪𖾑, 𖼾𖽨𖾏 𖼡𖽙𖾏 𖼾𖽨𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼮𖽡𖽝𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽘𖾏. 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽻𖾐 𖼘𖽙 𖼦𖽡 𖼸𖽱 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼽𖽔𖾑 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼮𖽡𖾑 𖼽𖽡𖾑 𖼖𖽔𖾑 𖼙𖽻， 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏. 9 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼦𖽺𖾏 𖼐𖽱𖾏, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽑𖽡𖽠𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖼯𖽡𖽙𖾏 𖼋𖽔𖾐.”
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼇𖽳𖾑 𖼮𖽡𖾑 𖼽𖽡𖾑 𖼖𖽔𖾑
