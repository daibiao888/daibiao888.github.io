14
𖼅𖽡 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼷𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽑𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽡𖽝𖾑𖽑 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼺𖽘𖾏； 2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼺𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 , 𖼺𖽹𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 .
𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼎𖽙 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏
（𖼖𖽡𖾏 11：1 - 21）
3 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏（𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼋𖽡𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 ”） 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 . 4 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼢𖽻， 𖼟𖽪𖾏： 𖼑𖽨𖾏、𖼾𖽘𖾏、𖼮𖽑𖽷𖾏 、 5 𖼷𖽹𖾏、𖼇𖽔𖾐 𖼷𖽑𖽹𖾏、𖼦𖽡𖾑 𖽐𖼞𖽻𖾏、𖼮𖽑𖽷𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏、𖼖𖽡𖾑 𖼮𖽢𖾐、𖼖𖽡𖾏 𖼞𖽻𖾏、𖼳𖽹𖾏、 6 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼎𖽺𖾏 𖽐𖼊𖽡𖾏 𖼼𖽙𖾏 𖽃𖽡𖾑 , 𖼅𖽡 𖼑𖽻𖾏 𖼢𖽻 𖼯𖽡𖽙𖾏 . 7 𖼈𖽡𖽝𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼼𖽙𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼇𖽺𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼑𖽻𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼘𖽙 𖼊𖽑𖽙𖾐、𖼖𖽔𖾑、𖼖𖽔𖾑 𖼡𖽔𖾏 𖼷𖽔𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼼𖽙𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏, 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏； 8 𖽐𖼀𖽔𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖽐𖼊𖽡𖾏 𖼼𖽙𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼋𖽡𖾏 𖼯𖽦𖾏 𖼐𖽡𖾑 𖽐𖼡𖽹𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼗𖽠𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽔𖾏 𖼺𖽔𖾏 .
9 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖼑𖽻𖾏 𖼢𖽻 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 : 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼷𖽷𖾏 𖼖𖽔𖾏 𖼑𖽻𖾏 𖼢𖽻 𖼯𖽡𖽙𖾏； 10 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼷𖽷𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 , 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏.
11 𖼐𖽻 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼅𖽡 𖼑𖽻𖾏 𖼢𖽻 𖼯𖽡𖽙𖾏 . 12 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼅𖽡 𖼑𖽻𖾏 𖽃𖽡𖾑，𖼟𖽪𖾏： 𖼚𖽘𖾑 𖼄𖽔𖾐 𖽐𖼀𖽔𖾏、𖼚𖽘𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼚𖽡𖾑、𖼚𖽘𖾑 𖼖𖽡𖽝𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏、 13 𖼙𖽡𖽝𖾏、𖼙𖽡𖽝𖾏 𖼊𖽻𖾏 𖼮𖽢𖾏、𖼙𖽡𖽝𖾏 𖽐𖼷𖽔𖾐 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽡𖾑， 14 𖼖𖽡𖾑 𖽃𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽡𖾑， 15 𖼐𖽻 𖼞𖽡𖾏 𖼎𖽻𖾑 、 𖼽𖽱𖾏 𖽂𖽙、𖼚𖽘𖾑 𖼑𖽻𖾏 𖽐𖼀𖽳𖾏 、 𖼚𖽘𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽡𖾑， 16 𖼊𖽔𖾑 𖼞𖽙𖾏 𖼞𖽙𖾏、𖼘𖽻𖾏 𖼘𖽑𖽱𖾏、𖼮𖽑𖽙𖾑 𖼮𖽑𖽙𖾐、 17 𖼤𖽪𖾏 𖼑𖽻𖾏 𖽐𖼀𖽳𖾏、𖼚𖽘𖾑 𖼚𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏、𖽃𖽙𖾏 𖼑𖽻𖾏 𖽐𖼀𖽳𖾏、 18 𖼁𖽔𖾏、𖼁𖽔𖾏 𖼑𖽻𖾏 𖽐𖼀𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽡𖾑 , 𖼐𖽻 𖽐𖼮𖽡𖽠𖾏 𖽐𖼚𖽡𖾏 𖽐𖼎𖽙 𖼑𖽘𖾏 𖽐𖼀𖽔𖾑 .
19 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼈𖽡𖽝𖾏 𖽐𖼞𖽘 𖼊𖽜𖾑 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼅𖽡 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻𖾏 . 20 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽡𖾑 , 𖼅𖽡 𖼑𖽻𖾏 𖼢𖽻 𖼯𖽡𖽙𖾏 .
