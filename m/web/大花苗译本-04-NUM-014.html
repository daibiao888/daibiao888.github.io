𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏； 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 . 2 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼎𖽻𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼷𖽙𖾑 𖼋𖽔 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼐𖽡𖾑 . 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 , 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼖𖽺𖾑 ? 𖼀𖽡𖾐 𖼐𖽤𖾐 𖽐𖼎𖽙 𖼣𖽔𖾑 𖼽𖽻𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏, 𖼀𖽡𖾐 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼘𖽪 ?” 4 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼺𖽹𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 , 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼦𖽡𖾑!”
5 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 . 6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼐𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼇𖽪 𖼐𖽡 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼖𖽳 , 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏， 7 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽑𖽡𖾏 : “𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 . 8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼙𖽔 𖼀𖽡𖾐 , 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 , 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐，𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼄𖽡𖾏 𖽐𖼎𖽙 𖼄𖽪𖾑 . 9 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼷𖽜𖾑 𖼇𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 . 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏 . 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 .” 10 𖼈𖽡𖽝𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽑𖽡𖾏 : “𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼺𖽘𖾏 .” 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐.

39 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 . 40 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼀𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼋𖽘𖾏 , 𖼀𖽡𖾐 𖼯𖽪𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼏𖽵 𖽃𖽡𖾑 .” 41 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼷𖽜𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼺𖽘𖾏 𖼖𖽺𖾑 ? 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼎𖽳𖾏 𖼎𖽪𖾑 𖼋𖽘𖾏 . 42 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 . 43 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐， 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼐𖽡𖽝𖾑𖽑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼺𖽷𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 .” 44 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼖𖽪𖾑 𖼾𖽡 . 45 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑 , 𖽐𖼡𖽜𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖽐𖼊𖽻𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽡𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽔𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼦𖽙 𖽃𖽳𖾐 𖼄𖽔𖾐 .
