5 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼅𖽔𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼋𖽔𖾏，𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ． 6 𖼊𖽪𖾑 𖼳𖽘𖾐 𖼐𖽡𖾑 𖽃𖽡𖾐 𖽐𖼎𖽙𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏， 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽡𖾐 𖼗𖽪𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏， 𖼎𖽺𖾏 𖼏𖽘𖾏 𖼀𖽡𖾏 𖼮𖽨𖾑． 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏：“ 𖼑𖽔𖾏 𖼘𖽪！ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪？ 8 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼀𖽡𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼀𖽡𖾐 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽡𖾐 𖼗𖽪𖾏 𖼖𖽺𖾑？ 9 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼀𖽑𖽔𖾏 𖼊𖽑𖽡 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 、𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼊𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 、𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼽𖽡𖾐 𖼖𖽗 𖼊𖽜𖾑 𖼐𖽜𖾐 、𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼄𖽡𖾐 𖼺𖽙𖾐 𖼀𖽙𖾑 𖼊𖽔𖾏 𖼄𖽡𖾐 、𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 ，𖽐𖼎𖽙 𖼮𖽢𖾑 𖼀𖽑𖽔𖾏 𖼊𖽙𖾑 𖼮𖽢𖾑 、𖽐𖼎𖽙 𖼀𖽞𖾐 𖼊𖽪𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼳𖽡𖾏 𖼽𖽔𖾏 、 10 𖽐𖼎𖽙 𖼇𖽪 𖼖𖽱𖾐 𖼮𖽢𖾑 、𖽐𖼎𖽙 𖼀𖽑𖽘 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 、𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼖𖽡𖾏 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾐 𖼖𖽡𖾏 𖼐𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 、𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼘𖽙 𖼄𖽔𖾐 𖼙𖽻， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼽𖽨𖾏 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏， 11 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼞𖽑𖽳 𖼖𖽡𖾐 𖼊𖽑𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼖𖽔𖾑 𖼀𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼁𖽜 𖼐𖽑𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽙𖾏 𖼀𖽡𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽡𖾐 𖼗𖽪𖾏，𖼦𖽔𖾑 𖼗𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐．” 12 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏，𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼑𖽪 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼐𖽑𖽡𖾏： “ 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑？” 13 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐（𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “ 𖼮𖽡𖽠𖾑 𖼄𖽪𖾑 ”） 𖼢𖽻 𖼮𖽡𖽠𖾑 𖼋𖽘𖾏．”
𖼀𖽡𖾐 𖼊𖽳 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏
14 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏，𖼮𖽘𖾏 𖼺𖽘𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 ． 15 𖼅𖽡 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼢𖽻 𖼮𖽡𖽠𖾑 𖼋𖽘𖾏，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼢𖽻 𖼮𖽡𖽠𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽘𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼯𖽢𖾏 𖼋𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼊𖽔𖾐． 16 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽙 𖽃𖽳𖾐 𖼗𖽪𖾏：
17 ‘ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏：

21 𖼸𖽙 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼨𖽑𖽪𖾏 𖼮𖽵𖾑 𖽐𖼷𖽷𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏 ．’
22 “ 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑：𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏，𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑，𖼄𖽔𖾐 𖼐𖽱𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽹𖾏，𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼯𖽡𖽙． 23 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼎𖽺𖾏 𖼊𖽡𖾏 𖼮𖽡𖽝𖾏，𖼊𖽻𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼅𖽡 𖼎𖽺𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏， 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼮𖽦𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑𖽑 𖼖𖽡𖾐 ，𖼊𖽔𖾏 𖼺𖽘𖾏． 24 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼋𖽔 𖼑𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼚𖽠𖾏 𖼺𖽘𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽔 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏． 25 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ：
‘𖼞𖽪𖾑 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏， 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙．
26 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑，𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏，
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐．
27 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼗𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏，
𖼟𖽡 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼗𖽱𖾏 𖼺𖽘𖾏．
28 𖼟𖽡 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼋𖽘𖾏，
𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 ．’
29 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡，𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼑𖽪𖾏，𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼟𖽡𖾏 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼋𖽡𖽝，𖼐𖽱𖾏 𖼋𖽔 𖼗𖽠𖾏 𖼋𖽘𖾏，𖼊𖽱𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖼸𖽙 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼺𖽷𖾐． 30 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐，𖼐𖽱𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼏𖽵 𖼗𖽪𖾏，𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐， 𖼺𖽹𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑． 31 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼺𖽪𖾐 𖼑𖽔𖾏 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑，𖼦𖽔𖾑 𖼗𖽔𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼐𖽑𖽡𖾏：
‘𖼐𖽱𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 ，
𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼗𖽱𖾏 𖼺𖽘𖾏．’
32 𖼷𖽷𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖾑 ， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼋𖽘𖾏 ，𖼀𖽡𖾐 𖼑𖽜 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏． 33 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏， 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼏𖽵 𖼊𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼙𖽔 𖼁𖽙𖾏 𖼙𖽔 𖼐𖽑𖽙𖾑 𖽃𖽡𖾑，𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏． 34 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 ，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏：
‘𖼮𖽵𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：
𖼟𖽡 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏，
35 𖼋𖽙𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐．’
