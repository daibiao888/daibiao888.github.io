𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫
𖽃𖽡𖾐 𖼸𖽙𖾐 𖽐𖼞𖽻𖾏,𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏.
（𖽐𖼞𖽻𖾏 57：7 - 11；60：5 - 12）
108
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼞𖽪𖾑 𖼖𖽡𖽾( “𖼖𖽡𖽾”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼚𖽻 𖼮𖽡𖾏”) 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔！
2 𖼢𖽜𖾏 𖽐𖼎𖽔 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽷𖾏 𖼳𖽠𖾑！
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼯𖽷𖾏 𖼮𖽡𖾏 𖼡𖽘𖾐 𖽐𖼊𖽪𖾐！
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡 ,
𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡！
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 .
 
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜！
6 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼊𖽡𖾐 𖼀𖽡𖾐, 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏.
 
7 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼇𖽺𖾐 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼺𖽘𖾏,
𖽐𖼮𖽔𖾏 𖼺𖽪𖾑 𖼞𖽙 𖼖𖽪𖾑 𖼦𖽘𖾑 .
8 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝 ,
𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝 ,
𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽻𖾏 𖼋𖽻𖾏 ,
𖼽𖽨 𖼊𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.
9 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼞𖽑𖽻𖾏 𖼎𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏.”
 
10 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼏𖽔𖾏?
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑?
11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼾𖽙 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼀𖽡𖾐 𖼺𖽘𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼸𖽻𖾏 𖼏𖽙 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏 𖼋𖽘𖾏 𖼖𖽔𖾏?
12 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐 𖽐𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽘𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 .
13 𖼀𖽡𖾐 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼀𖽡𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏.
