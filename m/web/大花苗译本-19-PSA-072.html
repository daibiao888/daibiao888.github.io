5 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼐𖽡𖽙𖾐 𖼺𖽷𖾐, 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼐𖽡𖽙𖾐 𖼺𖽷𖾐,

6 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼘𖽻𖾑𖽑 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏, 𖼺𖽱𖾏 𖼑𖽘 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼤𖽜𖾏 𖽐𖼡𖽜𖾐 𖼺𖽘𖾏,

7 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏,

8 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽔𖾑 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼐𖽡𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽡𖾑,
𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖽃𖽔𖾐 𖼟𖽱𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼺𖽻𖾏.
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑,
𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼾𖽺𖾏 𖽃𖽔 𖼗𖽔𖾏.
10 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 𖽐𖼎𖽙 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽔𖾏,
𖼳𖽷𖾏 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼀𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽘𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏.
11 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏,

12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,
