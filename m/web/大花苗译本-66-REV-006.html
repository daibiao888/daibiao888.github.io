𖼡𖽑𖽜𖾐 𖼚𖽻𖾏 𖼗𖽪𖾏 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏
6
𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼡𖽑𖽜𖾐 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖼖𖽪𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 ，𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 ，𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼺𖽙𖾐 𖽐𖼊𖽑𖽡𖾏 ，𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼋𖽔𖾏！” 2 𖼞𖽪𖾑 𖼑𖽔𖾏 ，𖼁𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼚𖽠𖾐，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼐𖽑𖽜𖾑， 𖼅𖽔𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏，𖼐𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼳𖽠𖾏．
3 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖽃𖽡𖾑，𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽡𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 ： “𖼟𖽡 𖼋𖽔𖾏！” 4 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖽃𖽡𖾑，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏， 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏，𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏．
5 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖼷𖽷𖾐 𖼖𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖽃𖽡𖾑，𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼷𖽷𖾐 𖼋𖽪 𖽃𖽡𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 ： “𖼟𖽡 𖼋𖽔𖾏！” 𖼞𖽪𖾑 𖼑𖽔𖾏 ， 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼚𖽪𖾐，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼸𖽱𖾏 𖼞𖽡𖾏． 6 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，𖼺𖽱𖾏 𖼺𖽱𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽑𖽡𖾏：“𖽃𖽡𖾐 𖼳𖽜𖾐 𖼳𖽻𖾐 𖽐𖼡𖽜𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼑𖽡𖽝𖾏（𖽃𖽡𖾐 𖼛𖽡𖾐 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙），𖼷𖽷𖾐 𖼳𖽜𖾐 𖼅𖽙 𖽐𖼡𖽜𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼑𖽡𖽝𖾏，𖼛𖽙𖾏 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼞𖽡𖾏．”
7 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖼚𖽻𖾐 𖼖𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖽃𖽡𖾑，𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼚𖽻𖾐 𖼋𖽪 𖽃𖽡𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼋𖽔𖾏！” 8 𖼞𖽪𖾑 𖼑𖽔𖾏 ，𖼁𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼀𖽡𖾏 𖼮𖽑𖽻𖾏，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖽃𖽡𖾑， 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 “𖼛𖽙𖾏 𖼋𖽔 ”，𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐； 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏，𖼞𖽳𖾑 𖼯𖽵𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼚𖽻𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 ， 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏、𖼮𖽑𖽹𖾐 𖼡𖽑𖽜𖾏 、 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼞𖽡𖾏 、 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏， 𖼅𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．
9 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖼀𖽜𖾐 𖼖𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖽃𖽡𖾑，𖼞𖽪𖾑 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏，𖼑𖽜 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐， 10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼘𖽪𖾏！𖼟𖽡 𖼦𖽡𖾐 𖼨𖽑𖽢𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼾𖽔𖾏，𖼦𖽡𖾐 𖼑𖽜 𖼀𖽡𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼀𖽡𖾐 𖽐𖼮𖽑𖽘𖾑，𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑？” 11 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 ； 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼽𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽙𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼺𖽷𖾐，𖼋𖽙𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼊𖽔𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼺𖽘𖾏．
