𖼐𖽔 𖼦𖽪𖽾 𖽐𖼊𖽠𖾑
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏
1
𖼐𖽑𖽡𖾏 𖼸𖽙 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐, 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼽𖽡𖾑 𖼖𖽳 𖼞𖽙𖾑 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽔 𖼦𖽪𖽾 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏.
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏、 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼼𖽷𖾏 𖼁𖽻𖾏、 𖼅𖽔𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 .
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐,
𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏．
 
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖽐𖼎𖽙 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑 𖼋𖽔𖾏,
𖼚𖽻𖾏 𖽃𖽻𖾐 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼡𖽘𖾐 𖼋𖽠𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 .
4 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏,
𖼐𖽱𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼺𖽘𖾏．
𖼀𖽔𖾑 𖼳𖽔𖾑 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼄𖽡 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡𖾏,
𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼑𖽡𖾏 𖼁𖽘𖾏 𖽐𖼡𖽜𖾐 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏.
5 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼑𖽜 𖼐𖽱𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 ,
𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼖𖽔𖾏 𖼾𖽘𖾏 𖼙𖽠𖾏．
𖼊𖽡𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.
6 𖼐𖽱𖾏 𖽂𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼳𖽠𖾑 𖼋𖽻𖾏 𖼖𖽺𖾑?
𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼋𖽻𖾏 𖼘𖽪?
𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼘𖽡𖾑𖽑 𖽐𖼮𖽢𖾑𖽑,
𖼈𖽳𖾐 𖼗𖽘𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼋𖽠 𖽐𖼀𖽑𖽔𖾐 𖼺𖽘𖾏.
 
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏,
𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑.
8 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽻𖾐 𖼷𖽑𖽪𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼎𖽺𖾏 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏.
9 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪 ！ 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼊𖽑𖽡𖽝𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽺𖾑?
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼅𖽡 𖼖𖽻𖾐 𖼷𖽑𖽪𖾐 𖼺𖽘𖾏, 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽠𖾑.
10 𖼅𖽡 𖼺𖽱𖾏 𖼞𖽑𖽻𖾐 𖼟𖽡𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏,
𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼢𖽻 𖼺𖽘𖾏 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖽃𖽔𖾏 𖽐𖼊𖽻𖾏 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑， 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏.
 
11 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼋𖽔𖾏,
𖽐𖼊𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 , 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼐𖽱𖾏 𖼸𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽺𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼈𖽡𖽝𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏, 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏.
𖼽𖽨 𖼊𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼈𖽡𖽝𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼋𖽘 .
13 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼟𖽡 𖼊𖽪𖾑 𖼖𖽡𖾐 𖼡𖽔𖾐,
𖽐𖼀𖽱𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏.”
 
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼋𖽠𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼋𖽘𖾏, 𖼡𖽑𖽳𖾏 𖼎𖽑𖽻𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖽐𖼷𖽷𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾏 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 𖼡𖽜𖾐,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼞𖽔𖾏 𖼁𖽠 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖽐𖼎𖽙 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼘𖽡𖾑𖽑,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏, 𖼑𖽜 𖼟𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼊𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏.”
15 𖼑𖽔𖾏 𖼘𖽪! 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼻𖽻𖾏、 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼐𖽑𖽡𖾏:
“𖼽𖽨 𖼊𖽔𖾏 𖼘𖽪! 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼻𖽙𖾑 𖼟𖽡 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐,
𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 !
𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏,
𖼐𖽱𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏.”
