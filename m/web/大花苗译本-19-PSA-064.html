2 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏,

4 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼀𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,

5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾐 𖼙𖽳𖾐 𖼦𖽡𖾏 𖼷𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,

7 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼐𖽑𖽜𖾑 𖼀𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏,
