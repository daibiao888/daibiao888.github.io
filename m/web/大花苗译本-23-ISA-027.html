𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽻𖾏 𖼮𖽨𖾏
27
𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼟𖽪𖾏 𖼼𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼨𖽑𖽢𖾐 𖼖𖽡𖾏 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾐 ( “ 𖼖𖽡𖾏 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾐 ” 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐), 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽘 𖼀𖽜𖾑 𖼇𖽺𖾏 𖽃𖽡𖾑, 𖼨𖽑𖽢𖾐 𖼖𖽡𖾏 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾐, 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼞𖽑𖽡𖾐 𖼖𖽡𖾏 𖽐𖼞𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼊𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽪𖾑 𖼼𖽘𖾏.
2 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖼅𖽡 𖼽𖽔𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼐𖽑𖽡𖾏:
3 “𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼯𖽡𖽙𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽐𖼊𖽙𖾑 𖽃𖽻𖾐,
𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑,
𖽐𖼮𖽑𖽹𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼟𖽡𖾏.
4 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐．
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼁𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏,
𖼞𖽪𖾑 𖼮𖽙𖾏 𖼚𖽔𖾐 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏,
𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏.
5 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽔𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑, 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏 ,
𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏,
𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏.”
 
6 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼽𖽔𖾑 𖽐𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼟𖽻𖾏 𖼋𖽠𖾏 𖼁𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐.
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
7 𖼮𖽵𖾑 𖽐𖼊𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 , 𖼋𖽡𖽝𖾏 𖼺𖽱𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼘𖽪?
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏,
𖼋𖽡𖽝𖾏 𖼺𖽱𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼘𖽪?
8 𖼟𖽡 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏,
𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼮𖽢𖾏,
𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏.
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏,
𖼐𖽱𖾏 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑,
𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼇𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼯𖽷𖾏 𖼺𖽘𖾏,
𖼟𖽙𖾏 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽘𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏.
10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 ,
𖼺𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼑𖽻𖾏 𖽐𖼡𖽜𖾐, 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖼑𖽻𖾏 𖼋𖽘 𖽐𖼮𖽢𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽡 𖽐𖼊𖽻𖾏.
11 𖽃𖽔𖾑 𖼯𖽡 𖽐𖼊𖽻𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 , 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏,
𖽃𖽔 𖼁𖽙𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼯𖽡𖽙𖾏 𖼅𖽻 𖼚𖽻𖾑．
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼊𖽡𖽝𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑
12 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！ 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏𖼽𖽨𖾏 𖼇𖽔 𖼖𖽔𖾑 𖼊𖽡𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔𖾐 𖼟𖽱𖾐, 𖼞𖽑𖽡𖽠𖾏 𖼅𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 , 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐. 13 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼞𖽡𖾑 𖼊𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼳𖽪 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 , 𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼋𖽔𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
