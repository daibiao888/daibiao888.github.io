𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼐𖽤𖾐
（𖼊𖽑𖽹𖾏 19：1-12；𖼖𖽪𖾏 16：18）
10
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼳𖽠𖾑 𖼅𖽻𖾏， 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏． 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾑 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏． 2 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽻𖾏 𖼐𖽤𖾐 𖼋𖽡𖽝𖾏 𖽐𖼎𖽻𖾏 𖼢𖽻 𖼖𖽔𖾏 𖽐𖼎𖽻𖾏 𖼦𖽡𖾐 𖼢𖽻？” 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖼖𖽡𖾏 𖼯𖽤𖾏 ？” 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼯𖽪𖾏 𖼀𖽡𖾐 𖼺𖽻𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼐𖽤𖾐， 𖼎𖽺𖾏 𖽐𖼎𖽻𖾏 𖼐𖽤𖾐 𖼋𖽘．” 5 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏， 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼺𖽻𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼅𖽡 ． 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼷𖽷𖾏 𖽃𖽔 𖼁𖽙𖾏． 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼺𖽘𖾏， 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏， 8 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼊𖽔𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 ， 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼋𖽘𖾏． 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼺𖽘𖾏．” 10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 ，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑． 11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐， 𖼺𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 ， 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐． 12 𖼐𖽤𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼡𖽔𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐， 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼋𖽘𖾏．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽜 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
（𖼊𖽑𖽹𖾏 19：13-15；𖼖𖽪𖾏 18：15-17）
13 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽳𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖼽𖽔𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽑𖽱𖾏 𖼚𖽑𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 14 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐， 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏： “𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 15 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙， 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 ， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏．” 16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼁𖽔 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
𖼊𖽪𖾐 𖼅𖽔𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏
（𖼊𖽑𖽹𖾏 19：16-30；𖼖𖽪𖾏 18：18-30）
17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑 𖽃𖽡𖾑， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽘 𖼋𖽔𖾏， 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏， 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼸𖽠𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 ？” 18 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏？ 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏，𖼯𖽔 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏． 19 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼯𖽡𖽙：‘ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐； 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 ； 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼺𖽘𖾐 𖽃𖽔𖾏 𖼸𖽜； 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑； 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐； 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏．’ ” 20 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼘𖽪， 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼳𖽻𖾐， 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖾑 𖼊𖽪𖾏 𖼻𖽙𖾑 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏．” 21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽔𖾏 𖼐𖽱𖾏 ，𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼯𖽠 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼺𖽷𖾐， 𖼅𖽻𖾏 𖼋𖽺 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏， 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏， 𖼯𖽡𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼟𖽡 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐， 𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐．” 22 𖼐𖽱𖾏 𖼑𖽜 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑， 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼁𖽔𖾏 𖼉𖽙，𖼊𖽳𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝．
23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽔𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝．” 24 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼅𖽡， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽔𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝． 25 𖼘𖽙 𖼊𖽑𖽙𖾐 𖼁𖽺𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼞𖽻𖾐， 𖼖𖽡 𖼟𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙．” 26 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 ？” 27 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ，𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏， 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼊𖽪𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏．” 28 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼑𖽔𖾏 𖼘𖽪， 𖼀𖽡𖾐 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼺𖽘𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼟𖽡 𖼋𖽘𖾏．” 29 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼇𖽳𖾑 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏、 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼞𖽱𖾑 𖼋𖽡𖾏、 𖼞𖽱𖾑 𖼻𖽷𖾑、𖽃𖽔𖾑 𖼐𖽡𖽝𖾐、𖽃𖽔𖾑 𖼈𖽹𖾐、 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏， 30 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼀𖽔𖾏 𖽐𖼀𖽠，𖼟𖽪𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏、 𖼞𖽱𖾑 𖼋𖽡𖾏、 𖼞𖽱𖾑 𖼻𖽷𖾑、𖽃𖽔𖾑 𖼐𖽡𖽝𖾐、 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏，𖼊𖽑𖽡𖽝𖾐 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 ，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔， 𖼐𖽡𖽙𖾐 𖼷𖽑𖽷𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼗𖽙𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏． 31 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏，𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼐𖽡𖽙𖾐 𖼊𖽑𖽔𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽑𖽔𖾏．”

46 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑． 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼀𖽔𖾑 𖼊𖽡𖾐 𖼄𖽹𖾐 ， 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼄𖽹𖾐 𖼊𖽪𖾐， 𖼼𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 𖼳𖽻𖾏 𖼡𖽜𖾑． 47 𖼐𖽱𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽔𖾏， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼘𖽪！ 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑．” 48 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽔𖾏 𖼐𖽱𖾏， 𖼽𖽔𖾑 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏， 𖼐𖽱𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼘𖽪！ 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑．” 49 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼳𖽠𖾑 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏．”𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼷𖽷𖾑 𖼐𖽔𖾐 𖼄𖽱𖾐 𖽃𖽡𖾑， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼮𖽙𖾏 𖼺𖽡𖽝𖾐， 𖼳𖽠𖾑 𖼗𖽙𖾏 ，𖼮𖽵𖾑 𖼨𖽑𖽪𖾏 𖼟𖽡．” 50 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏， 𖼚𖽑𖽡𖽝𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏， 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐． 51 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 ？” 𖼐𖽔𖾐 𖼄𖽱𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 ，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 ．” 52 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼅𖽻𖾏， 𖼟𖽡 𖽐𖼮𖽜 𖼮𖽨𖾏 𖼟𖽡 𖼋𖽘𖾏．” 𖼐𖽱𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏， 𖼎𖽑𖽻𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 ．
