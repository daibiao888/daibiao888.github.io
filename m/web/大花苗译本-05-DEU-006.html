𖼋𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏、𖼼𖽘𖾏 𖼻𖽡𖽝𖾑、𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 , 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼅𖽻𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽡𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 , 2 𖼻𖽻𖾏 𖼟𖽙𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼊𖽪𖾐 𖼘𖽑𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑、𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖽃𖽡𖾑 , 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 . 3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ！ 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻 , 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 , 𖼟𖽙𖾏 𖼟𖽡 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼄𖽡𖾏 𖽐𖼎𖽙 𖼄𖽪𖾑 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼏𖽵 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 .
4 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ! 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽔 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽔𖾐 . 5 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜、𖽃𖽔𖾑 𖼛𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜、𖼛𖽘𖾏 𖼼𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼮𖽑𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 6 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏， 7 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏． 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽡 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 , 𖼅𖽻𖾏 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 , 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼳𖽙𖾏 𖼀𖽱𖾏 , 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 . 8 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼺𖽻𖾏 , 𖽐𖼊𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼇𖽻𖾏 𖼐𖽡𖽝𖾑𖽑 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼽𖽘𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏 . 9 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼀𖽔𖾐 𖽃𖽔 𖼛𖽻𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼊𖽪𖾑 𖼇𖽻𖾑 𖽃𖽔 𖼛𖽻𖾏 .
