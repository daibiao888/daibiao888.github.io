𖼎𖽑𖽻𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼽𖽙 𖼀𖽳 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼾𖽨𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼗𖽙 . 2 𖼽𖽙 𖼀𖽳 𖼐𖽑𖽡𖾏:
3 “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼞𖽪𖾑 ,
𖽐𖼎𖽙 𖼄𖽑𖽙𖾏 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏 𖽃𖽔𖾑 𖼾𖽠 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏．
4 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼦𖽡𖾐 𖽐𖼎𖽑𖽦𖾑 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼦𖽡𖾐 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 .
5 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼙𖽠𖾏 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼮𖽳 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼐𖽱𖾏 .
6 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼚𖽪𖾐 𖼖𖽡𖾐 𖽐𖼮𖽡𖽝𖾐 𖼦𖽔𖾏 𖼙𖽠𖾏 ,
𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 .
7 𖼑𖽔𖾏 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏 ,
𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏.
8 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼸𖽠𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼮𖽑𖽙 𖼖𖽡𖾏 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾐 𖽃𖽡𖾑 (“𖼖𖽡𖾏 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾐”𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼗𖽪𖾏， 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 “𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐”) ,
𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑 .
9 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼄𖽑𖽙𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼮𖽡𖾏 𖽃𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 ,
𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽙𖾏 𖼮𖽡𖾏 𖼡𖽘𖾐 𖽐𖼊𖽪𖾐 𖼑𖽡𖾏 𖼮𖽡𖾏 .
10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼟𖽪𖾏 𖼾𖽨 𖼞𖽪𖾑 𖽃𖽡𖾑 𖽃𖽔 𖼛𖽻𖾏 𖼡𖽙𖾏 𖼮𖽡𖽝𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼁𖽙𖾏 .
 
11 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼺𖽷𖾐 𖼋𖽔 𖼺𖽘𖾏 ?
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼊𖽪𖾏 𖼀𖽘𖾏 𖼺𖽘𖾏 ?
12 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽔𖾏 𖼇𖽻𖾑 𖼯𖽡𖽻 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽪𖾑 ?
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽔𖾏 𖼄𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼾𖽨 𖼞𖽪𖾑 ?
13 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 , 𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼢𖽻 𖼀𖽱𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 ,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼷𖽙𖾑 𖽐𖼷𖽙𖾑 𖼳𖽙𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 .
14 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼸𖽠𖾐 ,
𖼟𖽪𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 .
15 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐 ,
𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼳𖽙𖾏 .
16 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼺𖽷𖾐 𖼷𖽹𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏 ,
𖼎𖽙𖾑 𖼚𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 , 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 ?
17 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽐𖼮𖽳 𖼺𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽙 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼊𖽻𖾏 𖼳𖽙𖾏 .
18 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼞𖽱𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼋𖽡𖽝𖾏 𖼋𖽪 ,
𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼼𖽙 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 .
19 𖼘𖽑𖽙𖾐 𖼳𖽻𖾐 𖼞𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 .
𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼽𖽔𖾑 𖼻𖽻𖾏
20 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 , 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑 ?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 , 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑 ?
21 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼋𖽔 , 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼋𖽔 ,
𖽐𖼎𖽑𖽦𖾑 𖼋𖽔 、 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 .
22 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼙𖽔 , 𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝 .
23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 ,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼙𖽻𖾏 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼮𖽡𖽝𖾏 ,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼺𖽷𖾐 𖼘𖽪 ?
