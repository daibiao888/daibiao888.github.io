𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖽐𖼎𖽙 𖼘𖽙 𖼊𖽳, 𖼊𖽑𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏, 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼙𖽠𖾏. 2 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼑𖽡𖾏 𖼯𖽦𖾏、𖼞𖽪𖾐、𖼑𖽡𖽝𖾏 𖼷𖽑𖽷𖾑 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐. 3 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼚𖽔𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼊𖽑𖽡𖽝𖾐 𖼣𖽹𖾏 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽃𖽡𖾑, 4 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽪𖾐 𖼁𖽱𖾏 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏. 5 𖼘𖽙 𖼊𖽳 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼑𖽨𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐、𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐、𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐. 6 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼷𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐； 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐. 7 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼳𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐. 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼘𖽙 𖼊𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼯𖽦𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖽐𖼎𖽙 𖼘𖽙 𖼊𖽳 𖼦𖽡𖾐 𖼇𖽺𖾐
8 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼘𖽙 𖼊𖽳 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏, 𖼟𖽡 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼯𖽦𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼊𖽔𖾐. 9 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼘𖽪? 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼦𖽡𖾐 𖼇𖽺𖾐！ 𖼟𖽡 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏, 𖼞𖽪𖾑 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏； 𖼟𖽡 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏.” 10 𖼘𖽙 𖼊𖽳 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼺𖽙𖾐 𖽃𖽳𖾐, 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼐𖽻𖾏, 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼖𖽻𖾐 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐、𖽃𖽳 𖼄𖽙 𖼖𖽔𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏, 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 11 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏𖼘𖽙 𖼊𖽳 𖼺𖽹𖾑 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏, 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐． 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏. 12 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐, 𖼘𖽙 𖼊𖽳 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖽃𖽡𖾑 𖼼𖽙𖾏 𖼙𖽻, 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼮𖽑𖽹𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐. 13 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
14 𖼘𖽙 𖼊𖽳 𖼇𖽳𖾑 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽑𖽡𖾏: “𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑, 𖼟𖽡 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏； 15 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏. 16 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼮𖽡𖽻𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼸𖽠𖾏 𖼾𖽱 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐, 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼸𖽠𖾏 𖼾𖽱 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏. 17 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼟𖽡.” 18 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼎𖽺𖾏 𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐, 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝， 𖼄𖽔𖾏 𖼖𖽡𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼮𖽡𖽝𖾏, 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘.
