𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼀𖽡𖾐, 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐,
𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖾏, 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐. （𖼳𖽡𖾏 𖼖𖽔𖾑）
2 𖼻𖽻𖾏 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑,
𖼉𖽻𖾏 𖼞𖽪𖽙 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏.
3 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡，
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡．
 
4 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼙𖽔 𖼚𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐. （𖼳𖽡𖾏 𖼖𖽔𖾑）
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪， 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡！
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡！
 
6 𖼊𖽡𖾐 𖼋𖽠𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖽐𖼎𖽙 𖼻𖽻𖾐 𖼷𖽷𖾏,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏, 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐.
7 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼺𖽘𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐,
𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏！
