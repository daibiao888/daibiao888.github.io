14
𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼖𖽔 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 , 𖼺𖽘𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 、 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼋𖽠𖾏 ,𖼑𖽜 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 . 2 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼎𖽺𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 . 3 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼎𖽺𖾏 𖼷𖽙𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼐𖽤𖾐 , 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 . 4 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼳𖽻𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 : 𖼳𖽔𖾑 𖼄𖽪𖾐 𖼽𖽔𖾏 、 𖼳𖽙 𖼀𖽔𖾏 、 𖼐𖽔 𖼊𖽗𖾑 、 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 、 5 𖼽𖽡𖾏 𖼳𖽢 、 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽪𖾑 𖼽𖽔𖾏 、 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 、 6 𖼐𖽙 𖼮𖽢𖾑 、 𖼐𖽡 𖼇𖽺𖾑 、 𖼽𖽔𖾐 𖼇𖽺𖾑 𖼽𖽔𖾏 、 7 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼄𖽔𖾐 、 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼊𖽔𖾏 、 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼖𖽡𖽝 .
𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐
（𖼺𖽔𖾐 𖽃𖽔𖾐 5：17 - 25）
8 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽻𖾏 𖼚𖽜𖾑 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 . 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 , 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼸𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 . 9 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼋𖽘 , 𖼦𖽔𖾏 𖽐𖼮𖽵 𖼚𖽔𖾏 𖼖𖽡𖾏 𖼇𖽔 𖼽𖽡𖽾𖾑 𖼖𖽪𖾑 𖼦𖽘𖾑 . 10 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖼘𖽪 ? 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼘𖽪 ?” 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼯𖽡𖽙𖾏 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .” 11 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼸𖽙 𖼀𖽔𖾑 𖼖𖽡 · 𖼀𖽑𖽡𖾐 𖼖𖽔𖾑 𖼳𖽡𖽾𖾑 , 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼊𖽱𖾑𖽑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽱𖾑𖽑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼀𖽔𖾑 𖼖𖽡 · 𖼀𖽑𖽡𖾐 𖼖𖽔𖾑 𖼳𖽡𖽾𖾑 . 12 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼁𖽠𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽡𖽝𖾏 , 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼋𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼁𖽠𖾏 𖼺𖽘𖾏 .
13 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼦𖽔𖾏 𖽐𖼮𖽵 𖼚𖽔𖾏 𖼖𖽡𖾏 𖼇𖽔 𖼽𖽡𖽾𖾑 𖼖𖽪𖾑 𖼦𖽘𖾑 . 14 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼷𖽷𖾑 𖼀𖽡𖾐 𖼞𖽘𖾐 𖼻𖽻𖾐 𖼀𖽑𖽘𖾑 𖼀𖽡 𖼋𖽪𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 15 𖼟𖽡 𖽃𖽡𖾐 𖼐𖽑𖽙𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼀𖽡𖾐 𖼞𖽘𖾐 𖼻𖽻𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼏𖽙 𖼋𖽘𖾏 .” 16 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 , 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽙 , 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼀𖽡𖽝𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼺𖽳 . 17 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼡𖽑𖽔𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 .
