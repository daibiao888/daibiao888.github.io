13
𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼑𖽜 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼾𖽡𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼡𖽑𖽜𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼷𖽑𖽠𖾏 , 𖽐𖼷𖽔𖾑 𖼺𖽘𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽐𖼎𖽙 𖼦𖽡𖾏 𖼡𖽳𖾏.
2 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼊𖽺𖾐 𖼁𖽠𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏, 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖽐𖼎𖽙 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏. 3 𖼮𖽔𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼋𖽘 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑. 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼐𖽱𖾏 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏. 4 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖽃𖽡𖾑, 𖼖𖽔𖾑 𖼑𖽜 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. 5 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏:𖼞𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐, 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼊𖽔𖾐, 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽜𖾑. 6 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼀𖽡𖾏 𖼡𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽡𖽳𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑? 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽺 𖼗𖽔 𖽐𖼞𖽔𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐.”
𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏
7 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼘𖽪!𖼽𖽔𖾑 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 !
𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼽𖽺 𖼗𖽔 .
𖽐𖼊𖽻𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏, 𖼾𖽘𖾏 𖼎𖽺𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏,
𖼞𖽪𖾑 𖼖𖽔𖾑 𖼎𖽙𖾑 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 .
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼷𖽷𖾐 𖽐𖼀𖽠 𖽃𖽡𖾑 𖽃𖽔𖾐 𖽐𖼀𖽠, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼯𖽔 𖼅𖽔𖾏 𖼷𖽷𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏.
9 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼟𖽙𖾏 𖼷𖽷𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼐𖽡𖾑 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏,
𖼷𖽑𖽷𖾑 𖼮𖽡𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼮𖽡𖽺𖾏 𖼑𖽡𖽝𖾏,
𖽐𖼮𖽔𖾏 𖼮𖽡𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼮𖽡𖽺𖾏 𖼞𖽪𖾐.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏.
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: ‘ 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏.’
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: ‘ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. ’ ”
