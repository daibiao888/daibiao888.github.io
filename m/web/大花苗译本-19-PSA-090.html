𖽐𖼊𖽠𖾑 𖽐𖼞𖽻𖾏 𖼚𖽻𖾐 𖼑𖽘
𖼷𖽻𖾑 𖼖𖽔 𖼮𖽵𖾑 𖼅𖽳𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
90
𖼮𖽵𖾑 𖼘𖽪！ 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖽃𖽔𖾏 𖼀𖽡𖾐 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐.
2 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽻𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼅𖽔𖾏 𖼗𖽙 ,
𖼊𖽡𖾐 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼊𖽻𖾏 𖼟𖽡 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏,
𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
 
3 𖼟𖽡 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽙𖾑 𖼁𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 , 𖼐𖽑𖽡𖾏:
“𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼗𖽙𖾏 .”
4 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼑𖽔𖾏 𖼗𖽙𖾏 , 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼳𖽡𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼑𖽘 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼯𖽡𖽹𖾐.
 
5 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖽐𖼊𖽱𖾑𖽑 𖼙𖽠𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼀𖽱𖾏 𖽃𖽡𖾐 𖼯𖽷𖾏 ,
𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑，
6 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖽃𖽔𖾏 𖼟𖽻𖾏 𖼘𖽑𖽘𖾑 𖼘𖽑𖽙𖾐,
𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼤𖽜𖾏 𖼺𖽘𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼙𖽠𖾏.
 
7 𖼀𖽡𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼑𖽜 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼐𖽔𖾑 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏.
8 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏.
 
9 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼟𖽱𖾏 𖼮𖽑𖽡𖽻𖾏 𖼗𖽙𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼀𖽡𖾐 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼺𖽱𖾏 𖼮𖽙𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼀𖽘𖾏 𖼊𖽔𖾐.
10 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽡𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏,
𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼸𖽠𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖼷𖽻𖾑 𖼟𖽪𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖽐𖼊𖽱𖾏 𖼊𖽳𖾏 𖼊𖽔𖾐,
𖽃𖽡𖾐 𖽐𖼮𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏, 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽘𖾏 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
 
11 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏？
𖼡𖽔 𖼋𖽱𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼟𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼖𖽺𖾑?
12 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼮𖽔𖾑 𖼺𖽙𖾏 𖼾𖽨𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏,
𖼸𖽠𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
 
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼺𖽷𖾐 𖼖𖽺𖾑?
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼺𖽷𖾐 𖼗𖽙 , 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏.
14 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽐𖼷𖽙𖾑 𖽐𖼷𖽙𖾑 𖼊𖽻𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼻𖽻𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑.
15 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼑𖽻𖾏 𖽃𖽦𖾐，
𖽐𖼎𖽙 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑.
16 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏.
17 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼮𖽵𖾑 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑．
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼑𖽜 𖼀𖽡𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
