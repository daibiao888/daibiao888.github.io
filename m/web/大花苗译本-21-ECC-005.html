𖼟𖽡 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼑𖽻, 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏. 2 𖼟𖽡 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽷𖾏 𖼛𖽙 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼷𖽹𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼯𖽠 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏.
3 𖼑𖽪𖾏 𖼺𖽜𖾑 𖼮𖽡𖽻𖾐, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑； 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 .
4 𖼟𖽡 𖼏𖽵 𖼗𖽪𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼏𖽵 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾑 𖼊𖽳𖾑𖽑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽻𖾏. 5 𖼟𖽡 𖼏𖽵 𖼗𖽪𖾏 𖼮𖽡𖽝𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼁𖽻𖾏, 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼏𖽵. 6 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽔𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 (“𖼊𖽪𖾐 𖼖𖽑𖽘𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “𖼊𖽪𖾐 𖼷𖽷𖾐”) 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼏𖽵 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼟𖽡 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼖𖽺𖾑？ 7 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼗𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑, 𖼟𖽡 𖼮𖽔 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 8 𖼟𖽡 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼚𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽙 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 , 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖽐𖼎𖽙 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼸𖽠𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼺𖽷𖾐. 9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽻𖾏 𖼊𖽪𖾑 𖼾𖽨 .
𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼦𖽡𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐
10 𖼦𖽠𖾑 𖼉𖽳 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐, 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼊𖽻𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏, 𖼦𖽠𖾑 𖼉𖽳 𖼅𖽔𖾏 𖼊𖽻𖾏, 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼊𖽻𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑.
11 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼎𖽺𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑? 𖼟𖽪𖾏 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼡𖽔 𖼅𖽔 𖼊𖽔𖾐.
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽻𖾏 𖼮𖽡𖽻𖾐 𖼑𖽻𖾏 𖼯𖽠 , 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼀𖽱𖾏； 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼑𖽡𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼀𖽱𖾏.
13 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼮𖽷𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐, 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼦𖽺𖾏 𖼾𖽨𖾏. 14 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼳𖽻𖾏 𖽐𖼚𖽔 , 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘 , 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼮𖽔𖾑 𖼘𖽑𖽘𖾑 𖼾𖽨 𖼊𖽪𖾐, 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐. 15 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏 𖼗𖽙𖾏, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏 𖼅𖽻𖾏 𖼙𖽠𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼯𖽡𖽙𖾏 𖼅𖽻. 16 𖼐𖽱𖾏 𖼼𖽘𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼗𖽙𖾏, 𖼐𖽱𖾏 𖼙𖽠𖾏 𖼼𖽘𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏． 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽡𖽝． 𖼐𖽱𖾏 𖼑𖽜 𖼮𖽢𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽺𖾑? 17 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏, 𖼮𖽡𖽻𖾐 𖼅𖽔𖾏 𖽐𖼊𖽱𖾏 𖼊𖽳𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖽐𖼎𖽙 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐.
