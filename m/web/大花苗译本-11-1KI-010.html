（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 9：1 - 12）
10
𖼳𖽷𖾏 𖼀𖽔𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼐𖽔𖾑 𖽐𖼊𖽙𖾏 𖽐𖼷𖽷𖾏. 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼊𖽔𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼋𖽔𖾏 𖼑𖽪 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞. 2 𖼐𖽱𖾏 𖼨𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼘𖽙 𖼊𖽑𖽙𖾐 𖼮𖽹𖾑 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏, 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑. 𖼐𖽱𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏. 3 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼑𖽪 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖼊𖽡𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼗𖽙𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏. 4 𖼳𖽷𖾏 𖼀𖽔𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽙𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾑， 5 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾐 𖼎𖽻𖾏 𖼸𖽱𖾏 𖼞𖽡 𖼏𖽻𖾏、 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏、 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼳𖽠𖾑 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏、 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼷𖽑𖽷𖾏、 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼮𖽡𖽠𖾑, 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼊𖽑𖽹𖾑 𖽃𖽔𖾏 𖼷𖽜𖾏 ( 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑” ) 𖼎𖽺𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼗𖽙𖾏.
6 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼡𖽑𖽙𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏. 7 𖼞𖽪𖾑 𖼺𖽪𖾐 𖼁𖽱 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑, 𖼸𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽔𖾏 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 𖼋𖽘𖾏, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼸𖽙 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼺𖽷𖾐, 𖼟𖽡 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼟𖽡 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼮𖽑𖽡𖽻𖾏 𖼇𖽔𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼐𖽙𖾑𖽑. 8 𖼟𖽡 𖼉𖽻 𖼷𖽻𖾏, 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽻 𖼟𖽡 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳, 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏. 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏！ 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼟𖽡, 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽪𖽙 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽘𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳. 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼳𖽷𖾏 𖼀𖽔𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 𖼞𖽪𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏. 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑.
