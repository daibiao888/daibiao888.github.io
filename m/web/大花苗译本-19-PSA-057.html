𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼻𖽡𖽝𖾏 𖼮𖽵𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼀𖽜𖾑 𖼺𖽻𖾐 𖼘𖽙, 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼐𖽡𖾑, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏．𖼯𖽡𖽙𖾏 “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑” 𖼳𖽘𖾐.
（𖼺𖽔𖾐 𖽃𖽡𖾐 24：1 - 22）
57
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑 , 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡．
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼋𖽙𖾏 𖼸𖽙 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏.
2 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏.
3 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼞𖽪𖾑 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 . （𖼳𖽡𖾏 𖼖𖽔𖾑）
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼊𖽑𖽡𖽝𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼎𖽻𖾏 𖼞𖽪𖾑 .
 
4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖼞𖽪𖾑 𖼺𖽱𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼄𖽑𖽪𖾑、𖼊𖽪𖾑 𖼈𖽙𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼟𖽪𖾏 𖽐𖼮𖽳𖾏 .
 
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏, 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜！
6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼮𖽢𖾐 𖼸𖽙𖾏 𖼮𖽡𖽝𖾏, 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼮𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖼮𖽡𖽝𖾏, 𖼾𖽨𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
7 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏, 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔！
8 𖼞𖽪𖾑 𖼖𖽡𖽾 𖼘𖽪 ! ( “𖼖𖽡𖽾”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“ 𖽐𖼚𖽻 𖼮𖽡𖾏”) 𖼟𖽡 𖼽𖽔𖾑 𖼯𖽷𖾏 𖼳𖽠𖾑！
𖼢𖽜𖾏 𖽐𖼎𖽔 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽷𖾏 𖼳𖽠𖾑， 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼯𖽷𖾏 𖼮𖽡𖾏 𖼡𖽘𖾐 𖽐𖼊𖽪𖾐.
9 𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 ,
𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡．
10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏, 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏.
11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏, 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏；
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜．
