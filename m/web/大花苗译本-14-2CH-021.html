2 𖼽𖽙 𖼖𖽔 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 , 𖼟𖽪𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾐 、 𖼽𖽦𖾑 𖼳𖽡𖽝 、 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 、 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾐 、 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼖𖽳 、 𖼳𖽷𖾏 𖼇𖽔 𖼊𖽑𖽡 𖼽𖽔𖾐 , 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 . 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖼄𖽔𖾐 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 𖼟𖽪𖾏 𖼞𖽪𖾐、 𖼑𖽡𖽝𖾏、 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 , 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 𖼈𖽡𖽝𖾏 𖼄𖽔𖾐 𖼞𖽪𖽙 𖼎𖽻𖾏 𖼽𖽙 𖼖𖽔 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 . 4 𖼽𖽙 𖼖𖽔 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼾𖽨𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 , 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 .
5 𖼽𖽙 𖼖𖽔 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 . 6 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 , 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼾𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽙𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐 , 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 . 7 𖼈𖽡𖽝𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼾𖽡𖾏 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖼏𖽵 , 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼺𖽘𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 .
𖼊𖽻𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏
