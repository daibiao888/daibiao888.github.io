𖼽𖽙 𖼀𖽳 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 :
2 “𖼅𖽡 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 !
𖼅𖽡 𖼋𖽔 𖼙𖽠𖾏, 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽡𖾐 𖼸𖽻 𖼛𖽙𖾏 𖼋𖽔 .
3 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 , 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼅𖽡 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 , 𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼘𖽪 ?
4 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼑𖽻 𖼊𖽡𖾐 ,
𖼎𖽺𖾏 𖼊𖽻𖾏 𖼽𖽺 𖼗𖽔 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 .
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 , 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 .
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼋𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝 𖼢𖽻𖾏 .
6 𖼸𖽜 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏（“𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪 𖽃𖽔𖾏 “𖼮𖽵𖾑 𖼳𖽺”） 𖼯𖽡𖽙𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .
7 𖼟𖽡 𖼮𖽔𖾑 𖼑𖽪 𖼯𖽦𖾏 , 𖼯𖽦𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 ,
𖼎𖽺𖾏 𖼑𖽪 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 , 𖼊𖽪𖾑 𖼐𖽻 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 .
8 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 ,
𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼊𖽪𖾑 𖽐𖼀𖽳𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼟𖽙𖾏 𖽐𖼎𖽑𖽹𖾏 .
9 𖼑𖽔𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 ,
𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑 ?
10 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑
𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼳𖽦𖾑 𖼀𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .
11 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 ?
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖽝𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼘𖽪 ?
12 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼘𖽑𖽙𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 .
𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼸𖽠𖾏 𖽐𖼞𖽡𖽝𖾏
13 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼛𖽘𖾏 𖼼𖽙𖾏 ,
𖼐𖽱𖾏 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼊𖽘𖾐 .
14 𖼑𖽔𖾏 𖼘𖽪！ 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 , 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼎𖽪𖾑 𖼮𖽡𖽝𖾏 , 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼮𖽙𖾏 .
15 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 , 𖽃𖽻𖾐 𖼎𖽺𖾏 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏 ,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼟𖽙𖾏 𖽃𖽻𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖽃𖽻𖾐 𖼎𖽺𖾏 𖽐𖼚𖽱𖾏 𖽐𖼷𖽜𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 .
16 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼸𖽠𖾏 𖽐𖼞𖽡𖽝𖾏 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽑𖽵𖾑 , 𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼁𖽡𖽝 .
17 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼸𖽠𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼯𖽡𖽙𖾏 𖼊𖽠𖾏 𖼊𖽡𖾏 𖼮𖽢𖾏 𖼙𖽠𖾏 ,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼚𖽑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 .
18 𖼐𖽱𖾏 𖼚𖽠𖾏 𖼺𖽘𖾏 𖼖𖽑𖽔𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 ,
𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼚𖽔𖾑 .
19 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼯𖽡𖽙𖾏 𖼊𖽠𖾏 𖼊𖽡𖾏 𖼮𖽢𖾏 𖼙𖽠𖾏 ,
𖼎𖽺𖾏 𖼊𖽑𖽠𖾐 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 .
20 𖼐𖽱𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 ,
𖼎𖽺𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼻𖽻𖾏 𖼮𖽷𖾑 .
21 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 ,
𖼚𖽠𖾏 𖼺𖽘𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼑𖽡𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 .
22 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼊𖽙𖾐 ,
𖼐𖽱𖾏 𖼨𖽑𖽳𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽐𖼚𖽙 .
23 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 ,
