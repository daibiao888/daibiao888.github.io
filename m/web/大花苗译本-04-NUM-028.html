𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
（𖼋𖽠𖾏 29：38 - 46）
28
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 , 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 , 𖼅𖽡 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 .
3 “𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏 , 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼋𖽪 , 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 . 4 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏, 𖼐𖽑𖽪𖾐 𖼢𖽹𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏； 5 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏, 𖼊𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖼳𖽡 𖼚𖽻𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 . 6 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼙𖽔 𖼏𖽵 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑， 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 7 𖼑𖽜 𖼊𖽔𖾑 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼐𖽡𖾑 , 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼳𖽡 𖼚𖽻𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑． 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽐𖼚𖽙，𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 . 8 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼐𖽑𖽪𖾐 𖼢𖽹𖾐 , 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖼊𖽔𖾑 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼊𖽔𖾑 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏， 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 , 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑
9 “𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 , 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖽃𖽔𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏, 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽔𖾐 𖽐𖼀𖽠 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼛𖽙𖾏 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖼎𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 . 10 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 , 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼞𖽑𖽱𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 .
𖼷𖽑𖽔𖾏 𖼖𖽑𖽡𖾏 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑
11 “𖼷𖽑𖽔𖾏 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 , 𖼅𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏, 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼗𖽻𖾏 𖼾𖽘𖾏 , 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏、𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 , 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 . 12 𖼑𖽜 𖼷𖽑𖽔𖾏 𖼋𖽪𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖼷𖽷𖾐 𖽐𖼀𖽠 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼚𖽙𖾏 , 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼊𖽑𖽡𖽝𖾐 , 𖼑𖽜 𖼊𖽪𖾑 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽔𖾐 𖽐𖼀𖽠 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼛𖽙𖾏, 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼊𖽑𖽡𖽝𖾐； 13 𖼑𖽜 𖼷𖽑𖽔𖾏 𖼋𖽪 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼚𖽙𖾏, 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼊𖽑𖽡𖽝𖾐 . 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 , 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾏 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 14 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼽𖽔𖾑 𖽃𖽡𖾐 𖽐𖼊𖽘𖾏 𖼳𖽡 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 , 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼳𖽡 𖼷𖽷𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑， 𖽃𖽡𖾐 𖼋𖽪 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼳𖽡 𖼚𖽻𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 . 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼖𖽑𖽡𖾏 𖼑𖽡𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼙𖽱 𖼙𖽱 𖼷𖽑𖽔𖾏 𖼖𖽑𖽡𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 . 15 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 , 𖼖𖽑𖽘𖾑 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼖𖽑𖽘𖾑
（𖼖𖽡𖾏 23：5 - 14）
16 “𖼖𖽪𖾑 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏 𖼟𖽻𖾏 𖼚𖽻𖾐 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 . 17 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼐𖽡𖾑 𖼟𖽻𖾏 𖼀𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 , 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 𖼳𖽤𖾐 𖼐𖽑𖽪𖾐 . 18 𖼐𖽑𖽪𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 . 19 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 , 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼞𖽡𖾏 𖼗𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 , 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏 , 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 . 20 𖼡𖽻𖾐 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼎𖽙 𖼖𖽑𖽘𖾑 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼛𖽙𖾏 , 𖼑𖽜 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖼷𖽷𖾐 𖽐𖼀𖽠； 𖼑𖽜 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽔𖾐 𖽐𖼀𖽠； 21 𖼑𖽜 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖽃𖽡𖾑 , 𖼷𖽑𖽔𖾏 𖼋𖽪𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠. 22 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 𖼑𖽜 𖼅𖽡 𖼚𖽠𖾏 𖼾𖽔𖾏 . 23 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 , 𖼅𖽡 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖼋𖽡𖾏 𖼐𖽡𖾑 . 24 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 , 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑 , 𖼞𖽑𖽡𖽠𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 𖼐𖽡𖾑 𖼉𖽺𖾏 𖼢𖽻𖾏 . 25 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 .
（𖼋𖽠𖾏 34：22；𖼖𖽡𖾏 23：15 - 21；𖼎𖽺𖾏 16：9 -12）
26 “𖼳𖽤𖾏 𖼋𖽪𖾏 𖼳𖽤𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼺𖽪𖾐 𖼳𖽦𖾑 , 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼖𖽑𖽘𖾑 𖼡𖽻𖾐 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 . 27 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏, 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 , 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 28 𖼡𖽻𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼖𖽑𖽘𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼛𖽙𖾏 , 𖼑𖽜 𖼷𖽑𖽔𖾏 𖼋𖽪𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖼷𖽷𖾐 𖽐𖼀𖽠, 𖼑𖽜 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 , 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽔𖾐 𖽐𖼀𖽠； 29 𖼑𖽜 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 , 𖼷𖽑𖽔𖾏 𖼋𖽪𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠. 30 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏 𖼑𖽜 𖼅𖽡 𖼚𖽠𖾏 𖼾𖽔𖾏 . 31 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 𖼺𖽘𖾏，𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼋𖽡𖾏 𖼐𖽡𖾑， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏.”
