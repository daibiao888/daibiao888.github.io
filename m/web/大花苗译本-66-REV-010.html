𖼞𖽪𖾑 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏，𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼀𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐，𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼅𖽔𖾏 𖼊𖽺𖾐 𖼼𖽘𖾏 𖼦𖽻𖾏 𖽃𖽻𖾐，𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐，𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼎𖽻𖾐． 2 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼑𖽘 𖼣𖽔𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏．𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼷𖽠𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑，𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼖𖽡 𖼇𖽳𖾏 𖼷𖽠𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐． 3 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏，𖼺𖽱𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼢𖽔𖾏．𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼺𖽙𖾐 𖼋𖽠𖾏 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏． 4 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼺𖽙𖾐 𖼋𖽠𖾏 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑， 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽑𖽡𖾏：“ 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼺𖽙𖾐 𖼐𖽑𖽡𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼘𖽙𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏．” 5 𖼷𖽷𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼷𖽠𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼎𖽙 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖼳𖽤𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏， 6 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏、𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐、𖼳𖽡𖽠𖾑 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼳𖽡𖽠𖾑、𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏 𖽃𖽡𖾑 𖼏𖽵 𖼗𖽪𖾏 ，𖼐𖽑𖽡𖾏：“𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼋𖽘𖾏． 7 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 ”．
8 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏，𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 ， 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼅𖽻𖾏， 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼷𖽠𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼎𖽙 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏，𖼞𖽑𖽡𖽠𖾏 𖼑𖽘 𖽐𖼊𖽠𖾑 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏．” 9 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ：“ 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼑𖽘 𖼣𖽔𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑．” 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼅𖽻𖾏， 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏．𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼽𖽔𖾑 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏．” 10 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼑𖽘 𖼣𖽔𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖽃𖽡𖾑 𖼗𖽙𖾏， 𖼑𖽻𖾏 𖼋𖽘 𖼺𖽘𖾏，𖼋𖽡𖽝 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏，𖼑𖽻𖾏 𖼋𖽘 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏． 11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐、𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐、𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 ．”
