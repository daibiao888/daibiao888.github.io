8
𖼡𖽔 𖼋𖽱𖾏 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼖𖽺𖾑?
𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖼑𖽪𖾏 𖼑𖽡𖾏 𖼚𖽠𖾏 𖼐𖽑𖽡𖾏 𖼖𖽺𖾑?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏.
2 𖼞𖽪𖾑 𖼞𖽑𖽻𖾑 𖼟𖽡 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼐𖽔𖾑. 3 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽜𖾐 𖼑𖽜𖾐 𖼞𖽑𖽜𖾐 𖼷𖽑𖽪𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽪𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼖𖽔𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏. 4 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼡𖽔 𖼋𖽱𖾏 𖼞𖽔𖾐 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?” 5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼻𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼟𖽱𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳, 𖼸𖽠𖾏 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼯𖽡𖽹𖾏 𖼋𖽱𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼨𖽑𖽢𖾐 𖼇𖽺𖾏” ). 6 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑, 𖼊𖽪𖾏 𖼅𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼸𖽺𖾏 𖼐𖽑𖽤𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 . 7 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼑𖽡𖾏 𖼑𖽪𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼖𖽺𖾑? 8 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 (“𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 ” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼮𖽢𖾏”), 𖼷𖽷𖾐 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 , 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼐𖽡𖾑 𖽐𖼮𖽷𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏. 9 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖾑 𖼙𖽔 𖼁𖽙𖾏, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼯𖽡𖽹𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑, 𖼟𖽙𖾏 𖼾𖽨𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏.

10 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼊𖽻𖾏 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏, 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑. 11 𖼊𖽡𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏 𖼈𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼞𖽑𖽜 𖼑𖽜 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏. 12 𖼊𖽪𖾐 𖼾𖽔𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼡𖽔𖾏 𖼀𖽔𖾏 𖼟𖽻𖾏 𖼈𖽡𖽝𖾏, 𖼎𖽙𖾑 𖼊𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽃𖽡𖾑, 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏. 13 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏, 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼺𖽱𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
