𖼖𖽡𖽾 𖼮𖽑𖽳𖾑 𖼞𖽪𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐. 𖼑𖽔𖾏 𖼘𖽪！ 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏, 𖼽𖽔 𖼳𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼺𖽔𖾐 𖼐𖽡 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽔 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽑𖽡𖾐 𖼖𖽔𖾑 𖼊𖽑𖽡 . 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏. 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼮𖽡𖽻𖾏 𖼺𖽷𖾐, 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽑𖽻𖾏, 𖼀𖽡𖾐 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏. 4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.” 5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑, 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼘𖽪！ 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏, 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼡𖽑𖽙𖾑 𖽐𖼊𖽱𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼯𖽡𖽙 . 6 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽔𖾏 𖽃𖽡𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽔𖾑 𖼯𖽢𖾏 𖼟𖽡𖾏. 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼙𖽻 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏, 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽑𖽻𖾏, 𖼈𖽡𖽝𖾏 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏. 8 𖼅𖽡 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 9 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼖𖽠𖾑 𖼅𖽡 𖼋𖽠𖾏 𖼙𖽠𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐. 10 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼨𖽑𖽢𖾐 𖼅𖽡 , 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 11 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼅𖽡 𖼖𖽪𖾑 𖼀𖽑𖽻𖾏, 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼨𖽑𖽢𖾐 𖼅𖽡, 12 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏.” 13 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼀𖽡𖾐 𖼐𖽔 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽑𖽡𖾐 𖼖𖽔𖾑 𖼊𖽑𖽡 𖼋𖽔 𖼙𖽠𖾏. 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽺𖾐! 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼟𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼖𖽻𖾐 𖼋𖽘 𖽐𖼮𖽢𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?”
