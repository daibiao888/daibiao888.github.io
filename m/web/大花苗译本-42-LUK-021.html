𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼗𖽔𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑
（𖼞𖽑𖽙𖾐 12：41-44）
21
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼑𖽔𖾏 ，𖼁𖽙𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼗𖽔𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼑𖽡𖽝𖾏． 2 𖼎𖽺𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾏 𖼋𖽔𖾏，𖼗𖽔𖾏 𖽃𖽔𖾐 𖼛𖽡𖾐 𖼣𖽔𖾏 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏． 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾏 𖼐𖽡𖾑，𖼗𖽔𖾏 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐． 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 ， 𖼯𖽡𖽙𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼋𖽔𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 ，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼐𖽡𖾑 𖼳𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼳𖽦𖾑 𖼋𖽔𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏．”
𖽃𖽔𖼁𖽙𖾏𖽐𖼷𖽔𖾏𖼗𖽔𖾏𖼋𖽪𖼷𖽑𖽷𖾐
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑
（𖼊𖽑𖽹𖾏 24：1-2；𖼞𖽑𖽙𖾐 13：1-2）
5 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏，𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼘𖽙𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼎𖽻𖾏 𖼦𖽡𖾐 𖼇𖽔𖾑 ． 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏 𖼐𖽡𖾑，𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 ，𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼮𖽡𖽝𖾏，𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏．”
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏
（𖼊𖽑𖽹𖾏 24：3-14；𖼞𖽑𖽙𖾐 13：3-13）
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 ，𖼯𖽡𖽹𖾏 𖼋𖽱 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑 ？ 𖼊𖽑𖽻𖾑 𖼋𖽱 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 ？ 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑？” 8 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼏𖽻 𖼮𖽑𖽵𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖼻𖽡𖽝𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏：‘ 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪’ ，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 ：‘𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏’， 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐！ 9 𖼅𖽡 𖼐𖽑𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼁𖽔𖾏 𖼏𖽙 ，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽪𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 ，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏， 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏． ” 10 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼉𖽺𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 ， 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖽙 𖼉𖽺𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 ． 11 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐， 𖼅𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖽐𖼎𖽙 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼞𖽡𖾏 ，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼟𖽪𖾏 𖼑𖽡 𖽐𖼮𖽑𖽹𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏． 12 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖼄𖽔𖾐 𖼅𖽡，𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼅𖽡，𖼄𖽔𖾐 𖼅𖽡 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱， 𖼎𖽺𖾏 𖼞𖽱𖾏 𖼅𖽡 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏，𖼎𖽺𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼮𖽤𖾐 𖼅𖽡 𖽐𖼮𖽡𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐． 13 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏． 14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼏𖽔𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼅𖽻𖾏 𖼊𖽑𖽔𖾑 𖼐𖽑𖽡𖾏． 15 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽻𖾏 𖼅𖽡，𖼟𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼅𖽡 𖽐𖼎𖽳𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼅𖽡， 𖼐𖽑𖽡𖾏 𖼦𖽡𖾐 𖼳𖽠𖾏 𖼅𖽡． 16 𖼅𖽡 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏、𖼞𖽱𖾑 𖼋𖽡𖾏、𖼑𖽜𖾏 𖼮𖽘𖾏、𖼽𖽺 𖼗𖽔，𖼽𖽔𖾑 𖼄𖽔𖾐 𖼅𖽡 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼉𖽻 𖼸𖽻𖾏 ，𖼅𖽡 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏． 17 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏，𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽪𖾐 𖼅𖽡 ． 18 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖽃𖽡𖾐 𖼷𖽙𖾏 𖼚𖽻𖾐 𖼇𖽻𖾏，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼚𖽙𖾐 𖼺𖽘𖾏． 19 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽠𖾏 𖼡𖽑𖽠𖾑 𖼅𖽡 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼮𖽡𖽝𖾏．”
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐
（𖼊𖽑𖽹𖾏 24：15-21；𖼞𖽑𖽙𖾐 13：14-19）

𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏
（𖼊𖽑𖽹𖾏24：29-31；𖼞𖽑𖽙𖾐13：24-27）
25 “𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖽐𖼎𖽙 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏，𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏，𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽔𖾑 𖼳𖽠𖾑 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼟𖽡𖾏 𖽐𖼷𖽑𖽷𖾏 𖼳𖽱𖾑，𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝． 26 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙，𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐， 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼦𖽡𖾐 𖼻𖽡𖽝𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑． 27 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐， 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏． 28 𖽂𖽪𖾏 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑，𖼅𖽡 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼟𖽪𖾏 𖼮𖽨𖾏 𖽐𖼊𖽪𖾐 𖼀𖽡𖾏 𖼮𖽨𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．”
𖼽𖽔𖾑 𖼯𖽷𖾏 𖼻𖽙𖾑
（𖼊𖽑𖽹𖾏 24：32-35；𖼞𖽑𖽙𖾐 13：28-31）
29 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “𖼅𖽡 𖼑𖽔𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐，𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖽐𖼊𖽻𖾏． 30 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼘𖽑𖽘𖾑 𖽃𖽔 𖽐𖼚𖽻𖾏，𖼅𖽡 𖽂𖽪𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏， 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏． 31 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽙𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼊𖽡𖾏 𖼈𖽳𖾏 𖼋𖽘𖾏． 32 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏，𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖽃𖽔𖾏 ． 33 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼺𖽘𖾏 ，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏．
34 𖼅𖽡 𖼾𖽨𖾏 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏，𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼅𖽡 𖽐𖼮𖽙𖾑 𖼑𖽻𖾏、 𖼢𖽻 𖼮𖽡𖽠𖾑 ，𖽐𖼎𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼊𖽳𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼑𖽪𖾏 𖼡𖽑𖽹𖾐 𖼮𖽘𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏， 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖽐𖼮𖽡𖽻𖾏 𖼅𖽡． 35 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑，𖼽𖽔𖾑 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑． 36 𖼅𖽡 𖼽𖽔𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼯𖽷𖾏 𖼻𖽙𖾑，𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖼟𖽙𖾏 𖼅𖽡 𖼸𖽠𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼗𖽙𖾏 𖼐𖽡𖾑，𖼊𖽻𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐．”
