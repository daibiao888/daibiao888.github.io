𖼽𖽙 𖼀𖽳 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 :
2 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼳𖽷𖾏 , 𖼟𖽡 𖼮𖽡𖽻𖾐 𖽐𖼀𖽘𖾏 !
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 , 𖼟𖽡 𖼮𖽡𖽻𖾐 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 !
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽡 𖼮𖽡𖽻𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 ! 𖼟𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 .
4 𖼟𖽡 𖼻𖽡𖽝𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼗𖽙𖾏 ?
𖼡𖽔 𖼋𖽱𖾏 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼖𖽺𖾑 ?
5 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 .
6 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼙𖽪 ,
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼷𖽑𖽪𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 .
7 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼊𖽑𖽔𖾑 𖼎𖽻𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 ,
𖼞𖽑𖽡𖽠𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽘𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏 .
8 𖼞𖽑𖽡𖽠𖾏 𖽃𖽻𖾐 𖼡𖽑𖽠𖾑 𖼎𖽻𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽔𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 ,
𖼈𖽡𖽝𖾏 𖽃𖽻𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼑𖽡𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 .
9 𖼐𖽱𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 ,
𖼞𖽑𖽡𖽠𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼊𖽑𖽔𖾑 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 .
10 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖼇𖽺𖾐 𖼊𖽠𖾏 𖽐𖼚𖽜𖾏 𖼗𖽙𖾏 ,
𖼸𖽙 𖼚𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽐𖼎𖽙 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽐𖼚𖽜𖾏 .
11 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 .
12 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 (“ 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼩𖽪𖾏”) ,
𖼐𖽱𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼦𖽔𖾐 𖼀𖽳 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏 .
13 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼀𖽘𖾏 𖼟𖽙𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼮𖽑𖽦𖾐 𖼻𖽻𖾏 ,
𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽘 𖼀𖽜𖾑 𖼇𖽺𖾏 𖽃𖽡𖾑 .
14 𖼑𖽔𖾏 𖼘𖽪！𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼊𖽔𖾐 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐！
𖼐𖽱𖾏 𖼺𖽙𖾐 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼖𖽺𖾑 ?”
