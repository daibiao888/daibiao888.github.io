𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼐𖽱𖾏, 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼏𖽘 𖼀𖽜𖾑.
2 𖼟𖽡 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏,
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽑𖽪𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏, 𖼺𖽱𖾏 𖼮𖽡𖽝𖾐 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼾𖽘𖾏.
3 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑.
4 𖼅𖽡 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏,
𖼑𖽜 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖽐𖼮𖽡𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏, 𖼨𖽑𖽢𖾏 𖼨𖽳𖾑𖽑 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼙𖽔 𖼚𖽙𖾏！
 
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖼑𖽡𖾏 𖼉𖽺𖾏,
𖽃𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏.
6 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼗𖽔 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽐𖼞𖽔𖾏,
𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼊𖽻𖾏 𖼀𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼷𖽻𖾏,
𖼯𖽔 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡.
 
7 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖼟𖽡 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
8 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽑𖽪𖾏 𖼎𖽺𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏, 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖼗𖽙𖾏 𖼑𖽘,
𖼖𖽪𖾑 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼖𖽔𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 .
9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼟𖽙𖾏 𖼑𖽘 𖼗𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼟𖽡 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
10 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼳𖽻𖾏 𖽐𖼊𖽻𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏.
 
11 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏,
𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼄𖽻𖾏 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼸𖽻𖾏.
12 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼏𖽙 𖼀𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼀𖽜𖾑 𖼗𖽠𖾏 𖼋𖽘𖾏,
𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼇𖽺𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏.
13 𖼅𖽡 𖼀𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼾𖽘𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖽐𖼡𖽔𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼾𖽜𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐, 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼚𖽻𖾐 𖽐𖼡𖽔𖾐 𖼾𖽜𖾏 𖼞𖽪𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.
14 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖽐𖼚𖽙 𖼖𖽠𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑,
𖼺𖽱𖾏 𖽐𖼀𖽪𖾏 𖼽𖽘𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼺𖽔𖾐 𖼄𖽞.
 
15 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐,
𖼀𖽔𖾑 𖼳𖽔𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼇𖽻𖾏 𖼩𖽜 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐.
16 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼇𖽻𖾏 𖼩𖽜 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼘𖽪！
𖽃𖽔𖾏 𖼯𖽤𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼀𖽡𖾏 𖼛𖽘𖾏 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 𖼘𖽪?
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑, 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏！
 
17 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼷𖽑𖽡𖽝𖾐 𖼷𖽑𖽡𖽝𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏,
𖼮𖽵𖾑 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
18 𖼟𖽡 𖼙𖽔 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑,
𖼟𖽡 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏,
𖼟𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼯𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐.
19 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑,
