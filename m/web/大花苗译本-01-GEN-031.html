22 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐, 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑: “𖼽𖽔𖾐 𖼞𖽙 𖼏𖽘 𖼀𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏.” 23 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽻𖾏 𖼘𖽠𖾑, 𖼘𖽠𖾑 𖼅𖽻𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐, 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼘𖽠𖾑 𖼯𖽢 𖼋𖽘𖾏. 24 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏 𖼻𖽻𖾏 𖼐𖽑𖽡𖾏 𖼮𖽷𖾑.”
25 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼖𖽠𖾑 𖼯𖽢 𖼽𖽔𖾐 𖼞𖽙． 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐； 𖼖𖽔𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐. 26 𖼖𖽔𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼘𖽪? 𖼟𖽡 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼑𖽡𖽝 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼷𖽑𖽹𖾏 𖼅𖽡 𖼙𖽠𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐. 27 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼏𖽘 𖼀𖽜𖾑, 𖼑𖽡𖽝 𖼙𖽠𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼚𖽙𖾏、𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏、𖽐𖼊𖽻𖾏 𖽐𖼎𖽔𖾏、𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼺𖽘𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼺𖽷𖾐 𖼅𖽻? 28 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽱𖾑 𖼘𖽔𖾑𖽑 𖼊𖽑𖽡𖽝𖾐 𖽐𖼷𖽑𖽹𖾏 𖼡𖽑𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼡𖽔 𖼏𖽔 𖼋𖽡𖽝. 29 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼦𖽺𖾏 𖼟𖽡, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔 𖼄𖽑𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:‘ 𖼟𖽡 𖼽𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏 𖼻𖽻𖾏 𖼐𖽑𖽡𖾏 𖼮𖽷𖾑.’ 30 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼯𖽡𖽙𖾏 , 𖼦𖽡𖾐 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼷𖽷𖾐 𖼈𖽡𖽝𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼎𖽺𖾏 𖼑𖽡𖽝 𖼞𖽪𖾑 𖼋𖽢𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼙𖽠𖾏 𖼘𖽪?” 31 𖼽𖽔𖾐 𖼞𖽙 𖼊𖽡𖾐 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼦𖽔𖾏 𖼟𖽡 𖽐𖼷𖽑𖽹𖾏 𖼙𖽠𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼏𖽘 𖼀𖽜𖾑. 32 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼟𖽡 𖼋𖽢𖾏 𖼮𖽵𖾑 𖼳𖽺, 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼗𖽙𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏； 𖼷𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽡 𖼯𖽜𖾏 𖽃𖽡𖾐 𖼯𖽜𖾏, 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼋𖽪 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼁𖽡𖽝, 𖼯𖽡𖾏 𖼞𖽑𖽡𖽠𖾏 𖼅𖽻𖾏．” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼑𖽡𖽝 𖼋𖽢𖾏 𖼁𖽠𖾏 𖽃𖽡𖾑.
33 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙、𖼖𖽡𖾏 𖼽𖽔𖾏, 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐, 𖼊𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏, 𖼎𖽑𖽻𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼑𖽡𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐. 34 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼞𖽑𖽡𖽠𖾏 𖼋𖽢𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼁𖽠𖾏 𖼈𖽺𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼘𖽙 𖼊𖽑𖽙𖾐 𖼑𖽡𖾏 𖼡𖽘𖾐 𖽂𖽙 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽡𖽝𖾏, 𖼐𖽱𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑, 𖼖𖽔𖾑 𖼀𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖽐𖼷𖽻𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐, 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏. 35 𖼖𖽔𖾑 𖼮𖽡𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼐𖽑𖽡𖾏: “𖼷𖽔𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏. 𖼮𖽑𖽙 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼋𖽢𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼁𖽠𖾏, 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼗𖽙𖾏.
