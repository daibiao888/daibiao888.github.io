𖼽𖽙 𖼐𖽔 𖼐𖽡𖽙𖾐 𖽐𖼀𖽳𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼮𖽵𖾑
2
𖼽𖽙 𖼐𖽔 𖼐𖽡𖽙𖾐 𖽐𖼀𖽳𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 2 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼟𖽡 𖼎𖽺𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑,
𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 ,
𖼟𖽡 𖼎𖽺𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
3 𖼟𖽡 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐 ,
𖼟𖽪𖾏 𖼖𖽪𖾑 𖼡𖽘𖾐 𖼳𖽡𖽠𖾑 ,
𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑,
𖼟𖽡 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙 𖼈𖽳𖾑 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼺𖽘𖾏.
4 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏： 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼗𖽠𖾏 𖼈𖽡𖽝𖾏 ,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏.
5 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼞𖽪𖾑, 𖼽𖽔𖾑 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼞𖽪𖾑 ,
𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏，
𖼳𖽡𖽠𖾑 𖽐𖼡𖽜𖾐 𖼞𖽑𖽡𖾏 𖼉𖽜𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏.
6 𖼞𖽪𖾑 𖽐𖼡𖽜𖾏 𖼸𖽙 𖼎𖽻𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏,
𖼊𖽡𖾐 𖼑𖽡𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼡𖽙𖾏 𖼮𖽡𖽝𖾏.
𖽃𖽔𖾏! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪!
𖼟𖽡 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼮𖽨𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
7 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽻 𖼼𖽙 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
𖼞𖽪𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏,
𖼸𖽙 𖼚𖽔𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼻𖽡𖽝𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝 ,
𖼇𖽳𖾑 𖼡𖽔𖾏 𖼺𖽘𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼋𖽔𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼟𖽡 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏.
𖼯𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. ”
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽪𖾑 𖽐𖼀𖽳𖾏, 𖽐𖼀𖽳𖾏 𖼎𖽺𖾏 𖽐𖼊𖽡𖾏 𖼽𖽙 𖼐𖽔 𖼎𖽻𖾏 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐.
