𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼊𖽡𖾐 𖼐𖽔 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼅𖽡. 2 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑, 𖼳𖽡𖾑 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽔𖾐 𖼄𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏, 𖼊𖽔𖾑 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼺𖽘𖾏. 3 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽘𖾏 𖼮𖽡𖽝𖾏 𖼉𖽳 𖽐𖼮𖽑𖽠𖾏𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼊𖽡𖾐 𖼐𖽔, 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼖𖽜𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽡𖾑, 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼐𖽱𖾏. 4 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼦𖽔𖾐 𖼄𖽙 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙 𖼟𖽡 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼖𖽜𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽡𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼐𖽤𖾐.”
5 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽑𖽹𖾏 𖼊𖽡𖾐 𖼐𖽔 𖼊𖽻𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼺𖽘𖾏, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼅𖽡 𖼾𖽨 𖼯𖽦𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐, 𖼽𖽔𖾐 𖼞𖽙 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏, 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽷𖾐 𖼗𖽙. 6 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼦𖽔𖾐 𖼄𖽙 𖼋𖽔𖾏 𖽐𖼎𖽑𖽦𖾑 𖼽𖽔𖾐 𖼞𖽙, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏. 7 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼅𖽡 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼗𖽙𖾏, 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼚𖽘𖾐, 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏, 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏.
8 𖼦𖽔𖾐 𖼄𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼅𖽡 𖼖𖽜𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽡𖾑, 𖼮𖽑𖽙 𖼅𖽡 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼐𖽤𖾐. 9 𖼅𖽡 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖼮𖽑𖽻𖾐 𖼟𖽙； 𖼅𖽡 𖼄𖽔𖾐 𖽐𖼷𖽑𖽹𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐, 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖽐𖼷𖽑𖽹𖾏. 10 𖼅𖽡 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐，𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑, 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏, 𖼅𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.” 11 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖽐𖼎𖽙 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏: “𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏, 𖼅𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡. 12 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼷𖽙𖾏、𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡 , 𖼯𖽔 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽐𖼷𖽑𖽹𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼐𖽤𖾐.”
13 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼅𖽡 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼊𖽡𖾐 𖼐𖽔 𖼊𖽻𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼺𖽘𖾏, 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽡𖾐 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼦𖽔𖾐 𖼄𖽙， 14 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼄𖽔𖾐 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼐𖽤𖾐. 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏. 15 𖼮𖽔 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾐 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼯𖽪𖾏, 𖼟𖽪𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼾𖽠 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏, 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐. 16 𖼀𖽡𖾐 𖼄𖽔𖾐 𖽐𖼷𖽑𖽹𖾏 𖼎𖽻𖾏 𖼅𖽡, 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼷𖽙𖾏 𖼅𖽡 𖽐𖼷𖽑𖽹𖾏, 𖼀𖽡𖾐 𖼯𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐, 𖽃𖽔𖾐 𖼸𖽻 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏. 17 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏, 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼅𖽻𖾏 𖼋𖽘𖾏.”
