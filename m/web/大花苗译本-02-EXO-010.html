𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 , 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽻 𖼷𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏 , 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑， 2 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 , 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖽃𖽡𖾑 , 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 , 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .”
3 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼾𖽨𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼺𖽷𖾐 𖼖𖽺𖾑 ? 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 , 𖼻𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑 . 4 𖼟𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 , 𖼀𖽡 𖼟𖽡 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼷𖽷 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖽐𖼚𖽙， 5 𖽐𖼷𖽻𖾑 𖼀𖽪𖾑 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼖𖽔𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼑𖽘 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽐𖼊𖽻𖾏 𖼳𖽙𖾏 𖽃𖽡𖾑 , 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 . 6 𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽐𖼎𖽙 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽻 𖼷𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼷𖽷𖾑 𖼷𖽷 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼋𖽘𖾏 , 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼗𖽙𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼑𖽔𖾏 , 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼖𖽡𖾏 𖼐𖽡𖾑 .” 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼇𖽳𖾑 𖼇𖽔 𖼘𖽻𖾐 𖼺𖽘𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏.
