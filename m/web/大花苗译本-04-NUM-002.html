𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽑𖽡𖾏 : 2 “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 𖼀𖽡𖾑 𖼊𖽡𖾐 , 𖼐𖽡𖽙𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼺𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼊𖽱𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼮𖽪𖾐 𖼾𖽡 .
3 “𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 𖼏𖽙 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼮𖽪𖾐 𖼾𖽡 , 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 . 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼐𖽔 𖼊𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼐𖽔 𖼳𖽪𖽝𖾏 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 4 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼉𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐 𖼚𖽻𖾏 𖼀𖽔𖾏 𖼗𖽜𖾏 . 5 𖼐𖽡𖽙𖾐 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼮𖽪𖾐 𖼾𖽡 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼳𖽡𖾏 𖽃𖽡𖾑． 𖼅𖽔𖾏 𖼺𖽪𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼐𖽔 𖼊𖽑𖽗𖾐 𖼐𖽡𖽝 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 6 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼉𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 . 7 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼳𖽡𖾑 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔 𖽃𖽔𖾏 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 8 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼉𖽻𖾏 𖼳𖽤𖾏 𖼷𖽑𖽡𖽝𖾐 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 .
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼸𖽙𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼾𖽡𖾏 𖼉𖽻𖾏 𖼚𖽻𖾏 𖼷𖽑𖽡𖽝𖾐 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 , 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼺𖽪𖾐 𖼁𖽱 𖽃𖽡𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 .
10 “𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼖𖽨 𖼀𖽡𖽝𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 . 𖼅𖽔𖾏 𖼳𖽷𖾏 𖼊𖽨𖾑 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽱 𖽃𖽔𖾏 𖼖𖽨 𖼀𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 11 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼉𖽻𖾏 𖼚𖽻𖾏 𖼷𖽑𖽡𖽝𖾐 𖼀𖽜𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 . 12 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼮𖽪𖾐 𖼾𖽡 𖼟𖽪𖾏 , 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 . 𖼺𖽪𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼊𖽹𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽷𖾏 𖼖𖽪𖾏 𖼄𖽡𖽝 𖽃𖽔𖾏 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 13 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼉𖽻𖾏 𖼯𖽢𖾏 𖼷𖽑𖽡𖽝𖾐 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 . 14 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼮𖽢𖾑 𖼊𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 . 𖼊𖽨𖾑 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼺𖽔𖾐 𖽃𖽔𖾏 𖼮𖽢𖾑 𖼊𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼉𖽻𖾏 𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐 𖼚𖽻𖾏 𖼀𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 . 16 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼖𖽨 𖼀𖽡𖽝𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼀𖽜𖾐 𖼉𖽻𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 . 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼸𖽻 𖽃𖽡𖾐 𖼸𖽻 𖽃𖽡𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 .
17 “𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 , 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼾𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 . 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼮𖽪𖾐 𖼾𖽡 , 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏， 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 .
18 “𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼖𖽪𖾑 𖼾𖽡 𖼑𖽡𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 . 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼦𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 19 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼉𖽻𖾏 𖼖𖽡 𖼀𖽜𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 . 20 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼳𖽡𖾏 𖽃𖽡𖾑 . 𖼀𖽡𖾐 𖼊𖽔𖾏 𖼳𖽱 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼮𖽢𖾑 𖼄𖽔𖾐 𖼖𖽡𖽝 𖽃𖽔𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 21 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 . 22 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼮𖽡𖾑 𖼊𖽙𖾑 𖼐𖽡 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼊𖽗𖾏 𖽃𖽔𖾏 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 23 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼉𖽻𖾏 𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 . 24 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽙𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼖𖽪𖾑 𖼾𖽡 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼉𖽻𖾏 𖼾𖽡𖾏 𖼷𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏， 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼷𖽷𖾐 𖼸𖽻 𖽃𖽡𖾐 𖼸𖽻 𖽃𖽡𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 .
25 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽗𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼑𖽡𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 . 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼳𖽔𖾑 𖼊𖽹𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼽𖽡𖾑 𖼳𖽡𖽝𖾏 𖽃𖽔𖾏 𖼊𖽗𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 26 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 𖼳𖽤𖾏 𖼀𖽔𖾏 𖼗𖽜𖾏 . 27 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼮𖽪𖾐 𖼾𖽡 , 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 . 𖽃𖽳 𖼖𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽑𖽔 𖼮𖽡𖽝 𖽃𖽔𖾏 𖼽𖽔𖾏 𖼳𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 28 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼉𖽻𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽜𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 . 29 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼽𖽡𖾑 𖼐𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 30 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼉𖽻𖾏 𖼷𖽷𖾐 𖼷𖽑𖽡𖽝𖾐 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 . 31 𖼊𖽗𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼾𖽱 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼀𖽜𖾐 𖼉𖽻𖾏 𖼳𖽤𖾏 𖼷𖽑𖽡𖽝𖾐 𖼚𖽻𖾏 𖼀𖽔𖾏 𖼗𖽜𖾏， 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 . ”
