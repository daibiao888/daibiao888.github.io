（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 13：1 - 14：1）
15
𖼐𖽡 𖼀𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽘𖾑 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 2 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏． 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖼮𖽢𖾑, 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏. 3 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽘𖾑 𖽃𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 4 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼑𖽜 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼑𖽪𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼅𖽔𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐. 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼦𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 , 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏, 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏. 6 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽘𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏,𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼯𖽪 𖼯𖽪𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏. 7 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽘𖾑 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 《𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏》 𖼮𖽡𖽝𖾏, 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽘𖾑 𖼯𖽪 𖼯𖽪𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏. 8 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽘𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏, 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻, 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 15：16 - 16：6）
