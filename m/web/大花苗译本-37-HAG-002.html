6 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼗𖽡𖾏 𖼐𖽡𖾑 : “𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼗𖽡𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼎𖽺𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐、 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼎𖽙 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐. 7 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼑𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽹𖾑 𖼋𖽔𖾏 ( 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼡𖽑𖽙𖾑 𖼉𖽳 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼗𖽙𖾏 𖼸𖽙”), 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 , 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 . 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.” 8 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝 , 𖼞𖽪𖾐 𖼖𖽔𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝 . 9 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 , 𖼖𖽔𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼘𖽑𖽙𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏. 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼺𖽘𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 . 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
𖼏𖽵 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
