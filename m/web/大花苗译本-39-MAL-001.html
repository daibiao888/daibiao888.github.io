1
𖼅𖽜 𖼡𖽑𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼄𖽔𖾐 𖼖𖽔𖾑 𖼮𖽡𖾑 𖼡𖽑𖽔𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 .
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼙𖽔 𖽐𖼮𖽑𖽠𖾏 𖼅𖽡 .” 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖽐𖼮𖽑𖽠𖾏 𖼀𖽡𖾐 𖼖𖽺𖾑?” 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼺𖽻𖾐 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼽𖽔𖾐 𖼞𖽙, 3 𖼳𖽪𖾐 𖼽𖽡𖾑 𖼺𖽻𖾐 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏, 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽃𖽡𖾑 𖼚𖽡𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏.” 4 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼈𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼷𖽑𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏.” 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : “𖼯𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏.” 5 𖼅𖽡 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼉𖽺𖾏 𖼢𖽻𖾏, 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐！”
𖼖𖽔𖾏 𖼡𖽑𖽔𖾐 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏
6 𖼅𖽡 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑 𖼘𖽪 ! 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏: “𖼊𖽪𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼉𖽺𖾏, 𖼡𖽑𖽜𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑. 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼉𖽺𖾏 𖼋𖽘𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑? 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽵𖾑 𖼋𖽘𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?” 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼖𖽺𖾑?” 7 𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼏𖽘 , 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼖𖽺𖾑? ” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽔𖾏 𖽐𖼮𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼸𖽱𖾏 𖼞𖽡 𖼏𖽻𖾏 . 8 𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽡 𖼛𖽜 𖼖𖽑𖽘𖾑 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ? 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽃𖽡𖾑 𖼖𖽑𖽘𖾑 , 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ? 𖼟𖽡 𖼖𖽑𖽘𖾑 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑, 𖼐𖽱𖾏 𖼋𖽡𖽝𖾏 𖼙𖽔 𖼟𖽡 , 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼘𖽪? 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
