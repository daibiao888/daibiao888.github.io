𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱 𖼏𖽙, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼽𖽔𖾏 𖼮𖽡 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐, 𖼟𖽡 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏.” 2 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼮𖽡 𖼐𖽑𖽡𖾏: “𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼀𖽻𖾐.” 𖼽𖽔𖾏 𖼮𖽡 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼟𖽡 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑.”
𖼺𖽻𖾐 𖼘𖽙 𖼚𖽔𖾏 𖼽𖽡𖽾𖾐 · 𖼊𖽙𖾑 𖽃𖽳𖾐 𖼁𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽜𖾐
3 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼐𖽱𖾏 𖼐𖽡𖽝𖾑 𖼮𖽑𖽡𖾏, 𖼊𖽱𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼄𖽔𖾐 𖼮𖽡𖽝𖾏, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽐𖼚𖽙. 𖼺𖽻𖾐 𖼘𖽙 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼸𖽷𖾑 𖼅𖽪𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐. 4 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱, 𖼋𖽔𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽪𖾑 𖼐𖽡𖽝𖾏 𖼮𖽪𖾐 𖼾𖽡, 𖼺𖽻𖾐 𖼘𖽙 𖼦𖽡 𖼸𖽱 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽡 𖼀𖽙𖾑 𖼮𖽪𖾐 𖼾𖽡. 5 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼏𖽙 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏. 6 𖼺𖽻𖾐 𖼘𖽙 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼡𖽜𖾑 𖼷𖽹𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽂𖽪𖾑 𖼖𖽡𖽾, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼊𖽡𖾐 𖼐𖽱𖾏. 7 𖼺𖽻𖾐 𖼘𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼅𖽻𖾏 𖼑𖽪 𖼐𖽱𖾏.” 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽡𖽙𖾐 𖼽𖽡𖽾𖾐 · 𖼊𖽙𖾑 𖽃𖽳𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐.” 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽻𖾐 𖼘𖽙 𖼘𖽑𖽪𖾑 𖼺𖽘𖾏 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑, 𖼎𖽺𖾏 𖼐𖽑𖽘𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽑𖽙𖾏, 𖼨𖽑𖽳𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽡𖽙𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏, 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽢𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼐𖽜𖾐 𖼸𖽷𖾏 𖼅𖽪𖾏 𖼄𖽔𖾐 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 𖽃𖽡𖾑, 𖼑𖽜 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖼟𖽙𖾏 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏.” 9 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼐𖽜𖾐 𖼸𖽷𖾏 𖼅𖽪𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐. 𖼟𖽡 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼋𖽔 𖼖𖽺𖾑?”
