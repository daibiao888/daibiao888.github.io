𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 .
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑 𖽐𖼎𖽙 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏！
 
3 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼘𖽪！ 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽺𖾑?
𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽺𖾑?
4 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼊𖽪𖾐 𖼋𖽻 𖼑𖽡𖾏 𖼺𖽪𖾐 𖼈𖽙𖾏 𖼟𖽪𖾏 𖽐𖼮𖽳𖾏
𖽐𖼎𖽙 𖼘𖽙 𖼊𖽑𖽺 ( “𖼘𖽙 𖼊𖽑𖽺 ” 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼊𖽑𖽪𖾑 𖽐𖼷𖽷𖾏 ) 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡 𖼋𖽠𖾏.
 
5 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼄𖽡𖾐 𖼳𖽳 ,
𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼊𖽔 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼚𖽙, 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！
6 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏
𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 .
7 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏,
𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖽂𖽪𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼁𖽔𖾏 𖼏𖽙 .
