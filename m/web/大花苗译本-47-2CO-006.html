“𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏， 𖼞𖽪𖾑 𖼙𖽔 𖼑𖽻 𖼊𖽡𖾐 𖼟𖽡 𖼋𖽘𖾏，
𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 ， 𖼞𖽪𖾑 𖼙𖽔 𖽐𖼀𖽘𖾏 𖼟𖽡 𖼋𖽘𖾏 ．”
𖼑𖽔𖾏 𖼘𖽪！ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏， 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏． 3 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼡𖽔 𖼖𖽡𖾏 𖼋𖽱𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏， 𖽐𖼮𖽑𖽹𖾏 𖼋𖽡𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑； 4 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏， 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼀𖽡𖾐 𖼾𖽨𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏， 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 、 𖼳𖽻𖾏 𖽐𖼊𖽻𖾏、 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏、𖼑𖽻𖾏 𖽃𖽦𖾐、 5 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏、 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏、 𖽐𖼷𖽑𖽙𖾑 𖼮𖽑𖽙、 𖼘𖽑𖽻𖾐 𖽐𖼡𖽔𖾏、 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼀𖽱𖾏、 𖼊𖽪𖾏 𖼡𖽜𖾑， 6 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏、𖼀𖽻𖾐 𖼟𖽳、 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝、 𖼅𖽳𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐、 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼛𖽠𖾏 𖼚𖽑𖽡𖾏、 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼮𖽑𖽵𖾑、 7 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝、 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼏𖽙 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏、 8 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑、 𖼊𖽻𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼊𖽻𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏、 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝、 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽻𖾐 𖼯𖽡𖽙、 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼯𖽡𖽙 𖼺𖽷𖾐 、𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼛𖽙𖾏 𖼳𖽦𖾑 𖼺𖽷𖾐、 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑、 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏、 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏，𖼈𖽡𖽝𖾏 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼯𖽡𖽙．
11 𖼞𖽙𖾑 𖼖𖽡 𖽐𖼊𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！ 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼅𖽡， 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝， 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖼯𖽡𖽙𖾏． 12 𖼅𖽡 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼀𖽡𖾐， 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏． 13 𖼅𖽡 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼇𖽘𖾑 𖼯𖽡𖽙𖾏 𖼀𖽡𖾐， 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼐𖽑𖽡𖾏．
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼞𖽱𖾑 𖽃𖽡𖾐 𖼸𖽱𖾏 𖼖𖽡𖾑 𖼡𖽔𖾐
14 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼦𖽡𖾐 𖽐𖼎𖽙𖾏 𖼞𖽱𖾑 𖽃𖽡𖾐 𖼸𖽱𖾏 𖼖𖽡𖾑 𖼡𖽔𖾐． 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼷𖽷𖾐 𖼘𖽪？ 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽐𖼎𖽙 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼘𖽪？ 15 𖼮𖽡𖾑 𖼊𖽪 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼖𖽡𖽝 （“𖼀𖽡𖾐 𖼖𖽡𖽝” 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼷𖽷𖾏）𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏 𖼘𖽪？ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼮𖽵𖾑， 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼘𖽪？ 16 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼊𖽺𖾐 𖼁𖽠𖾏 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼺𖽱𖾏 𖼘𖽪？ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏：
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐， 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，

17 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼮𖽵𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
“𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 ，

𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑，

18 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼅𖽡 𖼉𖽺𖾏，
𖼅𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏．”
𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏．
