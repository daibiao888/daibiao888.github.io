𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻𖾏 𖽐𖼮𖽑𖽘𖾑
10 “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼡𖽔 𖼋𖽱𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼷𖽜𖾑 𖼚𖽑𖽪𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑 , 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼇𖽳𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏. 11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 , 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼮𖽑𖽘𖾑 𖽐𖼚𖽙． 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖽐𖼮𖽑𖽘𖾑 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼅𖽡 , 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼑𖽜 𖼅𖽡 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼚𖽠𖾏 𖼾𖽔𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽘𖾑 𖽐𖼚𖽙 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑， 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 . 12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 , 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 . 13 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼡𖽔 𖼋𖽱𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼮𖽔𖾑 𖽐𖼊𖽻𖾏 𖼊𖽻𖾏 𖼑𖽘𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼢𖽻 , 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼈𖽳𖾑 𖽐𖼀𖽙 .
14 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖼳𖽦𖾑 , 𖼐𖽱𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼮𖽑𖽘𖾑 𖽐𖼚𖽙 . 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑 , 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 . 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼑𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼇𖽳𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 . 15 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼋𖽔 𖼾𖽨𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼄𖽔𖾐 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 , 𖼽𖽔𖾑 𖽐𖼷𖽑𖽔𖾏 𖼾𖽨𖾏 𖼞𖽡𖾑 𖼛𖽔𖾏 𖼮𖽑𖽙𖾏 , 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼸𖽙 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 , 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 16 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼷𖽑𖽔𖾏 𖼮𖽑𖽙𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 .”
