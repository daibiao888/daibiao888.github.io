𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑
（𖼊𖽑𖽹𖾏 12：1-8；𖼞𖽑𖽙𖾐 2：23-28）
6
𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 ，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽡𖽻𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼳𖽻𖾐，𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖼊𖽪𖾏 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐 𖼳𖽻𖾐，𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼦𖽡𖾏 𖼄𖽙𖾏 𖼑𖽻𖾏． 2 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏？” 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼮𖽑𖽹𖾐 𖼋𖽘，𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼅𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼟𖽡𖽠 𖼺𖽷𖾐 𖼘𖽪𖾏？ 4 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 ，𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼑𖽻𖾏，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼑𖽻𖾏，𖼋𖽡𖾏 𖼳𖽡𖽝𖾐 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏，𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼑𖽻𖾏．” 5 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽔𖾑
（𖼊𖽑𖽹𖾏 12：9-14；𖼞𖽑𖽙𖾐 3：1-6）
6 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏． 7 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖽐𖼎𖽙 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼊𖽔𖾐 𖼡𖽔 𖼅𖽔 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼋𖽡𖽝𖾏 𖼞𖽑𖽙𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼞𖽑𖽙𖾏，𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼡𖽘𖾐 𖼅𖽻𖾏 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼐𖽱𖾏． 8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“ 𖼳𖽠𖾑 𖼗𖽙𖾏，𖼳𖽠𖾑 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐．” 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼋𖽘𖾏． 9 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼑𖽪 𖼅𖽡，𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏，𖼮𖽨𖾏 𖼳𖽦𖾑 𖼐𖽑𖽙𖾏 𖼳𖽦𖾑，𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼼𖽘𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？” 10 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐，𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“𖼳𖽤𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏．” 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼳𖽤𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 ，𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏． 11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼊𖽠𖾏 𖽐𖼎𖽙𖾐，𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 ．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏
（𖼊𖽑𖽹𖾏 10：1-4；𖼞𖽑𖽙𖾐 3：13-19）
12 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 13 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏，𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐，𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼄𖽻𖾏． 14 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼄𖽞 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼀𖽡𖾐 𖼊𖽳，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼊𖽳 𖼖𖽡𖽝，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼽𖽙 𖼦𖽗 ， 𖼇𖽺𖾑 𖼖𖽡 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼊𖽙𖾑 𖼘𖽙 𖼄𖽹𖾐， 15 𖼄𖽔𖾐 𖼊𖽑𖽹𖾏 𖽐𖼎𖽙 𖼊𖽙𖾑 𖼄𖽔𖾐，𖼽𖽔𖾏 𖼖𖽳 𖼇𖽺𖾑 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙，𖽐𖼎𖽙 𖼇𖽞𖾏 𖼵𖽪𖽺𖾏 𖼊𖽗𖾐𖼳𖽡𖾑 𖼄𖽞 ， 16 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽨 𖼊𖽔𖾏，𖽐𖼎𖽙 𖼮𖽢𖾑 𖼖𖽡𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼟𖽪𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑．
𖼺𖽹𖾑𖼨𖽑𖽪𖾏𖼊𖽪𖾐𖽐𖼎𖽙𖾏
（𖼊𖽑𖽹𖾏 4：23-25）
17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼡𖽜𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼗𖽙，𖼳𖽠𖾑 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏， 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼳𖽠𖾑，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼎𖽑𖽻𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐，𖼊𖽑𖽡𖽝𖾐 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙、 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 𖼋𖽔𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼑𖽻 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏，𖼎𖽺𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼞𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐． 18 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼏𖽻 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖽐𖼊𖽠𖾏 𖼮𖽡𖽝𖾏，𖼖𖽔𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼊𖽑𖽡𖽝𖾐． 19 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽠𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼞𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽙𖾏 𖼻𖽻𖾏 ．
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
（𖼊𖽑𖽹𖾏 5：1-12）
20 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：
“𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽡𖽝．
21 𖼅𖽡 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏．
𖼅𖽡 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏 𖽃𖽔𖾏 𖽐𖼷𖽑𖽷𖾏．
22 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼳𖽪𖾐 𖼅𖽡，𖼗𖽔𖾏 𖼅𖽡 𖼺𖽘𖾏 ， 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼅𖽡，𖼞𖽑𖽡𖽠𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼇𖽳𖾏 𖼅𖽡 𖽐𖼷𖽷𖾏 𖼺𖽘𖾏，𖼅𖽡 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏． 23 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼟𖽡𖾏 𖼚𖽑𖽡𖽝𖾏 𖼚𖽜𖾑𖽑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐，𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏．
24 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼙𖽔 𖼊𖽻𖾏 𖼅𖽡 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼋𖽘𖾏．
25 𖼅𖽡 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 ．
𖼅𖽡 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼙𖽔 𖼚𖽙𖾏 𖽃𖽔𖾏 𖽐𖼷𖽑𖽷𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏．
26 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼻𖽻𖾏，𖼅𖽡 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． ”
𖼾𖽡𖾏𖼻𖽘𖾏𖼻𖽻𖾏𖼮𖽑𖽡𖽠𖾏
𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑
（𖼊𖽑𖽹𖾏 5：38-48；7：12）
27 “𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑，𖼅𖽡 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼅𖽡，𖼅𖽡 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏． 28 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼅𖽡， 𖼅𖽡 𖼽𖽔𖾑 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼅𖽡 ，𖼅𖽡 𖼽𖽔𖾑 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏． 29 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼟𖽡 𖼺𖽘𖾑 𖼚𖽑𖽪𖾏 𖼐𖽡𖾑，𖼽𖽔𖾑 𖼄𖽔𖾐 𖼺𖽘𖾑 𖽃𖽡𖾑 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏．𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽔𖾏 𖼅𖽡 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏， 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼦𖽔𖾏 𖼅𖽡 𖼮𖽑𖽙𖾏 𖼄𖽡𖾐 𖼋𖽱𖾐 𖼊𖽑𖽡𖽝𖾐． 30 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼊𖽜 ， 𖼟𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 ，𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽔𖾏 𖼟𖽡 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼙𖽠𖾏，𖼦𖽡𖾐 𖼅𖽪𖾏 𖼽𖽔𖾑 𖼺𖽷𖾐 𖼗𖽙．
