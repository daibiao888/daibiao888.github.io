𖼟𖽻𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼟𖽻𖾏 𖼖𖽑𖽡𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 2 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽱𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼇𖽔 𖼘𖽻𖾐, 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 3 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：
𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼘𖽪！
𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐．
𖼟𖽡 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽡𖾑 ,
𖼟𖽡 𖼙𖽔 𖼐𖽑𖽡𖾏: 𖼖𖽪𖾑 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝 ,
𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏.
4 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖽐𖼡𖽜𖾏 𖼮𖽡𖽝𖾏 𖼟𖽡 𖼀𖽔𖾐 𖽃𖽔𖾏 𖼯𖽹𖾏 ,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖽐𖼀𖽳𖾏 𖼘𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽔𖾏 𖼷𖽷𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼘𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽔𖾏 𖼷𖽷𖾏 𖽃𖽡𖾑,
𖼎𖽑𖽻𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽤𖾐 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏 .
5 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼀𖽳𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏,
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽻𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽱𖾏.
𖼞𖽪𖾑 𖼄𖽔𖾐 𖼟𖽡 𖼎𖽻𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏, 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼋𖽘 .
