𖼎𖽺𖾏 𖼺𖽻𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏
（𖼎𖽺𖾏 10：1 - 5）
34
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼽𖽔𖾑 𖼸𖽻 𖼋𖽠𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 , 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼟𖽡 𖼙𖽔 𖼮𖽻𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼋𖽡𖾏 𖽃𖽔 𖼅𖽔 𖽐𖼊𖽠𖾑 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼐𖽡𖾑 . 2 𖼀𖽡 𖼟𖽡 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 𖼮𖽡𖽝𖾏 , 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 . 3 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 , 𖽃𖽡𖾐 𖼎𖽻𖾐 𖼙𖽱 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼞𖽙𖾏 𖼎𖽻𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖽐𖼀𖽙𖾏 𖼑𖽨𖾐 𖼑𖽻𖾏 𖽐𖼡𖽜𖾐 .” 4 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼸𖽻 𖼋𖽠𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 , 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐． 𖼐𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 ,𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 . 5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼚𖽙 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 , 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 . 6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽑𖽡𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 :
“𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 ,
𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 、𖼅𖽔𖾏 𖼅𖽳𖾏，
𖼦𖽡𖾐 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 ,
𖼎𖽺𖾏 𖼅𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 .
7 𖼑𖽜 𖼷𖽑𖽡𖽝𖾐 𖼉𖽻𖾏 𖼷𖽑𖽷𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 ,
𖼯𖽔𖾏 𖼾𖽔𖾏、𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖽐𖼎𖽙 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 ,
𖼽𖽔𖾑 𖼖𖽠𖾑 𖼾𖽱 𖼐𖽱𖾏 𖼾𖽔𖾏 ,
𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼸𖽙 𖼊𖽪𖾐 , 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾐 𖼚𖽻𖾐 𖼷𖽑𖽷𖾏 .”
8 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼞𖽑𖽜 𖼑𖽜 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 . 9 𖼐𖽑𖽡𖾏 : “𖼮𖽵𖾑 𖼘𖽪 ! 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏 , 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼅𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏． 𖼎𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼯𖽔𖾏 𖼀𖽡𖾐 𖼾𖽔𖾏 𖽐𖼎𖽙 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽙𖾏 .
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼁𖽠𖾏
（𖼋𖽠𖾏 23：14 - 19；𖼎𖽺𖾏 7：1 - 5；16：1 - 17）
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖼑𖽔𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 , 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼋𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 . 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼚𖽻𖾐 𖼺𖽘𖾐 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 .
11 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖽃𖽡𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏． 𖼑𖽔𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼦𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼀𖽡𖾐 𖼖𖽡𖾏 𖼳𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼳𖽡𖾑 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 . 12 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽪𖾑 𖼸𖽙𖾏 . 13 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼁𖽠𖾏 , 𖽐𖼊𖽙𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐. 14 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 , 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 . 15 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 ,𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 , 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾏 𖼟𖽡 , 𖼟𖽡 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽑𖽘𖾑 . 16 𖼎𖽺𖾏 𖼑𖽜 𖼟𖽡 𖼊𖽪𖾐 𖼷𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 , 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 , 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼖𖽔𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 .
17 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼘𖽡𖾑𖽑 𖼁𖽠𖾏 𖼼𖽘𖾏（“𖼁𖽠𖾏 𖼼𖽘𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏” ）．
18 “𖼟𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 , 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 , 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼀𖽡 𖼖𖽑𖽡𖾏 𖽐𖼚𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼀𖽡 𖼖𖽑𖽡𖾏 𖼐𖽡𖾑 𖽐𖼚𖽙 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼗𖽙𖾏 .
