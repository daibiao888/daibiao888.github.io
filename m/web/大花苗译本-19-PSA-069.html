𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼮𖽵𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 “𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏”𖼳𖽘𖾐.
69
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼈𖽳𖾑 𖼯𖽢 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼚𖽘𖾐.
2 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼁𖽘𖾏 𖽃𖽘𖾑 𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼷𖽠𖾐,
𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼊𖽙𖾐, 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼈𖽳𖾑 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼺𖽘𖾏.
3 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼐𖽔𖾑 𖼯𖽻 𖼚𖽑𖽔𖾏, 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡,
𖼞𖽪𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽃𖽔𖾑 𖼅𖽔 𖼇𖽔 𖼙𖽻 𖼺𖽘𖾏.
 
4 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑, 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑 𖼚𖽻𖾐 𖼇𖽻𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏．
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼦𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏, 𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏.
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼟𖽡 𖼀𖽻𖾐 𖼯𖽡𖽙,
𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼈𖽺𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼟𖽡 .
6 𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽡 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏；
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖼷𖽑𖽜𖾑 𖼟𖽡 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑．
7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖼐𖽔𖾑 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏,
𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
8 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼗𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼐𖽱𖾏 𖽐𖼀𖽙𖾏 𖼊𖽪𖾐 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐.
 
9 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖽐𖼞𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼟𖽡 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼷𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑.
10 𖼞𖽪𖾑 𖼐𖽡𖽝𖾑 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔, 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼗𖽙𖾏 𖼯𖽵𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼋𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼎𖽙𖾑 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
11 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑,
𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼚𖽙𖾏.
12 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼖𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼨𖽑𖽪𖾏.
 
13 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼷𖽻𖾏,
𖼸𖽙𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑！
14 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼁𖽘𖾏 𖽃𖽘𖾑 𖽐𖼞𖽙𖾏,
𖼦𖽡𖾐 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡 𖼋𖽙 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐.
15 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼟𖽙𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼈𖽳𖾑 𖽐𖼀𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑,
𖼦𖽡𖾐 𖼟𖽙𖾏 𖼡𖽘𖾐 𖼳𖽡𖽠𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼞𖽪𖾑,
𖼦𖽡𖾐 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽙𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽙𖾏 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏.
 
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼯𖽡𖽙𖾏,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼷𖽻𖾏, 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑.
17 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼈𖽳𖾑 𖽐𖼀𖽙 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏,
𖼞𖽪𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑，
18 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑, 𖼮𖽨𖾏 𖼎𖽙𖾑 𖼞𖽪𖾑，
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑．
 
19 𖼟𖽡 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏、𖼟𖽡𖾏 𖽐𖼮𖽔𖾏、𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽠𖾏,
𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
20 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽘𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏．
𖼞𖽪𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽻𖾑 𖼅𖽳𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏,
