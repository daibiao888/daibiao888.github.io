𖼽𖽔𖾐 𖼞𖽙 𖽐𖼮𖽷𖾐 𖼖𖽔𖾑 𖼮𖽡𖽝
29
𖼽𖽔𖾐 𖼞𖽙 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 2 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐, 𖼅𖽔𖾏 𖼷𖽷𖾐 𖽐𖼀𖽙𖾐 𖼾𖽘𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽪𖾐 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐, 𖽃𖽔𖾐 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖾑 𖽃𖽻𖾐, 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼡𖽙𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐. 3 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼉𖽺𖾏, 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼦𖽡𖾐 𖼗𖽱𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼺𖽘𖾏 𖼀𖽪𖾐 𖼾𖽘𖾏, 𖼀𖽪𖾐 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼗𖽱𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼡𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼮𖽡𖽝𖾏.
4 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 , 𖼅𖽡 𖼋𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼦𖽔𖾐 𖼖𖽔 𖼋𖽔𖾏.” 5 𖼐𖽱𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏:“ 𖼐𖽔 𖼦𖽳𖾏 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼀𖽻𖾐.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼀𖽻𖾐 𖼯𖽡𖽙 .” 6 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼋𖽡𖽝𖾏 𖼐𖽱𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼘𖽪?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽡𖽝𖾏 𖼋𖽪 . 𖼑𖽔𖾏 𖼘𖽪 ! 𖼐𖽱𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼖𖽠𖾑 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼋𖽔𖾏 𖼋𖽘 .” 7 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼐𖽡𖽙𖾐 𖼺𖽡𖽝𖾐 𖼺𖽷𖾐, 𖼦𖽡𖾐 𖼾𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼦𖽡 𖼸𖽱, 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼀𖽪𖾐 𖼾𖽘𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼾𖽨.” 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼦𖽡𖾐 𖽐𖼮𖽻 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼟𖽙𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼦𖽡 𖼸𖽱 𖽐𖼷𖽻𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼦𖽡𖾐 𖼗𖽱𖾐 𖼺𖽘𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽪𖾐 𖼾𖽘𖾏.”
9 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑, 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼖𖽠𖾑 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼋𖽔𖾏 𖼋𖽘 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼾𖽘𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨 𖼯𖽡𖽙𖾏. 10 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼑𖽡𖽝𖾏 𖼑𖽪𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼑𖽡𖽝𖾏 𖼑𖽪𖾏 𖼑𖽡𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐, 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼗𖽱𖾐 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼺𖽘𖾏, 𖼀𖽪𖾐 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼑𖽡𖽝𖾏 𖼑𖽪𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐. 11 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼡𖽑𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑. 12 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝 , 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼷𖽔𖾑 𖼮𖽑𖽪𖾐, 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖼷𖽔𖾑 𖼊𖽪𖾐, 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼎𖽺𖾏 𖼏𖽘 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐.
13 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼷𖽔𖾑 𖼮𖽑𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼡𖽘𖾐 𖼄𖽻𖾏, 𖼎𖽺𖾏 𖼏𖽘 𖼅𖽻𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 , 𖼁𖽔 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼡𖽑𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏. 𖼽𖽔𖾐 𖼞𖽙 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑. 14 𖼖𖽔𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼾𖽙 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏．” 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏.
𖼽𖽔𖾐 𖼞𖽙 𖼑𖽜 𖼖𖽔𖾑 𖼮𖽡𖽝 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑
15 𖼖𖽔𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖼈𖽡𖽝𖾏 (“𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼞𖽱𖾑 𖼋𖽡𖾏”), 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑? 𖼮𖽑𖽙 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 , 𖼟𖽡 𖼖𖽔𖾑 𖼽𖽔𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽃𖽔𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙?” 16 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼷𖽑𖽹𖾏, 𖼖𖽜𖾑 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏, 𖼖𖽜𖾑 𖼣𖽔𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝. 17 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼖𖽪𖾑 𖽃𖽔 𖼅𖽔 𖼦𖽡𖾐 𖼮𖽨𖾏, 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼺𖽜𖾏 𖼟𖽡𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼻𖽻𖾏 𖼼𖽘𖾏. 18 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝, 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼯𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼖𖽔𖾑 𖼣𖽔𖾏 𖽐𖼷𖽑𖽹𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏.” 19 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 . 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐.” 20 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽜 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏； 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝， 𖼑𖽔𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼐𖽑𖽪𖾐 𖼊𖽔𖾐.
21 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “ 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼀𖽪𖾑 𖼋𖽘𖾏, 𖼮𖽑𖽙 𖼟𖽡 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼐𖽤𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏.” 22 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏, 𖼞𖽑𖽔𖾏 𖼨𖽑𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐. 23 𖼸𖽙 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼄𖽔𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼺𖽘𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙, 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏. 24 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼳𖽡 𖼀𖽑𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼷𖽑𖽹𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑. 25 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑, 𖼽𖽔𖾐 𖼞𖽙 𖽂𖽪𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏 ，𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽺𖾑? 𖼞𖽪𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽜 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼘𖽪? 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼮𖽑𖽵𖾑 𖼞𖽪𖾑?” 26 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽜𖾑 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼺𖽪𖾐 𖼄𖽔𖾐 𖼖𖽜𖾑 𖼣𖽔𖾏 𖽐𖼷𖽑𖽹𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑. 27 𖼟𖽡 𖼽𖽔𖾑 𖼖𖽜𖾑 𖼐𖽡𖾑 𖼀𖽪𖾑 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼡𖽑𖽔𖾏 𖼮𖽑𖽦𖾐, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼟𖽡, 𖼟𖽡 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏.” 28 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼀𖽪𖾑 𖼋𖽘𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼡𖽑𖽔𖾏 𖼮𖽑𖽦𖾐, 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼄𖽔𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽔𖾏 𖼐𖽤𖾐 𖼊𖽑𖽡𖽝𖾐. 29 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼀𖽑𖽡 𖼖𖽔𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼷𖽑𖽹𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑. 30 𖼽𖽔𖾐 𖼞𖽙 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼀𖽱𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏； 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏.

31 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 ( “𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼊𖽻𖾏 𖼳𖽪𖾐 ”), 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏, 𖼈𖽡𖽝𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏. 32 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖼊𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼖𖽨 𖼀𖽡𖽝𖾏 (“𖼖𖽨 𖼀𖽡𖽝𖾏” 𖼟𖽪𖾏 “𖼑𖽔𖾏! 𖼷𖽔 𖼊𖽪𖾐”) 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼈𖽻𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑.” 33 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽤𖾑 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼮𖽑𖽠𖾏, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 (“𖼳𖽡𖾑 𖼄𖽡𖽝𖾏”𖼟𖽪𖾏 “𖼐𖽙𖾑𖽑 𖼊𖽻𖾏”). 34 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽤𖾑 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖼊𖽡𖾏 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 (“𖼖𖽡𖾏 𖽂𖽺𖾏” 𖼟𖽪𖾏 “𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 ” ). 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼑𖽜 𖼈𖽻𖾑 𖼳𖽻𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏.” 35 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽤𖾑 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖼐𖽑𖽡𖾏: “𖽃𖽡𖾐 𖼟𖽻𖾐 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.” 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼊𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼽𖽨 𖼊𖽔𖾏 (“𖼽𖽨 𖼊𖽔𖾏” 𖼟𖽪𖾏 “𖼷𖽻𖾑 𖼖𖽔”), 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼳𖽙𖾏 𖼮𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏.
