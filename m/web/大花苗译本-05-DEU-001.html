16 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽑𖽡𖾏 ： 𖼅𖽡 𖼑𖽻 𖼑𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 , 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 , 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 . 17 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽪𖾏 𖼡𖽻𖾏 𖼑𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽺𖾐 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼢𖽜 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽡𖽝 , 𖼅𖽡 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼊𖽔𖾑 , 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼨𖽑𖽢𖾐 . 18 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼞𖽑𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖼋𖽘𖾏 .
