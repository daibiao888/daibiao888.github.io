𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼳𖽹𖾐 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 27：1-2，11-14；𖼖𖽪𖾏 23：1-5；𖼽𖽙 18：28-38）
15
𖼳𖽠𖾑 𖼊𖽔𖾐 𖽐𖼷𖽙𖾑 ，𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑、 𖼊𖽪𖾐 𖼗𖽻、𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼑𖽪𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖼊𖽳𖾏，𖼡𖽑𖽹𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 ，𖼮𖽤𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑． 2 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “ 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 ？” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽙．” 3 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼛𖽠𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐． 4 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼑𖽔𖾏， 𖼸𖽻𖾏 𖼛𖽠𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼟𖽡 𖼦𖽡𖾐 𖼊𖽡𖾐 𖼋𖽪 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼘𖽪𖾏？” 5 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑．
（𖼊𖽑𖽹𖾏 27：15-26；𖼖𖽪𖾏 23：13-25；𖼽𖽙 18：39-19：16）
6 𖼷𖽑𖽔𖾏 𖼟𖽻𖾏 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼐𖽡𖾑， 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 ，𖼮𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ． 7 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑， 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽜𖾑 𖽐𖼷𖽑𖽙𖾑 𖼑𖽪𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼄𖽔𖾐，𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼞𖽱𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏． 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽜𖾑 𖽐𖼷𖽑𖽙𖾑 𖼑𖽪𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼙𖽔 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． 8 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽐𖼊𖽜， 𖼟𖽙𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼟𖽻𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖽃𖽡𖾑． 9 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼖𖽔𖾏？” 10 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽔𖾑， 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽤𖾐 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏． 11 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼎𖽺𖾏 𖼷𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼽𖽔𖾑 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 12 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖾏， 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏？” 13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼨𖽑𖽪𖾏：“ 𖽐𖼮𖽦𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐！” 14 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼖𖽺𖾑？ 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽔𖾑 𖼖𖽺𖾑？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼨𖽑𖽪𖾏 ：“𖽐𖼮𖽦𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐！” 15 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽔 ， 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼎𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽦𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐．
𖼊𖽪𖾐 𖼏𖽙𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼚𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 27：27-31；𖼽𖽙 19：2-3）
16 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽻 𖽐𖼚𖽙 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏， 𖼨𖽑𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼦𖽡 𖼸𖽱 ． 17 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽘𖾑， 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼳𖽡𖽝𖾐 𖽃𖽡𖾐 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏． 18 𖼎𖽺𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼋𖽪， 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏！” 19 𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏， 𖽐𖼊𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏， 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏． 20 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼚𖽙𖾏 𖼐𖽱𖾏 𖼊𖽘𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼚𖽠𖾏 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑 𖼺𖽘𖾏， 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼮𖽑𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽘𖾑． 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏， 𖼽𖽔𖾑 𖽐𖼮𖽦𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐．
𖽐𖼮𖽦𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐
（𖼊𖽑𖽹𖾏 27：32-44；𖼖𖽪𖾏 23：26-43；𖼽𖽙 19：17-27）

𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽔 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼺𖽘𖾏
27 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼸𖽜 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼮𖽦𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐． 28 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏， 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼖𖽡 𖼇𖽳𖾏． 29 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏， 𖼎𖽺𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽑𖽡𖾏：“𖽃𖽔𖾏 𖼦𖽔𖾏！ 𖼟𖽡 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽔𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏， 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏 𖽃𖽡𖾑， 30 𖼾𖽨𖾏 𖼮𖽨𖾏 𖼾𖽨𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏！” 31 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼖𖽔𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼚𖽙𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏， 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽱𖾏 𖼮𖽨𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼾𖽨𖾏． 32 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽡𖾑 𖼊𖽪， 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏， 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼑𖽔𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖽐𖼮𖽜．” 𖼷𖽷𖾑 𖼸𖽜 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽐𖼮𖽦𖾏 𖼖𖽔𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏．

（𖼊𖽑𖽹𖾏 27：45-56；𖼖𖽪𖾏 23：44-49；𖼽𖽙 19：28-30）
