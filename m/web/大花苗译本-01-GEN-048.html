𖼽𖽔𖾐 𖼞𖽙 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽐𖼎𖽙 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑

“𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏 𖼊𖽪𖾑 𖼾𖽨 𖼞𖽪𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏.
16 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑, 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼐𖽡𖾑.
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
𖼎𖽺𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
17 𖼽𖽙 𖼺𖽳 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽑𖽳𖾑 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼦𖽡𖾐 𖼗𖽱𖾐 𖼚𖽔𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏. 18 𖼽𖽙 𖼺𖽳 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾑 𖼈𖽹𖾐, 𖼦𖽡𖾐 𖼾𖽙 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼮𖽙𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏.” 19 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼦𖽡𖾐 𖼸𖽙𖾏, 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼀𖽻𖾐, 𖼊𖽪𖾐 𖼘𖽪！ 𖼞𖽪𖾑 𖼀𖽻𖾐. 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽡𖽝𖾑 𖽃𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼽𖽔𖾑 𖼘𖽑𖽙𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼽𖽔𖾑 𖼊𖽡𖽝𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.” 20 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼅𖽡 𖽐𖼷𖽷𖾏 𖼏𖽵 𖼅𖽳𖾏, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼺𖽱𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝, 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽘𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼎𖽻𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼨𖽑𖽪𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑. 21 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽔 𖼋𖽘𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐, 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 22 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼎𖽑𖽻𖾏 𖼐𖽑𖽜𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏, 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 , 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽡𖽻𖾐 𖼊𖽻𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠.”
