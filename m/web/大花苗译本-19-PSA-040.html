𖼞𖽪𖾑 𖼙𖽔 𖽐𖼷𖽜𖾏 𖼺𖽡𖽝𖾐 𖼋𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽱𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜．
2 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽻𖾐,
𖼎𖽑𖽻𖾏 𖼁𖽘𖾏 𖽃𖽘𖾑 𖽐𖼞𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾏 𖼮𖽤𖾐 𖼞𖽪𖾑 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼡𖽔𖾑 𖼏𖽔𖾏 𖼞𖽡𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
3 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼮𖽑𖽦𖾐,
𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
 
4 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼦𖽡𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼚𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏！
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏,
𖼊𖽑𖽡𖽝𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽜 𖼀𖽡𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼦𖽡 𖽐𖼮𖽔,
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏, 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼾𖽱 𖼋𖽘 𖼦𖽡𖾐 𖼋𖽻𖾏.
 
6 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼙𖽔,
𖼟𖽡 𖼙𖽔 𖼡𖽑𖽜𖾐 𖼊𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏．
𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽡 𖼽𖽔𖾑.
7 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼋𖽘，
𖼞𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼙𖽔 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏.”
8 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼸𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏,
𖼟𖽡 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐．
 
9 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑．
𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏．
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼯𖽡𖽙.
10 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼈𖽺𖾏 𖼟𖽡 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏,
𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼋𖽘𖾏,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾐 𖼙𖽔 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝.
 
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏；
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑．
（𖽐𖼞𖽻𖾏 70）
12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼾𖽱 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼞𖽪𖾑,
𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼖𖽠𖾑 𖼯𖽢 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼋𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑 𖼚𖽻𖾐 𖼇𖽻𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼡𖽑𖽜𖾐 𖼅𖽳𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑；
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑．
14 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑, 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼞𖽪𖾑 𖼳𖽦𖾑 𖽃𖽡𖾑, 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑, 𖼑𖽡𖾏 𖼡𖽜𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏！
15 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾏 𖼦𖽔𖾏！𖽃𖽔𖾏 𖼦𖽔𖾏！” 𖽃𖽡𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼐𖽔𖾑 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏．
 
16 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡 , 𖼑𖽜 𖼟𖽡 𖽃𖽺𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑；
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏 𖽃𖽡𖾑, 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐．”
17 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏, 𖼮𖽵𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏,
𖼟𖽡 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏．
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽔𖾑 𖼊𖽳𖾑𖽑．
