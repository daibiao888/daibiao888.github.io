8
𖼳𖽪𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾐 𖼖𖽳 𖼊𖽔 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 :
2 “𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼊𖽑𖽻𖾑 𖼋𖽱 ?
𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼘𖽑𖽙𖾐 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑 ?
3 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼺𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 ?
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼺𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ?
4 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 ,
𖼐𖽱𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 .
5 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼘𖽑𖽻𖾐 𖽐𖼡𖽔𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 ,
6 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼟𖽡 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 ,
𖼟𖽙𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 .
7 𖼟𖽡 𖼺𖽪𖾐 𖼁𖽱 𖽃𖽺𖾐 𖼳𖽻𖾐 𖼈𖽡𖽝𖾏 ,
𖼸𖽙 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼘𖽑𖽙𖾐 𖽐𖼷𖽱𖾏 .
 
8 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽪 𖼷𖽑𖽷𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 ,
𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼺𖽘𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 .
9 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼑𖽘 𖼡𖽑𖽔𖾐 𖼑𖽡𖽝 𖼅𖽔𖾏 𖼊𖽔𖾐 , 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼞𖽔𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 ,
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 .
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 , 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 ,
𖼎𖽑𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼗𖽙𖾏 𖼖𖽺𖾑 ?
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏
11 𖼀𖽑𖽪 𖼷𖽑𖽻𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼷𖽑𖽷𖾑 , 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼘𖽑𖽘𖾑 𖼖𖽺𖾑 ?
𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽻𖾐 , 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼘𖽑𖽙𖾐 𖼘𖽪 ?
12 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼊𖽻𖾏 𖼤𖽜𖾏 𖽃𖽡𖾑 ,
𖼐𖽡𖽙𖾐 𖼀𖽔𖾏 𖼮𖽘𖾑 𖽐𖼡𖽜𖾐 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 .
13 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼐𖽱𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 .
14 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼞𖽘𖾐 𖼽𖽘𖾏 𖼳𖽔𖾐 𖼖𖽪𖾑 𖼸𖽙𖾏 .
15 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 ,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽔𖾑 𖼏𖽔𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼈𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 .
16 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼘𖽘𖾑𖽑 𖽐𖼷𖽔𖾐 ,
𖽃𖽔𖾑 𖼯𖽡 𖼘𖽘𖾑𖽑 𖼀𖽪𖾑 𖼖𖽪𖾑 𖼉𖽘𖾏 .
17 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼗𖽔 𖼞𖽑𖽡 𖼙𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼁𖽠𖾏 𖼈𖽳𖾐 ,
𖼗𖽔 𖼁𖽺𖾏 𖼊𖽡𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 .
18 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼚𖽙𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼡𖽑𖽙𖾑 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼐𖽱𖾏 , 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼟𖽡 .
19 𖼑𖽔𖾏 𖼘𖽪 ! 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 ,
𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼘𖽘𖾑𖽑 𖼋𖽠𖾏 .
20 𖼑𖽔𖾏 𖼘𖽪 ！ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .
21 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 ,
𖼎𖽑𖽻𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼟𖽡 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 .
22 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼀𖽔𖾏 𖼐𖽑𖽘𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 ,
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏 .”
