11
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼺𖽷𖾐 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 ,
𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 .
2 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼀𖽜𖾑 𖼙𖽠𖾏, 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽔𖾑 𖼖𖽡,
𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼎𖽻𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏.
3 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼅𖽻𖾏 𖼞𖽡𖾑 ,
𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼁𖽔 𖼮𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
4 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽪𖾐 𖼖𖽑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 (“𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖼊𖽜𖾑 𖼐𖽜𖾐 ”) 𖼮𖽤𖾐 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽙𖾏 𖼺𖽘𖾏 𖼊𖽪𖾑 𖼑𖽨𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼖𖽡𖾐 𖼡𖽔𖾐,
𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑 𖼗𖽙𖾏 𖼊𖽪𖾑 𖼾𖽨 𖼐𖽱𖾏 𖼸𖽻𖾏.
𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐,
𖼽𖽔𖾏 𖼳𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑 .
6 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼽𖽔𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 ,
𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼎𖽪𖾑 𖽃𖽔 𖼛𖽻𖾏 , 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 ,
𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼾𖽨𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑.
7 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼷𖽪𖾑 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼞𖽪𖾑 ,
𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽙𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼯𖽡𖽙 ,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼐𖽱𖾏 .
𖼎𖽺𖾏 𖼺𖽘𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏
8 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼘𖽪 ! 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼟𖽡?
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪! 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼟𖽡?
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼺𖽱𖾏 𖼽𖽔 𖼄𖽔𖾐 ?
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼺𖽱𖾏 𖼳𖽡𖾐 𖼀𖽡𖽝 ?
𖼞𖽪𖾑 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼞𖽪𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
9 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 ,
𖼟𖽪𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼅𖽡 .
 
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 .
𖼐𖽱𖾏 𖽂𖽪𖾏 𖼢𖽔𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽡𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼗𖽙𖾏 .
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼐𖽻 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼗𖽙𖾏 ,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖽐𖼡𖽔𖾐 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏.
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 .
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
12 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼎𖽑𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑 ,
𖼈𖽡𖽝𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐,
𖼟𖽪𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼻𖽡𖽝𖾏 𖼢𖽻 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐.”
