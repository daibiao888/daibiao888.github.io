24
𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼁𖽙𖾏 𖼞𖽡𖾑 ，𖼸𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏，𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏， 2 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼡𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏， 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼑𖽜𖾏． 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽠𖾑𖽑 𖽐𖼊𖽱𖾏 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼷𖽷𖾏 𖼖𖽘𖾑，𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 ，𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼊𖽡𖾏 𖽃𖽻𖾏． 5 𖼸𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼊𖽱𖾐 𖼚𖽑𖽪𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐．𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽑𖽦𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼖𖽺𖾑？ 6 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼋𖽘𖾏．𖼅𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖽙𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 7 𖼐𖽑𖽡𖾏 ：‘𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼋𖽡𖾏，𖽐𖼮𖽦𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐，𖼊𖽻𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏．’ ” 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏，
𖽐𖼊𖽪𖾏𖼊𖽪𖾐𖼷𖽷𖾐𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙𖼡𖽑𖽙𖾑 𖼦𖽡𖾏𖽐𖼷𖽘𖾏
