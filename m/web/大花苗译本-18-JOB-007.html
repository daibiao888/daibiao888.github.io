7
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼘𖽪 ?
𖼐𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼘𖽪 ?
2 𖼺𖽱𖾏 𖼡𖽑𖽜𖾑 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 ,
𖼺𖽱𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼊𖽻𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 .
3 𖼞𖽪𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼟𖽱𖾏 𖼀𖽡 𖼯𖽠 𖼖𖽑𖽡𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 ,
𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽡𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 .
4 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 :
𖼞𖽪𖾑 𖼡𖽔𖾏 𖼊𖽘𖾑 𖼋𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 ? 𖼄𖽑𖽙𖾏 𖼗𖽡𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ?
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼷𖽜𖾑 𖼅𖽻𖾏 𖽐𖼷𖽜𖾑 𖼗𖽙𖾏 , 𖼸𖽙 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 .
5 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽘𖾑 𖼞𖽘𖾏 𖽐𖼎𖽙 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 ,
𖼞𖽪𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼅𖽔 𖼦𖽡𖾏 𖼻𖽡𖽝𖾏 , 𖼎𖽺𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑 𖼟𖽻𖾐 .
6 𖼞𖽪𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼇𖽺𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼖𖽪𖾑 𖽐𖼡𖽙𖾏 ,
𖼞𖽑𖽔𖾏 𖽐𖼞𖽡𖾏 𖼎𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽘𖾏 .
 
7 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 , 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼀𖽘𖾏 𖼊𖽔𖾐 ,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼙𖽔 𖼚𖽙𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 .
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼞𖽪𖾑 , 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 ,
𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽪𖾑 , 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼋𖽘 .
9 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 ,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏 .
10 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 ,
𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽡𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼋𖽘𖾏 .
 
11 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼖𖽡𖽾 𖽐𖼚𖽙 𖼑𖽡𖾏 𖼊𖽳𖾏 ,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 , 𖼽𖽔𖾑 𖽐𖼊𖽙𖾏 𖼋𖽠𖾏 𖼞𖽡𖾑 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 .
12 𖼞𖽪𖾑 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 ?
𖼋𖽡𖽝𖾏 𖼾𖽙 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 ?
𖼟𖽡 𖽃𖽔𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼘𖽪𖾏?
13 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖼸𖽱𖾏 𖼸𖽘𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖼖𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼚𖽠𖾏 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏，
14 𖼟𖽡 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼮𖽡𖽠𖾏 𖼯𖽷𖾏 𖼞𖽪𖾑 ,
𖼯𖽡𖽙𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼞𖽪𖾑 ，
15 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼀𖽘𖾏 𖼋𖽔 𖼺𖽘𖾏 ,
𖽐𖼮𖽑𖽠𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼷𖽷𖾐 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼮𖽡𖽝𖾏 .
16 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 , 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼯𖽡𖽝𖾏 𖼗𖽡𖾏．
𖼟𖽡 𖼯𖽔𖾏 𖼞𖽪𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 .
17 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼟𖽡 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐？
𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐？
18 𖼷𖽑𖽔𖾏 𖼊𖽡𖾏 𖼟𖽡𖾏 , 𖼟𖽡 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼐𖽱𖾏 ,
𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 ？
19 𖼟𖽡 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 ,
𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼡𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖽻𖾐 𖼺𖽘𖾏 𖼖𖽺𖾑 ?
20 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪 ! 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 , 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏 ?
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼟𖽡 𖼊𖽪𖾑 𖼷𖽑𖽷𖾐 ？
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 ?
21 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼯𖽔𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖼺𖽘𖾏 ？
𖼋𖽠𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 ?
𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 ,
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽡 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 , 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼋𖽘 .”
