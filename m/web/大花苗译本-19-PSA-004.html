𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏, 𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽷𖾑 𖼺𖽙𖾑.
4
𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼘𖽪!
𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑！
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑, 𖼑𖽻 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏！
 
2 𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑, 𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼖𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑？
𖼅𖽡 𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑, 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼖𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑？（𖼳𖽡𖾏 𖼖𖽔𖾑）
3 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼇𖽺𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼁𖽡𖽝 𖼋𖽘𖾏,
𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽻 𖼞𖽪𖾑.
 
4 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼐𖽡𖽙𖾐 𖼸𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖽐𖼊𖽱𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
5 𖼟𖽡 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑,
𖼎𖽺𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
6 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏: “𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏？
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖾏, 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐.”
7 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼇𖽔𖾑 𖼊𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖼑𖽡𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
 
8 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼀𖽱𖾏 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐.
