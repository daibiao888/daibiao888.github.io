𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 1：34 - 37）
36
𖼽𖽡𖾑 𖼺𖽻𖾐 𖼟𖽪𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐. 2 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼷𖽙𖾏 𖼮𖽢𖾑 𖼐𖽔 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐, 𖼟𖽪𖾏 𖼦𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽪𖽝 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼽𖽔𖾏 𖼊𖽔𖾏, 𖼳𖽡𖾑 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖾏 𖼀𖽡𖽝𖾏 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖽐𖼷𖽑𖽹𖾏, 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼐𖽔 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾑 𖼦𖽳 𖼖𖽡𖾏 𖼀𖽔𖾑 𖼄𖽔𖾐， 3 𖼎𖽺𖾏 𖼷𖽙𖾏 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽡 𖼀𖽺𖾏 𖼽𖽙 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼀𖽔𖾑 𖼳𖽷𖾏 𖼄𖽙. 4 𖼽𖽔𖾏 𖼊𖽔𖾏 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔, 𖼀𖽔𖾑 𖼳𖽷𖾏 𖼄𖽙 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖼖𖽨 𖽃𖽳𖾐； 5 𖽃𖽔𖾑 𖼦𖽳 𖼖𖽡𖾏 𖼀𖽔𖾑 𖼄𖽔𖾐 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼽𖽦𖾑 𖽂𖽪𖾑 𖼳𖽷𖾑、𖼽𖽔𖾐 𖼖𖽔、𖼞𖽑𖽙𖾐 𖼖𖽔𖾑. 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼳𖽻𖾏.
6 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐, 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐, 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼐𖽔 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑, 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽠𖾏, 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼺𖽘𖾏. 7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼀𖽙𖾏 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼷𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐. 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼟𖽪𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏.
9 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼟𖽪𖾏 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐, 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐. 10 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼷𖽷𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐: 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼽𖽔𖾏 𖼊𖽔𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔, 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼀𖽔𖾑 𖼳𖽷𖾏 𖼄𖽙 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼖𖽨 𖽃𖽳𖾐. 11 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼊𖽑𖽡 𖼄𖽔𖾏、𖽃𖽔𖾑 𖼄𖽙、𖼳𖽡𖾐 𖼀𖽙𖾑、𖼮𖽢𖾑 𖼊𖽑𖽗𖾐、𖼮𖽡𖾑 𖼐𖽔 𖼺𖽷𖾑. 12 𖼊𖽑𖽡 𖼐𖽔 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏, 𖼐𖽱𖾏 𖼑𖽜 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡. 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽤𖾐 𖼽𖽔𖾏 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏. 13 𖼖𖽨 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽔 𖼦𖽔𖾐、𖼳𖽡𖽝𖾏 𖼖𖽔𖾑、𖼳𖽔𖾑 𖼄𖽔𖾐、𖼄𖽡𖾐 𖼺𖽔𖾐. 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽤𖾐 𖼀𖽔𖾑 𖼳𖽷𖾏 𖼄𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏. 14 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐 𖽃𖽔𖾑 𖼦𖽳 𖼖𖽡𖾏 𖼀𖽔𖾑 𖼄𖽔𖾐, 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼀𖽡𖽝𖾏 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼐𖽔 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏, 𖼐𖽱𖾏 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼽𖽦𖾑 𖽂𖽪𖾑 𖼳𖽷𖾑、𖼽𖽔𖾐 𖼖𖽔、𖼞𖽑𖽙𖾐 𖼖𖽔𖾑.
15 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑, 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐： 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐, 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼅𖽔𖾏 𖼊𖽑𖽡 𖼄𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖽃𖽔𖾑 𖼄𖽙 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼳𖽡𖾐 𖼀𖽙𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼮𖽡𖾑 𖼐𖽔 𖼺𖽷𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、 16 𖼞𖽑𖽙𖾐 𖼖𖽔𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼮𖽢𖾑 𖼊𖽑𖽗𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑. 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽡𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑, 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏. 17 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽨 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼅𖽔𖾏 𖼐𖽔 𖼦𖽔𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼳𖽡𖽝𖾏 𖼖𖽔𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼳𖽔𖾑 𖼄𖽔𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼄𖽡𖾐 𖼺𖽔𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑. 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼎𖽑𖽻𖾏 𖼖𖽨 𖽃𖽳𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑, 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼀𖽔𖾑 𖼳𖽷𖾏 𖼄𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏. 18 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐 𖽃𖽔𖾑 𖼦𖽳 𖼖𖽡𖾏 𖼀𖽔𖾑 𖼄𖽔𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼅𖽔𖾏 𖼽𖽦𖾑 𖽂𖽪𖾑 𖼳𖽷𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼽𖽔𖾐 𖼖𖽔 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼞𖽑𖽙𖾐 𖼖𖽔𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑. 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐, 𖼽𖽔𖾏 𖼐𖽔 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏, 𖽃𖽔𖾑 𖼦𖽳 𖼖𖽡𖾏 𖼀𖽔𖾑 𖼄𖽔𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑. 19 𖼋𖽡𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼟𖽪𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑.

20 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽙 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐： 𖼟𖽪𖾏 𖼘𖽙 𖼊𖽑𖽗𖾑、𖼳𖽙 𖼀𖽔𖾑、𖼮𖽡𖾏 𖼀𖽡𖽝𖾏、𖼽𖽔𖾏 𖼐𖽔、 21 𖼊𖽡𖾐 𖼳𖽪𖽝𖾏、𖼽𖽡𖾐 𖼮𖽑𖽔、𖼊𖽡𖾐 𖼳𖽗𖾑． 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽡𖾐 𖼦𖽙 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑. 22 𖼘𖽙 𖼊𖽑𖽗𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼦𖽙 𖼖𖽡𖾏、𖼳𖽡𖾑 𖼄𖽔𖾏； 𖼘𖽙 𖼊𖽑𖽗𖾑 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼊𖽑𖽡 𖼐𖽔. 23 𖼳𖽙 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼖𖽳 𖽂𖽺、𖼄𖽔𖾐 𖼐𖽔 𖼳𖽢、𖼽𖽡𖾑 𖼀𖽔𖾑 𖼖𖽪、𖼳𖽷𖾏 𖼀𖽙𖾑、𖽃𖽔𖾑 𖼐𖽔. 24 𖼮𖽡𖾏 𖼀𖽡𖽝𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼽𖽔𖾐、𖼽𖽔𖾏 𖼐𖽔, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼾𖽨 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐𖼮𖽡𖾏 𖼀𖽡𖽝𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑, 𖽐𖼮𖽷𖾐 𖼏𖽻 𖽃𖽻𖾐 𖼷𖽠𖾑𖽑 𖼟𖽪𖾏 𖼳𖽙𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼽𖽔𖾏 𖼐𖽔 𖼐𖽡𖾑. 25 𖼽𖽔𖾏 𖼐𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼳𖽪𖽝𖾏； 𖼽𖽔𖾏 𖼐𖽔 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼦𖽳 𖼖𖽡𖾏 𖼀𖽔𖾑 𖼄𖽔𖾐. 26 𖼊𖽡𖾐 𖼳𖽪𖽝𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽡𖽾𖾑 𖼊𖽗𖾏、𖼽𖽡𖾑 𖼳𖽷𖾏 𖼀𖽔𖾑、𖼽𖽡 𖼖𖽔、𖼮𖽡𖾑 𖼖𖽔. 27 𖼽𖽡𖾐 𖼮𖽑𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼀𖽑𖽡 𖼦𖽗𖾐、𖼺𖽔𖾐 𖼇𖽔𖾑、𖼽𖽔𖾏 𖼞𖽔𖾑. 28 𖼊𖽡𖾐 𖼳𖽗𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽂𖽪𖾑 𖼺𖽷𖾑、𖼽𖽔𖾏 𖼖𖽔. 29 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼦𖽙 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐： 𖼟𖽪𖾏 𖼘𖽙 𖼊𖽑𖽗𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼳𖽙 𖼀𖽔𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼮𖽡𖾏 𖼀𖽡𖽝𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼽𖽔𖾏 𖼐𖽔 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑, 30 𖼊𖽡𖾐 𖼳𖽪𖽝𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼽𖽡𖾐 𖼮𖽑𖽔 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼊𖽡𖾐 𖼳𖽗𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑.𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼦𖽙 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑. 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑.
𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 1：43 - 54）
31 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐： 32 𖼀𖽡𖾐 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽻 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼊𖽑𖽡 𖼦𖽔𖾐 𖼀𖽔𖾑. 33 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼋𖽔 𖼙𖽠𖾏, 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽡𖽝𖾏 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼀𖽔𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 34 𖼽𖽙 𖼀𖽔𖾑 𖼋𖽔 𖼙𖽠𖾏, 𖼊𖽑𖽡 𖼄𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽪𖾏 𖼳𖽗𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 35 𖼦𖽪𖾏 𖼳𖽗𖾑 𖼋𖽔 𖼙𖽠𖾏, 𖼀𖽡𖾐 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼦𖽔𖾐 𖼊𖽔 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼷𖽷𖾑 𖼦𖽔𖾐 𖼊𖽔 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽻 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖽂𖽺𖾏 𖼊𖽳. 36 𖼦𖽔𖾐 𖼊𖽔 𖼋𖽔 𖼙𖽠𖾏, 𖼄𖽔𖾐 𖼺𖽷𖾏 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾑 𖼖𖽔𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 37 𖼺𖽘𖾑 𖼖𖽔𖾑 𖼋𖽔 𖼙𖽠𖾏, 𖼖𖽪𖾑 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼖𖽡𖾏 𖼦𖽙 𖼀𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽻𖾐 𖼘𖽙 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 38 𖼺𖽻𖾐 𖼘𖽙 𖼋𖽔 𖼙𖽠𖾏, 𖼽𖽔𖾏 𖼞𖽳 𖼀𖽙𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼖𖽳 · 𖼦𖽔𖾐 𖼐𖽔 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 39 𖼽𖽔𖾏 𖼞𖽳 𖼀𖽙𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼖𖽳 · 𖼦𖽔𖾐 𖼐𖽔 𖼋𖽔 𖼺𖽘𖾏, 𖼦𖽔𖾐 𖼊𖽔 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽻 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽔𖾑 𖽂𖽪𖾑．𖼐𖽱𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼄𖽡𖾐 𖼳𖽡𖾑 𖼊𖽑𖽔𖾑 𖼀𖽡𖽝, 𖼟𖽪𖾏 𖼄𖽡𖾐 · 𖼺𖽔𖾐 𖼦𖽳 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼞𖽱𖾏 𖼞𖽱𖾑 𖽐𖼷𖽑𖽹𖾏, 𖼄𖽔𖾐 𖼊𖽑𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏.
40 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐、𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐、𖽐𖼷𖽷𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐: 𖼟𖽪𖾏 𖼊𖽑𖽡 𖼐𖽔 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼽𖽔𖾏 𖼖𖽳 𖽂𖽔𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼽𖽦𖾑 𖼊𖽑𖽡𖽝 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、 41 𖽃𖽔𖾑 𖼦𖽳 𖼖𖽡𖾏 𖼀𖽔𖾑 𖼄𖽔𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼽𖽡𖾑 𖼖𖽔𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼀𖽡𖾐 𖼐𖽹𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、 42 𖼮𖽡𖾑 𖼐𖽔 𖼺𖽷𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼊𖽑𖽡 𖼄𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼄𖽡𖾐 𖼀𖽡𖾐 𖼺𖽔𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、 43 𖼄𖽔𖾐 𖼮𖽡𖾑 𖼊𖽡𖽝 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑、𖼽𖽡𖾑 𖼖𖽔 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑. 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑, 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏, 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑. 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐.
