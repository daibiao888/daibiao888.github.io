𖼦𖽔𖾑 𖼗𖽔𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼑𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐:
 
𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 𖼑𖽡𖾏 𖽐𖼞𖽙𖾏 𖼘𖽪! 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏,
𖼋𖽡𖾏 𖼡𖽘𖾐 𖼄𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽹𖾏 𖼀𖽑𖽪𖾐 𖼖𖽪𖾏 𖼺𖽷𖾑 （ 𖼺𖽹𖾏 𖼀𖽑𖽪𖾐 𖼖𖽪𖾏 𖼺𖽷𖾑 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼊𖽑𖽡 ）𖼊𖽡𖾐 𖼊𖽻𖾏 𖼗𖽙𖾏 .
2 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪𖾏 𖼳𖽡𖽠𖾑,
𖼟𖽪𖾏 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽱𖾏 𖽐𖼞𖽙𖾏 𖼊𖽻𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 𖽃𖽡𖾑 𖼘𖽪 !
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏.
3 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼳𖽡𖾑 𖼦𖽳 𖼑𖽡𖾏 𖼡𖽻𖾐 𖼡𖽜𖾏,
𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼑𖽡𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼟𖽪𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏.
4 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼘𖽪！ 𖼟𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼐𖽑𖽡𖾏, 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽑𖽡𖽝𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼑𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼖𖽑𖽘𖾑 𖼾𖽨 ,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼊𖽪𖾑 𖼾𖽨 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼊𖽪𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼊𖽪𖾑 𖼾𖽨 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼷𖽑𖽹𖾏.”
5 𖼋𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 ,
𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝.
6 𖼅𖽡 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 ,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏.
7 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼅𖽡 𖼙𖽔 𖼚𖽙𖾏 𖼖𖽔𖾏?
𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼷𖽹𖾏 𖼅𖽔𖾏 𖼋𖽘 , 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼨𖽑𖽳𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐.
8 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼯𖽡𖽙𖾏,
𖼐𖽱𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐,
𖼐𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽺 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏,
𖽐𖼮𖽷𖾐 𖼏𖽻 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼏𖽵 𖼐𖽔𖾑 𖼖𖽺𖾑?
9 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼏𖽵 𖼯𖽡𖽙𖾏,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼊𖽻𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.
10 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪(“𖼖𖽡𖾑 𖼉𖽪𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏”),
𖼟𖽡 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖽐𖼚𖽱𖾏 𖼀𖽪𖾑 𖼟𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼋𖽘𖾏.
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼋𖽘𖾏,
𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼐𖽔 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼋𖽘𖾏,
𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽃𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑.
12 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 (“𖼖𖽡𖾑 𖼉𖽪𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 ”)𖼟𖽪𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼘𖽪！
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼋𖽡𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼺𖽹𖾏 𖼀𖽑𖽪𖾐 𖼖𖽪𖾏 𖼺𖽷𖾑 ,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼳𖽙𖾏.”
𖼮𖽑𖽡𖽻𖾏 𖼳𖽤𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏
13 𖼑𖽔𖾏 𖼘𖽪！ 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼽𖽔𖾏 𖼳𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼙𖽻 , 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏.
14 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 𖼑𖽡𖾏 𖽐𖼞𖽙𖾏 𖼘𖽪! 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼋𖽘 .
15 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼳𖽤𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏.𖼳𖽤𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼽𖽔𖾑 𖼺𖽱𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼨𖽑𖽪𖾏 :
16 “𖼟𖽡 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼐𖽡𖾑 𖼘𖽪 !
𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼢𖽜𖾏 𖽐𖼎𖽔 𖼗𖽙𖾏 𖽐𖼮𖽡 𖽐𖼷𖽻𖾏 𖽐𖼚𖽙 𖼙𖽻 ,
𖽐𖼎𖽷𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼑𖽻 , 𖼮𖽡𖽻𖾐 𖼨𖽑𖽪𖾏 𖼸𖽙𖾐 𖽐𖼞𖽻𖾏, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽐𖼊𖽱𖾏 𖼮𖽳𖾐 𖼟𖽡.”
17 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 , 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼊𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 (“𖼊𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏”)，𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼞𖽪𖽙 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 (“𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏”), 18 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼮𖽷𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼮𖽻𖾏, 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼏𖽔𖾏 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 .
