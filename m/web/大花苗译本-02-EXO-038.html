38
𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼞𖽪𖾐 , 𖽐𖼊𖽡𖾑 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 ,𖼇𖽘𖾑 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼺𖽡𖽝𖾐 𖼷𖽷𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 2 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽡𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽪𖾐 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼋𖽪 𖼞𖽪𖾑 𖼞𖽪𖾐 , 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼊𖽡𖾏 , 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖼡𖽑𖽠𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼮𖽡𖽝𖾏 . 3 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼖𖽪𖾑 𖼋𖽺𖾏、𖼚𖽡𖽝𖾑 𖼞𖽻𖾏、𖼖𖽪𖾑 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑、𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖽐𖼡𖽹𖾏、𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 , 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 . 4 𖼎𖽺𖾏 𖼑𖽜 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼋𖽻𖾏 𖼯𖽡𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼸𖽙𖾏 , 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽻𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼀𖽡𖾑 𖼊𖽡𖾐 , 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼸𖽙 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼚𖽔𖾏 . 5 𖼑𖽜 𖼋𖽻𖾏 𖼮𖽡𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼸𖽙𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽪𖾐 𖼘𖽑𖽡𖾑 𖼚𖽻𖾐 𖼖𖽪𖾐 𖽃𖽔 𖼉𖽙𖾏 , 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼙𖽡 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 . 6 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 , 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏， 7 𖼞𖽑𖽡𖽠𖾏 𖼡𖽠𖾏 𖼙𖽡 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾐 𖼺𖽘𖾐 𖽐𖼚𖽙 𖽃𖽔 𖼉𖽙𖾏 𖼞𖽱𖾑 𖽃𖽔𖾑 𖼏𖽘 . 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼊𖽙𖾏 𖽐𖼚𖽙 .
（𖼋𖽠𖾏 30：18）
8 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼋𖽺𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾑 𖼊𖽠𖾏 𖼋𖽡𖾏 , 𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽜𖾐 𖼖𖽪𖾑 𖼋𖽺𖾏 , 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽔 𖼛𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼑𖽔𖾏 𖼚𖽑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 .
𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏
（𖼋𖽠𖾏 27：9 - 19）
9 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 . 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 , 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 𖽃𖽔𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏, 𖼇𖽘𖾑 𖽃𖽡𖾐 𖼀𖽔𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 10 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼋𖽪𖾏 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼗𖽪𖾏 , 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 . 11 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏, 𖼇𖽘𖾑 𖽃𖽡𖾐 𖼀𖽔𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼋𖽪𖾏 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼗𖽪𖾏, 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 .
12 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼅𖽔𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏, 𖼇𖽘𖾑 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼋𖽪𖾏 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽻𖾏 𖼗𖽪𖾏 , 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 . 13 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 , 𖼇𖽘𖾑 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 14-15 𖽃𖽔 𖼛𖽻𖾏 𖼺𖽘𖾑 𖼐𖽡𖾑 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖼟𖽻𖾏 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼺𖽘𖾑 𖽃𖽡𖾑 𖼖𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 . 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼷𖽷𖾐 𖼋𖽪 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾐 𖼖𖽪𖾐 . 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼅𖽔𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖼟𖽻𖾏 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼷𖽷𖾐 𖼋𖽪 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾐 𖼖𖽪𖾐 . 16 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏, 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 𖽃𖽔𖾏 . 17 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 , 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 , 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼡𖽑𖽠𖾑 . 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 . 18 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼁𖽘𖾏 , 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖽐𖼎𖽙 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 𖽐𖼊𖽙𖾏 , 𖼇𖽘𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼺𖽡𖽝𖾐 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼺𖽱𖾏 . 19 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼚𖽻𖾐 𖼋𖽪 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼖𖽪𖾐 , 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼊𖽪𖾑 𖽐𖼡𖽜𖾐 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼊𖽔𖾐 , 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼡𖽑𖽠𖾑 . 20 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼘𖽑𖽻𖾐 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼘𖽑𖽻𖾐 , 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 .
