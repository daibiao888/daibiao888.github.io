𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐
6
𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐, 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼚𖽻𖾑 𖼖𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 2 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼅𖽔𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼇𖽪 𖼉𖽔𖾏 𖼾𖽨𖾏 , 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼊𖽡𖾏 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼊𖽡𖾏 𖽐𖼀𖽙 𖼊𖽠𖾏, 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼽𖽘𖾏. 3 𖼦𖽡𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼨𖽑𖽪𖾑 𖼯𖽢𖾏! 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏! 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏! 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑!”
4 𖼑𖽜 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏, 𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼑𖽡𖾏 𖼊𖽠𖾏 𖼖𖽔𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼋𖽘𖾏, 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼀𖽘𖾏 𖽐𖼮𖽡𖽙𖾏.
5 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏! 𖼞𖽪𖾑 𖼛𖽙𖾏 𖼋𖽔 𖼋𖽘𖾏! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.”
6 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼇𖽪 𖼽𖽘𖾏 𖼸𖽙 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡 𖼊𖽑𖽜𖾏 𖼟𖽪𖾏 𖼾𖽱𖾏 𖼟𖽡𖾏 𖼞𖽡𖾏 𖼯𖽔𖾏, 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼞𖽡𖾑 𖼐𖽡 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽹𖾏 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏. 7 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡 𖼊𖽑𖽜𖾏 𖼊𖽑𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼊𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡 𖼊𖽑𖽜𖾏 𖼐𖽡𖾑 𖼊𖽑𖽻𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏．”
8 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼮𖽵𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑? 𖼡𖽔 𖼋𖽱𖾏 𖼯𖽪𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼖𖽺𖾑?” 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑! 𖼷𖽑𖽜𖾑 𖼷𖽷𖾐 𖼞𖽪𖾑!” 9 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏:
𖼅𖽡 𖼐𖽙𖾑𖽑 𖼯𖽡𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏, 𖼈𖽡𖽝𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏,
𖼑𖽔𖾏 𖼯𖽡𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼯𖽡𖽙𖾏,𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐.
10 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼡𖽔𖾐 𖼗𖽘𖾐,
𖽃𖽔𖾑 𖼅𖽔 𖼇𖽔 𖼙𖽻,
𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼁𖽙𖾏,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏,
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏,
𖼎𖽙𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼗𖽙, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏.”
11 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼘𖽪! 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏:
“𖼸𖽙 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐,
𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼟𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.”
12 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼙𖽠𖾏,
𖼐𖽡𖽙𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖽐𖼚𖽙 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .
13 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖽐𖼚𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼊𖽔𖾐 𖼈𖽡𖽝𖾏,
𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼺𖽱𖾏 𖼷𖽷𖾑 𖼄𖽡𖾏 𖽐𖼷𖽳𖾑、 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏, 𖼊𖽻𖾏 𖽐𖼊𖽙𖾑 𖼺𖽘𖾏 𖼈𖽡𖽝𖾏,
𖽃𖽔𖾑 𖼮𖽑𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼯𖽡𖽙𖾏 𖼺𖽷𖾐,
𖼊𖽪𖾑 𖼟𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖼮𖽑𖽱𖾏.
