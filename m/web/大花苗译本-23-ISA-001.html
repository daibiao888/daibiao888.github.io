𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖽐𖼊𖽠𖾑
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼦𖽡𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻
1
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 、 𖼽𖽙 𖼊𖽑𖽗𖾐 、 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 、 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏, 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 .
2 𖽐𖼊𖽪𖾏 𖼘𖽪 ！ 𖼽𖽔𖾑 𖼑𖽻．𖼊𖽡𖾐 𖼘𖽪！ 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻．
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼊𖽪𖾑 𖼾𖽨 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏, 𖼾𖽨 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏.
3 𖼊𖽪𖾑 𖼑𖽨𖾏 𖼯𖽜𖾏 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑,
𖼑𖽜𖾏 𖼖𖽔𖾑 𖼯𖽜𖾏 𖽐𖼎𖽑𖽹𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼚𖽘𖾐,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼯𖽜𖾏 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑,
𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼞𖽪𖾑 .”
 
4 𖽃𖽺𖾐! 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼾𖽔𖾏 𖽐𖼎𖽘𖾏 𖽃𖽡𖾑,
𖼮𖽘𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼗𖽱𖾏 𖽐𖼚𖽔𖾏 !
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏, 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑,
𖼊𖽱𖾐 𖼊𖽡𖾑 𖼢𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 ,𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼑𖽡𖽻𖾏.
 
5 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼯𖽪 𖼯𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼯𖽪 𖼯𖽪𖾏 𖼏𖽻 𖽐𖼊𖽻𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑 ?
𖼅𖽡 𖽃𖽡𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼙𖽱 𖼊𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐,
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏.
6 𖼎𖽑𖽻𖾏 𖼡𖽘𖾐 𖼋𖽠𖾏 𖼸𖽙 𖼋𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼄𖽙𖾐,
𖼮𖽑𖽡𖽠𖾏 𖽃𖽙𖾏 𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖼀𖽡𖾏 𖼡𖽔𖾏 𖼮𖽑𖽦𖾐 𖼞𖽡𖾑𖽑,
𖼮𖽑𖽡𖽠𖾏 𖼄𖽙𖾐 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽹𖾏 𖼨𖽑𖽪𖾑 𖼺𖽷𖾐, 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼚𖽜𖾑 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼅𖽔𖾏.
 
7 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼖𖽔𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼺𖽘𖾏,
𖼅𖽡 𖼖𖽪𖾑 𖼙𖽻 𖼏𖽻 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏．
𖼅𖽡 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼏𖽻 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏,
𖼎𖽺𖾏 𖼏𖽻 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽠𖾐 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏, 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼙𖽠𖾏.
8 𖼮𖽔 𖼷𖽷𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼙𖽻 𖼮𖽡𖽝𖾏 𖼊𖽔𖾐 (“𖼙𖽻”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏”),
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼖𖽔𖾑 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 ,
𖼊𖽡𖾐 𖼞𖽡𖾐 𖽐𖼀𖽔𖾐 𖼑𖽡𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖽐𖼡𖽜𖾐,
𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼮𖽡𖽝𖾏.
9 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼀𖽡𖾐 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏,
𖼀𖽡𖾐 𖽐𖼷𖽙𖾑 𖽐𖼷𖽙𖾑 𖼺𖽱𖾏 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 、 𖽃𖽳 𖼄𖽙 𖼖𖽔𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏.
𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼈𖽡𖽝𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏
10 𖼅𖽡 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽡𖾑 𖼘𖽪 !
𖼽𖽔𖾑 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 !
𖼅𖽡 𖽃𖽳 𖼄𖽙 𖼖𖽔𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼘𖽪 !
𖼽𖽔𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏！
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼖𖽑𖽘𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?
𖼊𖽪𖾑 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼛𖽙 𖼑𖽡𖾏 𖼛𖽙𖾏,
𖼞𖽪𖾑 𖼇𖽔𖾑 𖼷𖽻𖾏 𖼋𖽘𖾏,
𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖽐𖼮𖽑𖽘𖾑、𖼾𖽘𖾏 𖼁𖽜𖾏 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑、 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏 𖽐𖼮𖽑𖽘𖾑,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐.
 
12 𖼅𖽡 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑,
𖼡𖽔 𖼋𖽱𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼐𖽡𖾑,
𖼟𖽙𖾏 𖼅𖽡 𖼯𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼖𖽺𖾑?
13 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖼋𖽡𖾏 𖼀𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼐𖽡𖾑,
𖽐𖼷𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .
𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖽐𖼎𖽙 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼖𖽔𖾏 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝.
𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼺𖽘𖾏,𖼎𖽺𖾏 𖼻𖽙𖾑 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼯𖽔𖾏 .
14 𖼅𖽡 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖽐𖼎𖽙 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼋𖽡𖾏 𖼐𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖽐𖼎𖽘𖾏 𖼐𖽑𖽤𖾑,
𖼞𖽪𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼟𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼷𖽑𖽔𖾏.
15 𖼅𖽡 𖼮𖽘𖾏 𖼋𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼀𖽙 𖽃𖽔𖾑 𖼅𖽔 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏,
𖼟𖽪𖾏 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽻 .
𖼅𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑.
𖼡𖽑𖽔𖾐 𖼟𖽙𖾏 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
16 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼷𖽔𖾑 𖽐𖼷𖽑𖽔𖾏 𖼾𖽨𖾏, 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼾𖽨𖾏,
𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼅𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳,
𖼽𖽔𖾑 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏，
17 𖼾𖽻 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖼚𖽠𖾏 𖼮𖽨𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏,
𖽐𖼀𖽘𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖼑𖽜 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼚𖽠𖾏 𖼑𖽪𖾏 .”
 
18 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼋𖽔𖾏, 𖼀𖽡𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏 𖼞𖽡𖾑．
𖼅𖽡 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼞𖽡𖾏 𖼯𖽔𖾏 𖼈𖽡𖽝𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖼚𖽠𖾐 𖼺𖽱𖾏 𖽐𖼀𖽪𖾏,
𖼺𖽱𖾏 𖼞𖽔𖾐 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼞𖽡𖾏 𖼯𖽔𖾏 𖼈𖽡𖽝𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖼚𖽠𖾐 𖼺𖽱𖾏 𖼚𖽻𖾐 𖼾𖽘𖾐.
19 𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼑𖽻 𖼸𖽙𖾏,
𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽃𖽡𖾑 𖼑𖽻𖾏 .
20 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏, 𖼎𖽺𖾏 𖼎𖽙𖾑 𖽐𖼷𖽜𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
21 𖽃𖽺𖾐 ! 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏！
𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼮𖽔 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼊𖽔𖾐.
22 𖼟𖽡 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼟𖽡 𖼋𖽡𖾏 𖼮𖽡𖽠𖾑 𖼳𖽡𖽠𖾏 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏.
23 𖼟𖽡 𖼸𖽻𖾏 𖼉𖽻 𖼷𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 ,
𖽐𖼎𖽙 𖼸𖽜 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔 ,
𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 ,
𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼾𖽨𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏．
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽜 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑𖼷𖽡𖽝𖾏,
𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖽐𖼊𖽠𖾑 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼸𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
𖼳𖽡 𖼣𖽔𖾑 𖼮𖽔𖾑 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏
24 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼮𖽵𖾑 — 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫，
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:
“𖽃𖽺𖾐 ! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾏 𖽃𖽡𖾑 𖽃𖽔𖾐 𖼮𖽡𖾑 𖽐𖼷𖽔𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼼𖽷𖾏 𖼁𖽻𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 .
25 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼟𖽡 ,
𖼷𖽑𖽷𖾑 𖼮𖽡𖽺𖾏 𖼟𖽡 𖼋𖽡𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼺𖽘𖾏,
𖼺𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼮𖽑𖽻𖾑 𖼗𖽙𖾏 𖼮𖽡𖽺𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼋𖽡𖾏 𖼡𖽔𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑.
26 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼉𖽻 𖼷𖽻𖾏 𖼎𖽙𖾑 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏, 𖼺𖽱𖾏 𖼺𖽪𖾐 𖼁𖽱 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼎𖽙𖾑 𖽐𖼷𖽱𖾏 𖼟𖽡 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼸𖽠𖾐, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼺𖽪𖾐 𖼁𖽱 𖽃𖽡𖾐 𖼼𖽘𖾏.
𖽃𖽔𖾑 𖼙𖽡𖽝𖾏, 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐.”
27 𖼳𖽡 𖼣𖽔𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼙𖽻 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾐 𖼙𖽻 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏.
28 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑, 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼛𖽙𖾏 𖼋𖽔 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.
29 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼇𖽘𖾑 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽺𖾐 𖼙𖽔 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼅𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼺𖽹𖾑 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
30 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼚𖽻𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡,
