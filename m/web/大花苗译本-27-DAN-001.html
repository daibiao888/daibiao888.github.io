𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼙𖽻 𖼮𖽡𖽝𖾏. 2 𖼮𖽵𖾑 𖼄𖽔𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖼊𖽑𖽡𖽝𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼅𖽻 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼐𖽔 𖼊𖽡𖾐, 𖼳𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼞𖽔𖾏 𖼮𖽡𖽝𖾏, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼞𖽔𖾏 𖼼𖽪 𖼮𖽡𖽝𖾏.
𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔 𖼊𖽻𖾏 𖼺𖽹𖾑
