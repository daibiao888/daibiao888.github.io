𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾏
（𖼀𖽳 4：1 -14，22）
4
𖼊𖽑𖽡 𖼄𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 :
2 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼟𖽡 𖼎𖽺𖾏 𖽐𖼞𖽜𖾏 𖼑𖽻 𖼘𖽪𖾏 ?
𖼈𖽡𖽝𖾏 𖼡𖽔 𖼊𖽱𖾏 𖼸𖽠𖾏 𖽐𖼷𖽜𖾏 𖼋𖽻𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼖𖽺𖾑 ?
3 𖼟𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 ,
𖼎𖽺𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .
4 𖼟𖽡 𖼗𖽪𖾏 𖼙𖽔 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽻𖾏 𖼢𖽻 ,
𖼟𖽡 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼇𖽻𖾑 𖼯𖽡𖽻 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 .
5 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼗𖽙𖾏 𖼷𖽠𖾏 𖼟𖽡 , 𖼟𖽡 𖼎𖽺𖾏 𖼀𖽑𖽪𖾑 𖼺𖽡𖽝𖾐 ,
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼟𖽡 , 𖼟𖽡 𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 .
𖼐𖽑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑
6 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 , 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼘𖽪 ?
𖼟𖽡 𖼋𖽙𖾏 𖼑𖽔𖾏 , 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼘𖽪 ?
7 𖼮𖽑𖽙 𖼟𖽡 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖽃𖽡𖾐 𖽐𖼊𖽱𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 ?
8 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽙𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 ,
𖼯𖽡𖽙 𖼋𖽙𖾏 𖼋𖽻𖾏 𖼚𖽠𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 .
9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽂𖽪𖾏 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼀𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏 ,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽂𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘 .
10 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖽐𖼮𖽦 𖼺𖽘𖾏 ,
𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼐𖽑𖽡𖽝𖾑 𖼞𖽑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 .
11 𖼡𖽔𖾐 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 ,
𖼡𖽔𖾑 𖼑𖽡𖽝 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼑𖽡𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼖𖽔𖾏 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏 .
 
12 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼊𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖽃𖽡𖾑 .
13 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽔𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 ,
𖼐𖽡𖽙𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼑𖽜 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼮𖽑𖽙 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏，
14 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼎𖽙 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼽𖽘𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 .
15 𖼅𖽔𖾏 𖼖𖽡𖽾 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏，
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼚𖽻𖾐 𖽐𖼷𖽑𖽙𖾑 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 .
16 𖼊𖽪𖾑 𖼖𖽡𖽾 𖽃𖽡𖾑 𖼳𖽠𖾑 𖽐𖼎𖽙𖾏 ,
𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼇𖽺𖾐 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 ,
𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏．
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽑𖽡𖾏 :
17 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼋𖽔 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽔𖾏 𖼘𖽪 ?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼋𖽔 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼐𖽱𖾏 𖼖𖽺𖾑 ?
18 𖼑𖽔𖾏 𖼘𖽪 ！𖼮𖽵𖾑 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼻𖽡𖽝𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 ,
𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 .
19 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼐𖽑𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽃𖽔 𖼗𖽔𖾏 ,
𖼊𖽠𖾏 𖽐𖼞𖽔𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 ,
𖼟𖽪𖾏 𖼖𖽡𖾏 𖼟𖽻𖾏 𖼐𖽹𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼐𖽹𖾏 𖼀𖽡𖾑 𖼞𖽘𖾐 𖽃𖽡𖾑 𖼖𖽺𖾑 ?
20 𖼐𖽱𖾏 𖼷𖽻𖾏 𖼐𖽡𖽙𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽳𖾐 𖽐𖼊𖽱𖾏 .
21 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼘𖽪𖾏 ?
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 , 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽔𖾑 𖼋𖽔 ．”
