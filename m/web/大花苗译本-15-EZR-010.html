10
𖼽𖽡𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼯𖽜𖾏 𖼾𖽔𖾏 𖼐𖽡𖽝𖾑， 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔 𖼾𖽠 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽡𖽝𖾑. 2 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼽𖽡𖾑 𖼖𖽗 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼽𖽦𖾑 𖼳𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐𖼳𖽷𖾏 𖼮𖽢𖾑 𖼐𖽡 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼷𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐, 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼦𖽡𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼺𖽷𖾐. 3 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼞𖽑𖽔𖾏 𖽐𖼎𖽻𖾏 𖼋𖽡𖾏 𖼐𖽤𖾐 𖼐𖽡𖾑 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏 , 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽔𖾏. 4 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏, 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼀𖽘𖾏 𖼟𖽡 , 𖼟𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖽃𖽔𖾏.” 5 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽵 𖼗𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼮𖽡𖽝𖾏.
6 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼳𖽷 𖼑𖽡𖾏 𖼊𖽪𖾐𖼽𖽙 𖼦𖽔𖾐 𖼐𖽔 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼉𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼦𖽻𖾏 𖽃𖽻𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼗𖽠𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐. 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖽐𖼷𖽻𖾏 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼗𖽠𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐. 8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 、𖼊𖽪𖾐 𖼗𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖽐𖼚𖽙 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼋𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽔𖾑 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏（𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐）， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼗𖽠𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 .
