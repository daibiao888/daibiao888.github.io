𖼷𖽑𖽙𖾑 𖼚𖽻𖾐 𖼋𖽪𖾏 𖼞𖽡𖾑 𖼊𖽳𖾏

7 𖼺𖽪𖾐 𖼁𖽱 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏，𖼅𖽔𖾏 𖼑𖽘 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖽐𖼮𖽑𖽘𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐， 𖼊𖽡𖾐 𖼷𖽷𖾐 𖽐𖼀𖽠 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠，𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼷𖽷𖾐 𖽐𖼀𖽠 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠，𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏．

10 𖼷𖽷𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼯𖽡，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾐，𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽻𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏，𖼀𖽻𖾐 𖼗𖽙 𖼸𖽠𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼑𖽡𖾏 𖼷𖽷𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠，𖼎𖽺𖾏 𖼀𖽻𖾐 𖼗𖽙 𖼸𖽠𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼷𖽠𖾑𖽑 𖼊𖽑𖽡𖽝𖾐． 11 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 “𖽃𖽦𖾐 𖼞𖽡𖾏 𖼚𖽔𖾏 ”；𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽻𖾐 𖼷𖽷𖾐 𖽐𖼀𖽠 𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 “𖽃𖽦𖾐 𖼞𖽡𖾏 𖼚𖽔𖾏” ，𖽃𖽻𖾐 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼦𖽻𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼋𖽔 𖼙𖽠𖾏．
