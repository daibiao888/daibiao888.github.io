𖼖𖽪𖾑 𖼙𖽻 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
32 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼊𖽻𖾏 𖼗𖽔 𖼦𖽔𖾏 𖼐𖽡𖽙𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏,
𖼖𖽪𖾑 𖼁𖽘𖾏 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏,

33 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：

𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏, 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏.
 
34 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖽐𖼡𖽙𖾏 𖼖𖽻𖾐 𖼞𖽪𖾑, 𖼸𖽺𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼞𖽪𖾑,

36 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :

37 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼀𖽙𖾏,
