7 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏,

8 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏

9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼊𖽻𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,

10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼋𖽔 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏,
𖼊𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖽐𖼎𖽙 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽻𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐,
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼗𖽪𖾏,

12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,

13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽡𖽝𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,

16 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖽐𖼚𖽔 𖼺𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼋𖽻𖾏,

17 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 𖼐𖽔𖾑 ,

18 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼷𖽑𖽔𖾐 𖼼𖽘𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 ,
