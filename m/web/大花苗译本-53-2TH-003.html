16 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪 ， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼯𖽡𖽹𖾏 𖼋𖽱𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼳𖽷𖾏，𖼐𖽱𖾏 𖼾𖽨𖾏 𖼄𖽔𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼎𖽻𖾏 𖼅𖽡！𖼊𖽻𖾏 𖼅𖽪𖾏 𖼮𖽵𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐！
