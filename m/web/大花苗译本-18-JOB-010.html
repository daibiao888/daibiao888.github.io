10
“𖼞𖽪𖾑 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 ,

𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 .
2 𖼞𖽪𖾑 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽡𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼾𖽔𖾏 ,
𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 , 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 .
3 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾏 , 𖼟𖽡 𖼎𖽺𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 , 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 ,
𖼈𖽡𖽝𖾏 𖼙𖽔 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 ,
𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 ?
4 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼋𖽡𖽝𖾏 𖼾𖽙 𖽐𖼡𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔?
𖼟𖽡 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼘𖽪 ?
5 𖼟𖽡 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼋𖽡𖽝𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 ,
𖼟𖽡 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼋𖽡𖽝𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 ?
6 𖼟𖽡 𖼖𖽠𖾑 𖼾𖽱 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 ,
𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼘𖽪𖾏 ?
7 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .
8 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽷𖾏 𖼞𖽪𖾑 , 𖼷𖽷𖾏 𖼞𖽪𖾑 𖼚𖽻𖾐 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼊𖽠𖾏 𖼋𖽡𖾏 ,
𖼈𖽡𖽝𖾏 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼡𖽙𖾏 𖼖𖽻𖾐 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼮𖽡𖾏 𖼙𖽱 .
9 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏，𖼟𖽡 𖼷𖽷𖾏 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼄𖽑𖽻𖾏 𖽃𖽘𖾑 𖽃𖽔 𖼗𖽔𖾏 ,
𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
10 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼘𖽡𖾑𖽑 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼺𖽱𖾏 𖼘𖽡𖾑𖽑 𖼞𖽔𖾏 𖼄𖽡𖾏,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼞𖽑𖽙𖾑 𖼺𖽱𖾏 𖼞𖽔𖾏 𖼄𖽡𖾏 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼘𖽪𖾏 ?
11 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽐𖼎𖽙 𖽐𖼡𖽹𖾏 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼐𖽑𖽘𖾑 ,
𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖽐𖼎𖽙 𖼳𖽠𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 .
12 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽐𖼎𖽙 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 ,
𖼟𖽡 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾.
13 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼐𖽡𖾑 , 𖽐𖼷𖽙𖾑 𖼙𖽔 𖼈𖽺𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 ,
𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖽐𖼷𖽙𖾑 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽡𖾑 .
14 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 , 𖼟𖽡 𖼎𖽺𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 .
15 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 , 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 , 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 ,
𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 ,
𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 .
16 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼾𖽨𖾏 𖼻𖽻𖾏 , 𖼟𖽡 𖽐𖼎𖽑𖽦𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 ,
𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 .
17 𖼟𖽡 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼛𖽠𖾏 𖼞𖽪𖾑 ,
𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 , 𖼺𖽱𖾏 𖼏𖽙 𖼦𖽡𖾏 𖼘𖽑𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 .
 
18 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼖𖽺𖾑 ?
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼊𖽪𖾏 𖼀𖽘𖾏 𖼺𖽘𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 .
19 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖽂𖽪𖾏 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼊𖽻𖾏 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 .
20-21 𖼞𖽪𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼯𖽠 𖼘𖽪 ?
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼮𖽳 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽡𖽝𖾏 𖼯𖽔𖾏 𖼞𖽪𖾑 ,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼺𖽷𖾐 𖼗𖽙 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 ,
𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼚𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 ,
𖼊𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖽃𖽡𖾐 𖼀𖽘𖾏 𖼣𖽔𖾏 .
22 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 ,
𖼟𖽪𖾏 𖼋𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 ,
𖼉𖽺𖾏 𖽃𖽡𖾑 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼺𖽱𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 .”
