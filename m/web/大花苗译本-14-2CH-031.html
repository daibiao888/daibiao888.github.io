2 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼇𖽺𖾐 𖼏𖽵 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼸𖽻𖾏 𖼮𖽡𖽝𖾏 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 , 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 ( “𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼖𖽪𖾑 𖼾𖽡”) 𖽃𖽔 𖼛𖽻𖾏 𖽐𖼚𖽙 𖼉𖽔𖾏 𖼾𖽨𖾏 、 𖽐𖼮𖽢𖾏 𖼙𖽔 、 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 3 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼞𖽑𖽡𖽠𖾏 𖼋𖽠𖾏 𖽐𖼀𖽠 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖽐𖼎𖽙 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 、𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 , 𖼊𖽑𖽡𖽝𖾐 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼑𖽡𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏. 4 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑.
