4 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,

𖼦𖽡𖾐 𖼎𖽺𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼑𖽡𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼈𖽻𖾑,
𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏．
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑,

6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼟𖽡,

𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
7 𖼞𖽪𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼊𖽔𖾐,
