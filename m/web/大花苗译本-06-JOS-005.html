𖽃𖽔𖾐 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏
2 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 , 𖼎𖽺𖾏 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 .” 3 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏, 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 , 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏. 4 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖼐𖽡𖾑 , 𖼟𖽪𖾏 𖼑𖽜 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼋𖽻𖾏 𖽐𖼮𖽷𖾏 , 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼗𖽙𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼞𖽑𖽔𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼺𖽘𖾏 . 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼊𖽻𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼘𖽘𖾑𖽑 𖼾𖽨 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼙𖽔 𖼊𖽻𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖼺𖽷𖾐 . 6 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼅𖽻𖾏 𖼺𖽘𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 , 𖼸𖽙 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼋𖽻𖾏 𖽐𖼮𖽷𖾏 , 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 , 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 . 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽵 𖼗𖽪𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼙𖽔 𖼏𖽵 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼐𖽡𖾑 , 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼄𖽡𖾏 𖽐𖼎𖽙 𖼄𖽪𖾑 . 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼗𖽙𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼙𖽔 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖼺𖽷𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 , 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 .
8 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼞𖽑𖽔𖾏 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾐 𖼖𖽪𖾑 𖼾𖽡 𖽐𖼚𖽙 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 . 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼎𖽑𖽻𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼺𖽘𖾏 𖼣𖽹 𖼮𖽡 𖼑𖽡𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏 .” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖽐𖼷𖽷𖾏 𖼮𖽡 𖼮𖽢 ( “𖼮𖽡 𖼮𖽢 ”𖼟𖽪𖾏 “𖼦𖽡𖾏 𖼚𖽙𖾑 ” ), 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏.
𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏
