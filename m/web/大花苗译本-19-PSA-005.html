𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼚𖽠𖾏 𖼛𖽡𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏．𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑.
5
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏,
𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏！
2 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡.
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏,
𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼋𖽙𖾏.
 
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏,
𖽐𖼚𖽪𖾏 𖼸𖽳 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐.
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼳𖽪𖾐.
6 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼮𖽘𖾑 𖼺𖽘𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐.
 
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼻𖽡𖽝𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼷𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽡 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏.
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖼸𖽙𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽜 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏.
 
9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼷𖽑𖽪𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼘𖽑𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼦𖽡𖾑 𖽐𖼷𖽘𖾏 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐.
10 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼛𖽠𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽜𖾑 𖼟𖽡 𖼺𖽘𖾏 .
 
11 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏, 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼎𖽺𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼻𖽡𖽝𖾏 𖼟𖽡 𖼙𖽔 𖼚𖽙𖾏.
12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼺𖽱𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏.
