𖼾𖽘𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼙𖽠𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏
（𖼊𖽑𖽹𖾏 18：12-14）
15
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼋𖽔𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 ，𖼽𖽔𖾑 𖼑𖽻 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏． 2 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼷𖽹𖾏 𖼯𖽡𖽙𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏．” 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： 4 “ 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼋𖽪𖾏 𖼾𖽘𖾏，𖼛𖽙𖾏 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼋𖽪，𖼋𖽡𖽝𖾏 𖼾𖽙 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼯𖽢𖾏 𖼋𖽪𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼮𖽡𖽝𖾏，𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽪𖾑 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑，𖼸𖽙 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼖𖽺𖾑？ 5 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼙𖽔 𖼟𖽡𖾏 𖼚𖽙𖾏 𖽃𖽔𖾏 𖽐𖼷𖽑𖽷𖾏，𖼞𖽱𖾑 𖼊𖽪𖾑 𖼾𖽘𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏 𖼺𖽷𖾐 𖼗𖽙 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏， 6 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼽𖽺 𖼗𖽔 𖼼𖽷𖾏 𖼼𖽙 𖼋𖽔𖾏，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼙𖽠𖾏 𖼖𖽔𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼋𖽘𖾏，𖼅𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼮𖽑𖽔𖾏 𖼙𖽔．’ 7 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼾𖽔𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻，𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼮𖽑𖽔𖾏 𖼙𖽔，𖼽𖽔𖾑 𖼙𖽔 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼯𖽢𖾏 𖼗𖽜𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼦𖽡𖾐 𖼅𖽙𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼺𖽷𖾐． ”
𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼙𖽠𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏
8 “𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏，𖼟𖽪𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼛𖽡𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑，𖼮𖽔𖾑 𖼛𖽙𖾏 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐，𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽠𖾏 𖼎𖽻𖾐， 𖼮𖽑𖽡𖾐 𖽐𖼞𖽔𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏，𖼸𖽙 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼖𖽺𖾑 ？ 9 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼽𖽺 𖼗𖽔 𖼼𖽷𖾏 𖼼𖽙 𖼋𖽔𖾏，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼋𖽘𖾏，𖼅𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼮𖽑𖽔𖾏 𖼙𖽔．’ 10 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼾𖽔𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻，𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼑𖽜 𖼐𖽱𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼖𖽡𖾏 𖼐𖽡𖾑．”
𖼊𖽪𖾐 𖽐𖼚𖽔𖾏 𖼀𖽡𖾏 𖼚𖽑𖽔𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏
11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐， 12 𖼷𖽔𖾑 𖼏𖽷𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏：‘𖼉𖽺𖾏 𖼘𖽪 ！𖼀𖽡𖾐 𖽃𖽡𖾐 𖼾𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏， 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼇𖽺𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑．’ 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 13 𖼦𖽡𖾐 𖼋𖽙𖾏 𖼀𖽡𖾑 𖼯𖽠𖾏 𖼐𖽑𖽪𖾐，𖼷𖽷𖾑 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼏𖽷𖾏 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼁𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜，𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼙𖽠𖾏．𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼗𖽱𖾏 𖽃𖽔𖾏，𖼖𖽡𖾏 𖼅𖽪𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏． 14 𖼎𖽑𖽔𖾑 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 ，𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻 𖽃𖽡𖾐 𖼋𖽪 𖼮𖽷𖾑 𖼑𖽡𖽝𖾏 𖼮𖽷𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼊𖽔𖾑 𖼋𖽡𖽝 . 15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼾𖽨 𖽐𖼀𖽔𖾏 𖼚𖽔𖾏 𖽐𖼎𖽘𖾏． 16 𖼐𖽱𖾏 𖼊𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾑 𖼊𖽠𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖼀𖽪𖾐 𖽐𖼀𖽔𖾏 𖽃𖽡𖾑 𖼑𖽻𖾏 ，𖽐𖼮𖽠𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼮𖽑𖽹𖾐，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏．
