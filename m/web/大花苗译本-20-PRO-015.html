15
𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼅𖽔𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼸𖽠𖾏 𖼟𖽙𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼛𖽙𖾏 𖼺𖽘𖾏,
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼼𖽔𖾏, 𖼸𖽠𖾏 𖼊𖽱𖾑 𖼟𖽙𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐.
2 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼀𖽻𖾐 𖼟𖽳 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼡𖽔 𖼏𖽔 .
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏、𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 , 𖼐𖽱𖾏 𖼞𖽑𖽔𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 .
4 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼊𖽻𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼦𖽡𖾐 𖼙𖽠𖾏 𖼼𖽘𖾏, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼚𖽔 .
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼉𖽺𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼯𖽷𖾏 .
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼅𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼊𖽳𖾏 𖽐𖼮𖽑𖽹𖾏.
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽔𖾐 𖼋𖽠𖾏 𖼀𖽻𖾐 𖼟𖽳 ；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 .
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 , 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼊𖽻𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 .
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑, 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐；
𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖼺𖽘𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 .
11 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼖𖽔𖾏 𖼢𖽳 𖼙𖽪 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼯𖽡𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏？
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼗𖽔 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 ,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳.
13 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 , 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼎𖽺𖾏 𖼚𖽙𖾏 𖽃𖽔𖾏 𖽐𖼷𖽑𖽷𖾏；
𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏, 𖼖𖽡𖽾 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐.
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼀𖽻𖾐 𖼟𖽳 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼡𖽔 𖼏𖽔 .
15 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽡𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏.
16 𖼅𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼯𖽠, 𖼈𖽡𖽝𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼇𖽔𖾑 𖼊𖽪𖾑 𖼅𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼈𖽡𖽝𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 .
17 𖼑𖽻𖾏 𖼻𖽻𖾐 𖼯𖽔 𖼊𖽔𖾐 𖼈𖽡𖽝𖾏, 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏,
𖼇𖽔𖾑 𖼊𖽪𖾑 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼑𖽨𖾐 𖼛𖽙 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼳𖽪𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏.
18 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽜𖾏 𖼋𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼑𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
19 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼺𖽱𖾏 𖼞𖽑𖽻𖾐 𖼞𖽡𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼺𖽙𖾑 𖼉𖽘𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑, 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏
20 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽙𖾏 𖼉𖽺𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐.
21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽔𖾏 𖼙𖽔 𖼚𖽙𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼸𖽙𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽔𖾏 𖼑𖽪𖾏 .
22 𖼦𖽡𖾐 𖼺𖽪𖾐 𖼦𖽡𖾏 𖼊𖽳𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼊𖽱𖾏 𖼖𖽔𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 ；
𖼊𖽪𖾐 𖼸𖽠𖾐 𖼮𖽡𖽻𖾐, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑.
23 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼸𖽠𖾏 𖼊𖽡𖾐, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑；
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼐𖽑𖽡𖾏, 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼋𖽡𖽝 .
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽑𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼞𖽡𖾑 𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼳𖽔𖾑,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐.
25 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼑𖽡𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐.
26 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐；
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏.
27 𖼦𖽠𖾑 𖼉𖽳 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 , 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽱𖾏 𖼦𖽺𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏.
28 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼊𖽡𖾐；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
29 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽜𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐,
𖼈𖽡𖽝𖾏 𖼑𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 .
30 𖽃𖽔𖾑 𖼅𖽔 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼄𖽻𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼷𖽑𖽘𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾐 𖼐𖽻𖾏 .
31 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
32 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 , 𖼟𖽪𖾏 𖼑𖽔𖾏 𖽐𖼮𖽠𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
33 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏；
𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼺𖽪𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼳𖽻𖾐 𖼢𖽜 .
