𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽳𖾏 𖼊𖽻𖾏 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏
𖼽𖽔𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏,𖼸𖽙𖾏 𖼽𖽦𖾑 𖼊𖽪𖾏 𖼊𖽪𖽝𖾏 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
77
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼞𖽪𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 .
2 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼮𖽵𖾑 ,
𖼞𖽪𖾑 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼳𖽙𖾏 𖼮𖽘𖾏 𖼋𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏.
 
3 𖼞𖽪𖾑 𖼮𖽳𖾐 𖽐𖼊𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼞𖽪𖾑 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏,
𖼞𖽪𖾑 𖽐𖼊𖽱𖾏 𖼅𖽻𖾏 𖽐𖼊𖽱𖾏 𖼗𖽙𖾏 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖼟𖽡𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 . （𖼳𖽡𖾏 𖼖𖽔𖾑）
4 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼡𖽜𖾏 𖽃𖽔𖾑 𖼅𖽔 ,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼋𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 .
5 𖼞𖽪𖾑 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏,
𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏.
6 𖼞𖽪𖾑 𖽐𖼊𖽱𖾏 𖼸𖽙 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏,
𖼎𖽺𖾏 𖼾𖽨𖾏 𖼑𖽪 𖼾𖽨𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼻𖽻𖾏 𖼳𖽹𖾐 𖽐𖼊𖽱𖾏 ：
7 “𖼋𖽡𖽝𖾏 𖼾𖽙 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 ,
𖼦𖽡𖾐 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
8 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏 ?
𖼐𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏 𖼖𖽔𖾏 ?
9 𖼋𖽡𖽝𖾏 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏,
𖼖𖽔𖾏 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼐𖽔𖾑 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼘𖽪?” （𖼳𖽡𖾏 𖼖𖽔𖾑）
10 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐.”
 
11 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏.
12 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼊𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏,
𖼮𖽳𖾐 𖽐𖼊𖽱𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏.
13 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼯𖽡𖽙𖾏,
𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼘𖽑𖽙𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪?
14 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑,
𖼟𖽡 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏.
15 𖼟𖽡 𖼙𖽔 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼮𖽨𖾏 𖼎𖽙𖾑 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
16 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖽃𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼁𖽙𖾏 𖼟𖽡 , 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏,
𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏.
17 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼚𖽪𖾐 𖼘𖽑𖽡𖾑 𖼋𖽠𖾏 𖽃𖽻𖾐 𖼗𖽙 ,
𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 ,
𖼟𖽡 𖼺𖽪𖾐 𖼈𖽙𖾏 𖼎𖽺𖾏 𖼽𖽘𖾏 𖼚𖽔𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐.
18 𖼟𖽡 𖼺𖽙𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖽐𖼚𖽙 ,
𖼺𖽙𖾐 𖼀𖽡𖾏 𖼗𖽺𖾏 𖼮𖽡𖾏 𖽐𖼷𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼊𖽡𖾐 𖼎𖽺𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 .
19 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼟𖽡 𖼸𖽙𖾏 𖼞𖽘𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼟𖽡 𖼊𖽠𖾏 𖼐𖽜𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐.
20 𖼟𖽡 𖼙𖽔 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.
