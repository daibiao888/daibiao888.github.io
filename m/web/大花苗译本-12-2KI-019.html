𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏, 𖼎𖽺𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏, 𖽐𖼀𖽔𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 2 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽃𖽔𖾏 𖽐𖼊𖽠𖾑 𖼳𖽳𖾐 𖼀𖽳 𖼐𖽔, 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽪𖾐 𖼗𖽻 𖼖𖽔𖾏 𖽐𖼀𖽔𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐, 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏. 3 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔、 𖼖𖽔𖾏 𖼨𖽑𖽢𖾐、 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼺𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼳𖽻𖾏. 4 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼖𖽔𖾑 𖼀𖽳 𖼳𖽔𖾑 𖼮𖽡𖾑 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏, 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼖𖽔𖾏 𖼡𖽑𖽔𖾐． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼐𖽡𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.” 5 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏. 6 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼐𖽙𖾑𖽑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼞𖽪𖾑, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏. 7 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽡𖽠𖾏 ( “𖼮𖽡𖽠𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“ 𖼟𖽙𖾏 𖼖𖽡𖽾 𖼁𖽺𖾏 𖼚𖽔𖾏” )𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼄𖽻𖾏, 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏． 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.”
𖼽𖽔𖾏 𖼳𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑
（𖼺𖽹 37： 8 - 20）
8 𖼖𖽔𖾑 𖼀𖽳 𖼳𖽔𖾑 𖼮𖽡𖾑 𖼺𖽷𖾐 𖼅𖽻, 𖽐𖼮𖽷𖾐 𖼸𖽪𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼊𖽻𖾏 𖼖𖽡 𖼐𖽔, 𖼐𖽱𖾏 𖼺𖽪𖾐 𖽐𖼷𖽙𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽹𖾏 𖼾𖽡 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼖𖽔𖾑 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 9 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼞𖽪𖾐 𖼳𖽷 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽳 𖼦𖽔𖾐 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 , 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: 10 “𖼅𖽡 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽵𖾑 𖼟𖽡 𖼐𖽑𖽡𖾏： 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏. 11 𖼑𖽔𖾏 𖼘𖽪! 𖼟𖽡 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼽𖽔𖾏 𖼳𖽪 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼺𖽷𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏? 12 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼡𖽑𖽙𖾑 𖼖𖽻𖾐 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼞𖽙𖾑 𖼺𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼦𖽔𖾐 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼖𖽡𖾏 𖼺𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽑𖽡 · 𖼖𖽔𖾑 𖼺𖽔𖾐, 𖼋𖽡𖾏 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼋𖽡𖾏 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼘𖽪? 13 𖼦𖽔𖾐 𖼄𖽔𖾐 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼀𖽔 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼳𖽡𖾑 𖼇𖽔 𖽂𖽫𖾐 𖼽𖽡𖽾𖾑 𖼙𖽻 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼳𖽡𖾑 𖼐𖽔 𖽐𖼎𖽙 𖼽𖽡𖾑 𖽂𖽔𖾐 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?”
14 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖽐𖼊𖽠𖾑 𖼄𖽻𖾏 𖼗𖽙𖾏, 𖼟𖽡𖽠 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽠𖾑 𖼄𖽻𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼡𖽑𖽜𖾐 𖼮𖽡𖽝𖾏. 15 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽡 𖼙𖽔 𖼷𖽷𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐. 16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻， 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼡𖽑𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏， 𖼽𖽔𖾑 𖼑𖽻 𖼳𖽡𖾑 𖼐𖽔 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽔𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏. 17 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼽𖽔𖾏 𖼳𖽪 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖽙 𖽐𖼎𖽙 𖼞𖽪𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏， 18 𖼞𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼁𖽠𖾏 𖼼𖽘𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼯𖽘𖾏 𖼋𖽠 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏, 𖼟𖽪𖾏 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼈𖽳𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽻𖾐 𖼊𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏. 19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.”
𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
（𖼺𖽹 37：21 - 38）
20 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑, 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼐𖽔 𖼮𖽡𖾑 𖼖𖽡𖾏, 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏. 21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:
𖼳𖽡 𖼣𖽔𖾑 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼟𖽡, 𖼋𖽔𖾏 𖼊𖽔𖾐 𖼡𖽔𖾏 𖼚𖽙𖾏 𖼟𖽡,
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼇𖽜𖾏 𖼇𖽜𖾏 𖼖𖽡𖾑 𖼇𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡.
 
22 𖼟𖽡 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏? 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼡𖽔 𖼋𖽱𖾏?
𖼮𖽘𖾏 𖼺𖽘𖾐 𖼗𖽙, 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖽃𖽔𖾑 𖼅𖽔, 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼘𖽪?
𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑!

27 𖼟𖽡 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼟𖽡 𖼁𖽺𖾏 𖼋𖽔𖾏,

29 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！ 𖼞𖽪𖾑 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼎𖽻𖾏 𖼅𖽡, 𖼅𖽡 𖼳𖽡𖽻𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼘𖽘𖾑𖽑, 𖼦𖽡𖾏 𖼑𖽡𖽻𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐, 𖼎𖽺𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖽃𖽔𖾐 𖼡𖽜𖾐, 𖼅𖽡 𖼽𖽔𖾑 𖼯𖽡𖽙 𖼮𖽘𖾑 𖼳𖽻𖾏 𖼤𖽜𖾏, 𖼯𖽡𖽙 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏, 𖼑𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑. 30 𖼽𖽨 𖼊𖽔𖾏 𖼾𖽡𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼗𖽔 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐. 31 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏, 𖼎𖽑𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏. 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼳𖽙𖾑, 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑. 32 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼸𖽙𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑： 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼀𖽙𖾑 𖼈𖽙𖾏, 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼚𖽔𖾏 𖼙𖽻 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖽐𖼊𖽻𖾏 𖼙𖽻. 33 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼋𖽱 𖼋𖽔𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼋𖽱 𖼺𖽷𖾑 𖼡𖽜𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏. 34 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼊𖽔𖾐 𖼊𖽜𖾑, 𖼎𖽺𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽔𖾐 𖼊𖽜𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼖𖽪𖾑 𖼙𖽻.”
