52
𖼳𖽡 𖼣𖽔𖾑 𖼘𖽪！ 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏! 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏！
𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 𖼮𖽡𖽝𖾏,
𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！ 𖼐𖽑𖽘𖾑 𖼟𖽡 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼮𖽡𖽝𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽳 𖼐𖽡𖾑 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
2 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！ 𖼽𖽔𖾑 𖽐𖼮𖽑𖽡𖽙𖾏 𖼺𖽘𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏,
𖼳𖽠𖾑 𖼗𖽙𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐,
𖼳𖽡 𖼣𖽔𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 (“𖼖𖽡𖾑 𖼉𖽪𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏”) 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼘𖽪！
𖼽𖽔𖾑 𖼚𖽠𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽻𖾏.
 
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: “𖼅𖽡 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼊𖽻𖾏 𖼋𖽺, 𖼅𖽔𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏.” 4 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: “𖼺𖽪𖾐 𖼁𖽱 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 , 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑? 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏, 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏. 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏, 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼑𖽔𖾏 𖼘𖽪 ! 𖼟𖽪𖾏 𖼞𖽪𖾑 .”
 
7 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼻𖽻𖾏 𖼄𖽻𖾏, 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 ,
𖼎𖽻𖾏 𖼄𖽻𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖽃𖽡𖾑,
𖽐𖼎𖽙 𖼳𖽡 𖼣𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽘𖾏.”
𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼊𖽠𖾏 𖼐𖽜𖾑 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 .
8 𖼑𖽻 𖼘𖽪 ! 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽃𖽡𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼗𖽙 , 𖽃𖽡𖾐 𖼸𖽻 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽙𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏.
9 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼑𖽡𖾏 𖼡𖽔𖾑 𖼚𖽻𖾏 𖼦𖽡𖾏 𖼚𖽘𖾏 𖼘𖽪！
𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏, 𖽃𖽡𖾐 𖼸𖽻 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼋𖽘𖾏,
𖼚𖽠𖾏 𖼮𖽨𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼋𖽘.
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼋𖽻𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽔𖾑 𖼅𖽔𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑,
𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼋𖽘𖾏.
11 𖼅𖽡 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏！ 𖼇𖽳𖾑 𖼺𖽘𖾏！
𖼎𖽑𖽻𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏.
𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏．
𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼘𖽪,
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼾𖽨𖾏.
12 𖼅𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼏𖽠𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼏𖽘 𖼀𖽜𖾑．
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼅𖽡 𖼊𖽡𖾑 𖼢𖽻𖾏.
 
13 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐,
𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐.
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽜 𖼐𖽱𖾏 (“𖼐𖽱𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼟𖽡”) 𖼊𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼉𖽘𖾏 𖼾𖽔 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼐𖽱𖾏 𖼼𖽘𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐.
15 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼷𖽔𖾑 𖼨𖽑𖽪𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏．
