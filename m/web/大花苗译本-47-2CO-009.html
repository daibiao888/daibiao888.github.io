6 “𖼯𖽠 𖼯𖽡𖽙 𖼯𖽠 𖼳𖽻𖾏， 𖼮𖽡𖽻𖾐 𖼯𖽡𖽙 𖼮𖽡𖽻𖾐 𖼳𖽻𖾏”， 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏． 7 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼯𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽜𖾏 𖼑𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽔𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑， 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 ． 8 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽠𖾏 𖼄𖽔𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼮𖽡𖽻𖾐 𖼮𖽡𖽻𖾐 𖼮𖽑𖽨𖾏 𖼎𖽻𖾏 𖼅𖽡， 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏， 𖼎𖽺𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 ． 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：
“𖼐𖽱𖾏 𖽃𖽺𖾐 𖼯𖽪𖾏 𖼄𖽔𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑 𖽐𖼀𖽘𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏，
𖼐𖽱𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼷𖽷𖾐 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏．”
