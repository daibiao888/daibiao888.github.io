𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽙 𖼗𖽔 𖽃𖽡𖾑 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼺𖽳 𖼊𖽑𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐”).
18 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 ,
𖼎𖽺𖾏 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 .
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐 .
19 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼽𖽔𖾐 𖼞𖽙 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 ,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖽐𖼚𖽙 𖼙𖽻 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽐𖼎𖽙𖾏 .”
20 𖼀𖽔𖾑 𖼖𖽔 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 , 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 , 𖼐𖽑𖽡𖾏 :
“𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼊𖽑𖽪 𖼇𖽻𖾑 ,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔 .
21 𖼀𖽔𖾑 𖼖𖽔 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼮𖽡𖾑 𖼐𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 , 𖼐𖽑𖽡𖾏 :
𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 𖽐𖼎𖽜𖾏 ,
𖼟𖽡 𖼖𖽪𖾑 𖼉𖽳𖾏 𖼟𖽪𖾏 𖼁𖽔 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼷𖽔𖾏 𖽐𖼚𖽙 .
22 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼮𖽡𖾑 𖼐𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽻 𖼼𖽙 ,
𖼸𖽙 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼙𖽠𖾏 .”
23 𖼀𖽔𖾑 𖼖𖽔 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 , 𖼐𖽑𖽡𖾏 :
“ 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏! 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 , 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 ?
24 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼊𖽑𖽡 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼋𖽔𖾏 ,
𖼟𖽡𖾏 𖼦𖽺𖾏 𖼽𖽔𖾏 𖼳𖽪 ,
𖼟𖽡𖾏 𖼦𖽺𖾏 𖼳𖽡𖾑 𖼀𖽳 ,
𖼐𖽱𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 .”
25 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼀𖽔𖾑 𖼖𖽔 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 , 𖼀𖽔𖾑 𖼖𖽳 𖼖𖽔𖾏 𖼺𖽷𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏 .
