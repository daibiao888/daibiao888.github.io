3
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 ! 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼗𖽪𖾏、 𖼦𖽡𖾏 𖼡𖽳𖾏、 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑．
2 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐,
𖼦𖽡𖾐 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼦𖽡𖾐 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
3 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼟𖽪𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏,
𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽳𖾑 𖼘𖽳𖾐,
𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼷𖽷𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑.
4 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼗𖽔 𖼦𖽡𖾐 𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼅𖽔、 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑, 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼚𖽠𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑.
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼯𖽡𖽙𖾏,
𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏,
𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ,
𖼯𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
 
6 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽻𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽻 𖽐𖼊𖽑𖽘𖾐 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑,
𖼞𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼟𖽙𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏.
7 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼮𖽔𖾑 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 ,
𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏.
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼏𖽵 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼾𖽨𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼾𖽨𖾏.”
 
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽜𖾏 𖼋𖽙𖾏 𖼞𖽪𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏.
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖼦𖽡 𖼸𖽱 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 ,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏, 𖼽𖽔𖾑 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.
 
9 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏,
𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑 .
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 ( “𖼖𖽡𖾑 𖼉𖽪𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 ”) 𖼞𖽪𖾑 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖽃𖽡𖾑,
𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼳𖽷 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼀𖽘𖾑𖽑 𖽃𖽡𖾑 𖼋𖽔𖾏,
𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 .
 
11 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼟𖽡 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖼾𖽨𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼾𖽨𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼷𖽑𖽷𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼟𖽡 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏.
12 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼷𖽷𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼮𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏.
13 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑, 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼢𖽻 𖼀𖽱𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽠𖾏.”
𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼮𖽵𖾑 𖼅𖽳𖾏
14 𖼳𖽡 𖼣𖽔𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪 ！ 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏．
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ！ 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏．
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪! 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼙𖽔 𖼛𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑.
15 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼋𖽘 ,
𖼖𖽠𖾑 𖼋𖽠𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 .
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
16 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼐𖽑𖽡𖾏:
“𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 !
𖼳𖽡 𖼣𖽔𖾑 𖼘𖽪 ! 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼚𖽻𖾑 𖼖𖽠𖾏 !
17 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽡𖽙𖾐 𖼞𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼘𖽑𖽙𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼐𖽡𖽙𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 ,
𖼎𖽺𖾏 𖼑𖽜 𖼟𖽡 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏.
18 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼟𖽡, 𖼑𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽳𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼞𖽱𖾑 𖼏𖽷𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖼐𖽱𖾏 𖼸𖽻𖾏.
19 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑, 𖼦𖽡 𖼸𖽱 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑.
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔, 𖼅𖽔𖾏 𖼻𖽻𖾏 𖽐𖼷𖽷𖾏.
20 𖼸𖽙 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼺𖽷𖾐 𖼗𖽙, 𖼦𖽡 𖼸𖽱 𖼅𖽡 ,
𖼞𖽪𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖽐𖼷𖽷𖾏, 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 .”
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
