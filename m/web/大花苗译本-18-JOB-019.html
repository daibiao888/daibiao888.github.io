10 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏 ,

11 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 ,
𖼑𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 .
12 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 ,

𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼮𖽪𖾐 𖼾𖽡 .
𖼽𖽙 𖼀𖽳 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽙𖾏 𖼽𖽺 𖼗𖽔 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐
13 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼮𖽡𖽺𖾏 𖼡𖽔𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 ,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼀𖽻𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼑𖽡𖽻𖾏 .
14 𖼞𖽪𖾑 𖼑𖽜𖾏 𖼮𖽘𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼊𖽪𖾏 𖼺𖽘𖾏 ,
𖼞𖽪𖾑 𖼽𖽺 𖼗𖽔 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 .
