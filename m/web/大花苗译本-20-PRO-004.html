𖼽𖽔𖾑 𖼑𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏
4
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼘𖽪！ 𖼽𖽔𖾑 𖼑𖽻 𖼉𖽺𖾏 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖼽𖽔𖾑 𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐；
2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡 , 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖽐𖼞𖽡𖽝𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 (“𖼡𖽑𖽔𖾐 𖼇𖽺𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏”).
3 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼊𖽪𖾐,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
4 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼡𖽑𖽔𖾐 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼮𖽡𖽝𖾏,
𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏；
5 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 .
6 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏 𖼟𖽡 ,
𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 .
7 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖽐𖼚𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐( 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “ 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼅𖽻𖾏 𖼘𖽑𖽪𖾑 𖼀𖽻𖾐 𖼊𖽘𖾐 ” ).
8 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐,
𖼁𖽔 𖼮𖽡𖽝𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .
9 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼄𖽔𖾐 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼟𖽡 .
 
10 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻，𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏 ,
𖼟𖽡 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏.
11 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼋𖽘𖾏,
𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
12 𖼟𖽡 𖼅𖽻𖾏 𖼞𖽡𖾑, 𖼷𖽑𖽔𖾏 𖼏𖽔𖾏 𖼞𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏,
𖼟𖽡 𖼏𖽘, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 .
13 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼋𖽻 𖼺𖽘𖾏；
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑.
14 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 .
15 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼽𖽔𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼅𖽻𖾏 𖼙𖽠𖾏.
16 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼀𖽱𖾏,
𖼦𖽡𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼢𖽻 , 𖼀𖽱𖾏 𖼖𖽔𖾏 𖼀𖽱𖾏 𖼦𖽡𖾐 𖼡𖽘𖾐,
17 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔𖾏 𖼳𖽡𖽝𖾐 𖼑𖽻𖾏 ；
𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖽃𖽔𖾏 𖼮𖽡𖽠𖾑 𖼦𖽻𖾏 .
 
18 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 , 𖼺𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼁𖽙𖾏 𖼞𖽡𖾑 ,
𖽃𖽺𖾐 𖼮𖽡𖾏 𖽃𖽺𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑, 𖼸𖽙 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼋𖽘 𖼐𖽑𖽪𖾐.
19 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼺𖽱𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼾𖽨𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼢𖽻 .
 
20 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡𖾏 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 ,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 .
21 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 ,
𖼽𖽔𖾑 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏.
22 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼋𖽘𖾏,
𖼎𖽺𖾏 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼮𖽡𖾏 𖼙𖽱 .
23 𖼟𖽡 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼎𖽑𖽻𖾏 𖼼𖽙 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐” ),
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏, 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 .
24 𖼟𖽡 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼗𖽔𖾏 𖼺𖽘𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽠𖾏 𖼼𖽘𖾏.
