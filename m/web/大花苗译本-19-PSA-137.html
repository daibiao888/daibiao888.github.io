𖼀𖽡𖾐 𖼙𖽔 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪𖾏,
𖽂𖽪𖾏 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼳𖽡 𖼣𖽔𖾑 𖼎𖽺𖾏 𖼐𖽡𖽝𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏.
2 𖼀𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖼢𖽜𖾏 𖼮𖽘𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼙𖽱 𖼙𖽱 𖼮𖽡𖽝𖾏.
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼀𖽡𖾐,
𖼽𖽔𖾑 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖽐𖼞𖽻𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼀𖽡𖾐, 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏, 𖼐𖽑𖽡𖾏:
“𖼨𖽑𖽪𖾏 𖽃𖽡𖾐 𖼸𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖽐𖼞𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼑𖽻！”
 
4 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏
𖼨𖽑𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽻𖾏 𖼖𖽺𖾑?
5 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！
𖼞𖽪𖾑 𖼮𖽔𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼟𖽡, 𖼯𖽪𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏！
6 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡, 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼙𖽔 𖽃𖽡𖾑,
𖼯𖽪𖾏 𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼘𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾏 𖽐𖼎𖽪𖾏 𖼮𖽡𖽝𖾏！
7 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔,
𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 ! 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏!
𖼎𖽑𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙 𖼊𖽠𖾏！”
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖽃𖽡𖾑！
8 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼖𖽪𖾑 𖼙𖽻 ( “𖼙𖽻” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏”) 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽃𖽡𖾑 𖼘𖽪 ,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼟𖽡 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖽃𖽡𖾑, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
9 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽔𖾑 𖼟𖽡 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼇𖽜𖾏 𖼇𖽜𖾏 𖽐𖼮𖽠𖾏 𖼎𖽻𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽃𖽡𖾑,
