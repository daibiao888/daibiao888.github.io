𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼾𖽨𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪𖾏 ! 𖼟𖽡 𖼙𖽔 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼞𖽪𖾑,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼑𖽻 𖼟𖽡 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼋𖽘 ,
𖼟𖽡 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑, 𖼎𖽺𖾏 𖼳𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏.
𖼞𖽪𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼚𖽙𖾏,
𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽪𖾏 𖼚𖽙𖾏 𖼞𖽪𖾑 .
8 𖼷𖽑𖽔𖾏 𖼟𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑,
𖼞𖽪𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽔𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖽐𖼎𖽙 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 !”
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼏𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑.
9 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼦𖽡𖾐 𖼎𖽺𖾏 𖼻𖽡𖽝𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼮𖽡𖽝𖾏 𖽐𖼮𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼷𖽜𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼾𖽨𖾏 𖼯𖽵𖾏 𖼾𖽨𖾏.
𖼻𖽻𖾏 𖼽𖽺 𖼗𖽔 𖼖𖽔𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏
10 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑,
𖼚𖽻𖾐 𖼺𖽘𖾐 𖼊𖽪𖾏 𖼅𖽔𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏,
𖼟𖽪𖾏 𖼽𖽺 𖼗𖽔 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼊𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼢𖽻, 𖼐𖽑𖽡𖾏:
“𖼛𖽠𖾏 𖼑𖽪𖾏 𖼐𖽱𖾏, 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼛𖽠𖾏 𖼑𖽪𖾏 𖼐𖽱𖾏,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼮𖽑𖽵𖾑, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼳𖽠𖾏 𖼐𖽱𖾏.
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼼𖽷𖾏 𖼁𖽻𖾏.”
11 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐,
