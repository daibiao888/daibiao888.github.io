“𖼞𖽪𖾑 𖼟𖽪𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝，𖼞𖽪𖾑 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙 𖼮𖽑𖽪𖾑 𖼨𖽑𖽢𖾏 𖽃𖽡𖾑． 2 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼦𖽡𖾐 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽑𖽘𖾐 𖼺𖽘𖾏；𖼋𖽡𖾏 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽢𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽡𖽻𖾐． 3 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼅𖽡 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏． 4 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼚𖽙 ，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖽐𖼚𖽙，𖽃𖽔𖾑 𖼯𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏，𖼐𖽱𖾏 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐，𖼅𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼚𖽙，𖼅𖽡 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 5 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏，𖼅𖽡 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼯𖽡．𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼚𖽙，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐．𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏，𖼅𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏． 6 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼙𖽠𖾏，𖼡𖽑𖽔𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼯𖽘𖾏 𖼋𖽠 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏． 7 𖼅𖽡 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙，𖼞𖽪𖾑 𖼗𖽪𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖽐𖼚𖽙，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼙𖽱 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 ，𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑． 8 𖼅𖽡 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼞𖽪𖾑 𖼉𖽺𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏，𖼅𖽡 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏． 9 𖼞𖽪𖾑 𖼉𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼅𖽡 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖽐𖼚𖽙． 10 𖼅𖽡 𖼮𖽔𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑，𖼅𖽡 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖽐𖼚𖽙，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑，𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖽐𖼚𖽙．”
