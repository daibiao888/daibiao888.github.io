23
“𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼙𖽳𖾐 𖼦𖽡𖾏 𖼷𖽱𖾏 𖼐𖽑𖽡𖾏 𖼀𖽘𖾏 𖼮𖽢𖾏 𖼗𖽪𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 . 2 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏； 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽪𖾐 𖼸𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐, 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼑𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳； 3 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽪𖾐 𖼡𖽑𖽠𖾑 𖼊𖽪𖾐 𖼳𖽻𖾏 . ”
4 “𖼮𖽔𖾑 𖽐𖼮𖽷𖾐 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼑𖽨𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼛𖽙𖾏 𖼙𖽠𖾏, 𖼽𖽔𖾑 𖼮𖽤𖾐 𖼗𖽙𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼙𖽠𖾏. 5 𖼮𖽔𖾑 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼟𖽡 𖼑𖽡𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼊𖽻𖾏 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼸𖽺𖾏 𖼮𖽡𖽝𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼮𖽑𖽳𖾑 𖽐𖼎𖽘𖾏 𖼐𖽑𖽤𖾑 𖼺𖽘𖾏 . ”
6 “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽪𖾐 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼺𖽘𖾏 𖼑𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 . 7 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼟𖽙𖾏 𖼚𖽡𖾐 . 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 . 8 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼚𖽑𖽡𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼺𖽘𖾏 , 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼞𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼎𖽙𖾑 𖼡𖽘𖾐 𖼎𖽙𖾏 𖼇𖽻𖾑 𖼺𖽘𖾏 .
9 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 , 𖼀𖽻𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 . ”
𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏
10 “𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐 , 𖼳𖽻𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼮𖽡𖽝𖾏 , 11 𖼈𖽡𖽝𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼳𖽙𖾏 𖼋𖽘𖾏 , 𖼦𖽡𖾐 𖼯𖽡𖽙 𖼦𖽡𖾐 𖼳𖽻𖾏 , 𖼟𖽙𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖼑𖽻𖾏 𖼳𖽙𖾏 𖽃𖽡𖾑 , 𖼟𖽙𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼎𖽺𖾏 𖼑𖽻𖾏 . 𖼟𖽡 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽐𖼎𖽙 𖼉𖽘𖾏 𖼞𖽗𖾐 𖼖𖽗 , 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 . ”
12 “𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 , 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼳𖽙𖾏 , 𖼟𖽙𖾏 𖼑𖽨𖾏、𖼑𖽜𖾏 𖼖𖽔𖾑、𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼳𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 . ”
13 “𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼻𖽙𖾑 . 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼷𖽷𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 .”
𖼷𖽷𖾐 𖼋𖽪 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼻𖽙𖾑
（𖼋𖽠𖾏 34：18 - 26；𖼎𖽺𖾏 16：1 -17）
14 “𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼷𖽷𖾐 𖼟𖽻𖾐 , 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 . 15 𖼟𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 , 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖽃𖽡𖾑 , 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼀𖽡 𖼖𖽑𖽡𖾏 (“𖼽𖽔𖾏 𖼀𖽡 𖼖𖽑𖽡𖾏”𖼟𖽪𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼖𖽪𖾑 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏 , 𖼟𖽪𖾏 𖼖𖽑𖽡𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼖𖽪𖾑 𖼷𖽷𖾐、𖼚𖽻𖾐 𖼖𖽑𖽡𖾏 ) 𖼐𖽡𖾑 𖽐𖼚𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼏𖽵 , 𖼑𖽻𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 , 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼁𖽙𖾏 𖼞𖽪𖾑 . 16 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏, 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽱𖾏 𖼳𖽻𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼳𖽦𖾑 𖽃𖽺𖾐 𖼻𖽻𖾏. 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼋𖽘𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 , 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼊𖽡𖾏 . 17 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼾𖽠 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼷𖽷𖾐 𖼟𖽻𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . ”
18 “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑 𖽐𖼎𖽙 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼖𖽑𖽘𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼖𖽑𖽘𖾑 𖼑𖽡𖾏 𖼛𖽙𖾏 𖼷𖽷𖾐 𖼸𖽙 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 .
19 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼳𖽦𖾑 𖼽𖽔𖾑 𖼺𖽘𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 . 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼮𖽑𖽷𖾏 𖼄𖽡𖾏 𖼦𖽻𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏 .”
𖼏𖽵 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼨𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼁𖽺𖾏 𖼮𖽢𖾑 𖼐𖽔
20 “𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 , 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 , 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼸𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 . 21 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼋𖽔𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 , 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼷𖽜𖾑 𖼐𖽱𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼯𖽔𖾏 𖼅𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 .
22 𖼟𖽡 𖼮𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏 , 𖼸𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 , 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐 .
