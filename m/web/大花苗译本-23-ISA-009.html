5 𖼞𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼎𖽻𖾏 𖽃𖽡𖾑,

6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼗𖽙 ,

16 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼞𖽡𖾑 𖼺𖽘𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽡 𖼋𖽙 .
17 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼮𖽵𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼐𖽑𖽨𖾑 𖼗𖽪𖾏, 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼖𖽔𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼚𖽘𖾐 .
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼈𖽡𖽝𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼳𖽻𖾏 𖼎𖽙𖾑 ,

18 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑,
𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼞𖽡𖾑 𖼁𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏,
𖼐𖽡𖽙𖾐 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼯𖽡,
𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏, 𖼀𖽡 𖼙𖽻𖾏 𖼀𖽡𖾏 𖼗𖽜𖾏 𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐.
19 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼷𖽑𖽷𖾑 𖼸𖽙,
