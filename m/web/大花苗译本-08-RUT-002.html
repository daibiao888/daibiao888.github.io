2
𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖼑𖽡𖾏 𖼈𖽻𖾑 𖼽𖽡𖾑 𖼖𖽡 𖼄𖽡𖾐 𖼘𖽳 𖼑𖽡𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼑𖽜𖾏 𖼮𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 , 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼻𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽙𖾏 𖽐𖼷𖽷𖾏 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏.
𖼘𖽪𖾏 𖼊𖽳 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼡𖽻𖾐
2 𖼄𖽙𖾑 𖼽𖽔𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪𖾏 𖼊𖽳 𖽐𖼎𖽙 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼯𖽪𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼅𖽳𖾏，𖼎𖽺𖾏 𖼸𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼡𖽻𖾐.” 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪! 𖼟𖽡 𖼮𖽔 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏.” 3 𖼘𖽪𖾏 𖼊𖽳 𖼅𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼋𖽔𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼡𖽻𖾐. 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼸𖽪𖾏 𖼚𖽔𖾏𖼽𖽡𖾑 𖼖𖽡 𖼄𖽡𖾐 𖼘𖽳 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐. 4 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼎𖽑𖽻𖾏 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖼋𖽔𖾏, 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐！” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡！” 5 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼑𖽪 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽜𖾑 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼾𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 ? ” 6 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽪𖾏 𖼖𖽜𖾑 𖼄𖽙𖾑 𖼽𖽔𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖼎𖽑𖽻𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑. 7 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼯𖽪𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼐𖽡𖾑, 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐 𖼅𖽙 𖼟𖽪𖾏 𖼡𖽑𖽹𖾐 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼷𖽻𖾏 𖼺𖽘𖾏 𖼳𖽙𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑. 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼀𖽘𖾏 𖼐𖽡𖾑, 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼉𖽺𖾏 𖼗𖽠𖾏 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏, 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼡𖽻𖾐.”
8 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖽐𖼎𖽙 𖼘𖽪𖾏 𖼊𖽳 𖼐𖽑𖽡𖾏: “𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪! 𖼑𖽻 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼡𖽻𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼡𖽻𖾐. 9 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼯𖽡𖽙𖾐 𖼙𖽡𖽝𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏. 𖼞𖽪𖾑 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼐𖽑𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼟𖽡. 𖼟𖽡 𖼮𖽔𖾑 𖽐𖼡𖽑𖽜𖾏 𖽃𖽻𖾐 𖼋𖽘𖾏 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼦𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼮𖽑𖽳𖾑 𖼗𖽙𖾏 𖽃𖽡𖾑．” 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼘𖽪𖾏 𖼊𖽳 𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼮𖽑𖽔𖾏 𖼙𖽔, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏 , 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑?” 11 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏:“ 𖼋𖽳 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼈𖽻𖾑 𖼋𖽔 𖼙𖽠𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾐 𖼁𖽙𖾐 𖽃𖽡𖾑, 𖼊𖽑𖽡𖽝𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼇𖽳𖾑 𖼟𖽡 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖽐𖼎𖽙 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽡𖾐, 𖼚𖽔𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼀𖽻𖾐 𖽃𖽡𖾑, 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏. 12 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼎𖽙𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡. 𖼟𖽡 𖼋𖽔𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼀𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼷𖽻𖾏 𖼎𖽙𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡.” 13 𖼘𖽪𖾏 𖼊𖽳 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪! 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑.”
