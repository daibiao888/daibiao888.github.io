𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 33：1 - 20）

𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼳𖽡𖽝𖾏 𖼳𖽡𖾑 𖼀𖽔𖾑. 2 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼾𖽻 𖼸𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼗𖽠𖾏 𖽃𖽡𖾑 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖼳𖽪𖾐. 3 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼀𖽡𖾏 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼑𖽜 𖼀𖽔𖾑 𖼖𖽡 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘, 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼁𖽠 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼾𖽻 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽐𖼎𖽙 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼼𖽘𖾏 𖼼𖽔𖾏. 4 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘. 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏.” 5 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼊𖽡𖾑 𖼯𖽡𖽳𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐， 𖼑𖽜 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘, 6 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑, 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏, 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐, 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽃𖽔𖾏 𖼸𖽷𖾑 𖼅𖽪𖾏, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼊𖽱𖾑 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐. 7 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏. 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽔𖾏 𖼳𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼞𖽑𖽡𖽠𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽑𖽡𖽝𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑, 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏. 8 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽹𖾏 𖼮𖽑𖽡𖽠𖾏 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐. 9 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏. 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼮𖽑𖽵𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼐𖽑𖽤𖾑 𖼇𖽔𖾑 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼡𖽑𖽙𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖽃𖽡𖾑.
