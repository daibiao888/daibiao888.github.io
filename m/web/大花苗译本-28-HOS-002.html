2
𖼅𖽡 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼄𖽡𖾐 (“𖽃𖽔𖾐 𖼄𖽡𖾐 ” 𖼟𖽪𖾏 “𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏”), 𖼨𖽑𖽪𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼻𖽷𖾑 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽪𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 (“𖼖𖽪𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 ”𖼟𖽪𖾏 “𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐”).
2 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏, 𖼅𖽡 𖼽𖽔𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼐𖽤𖾐,
𖼞𖽪𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖼈𖽻𖾑.
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏
𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏.
3 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼚𖽠𖾏 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼺𖽘𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼺𖽱𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏, 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 ,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼡𖽑𖽔𖾐 𖼡𖽑𖽜𖾏 𖼋𖽔 𖼙𖽠𖾏.
4 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼗𖽙 .
5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼺𖽘𖾏,
𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼋𖽘 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼳𖽡𖽝𖾐、 𖽃𖽻𖾐、 𖼚𖽻𖾐 𖼾𖽘𖾐、 𖼅𖽘𖾏、 𖼛𖽙𖾏、 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼦𖽻𖾏 𖼷𖽷𖾐,
𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼯𖽡𖽙𖾏.
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼺𖽙𖾑 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖽐𖼊𖽪𖾏 𖼮𖽑𖽤 𖼙𖽻𖾏 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏.
7 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼸𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼎𖽙𖾑 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑, 𖼈𖽡𖽝𖾏 𖼖𖽠𖾑 𖼦𖽡𖾐 𖼯𖽢 ,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏,
𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼈𖽻𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 .”
8 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼡𖽻𖾐 𖼖𖽻𖾏、 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖽐𖼎𖽙 𖼛𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 ,
𖼎𖽺𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼅𖽻 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼀𖽔𖾑 𖼖𖽡.
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏, 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼡𖽻𖾐 𖼖𖽻𖾏、 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 ,
𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼟𖽪𖾏 𖼚𖽻𖾐 𖼾𖽘𖾐 𖽐𖼎𖽙 𖼅𖽘𖾏 𖼦𖽔𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 .
10 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
11 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼛𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑、 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐、𖼺𖽡𖽝𖾐 𖽃𖽡𖾐、 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏,
𖼟𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼊𖽪𖾏 𖽐𖼮𖽳 𖼳𖽙𖾏 𖼮𖽡𖽝𖾏.
12 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐,
𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏:“𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 .”
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼊𖽻𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏,
𖼟𖽙𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼑𖽻𖾏 .
13 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼎𖽻𖾏 𖼀𖽔𖾑 𖼖𖽡 𖽃𖽡𖾑,
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼎𖽻𖾏 𖽐𖼡𖽜𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼼𖽘𖾏,
𖼸𖽙𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑, 𖼈𖽡𖽝𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 .
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐
14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽻𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏,
𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏.
15 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 ,
𖼎𖽺𖾏 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼽𖽔𖾏 𖼞𖽙 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼊𖽡𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼺𖽱𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏𖼣𖽹 𖼮𖽡 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼳𖽷𖾑(“𖼽𖽡𖾑 𖼳𖽷𖾑”𖼟𖽪𖾏 “𖼞𖽪𖾑 𖼈𖽻𖾑”), 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼀𖽔𖾑 𖼖𖽡 (“𖼀𖽔𖾑 𖼖𖽡” 𖼟𖽪𖾏 “𖼞𖽪𖾑 𖼮𖽵𖾑”). 17 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼀𖽔𖾑 𖼖𖽡 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏, 𖼋𖽡𖾏 𖽐𖼷𖽷𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏. 18 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏、 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏,𖼊𖽑𖽡𖽝𖾐 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼞𖽘 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏. 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼖𖽙𖾑 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼎𖽙 𖼐𖽜𖾑𖽑 , 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪(𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼋𖽡𖽝𖾏 𖼋𖽪 𖼢𖽻 𖼀𖽱𖾏”). 19 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼏𖽵 𖼟𖽡 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼐𖽤𖾐, 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳、 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏、 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏、 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽑𖽡𖾏 𖼏𖽵 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼐𖽤𖾐. 20 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼏𖽵 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼐𖽤𖾐, 𖼟𖽡 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
22 𖼊𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼡𖽻𖾐 𖼖𖽻𖾏、 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖽐𖼎𖽙 𖼛𖽙𖾏,
𖼋𖽡𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 (“𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 ” 𖼟𖽪𖾏 “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽙”).
23 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼯𖽡𖽙 𖼎𖽻𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 .
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏:“𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏.”
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.”
