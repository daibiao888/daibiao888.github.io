𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑
（𖼋𖽠𖾏 28：1 - 14）
39
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 , 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼐𖽑𖽘𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 . 𖼎𖽺𖾏 𖼑𖽜 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 ， 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽡𖾑 𖽃𖽔𖾏 .
2 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼞𖽪𖾐 𖽐𖼎𖽙 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖼊𖽑𖽡𖽝𖾐 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳（𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 — 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼮𖽵𖾑 𖼚𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 ） . 3 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾐 𖽐𖼊𖽪𖾏 𖼟𖽙𖾏 𖼑𖽡𖽝𖾏 , 𖼷𖽑𖽳𖾏 𖽃𖽔𖾏 𖼺𖽙𖾑 𖼗𖽙𖾏 , 𖽐𖼎𖽙 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 、𖼺𖽙𖾑 𖼅𖽘𖾏, 𖼟𖽙𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼳𖽡𖾑 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 . 4 𖼎𖽺𖾏 𖼑𖽜 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼸𖽙𖾐 𖼘𖽑𖽘𖾐 𖼟𖽪𖾏 𖼚𖽑𖽙𖾏 𖼦𖽡𖾑 𖼁𖽱𖾏 , 𖼺𖽠𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖽃𖽔𖾐 𖼺𖽘𖾐 . 5 𖼸𖽙𖾏 𖼘𖽑𖽘𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽐𖼊𖽙𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼺𖽙𖾑 𖽃𖽔𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼮𖽡𖽝𖾏 , 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖼺𖽠𖾏 𖼦𖽡𖾏 𖼊𖽡𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 , 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼞𖽪𖾐 𖽐𖼎𖽙 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖼊𖽑𖽡𖽝𖾐 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 𖽃𖽔𖾏 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽡𖾑 .
6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽳𖾏 𖼋𖽠𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼄𖽔𖾐 𖼐𖽻𖾐 𖼖𖽡𖽝𖾐 ,𖼯𖽡𖽙𖾏 𖼗𖽙𖾏 𖼘𖽙𖾏 𖼎𖽑𖽹𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼸𖽙𖾏 𖼟𖽱𖾏 𖼞𖽪𖾐 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽳𖾏 𖽐𖼊𖽠𖾑 𖼁𖽘𖾏 , 𖼞𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖽐𖼷𖽷𖾏 𖼎𖽻𖾏 . 7 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼐𖽡𖾑 , 𖼊𖽡𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖽃𖽔𖾐 𖼸𖽙𖾐 𖼘𖽑𖽘𖾐 𖼦𖽡𖾏 𖼁𖽱𖾏 , 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖾏 . 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽡𖾑 .
