𖼞𖽑𖽻𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫
95
𖼋𖽔𖾏 𖼘𖽪！ 𖼀𖽡𖾐 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼙𖽔 𖼨𖽑𖽪𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐．
2 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼐𖽱𖾏,
𖼎𖽑𖽻𖾏 𖼳𖽘𖾐 𖽐𖼞𖽻𖾏 𖼨𖽑𖽪𖾏 𖼙𖽔 𖼐𖽱𖾏．
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐, 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼮𖽵𖾑 𖼳𖽺 .
4 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽙𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼇𖽻𖾑 𖼎𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼁𖽡𖽝 .
5 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼐𖽱𖾏 , 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏,
𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏.
 
6 𖼋𖽔𖾏 𖼘𖽪！ 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏,
𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼀𖽡𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 .
7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼀𖽡𖾐 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼖𖽪𖾑 𖽐𖼎𖽘𖾏 𖽐𖼡𖽜𖾐, 𖼟𖽪𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼄𖽑𖽙𖾏 𖼑𖽔𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼗𖽪𖾏 ,
8 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏, 𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼀𖽔𖾑 ,
𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑.
9 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖼈𖽡𖽝𖾏 𖽐𖼮𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽵𖾑 𖼞𖽪𖾑 .
10 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽐𖼊𖽪𖾐 𖽐𖼎𖽹𖾏 , 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼷𖽑𖽷𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏:
“𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 ,
𖼎𖽺𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑！”
11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼏𖽵 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏！”
