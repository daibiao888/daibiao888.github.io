30
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖽃𖽺𖾐! 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖼐𖽡𖾑．
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑,𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏,𖼦𖽡𖾐 𖼾𖽙 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼞𖽪𖾑 𖼖𖽡𖽾 ,
𖼸𖽙 𖼟𖽪𖾏 𖼾𖽔𖾏 𖼦𖽡𖾐 𖼮𖽳𖾐 𖼾𖽔𖾏.
2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽠𖾑 𖼊𖽠𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼙𖽠𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼞𖽪𖾑 ,
𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼾𖽨𖾏,
𖽐𖼚𖽺𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏, 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼅𖽡 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖽐𖼚𖽺𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼚𖽑𖽪𖾏.
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼐𖽡𖽙𖾐 𖼺𖽙𖾐 𖼣𖽔𖾑 𖼋𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼦𖽔𖾐 𖼐𖽪𖽺𖾏 𖼺𖽷𖾑 ．
5 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼊𖽔𖾐.”
6 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼯𖽦𖾏 𖼑𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐 :
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼮𖽹𖾑 𖼎𖽻𖾏 𖼡𖽔𖾑 𖼯𖽱𖾏 𖼀𖽡𖾏 𖼊𖽪𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼊𖽡𖾑 𖼢𖽻𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼮𖽹𖾑 𖼎𖽻𖾏 𖼘𖽙 𖼊𖽑𖽙𖾐 𖼊𖽡𖾏 𖽐𖼮𖽙𖾑 ,
𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖼟𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐、 𖼊𖽪𖾑 𖼡𖽔𖾐 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐、 𖼀𖽡𖾑 𖼐𖽘𖾐 𖽐𖼚𖽪𖾏、 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼽𖽘𖾏 𖽃𖽡𖾑,
𖼚𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼙𖽠𖾏.
7 𖼣𖽹 𖼮𖽡 𖼑𖽡𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 “𖼖𖽔𖾑 𖼦𖽔𖾐 𖼀𖽳 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙 .”
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏
8 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼅𖽻𖾏,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼸𖽻 𖼞𖽳𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼮𖽡𖽝𖾏,
𖼺𖽻𖾏 𖼎𖽻𖾏 𖽐𖼊𖽠𖾑 𖼮𖽡𖽝𖾏,
𖼟𖽙𖾏 𖼻𖽻𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 𖼡𖽜𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏.
9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 , 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐,
𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏.
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼺𖽪𖾐 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼋𖽘𖾏.”
𖽐𖼎𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼽𖽔𖾑 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾏 𖼊𖽔𖾐,
𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐.
11 𖼅𖽡 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼐𖽡𖾑 , 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖼐𖽡𖾑 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼨𖽑𖽪𖾑.”
12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼨𖽑𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:
“𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽡𖾑 ,
𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖽐𖼎𖽙 𖼖𖽡𖾐 𖼉𖽳𖾐, 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼏𖽔𖾏，
13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼋𖽡𖾏 𖼾𖽔𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 ，
𖼺𖽱𖾏 𖼮𖽑𖽤 𖼟𖽪𖾏 𖼋𖽠 𖼞𖽘𖾐 𖼐𖽔𖾑 𖼀𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑,
𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼊𖽔𖾐 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼢𖽻 𖼙𖽠𖾏.
14 𖼐𖽱𖾏 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏, 𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼑𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽃𖽔 𖼗𖽔𖾏,
𖽃𖽡𖾐 𖽐𖼮𖽷𖾐 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾐 𖼡𖽔𖾑 𖼷𖽑𖽹𖾏 𖼯𖽡𖽙𖾏 𖼗𖽙 𖼉𖽺𖾏 𖽃𖽔𖾑 𖼷𖽙𖾏 𖽐𖼚𖽙 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏,
𖼯𖽡𖽙𖾏 𖼗𖽙 𖼦𖽺𖾏 𖼁𖽘𖾏 𖽃𖽻𖾐 𖽐𖼚𖽙 𖼑𖽡𖾏 𖽃𖽻𖾐.”
15 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼨𖽑𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :

𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑， 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
19 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪! 𖼟𖽡 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏 𖼋𖽘𖾏. 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽙𖾑𖽑 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼟𖽡. 20 𖼮𖽵𖾑 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼳𖽡𖽝𖾐, 𖼄𖽔𖾐 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽃𖽻𖾐 𖼈𖽡𖽝𖾏, 𖼟𖽡 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼚𖽺𖾏 𖼙𖽠𖾏, 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼽𖽔𖾑 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏. 21 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏, 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽙𖾑𖽑 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏, 𖼅𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.” 22 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼅𖽻 𖼡𖽑𖽠𖾑 𖼘𖽙𖾏 𖼎𖽻𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼅𖽻 𖼘𖽙𖾏 𖼎𖽻𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼘𖽡𖾑𖽑 𖽃𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏, 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖽐𖼎𖽙 𖼊𖽺𖾐 𖼁𖽠𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽠𖾏 𖼙𖽠𖾏 𖼦𖽡𖾑！”
