𖼗𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐
23
𖼟𖽡 𖼮𖽔𖾑 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏,
𖼽𖽔𖾑 𖼑𖽔𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏.
2 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽠𖾑 𖼑𖽻𖾏 𖼦𖽻𖾏,
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼮𖽡𖽝𖾏,
3 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽠𖾑 𖼉𖽳 𖼐𖽱𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼮𖽵𖾏 𖼦𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐.
4 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼅𖽔𖾏 𖼊𖽻𖾏,
𖼟𖽡 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼮𖽙𖾏 𖼺𖽘𖾏 𖼊𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽡𖾑 .
5 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖽐𖼮𖽦𖾏 𖼮𖽻𖾑 𖼎𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 𖼖𖽺𖾑?
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼽𖽔𖾑 𖼋𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏, 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼚𖽘𖾑 𖼽𖽘𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼙𖽠𖾏.
 
6 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼚𖽪𖾏 𖼇𖽳𖾏 𖼑𖽡𖾏 𖼉𖽔𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼉𖽳 𖼐𖽱𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼮𖽵𖾏 𖼦𖽘𖾏.
7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ,
𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏,
𖼐𖽱𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 : “ 𖼷𖽑𖽜𖾑 𖼑𖽻𖾏 𖼷𖽑𖽜𖾑 𖼦𖽻𖾏.”
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼊𖽱𖾐 𖼊𖽡𖾑 𖼢𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 .
8 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽻𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼻𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼯𖽢𖾏 𖼙𖽠𖾏.
9 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽻 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼗𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏.
10 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽹𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼳𖽪𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐,
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏.
12 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳.
13 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏,
𖼟𖽡 𖼎𖽑𖽻𖾏 𖼡𖽠𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼈𖽡𖽝𖾏, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽙 𖼋𖽔 .
14 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽠𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏,
𖼎𖽺𖾏 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼦𖽡𖾐 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏.
15 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼙𖽔 .
16 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼗𖽙𖾏 𖼳𖽦𖾑.
17 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼊𖽪𖾐 𖼾𖽔𖾏,
𖼮𖽔 𖼽𖽔𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
18 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽘𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼟𖽡 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽪𖾏 𖼺𖽘𖾏.
 
19 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳,
𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏.
20 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽙𖾑 𖼮𖽡𖽠𖾑 𖽐𖼮𖽙𖾑 𖽐𖼡𖽹𖾏；
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 .
21 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽙𖾑 𖼮𖽡𖽠𖾑, 𖽐𖼮𖽙𖾑 𖼑𖽻𖾏 𖼦𖽻𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼀𖽱𖾏 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 .
 
22 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼟𖽡, 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼗𖽻𖾏 𖼋𖽘𖾏；
𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼗𖽻𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼐𖽱𖾏 .
23 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼟𖽡𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼅𖽔𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽺；
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳、𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼉𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼐𖽱𖾏 𖼙𖽔 𖼚𖽙𖾏.
25 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼟𖽡 𖽃𖽔𖾐 𖼈𖽹𖾐 𖼙𖽔 𖼚𖽙𖾏.
𖼟𖽙𖾏 𖼖𖽜𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
26 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 ；
𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼙𖽔 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 .
27 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽙𖾐,
𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼘𖽑𖽻𖾑 .
28 𖼐𖽱𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖼺𖽱𖾏 𖼸𖽜 ,
𖼐𖽱𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼮𖽑𖽨𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
 
29 𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔? 𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏 ?
𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏? 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏 ?
𖼡𖽔 𖼋𖽱𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐? 𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼖𖽡𖽝𖾐 ?
30 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼇𖽔𖾑 ,
𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏 𖽃𖽡𖾑.
31-32 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽠𖾑 𖼮𖽡𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐,
𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 ,
𖼦𖽻𖾏 𖼟𖽡𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼈𖽡𖽝𖾏,
𖽃𖽔𖾑 𖼡𖽜𖾐 𖼋𖽙𖾏 𖼟𖽡 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐, 𖼮𖽑𖽡𖽙𖾐 𖼟𖽡 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏.
33 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑.
𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐.
34 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽘𖾐 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖽐𖼞𖽙𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼮𖽘𖾏 𖽐𖼊𖽻𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏.
35 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 : “𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 , 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼄𖽙𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽠𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 ；
𖼞𖽪𖾑 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼯𖽷𖾏 𖼨𖽑𖽪𖾑,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼮𖽡𖽠𖾑 𖼦𖽻𖾏 𖼺𖽷𖾐.”
