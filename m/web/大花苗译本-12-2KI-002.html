2
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖼷𖽹𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼮𖽡𖾏 𖽐𖼊𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼎𖽑𖽻𖾏 𖼮𖽡 𖼮𖽢 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏. 2 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏, 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽙𖾏．” 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏, 𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼳𖽦𖾑 𖼏𖽵 𖼗𖽪𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼟𖽡.” 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏. 3 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼇𖽳𖾑 𖼟𖽡, 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼀𖽻𖾐, 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐.” 4 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑, 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽙𖾏.” 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏, 𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼳𖽦𖾑 𖼏𖽵 𖼗𖽪𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼟𖽡.” 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑. 5 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏, 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼇𖽳𖾑 𖼟𖽡, 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼀𖽻𖾐, 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐.”
6 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏, 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽙𖾏.” 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏, 𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼳𖽦𖾑 𖼏𖽵 𖼗𖽪𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼟𖽡.” 7 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼅𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼳𖽠𖾑 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼺𖽘𖾑 𖼀𖽡 𖼋𖽡𖾏, 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏. 8 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼯𖽡𖽙𖾏 𖼗𖽙, 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐, 𖽃𖽻𖾐 𖼎𖽺𖾏 𖼇𖽺𖾐 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖼋𖽡𖾏 𖼙𖽠𖾏. 9 𖼋𖽡𖾏 𖼙𖽠𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼊𖽻𖾏 𖼷𖽹𖾏 𖼇𖽳𖾑 𖼟𖽡, 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼟𖽡 𖼮𖽔𖾑 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑.” 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼟𖽡 𖼎𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼞𖽪𖾑.” 10 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽐𖼊𖽜 𖽃𖽔𖾏 𖼸𖽙 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼈𖽡𖽝𖾏, 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼷𖽹𖾏 𖼇𖽳𖾑 𖼟𖽡 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏, 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏.”
𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏

19 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑𖽐𖼚𖽙 𖼙𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽻𖾐 𖽐𖼚𖽪𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼦𖽡𖾐 𖼳𖽦𖾑 𖼺𖽷𖾐 𖼷𖽹𖾏 𖼉𖽜 𖼀𖽻𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏.” 20 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖽐𖼊𖽡𖾏 𖽐𖼮𖽳𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏. 21 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼸𖽙 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼞𖽙𖾏 𖽃𖽻𖾐, 𖼄𖽔𖾐 𖽐𖼮𖽳𖾑 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: “𖼞𖽪𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽡𖾏 𖽃𖽻𖾐 𖼐𖽡𖾑, 𖼎𖽑𖽻𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼡𖽻𖾐 𖼋𖽘.” 22 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼁𖽘𖾏 𖽃𖽻𖾐 𖽃𖽡𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏. 23 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑, 𖼅𖽔𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼙𖽻 𖼋𖽠𖾏 𖼋𖽔𖾏, 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼚𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽡𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 ! 𖼷𖽷𖾑 𖼚𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽡𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 !” 24 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼎𖽺𖾏 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 𖽃𖽔𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼚𖽹𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏, 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼺𖽘𖾏. 25 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼎𖽑𖽻𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼄𖽡 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼄𖽡 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼸𖽙 𖼚𖽔𖾏𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏.
