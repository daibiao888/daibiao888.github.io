28
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽠𖾑 𖼈𖽡𖽝𖾏 𖼏𖽘 𖼀𖽜𖾑 ；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼮𖽙𖾏 𖼚𖽔𖾐 𖼘𖽑𖽙𖾐 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏.
2 𖼞𖽪𖽙 𖼑𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏, 𖼎𖽺𖾏 𖼘𖽑𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐；
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼟𖽳 , 𖼞𖽪𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 .
3 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼸𖽹𖾏 𖼯𖽵𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼊𖽻𖾏,
𖼺𖽱𖾏 𖼑𖽘 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽱𖾑𖽑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏.
4 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼎𖽺𖾏 𖼷𖽻𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖼮𖽔 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏.
6 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾐 𖼄𖽔𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
7 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽙𖾑 𖼑𖽻𖾏 𖽐𖼮𖽙𖾏 𖼦𖽻𖾏 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽙𖾏 𖽐𖼮𖽷𖾏 𖼳𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏,
𖼟𖽪𖾏 𖼑𖽜 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏.
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑,
𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽪𖾐.
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼙𖽠𖾏,
𖼮𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼸𖽠𖾏 𖼑𖽔𖾏 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 .
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼳𖽠𖾏, 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏.
13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽜𖾏 𖼾𖽔𖾏 𖼇𖽳𖾑 𖼾𖽔𖾏 𖼺𖽘𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐.
14 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼼𖽔𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
15 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽹𖾏 𖼯𖽵𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏,
𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼡𖽔𖾏 𖽐𖼎𖽻𖾏 𖼙𖽠𖾏, 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼚𖽹𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼡𖽜𖾑 𖼑𖽻𖾏.
16 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑, 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 .
17 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼏𖽘 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 .
18 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼊𖽻𖾏 𖼢𖽻 .
19 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏.
20 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐.
21 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽑𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼻𖽻𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼊𖽔𖾐 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏.
22 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼇𖽳𖾏， 𖽐𖼮𖽑𖽠𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏,
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
23 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽔 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑 .
24 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝 𖼾𖽨𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏 : “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼾𖽔𖾏.”
𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼸𖽜 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐.
25 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼉𖽳 𖼗𖽔 𖼁𖽡𖽝 , 𖽐𖼷𖽑𖽙𖾑 𖼑𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼊𖽻𖾏.
26 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽙 , 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽔𖾏 𖼑𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏.
27 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼺𖽘𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽡𖽻𖾐 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏.
28 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
