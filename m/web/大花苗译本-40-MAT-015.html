𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼸𖽪𖾏
（𖼞𖽑𖽙𖾐 7：1-23）
15
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏，𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼐𖽑𖽡𖾏： 2 “𖼟𖽡 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼸𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽺𖾑？𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖽐𖼷𖽔𖾑 𖼋𖽡𖾏 ． ” 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼸𖽪𖾏，𖼦𖽡𖾐 𖼻𖽙𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼖𖽺𖾑？ 4 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏：‘ 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 ’， 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：‘ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔．’ 5 𖼅𖽡 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼐𖽑𖽡𖾏：‘ 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼎𖽙 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏：“𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼮𖽵𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏．” 6 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏．’ 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼅𖽡 𖼸𖽪𖾏， 𖼗𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼺𖽘𖾏． 7 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏，𖼅𖽡 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏． 8 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：
“𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼞𖽪𖾑，
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼡𖽔𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝．
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏，𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐，
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙．”
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐
10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏． 11 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．” 12 𖼊𖽑𖽻𖾏 𖼐𖽡𖾑，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼦𖽡𖾐 𖼙𖽔 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼀𖽻𖾐？” 13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 ： “𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼯𖽡𖽙，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽙𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼙𖽠𖾏． 14 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏！𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼨𖽑𖽳𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼨𖽑𖽳𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐，𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽑𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼁𖽻𖾏 𖼡𖽑𖽙𖾑．” 15 𖼀𖽡𖾐 𖼊𖽳 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼷𖽑𖽜𖾑 𖼄𖽔𖾐 𖼗𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑 𖼚𖽠𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐．” 16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼊𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼺𖽷𖾐 𖼘𖽪𖾏？ 17 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏，𖼟𖽪𖾏 𖼁𖽺𖾏 𖼁𖽻 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼁𖽻𖾏 𖼡𖽑𖽙𖾑 𖼮𖽑𖽡𖾑 𖼙𖽠𖾏 𖼘𖽪𖾏？ 18 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 ， 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐，𖼎𖽺𖾏 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． 19 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏、𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏、𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 、𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏、𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼺𖽘𖾐 𖽃𖽔𖾏 𖼸𖽜、𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑、𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏． 20 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼷𖽔𖾑 𖼋𖽡𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏，𖼦𖽡𖾐 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．”
𖼮𖽢𖾑 𖼐𖽔 𖽃𖽔 𖼁𖽙𖾏 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏
（𖼞𖽑𖽙𖾐 7：24-30）
21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏，𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙、𖼳𖽡𖾑 𖼊𖽪𖽝𖾏𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖽐𖼚𖽙． 22 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽢𖾑 𖼐𖽔 𖽃𖽔 𖼁𖽙𖾏， 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼎𖽑𖽻𖾏 𖼼𖽙 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪 ！ 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏，𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏，𖼐𖽱𖾏 𖽃𖽦𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ． ” 23 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐．𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼊𖽜𖾏 𖼐𖽑𖽡𖾏：“𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏，𖼮𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼙𖽠𖾏．” 24 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼗𖽙 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼙𖽠𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽔𖾐．” 25 𖼈𖽡𖽝𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼋𖽔𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖽐𖼀𖽘𖾏 𖼞𖽪𖾑．” 26 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼄𖽔𖾐 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼡𖽜𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼣𖽔𖾏 𖼚𖽡𖾑 𖼑𖽻𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏．” 27 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪 ！ 𖼾𖽙 𖼯𖽡𖽙𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼣𖽔𖾏 𖼚𖽡𖾑 𖼖𖽔𖾏 𖼑𖽻𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼡𖽘𖾐 𖼞𖽡 𖼏𖽻𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼗𖽙 𖼯𖽡𖽙𖾏 ． ” 28 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖽃𖽔 𖼁𖽙𖾏， 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽡𖽝， 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑，𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽘𖾏 ． ” 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑， 𖼐𖽱𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼻𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏．
29 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑， 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑， 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐，𖼎𖽺𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 ． 30 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑，𖼨𖽑𖽳𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐、 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾐 𖼊𖽠𖾏 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐、 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽜 𖼅𖽜 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐，𖼨𖽑𖽳𖾏 𖽐𖼮𖽡𖽻 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 31 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽜 𖼅𖽜 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾐 𖼊𖽠𖾏 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏，𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼸𖽠𖾏 𖼅𖽻， 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼸𖽠𖾏 𖼁𖽙𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐 𖼮𖽡𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖼮𖽻𖾏
（𖼞𖽑𖽙𖾐 8：1-10）
32 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏．𖼞𖽪𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽜𖾏 𖼮𖽑𖽹𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏，𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖽃𖽔𖾐 𖼘𖽑𖽪𖾐 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖽐𖼮𖽡𖾏 𖼉𖽪𖾏 𖼢𖽻．” 33 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼐𖽡𖾑，𖼀𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼊𖽻𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼷𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼑𖽻𖾏 𖼮𖽻𖾏 𖼘𖽪？” 34 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼳𖽡𖽝𖾐？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽔𖾏 𖼳𖽤𖾏 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽐𖼀𖽳𖾏 𖼊𖽑𖽡𖽝𖾐 ． ” 35 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼼𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐， 36 𖼞𖽑𖽡𖽠𖾏 𖼳𖽤𖾏 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖽐𖼎𖽙 𖽐𖼀𖽳𖾏，𖼮𖽑𖽔𖾏 𖼙𖽔 𖼋𖽘𖾏，𖽐𖼀𖽹𖾏 𖼮𖽡𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 ， 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏． 37 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼮𖽻𖾏 𖼋𖽘𖾏， 𖼳𖽻𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼟𖽪𖾏 𖼳𖽙𖾏，𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼊𖽠𖾐． 38 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏，𖼋𖽠𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼺𖽘𖾏，𖼅𖽔𖾏 𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐． 39 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏，𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏𖼄𖽔𖾐 𖼮𖽢𖾑 𖼊𖽗𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏．
