𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏.
27
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼞𖽡𖾑, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏,
𖼞𖽪𖾑 𖽐𖼮𖽑𖽹𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑?
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 ( “𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “ 𖼛𖽘𖾏 𖼼𖽙𖾏”)
𖼞𖽪𖾑 𖽐𖼮𖽑𖽹𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑？
2 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑,
𖼖𖽔𖾑 𖼋𖽔𖾏 𖼑𖽻𖾏 𖼞𖽪𖾑 𖽐𖼡𖽹𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖽐𖼀𖽑𖽻𖾏 𖽐𖼀𖽙 𖼢𖽻 𖼙𖽠𖾏.
 
3 𖼅𖽔𖾏 𖼏𖽙 𖼮𖽪𖾐 𖼾𖽡 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼞𖽪𖾑 𖼈𖽡𖽝𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏,
𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖽐𖼮𖽷𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼞𖽪𖾑 𖼈𖽡𖽝𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
4 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏, 𖼞𖽪𖾑 𖼙𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼞𖽪𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼺𖽷𖾐,
𖼞𖽪𖾑 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽜.
 
5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑, 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼈𖽺𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖽐𖼚𖽙 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼮𖽡𖽝𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼮𖽑𖽳𖾑 𖼺𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑.
6 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖽐𖼚𖽙 𖼙𖽔 𖼚𖽙𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
 
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼞𖽪𖾑,
𖼎𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑.
8 𖼟𖽡 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 𖼚𖽑𖽪𖾏.”
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡 𖼚𖽑𖽪𖾏 .”
9 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼀𖽙 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼖𖽠𖾑 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼺𖽘𖾏,
𖼟𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑．
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼘𖽪！𖼦𖽡𖾐 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏.
10 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼈𖽡𖽝𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼳𖽻𖾏 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏.
 
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑,
𖼑𖽜 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏.
12 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼊𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼞𖽪𖾑.
13 𖼞𖽪𖾑 𖽐𖼮𖽜, 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐,
𖼽𖽔𖾑 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏.
14 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫! 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼚𖽔𖾐! 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐!
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
