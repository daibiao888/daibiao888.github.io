𖼽𖽨 𖼊𖽔𖾏 𖼇𖽳𖾑 𖼮𖽵𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑
8
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼞𖽑𖽡𖽠𖾏 𖼊𖽔𖾑 𖽐𖼚𖽘𖾏 𖼟𖽪𖾏 𖼞𖽳𖾏 𖽃𖽔𖾐 𖼅𖽔 𖽐𖼊𖽠𖾑(𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖽃𖽔𖾑 𖼅𖽔 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼯𖽡𖽙𖾏”)， 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼄𖽔𖾐 𖼦𖽳 𖽃𖽳𖾐 ·𖼳𖽔𖾑 𖼖𖽔𖾑 𖼖𖽳 · 𖼦𖽔𖾐 𖼳𖽷𖾑 ·𖼀𖽔𖾏 𖼺𖽷𖾑 (𖼟𖽪𖾏 “𖼇𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏， 𖼇𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽔𖾑 𖼋𖽘𖾏 ”). 2 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏, 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽂𖽪𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏.”
3 𖼞𖽪𖾑 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼐𖽤𖾐 (“𖼐𖽤𖾐”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“ 𖽃𖽔 𖼁𖽙𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 ”) 𖼀𖽱𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼊𖽪𖾐. 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖼦𖽳 𖽃𖽳𖾐 ·𖼳𖽔𖾑 𖼖𖽔𖾑 𖼖𖽳 · 𖼦𖽔𖾐 𖼳𖽷𖾑 ·𖼀𖽔𖾏 𖼺𖽷𖾑 . 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼸𖽠𖾏 𖼨𖽑𖽪𖾏 𖼑𖽡𖽝𖾏 𖼨𖽑𖽪𖾏 𖼉𖽺𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽑𖽹𖾏 𖼙𖽠𖾏.”
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 6 “𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽡𖾑 𖼘𖽙 𖼽𖽔𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖽐𖼚𖽱𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼙𖽔𖼖𖽡𖾏 𖼳𖽱 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐， 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼮𖽵𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙 𖽐𖼊𖽱𖾑𖽑 𖼗𖽙𖾏, 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼚𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽪𖾑 𖼮𖽑𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽱𖾏 𖽃𖽻𖾐, 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏 𖼮𖽑𖽡𖽻𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪. 8 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼊𖽱𖾑𖽑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 , 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐. 𖼽𖽡𖾑 𖼄𖽔𖾐 𖼐𖽪𖽺𖾏 𖼖𖽡𖾏 𖼘𖽪！ 𖼐𖽱𖾏 𖽐𖼊𖽑𖽔𖾑 𖽃𖽔𖾑 𖼊𖽡𖾏, 𖽐𖼀𖽙 𖼀𖽪𖾑 𖼟𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.
𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏
9 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪! 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼋𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼈𖽡𖽝𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼘𖽪！ 𖼽𖽔𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻．
𖼯𖽔𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼮𖽡𖽝𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼅𖽡 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼮𖽡𖽝𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
10 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑, 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏,
𖼯𖽔𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼏𖽵 , 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽡𖽝𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐.

16 𖼟𖽡 𖼽𖽔𖾑 𖼦𖽡𖾏 𖼚𖽙𖾑 𖽐𖼊𖽠𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼮𖽡𖽝𖾏, 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼘𖽙𖾏 𖼮𖽡𖽝𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏. 17 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏. 18 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼋𖽔𖾏 , 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑.
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
19 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼮𖽤𖾐 𖼳𖽘𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼖𖽪𖾐 , 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼮𖽑𖽡 𖽃𖽡𖾑.” 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼑𖽪 𖼾𖽨𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼽𖽔𖾑 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼖𖽺𖾑?” 20 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽐𖼎𖽙 𖽐𖼊𖽠𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽔𖾏 𖼊𖽪𖾑 𖽐𖼮𖽔𖾏 . 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼁𖽙𖾏 𖼞𖽡𖾑 .
21 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 , 𖼟𖽱𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 , 𖼟𖽱𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏. 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽱𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼾𖽨𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼾𖽨𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 22 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 , 𖼚𖽻𖾑 𖼗𖽡𖾏 𖼇𖽻𖾏 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐． 𖼑𖽔𖾏 𖼘𖽪 ! 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼚𖽪𖾐 𖼖𖽡𖾐 𖽐𖼮𖽡𖽝𖾐 𖼑𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼚𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼚𖽪𖾐 𖼖𖽡𖾐 𖽐𖼮𖽡𖽝𖾐 𖼑𖽡𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼙𖽠𖾏.
