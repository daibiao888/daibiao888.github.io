𖼽𖽙 𖼺𖽳 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼇𖽔 𖼘𖽻𖾐 , 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼖𖽠𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽨𖾏、𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏, 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽔𖾏 𖼋𖽘, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼞𖽙𖾑 𖼳𖽔𖾑 𖼊𖽡𖾐.” 2 𖼽𖽙 𖼺𖽳 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽑𖽦𖾑 𖼋𖽠𖾏 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽙, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐. 3 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽪 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏, 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.” 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏:“ 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼡𖽑𖽜𖾑 𖼐𖽱𖾏 𖽐𖼀𖽙𖾏 𖼯𖽦𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼋𖽔𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼡𖽑𖽜𖾑 𖼐𖽡𖽙𖾐 𖼞𖽙𖾑 𖼳𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.” 5 𖼇𖽔 𖼘𖽻𖾐 𖽐𖼎𖽙 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏:“𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼋𖽔𖾏 𖼸𖽙 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼋𖽘𖾏， 6 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,𖼯𖽪𖾏 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏； 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼚𖽔𖾏 𖼞𖽙𖾑 𖼳𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 𖼟𖽡 𖼮𖽔𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽜, 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼞𖽳𖾑 𖼞𖽪𖾑 𖼯𖽦𖾏.”
𖼽𖽙 𖼺𖽳 𖼨𖽑𖽳𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐
7 𖼽𖽙 𖼺𖽳 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼽𖽔𖾐 𖼞𖽙 𖼎𖽺𖾏 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐. 8 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽪 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏:“𖼟𖽡 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼳𖽡𖽻𖾏?” 9 𖼽𖽔𖾐 𖼞𖽙 𖼊𖽡𖾐 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾐 𖼊𖽡𖾐 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖽃𖽺𖾐 𖽃𖽦𖾐 𖽃𖽺𖾐 𖼯𖽠, 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏.” 10 𖼽𖽔𖾐 𖼞𖽙 𖼎𖽺𖾏 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐 , 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏.
11 𖼽𖽙 𖼺𖽳 𖼸𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼄𖽔𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖼟𖽪𖾏 𖼖𖽔 𖼺𖽳 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐, 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 12 𖼽𖽙 𖼺𖽳 𖼎𖽑𖽻𖾏 𖼡𖽻𖾐 𖼡𖽜𖾏 𖼊𖽪𖾑 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼮𖽡𖽝𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽃𖽡𖾐 𖽐𖼞𖽔𖾐 𖼾𖽡𖾏, 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏.
𖼽𖽙 𖼺𖽳 𖼎𖽑𖽻𖾏 𖼡𖽻𖾐 𖼘𖽑𖽪𖾑 𖼯𖽦𖾏
13 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐, 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼋𖽘, 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼎𖽺𖾏 𖼮𖽑𖽹𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝. 14 𖼽𖽙 𖼺𖽳 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐, 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐, 𖼽𖽙 𖼺𖽳 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼑𖽡𖽝𖾏 𖼚𖽔𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼟𖽻. 15 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼐𖽔 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼞𖽑𖽔𖾏 𖼳𖽜𖾐 𖼋𖽘 𖼋𖽘𖾏, 𖼣𖽹 𖼮𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼽𖽙 𖼺𖽳, 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼞𖽑𖽔𖾏 𖼯𖽡𖽙𖾏 𖼋𖽘 𖼋𖽘𖾏, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼡𖽻𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼘𖽪?” 16 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “ 𖼅𖽡 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼮𖽔𖾑 𖼯𖽡𖽙𖾏 𖼋𖽘 𖼋𖽘𖾏, 𖼅𖽡 𖼄𖽔𖾐 𖼯𖽦𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼑𖽜 𖼅𖽡 𖼯𖽦𖾏 𖼄𖽔𖾐 𖼡𖽻𖾐 𖼎𖽻𖾏 𖼅𖽡.” 17 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼖𖽠𖾑 𖼯𖽦𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼺𖽳 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼽𖽙 𖼺𖽳 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾐 𖼘𖽑𖽪𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜𖾏、𖼾𖽘𖾏 𖼮𖽑𖽷𖾏、𖼑𖽨𖾏、𖼊𖽑𖽡𖽝𖾐 𖼑𖽜𖾏 𖼖𖽔𖾑. 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼯𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼳𖽦𖾑, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼯𖽦𖾏 𖼅𖽻𖾏 𖼘𖽑𖽪𖾑 𖼡𖽻𖾐.
𖼎𖽑𖽻𖾏 𖼡𖽻𖾐 𖼅𖽔𖾏 𖼘𖽑𖽪𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏
18 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏, 𖼳𖽡𖽻𖾏 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼽𖽙 𖼺𖽳, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼦𖽡𖾐 𖼮𖽑𖽵𖾑 𖼀𖽡𖾐 𖼮𖽵𖾑, 𖼀𖽡𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼞𖽑𖽔𖾏 𖼯𖽡𖽙𖾏 𖼋𖽘 𖼋𖽘𖾏, 𖼯𖽦𖾏 𖼞𖽑𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏. 19 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼯𖽪𖾏 𖼁𖽙𖾏 𖼀𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔, 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼘𖽪? 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼡𖽻𖾐 𖼅𖽔𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐, 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼇𖽔 𖼘𖽻𖾐 𖼋𖽠𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼮𖽘𖾑 𖼡𖽻𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽔, 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏.”
20 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽙 𖼺𖽳 𖼑𖽜 𖼇𖽔 𖼘𖽻𖾐 𖼅𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐, 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏, 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼋𖽺 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼺𖽘𖾏, 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼁𖽡𖽝 𖼙𖽠𖾏. 21 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼽𖽙 𖼺𖽳 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼺𖽘𖾑 𖼐𖽡𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼺𖽘𖾑 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼙𖽻. 22 𖼯𖽔 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼊𖽡𖾐, 𖼽𖽙 𖼺𖽳 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽡𖽝𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼊𖽪𖾑 𖼮𖽡𖽝𖾏. 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼋𖽺 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐.
23 𖼽𖽙 𖼺𖽳 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼑𖽜 𖼇𖽔 𖼘𖽻𖾐 𖼅𖽔𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼅𖽡 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐, 𖼑𖽔𖾏 𖼘𖽪! 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼡𖽻𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡, 𖼅𖽡 𖼽𖽔𖾑 𖼯𖽡𖽙 𖼡𖽻𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐. 24 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼡𖽻𖾐, 𖼀𖽜𖾐 𖽐𖼀𖽠 𖼅𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼎𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐 , 𖼚𖽻𖾐 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼅𖽡 𖼮𖽘𖾑 𖼡𖽻𖾐, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼅𖽡 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏.” 25 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑.” 26 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽙 𖼺𖽳 𖼑𖽜 𖼣𖽹 𖼮𖽡 𖽐𖼡𖽜𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏, 𖼀𖽜𖾐 𖽐𖼀𖽠 𖼇𖽔 𖼖𖽻𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼺𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 .
𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏
27 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼑𖽡𖾏 𖼞𖽙𖾑 𖼳𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼮𖽷𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼘𖽘𖾑𖽑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝. 28 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼟𖽻𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏, 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏. 29 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼇𖽺𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼅𖽳𖾏, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼯𖽡 𖼀𖽔𖾐 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵, 𖼎𖽑𖽻𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽱𖾏 𖼞𖽪𖾑 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡. 30 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡, 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏.” 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼟𖽡 𖼗𖽪𖾏 𖽃𖽔𖾏.” 31 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼏𖽵 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑.” 𖼽𖽙 𖼺𖽳 𖼎𖽺𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽡𖽙𖾐 𖼸𖽱𖾏 𖼸𖽘𖾏 (“𖼐𖽡𖽙𖾐 𖼸𖽱𖾏 𖼸𖽘𖾏 ”𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “𖽐𖼚𖽜 𖼡𖽠𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑 ”) 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
