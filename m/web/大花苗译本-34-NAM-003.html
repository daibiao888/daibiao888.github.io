𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
3
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏! 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖼐𖽡𖾑 ,
𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽐𖼎𖽙 𖽐𖼚𖽪𖾏 𖼸𖽳 ,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏.
2 𖽐𖼚𖽠𖾏 𖼑𖽡𖾏 𖼳𖽘𖾐 𖼳𖽺𖾑 𖼳𖽺𖾑 , 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼀𖽡𖾑 𖼘𖽙𖾐 𖼟𖽡𖾏 𖼨𖽑𖽻 𖼨𖽑𖽻 ,
𖼑𖽜𖾏 𖼚𖽠 𖼚𖽠 , 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼏𖽘 𖽐𖼊𖽑𖽱 𖽐𖼊𖽑𖽱 .
3 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼦𖽡𖾏 𖼦𖽔𖾏 𖽐𖼮𖽵 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼮𖽦𖾑𖽑,
𖼄𖽑𖽪𖾑 𖼟𖽡𖾏 𖼮𖽡𖾏 𖼙𖽪, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼊𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐,
𖼡𖽻𖾑 𖼋𖽔 𖼷𖽱𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼛𖽻𖾏, 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼾𖽱 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼡𖽻𖾑 𖼋𖽔 𖼢𖽻 𖼙𖽠𖾏.
4 𖼑𖽜 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖽃𖽔𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾏 𖼐𖽜𖾐 𖼸𖽷𖾏 𖼅𖽪𖾏,
𖼎𖽑𖽻𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 ,
𖼎𖽑𖽻𖾏 𖼐𖽜𖾐 𖼸𖽷𖾏 𖼅𖽪𖾏 𖼮𖽑𖽵𖾑 𖼚𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 ( “ 𖼮𖽑𖽵𖾑 𖼚𖽡𖾐 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼋𖽺”).
 
5 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼑𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼀𖽑𖽔𖾐 𖼟𖽡 𖼊𖽡𖽝𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏,
𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏,
𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼺𖽱𖾏 𖼚𖽘𖾐.
6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖼳𖽪𖾐 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼟𖽡, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏.
7 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼏𖽘 𖼀𖽜𖾑 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼟𖽡, 𖼐𖽑𖽡𖾏:
𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏！ 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼑𖽜 𖼟𖽡 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼘𖽪?
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼟𖽡 𖼖𖽺𖾑?”
 
8 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼋𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽙 𖼽𖽔𖾏 𖼄𖽞 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼐𖽙 𖼽𖽔𖾏 𖼄𖽞 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 ,
𖼟𖽻𖾏 𖼯𖽪𖾏 𖼅𖽔𖾏 𖽃𖽻𖾐,
𖼳𖽡𖽠𖾑 (“𖼳𖽡𖽠𖾑” 𖼟𖽪𖾏 “𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏” ) 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏,
𖼎𖽺𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 .
9 𖼞𖽪𖾐 𖼳𖽷 𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽘 ,
𖼇𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼖𖽪𖾏 𖼀𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏.
10 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼀𖽜𖾑 𖼙𖽠𖾏, 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏,
𖼐𖽱𖾏 𖼣𖽔𖾑 𖼾𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼙𖽻 𖼷𖽑𖽔𖾏 𖼯𖽢𖾏 𖽃𖽔𖾏 𖼟𖽙𖾏 𖼟𖽡𖾏 𖼊𖽻𖾏 𖽐𖼮𖽠𖾏 𖼟𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏．
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 ,
𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼮𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖼊𖽪𖾏 𖼊𖽻𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽻𖾏 𖼡𖽑𖽹𖾐 𖼮𖽡𖽝𖾏.
 
11 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼢𖽻 , 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏,
𖼟𖽡 𖼎𖽺𖾏 𖼑𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼐𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔.
12 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼳𖽦𖾑,
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏, 𖼽𖽔𖾑 𖼉𖽜 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼑𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
13 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏,
𖼟𖽡 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼖𖽪𖾑 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻𖾏 𖼑𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼟𖽡 𖼊𖽪𖾑 𖼎𖽪𖾑 𖽃𖽔 𖼛𖽻𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏.
 
14 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 ,
𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 ,
𖼟𖽡 𖼽𖽔𖾑 𖽐𖼮𖽨𖾏 𖽃𖽔 𖼗𖽔𖾏 ,
𖼊𖽔𖾑 𖼖𖽪𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖼑𖽡 𖼋𖽺𖾏 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼞𖽑𖽙𖾏.
15 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼽𖽔𖾑 𖼷𖽑𖽷𖾑 𖼖𖽻𖾐 𖼟𖽡 ,
𖽃𖽔𖾑 𖼏𖽦𖾏 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼟𖽡 ,
𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼟𖽡 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼷𖽷 𖼟𖽻𖾏.
 
𖼯𖽔𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼷𖽷 𖼟𖽻𖾏,
𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼷𖽷 .
16 𖼟𖽡 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏, 𖼮𖽡𖽻𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐,
𖼈𖽡𖽝𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼷𖽷 𖼟𖽻𖾏 𖼑𖽻𖾏 𖼋𖽘 𖼎𖽺𖾏 𖼽𖽘𖾏 𖼙𖽠𖾏.
