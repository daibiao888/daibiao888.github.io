24
“ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼊𖽡𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 ,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼘𖽪 ?
2 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽹𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 , 𖼦𖽔𖾏 𖽐𖼀𖽙𖾏 𖼯𖽦𖾏 𖼗𖽙𖾏 𖼾𖽨 .
3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽤𖾐 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖼊𖽪𖾑 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼙𖽠𖾏 ,
𖽃𖽔𖾏 𖼊𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼑𖽨𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 .
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼇𖽳𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖼺𖽘𖾏 ,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼙𖽠𖾏 .
5 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼐𖽡𖾑 , 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼋𖽠𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 , 𖼘𖽑𖽻𖾐 𖽐𖼡𖽔𖾏 𖽐𖼎𖽑𖽦𖾑 𖼡𖽜𖾑 𖼑𖽻𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼟𖽙𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼊𖽻𖾏 𖼑𖽻𖾏 .
6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼡𖽻𖾐 𖼖𖽻𖾏 ,
𖼚𖽡𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 .
7 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 ,
𖽐𖼊𖽪𖾏 𖼐𖽙𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 .
8 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽻𖾏 𖼑𖽘 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼞𖽑𖽔𖾏 𖽐𖼊𖽪𖾐 𖼺𖽘𖾏 ,
𖼑𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼎𖽺𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 .
9 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐 𖼦𖽔𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 ,
𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼦𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 .
10 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 , 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼳𖽔𖾏 𖼯𖽡,
𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼮𖽑𖽳𖾑 𖼡𖽔𖾏 𖼷𖽻𖾏 𖼡𖽻𖾐 .
11 𖼐𖽡𖽙𖾐 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼟𖽻𖾏 𖼮𖽑𖽤 𖽐𖼚𖽙 𖼸𖽹𖾏 𖼛𖽙𖾏 , 𖼯𖽪𖾏 𖼮𖽡𖽠𖾑 ( 𖼯𖽪𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼞𖽔𖾏 𖼞𖽔𖾏 𖽃𖽔𖾏 𖼮𖽡𖽠𖾑 ) ,
𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼎𖽺𖾏 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 .
12 𖼐𖽡𖽙𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽃𖽡𖾑 𖽐𖼚𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 ,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼡𖽔 𖼏𖽔 .
13 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏 ,
𖼦𖽡𖾐 𖼀𖽻𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏,
𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏 .
14 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 ,
𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 , 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼸𖽜 .
15 𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼋𖽙𖾏 𖼞𖽡𖾑 𖼉𖽘𖾏 𖼞𖽡𖾏 𖼷𖽻𖾏 ,
𖼐𖽑𖽡𖾏 : 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼸𖽠𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 ,
𖼎𖽺𖾏 𖽐𖼀𖽙 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 .
16 𖼸𖽜 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼮𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖽐𖼞𖽔𖾏 ,
𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼚𖽺𖾏 𖽐𖼎𖽙𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 .
17 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼺𖽱𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼑𖽡𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 .
 
18 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖽐𖼊𖽘𖾐 𖼦𖽡𖾐 𖼇𖽔𖾑 𖽃𖽻𖾐 𖼇𖽺𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼊𖽘𖾐 𖼙𖽠𖾏 ,
𖽐𖼀𖽠 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼊𖽙𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 .
19 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖽐𖼎𖽙 𖼳𖽙𖾑 𖼉𖽠𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖽃𖽻𖾐 𖽐𖼎𖽙 𖽐𖼀𖽪𖾏 ,
𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 .
20 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼟𖽪𖾏 𖼾𖽨 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 ,
𖼞𖽘𖾑 𖼞𖽘𖾏 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼐𖽱𖾏 , 𖼑𖽻𖾏 𖼡𖽘𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼽𖽔𖾑 𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 .
