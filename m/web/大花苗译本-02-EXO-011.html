𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐

𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼮𖽙𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐 𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 , 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 . 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼅𖽻𖾏 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼖𖽠𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼞𖽑𖽔𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏. 2 𖼟𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔 𖼾𖽠 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙 𖼽𖽔𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼞𖽪𖾐、𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 .” 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏 . 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔, 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 .
4 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : ‘ 𖼖𖽔𖾑 𖼸𖽙 𖼋𖽘 𖼄𖽑𖽙𖾏 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 . 5 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼎𖽑𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 , 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼢𖽪𖾏 𖼡𖽜𖾑 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 , 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼨𖽑𖽳𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 . 6 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 , 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼗𖽙𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 . 7 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼊𖽪𖾑 𖼚𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽠𖾏 𖼺𖽘𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼯𖽦𖾏 . 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽺𖾐 𖼡𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏.’ 8 𖼟𖽡 𖼸𖽻𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 , 𖼐𖽑𖽡𖾏 : ‘ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼦𖽡𖾑．’ 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 .” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖼮𖽑𖽡𖾏 𖽐𖼷𖽑𖽔𖾏 𖼇𖽳𖾑 𖼇𖽔 𖼘𖽻𖾐 𖼺𖽘𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 . 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼇𖽔 𖼘𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼑𖽻 𖼅𖽡 , 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽡𖽻𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 .”
10 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽡𖽙𖾐 𖼇𖽔 𖼘𖽻𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼐𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏 , 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 .
