𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏
93
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏！
𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼚𖽔𖾐 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼯𖽡𖽙𖾏 𖼚𖽔𖾐 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏, 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽔𖾏 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼼𖽙 .
2 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼷𖽹𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 𖼮𖽡𖽝𖾏,
𖼟𖽡 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖽃𖽔 𖽐𖼷𖽔 𖼷𖽹𖾏 𖼅𖽔𖾏 𖼋𖽘 .
 
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼳𖽠𖾑 𖼺𖽡𖽝𖾐, 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼳𖽠𖾑 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏,
𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼳𖽠𖾑 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙 .
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼇𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼎𖽙𖾑, 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼊𖽪𖾑 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
 
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼟𖽡 𖽐𖼞𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
