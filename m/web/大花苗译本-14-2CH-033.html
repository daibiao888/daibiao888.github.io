𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻

（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 21：19 - 26）
21 𖼽𖽔𖾏 𖼄𖽞 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 . 22 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 , 𖼾𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 , 𖼽𖽔𖾏 𖼄𖽞 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 、 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏， 23 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽱𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼳𖽻𖾐 𖼢𖽜 , 𖼷𖽷𖾑 𖼽𖽔𖾏 𖼄𖽞 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 , 𖽃𖽺𖾐 𖽃𖽔𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 .
24 𖼐𖽱𖾏 𖼉𖽻 𖼷𖽻𖾏 𖽐𖼷𖽜𖾑 𖼐𖽱𖾏 , 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 . 25 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽔𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼽𖽔𖾏 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 , 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
