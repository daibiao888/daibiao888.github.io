25
𖼀𖽡𖾑 𖼊𖽡𖾐 𖼖𖽔𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐, 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏.
2 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏, 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽹𖾏, 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏.
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼊𖽡𖾐 𖼊𖽙𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 ,
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏.
4 𖼺𖽪𖾐 𖼷𖽑𖽷𖾑 𖼮𖽡𖽺𖾏 𖼺𖽘𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏,
𖼖𖽡𖾏 𖼞𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼗𖽙 .
5 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼐𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼻𖽡𖽝𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏.
6 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼾𖽨𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼷𖽷𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐.
7 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : “𖼷𖽑𖽜𖾑 𖼟𖽡 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏.”
𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 𖼙𖽠𖾏.
 
8 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 ,
𖽃𖽔𖾑 𖼡𖽜𖾐 𖼟𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼋𖽘𖾏.
9 𖼟𖽡 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏 𖼊𖽔𖾐,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
10 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽙𖾑𖽑 𖼖𖽔𖾏 𖼟𖽡 ,
𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼎𖽺𖾏 𖼀𖽜𖾑 𖼦𖽡𖾐 𖼛𖽡𖾏.
 
11 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼗𖽙𖾏 𖼗𖽪𖾏 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼞𖽡𖾑 ,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼷𖽷𖾑 𖼀𖽑𖽡𖽾 𖽐𖼞𖽙𖾐 𖼞𖽪𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼚𖽙 𖼸𖽙𖾏 𖼑𖽡𖽝𖾏.
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐, 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻 𖼸𖽙𖾏 𖽃𖽡𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏,
𖼺𖽱𖾏 𖼁𖽻 𖼞𖽪𖾐 𖽐𖼎𖽙 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏， 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼑𖽔𖾏 .
13 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏, 𖼅𖽔𖾏 𖼚𖽻𖾏 𖽐𖼀𖽪𖾏 𖼑𖽡𖾏 𖼀𖽘𖾏 𖼟𖽪𖾏 𖼸𖽡𖽝𖾏.
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼷𖽻𖾑 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼉𖽡𖽝𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼗𖽔 𖽃𖽡𖾑,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼅𖽔𖾏 𖼮𖽢𖾏 𖼅𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼉𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽘 .
 
15 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖼸𖽙 𖼡𖽘𖾐 𖼸𖽠𖾏 𖼞𖽑𖽻𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼑𖽻 𖼸𖽙𖾏,
𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼅𖽔𖾏 𖼸𖽠𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏.
16 𖼟𖽡 𖼊𖽻𖾏 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏, 𖼑𖽻𖾏 𖼇𖽔𖾑 𖼯𖽡𖾏 𖼷𖽻𖾏 𖼋𖽘𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼟𖽡 𖼑𖽻𖾏 𖼮𖽡𖽻𖾐 𖼺𖽘𖾏 𖼎𖽺𖾏 𖽐𖼊𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 .
17 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼽𖽔𖾑 𖼯𖽠 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼼𖽷𖾏 𖼼𖽙 𖽐𖼞𖽔𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼙𖽔,𖼎𖽺𖾏 𖼳𖽪𖾐 𖼟𖽡.
18 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼼𖽷𖾏 𖼼𖽙 ,
𖼺𖽱𖾏 𖼊𖽑𖽱𖾑 𖼙𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼟𖽪𖾏 𖽐𖼮𖽳𖾏、 𖼈𖽙𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏.
19 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏,
𖼺𖽱𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽡𖽝𖾑𖽑 𖼟𖽪𖾏 𖼗𖽱𖾏, 𖼺𖽱𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼚𖽑𖽘𖾏 𖼀𖽡 𖼢𖽙𖾏.
20 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼟𖽪𖾏 𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼐𖽙𖾏 𖼚𖽠𖾏 𖼮𖽑𖽙𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼘𖽡𖾑𖽑 𖼞𖽔𖾏 𖼦𖽡𖾐 𖼡𖽻𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐.
21 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼮𖽔𖾑 𖼮𖽑𖽹𖾐 𖼋𖽘, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼉𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼑𖽻𖾏,
𖼮𖽔𖾑 𖽐𖼡𖽑𖽜𖾏 𖼋𖽘𖾏, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽻𖾏.
22 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡 𖼊𖽑𖽜𖾏 𖼷𖽱𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼋𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 .
23 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼨𖽑𖽳𖾏 𖼑𖽘 𖼋𖽔𖾏,
𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽪𖾑 𖼚𖽑𖽪𖾏,
24 𖼁𖽻𖾏 𖼏𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼏𖽪𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼞𖽪𖾐,
𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼐𖽡𖽙𖾐.
