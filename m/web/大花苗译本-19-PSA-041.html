𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.

𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼊𖽪𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏.
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼖𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏．
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑.
3 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼄𖽙𖾐 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏 𖼸𖽘𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏,
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼻𖽻𖾏, 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼸𖽱𖾏 𖼸𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽱𖾏.
 
4 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑,
𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼞𖽪𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 𖼺𖽘𖾏.”
5 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑, 𖼐𖽑𖽡𖾏:
“𖼊𖽻𖾑𖽑 𖼋𖽱 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼋𖽔, 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏 𖼖𖽺𖾑?”
6 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼑𖽔𖾏 𖼞𖽪𖾑, 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼮𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼋𖽠𖾏 𖼚𖽔𖾏 𖼻𖽻𖾑 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏.
7 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼳𖽪𖾐 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖽃𖽡𖾐 𖼸𖽻 𖼦𖽡𖾏 𖽐𖼷𖽑𖽷𖾏 𖼀𖽡𖾏 𖼮𖽑𖽡𖾑 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑.
 
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑,
𖼐𖽱𖾏 𖼄𖽙𖾐 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏 𖼋𖽘𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏.
9 𖼽𖽺 𖼗𖽔 𖼟𖽪𖾏 𖼷𖽑𖽡𖽝𖾏 𖼀𖽻𖾐 𖼷𖽑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 ,
𖼟𖽪𖾏 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏、𖼟𖽪𖾏 𖼙𖽔 𖼑𖽻𖾏 𖼞𖽪𖾑 𖼳𖽡𖽝𖾐 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼮𖽑𖽳𖾑 𖼡𖽘𖾐 𖽃𖽔 𖼙𖽻𖾏 𖼊𖽠𖾏 𖼋𖽔𖾏 𖼞𖽪𖾑.
10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼻𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏！
 
11 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖼞𖽪𖾑,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏.
12 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼎𖽺𖾏 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
 
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏,
𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼸𖽙𖾏 𖼚𖽔𖾏 𖼗𖽡𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽐𖼊𖽡𖾏, 𖽃𖽔𖾏 𖼄𖽞！ 𖽃𖽔𖾏 𖼄𖽞！
