𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘
（𖼋𖽠𖾏 38：1 - 7）
27
“𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 . 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼞𖽪𖾐 , 𖽐𖼊𖽡𖾑 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼇𖽘𖾑 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼺𖽡𖽝𖾐 𖼷𖽷𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 2 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖼐𖽡𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽪𖾐 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼋𖽪 𖼞𖽪𖾐 , 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽘 𖼦𖽡𖾏 𖼊𖽡𖾏 𖼦𖽡𖾏 𖼻𖽡𖽝𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼋𖽻𖾏 𖼡𖽑𖽠𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼮𖽡𖽝𖾏 . 3 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼋𖽺𖾏 , 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼮𖽑𖽻𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖽐𖼚𖽙 , 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼚𖽡𖽝𖾑 𖼞𖽻𖾏、𖼋𖽺𖾏、𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖽐𖼡𖽹𖾏、𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 , 𖽃𖽔𖾑 𖼏𖽘 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 𖼊𖽔𖾐 . 4 𖼽𖽔𖾑 𖼑𖽜 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼋𖽻𖾏 𖼮𖽡𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼸𖽙𖾏 , 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼮𖽡𖾏 𖼡𖽜𖾏 𖼸𖽙𖾏 𖼋𖽻𖾏 𖼐𖽡𖾑 𖼚𖽻𖾐 𖼋𖽪 𖼞𖽪𖾑 𖼞𖽪𖾐 , 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼖𖽪𖾐 𖼋𖽻𖾏 𖽃𖽔 𖼉𖽙𖾏， 5 𖼞𖽑𖽡𖽠𖾏 𖼖𖽪𖾑 𖼮𖽡𖾏 𖼡𖽜𖾏 𖼸𖽙𖾏 𖼊𖽡𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽻𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼀𖽡𖾑 𖼊𖽡𖾐 , 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼮𖽡𖾏 𖼡𖽜𖾏 𖼸𖽙𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼸𖽙 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼚𖽔𖾏 . 6 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖼑𖽜 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾏 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 , 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏 . 7 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼙𖽡 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾐 𖼺𖽘𖾐 𖽐𖼚𖽙 𖽃𖽔 𖼉𖽙𖾏 , 𖼟𖽙𖾏 𖼻𖽻𖾏 𖼞𖽱𖾑 𖽃𖽔𖾑 𖼏𖽘 . 8 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼊𖽙𖾏 𖽐𖼚𖽙 , 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼟𖽡 𖽃𖽡𖾑 𖽃𖽔𖾏 .”
𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏
（𖼋𖽠𖾏 38：9 - 20）
9 “𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 . 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 𖽃𖽔𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏, 𖽐𖼊𖽡𖾑 𖽃𖽡𖾐 𖼀𖽔𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 10 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐, 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼗𖽪𖾏 , 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾑 𖼡𖽠𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 . 11 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 , 𖽐𖼊𖽡𖾑 𖽃𖽡𖾐 𖼀𖽔𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼋𖽪𖾏 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼗𖽪𖾏 , 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 . 12 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏, 𖼇𖽘𖾑 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼋𖽪𖾏 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽻𖾏 𖼗𖽪𖾏 . 13 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼽𖽔𖾑 𖼇𖽘𖾑 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 14 𖽃𖽔 𖼛𖽻𖾏 𖼺𖽘𖾑 𖼐𖽡𖾑 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖼽𖽔𖾑 𖼟𖽻𖾏 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼷𖽷𖾐 𖼋𖽪 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾐 𖼖𖽪𖾐 . 15 𖽃𖽔 𖼛𖽻𖾏 𖼺𖽘𖾑 𖽃𖽡𖾑 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼟𖽻𖾏 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼷𖽷𖾐 𖼋𖽪 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾐 𖼖𖽪𖾐 . 16 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔 𖼛𖽻𖾏 , 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖼊𖽑𖽡𖽝𖾐 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 , 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼁𖽘𖾏 𖽐𖼊𖽙𖾏 𖼊𖽡𖽝𖾑 , 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼚𖽻𖾐 𖼋𖽪 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼚𖽻𖾐 𖼖𖽪𖾐 . 17 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼚𖽑𖽙𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 , 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖽃𖽡𖾑 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 . 18 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼽𖽔𖾑 𖽐𖼊𖽡𖾑 𖽃𖽡𖾐 𖼀𖽔𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼇𖽘𖾑 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼺𖽡𖽝𖾐 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 𖽃𖽔𖾏 , 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 . 19 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 , 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼘𖽑𖽻𖾐 𖼟𖽪𖾏 𖼡𖽑𖽹𖾐 𖼖𖽑𖽔𖾏 , 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖽐𖼚𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼘𖽑𖽻𖾐 𖼟𖽪𖾏 𖼡𖽑𖽹𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 . ”
