𖼖𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼅𖽳𖾏
6
𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 :
𖼳𖽠𖾑 𖼗𖽙𖾏, 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼟𖽙𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼑𖽻 𖼟𖽡 𖼗𖽪𖾏 .
2 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼑𖽡𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾑 𖼮𖽤𖾐 𖼊𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 𖼘𖽪 !
𖼽𖽔𖾑 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
 
3 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼐𖽔𖾑 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼖𖽺𖾑?
𖼯𖽪𖾏 𖼟𖽙𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽙𖾏 𖽐𖼎𖽑𖽹𖾏.
4 𖼞𖽪𖾑 𖼙𖽔 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼎𖽑𖽻𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼟𖽡 ,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 、 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼣𖽔𖾏 ，𖼅𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏.
5 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪 ! 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽔𖾑 𖼖𖽳 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼖𖽔 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 ,
𖼎𖽺𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼳𖽺 𖼊𖽑𖽡𖽾 𖼸𖽙 𖼮𖽡 𖼮𖽢 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 ,
𖼻𖽻𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 .
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏
6 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏, 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪?
𖼋𖽡𖽝𖾏 𖼾𖽙 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼘𖽪?
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽡𖽝𖾏 𖼙𖽔 𖼡𖽔𖾑 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏,
𖼡𖽑𖽙𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼉𖽻𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖼛𖽙𖾏 𖼘𖽪?
𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼖𖽑𖽘𖾑 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼑𖽜 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼖𖽑𖽘𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼡𖽑𖽙𖾑 𖼾𖽨 𖽃𖽡𖾑 𖼘𖽪?
8 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪 ! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏?
𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼽𖽔𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪?
𖼯𖽔 𖼽𖽔𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑, 𖼉𖽳 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐,
𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏.
𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏．
𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼑𖽻 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼟𖽪𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏.
10 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑
𖽐𖼎𖽙 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼳𖽜𖾐 𖼊𖽱𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼺𖽷𖾐 𖼘𖽪?
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼸𖽱𖾏 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳
𖽐𖼎𖽙 𖽐𖼚𖽙 𖼐𖽑𖽘𖾐 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼖𖽺𖾑?
12 𖽐𖼚𖽙 𖼙𖽻 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼊𖽻𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽪𖾏 𖼸𖽳,
𖼉𖽺𖾏 𖽐𖼚𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼖𖽔𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼯𖽡𖽙𖾏.
13 𖽃𖽔𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖼟𖽡, 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼄𖽙𖾐,
𖼟𖽙𖾏 𖼟𖽡 𖼑𖽜 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼙𖽠𖾏.
𖼖𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳
14 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻𖾏, 𖼈𖽡𖽝𖾏 𖼑𖽻𖾏 𖼦𖽡𖾐 𖼮𖽻𖾏,
𖼟𖽡 𖽃𖽔𖾑 𖼚𖽘𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 .
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽻𖾏 𖽐𖼊𖽡𖾏 𖼮𖽡𖽝𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏.
15 𖼟𖽡 𖼀𖽳𖾐 𖼮𖽘𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏,
