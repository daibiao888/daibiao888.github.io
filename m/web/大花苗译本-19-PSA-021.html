𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽻𖾐 𖼋𖽘 𖼞𖽡𖾑 𖼀𖽔𖾑
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
21
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖼑𖽜 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏, 𖼐𖽱𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝！
2 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼡𖽑𖽙𖾑 𖽐𖼮𖽑𖽠𖾏 𖼊𖽻𖾏, 𖼟𖽡 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖽐𖼊𖽜 𖽃𖽡𖾑, 𖼟𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼦𖽡𖾐 𖼑𖽻 𖼊𖽡𖾐. （𖼳𖽡𖾏 𖼖𖽔𖾑）
3 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼙𖽔 𖼷𖽹𖾏 𖼐𖽱𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼞𖽪𖾐 𖽐𖼊𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏.
4 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼊𖽜 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖼟𖽡 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏,
𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼗𖽡𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏.
5 𖼐𖽱𖾏 𖼑𖽜 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼟𖽡 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼚𖽔𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼮𖽑𖽨𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏.
6 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑.
7 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼑𖽜 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙.
 
8 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼋𖽠𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑,
𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼋𖽠𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼟𖽡 𖽃𖽡𖾑.
9 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼟𖽪𖾏 𖼞𖽪𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐．
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼡𖽙𖾏 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼋𖽡𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽷𖾑 𖼟𖽙𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 .
10 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼺𖽘𖾏,
𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏.
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼖𖽔𖾑 𖼊𖽑𖽡𖽝𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼟𖽡 𖼈𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖼋𖽠𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑.
12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼏𖽘 𖼀𖽜𖾑,
𖽐𖼊𖽡𖾏 𖼈𖽙𖾏 𖼎𖽻𖾏 𖼐𖽑𖽜𖾑 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽑𖽪𖾏.
 
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼑𖽜 𖼾𖽨𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽺𖾐 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏！
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
