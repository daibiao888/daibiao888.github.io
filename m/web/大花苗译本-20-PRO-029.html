29
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽪 𖼯𖽪𖾏 𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏,
𖼐𖽱𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 , 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼞𖽑𖽙𖾏.
2 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼙𖽔 ；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽔𖾑 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏.
3 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼟𖽙𖾏 𖼉𖽺𖾏 𖽃𖽺𖾐 𖼙𖽔；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔 , 𖼟𖽪𖾏 𖽐𖼞𖽡𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐.
4 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 , 𖼟𖽙𖾏 𖼞𖽪𖽙 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏；
𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 , 𖼟𖽙𖾏 𖼞𖽪𖽙 𖼊𖽻𖾏 𖼢𖽻.
5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑 ,
𖼟𖽪𖾏 𖼊𖽑𖽔𖾑 𖼸𖽙𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼊𖽠𖾏.
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏,
𖼮𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼙𖽔 𖼛𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑.
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽔𖾏 𖽐𖼎𖽑𖽹𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼐𖽡𖾑 .
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼗𖽔 𖽃𖽡𖾐 𖼙𖽻 𖼙𖽱 𖽐𖼷𖽑𖽙𖾑 𖼑𖽪𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼺𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐.
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼮𖽔𖾑 𖼸𖽠𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼊𖽔𖾐, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 .
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑, 𖼳𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽐𖼮𖽑𖽠𖾏 𖼮𖽑𖽔𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑.
11 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼷𖽜𖾏 𖼋𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐.
12 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽔𖾑 𖼑𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾐.
13 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼦𖽡𖾐 𖽐𖼮𖽷𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽻𖾐 𖼮𖽡𖾏 .
14 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼨𖽑𖽢𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏,
𖼐𖽱𖾏 𖼞𖽪𖽙 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽡𖾏 𖽐𖼊𖽡𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏.
15 𖼎𖽑𖽻𖾏 𖼡𖽠𖾏 𖽐𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼸𖽠𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳,
𖼊𖽪𖾐 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼯𖽔𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼁𖽜𖾏 𖼺𖽡𖽝𖾐 𖼑𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
16 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽻𖾐, 𖼾𖽔𖾏 𖼖𖽔𖾏 𖼎𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽻𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼛𖽙𖾏 𖼋𖽔 .
17 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼟𖽡 𖼊𖽪𖾐, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 ,
𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
18 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 ( “𖼚𖽔𖾐 𖼼𖽘𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼅𖽜 𖼡𖽑𖽜𖾐 ” )𖼖𖽡𖾑 𖼉𖽪𖾏 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏；
𖼮𖽔 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
19 𖼮𖽔𖾑 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼊𖽔𖾐, 𖼡𖽑𖽜𖾑 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 ,
𖼐𖽱𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 .
20 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼙𖽔 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽹𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼖𖽺𖾑?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏 .
21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼮𖽙𖾏 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑,
𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖼐𖽡𖾑 𖽐𖼎𖽪𖾏 𖼡𖽜𖾑 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏.
22 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼮𖽡𖽻𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼾𖽔𖾏.
23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾏 𖼢𖽜𖾏 𖼗𖽙𖾏 ；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐 𖼢𖽜, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼚𖽻𖾏 𖼮𖽡𖾏 .
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖽐𖼎𖽙 𖼸𖽜 𖼇𖽺𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑, 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖼐𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 , 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏.
25 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼀𖽻𖾐 𖼁𖽻 𖼖𖽪𖾑 𖼸𖽙𖾏；
𖼮𖽔𖾑 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
26 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼊𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽳𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽹𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼋𖽔𖾏.
27 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 , 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼳𖽪𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼳𖽪𖾐.
