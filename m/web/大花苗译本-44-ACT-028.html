7 𖼡𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼚𖽡𖾐，𖼅𖽔𖾏 𖼀𖽡 𖼯𖽠 𖼮𖽡𖽙𖾐 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼮𖽵𖾑 𖼁𖽡𖽝，𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼮𖽵𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽪𖾏 𖼀𖽳 𖼖𖽨 ，𖼐𖽱𖾏 𖼷𖽹𖾏 𖼀𖽡𖾐，𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼷𖽹𖾏 𖼯𖽡𖽙𖾏 𖼀𖽡𖾐 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐． 8 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼀𖽪𖾏 𖼀𖽳 𖼖𖽨 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐，𖼄𖽙𖾐 𖼄𖽙𖾐 𖼚𖽪𖾐 𖽐𖼎𖽙 𖼄𖽙𖾐 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼖𖽡𖽝𖾐 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏， 𖼀𖽻𖾐 𖼘𖽙 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼐𖽱𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏． 9 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐，𖼎𖽺𖾏 𖼋𖽔𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏． 10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐， 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖽐𖼞𖽙𖾏，𖼎𖽺𖾏 𖼺𖽘𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖽃𖽡𖾑 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏．
