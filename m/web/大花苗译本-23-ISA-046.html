𖼀𖽡𖾐 𖼖𖽳 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼐𖽡 𖼀𖽑𖽙𖾑 𖼞𖽑𖽻𖾑 𖼚𖽔𖾑 ,
𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼁𖽠𖾏 𖼮𖽹𖾑 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽱𖾑 𖽃𖽡𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑, 𖼟𖽙𖾏 𖼯𖽦𖾏 𖼯𖽻 𖼼𖽙 .
2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑、 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐、
𖼦𖽡𖾐 𖼸𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽐𖼎𖽘𖾏 𖼐𖽑𖽤𖾑 𖼟𖽪𖾏 𖼮𖽹𖾑,
𖼾𖽨𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏.
 
3 “𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏.
𖼅𖽡 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽻𖾏 𖼞𖽪𖾑 𖼡𖽑𖽠𖾑 𖼁𖽔 ,
𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼁𖽔 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐.
4 𖼸𖽙 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑,
𖼸𖽙 𖼟𖽪𖾏 𖼅𖽡 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼚𖽠𖾐, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼀𖽔 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐．
𖼞𖽪𖾑 𖼙𖽔 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼁𖽔 ,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽔 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏.
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼺𖽱𖾏 𖼐𖽱𖾏
5 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼗𖽙𖾏 𖽐𖼮𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 ,
𖼟𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼸𖽷𖾐, 𖽐𖼮𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 , 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼦𖽡𖾏 𖼺𖽱𖾏 𖼖𖽺𖾑?
6 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖽐𖼚𖽙 𖼘𖽡𖾑𖽑 𖼋𖽠𖾏 𖼞𖽪𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼞𖽡𖾏 𖼙𖽪𖾏 𖼑𖽡𖽝𖾏,
𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼎𖽻𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏，𖼟𖽙𖾏 𖼷𖽷𖾏 𖼊𖽡𖽝𖾑 𖼁𖽠𖾏 𖼼𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽺𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼁𖽠𖾏 𖼼𖽘𖾏 𖼞𖽱𖾑 𖼮𖽡𖽝𖾏, 𖼞𖽱𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼊𖽜𖾐 𖼟𖽙𖾏 𖼞𖽑𖽙𖾑 𖼮𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏,
𖼐𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽡𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
 
8 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼊𖽱𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑, 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼀𖽡𖾏 𖼮𖽨𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 𖼘𖽪！ 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖽐𖼊𖽱𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏.
9 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼑𖽪𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼞𖽪𖾑 .
10 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖼡𖽑𖽳𖾏 𖽐𖼎𖽑𖽹𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼑𖽪𖾏,
𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑,
𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼙𖽔 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑.
11 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼨𖽑𖽪𖾏 𖼊𖽪𖾑 𖼚𖽘𖾑 𖼋𖽔𖾏,
𖼨𖽑𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼋𖽔𖾏．
𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽱𖾏 𖼮𖽡𖽝𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑.
 
12 𖼅𖽡 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏,
𖼇𖽳𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼚𖽡𖾐 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏，
13 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼡𖽔𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐．
𖼞𖽪𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽳𖾑𖽑 𖼮𖽡𖽝𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏,
𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏.”
