𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳、𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑、 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏、 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐，𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑， 2 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼑𖽡𖾏 𖼞𖽔𖾏 𖼄𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼘𖽑𖽘𖾑 𖼾𖽨 𖽐𖼮𖽑𖽠𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐， 𖼟𖽙𖾏 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼘𖽑𖽙𖾐 𖼟𖽡𖾏 𖼾𖽙， 𖼸𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏． 3 𖼅𖽡 𖼮𖽔𖾑 𖼙𖽔 𖼐𖽑𖽡𖽝𖾏 𖼮𖽵𖾑 𖼅𖽳𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼦𖽘𖾏， 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 4 𖼮𖽵𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽹𖾑 𖼮𖽡𖽝𖾏 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝． 5 𖼅𖽡 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏，𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽔𖾏 𖼖𖽡𖽾 𖽐𖼞𖽔𖾏， 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏， 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 ，𖼖𖽑𖽘𖾑 𖽃𖽔𖾏 𖼖𖽡𖽾 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼷𖽹𖾏 𖼳𖽻𖾏． 6 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：
“𖼅𖽡 𖼑𖽔𖾏， 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼞𖽪𖾐 𖽐𖼞𖽔𖾐 𖼊𖽠𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐
𖼟𖽪𖾏 𖼊𖽻𖾏 𖼺𖽹𖾑 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑，
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼐𖽱𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏．”
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑，𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏：
“𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏，
𖼎𖽺𖾏 𖽃𖽔𖾏 𖼞𖽪𖾐 𖽐𖼞𖽔𖾐 𖼊𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼮𖽡𖽙𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑．”
8 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
“𖽃𖽔𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽠𖾏，
𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼢𖽻．”
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼐𖽡𖾑 ，𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽠𖾏， 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼊𖽡𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏．
9 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼺𖽹𖾑，𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼾𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 ，𖼟𖽪𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏， 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼑𖽡𖾏 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼟𖽡𖾏 𖽐𖼊𖽙𖾏 𖽐𖼷𖽷𖾏．
10 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼖𖽡𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼺𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏，
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼎𖽺𖾏 𖽃𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏，
𖽃𖽔 𖼊𖽑𖽻𖾏 𖼖𖽡𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼊𖽻𖾏 𖼅𖽳𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐，
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼋𖽘𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐．
