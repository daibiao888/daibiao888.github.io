28 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼷𖽻𖾑 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼐𖽱𖾏 𖼈𖽻𖾑 𖼖𖽔𖾏 𖼷𖽻𖾑 𖼐𖽱𖾏,
29 𖼐𖽑𖽡𖾏 : “ 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐,
𖼮𖽔𖾑 𖼅𖽔𖾏 𖼟𖽡 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 .”
30 𖼻𖽻𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐, 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼖𖽔𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼯𖽡𖽙𖾏,
𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑.
31 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑,
