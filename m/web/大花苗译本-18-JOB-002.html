𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼀𖽙𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 , 𖼺𖽔𖾐 𖼊𖽔 𖼖𖽔𖾏 𖼋𖽔𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 . 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽪 𖼺𖽔𖾐 𖼊𖽔 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏 ?” 𖼺𖽔𖾐 𖼊𖽔 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏 .” 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽪 𖼺𖽔𖾐 𖼊𖽔 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖽝𖾏 𖼟𖽡 𖼙𖽔 𖼎𖽑𖽻𖾏 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼽𖽙 𖼀𖽳 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼙𖽔 ? 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼐𖽱𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏、 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏、 𖼀𖽜𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 . 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐 , 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼈𖽡𖽝𖾏 , 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .” 4 𖼺𖽔𖾐 𖼊𖽔 𖼊𖽡𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼘𖽑𖽪𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏 , 𖼯𖽪𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼺𖽘𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 . 5 𖼟𖽡 𖼮𖽔𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼡𖽹𖾏 , 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼟𖽡 𖼺𖽘𖾏 .” 6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼊𖽔 𖼐𖽑𖽡𖾏 : “𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼮𖽔 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼳𖽦𖾑 .”
7 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼺𖽔𖾐 𖼊𖽔 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽡𖾏 𖼙𖽠𖾏 , 𖼟𖽡𖾏 𖽐𖼊𖽻𖾏 𖼽𖽙 𖼀𖽳 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽠𖾏 𖼸𖽙 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼳𖽠𖾑 𖼷𖽑𖽘𖾐 𖼀𖽡𖾏 𖼮𖽡𖽙𖾏 . 8 𖼽𖽙 𖼀𖽳 𖼐𖽡𖽙𖾐 𖼮𖽑𖽻𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼷𖽑𖽹𖾏 𖼉𖽔𖾏 𖼞𖽻𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
9 𖼐𖽱𖾏 𖼐𖽤𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼺𖽷𖾐 𖼘𖽪𖾏 ? 𖼟𖽡 𖼇𖽳𖾑 𖼗𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 , 𖼋𖽔 𖼺𖽘𖾏 𖼦𖽡𖾑！” 10 𖼈𖽡𖽝𖾏 𖼽𖽙 𖼀𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼺𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼷𖽠𖾏 𖼡𖽘𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐． 𖽃𖽺𖾐 ! 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼊𖽻𖾏 𖼅𖽳𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼘𖽪𖾏 ？” 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 , 𖼽𖽙 𖼀𖽳 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 .
𖼽𖽙 𖼀𖽳 𖼑𖽡𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔
11 𖼽𖽙 𖼀𖽳 𖼑𖽡𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔 𖼊𖽑𖽡 𖼄𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 、 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾐 𖼖𖽳 𖼊𖽔 、 𖼐𖽔 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐𖼺𖽙𖾐 𖼇𖽔 , 𖼐𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼮𖽡𖽝𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽔𖾏 , 𖼑𖽜 𖼐𖽱𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼐𖽱𖾏 . 12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 , 𖼯𖽜𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼋𖽘𖾏 , 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼮𖽑𖽷𖾏 𖼺𖽘𖾏 𖼾𖽨𖾏 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 , 𖼉𖽺𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼨𖽑𖽢𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 . 13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼳𖽤𖾏 𖼄𖽑𖽙𖾏 , 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼑𖽻𖾏 𖽃𖽦𖾐.
