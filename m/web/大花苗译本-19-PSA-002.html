𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼛𖽙𖾏 𖼚𖽜𖾑
2
𖼉𖽻𖾏 𖼞𖽪𖽙 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏?
𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼊𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼖𖽺𖾑?
2 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏,
𖼉𖽻 𖼷𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼦𖽡𖾏 𖼊𖽳𖾏,
𖼽𖽔𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫
𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑.
3 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼽𖽔𖾑 𖼀𖽡𖾏 𖽐𖼊𖽡𖾏 𖼛𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐,
𖼘𖽱𖾑𖽑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏.”
 
4 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽙𖾏,
𖼮𖽵𖾑 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
5 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏.
6 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏.
 
7 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏, 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐,
𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼅𖽔𖾏 𖼟𖽡.
8 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏,
𖼄𖽔𖾐 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏.
9 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏 𖼖𖽑𖽻𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼨𖽑𖽻𖾐 𖼡𖽑𖽠𖾏.”
 
10 𖼐𖽔𖾑 𖼐𖽡𖾑, 𖼅𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼯𖽷𖾏 𖼨𖽑𖽪𖾑,
𖼅𖽡 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏.
11 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼎𖽺𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑.
12 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽻𖾐 𖼊𖽪𖾐, 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼅𖽡 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼺𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼇𖽺𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
