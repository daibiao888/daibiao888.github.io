8 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 ，𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡， 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼳𖽙𖾑， 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼋𖽡𖽝𖾏 𖼋𖽡𖽝 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝 ． 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖼅𖽳𖾏： 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼑𖽜 𖼅𖽡 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 ， 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝． 10 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽡𖾑 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼋𖽘𖾏， 11 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽙𖾏 𖼊𖽡𖽝𖾑， 𖼅𖽡 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼯𖽪𖾏 𖽃𖽔𖾏， 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼊𖽡𖽝𖾑． 12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼯𖽪𖾏 𖽃𖽔𖾏， 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾑，𖼊𖽻𖾏 𖼙𖽔 𖼷𖽹𖾏 𖼳𖽻𖾏，𖼦𖽡𖾐 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏． 13 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽡𖽝 𖼳𖽷𖾐， 𖼅𖽡 𖼁𖽡𖽝 𖼐𖽑𖽤𖾑， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼦𖽡 𖼋𖽳． 14 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼊𖽻𖾏， 𖽐𖼷𖽷𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏， 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏，𖼎𖽺𖾏 𖽐𖼷𖽷𖾑 𖽐𖼀𖽘𖾏 𖼅𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 ， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖾏 𖼐𖽑𖽤𖾑 𖼦𖽡 𖼋𖽳 𖼋𖽘𖾏； 15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：
