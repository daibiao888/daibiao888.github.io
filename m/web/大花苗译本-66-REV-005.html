𖼞𖽪𖾑 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼑𖽘 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼮𖽡𖽝𖾏， 𖽐𖼚𖽙 𖼡𖽻𖾏 𖼖𖽔𖾏 𖼺𖽻𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼊𖽠𖾑， 𖼎𖽑𖽻𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖼘𖽙𖾏 𖼟𖽪𖾏 𖼏𖽔𖾏 𖽐𖼎𖽜𖾏 𖼮𖽡𖽝𖾏． 2 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 ，𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖽐𖼸𖽷𖾏 𖼡𖽑𖽜𖾐 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖽃𖽡𖾑，𖼎𖽺𖾏 𖼡𖽑𖽜𖾐 𖼋𖽡𖾏 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖽃𖽡𖾑 𖼖𖽺𖾑 ？” 3 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 、𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 、 𖼼𖽘𖾏 𖼚𖽘𖾐 𖽐𖼚𖽡𖽝 𖼊𖽡𖾐，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼡𖽑𖽜𖾐、 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖽃𖽡𖾑 ． 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼸𖽷𖾏 𖼡𖽑𖽜𖾐 、𖽐𖼸𖽷𖾏 𖼑𖽔𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖽃𖽡𖾑，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑． 5 𖼊𖽪𖾐 𖼗𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖼐𖽡𖽝𖾑 𖼊𖽡𖾐 ， 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼽𖽨 𖼊𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐，𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐，𖼐𖽱𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼸𖽠𖾏 𖼡𖽑𖽜𖾐 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖽃𖽡𖾑 ，𖼡𖽑𖽜𖾐 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖽃𖽡𖾑．”
6 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼎𖽙 𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼗𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼊𖽻𖾏 𖼊𖽔𖾏 ， 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼞𖽪𖾐 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔， 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼖𖽡𖽾，𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼚𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼋𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑． 7 𖼊𖽪𖾑 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼐𖽡𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏，𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼞𖽑𖽡𖽠𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼗𖽙𖾏． 8 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖽃𖽡𖾑 𖼙𖽠𖾏，𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽐𖼎𖽙 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼚𖽻𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼗𖽻 𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽷𖾑 𖼺𖽙𖾑 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼷𖽙𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖽃𖽔𖾑 𖼋𖽪 𖼳𖽤𖾏，𖼋𖽡𖾏 𖽃𖽔 𖼋𖽪 𖼳𖽤𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏． 9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼮𖽑𖽦𖾐，𖼐𖽑𖽡𖾏：
“𖼟𖽡 𖽐𖼸𖽷𖾏 𖼞𖽑𖽡𖽠𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼮𖽡𖽝𖾏，
𖽐𖼸𖽷𖾏 𖼡𖽑𖽜𖾐 𖼋𖽡𖾏 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖼊𖽻𖾏 𖼊𖽔𖾏，
𖼟𖽡 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖽐𖼮𖽑𖽘𖾑 𖼎𖽑𖽻𖾏 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 、𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼗𖽪𖾏、𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 、𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽙𖾑 𖼗𖽙𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
10 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏，𖼎𖽙𖾑 𖼚𖽔𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏；
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏．”
𖼷𖽻𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼾𖽘𖾏 𖼁𖽜𖾏
11 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼑𖽔𖾏 ， 𖼐𖽑𖽙𖾑 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼗𖽻 𖼅𖽡 𖼟𖽻𖾏 𖼯𖽪𖾏，𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼷𖽑𖽡𖽝𖾐 𖼷𖽑𖽡𖽝𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏， 12 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏 ：
“ 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼊𖽻𖾏 𖼊𖽔𖾏 𖽐𖼸𖽷𖾏 𖼊𖽻𖾏
𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 、𖼀𖽪𖾑 𖼷𖽻𖾏、𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳、𖼛𖽘𖾏 𖼼𖽙𖾏 、
𖼐𖽑𖽨𖾑 𖼙𖽪𖾏、𖽐𖼚𖽻 𖼮𖽡𖾏、𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏 ．”
13 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏、𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐、𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼼𖽘𖾏 𖼚𖽘𖾐 𖽐𖼚𖽡𖽝 𖼊𖽡𖾐、𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑，𖽐𖼎𖽙 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐，𖽃𖽡𖾐 𖼸𖽻 𖼐𖽑𖽡𖾏：
“𖼊𖽻𖾏 𖼅𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 、 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 、𖽐𖼚𖽻 𖼮𖽡𖾏 、𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 ，
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼎𖽙 𖼾𖽘𖾏 𖼁𖽜𖾏，
𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏！”
14 𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔𖾏 𖼄𖽞！” 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼗𖽻 𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏．
