𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼞𖽳𖾑 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐
（𖼋𖽠𖾏 27：20 - 21）
24
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽗𖾐 𖼖𖽗 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽦𖾐 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽠𖾏 𖼎𖽻𖾐 , 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽠𖾏 𖼮𖽡𖽝𖾏 . 3 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼚𖽙 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼀𖽑𖽙𖾑 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽺𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 , 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼎𖽑𖽻𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 , 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽳𖾑 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼐𖽡𖾑 . 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 . 4 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼳𖽻𖾏 𖼞𖽳𖾑 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 .”
𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼊𖽜𖾐 𖼮𖽡𖽝𖾏
5 “𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 , 𖼷𖽑𖽡𖽝𖾐 𖽃𖽔𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 , 𖼷𖽑𖽔𖾏 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼯𖽡𖽙𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽔𖾐 𖽐𖼀𖽠 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐. 6 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼳𖽡𖽝𖾐 𖼊𖽜𖾐 𖼦𖽡𖾐 𖼮𖽳𖾐 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼼𖽘𖾏 , 𖽃𖽡𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼚𖽻𖾏 𖼮𖽡𖽙𖾐 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽪𖾑 𖼞𖽡 𖼏𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼦𖽡𖾏 𖼇𖽔𖾑 . 7 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼵𖽪𖾐 𖼳𖽤𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖽐𖼞𖽑𖽳𖾏 𖼳𖽡𖽝𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽔𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 , 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 8 𖼷𖽑𖽔𖾏 𖼞𖽑𖽱𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼽𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽜𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 . 9 𖼋𖽡𖾏 𖼳𖽡𖽝𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼑𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼊𖽻𖾏 .”
𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
10 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽔 𖼁𖽙𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 , 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼇𖽜𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐． 𖼖𖽜𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼾𖽡 𖽐𖼚𖽙 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 11 𖼖𖽜𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽔 𖼁𖽙𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼐𖽡𖾑 , 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼎𖽺𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 , 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 . 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼳𖽷𖾏 𖼘𖽙 𖼄𖽡 , 𖼟𖽪𖾏 𖼊𖽗𖾏 𖽃𖽡𖾐 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼊𖽡𖾐 𖼀𖽳 𖼖𖽡𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 . 12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼮𖽡𖽝𖾏 , 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 .
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 14 “𖼄𖽔𖾐 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼨𖽑𖽳𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖼾𖽡 , 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼋𖽔 𖼐𖽱𖾏 . 15 𖼟𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏 . 16 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼋𖽔 𖼐𖽱𖾏 . 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 , 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼺𖽘𖾏.
