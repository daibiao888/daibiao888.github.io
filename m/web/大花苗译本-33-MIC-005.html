𖼖𖽡𖾑 𖼉𖽪𖾏 ( “ 𖼖𖽡𖾑 𖼉𖽪𖾏 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 ”)𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 𖼘𖽪 !
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏,𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼀𖽡𖾐 ,
𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖽐𖼊𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏.
2 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖼙𖽻𖼽𖽡𖾑 𖼇𖽔 𖼊𖽑𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼘𖽪 !
𖼟𖽡 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐,
𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽜𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏, 𖼎𖽑𖽻𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖽃𖽡𖾑 𖼷𖽹𖾏 𖼅𖽔𖾏 𖼋𖽘 .
3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖼋𖽙𖾏 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼊𖽪𖾐 𖽃𖽡𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼺𖽘𖾏,
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼐𖽱𖾏 ”),
𖼐𖽱𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑.
𖼞𖽪𖽙 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼘𖽑𖽙𖾐 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏
4 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐,
𖼊𖽪𖾑 𖼾𖽨 𖼐𖽱𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏.
 
5 𖼷𖽷𖾑 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏 𖼗𖽙 𖼋𖽡𖽝𖾏 𖼋𖽪 .
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼀𖽡𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐,
𖼯𖽪𖾏 𖼖𖽻𖾐 𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼀𖽡𖾐 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏,
𖼾𖽡𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 .
6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼽𖽔𖾏 𖼳𖽪 𖼊𖽡𖾐
𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽡 𖼖𖽪 𖼊𖽡𖾐.
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼀𖽡𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼯𖽪𖾏 𖼖𖽻𖾐 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐 .
7 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼺𖽱𖾏 𖼀𖽡𖾑 𖼙𖽱 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼑𖽘 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖽐𖼡𖽜𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼻𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐.
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐
8 𖼽𖽔𖾐 𖼞𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼺𖽱𖾏 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼀𖽔𖾏 𖼮𖽘𖾑 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
𖼐𖽱𖾏 𖼮𖽔𖾑 𖼮𖽑𖽡𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽠𖾏 𖼯𖽪𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼺𖽘𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼮𖽨𖾏 𖽐𖼀𖽘𖾏.
9 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.
 
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼑𖽜𖾏 ,
𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏，
11 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼟𖽡 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼙𖽻 𖼺𖽘𖾏,
