6
𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼮𖽔𖾑 𖼑𖽜 𖼽𖽺 𖼗𖽔 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾏,
𖽐𖼀𖽘𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽔𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾏，
2 𖼟𖽡 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼗𖽪𖾏 𖼅𖽘 𖼮𖽡𖽝𖾏,
𖼊𖽻𖾏 𖼾𖽨𖾏 𖼗𖽪𖾏 𖼄𖽔𖾐 𖼊𖽻𖾏.
3 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼽𖽺 𖼗𖽔 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼾𖽨𖾏,
𖼟𖽡 𖼽𖽔𖾑 𖼳𖽻𖾐 𖼢𖽜 , 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼅𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼽𖽺 𖼗𖽔 .
4 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽻𖾏 𖼀𖽱𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏.
5 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼾𖽨𖾏 𖼺𖽱𖾏 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽪𖾐 𖽐𖼊𖽻𖾏 𖽐𖼡𖽹𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼺𖽱𖾏 𖼐𖽻 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽪𖾐 𖼮𖽢𖾐 𖼐𖽻 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏 𖼘𖽪！
𖼟𖽡 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼊𖽪𖾑 𖼞𖽡 𖽐𖼷𖽻 𖼑𖽡𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏, 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳.
7 𖼞𖽡𖾑 𖽐𖼷𖽻 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
8 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼳𖽙𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼡𖽜𖾑 𖼑𖽻 ,
𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼡𖽻𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼡𖽻𖾐 𖼡𖽜𖾏.
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼀𖽱𖾏 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
𖼟𖽡 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽱𖾏 𖼯𖽷𖾏 𖼖𖽺𖾑?
10 𖼎𖽺𖾏 𖼀𖽱𖾏 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏, 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏,
𖼁𖽔 𖼋𖽡𖾏 𖼢𖽻 𖼀𖽱𖾏 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏，
11 𖼟𖽡 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼋𖽔𖾏 𖼇𖽺𖾏 𖼺𖽱𖾏 𖼸𖽜 ,
𖼟𖽡 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 .
𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏
12 𖼡𖽔𖾑 𖼯𖽱𖾏 𖼸𖽜 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑,
𖽃𖽔𖾏 𖼑𖽪𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ；
13 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼎𖽑𖽻𖾏 𖼊𖽠𖾏 𖼢𖽳 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼡𖽑𖽳𖾏,
14 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐,
𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 , 𖼙𖽳𖾐 𖼷𖽱𖾏 𖼑𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼐𖽱𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐, 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼞𖽑𖽙𖾏.
16 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽪𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼼𖽘𖾏,
𖼊𖽑𖽡𖽝𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼳𖽪𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼳𖽤𖾐 𖼼𖽘𖾏 ：
17 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏、𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑、
𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑、
18 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏、
𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼏𖽘 𖼇𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏、
19 𖼊𖽪𖾐 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝 𖼷𖽑𖽔𖾐 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼊𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼙𖽳𖾐 𖼷𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐
20 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏，
21 𖼽𖽔𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼡𖽑𖽹𖾐 𖼋𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏,
𖼚𖽑𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖼚𖽘𖾐 𖼮𖽡𖽝𖾏.
22 𖼟𖽡 𖼋𖽙𖾏 𖼞𖽡𖾑, 𖼋𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 ；
𖼟𖽡 𖼢𖽻 𖼀𖽱𖾏, 𖼋𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 ；
𖼟𖽡 𖼀𖽱𖾏 𖼯𖽷𖾏 𖼋𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼦𖽔𖾑 𖼗𖽔𖾏 .
23 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐, 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 (“𖼡𖽑𖽔𖾐 𖼇𖽺𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “ 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏”) 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑，
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑，
24 𖼸𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼀𖽜𖾑 𖼛𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼑𖽡𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑 .
