17 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽜 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏 𖼋𖽘𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 .” 18 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏 .” 19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏 . 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 , 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏, 𖼽𖽔𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼡𖽔 𖼋𖽱𖾏 , 𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼡𖽔 𖼋𖽱𖾏 .” 20 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼚𖽑𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼚𖽑𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 .” 21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖼑𖽔𖾏 𖼘𖽪 ! 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼟𖽡 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼨𖽑𖽪𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 . 22 𖼞𖽪𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽘𖾐 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼟𖽡 , 𖼋𖽙𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏. 23 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼙𖽠𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼊𖽡𖾑 𖼢𖽻𖾏 , 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼚𖽑𖽪𖾏 .”
