𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐 𖽐𖼊𖽠𖾑
𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐 𖼷𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏
1
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 、 𖼽𖽙 𖼊𖽑𖽗𖾐 、 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 、 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼀𖽺𖾏 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐 .
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽻𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖼷𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐, 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼅𖽔𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.” 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐 𖼅𖽻𖾏 𖼷𖽙𖾏 𖼊𖽡 𖼖𖽔𖾑 𖼽𖽡𖾑 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏𖼞𖽙𖾑 𖼄𖽡𖽝 . 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑,𖼑𖽜 𖼐𖽱𖾏 𖼳𖽻𖾏 𖽃𖽡𖾑 𖼖𖽜𖾐 𖼊𖽪𖾐. 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼊𖽡𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 ,𖽃𖽡𖾑 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼽𖽦𖾑 𖼦𖽪𖾏 𖼑𖽡𖾏 𖼾𖽡𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼊𖽪𖾏 𖼮𖽘𖾑. 5 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 𖼖𖽪𖾑 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾑 𖼐𖽜𖾑𖽑.”
𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏𖼘𖽙 · 𖼖𖽪𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐
6 𖼞𖽙𖾑 𖼄𖽡𖽝 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼳𖽻𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽑𖽹𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼘𖽙 · 𖼖𖽪𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 (𖼘𖽙 · 𖼖𖽪𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 𖼟𖽪𖾏 “ 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 ” ), 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 7 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼾𖽡𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏, 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽡𖽝𖾏 𖼊𖽪𖾑 𖼐𖽑𖽜𖾑、 𖽃𖽔𖾑 𖼏𖽦𖾏、 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏、 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽐𖼎𖽙 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏.”
𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏𖼘𖽙 · 𖽃𖽔𖾐 𖼄𖽡𖾐
8 𖼞𖽙𖾑 𖼄𖽡𖽝 𖼟𖽙𖾏𖼘𖽙 · 𖼖𖽪𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 𖼊𖽪𖾏 𖼄𖽡𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽤𖾑 𖼳𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐. 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼘𖽙 · 𖽃𖽔𖾐 𖼄𖽡𖾐 (𖼘𖽙 · 𖽃𖽔𖾐 𖼄𖽡𖾐𖼟𖽪𖾏 “𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏”), 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.”
𖼏𖽵 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏
10 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐, 𖽐𖼮𖽔𖾏 𖼋𖽘 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼾𖽱 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.𖽃𖽔 𖼊𖽑𖽻𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: 𖼅𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐. 11 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼦𖽡 𖼸𖽱, 𖼑𖽜 𖼾𖽨𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐, 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.”
