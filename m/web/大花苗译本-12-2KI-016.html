（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 28： 1 - 27）
16
𖼖𖽡𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼊𖽑𖽗𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 2 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳. 3 𖼎𖽺𖾏 𖼾𖽻 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏, 𖼎𖽺𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐, 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑， 4 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐, 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾐 𖼄𖽡𖾐 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏.
