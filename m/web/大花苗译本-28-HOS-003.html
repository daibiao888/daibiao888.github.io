𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐 𖼎𖽺𖾏 𖼷𖽙𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐
3
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼈𖽻𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏 𖽐𖼮𖽑𖽠𖾏, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼮𖽑𖽠𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼈𖽡𖽝𖾏, 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼳𖽡𖽝𖾐.” 2 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼟𖽻𖾏 𖼀𖽜𖾐 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖽝𖾏,𖽃𖽡𖾐 𖼦𖽙 𖼄𖽺 𖽃𖽳𖾐 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼅𖽙 , 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑 . 3 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏,𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏,𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼗𖽔 𖼐𖽤𖾐 , 𖼞𖽪𖾑 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼟𖽡 .” 4 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼾𖽨𖾏 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 ,𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏、 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐、 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑、 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼁𖽠𖾏、 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳、 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼁𖽠𖾏.
5 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼗𖽙𖾏 , 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 . 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏.
