（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 11：4 - 16）
23
𖼸𖽙 𖼚𖽔𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼊𖽑𖽱𖾏 𖼛𖽘𖾏 𖼊𖽑𖽱𖾏 𖼼𖽙 , 𖼨𖽑𖽪𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼘𖽙 𖼦𖽗𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾐 、 𖼽𖽙 𖼦𖽔𖾐 𖼐𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 、 𖽃𖽙 𖼀𖽡𖾐 𖼊𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾐 、 𖼽𖽔𖾏 𖼊𖽔𖾏 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼄𖽔𖾐 𖼳𖽡𖾑 𖼽𖽔𖾐 、 𖼳𖽡𖾏 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼇𖽔 𖼋𖽔𖾏 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 . 2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 , 𖼎𖽑𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼨𖽑𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 .
3 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 . 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼑𖽔𖾏 𖼘𖽪 ! 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑.” 4 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼡𖽑𖽙𖾑 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 , 𖼷𖽷𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽃𖽔 𖼛𖽻𖾏 , 5 𖼷𖽷𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 , 𖼷𖽷𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼮𖽷𖾐 𖽃𖽔 𖼛𖽻𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖽐𖼚𖽙 . 6 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼙𖽻 𖼋𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏 , 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 , 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼊𖽔𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 . 7 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 , 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼮𖽵 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼟𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏 . 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 , 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 .”
8 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼨𖽑𖽳𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽡𖾑 𖼋𖽔𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽙𖾏 𖼼𖽙 . 9 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾑 𖼄𖽑𖽪𖾑 、 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 、 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼮𖽡𖽝𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽡𖾑 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 . 10 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 , 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 , 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 . 11 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽳𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼄𖽔𖾐 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 , 𖼄𖽔𖾐 𖽐𖼊𖽠𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼐𖽱𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼉𖽻𖾏 𖼳𖽡𖽻𖾏 !”
𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼊𖽻𖾏 𖼊𖽔𖾏
12 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼏𖽘 𖽐𖼎𖽙 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼎𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏， 13 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 , 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏 𖼙𖽔 𖼚𖽙𖾏 , 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼯𖽡𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼢𖽜𖾏 𖽐𖼎𖽔 𖼨𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 . 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼎𖽺𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏 , 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏 ! 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏 !” 14 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼨𖽑𖽳𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏 , 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼄𖽔𖾐 𖼐𖽱𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽃𖽡𖾑 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 !” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽐𖼚𖽙 𖼊𖽔𖾏 𖼐𖽱𖾏 .”

𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏

16 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 . 17 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼀𖽔𖾑 𖼖𖽡 𖽐𖼞𖽔𖾏 𖼁𖽠 , 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖽐𖼞𖽔𖾏 𖼁𖽠 𖼺𖽘𖾏 , 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 𖽐𖼎𖽙 𖼁𖽠𖾏 𖼼𖽘𖾏 , 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼄𖽔𖾐 𖼀𖽔𖾑 𖼖𖽡 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼄𖽔𖾐 𖼊𖽑𖽗𖾐 𖼊𖽔𖾏 𖼺𖽘𖾏 . 18 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼷𖽷𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 , 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 . 𖼸𖽻𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 , 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 , 𖼙𖽔 𖼚𖽙𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 . 19 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖽃𖽔 𖼛𖽻𖾏 𖽃𖽡𖾑 𖼻𖽙𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼮𖽡𖽝𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽜 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼳𖽷𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏.
20 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 , 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼟𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 , 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖽃𖽔 𖼸𖽷 𖽃𖽔 𖼛𖽻𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 , 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐. 21 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 , 𖽃𖽡𖾐 𖼙𖽻 𖼙𖽱 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 . 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼊𖽔𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏 .
