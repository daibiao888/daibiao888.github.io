4
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐． 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼦𖽡 𖼸𖽱 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 ， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽡𖽠𖾑， 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑． 2 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐． 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖽃𖽡𖾑， 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： 3 “ 𖼅𖽡 𖼑𖽻， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼮𖽘𖾑，𖼐𖽱𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼀𖽳𖾐 𖼮𖽘𖾑． 4 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼷𖽷𖾏 𖼖𖽘𖾑 𖽃𖽡𖾑， 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑， 𖼐𖽻 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏． 5 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽡𖽝𖾑， 𖼋𖽔𖾏 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏 ，𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽡𖽝𖾑 ， 6 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏 ， 𖼮𖽑𖽘𖾑 𖼻𖽡𖽝𖾐 𖼎𖽺𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼙𖽠𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼗𖽔 𖼐𖽔𖾑． 7 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼞𖽙𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏， 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼘𖽑𖽙𖾐 𖼗𖽙𖾏 𖼐𖽹𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏，𖼐𖽱𖾏 𖼷𖽷𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐． 8 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼻𖽻𖾏 𖼊𖽡𖾐， 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼘𖽑𖽙𖾐 𖼋𖽘 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐， 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽠 ，𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖽐𖼀𖽠𖾏， 𖽃𖽡𖾐 𖼀𖽔𖾏 𖽐𖼀𖽠．” 9 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼑𖽻， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻！”
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏
（𖼊𖽑𖽹𖾏 13：10-17；𖼖𖽪𖾏 8：9-10）
10 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼾𖽨𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼋𖽘𖾏 𖽃𖽡𖾑， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 ？” 11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼮𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼊𖽔𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼡𖽑𖽔𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ，𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 ， 12 𖼟𖽙𖾏
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼖𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼯𖽡𖽙𖾏， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼁𖽙𖾏 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏，
𖼑𖽻 𖼖𖽔𖾏 𖼑𖽻 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 ，
𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙𖾏， 𖼾𖽔𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼋𖽘．”
（𖼊𖽑𖽹𖾏 13：18-23；𖼖𖽪𖾏 8：11-15）
13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 ，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼖𖽺𖾑？ 14 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼀𖽳𖾐 𖼮𖽘𖾑 𖼀𖽳𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏． 15 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼷𖽠𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 𖽃𖽡𖾑， 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻 𖼞𖽘𖾐 𖼞𖽡𖾏， 𖼺𖽔𖾐 𖼊𖽔 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼋𖽔𖾏 𖼄𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼦𖽔𖾏 𖼙𖽠𖾏． 16 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼈𖽳𖾐 𖼗𖽘𖾏， 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏，𖼎𖽺𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖽃𖽺𖾐 𖼙𖽔𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏， 17 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐， 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼊𖽔𖾐， 𖼸𖽙 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽘𖾐 𖼞𖽡𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏， 𖽂𖽪𖾏 𖼢𖽻 𖼙𖽠𖾏 𖼋𖽘． 18 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼷𖽠𖾏 𖼞𖽙𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 ， 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻 𖼞𖽘𖾐 𖼞𖽡𖾏， 19 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏 、𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑 𖼮𖽑𖽵𖾑 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽐𖼀𖽜𖾏 𖼺𖽡𖽝𖾐 𖼁𖽺𖾏 𖼋𖽔𖾏， 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼐𖽹𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖼷𖽷𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐． 20 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽃𖽡𖾑， 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼎𖽺𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏， 𖼎𖽺𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐， 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽠𖾏、 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖽐𖼀𖽠、 𖽃𖽡𖾐 𖼀𖽔𖾏 𖽐𖼀𖽠．”
（𖼖𖽪𖾏 8：16-18）
21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼋𖽔𖾏， 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼡𖽘𖾐 𖼊𖽱𖾏 、 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼸𖽱𖾏 𖼡𖽘𖾐 𖼸𖽘𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼖𖽔𖾏？ 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼷𖽜𖾏 𖼖𖽺𖾑 ？ 22 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼮𖽡𖽝𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏． 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏， 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏． 23 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼑𖽻，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻！
24 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼐𖽙𖾑𖽑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏． 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼙𖽪𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼙𖽪𖾏 𖼎𖽻𖾏 𖼅𖽡， 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼺𖽷𖾐． 25 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏， 𖼽𖽔𖾑 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏； 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼦𖽔𖾏 𖼙𖽠𖾏．”
𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽳𖾐 𖼮𖽘𖾑 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐
26 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙， 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼮𖽘𖾑 𖼀𖽳𖾐 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐， 27 𖼐𖽱𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 ， 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏， 𖼋𖽡𖾏 𖼮𖽘𖾑 𖼡𖽻𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼟𖽻𖾏 𖼘𖽑𖽘𖾑 𖼘𖽑𖽙𖾐，𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑． 28 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼾𖽨 𖼋𖽠𖾏 𖼡𖽻𖾐，𖼺𖽪𖾐 𖽃𖽔𖾏 𖼽𖽨𖾐 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼷𖽑𖽠𖾑 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐， 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐 𖼀𖽪𖾑 𖽃𖽔 𖼛𖽡𖾏 𖼟𖽪𖾏 𖼢𖽻𖾏． 29 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽻𖾐 𖼳𖽦𖾑 𖼋𖽘𖾏， 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼖𖽡𖽝𖾐 𖼅𖽻𖾏 𖼤𖽜𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．”
（𖼊𖽑𖽹𖾏 13：31-32；𖼖𖽪𖾏 13：18-19）
30 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？ 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼗𖽙𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼖𖽺𖾑？ 31 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽘𖾑 𖼻𖽻𖾐 𖼛𖽙𖾐 𖽃𖽔 𖼛𖽡𖾏， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐， 𖼐𖽱𖾏 𖼳𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼮𖽘𖾑 𖼡𖽻𖾐 ． 32 𖼈𖽡𖽝𖾏 𖼀𖽳𖾐 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼻𖽻𖾐， 𖼘𖽑𖽘𖾑 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐， 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏 𖼳𖽻𖾏 𖼉𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡．”
（𖼊𖽑𖽹𖾏 13：34）
33 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻 𖼊𖽻𖾏， 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐； 34 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏． 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽡𖽙𖾐， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼚𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏．
𖼮𖽢𖾏 𖽐𖼎𖽙 𖼳𖽡𖽠𖾑 𖼊𖽪𖾏 𖼑𖽻 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 8：23-27；𖼖𖽪𖾏 8：22-25）
35 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼚𖽔𖾏 𖼀𖽑𖽘𖾑 𖼀𖽡 𖼋𖽡𖾏．” 36 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽐𖼞𖽙𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼎𖽺𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏 𖼨𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼞𖽙𖾏 𖽐𖼎𖽙 𖼅𖽻𖾏． 37 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼮𖽢𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼋𖽔𖾏， 𖽃𖽻𖾐 𖼟𖽡𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖽐𖼊𖽙𖾏 𖼗𖽙𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽙𖾏， 𖽐𖼞𖽙𖾏 𖼖𖽔𖾑 𖼀𖽪𖾑 𖽃𖽻𖾐 𖼋𖽘𖾏． 38 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾑 𖽐𖼷𖽘𖾏 𖽐𖼞𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼮𖽡𖽻𖾏 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏． 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 ，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼛𖽙𖾏 𖼳𖽦𖾑 𖼋𖽘𖾏， 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼀𖽡𖾐 𖼘𖽪 ？” 39 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼯𖽷𖾏 𖼗𖽙𖾏 𖼖𖽔𖾏 𖼮𖽢𖾏 𖽐𖼎𖽙 𖼳𖽡𖽠𖾑 𖼐𖽑𖽡𖾏：“ 𖽐𖼮𖽦 𖼊𖽘𖾏 𖼦𖽡𖾑！𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼦𖽡𖾑！” 𖼮𖽢𖾏 𖼎𖽺𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏，𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝． 40 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼮𖽜 𖼖𖽺𖾑？” 41 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝𖾏， 𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽔𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽢𖾏 𖽐𖼎𖽙 𖼳𖽡𖽠𖾑 𖼖𖽔𖾏 𖼑𖽻 𖼐𖽱𖾏？”
