𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼞𖽑𖽙𖾐 1：12-13；𖼖𖽪𖾏 4：1-13）
4
𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼳𖽟𖾏 𖼖𖽡𖽾 𖼨𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏，𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼐𖽱𖾏． 2 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼄𖽑𖽙𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑， 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽑𖽹𖾐 𖼋𖽘． 3 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐， 𖼟𖽙𖾏 𖼋𖽡𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼳𖽡𖽝𖾐．” 4 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼺𖽻𖾏：
‘ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏， 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼊𖽔𖾐，
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼷𖽑𖽔𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼉𖽺𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏．’ ”
5 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖽃𖽔𖾏 𖼎𖽪𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏． 6 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼚𖽑𖽡𖽝𖾏 𖼘𖽻𖾑𖽑 𖼅𖽻𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：
‘ 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡，
𖼷𖽹𖾏 𖼟𖽡 𖼎𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏，
𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼟𖽡 𖽐𖼊𖽱𖾑𖽑 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐．’ ”
7 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼺𖽻𖾏 𖼺𖽷𖾐：‘ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ．’ ” 8 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼑𖽔𖾏． 9 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼮𖽔𖾑 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼞𖽪𖾑， 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽻𖾏 𖼟𖽡．” 10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼺𖽔𖾐 𖼊𖽔，𖼟𖽡 𖼀𖽜𖾑 𖼙𖽠𖾏！ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：
‘ 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，
𖼯𖽔 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐．’ ”
11 𖽃𖽔𖾏 𖼐𖽡𖾑， 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾 𖼙𖽻
（𖼞𖽑𖽙𖾐 1：14-15；𖼖𖽪𖾏 4：14-15）
12 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏𖼽𖽙 𖼦𖽗 𖼊𖽻𖾏 𖼄𖽔𖾐 𖼙𖽠𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼙𖽠𖾏． 13 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼺𖽘𖾏， 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾 𖼙𖽻 𖼙𖽠𖾏，𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑， 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏． 14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏：
15 “ 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖽐𖼎𖽙𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，
𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 ， 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，
𖼟𖽪𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖽙𖾐．
16 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏
𖼑𖽔𖾏 𖼁𖽙𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，
𖼅𖽔𖾏 𖼮𖽡𖾏 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔．”
17 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼳𖽠𖾑 𖼁𖽱𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏， 𖼐𖽑𖽡𖾏：“𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏， 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼚𖽻𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏
（𖼞𖽑𖽙𖾐 1：16-20；𖼖𖽪𖾏 5：1-11）
18 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑， 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏， 𖼳𖽡𖾑 𖼄𖽞 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼊𖽳 𖼖𖽡𖽝 𖼗𖽔𖾏 𖼸𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼾𖽠𖾏 𖽐𖼀𖽳𖾏． 19 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼾𖽠𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼀𖽳𖾏．” 20 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏， 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏． 21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏， 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼊𖽑𖽡𖽝𖾐， 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼊𖽑𖽹𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼦𖽗， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼊𖽑𖽹𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖽐𖼷𖽷𖾑 𖼖𖽪𖾑 𖼸𖽙𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 22 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼇𖽳𖾑 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖼺𖽘𖾏， 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼞𖽑𖽙𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐
（𖼖𖽪𖾏 6：17-19）
23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑， 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼄𖽻𖾏 𖼻𖽻𖾏， 𖼞𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐． 24 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖽐𖼊𖽙𖾏 𖽐𖼷𖽷𖾏 𖼚𖽔𖾏 𖼳𖽱𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼸𖽻𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼄𖽙𖾐、 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏、 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼊𖽑𖽡 𖼗𖽻𖾏 、 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 25 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏、 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼀𖽙𖾑 𖼖𖽡𖾏、𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐、 𖼽𖽨 𖼊𖽑𖽹𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼋𖽔𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐．
