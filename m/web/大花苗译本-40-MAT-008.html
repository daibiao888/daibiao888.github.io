𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏
（𖼞𖽑𖽙𖾐 1：40 - 45；𖼖𖽪𖾏 5：12-16）
8
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏，𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐． 2 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼄𖽙𖾐 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏，𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽪𖾏， 𖼟𖽡 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏．” 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏，𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼯𖽪𖾏 𖼯𖽡𖽙𖾏，𖼟𖽡 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏！” 𖼐𖽱𖾏 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏． 4 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼡𖽔 𖼋𖽱𖾏，𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 ，𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏，𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏．”
𖼘𖽙 𖼄𖽔𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏
（𖼖𖽪𖾏 7：1-10）
5 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾 𖼙𖽻， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽑𖽡𖾏： 6 “ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑 𖼄𖽙𖾐 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑 𖼀𖽱𖾏 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏，𖽃𖽦𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝．” 7 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼞𖽑𖽙𖾏 𖼐𖽱𖾏．” 8 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 ：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 ，𖼯𖽔 𖼽𖽔𖾑 𖼟𖽡 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏，𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏． 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽵𖾏 𖼞𖽪𖾑， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽵𖾏， 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏：‘ 𖼅𖽻𖾏 ’！𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏： ‘ 𖼋𖽔𖾏！’ 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏 ： ‘ 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑！’ 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 ．” 10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏 𖼎𖽺𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑，𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 ： “ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑． 11 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐，𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼋𖽔𖾏， 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐、 𖼽𖽡𖾑 𖼺𖽔𖾐、 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽡𖾐 𖼸𖽻 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏， 12 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾑 ， 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼻𖽻𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 ， 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼟𖽡𖾏 𖼋𖽙𖾏 𖼐𖽡𖽝𖾑𖽑 𖼷𖽷𖾏 𖼞𖽹𖾑．” 13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼦𖽡𖾑！𖼟𖽡 𖽐𖼮𖽜 𖼖𖽡𖾏 𖼯𖽤𖾏， 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼎𖽻𖾏 𖼟𖽡．” 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼀𖽡𖾐 𖼊𖽳 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽹𖾏
（𖼞𖽑𖽙𖾐 1：29-34；𖼖𖽪𖾏 4：38-41）
14 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼞𖽔𖾏，𖼐𖽱𖾏 𖼁𖽙𖾏 𖼀𖽡𖾐 𖼊𖽳 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽹𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼞𖽪𖾐 𖼘𖽑𖽳𖾐 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏 ． 15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏，𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾐 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏，𖽃𖽔 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑． 16 𖼸𖽙 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼋𖽘 ， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏 𖽃𖽡𖾑 ，𖼋𖽔𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼐𖽱𖾏 𖼮𖽔 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼊𖽔𖾐，𖼎𖽺𖾏 𖼖𖽠𖾑 𖼖𖽡𖽾 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏，𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐． 17 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏：“𖼐𖽱𖾏 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐 𖼞𖽡𖾑 𖼯𖽻 𖼞𖽡𖾑 𖼚𖽑𖽔𖾏，𖼞𖽱𖾑 𖼎𖽷𖾏 𖼺𖽘𖾏 𖼀𖽡𖾐 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐．”
𖼸𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐
（𖼖𖽪𖾏 9：57-62）
18 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽻𖾏 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼀𖽡 𖼋𖽡𖾏． 19 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼋𖽔𖾏，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽡 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐．” 20 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽡𖾑 𖼋𖽱𖾏 𖼅𖽔𖾏 𖼡𖽑𖽙𖾑，𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼉𖽳𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖽐𖼮𖽡𖽻𖾏．” 21 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪 ！ 𖼯𖽪𖾏 𖼞𖽪𖾑 𖼺𖽪𖾐 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼊𖽱𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏．” 22 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐！𖼯𖽔𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼅𖽻𖾏 𖼊𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏．”
𖼮𖽢𖾏 𖽐𖼎𖽙 𖼳𖽡𖽠𖾑 𖼊𖽪𖾏 𖼑𖽻 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼞𖽑𖽙𖾐 4：35-41；𖼖𖽪𖾏 8：22-25）
23 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐． 24 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼮𖽢𖾏 𖼘𖽑𖽙𖾐，𖽃𖽻𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼖𖽔𖾑 𖼀𖽪𖾑 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏，𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽻𖾏 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏． 25 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖼮𖽘𖾏 𖼐𖽱𖾏 𖼯𖽷𖾏 𖼗𖽙𖾏，𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼮𖽨𖾏 𖼀𖽡𖾐， 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 𖼋𖽘𖾏！” 26 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼯𖽠 𖼐𖽡𖾑 𖼘𖽪！𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽡 𖽐𖼮𖽑𖽹𖾏 𖼖𖽺𖾑？” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 ， 𖼖𖽔𖾏 𖼮𖽢𖾏 𖽐𖼎𖽙 𖼳𖽡𖽠𖾑，𖼮𖽢𖾏 𖽐𖼎𖽙 𖽃𖽻𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼎𖽺𖾏 𖼞𖽑𖽔𖾏 𖽐𖼮𖽦 𖼙𖽠𖾏 𖼋𖽘𖾏． 27 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽘𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏？ 𖼮𖽢𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼖𖽔𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼗𖽪𖾏．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐
（𖼞𖽑𖽙𖾐 5：1-20；𖼖𖽪𖾏 8：26-39）
28 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽡𖾏 𖼳𖽡𖽠𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏𖼮𖽢𖾑 𖼊𖽔 𖼖𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏， 𖼎𖽺𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼷𖽘𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼸𖽱𖾏 𖼐𖽱𖾏，𖽃𖽺𖾐 𖽐𖼚𖽪𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼎𖽺𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽔𖾐 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖽃𖽡𖾑． 29 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 ： “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 ，𖼀𖽡𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼺𖽷𖾐，𖼟𖽡 𖼋𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼘𖽪𖾏？” 30 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼀𖽙𖾐 𖽐𖼀𖽔𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼑𖽙 𖼻𖽻𖾐． 31 𖽐𖼀𖽙𖾏 𖼀𖽡𖾐 𖼚𖽘𖾐 𖽃𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼀𖽡𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏 ， 𖼎𖽺𖾏 𖼯𖽪𖾏 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼀𖽙𖾏 𖽐𖼀𖽔𖾏 𖼉𖽺𖾏！” 32 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽻𖾏！” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼀𖽙𖾏 𖽐𖼀𖽔𖾏．𖽐𖼀𖽙𖾏 𖽐𖼀𖽔𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼏𖽘 𖽐𖼮𖽵 𖼘𖽻𖾑𖽑 𖽃𖽔𖾑 𖼷𖽔𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼷𖽑𖽪𖾐 𖼋𖽔 𖼺𖽘𖾏． 33 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼾𖽨 𖽐𖼀𖽔𖾏 𖼎𖽺𖾏 𖼏𖽘 𖼀𖽜𖾑 ，𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜， 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽷𖾐，𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． 34 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜， 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼙𖽻 𖼋𖽔𖾏 𖼸𖽱𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼁𖽙𖾏 𖼐𖽱𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖽐𖼊𖽜 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏．
