𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑，𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽢𖾑 𖼐𖽔𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽙𖾏 𖼐𖽤𖾐，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 2 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖼅𖽻𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏． 3 𖼮𖽡𖽠𖾑 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖼋𖽘𖾏 𖼋𖽘𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽡𖽠𖾑 𖼋𖽘𖾏．” 4 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔 𖼁𖽙𖾏，𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽔𖾏 𖼋𖽪 𖼞𖽳𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼘𖽪？𖼞𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼮𖽡𖽻𖾏 𖼺𖽷𖾐．” 5 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼡𖽑𖽔𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏．” 6 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾑 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏，𖼅𖽔𖾏 𖼚𖽻𖾏 𖼗𖽪𖾏 𖼚𖽘𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽻𖾐 𖼊𖽜𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖽃𖽡𖾐 𖼗𖽪𖾏 𖽐𖼊𖽡𖾏 𖼋𖽻𖾏 𖽃𖽔𖾐 𖼷𖽷𖾐 𖼊𖽑𖽻𖾐 𖽃𖽻𖾐． 7 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼘𖽡𖾑𖽑 𖽃𖽻𖾐 𖼟𖽙𖾏 𖼀𖽪𖾑 𖼚𖽘𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐．” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼘𖽡𖾑𖽑 𖽃𖽻𖾐 𖼟𖽙𖾏 𖼀𖽪𖾑． 8 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼦𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼡𖽜𖾑．” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼡𖽜𖾑． 9 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼡𖽜𖾑 𖼐𖽑𖽡𖽝𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼚𖽑𖽡𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼮𖽡𖽠𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏，𖼮𖽔𖾑 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼡𖽑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽺𖾏 𖽃𖽻𖾐 𖽃𖽡𖾑 𖼀𖽻𖾐 𖼊𖽔𖾐．𖼊𖽪𖾐 𖼯𖽵𖾏 𖼡𖽜𖾑 𖼨𖽑𖽪𖾏 𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖼋𖽔𖾏， 10 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽪𖾐 𖼊𖽜𖾐 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏，𖼋𖽙𖾏 𖼟𖽙𖾏 𖼡𖽑𖽔𖾏 𖼦𖽻𖾏 𖼷𖽻𖾏，𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼊𖽜𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏，𖼟𖽡 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑！” 11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽢𖾑 𖼐𖽔 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑，𖼟𖽪𖾏 𖼺𖽪𖾐 𖼁𖽱 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑，𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼗𖽙𖾏，𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼅𖽡 𖼎𖽺𖾏 𖽐𖼮𖽜 𖼐𖽱𖾏． 12 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 、𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 、 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼅𖽡 𖼘𖽻𖾑𖽑 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾，𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼀𖽡𖾑 𖼯𖽠 𖼐𖽑𖽪𖾐．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏
（𖼊𖽑𖽹𖾏 21：12-13；𖼞𖽑𖽙𖾐 11：15-17；𖼖𖽪𖾏 19：45-46）
13 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐． 14 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽱𖾏 𖼁𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽺 𖼑𖽨𖾏 、 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖽐𖼎𖽙 𖽐𖼡𖽔𖾐，𖼎𖽺𖾏 𖼁𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽪𖾑𖽑 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 15 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼖𖽑𖽔𖾏 𖼡𖽠𖾑 𖽃𖽔𖾏 𖽐𖼚𖽠𖾏，𖼖𖽠𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖽐𖼎𖽙 𖼑𖽨𖾏 𖼟𖽙𖾏 𖼋𖽠𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼙𖽠𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼘𖽡𖾑𖽑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽪𖾑𖽑 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼊𖽑𖽠𖾐 𖽐𖼷𖽜𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡 𖼏𖽻𖾏． 16 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽺 𖽐𖼡𖽔𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“ 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏！𖼞𖽪𖾑 𖼉𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏！” 17 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼅𖽡 𖼎𖽺𖾏 𖽐𖼊𖽱𖾏 𖼸𖽙 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖽐𖼞𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 ．” 18 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑，𖼋𖽡𖽝𖾏 𖼟𖽡 𖼸𖽠𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼑𖽔𖾏 𖼖𖽺𖾑？” 19 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑，𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖽐𖼚𖽙 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏．” 20 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼷𖽑𖽙𖾏 𖼊𖽡𖽝𖾑，𖼋𖽡𖽝𖾏 𖼟𖽡 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖽐𖼚𖽙 𖼊𖽔𖾐 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏？” 21 𖼈𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏，𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑． 22 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙，𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼊𖽱𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽜 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖽐𖼎𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏．
23 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼮𖽜 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑． 24 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼾𖽨𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐． 25 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼽𖽔𖾑 𖼡𖽔 𖼋𖽱𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏．
