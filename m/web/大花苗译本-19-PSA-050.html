𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼘𖽑𖽙𖾐, 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼋𖽘 ,
𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏.
2 𖼎𖽑𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,

3 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏．
𖼅𖽔𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 ,
𖼅𖽔𖾏 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼷𖽑𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐.
4 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏,
5 𖼐𖽑𖽡𖾏: “𖼦𖽡 𖼸𖽱 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑,
𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖽃𖽡𖾑.”
6 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽃𖽡𖾑. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
7 “𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪 ！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏，
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼳𖽹𖾐 𖼑𖽪 𖼟𖽡 ,
𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
8 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽜 𖼟𖽡 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼖𖽔𖾏 𖼟𖽡 ,
𖼟𖽡 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
9 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼯𖽦𖾐 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖽃𖽔 𖼁𖽱 𖼮𖽑𖽷𖾏.
10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼡𖽔𖾏 𖼀𖽔𖾏 𖼮𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝,
𖼷𖽑𖽡𖽝𖾐 𖼉𖽻𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼑𖽡𖾏 𖼯𖽦𖾏 𖼖𖽔𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝.
11 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼞𖽪𖾑 𖼞𖽑𖽔𖾏 𖼀𖽻𖾐,
𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼖𖽔𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝 .
 
12 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖽃𖽔𖾑 𖼚𖽘𖾐, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝 .
13 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼖𖽔𖾑 𖼑𖽻𖾏 𖼊𖽪𖾑 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼖𖽔𖾑 𖼦𖽻𖾏 𖼊𖽪𖾑 𖽃𖽔 𖼁𖽱 𖼮𖽑𖽷𖾏 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑 𖼖𖽺𖾑?
14 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼁𖽻𖾏 𖼟𖽡 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼏𖽵 𖼟𖽪𖾏 𖼟𖽡 𖼏𖽵 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑.
15 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼞𖽪𖾑,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼟𖽡 , 𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼞𖽪𖾑.”
𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏
16 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼞𖽔𖾐 𖼾𖽻 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏,
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼖𖽺𖾑?
17 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 ,
𖼄𖽔𖾐 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼺𖽘𖾏.
18 𖼟𖽡 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼸𖽜, 𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔 ,
𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻𖾏 𖼅𖽔𖾏 𖽐𖼀𖽠 .
 
19 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼟𖽡 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼚𖽺𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑．
20 𖼟𖽡 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏,
𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 𖼾𖽨𖾏 𖼑𖽡𖽝𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐.
21 𖼟𖽡 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 , 𖼞𖽪𖾑 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼺𖽷𖾐,
𖼟𖽡 𖽐𖼊𖽱𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐,
𖽃𖽡𖾑 𖼉𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼖𖽔𖾏 𖼟𖽡 , 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
 
22 𖼅𖽡 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 ,
