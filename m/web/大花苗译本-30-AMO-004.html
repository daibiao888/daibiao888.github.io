𖼅𖽡 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼺𖽱𖾏 𖼀𖽔𖾑 𖼳𖽔𖾑 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽨𖾏 𖽃𖽡𖾑 𖼘𖽪！
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑.
𖼅𖽡 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏, 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼊𖽪𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏,
𖽐𖼎𖽙 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖽠𖾑 𖼗𖽙𖾏 , 𖼀𖽡𖾐 𖼦𖽻𖾏 !”
2 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼏𖽵 𖼗𖽪𖾏 , 𖼐𖽑𖽡𖾏:
“𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖼄𖽔𖾐 𖼅𖽡 𖽐𖼡𖽜𖾏 𖼙𖽠𖾏,
𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖽐𖼀𖽳𖾏 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖽐𖼡𖽜𖾏 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼷𖽷𖾐.
3 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼋𖽠𖾏 𖼗𖽠𖾏 ,
𖼊𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼦𖽔𖾐 𖼄𖽞 .
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼦𖽡𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻
4 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪 ! 𖼯𖽔𖾏 𖼅𖽡 𖼚𖽔𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼸𖽙 𖼮𖽡 𖼮𖽢 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼖𖽑𖽘𖾑 𖼅𖽡 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 ,
𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼖𖽑𖽘𖾑 𖼅𖽡 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠.
5 𖼯𖽔𖾏 𖼅𖽡 𖼖𖽑𖽘𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏, 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼖𖽑𖽘𖾑 ,
𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼅𖽡 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼐𖽙𖾑𖽑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖼯𖽡𖽙𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝
6 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼚𖽙 𖼙𖽻 𖼊𖽪𖾑 𖼐𖽡𖽝𖾑𖽑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼐𖽡𖽙𖾐 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠 .
𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑．
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
7 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖽃𖽡𖾑 𖼷𖽷𖾐 𖼖𖽑𖽡𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏，
𖼞𖽪𖾑 𖼟𖽙𖾏 𖼑𖽘 𖼦𖽡𖾐 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼟𖽙𖾏 𖼑𖽘 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 ,
𖼦𖽡𖾐 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑.
𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼑𖽘，
𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽘 , 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽘 𖽃𖽡𖾑 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼺𖽘𖾏 𖼋𖽘𖾏.
8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼷𖽷𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼦𖽡 𖼸𖽱 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽻𖾐,
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 𖼦𖽻𖾏.
𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑．
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
9 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼮𖽢𖾏 𖽐𖼞𖽡、 𖼗𖽱𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼮𖽑𖽻𖾏 𖼋𖽔𖾏 𖼟𖽡𖾏 𖼅𖽡,
𖼅𖽡 𖼖𖽪𖾑 𖽐𖼚𖽙 𖼉𖽘𖾏 𖼻𖽻𖾐 𖼷𖽷𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏、 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐、 𖼞𖽗𖾐 𖼖𖽗 𖽐𖼊𖽻𖾏,
𖼞𖽑𖽔𖾏 𖼏𖽻 𖼷𖽷𖾑 𖼷𖽷 𖼑𖽻𖾏 𖼺𖽘𖾏.
𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑．
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
10 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐,
𖼟𖽙𖾏 𖼅𖽡 𖼑𖽜𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏,
𖼅𖽡 𖼖𖽪𖾑 𖼾𖽡 𖽐𖼚𖽙 𖼡𖽻𖾑 𖼋𖽔 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼗𖽱𖾏 𖼷𖽑𖽻𖾐 𖽃𖽔𖾑 𖽐𖼀𖽱 .
𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑．
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
11 𖼞𖽪𖾑 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼅𖽡 ,
𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 、 𖽃𖽳 𖼄𖽙 𖼖𖽔𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼟𖽙𖾏 𖼅𖽡 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑．
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ! 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼟𖽡 .
