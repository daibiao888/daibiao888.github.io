𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼑𖽡𖾏 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖼋𖽔𖾏,
𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐,
𖼚𖽡𖽝𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐,
𖼅𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑?
 
“ 𖼟𖽪𖾏 𖼞𖽪𖾑 ,
𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼎𖽑𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏.”
 
2 𖼟𖽡 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 ?
𖼟𖽡 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼮𖽡𖽠𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼮𖽑𖽙𖾏 𖼖𖽺𖾑?
 
3 “𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼯𖽪𖾏 𖼮𖽡𖽠𖾑 ,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐．
𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽪𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽪𖾏 𖼺𖽘𖾏．
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 𖼀𖽹𖾑 𖼗𖽙𖾏 𖼘𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏 .
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐,
𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏.
5 𖼞𖽪𖾑 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽘𖾏,
𖼞𖽪𖾑 𖼟𖽡𖾏 𖼏𖽘𖾏 𖽐𖼎𖽘 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏．
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 𖽐𖼀𖽘𖾏,
𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑.
6 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼯𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼢𖽻 𖼮𖽡𖽠𖾑 , 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.”
𖼊𖽻𖾏 𖼅𖽳𖾏 𖼋𖽘𖾏 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
7 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼎𖽑𖽻𖾏 𖼀𖽡𖾐 𖽃𖽡𖾑,
𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼙𖽔 ,
𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽔𖾏,
𖼋𖽡𖾏 𖼅𖽳𖾏 𖼐𖽡𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼎𖽙 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
8 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽙 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼟𖽪𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑.”
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽨𖾏 𖼮𖽵𖾑 .
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽐𖼚𖽙 ,
𖼐𖽱𖾏 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽡𖽙𖾐 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼁𖽔 𖼡𖽑𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏.
 
10 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏, 𖼟𖽙𖾏 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼊𖽳𖾏．
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
11 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏,
𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏, 𖼐𖽑𖽡𖾏:
“𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼾𖽨 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼮𖽑𖽷𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼚𖽙 𖽐𖼮𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
12 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖽙𖾐 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾐 𖽃𖽔𖾏 𖼼𖽙 ,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽻𖾐 𖼇𖽺𖾐 𖼺𖽘𖾏,
𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
13 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼊𖽙𖾐,
