5
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏,
𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏, 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼑𖽡𖾏 𖼑𖽪𖾏.
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼛𖽙.
2 𖼐𖽱𖾏 𖽐𖼮𖽡𖽠𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏, 𖼺𖽹𖾑 𖽃𖽔𖾑 𖼈𖽳𖾑 𖼞𖽑𖽡𖽠𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏,
𖼯𖽡𖽙 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽑𖽙𖾏 𖼷𖽱𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼊𖽑𖽘𖾐,
𖼎𖽺𖾏 𖼸𖽻 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼮𖽡𖽠𖾑,
𖼋𖽙𖾏 𖼑𖽔𖾏 𖼷𖽷𖾏 𖼻𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏,
𖼈𖽡𖽝𖾏 𖼎𖽙𖾑 𖼷𖽷𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏.
 
3 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪!
𖼷𖽑𖽜𖾑 𖼅𖽡 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼇𖽺𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼾𖽙 𖽐𖼎𖽙 𖼦𖽡𖾐 𖼽𖽙.
4 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼺𖽘𖾏,
𖼋𖽡𖽝𖾏 𖼡𖽑𖽔𖾑 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼻𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏.
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼎𖽙𖾑 𖼷𖽷𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼖𖽺𖾑?
 
5 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏.
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼞𖽑𖽻𖾐 𖼞𖽡𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼺𖽙𖾑 𖼉𖽘𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏,
𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼟𖽻𖾏 𖼮𖽑𖽤 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼯𖽪𖾏.
6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼨𖽑𖽢𖾏 𖼷𖽑𖽳𖾏,
𖼦𖽡𖾐 𖼎𖽺𖾏 𖼮𖽑𖽪𖾑 𖼨𖽑𖽢𖾏, 𖼞𖽡𖾑 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼖𖽑𖽘𖾑 𖼘𖽑𖽙𖾐,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼛𖽻𖾏 𖽃𖽻𖾐 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼗𖽙𖾏 𖼑𖽘 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏.
 
7 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏, 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏,
𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
𖼐𖽱𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏,
𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖼎𖽙𖾑 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑,
𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖼎𖽙𖾑 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐.
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏
8 𖽃𖽺𖾐! 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽡𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏,
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽡𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏,
𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏,
𖼅𖽡 𖼮𖽔 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
9 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼐𖽙𖾑𖽑 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼺𖽘𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐.
10 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖼟𖽻𖾏 𖼸𖽱𖾏 𖼑𖽨𖾏 𖼯𖽡𖽙𖾏, 𖼮𖽔 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼊𖽑𖽳 𖼮𖽡𖽠𖾑 𖼊𖽔𖾐,
𖽃𖽡𖾐 𖼦𖽙 𖼄𖽺 𖽃𖽳𖾐 𖼮𖽘𖾑 𖼡𖽻𖾐 𖼮𖽔𖾑 𖼳𖽻𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼽𖽡𖾑 𖼇𖽔 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼡𖽻𖾐 𖼊𖽔𖾐.”
 
11 𖽃𖽺𖾐! 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐,
𖼑𖽜 𖼮𖽡𖽠𖾑 𖼊𖽻𖾏 𖼮𖽡𖽠𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼇𖽳𖾑 𖼸𖽙 𖼋𖽘 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏!
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏
𖽐𖼎𖽷𖾑 𖼢𖽜𖾏、 𖼷𖽑𖽙𖾑 𖼢𖽜𖾏、 𖽐𖼊𖽻𖾏 𖽐𖼎𖽔𖾏、 𖼷𖽑𖽙𖾑 𖼖𖽡𖾑 𖼖𖽪𖾏、 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑,
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
 
13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼎𖽺𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝.
14 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽳𖾏 𖼇𖽘𖾑 𖼐𖽱𖾏 𖼡𖽔𖾏 𖽐𖼎𖽦𖾑,
𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼮𖽔𖾏,
