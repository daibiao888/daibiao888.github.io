25
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼳𖽙𖾏 . 3 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾏 𖼨𖽳𖾑𖽑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 , 𖼳𖽻𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 . 4 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼳𖽙𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼳𖽙𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽢𖾏 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 . 5 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼮𖽡𖽝𖾏 𖼐𖽔𖾑 𖼾𖽨𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼤𖽜𖾏 , 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽢𖾏 , 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽡𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 . 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼐𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 . 6 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼋𖽠𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼡𖽑𖽜𖾑、𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑、𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏、𖼊𖽑𖽡𖽝𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 . 7 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼐𖽡𖾑 𖼡𖽻𖾐 𖼖𖽻𖾏 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 .”
𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏
