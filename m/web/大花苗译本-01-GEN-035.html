4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼁𖽠𖾏 𖽐𖼎𖽙 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 ；𖼽𖽔𖾐 𖼞𖽙 𖼞𖽑𖽡𖽠𖾏 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼇𖽘𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼺𖽘𖾏.
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼙𖽠𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 𖽃𖽡𖾑 𖽐𖼚𖽙 𖼙𖽻 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼅𖽡 𖼋𖽘𖾏. 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽡𖽙𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼖𖽪𖾏 𖼺𖽷𖾑, 𖼟𖽪𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏. 7 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘, 𖼊𖽡𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼖𖽳 · 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 (“𖼽𖽡𖾑 𖼖𖽳 · 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏” 𖼟𖽪𖾏 “𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏”), 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏. 8 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼄𖽡𖾏 𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 𖼋𖽔 𖼙𖽠𖾏, 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽡𖽝𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽙𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 · 𖼀𖽔𖾑 𖼞𖽪𖾐.
