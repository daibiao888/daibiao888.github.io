𖼻𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏， 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼊𖽪𖾏 𖼊𖽪𖽝𖾏 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
62
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏, 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽔𖾐,
𖼞𖽪𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏.
2 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼮𖽵𖾑,
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
 
3 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼖𖽪𖾑 𖼮𖽑𖽤 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐, 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼢𖽻, 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑？
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑 𖼸𖽻 𖼦𖽡𖾏 𖼊𖽳𖾏, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼊𖽑𖽠𖾐 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼏𖽵 𖼅𖽳𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
5 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏, 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽔𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏.
6 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼮𖽵𖾑,
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼼𖽙.
7 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼮𖽨𖾏 𖽐𖼀𖽘𖾏, 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏, 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
 
8 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼻𖽡𖽝𖾏 𖼐𖽱𖾏,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼮𖽢𖾑𖽑 𖼋𖽠𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔. （𖼳𖽡𖾏 𖼖𖽔𖾑）
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼳𖽷𖾐 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐,
𖼮𖽙𖾏 𖼎𖽻𖾏 𖼸𖽱𖾏 𖼞𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼳𖽷𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽷𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼀𖽘𖾏 𖼮𖽢𖾏.
10 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽔 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏,
𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼮𖽔𖾑 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐、𖽃𖽔𖾐 𖼟𖽻𖾐, 𖼞𖽪𖾑 𖼞𖽑𖽔𖾏 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏,
𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
12 𖼮𖽵𖾑 𖼘𖽪！ 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼟𖽡,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏.
