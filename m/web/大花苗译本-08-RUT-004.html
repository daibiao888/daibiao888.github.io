𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼷𖽙𖾏 𖼘𖽪𖾏 𖼊𖽳
4
𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼙𖽻 𖽃𖽔 𖼛𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼑𖽪𖾏, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼑𖽜𖾏 𖼮𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽡𖾏 𖽃𖽡𖾑 𖼮𖽑𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽱𖾑 𖼋𖽡𖾏 𖼘𖽪!𖼟𖽡 𖼋𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑.” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏. 2 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼟𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖼗𖽻 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼋𖽘𖾏. 3 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼑𖽜𖾏 𖼮𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽡𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽜𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼟𖽪𖾏 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 , 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼋𖽺 𖼀𖽡𖾐 𖼾𖽡𖾏 𖼐𖽡𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼄𖽡𖾐 𖼘𖽳 𖼑𖽡𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐. 4 𖼞𖽪𖾑 𖽐𖼊𖽱𖾏 𖼟𖽡 𖼅𖽔𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑, 𖼟𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼐𖽡𖽙𖾐 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼊𖽪𖾐 𖼗𖽻 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏, 𖼟𖽡 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼎𖽙𖾑 𖼖𖽡𖾏 𖼯𖽡𖾏 𖼅𖽔𖾏, 𖼮𖽔𖾑 𖼦𖽡𖾑 𖼯𖽪𖾏 𖼅𖽔𖾏 𖼎𖽙𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑.” 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼎𖽙𖾑.” 5 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼎𖽑𖽻𖾏 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽙𖾏（“𖼷𖽙𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼅𖽔𖾏”） 𖼊𖽪𖾐 𖼋𖽔 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼄𖽙𖾑 𖼽𖽔𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏𖼘𖽪𖾏 𖼊𖽳 , 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼮𖽡𖽝𖾏.” 6 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼅𖽔𖾏 𖼋𖽘𖾏, 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼞𖽪𖾑 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼇𖽺𖾐 𖼊𖽔𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼟𖽡 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼎𖽙𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼅𖽔𖾏 𖼎𖽙𖾑 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼅𖽔𖾏 𖼋𖽘𖾏.”
7 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼽𖽔𖾑 𖼏𖽵 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼅𖽔𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼞𖽑𖽻𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑. 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏. 8 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼎𖽙𖾑 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼦𖽡𖾑!” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼚𖽠𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼞𖽑𖽻𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏. 9 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼗𖽻 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏:“ 𖼅𖽡 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽙𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼄𖽡𖾐 𖼘𖽳 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼖𖽡𖽝 、 𖼄𖽔𖾐 𖼖𖽪𖽝 𖽃𖽡𖾑, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼗𖽙𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏． 10 𖼎𖽺𖾏 𖼷𖽙𖾏 𖼄𖽔𖾐 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼄𖽙𖾑 𖼽𖽔𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪𖾏 𖼊𖽳 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼐𖽡𖾑 𖼖𖽡𖾏 𖽐𖼊𖽡𖾏 𖼺𖽙𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼄𖽡𖾐 𖼘𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼾𖽡𖾏 𖼄𖽑𖽻𖾐 𖽐𖼷𖽷𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼛𖽙𖾏 𖼙𖽠𖾏. 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏.” 11 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼗𖽻 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏. 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼖𖽜𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼟𖽡 𖼾𖽡𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼺𖽱𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝 ， 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼯𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽪𖽙 𖽃𖽡𖾐 𖼼𖽘𖾐. 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼊𖽑𖽔𖾑 𖽂𖽪𖾏 𖽂𖽪𖾏 𖽐𖼊𖽙𖾏 𖽐𖼷𖽷𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖼊𖽻𖾏 𖼻𖽻𖾏 𖽐𖼷𖽷𖾏. 12 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽜𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼺𖽱𖾏 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼎𖽑𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼇𖽔 𖼖𖽳 𖼺𖽷𖾑 𖼐𖽱𖾏 𖼾𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.”
𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝𖾐 𖼘𖽪𖾏 𖼊𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
