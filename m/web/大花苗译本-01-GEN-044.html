6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼎𖽺𖾏 𖼖𖽠𖾑 𖼯𖽢 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼞𖽑𖽔𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.” 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼘𖽪? 𖼟𖽡 𖼡𖽑𖽜𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽻 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑. 8 𖼟𖽡 𖼑𖽔𖾏, 𖼀𖽡𖾐 𖼄𖽔𖾐 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖽐𖼚𖽙 𖼇𖽜𖾑, 𖼞𖽑𖽔𖾏 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖼁𖽻𖾏 𖼟𖽡 𖼺𖽷𖾐, 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼑𖽡𖽝 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐 𖼘𖽪? 9 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽜𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼚𖽙 𖼇𖽜𖾑, 𖼐𖽱𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼋𖽔, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏.” 10 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽔𖾑 𖼐𖽡𖾑 𖼀𖽡𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏! 𖼐𖽡𖽙𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼎𖽑𖽦𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼡𖽔 𖼋𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏, 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏.” 11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽜 𖼑𖽜 𖼷𖽙𖾏 𖼇𖽜𖾑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼡𖽑𖽜𖾐 𖼇𖽜𖾑 𖼮𖽡𖽝𖾏. 12 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼎𖽺𖾏 𖼷𖽱𖾏 𖼛𖽘𖾏 𖽐𖼎𖽑𖽦𖾑, 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼘𖽑𖽙𖾐, 𖼸𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽡𖾑, 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙 𖼖𖽪𖾑 𖼞𖽑𖽙𖾏 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼖𖽪𖾑 𖼇𖽜𖾑 𖽐𖼎𖽑𖽦𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏. 13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼮𖽑𖽙𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼮𖽑𖽳𖾑 𖽐𖼎𖽘𖾏 𖼎𖽻𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼮𖽹𖾑, 𖼺𖽷𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼙𖽻.
