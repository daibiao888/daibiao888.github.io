𖼗𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐
24
𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏.
2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
3 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏,
𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏．
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼀𖽻𖾐 𖼟𖽳,
𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.
 
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏.
6 𖼟𖽡 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏, 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 ,
𖼊𖽪𖾐 𖼸𖽠𖾐 𖼮𖽡𖽻𖾐, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏.
7 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐, 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼋𖽳𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏．
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏.
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼗𖽔 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽪𖾐.
10 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼮𖽔𖾑 𖼀𖽑𖽪𖾑 𖼺𖽡𖽝𖾐,
𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽺𖾏 𖼳𖽻𖾐.
 
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼮𖽤𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽔 , 𖼟𖽡 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼮𖽨𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏, 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼮𖽡𖽝𖾏.
12 𖼟𖽡 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏 : “𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐.”
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽡𖽝𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 𖼳𖽦𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼐𖽱𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
 
13 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽔𖾏 𖼄𖽪𖾑 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼗𖽙𖾏 𖼄𖽪𖾑 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼡𖽘𖾐 𖼄𖽪𖾑 𖽃𖽡𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
14 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼋𖽘𖾏, 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ,
𖼟𖽡 𖼮𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽘𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼟𖽡 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏 𖼺𖽘𖾏.
15 𖼟𖽡 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑！
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 .
16 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼢𖽻 𖼳𖽤𖾏 𖼟𖽻𖾏 𖼈𖽡𖽝𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏,
𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼊𖽑𖽠𖾐 𖼢𖽻 𖼺𖽘𖾏.
 
17 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽻𖾏 𖼢𖽻 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼙𖽔 ,
𖼐𖽱𖾏 𖼊𖽻𖾏 𖼢𖽻 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼗𖽙𖾏 𖼳𖽦𖾑.
18 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 ,
𖼟𖽙𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽑𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 .
 
19 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
20 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 .
 
21 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽙𖾑 𖼎𖽙𖾑 𖼎𖽺𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖽐𖼷𖽜𖾑 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔 .
22 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼳𖽠𖾑,
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼖𖽺𖾑?
 
23 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼗𖽪𖾏：
𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏.
24 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳.”
𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏, 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽪𖾐.
25 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼗𖽙𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
26 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖽃𖽡𖾑,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼡𖽑𖽻𖾐.
 
27 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏,
𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽐𖼷𖽻𖾏,
𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏.
 
28 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼼𖽷𖾏 𖼼𖽙 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐.
29 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 : “ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 ,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏.”
 
30 𖼞𖽪𖾑 𖼮𖽑𖽡𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽜 𖼑𖽡𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 ,
31 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼘𖽑𖽘𖾑 𖼀𖽪𖾑 𖼊𖽡𖾐,
𖽐𖼡𖽜𖾐 𖼞𖽡𖾏 𖼁𖽙𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼺𖽘𖾏 𖽐𖼎𖽹𖾏 𖼙𖽡𖽝𖾏,
𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼖𖽔𖾏 𖼼𖽠𖾏 𖼢𖽻 𖼙𖽠𖾏.
32 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏,
𖼞𖽪𖾑 𖼑𖽔𖾏 𖼎𖽺𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏.
33 𖼎𖽺𖾏 𖼀𖽱𖾏 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏, 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏,
𖼎𖽺𖾏 𖼁𖽔 𖼋𖽡𖾏 𖼢𖽻 𖼀𖽱𖾏 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏,
34 𖼟𖽡 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼋𖽔𖾏 𖼇𖽺𖾏 𖼺𖽱𖾏 𖼸𖽜 ,
𖼟𖽡 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 .
