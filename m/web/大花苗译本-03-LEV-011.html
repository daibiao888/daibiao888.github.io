11
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽑𖽡𖾏 : 2 “𖼅𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼢𖽻 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 : 3 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼇𖽺𖾏 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐、𖼯𖽦𖾏 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼼𖽙𖾏 , 𖼅𖽡 𖼖𖽔𖾏 𖼑𖽻𖾏 𖼢𖽻 . 4 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼼𖽙𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏： 𖼘𖽙 𖼊𖽑𖽙𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼼𖽙𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 5 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼷𖽔𖾏（“𖼖𖽔𖾑 𖼡𖽔𖾏 𖼷𖽔𖾏 ” 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼖𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼡𖽔𖾐 𖼈𖽳𖾐 𖼞𖽘𖾐 𖼡𖽔𖾏 𖼷𖽔𖾏） , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼼𖽙𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏， 6 𖼊𖽔𖾑 𖼖𖽔𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼼𖽙𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏， 7 𖼊𖽪𖾑 𖽐𖼀𖽔𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 , 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖽐𖼊𖽡𖾏 𖼼𖽙𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 8 𖼋𖽡𖾏 𖼯𖽦𖾏 𖼐𖽡𖾑 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏 , 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽔 𖽃𖽡𖾑 , 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 , 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .
9 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖼑𖽻𖾏 𖼢𖽻 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 : 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖽐𖼚𖽙、𖼳𖽡𖽠𖾑 𖽐𖼚𖽙、𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖽐𖼚𖽙、𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼷𖽷𖾏 𖽃𖽡𖾑 , 𖼖𖽔𖾏 𖼑𖽻𖾏 𖼢𖽻 . 10 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼚𖽙、𖽃𖽔𖾑 𖼟𖽱𖾐 𖽐𖼚𖽙、𖼊𖽑𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼎𖽳𖾏 𖽃𖽔𖾏 𖼡𖽹𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖽃𖽡𖾑 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼷𖽷𖾏 𖽃𖽡𖾑 , 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 . 11 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼷𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼐𖽡𖾑 , 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 , 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼋𖽡𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 . 12 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼷𖽷𖾏 𖽃𖽡𖾑 ,𖼋𖽡𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 .
13 𖼐𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼅𖽡 𖼑𖽻𖾏 𖽃𖽡𖾑，𖼟𖽪𖾏： 𖼚𖽘𖾑 𖼄𖽔𖾐 𖽐𖼀𖽔𖾏、𖼚𖽘𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼚𖽡𖾑、𖼚𖽘𖾑 𖼖𖽡𖽝𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏、 14 𖼙𖽡𖽝𖾏 𖽐𖼷𖽔𖾐、𖼙𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖼮𖽢𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽡𖾑、 15 𖼖𖽡𖾑 𖽃𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽡𖾑 , 16 𖼊𖽪𖾑 𖼐𖽻 𖼞𖽡𖾏 𖼎𖽻𖾑、𖼽𖽱𖾏 𖽂𖽙、𖼚𖽘𖾑 𖼑𖽻𖾏 𖽐𖼀𖽳𖾏、𖼚𖽘𖾑 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽡𖾑 , 17 𖼊𖽔𖾑 𖼞𖽙𖾏 𖼞𖽙𖾏、𖽃𖽙𖾏 𖼑𖽻𖾏 𖽐𖼀𖽳𖾏、𖼘𖽻𖾏 𖼘𖽑𖽱𖾏、 18 𖼮𖽑𖽙𖾑 𖼮𖽑𖽙𖾐、𖼤𖽪𖾏 𖼑𖽻𖾏 𖽐𖼀𖽳𖾏、𖼚𖽘𖾑 𖼚𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏、 19 𖼁𖽔𖾏、𖼁𖽔𖾏 𖼑𖽻𖾏 𖽐𖼀𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽡𖾑 , 𖼐𖽻 𖽐𖼮𖽡𖽠𖾏 𖽐𖼚𖽡𖾏 𖼊𖽑𖽡𖽝𖾐 𖼑𖽘𖾏 𖽐𖼀𖽔𖾑 .
20 𖼀𖽡𖾑 𖼞𖽘𖾐 𖼟𖽪𖾏 𖼎𖽳𖾏 𖽃𖽔𖾏 𖼡𖽹𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏、 𖼎𖽑𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼋𖽡𖾏 𖽐𖼞𖽘 𖽃𖽡𖾑 ,𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 . 21 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽳𖾏 𖽃𖽔𖾏 𖼡𖽹𖾏， 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏、𖼎𖽑𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼋𖽡𖾏 𖽐𖼞𖽘 , 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽠𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼀𖽔𖾐 , 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼚𖽑𖽡𖽝𖾏 𖼚𖽜𖾑𖽑 𖽃𖽡𖾑 , 𖼅𖽡 𖼑𖽻𖾏 𖼢𖽻𖾏 𖼯𖽡𖽙𖾏 . 22 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼷𖽷、𖼷𖽷 𖼟𖽻𖾏、𖼷𖽷𖾏 𖼖𖽱𖾑 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽡𖾑 , 𖼡𖽔𖾑 𖼯𖽱𖾏 𖼷𖽷 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽡𖾑 . 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼖𖽔𖾏 𖼑𖽻𖾏 𖼢𖽻𖾏 . 23 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽳𖾏 𖽃𖽔𖾏 𖼡𖽹𖾏， 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 , 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼞𖽘 𖽃𖽡𖾑 , 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 .
