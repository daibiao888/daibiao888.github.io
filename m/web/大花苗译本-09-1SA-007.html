𖼮𖽡𖾑 𖼖𖽡𖽝 · 𖼽𖽦𖾑 𖼖𖽡𖽾 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼡𖽜𖾏 𖼋𖽔𖾏, 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼞𖽱𖾑 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽔 𖼊𖽔 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼮𖽡𖽝𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽔 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼟𖽙𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏． 2 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 · 𖼽𖽦𖾑 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼗𖽡𖾏, 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
3 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼮𖽔𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼺𖽷𖾑 𖼊𖽑𖽔𖾑 𖼖𖽪 𖼎𖽑𖽻𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏, 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼊𖽜𖾑， 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼅𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.” 4 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼀𖽔𖾑 𖼖𖽡 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼺𖽷𖾑 𖼊𖽑𖽔𖾑 𖼖𖽪, 𖼯𖽔 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐.
𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐
5 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼄𖽡𖾐 𖼺𖽷𖾑 𖼀𖽔𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼑𖽜 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.” 6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼄𖽡𖾐 𖼺𖽷𖾑 𖼀𖽔𖾑, 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼊𖽪𖾏 𖼡𖽜𖾑, 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼄𖽡𖾐 𖼺𖽷𖾑 𖼀𖽔𖾑 𖼨𖽑𖽢𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐. 7 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼄𖽡𖾐 𖼺𖽷𖾑 𖼀𖽔𖾑, 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼅𖽡 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐. 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏, 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. 8 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼦𖽡𖾐 𖼳𖽙𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.” 9 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑, 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼐𖽱𖾏. 10 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑, 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏. 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼺𖽙𖾐 𖼢𖽔𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼮𖽡𖽠𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐. 11 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼄𖽡𖾐 𖼺𖽷𖾑 𖼀𖽔𖾑 𖼋𖽠𖾏 𖼋𖽔𖾏, 𖼖𖽠𖾑 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽐𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼀𖽳 · 𖼮𖽢 𖼑𖽡𖾏 𖼺𖽘𖾑 𖼀𖽡 𖼋𖽻.
12 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼷𖽑𖽙𖾏 𖼎𖽻𖾏 𖼄𖽡𖾐 𖼺𖽷𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼳𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼟𖽙𖾏 𖼮𖽡𖽙𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼀𖽡𖽝 · 𖼽𖽡𖾑 𖼳𖽡𖽝𖾏, 𖼐𖽑𖽡𖾏: “𖼸𖽙 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐.” 13 𖼎𖽑𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼯𖽵𖾏 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐. 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼋𖽻 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. 14 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼞𖽳 𖼖𖽪𖽝 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳, 𖼖𖽔𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏. 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼋𖽡𖾏 𖼙𖽻 𖼐𖽡𖾑 𖽃𖽡𖾑, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽔𖾏 𖼎𖽑𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖾏 𖼳𖽻𖾏 𖼺𖽷𖾐 𖼗𖽙. 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏.
