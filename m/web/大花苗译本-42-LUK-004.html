𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 4：1-11；𖼞𖽑𖽙𖾐 1：12-13）
4
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 ，𖼎𖽑𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼺𖽷𖾐 𖼗𖽙，𖼳𖽟𖾏 𖼖𖽡𖽾 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏， 2 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑． 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼀𖽪𖾑 𖼋𖽘𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽑𖽹𖾐 𖼋𖽘． 3 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐， 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼋𖽡𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼳𖽡𖽝𖾐．” 4 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏： “𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：
‘𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼊𖽔𖾐，𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏．’ ”
5 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐，𖽃𖽡𖾐 𖽐𖼮𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼡𖽑𖽳𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼑𖽔𖾏． 6 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “𖼋𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽐𖼎𖽙 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼙𖽔 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏． 7 𖼟𖽡 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑，𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼺𖽙𖾏 𖼟𖽡 𖼁𖽡𖽝．” 8 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：
‘𖼟𖽡 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，
𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐．’ ”
9 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖽃𖽔𖾏 𖼎𖽪𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 ，𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼚𖽑𖽡𖽝𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏． 10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：
‘ 𖼮𖽵𖾑 𖼑𖽜 𖼟𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡．
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼟𖽡 𖼮𖽡𖽝𖾏，
𖽐𖼮𖽑𖽹𖾏 𖼟𖽡 𖼊𖽻𖾏 𖽐𖼊𖽱𖾑𖽑 𖼊𖽠𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐．’ ”
12 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼟𖽡 𖼮𖽵𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．’ ” 13 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼺𖽘𖾏 ，𖼎𖽺𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼙𖽠𖾏．
𖼀𖽡𖾑𖼚𖽘𖾐𖼮𖽵𖾑𖽑 𖽐𖼮𖽔𖾏 𖼽𖽦𖾑𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 4：12-17；𖼞𖽑𖽙𖾐 1：14-15）
14 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 ，𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏，𖼐𖽱𖾏 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖼚𖽔𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑． 15 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽻𖾑 𖼐𖽱𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽡𖾐
（𖼊𖽑𖽹𖾏 13：53-58；𖼞𖽑𖽙𖾐 6：1-6）
16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳，𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼐𖽡𖽙𖾐 𖼘𖽑𖽙𖾐， 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐，𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱，𖼐𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼽𖽔𖾑 𖼟𖽡𖽠 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑． 17 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏，𖼐𖽱𖾏 𖼡𖽑𖽜𖾐 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：
18 “ 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼞𖽪𖾑，
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 ，
𖼷𖽷𖾐 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏，
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽙𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏，
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑，
𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽺𖾏 𖼯𖽵𖾏 ，𖼊𖽻𖾏 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏．
19 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼳𖽡𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼟𖽪𖾏 𖼮𖽵𖾑 𖽃𖽺𖾐 𖼙𖽔 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．”

31 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻，𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼡𖽑𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐． 32 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽙𖾑 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏． 33 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼏𖽻 𖼖𖽡𖽾 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏，𖼐𖽱𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏： 34 “𖽃𖽺𖾐！𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖼀𖽡𖾐 𖽐𖼎𖽙𖾏 𖼟𖽡 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼐𖽔𖾑？ 𖼟𖽡 𖼋𖽔𖾏 𖼖𖽻𖾐 𖼀𖽡𖾐 𖼘𖽪𖾏 ？ 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏， 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 ．” 35 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏，𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 ！” 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼎𖽺𖾏 𖽐𖼀𖽱𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼟𖽡𖾏 𖼦𖽹𖾏 𖼐𖽱𖾏． 36 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏，𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼑𖽪 𖼐𖽑𖽡𖾏： “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽺𖾑？ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽐𖼎𖽙 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏，𖼀𖽡𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏．” 37 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼊𖽙𖾏 𖽐𖼷𖽷𖾏 𖼚𖽔𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼀𖽡𖾐 𖼊𖽳 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽹𖾏
（𖼊𖽑𖽹𖾏 8：14-17；𖼞𖽑𖽙𖾐 1：29-34）
38 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 ，𖼚𖽔𖾏 𖼳𖽡𖾑 𖼄𖽞 𖽐𖼞𖽔𖾏，𖼳𖽡𖾑 𖼄𖽞 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽹𖾏 𖼄𖽙𖾐 𖼄𖽙𖾐 𖼚𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽑𖽤𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑． 39 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪， 𖼖𖽔𖾏 𖼋𖽡𖾏 𖼄𖽙𖾐 𖼄𖽙𖾐 𖼚𖽪𖾐 𖼐𖽡𖾑，𖼄𖽙𖾐 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼗𖽠𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 40 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼋𖽘𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼄𖽙𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏，𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼞𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏． 41 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐！” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ，𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪．
（𖼞𖽑𖽙𖾐 1：35-39）
42 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏．𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏，𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 ，𖼽𖽔𖾑 𖼞𖽑𖽻𖾑 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 ，𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏． 43 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼙𖽻 𖼡𖽑𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼄𖽻𖾏 𖼻𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼗𖽙 𖼟𖽪𖾏 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏．” 44 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏．
