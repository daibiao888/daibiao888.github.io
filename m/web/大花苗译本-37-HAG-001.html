𖼦𖽔𖾐 𖼞𖽹𖾑 𖽐𖼊𖽠𖾑
𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
1
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼖𖽡𖾏 𖽂𖽪𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼖𖽪𖾑 𖼚𖽻𖾏 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼦𖽔𖾐 𖼞𖽹𖾑 , 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽑𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼺𖽙𖾐 𖼘𖽙 𖼀𖽔𖾑 𖼀𖽳 , 𖼊𖽑𖽡𖽝𖾐 𖼽𖽙 𖼺𖽔𖾐 𖼊𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏:
2 “𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : ‘ 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏, 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼺𖽷𖾐.’ ” 3 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼦𖽔𖾐 𖼞𖽹𖾑 𖼐𖽑𖽡𖾏: 4 “𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐, 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖽐𖼊𖽑𖽘𖾐 𖽐𖼮𖽦𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼺𖽷𖾐 𖼘𖽪𖾏? 5 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼅𖽡 𖼽𖽔𖾑 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏. 6 𖼅𖽡 𖼀𖽳𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼯𖽠； 𖼅𖽡 𖼑𖽻𖾏 , 𖼈𖽡𖽝𖾏 𖼑𖽻𖾏 𖼦𖽡𖾐 𖼮𖽻𖾏； 𖼦𖽻𖾏, 𖼖𖽔𖾏 𖼦𖽻𖾏 𖼦𖽡𖾐 𖼇𖽔𖾑； 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼳𖽙𖾑. 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 , 𖼞𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼐𖽑𖽘𖾐 𖼟𖽪𖾏 𖽐𖼚𖽔 𖼊𖽙𖾏. 7 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼅𖽡 𖼽𖽔𖾑 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏. 8 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼅𖽻𖾏 𖽐𖼊𖽙𖾑 𖽐𖼊𖽻𖾏, 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼙𖽔 , 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
9 𖼅𖽡 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽻𖾏, 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼯𖽠 , 𖼅𖽡 𖼳𖽻𖾏 𖼗𖽙𖾏 𖼸𖽙 𖽐𖼞𖽔𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼷𖽑𖽔𖾐 𖼺𖽘𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑? 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏, 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼾𖽨𖾏 𖼞𖽳𖾑 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽔𖾑 ( “𖼞𖽳𖾑”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼏𖽘 ”), 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼑𖽜 𖼅𖽡 𖼑𖽪𖾏 𖼊𖽔𖾐, 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼗𖽙𖾏 𖼀𖽡𖾑 𖼙𖽱, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼡𖽻𖾐. 11 𖼞𖽪𖾑 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼞𖽡 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏、 𖼇𖽻𖾑 𖼎𖽻𖾐、 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼡𖽻𖾐、 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖽐𖼎𖽙 𖼛𖽙𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏、 𖼖𖽡𖾑 𖼉𖽪𖾏、 𖼯𖽦𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖽃𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑.”
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏
12 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽑𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼺𖽙𖾐 𖼘𖽙 𖼀𖽔𖾑 𖼀𖽳 𖽐𖼎𖽙 𖼽𖽙 𖼺𖽔𖾐 𖼊𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽻 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼑𖽻 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼦𖽔𖾐 𖼞𖽹𖾑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏. 13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼦𖽔𖾐 𖼞𖽹𖾑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐.” 14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼽𖽨 𖼊𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽑𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼺𖽙𖾐 𖼘𖽙 𖼀𖽔𖾑 𖼀𖽳 , 𖽐𖼎𖽙 𖼽𖽙 𖼺𖽔𖾐 𖼊𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾐 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 15 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽔𖾏 𖼖𖽡𖾏 𖽂𖽪𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼖𖽪𖾑 𖼚𖽻𖾏 𖼖𖽑𖽡𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼚𖽻𖾐.
