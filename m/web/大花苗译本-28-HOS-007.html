4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏,

𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼄𖽑𖽻𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐, 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼳𖽪𖾏 𖽃𖽡𖾑,
𖼦𖽡𖾐 𖼟𖽙𖾏 𖼋𖽠𖾏 𖼯𖽡 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
5 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑,

6 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖽐𖼚𖽺𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼞𖽪𖾐 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼷𖽙𖾏,
𖼺𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽡𖽝𖾐 𖼳𖽡𖽝𖾐 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏,

7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖾐 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼷𖽙𖾏,

8 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏,

9 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐,

10 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏,
𖽐𖼮𖽷𖾐 𖼏𖽻 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽘𖾏 𖼈𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,

11 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐,

12 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽻𖾏 𖼀𖽻𖾐 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 .
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.

𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽷𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽱𖾐 𖼊𖽡𖾑 𖼢𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 .

14 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼞𖽪𖾑 ,

15 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼞𖽪𖾑 .
16 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐,

𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼾𖽨𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼙𖽙𖾏 𖽐𖼊𖽪𖾏 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽙𖾏.
