𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
19
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏,
𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏.
2 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏,
𖼄𖽑𖽙𖾏 𖼐𖽡𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑 𖼡𖽑𖽔𖾐 𖼋𖽠𖾏 𖼀𖽻𖾐 𖼟𖽳.
3 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾐,𖼦𖽡𖾐 𖼅𖽔𖾏 𖼗𖽪𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼑𖽻.
4 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖽐𖼎𖽙𖾑 𖽐𖼷𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼐𖽱𖾏 𖼗𖽪𖾏 𖼡𖽑𖽔𖾐 𖼸𖽙 𖼊𖽡𖾐 𖼺𖽻𖾏.
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽜 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐.
5 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼺𖽱𖾏 𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖼋𖽠𖾏 𖼮𖽑𖽤𖾏 𖽐𖼞𖽔𖾏, 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼙𖽔 𖼚𖽙𖾏 𖼏𖽘 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑.
6 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼺𖽘𖾑 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼋𖽔𖾏, 𖼮𖽑𖽙𖾑 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼺𖽘𖾑 𖽃𖽡𖾑,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽻𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼀𖽘𖾏 𖼳𖽙𖾑.
 
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖼸𖽠𖾏 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼖𖽡𖽾 𖼯𖽷𖾏 𖼗𖽙𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏, 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽱𖾏 𖼅𖽳𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐,
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼁𖽙𖾏 𖼨𖽑𖽪𖾑.
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽘𖾐 𖼞𖽡𖾏（“𖼞𖽘𖾐 𖼞𖽡𖾏 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 ”） 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼷𖽷𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
10 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖼉𖽳 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖼉𖽳,
𖼡𖽘𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾑 𖼑𖽘 𖼄𖽪𖾑 .
11 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼯𖽵𖾏 𖼡𖽑𖽔𖾐,
𖼻𖽙𖾑 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼮𖽡𖽝𖾏 𖼯𖽡𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
12 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼖𖽺𖾑,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏.
13 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼦𖽡𖾐 𖼯𖽪𖾏 𖼋𖽡𖾏 𖼾𖽔𖾏 𖼐𖽡𖾑 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼞𖽪𖾑,
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
 
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏, 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼘𖽪！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 , 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏.
