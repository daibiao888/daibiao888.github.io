6 “𖼅𖽡 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼋𖽳 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽔𖾏 𖽐𖼚𖽜𖾏 , 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 .
7 “𖼅𖽡 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼳𖽠𖾑 𖼁𖽱𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼦𖽙 𖽃𖽳𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐， 8 𖼎𖽑𖽻𖾏 𖼦𖽙 𖽃𖽳𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼯𖽪𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽜𖾏 𖼚𖽔𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 , 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼊𖽔 𖼊𖽔， 9 𖼎𖽺𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼇𖽺𖾑 𖼖𖽪𖽝 , 𖼸𖽙 𖼚𖽔𖾏𖼦𖽔𖾐 𖼺𖽔𖾐 · 𖼽𖽡𖾑 𖼐𖽔． 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼅𖽡 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 .
10 “𖼅𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼦𖽔𖾐 𖼺𖽔𖾐 · 𖼽𖽡𖾑 𖼐𖽔 , 𖼯𖽪𖾏 𖼸𖽙 𖼚𖽔𖾏𖼳𖽷𖾏 𖼇𖽔𖾑 , 𖽃𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 . 11 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼳𖽷𖾏 𖼇𖽔𖾑 𖽐𖼡𖽜𖾏 𖼸𖽙 𖼽𖽔𖾏 𖼽𖽦 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼐𖽡 𖼖𖽡𖽝 𖼳𖽡𖽠𖾑 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 .” 12 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼡𖽜𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽡𖽠𖾑 𖽐𖼮𖽳𖾑．𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼚𖽻𖾐 𖼺𖽘𖾑 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 . ”
13 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : “𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 𖼎𖽻𖾏 𖼯𖽢𖾏 𖼳𖽡𖾏 𖽐𖼊𖽘𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 . 14 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽨 𖼀𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼊𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 , 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖽐𖼊𖽘𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 . 15 𖽃𖽔𖾐 𖼳𖽡𖾏 𖽐𖼊𖽘𖾏 𖼐𖽡𖾑 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 𖼀𖽘𖾑𖽑 𖼀𖽡 𖼋𖽡𖾏 , 𖼊𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 , 𖼟𖽪𖾏 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 .”
𖼺𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 17 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼇𖽺𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 , 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼐𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 . 18 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽹𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼗𖽙𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 . 19 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏 :
𖼽𖽨 𖼊𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼇𖽪 𖼐𖽡 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼮𖽢𖾑 𖼖𖽳 .
20 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼦𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽷𖾏 𖼄𖽪𖾐 𖼖𖽡𖾏 .
21 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼺𖽷𖾑 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼊𖽔 .
22 𖼊𖽗𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 , 𖼽𖽙 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽪𖾏 𖼮𖽡𖾑．
23 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 , 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏, 𖼽𖽡𖾑 𖼇𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼦𖽗𖾏 𖼐𖽡𖽝𖾏 .
24 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 , 𖼳𖽷𖾏 𖼇𖽪 𖼊𖽗𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼮𖽡𖾑 𖼄𖽪𖾐 𖼖𖽡𖾏 .
25 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 , 𖼀𖽑𖽔 𖼐𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼺𖽔𖾐 𖼇𖽔𖾑 .
26 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 , 𖽃𖽔𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽑𖽔 𖼊𖽑𖽡𖽝 .
27 𖼽𖽔𖾏 𖼳𖽳 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 , 𖼳𖽷𖾏 𖼘𖽙 𖼄𖽡𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼦𖽪 .
28 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 , 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼦𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽡𖾐 𖼊𖽔𖾏 𖼦𖽳 .”
29 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼄𖽔𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 .
