𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼇𖽔 𖼘𖽻𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽡 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 . 2 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖽃𖽡𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 . 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏 : 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 . 3 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 . 4 𖼈𖽡𖽝𖾏 𖼇𖽔 𖼘𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼑𖽻 𖼅𖽡 , 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼗𖽙𖾏 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼨𖽑𖽢𖾐 𖼣𖽹 𖼮𖽡 , 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼸𖽻𖾏 𖼏𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 . 5 𖼞𖽪𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼣𖽹 𖼮𖽡 , 𖼄𖽔𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .” 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏 . 7 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼅𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 , 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼅𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 .
𖼮𖽡𖾑 𖼀𖽜𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐

25 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽻𖾏𖼐𖽡 𖼘𖽙 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 .
