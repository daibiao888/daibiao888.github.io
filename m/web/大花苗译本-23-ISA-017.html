2 𖼽𖽔𖾏 𖼘𖽙 𖽃𖽳𖾐 𖼖𖽪𖾑 𖼙𖽻 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼯𖽦𖾏,
𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽠𖾏.
3 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 ,

𖼽𖽔𖾏 𖼖𖽔 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑,
𖼽𖽔𖾑 𖼺𖽱𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
4 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼽𖽔𖾑 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼛𖽙𖾏 𖼙𖽠𖾏,

5 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖽐𖼮𖽳 𖼺𖽷𖾐 𖽃𖽡𖾑,
𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼤𖽜𖾏 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐 𖼡𖽻𖾐,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼖𖽡𖾏 𖼇𖽔 𖼽𖽡𖽾𖾑 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼳𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼙𖽠𖾏.
6 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼮𖽡𖽻𖾐, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼛𖽠𖾏 𖼞𖽗𖾐 𖼖𖽗 𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐,
𖼮𖽔𖾑 𖼳𖽙𖾏 𖽃𖽔𖾐 𖼷𖽷𖾐 𖼖𖽪𖾐 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽐𖼊𖽻𖾏 𖼊𖽔𖾐,
𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼮𖽡𖽻𖾐 𖼮𖽔𖾑 𖼳𖽙𖾏 𖼚𖽻𖾐 𖼀𖽜𖾐 𖼖𖽪𖾐 𖼊𖽔𖾐,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
7 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼨𖽑𖽪𖾑.
8 𖼐𖽱𖾏 𖼸𖽻 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖽝𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑𖽃𖽔𖾐 𖽐𖼮𖽡𖾐, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 . 9 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼏𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼻𖽻𖾏 𖽐𖼚𖽙 𖽐𖼎𖽙 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏. 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼮𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼟𖽡 ,
𖼦𖽡𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼎𖽑𖽺𖾏 𖽃𖽔𖾑 𖼯𖽡 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 ,
𖼯𖽡𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽘𖾑 𖽐𖼊𖽻𖾏 𖼮𖽡𖽝𖾏 𖼈𖽡𖽝𖾏，
11 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙 𖽃𖽡𖾑, 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼘𖽑𖽙𖾐,
𖼎𖽺𖾏 𖼸𖽙 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼟𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽡𖽙 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼁𖽘𖾏,
𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼳𖽻𖾏 𖼤𖽜𖾏 𖽃𖽡𖾑 𖼽𖽘𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏.
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼳𖽠𖾏
12 𖽃𖽺𖾐! 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏, 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏,
𖼉𖽻𖾏 𖼞𖽪𖽙 𖽐𖼷𖽑𖽷𖾏 𖼳𖽱𖾑, 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼷𖽑𖽷𖾏 𖼳𖽱𖾑.
13 𖼉𖽻𖾏 𖼞𖽪𖽙 𖽐𖼷𖽑𖽷𖾏 𖼳𖽱𖾑, 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼷𖽑𖽷𖾏 𖼳𖽱𖾑,
𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽘 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼎𖽺𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑, 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼞𖽑𖽻𖾐 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖽐𖼀𖽱𖾏 𖼷𖽑𖽔𖾐,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖽐𖼀𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
14 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏, 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼸𖽙 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘 .
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼀𖽡𖾐 𖽃𖽡𖾑 𖼊𖽻𖾏, 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼀𖽡𖾐.
