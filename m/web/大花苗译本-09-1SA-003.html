3
𖼣𖽔𖾑 𖼽𖽻𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽺𖾐 𖼯𖽠, 𖼦𖽡𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼅𖽔𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐. 2 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐, 𖼽𖽡𖾑 𖼖𖽡 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼇𖽔 𖼙𖽻, 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐. 3 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼇𖽳𖾏 𖼋𖽔 𖼺𖽷𖾐, 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏. 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐. 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑.” 5 𖼎𖽺𖾏 𖼏𖽘 𖽐𖼮𖽡𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑.” 𖼽𖽡𖾑 𖼖𖽡 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡, 𖼟𖽡 𖼅𖽻𖾏 𖼀𖽱𖾏 𖼦𖽡𖾑.” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼀𖽱𖾏 𖼋𖽘𖾏. 6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐, 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖽐𖼮𖽡𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑.” 𖼽𖽡𖾑 𖼖𖽡 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐 𖼘𖽪！ 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡, 𖼟𖽡 𖼅𖽻𖾏 𖼀𖽱𖾏 𖼦𖽡𖾑.” 7 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼯𖽜𖾏 𖽐𖼎𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐. 8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐, 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖽐𖼮𖽡𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑.” 𖼽𖽡𖾑 𖼖𖽡 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏. 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼽𖽡𖾑 𖼖𖽡 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼅𖽻𖾏 𖼀𖽱𖾏 𖼦𖽡𖾑, 𖼮𖽔𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡, 𖼟𖽡 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼷𖽑𖽜𖾑 𖼐𖽑𖽡𖾏, 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼯𖽪𖾏 𖼑𖽻！” 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼀𖽱𖾏.
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏, 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼷𖽷𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼘𖽪! 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼘𖽪!” 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽜𖾑 𖼐𖽑𖽡𖾏, 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼯𖽪𖾏 𖼑𖽻.” 11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽙𖾑𖽑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏. 12 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼾𖽡𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖼸𖽙 𖼋𖽘𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼁𖽱𖾏 𖼸𖽙 𖼺𖽻𖾏 𖽃𖽔𖾏 𖼸𖽪𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑. 13 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽡𖾏 𖽐𖼊𖽡𖾏 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼾𖽡𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼖𖽔𖾏 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏. 14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼏𖽵 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼾𖽡𖾏 𖼐𖽑𖽡𖾏: 𖼽𖽡𖾑 𖼖𖽡 𖼾𖽡𖾏 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖼐𖽱𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼺𖽘𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏, 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏.”
15 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼀𖽱𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏, 𖼎𖽺𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼄𖽔𖾐 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏. 16 𖼽𖽡𖾑 𖼖𖽡 𖼨𖽑𖽪𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼘𖽪！” 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑.” 17 𖼽𖽡𖾑 𖼖𖽡 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏. 𖼟𖽡 𖼮𖽔𖾑 𖼄𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏. 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽑𖽤𖾑 𖼎𖽻𖾏 𖼟𖽡.” 18 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏. 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼐𖽱𖾏 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖼯𖽡𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼻𖽻𖾏.”
𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐
