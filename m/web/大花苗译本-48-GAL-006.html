6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏，𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼇𖽺𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏． 7 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼮𖽑𖽵𖾑 𖼾𖽨𖾏， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼦𖽡𖾐 𖼢𖽻． 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏． 8 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼀𖽳𖾐 𖼮𖽘𖾑， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼳𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼗𖽱𖾏，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼀𖽳𖾐 𖼮𖽘𖾑 ，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼳𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏． 9 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏 ， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼚𖽔𖾐， 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽑𖽪𖾑 𖼺𖽡𖽝𖾐， 𖼸𖽙 𖼋𖽘𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼊𖽻𖾏． 10 𖼑𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼗𖽙𖾏 ， 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 ， 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼮𖽜𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏．
𖼀𖽻𖾐 𖼘𖽙 𖼮𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐 𖼊𖽔𖾐
