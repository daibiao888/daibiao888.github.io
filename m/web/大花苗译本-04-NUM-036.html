5 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 , 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : “𖼽𖽙 𖼺𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼾𖽙 𖼞𖽡𖾑． 6 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼳𖽡𖾑 𖼘𖽙 𖼇𖽺𖾑 𖼦𖽔𖾐 𖼑𖽡𖾏 𖽐𖼀𖽙𖾏 𖽐𖼷𖽑𖽹𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 𖼡𖽔𖾏 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼯𖽡𖾏 𖼡𖽔𖾏 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼈𖽡𖽝𖾏 , 𖼽𖽔𖾑 𖼡𖽔𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾐 . 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼳𖽡𖾏 𖼐𖽡𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼳𖽡𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 . 8 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 , 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼐𖽤𖾐 , 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 . 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 , 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼳𖽡𖾏 𖼐𖽡𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 . 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽡𖾏 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 .”
