𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽜 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽪𖾐 𖼳𖽷 𖼑𖽡𖾏 𖼗𖽪𖾏 ,𖼨𖽑𖽪𖾏 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 “𖼳𖽔𖾏 𖽐𖼮𖽡” 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
7
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽪𖾑, 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏！
2 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽐𖼮𖽑𖽷𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏,
𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼀𖽡𖾏 𖼚𖽑𖽔𖾏 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏.
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼮𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏,
𖼮𖽔𖾑 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
4 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽪𖾏 𖼁𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼻𖽻𖾏,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐, 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏.
5 𖼎𖽺𖾏 𖼯𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼖𖽠𖾑 𖼞𖽪𖾑, 𖼸𖽙 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼯𖽢,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼯𖽪𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏,
𖼳𖽠𖾑 𖼟𖽡𖾏 𖼮𖽨𖾏 𖼀𖽡𖾏 𖼮𖽨𖾑, 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼯𖽷𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼟𖽡 𖼙𖽔 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼖𖽔𖾑 𖼨𖽑𖽢𖾐.
7 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼞𖽑𖽡 𖼙𖽻 𖼟𖽡,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼳𖽠𖾑 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐.
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏，
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑
𖽐𖼎𖽙 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑.
 
9 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽪𖾏 𖼮𖽘𖾑,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳.
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼘𖽪！ 𖼟𖽡 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽐𖼀𖽱𖾏.
10 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐,
𖼐𖽱𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳.
11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐.
 
12 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼺𖽷𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼽𖽔𖾑 𖼨𖽑𖽙𖾑 𖼟𖽪𖾏 𖽐𖼮𖽳𖾏,
𖼾𖽨𖾏 𖼊𖽪𖾑 𖼐𖽑𖽜𖾑 𖼖𖽔𖾏 𖽐𖼊𖽘𖾏 𖼮𖽡𖽝𖾏, 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏.
13 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼀𖽙𖾑 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼈𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏.
14 𖼑𖽔𖾏 𖼘𖽪！ 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽔𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖽐𖼞𖽜𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼊𖽔𖾐, 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐.
15 𖼐𖽱𖾏 𖽐𖼮𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽻𖾐, 𖽐𖼮𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐,
𖼉𖽡𖽝𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼮𖽡𖽠𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏.
16 𖼐𖽱𖾏 𖼋𖽡𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼷𖽠𖾏 𖼾𖽨𖾏 𖼋𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼀𖽻𖾐 𖼷𖽠𖾏 𖼾𖽨𖾏 𖼋𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏.
 
17 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼐𖽱𖾏,
𖼷𖽻𖾑 𖼖𖽔 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏.
