8 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡， 𖼀𖽡𖾐 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐， 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼳𖽡𖾏 𖼽𖽔𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻， 𖼸𖽺𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼳𖽠𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏， 𖼀𖽡𖾐 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼋𖽘， 9 𖼀𖽡𖾐 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼋𖽡𖽝，𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼾𖽨𖾏 𖼻𖽡𖽝𖾏 𖼾𖽨𖾏， 𖼯𖽔 𖼻𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼋𖽔 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖽃𖽡𖾑 𖼊𖽔𖾐． 10 𖼐𖽱𖾏 𖼙𖽔 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼋𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽡𖾏 𖽃𖽡𖾑， 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼺𖽷𖾐， 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼊𖽑𖽡𖽝𖾐． 11 𖼅𖽡 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐， 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽜 𖼀𖽡𖾐 𖼙𖽔 𖼅𖽳𖾏， 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼯𖽡𖽙𖾏．
