𖼽𖽔𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
3
𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑；
2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 、
𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼑𖽡𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖽐𖼎𖽙 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼟𖽡 .
3 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼇𖽳𖾑 𖼟𖽡 𖼺𖽘𖾏,
𖼽𖽔𖾑 𖼡𖽑𖽹𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐 𖼮𖽡𖽝𖾏, 𖼞𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏.
4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾏
𖼊𖽻𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏, 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐.
5 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐；
6 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽜𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼎𖽺𖾏 𖼋𖽡𖽝𖾏 𖼎𖽺𖾏 𖽐𖼞𖽳𖾏.
7 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼑𖽔𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼀𖽜𖾑 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐.
8 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑,
𖽐𖼊𖽪𖾐 𖼐𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼽𖽘𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏.
9 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐
𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼳𖽦𖾑, 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼼𖽪 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏,
𖼟𖽡 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼮𖽡𖽠𖾑 𖼅𖽔𖾏 𖼮𖽡𖽠𖾑 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖽐𖼚𖽔𖾏 𖼗𖽙𖾏.
 
11 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 (“𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 ”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖼇𖽔𖾏 𖼳𖽙𖾏” ),
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 ；
12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖽐𖼮𖽑𖽠𖾏, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼉𖽺𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑
13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 、𖽐𖼎𖽙 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽃𖽡𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
14 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐,
𖼐𖽱𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾐,
15 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼙𖽔 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼋𖽳𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳.
16 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼅𖽔𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 ,
𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼇𖽳𖾏 𖼅𖽔𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖽐𖼎𖽙 𖽐𖼚𖽻 𖼮𖽡𖾏 .
17 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 .
18 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼞𖽑𖽡𖽠𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
 
19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼷𖽑𖽙𖾏 𖼷𖽱𖾏 𖽐𖼊𖽪𖾏.
20 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼇𖽺𖾐 𖼺𖽘𖾏,
𖼟𖽙𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼗𖽙𖾏 𖼀𖽡𖾑 𖼙𖽱 .
21 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔．
22 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽡 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼻𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼚𖽑𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐.
23 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼋𖽡𖽝𖾏 𖼋𖽪 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 ,
𖼦𖽡𖾑 𖼊𖽻𖾏 𖽐𖼊𖽱𖾑𖽑 𖼊𖽠𖾏.
24 𖼟𖽡 𖼢𖽻 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏,
𖼟𖽡 𖼢𖽻 𖼀𖽱𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼻𖽻𖾏 𖼀𖽱𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 .
25 𖼊𖽳𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼋𖽔𖾏, 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏；
𖼊𖽪𖾐 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐, 𖼟𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏；
26 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏, 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏.
𖼻𖽻𖾏 𖽐𖼎𖽙 𖽐𖼚𖽪𖾏、𖼺𖽘𖾏 𖼅𖽳𖾏 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼁𖽻𖾏
27 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽡𖾏 𖼙𖽳𖾏, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼏𖽻 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖽃𖽡𖾑.
28 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼯𖽡𖽙 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙 𖼐𖽑𖽡𖾏 :
“𖼅𖽻𖾏 𖼦𖽡𖾑, 𖼀𖽡 𖼟𖽡 𖼎𖽺𖾏 𖼋𖽔𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 .”
29 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐,
𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼦𖽺𖾏 𖼐𖽱𖾏 .
30 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼙𖽔 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼟𖽡 ,
𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
31 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽹𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏；
32 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐,
𖼐𖽱𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
33 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏,
𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐.
34 𖼐𖽱𖾏 𖼚𖽙𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼚𖽙𖾏 𖼗𖽔 𖽃𖽡𖾑,
𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 .
35 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼳𖽠𖾑 𖼺𖽡𖽝𖾐 𖼈𖽡𖽝𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
