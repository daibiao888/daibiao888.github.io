10
𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏， 𖼦𖽡𖾐 𖼾𖽙 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼳𖽡𖽻𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼊𖽔𖾐 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑， 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖽃𖽡𖾑 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏． 2 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙， 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡 𖼮𖽘𖾑 𖽐𖼷𖽙𖾑 𖽐𖼷𖽙𖾑 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏． 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼮𖽔𖾑 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼊𖽔𖾐 𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼖𖽡𖾏 𖼯𖽡𖾏 ，𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼚𖽠𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼋𖽘𖾏． 3 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼀𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽔𖾏 𖼳𖽡𖽻𖾏 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼾𖽔𖾏， 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖽐𖼮𖽑𖽘𖾑 𖽐𖼎𖽙 𖼮𖽑𖽷𖾏 𖽐𖼮𖽑𖽘𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼾𖽔𖾏．
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 ，𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：
“ 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑，
𖼟𖽡 𖼙𖽔 𖼑𖽜 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑．
6 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏
𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼙𖽔．
7 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：
𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏；
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ！𖼞𖽪𖾑 𖼋𖽔𖾏 𖼋𖽘 ，𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏．”
8 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏：“ 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏， 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏， 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑，𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐．” 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼖𖽑𖽘𖾑． 9 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼋𖽘， 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏．” 𖼑𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏， 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐． 10 𖼀𖽡𖾐 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽡𖾑， 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼊𖽔𖾐， 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏．
11 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖼯𖽪 𖼯𖽪𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐， 𖼋𖽡𖾏 𖼀𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼐𖽡𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼾𖽔𖾏． 12 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼮𖽡𖾑 𖼊𖽪 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼊𖽔𖾐 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏，𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 ，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏， 13 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐． 14 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼊𖽔𖾐 𖼖𖽑𖽘𖾑 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑， 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽡𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏．
15 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：
16 “𖼮𖽵𖾑 𖼐𖽑𖽡𖾏： 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，
𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏：
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐，
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐．”
17 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
“𖼞𖽪𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 ，
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳．”
