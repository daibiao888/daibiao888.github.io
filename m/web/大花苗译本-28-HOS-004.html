𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,

𖼦𖽡𖾐 𖼅𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
2 𖼈𖽡𖽝𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑、 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑、 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏、
𖽃𖽔𖾏 𖼺𖽘𖾐 𖽃𖽔𖾏 𖼸𖽜、 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏、 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳、

3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑,𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏、 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏、
𖼐𖽻 𖼟𖽪𖾏 𖼾𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾏 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼯𖽠 ,
𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼋𖽔 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
 
4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏,𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽙𖾏 𖼷𖽑𖽡𖽝𖾏 𖼾𖽱 𖼷𖽑𖽡𖽝𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
5 𖼟𖽡 𖼷𖽷𖾑 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼊𖽻𖾏 𖼢𖽻 ,
𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼢𖽻 ,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼮𖽘𖾑 .
𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
6 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏,
𖼟𖽡 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼀𖽻𖾐 𖼟𖽳,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼟𖽡 ,
𖼟𖽙𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏. 𖼟𖽡 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏.
7 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,𖼎𖽺𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏,
𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏.
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ,
𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼖𖽔𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏,𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼮𖽻𖾏,
𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏,𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽑𖽨𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼦𖽡𖾐 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑.
11 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑,
𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐, 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
12 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼊𖽪𖾑 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽠𖾏 𖽃𖽔𖾏,
𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼋𖽠𖾏 𖼸𖽠𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
13 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑、 𖼐𖽡𖽙𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏、 𖼙𖽱𖾑 𖼙𖽱、 𖼷𖽷𖾑 𖼄𖽡𖾏 𖽐𖼷𖽳𖾑 𖽐𖼊𖽻𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏,
𖼅𖽡 𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 (“𖼐𖽤𖾐 𖼮𖽑𖽦𖾐”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼅𖽡 𖼊𖽪𖾐 𖼐𖽱𖾏 𖼐𖽤𖾐”)𖼖𖽔𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏.
14 𖼅𖽡 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏,

𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠 𖼾𖽨𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼐𖽡𖽙𖾐,
