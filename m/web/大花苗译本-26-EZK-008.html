8
𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼚𖽻𖾏 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼀𖽜𖾐, 𖼞𖽪𖾑 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏, 𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼗𖽻 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 . 2 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏, 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏, 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼚𖽔𖾑 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖼺𖽱𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼮𖽦𖾑𖽑. 3 𖼐𖽱𖾏 𖼳𖽤𖾐 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼊𖽔𖾑 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼡𖽹𖾏 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼮𖽡𖽝𖾏, 𖼖𖽡𖽾 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼮𖽑𖽳𖾑 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐、 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽡𖾑 𖼯𖽡𖽳𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽃𖽡𖾑 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼮𖽵𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐, 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑. 4 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐, 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼺𖽱𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼎𖽘𖾐 𖼋𖽡𖽝𖾏 𖼁𖽙𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏.” 𖼞𖽪𖾑 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏, 𖼁𖽙𖾏 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖽃𖽔 𖼛𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏, 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼅𖽔𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑. 6 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾏 𖼘𖽪！ 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑, 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏? 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐.”
𖼁𖽙𖾏 𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏
7 𖼐𖽱𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖽃𖽔 𖼛𖽻𖾏, 𖼞𖽪𖾑 𖼑𖽔𖾏 , 𖼁𖽙𖾏 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼋𖽜𖾏 𖼮𖽑𖽤 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑. 8 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖽠𖾏 𖼮𖽑𖽤 .” 𖼞𖽪𖾑 𖽃𖽡𖾐 𖽐𖼮𖽡𖽠𖾏 𖼮𖽑𖽤 , 𖼁𖽙𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔 𖼛𖽻𖾏. 9 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼁𖽺𖾏 𖼅𖽻𖾏, 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏.” 10 𖼞𖽪𖾑 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 𖼞𖽳𖾏 𖼮𖽡𖽝𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼞𖽘 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼁𖽠𖾏. 11 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖼁𖽠𖾏 𖼐𖽡𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏, 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖼗𖽻 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏, 𖼳𖽔𖾑 𖼇𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼺𖽔𖾐 𖼐𖽡 𖼽𖽔𖾏 𖼖𖽔𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼾𖽨𖾏 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼷𖽙𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏, 𖼀𖽘𖾏 𖼡𖽔𖾏 𖼮𖽠𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐. 12 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼊𖽪𖾐 𖼗𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼋𖽘𖾏 𖼖𖽔𖾏? 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽪 𖼯𖽪𖾏 𖼐𖽑𖽡𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼀𖽡𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏.” 13 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼽𖽔𖾑 𖼳𖽪𖾐 𖼺𖽷𖾐.”
14 𖼐𖽱𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏, 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏, 𖼑𖽜 𖼊𖽔 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼐𖽡𖽝𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝. 15 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏 𖼖𖽔𖾏? 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽷𖾐.” 16 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼊𖽡𖾑 𖼯𖽡𖽳𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽐𖼚𖽙 . 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏、 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼡𖽘𖾐 𖽃𖽔𖾏 𖽐𖼮𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽘 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼊𖽡𖾑 𖼢𖽻𖾏 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽱𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 , 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐. 17 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪! 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏? 𖼽𖽨 𖼊𖽔𖾏 𖼾𖽡𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽪𖾏 𖼺𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼺𖽷𖾐 𖼘𖽪𖾏? 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽪𖾏 𖼸𖽳, 𖼯𖽪 𖼯𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐. 𖼑𖽔𖾏 𖼘𖽪！ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼯𖽡 𖽐𖼊𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼀𖽱 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏. 18 𖽃𖽔𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼳𖽻𖾑 𖼅𖽳𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼐𖽱𖾏 𖼸𖽻𖾏.”
