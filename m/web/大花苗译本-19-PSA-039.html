𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏 𖼽𖽦𖾑 𖼊𖽪𖾏 𖼊𖽪𖽝𖾏.
39
𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼖𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼞𖽪𖾑 𖼖𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼚𖽱𖾐 𖼞𖽑𖽱𖾐 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏.”
2 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐, 𖼻𖽻𖾏 𖼗𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏,
𖼞𖽪𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐.
3 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽺𖾐 𖼳𖽙𖾑,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼯𖽡,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏：
 
4 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽔𖾑 𖼺𖽻𖾏，
𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠，
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼗𖽡𖾏！
5 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖽐𖼡𖽹𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼷𖽑𖽹𖾏 𖽐𖼚𖽡𖽝𖾏 𖼋𖽡.
𖼞𖽪𖾑 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏．
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽺𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼀𖽘𖾏 𖼊𖽔𖾐. （𖼳𖽡𖾏 𖼖𖽔𖾑）
6 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼊𖽔𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽠𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑,
𖼮𖽷𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼳𖽻𖾏.
 
7 𖼮𖽵𖾑 𖼘𖽪！ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏?
𖼞𖽪𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 ！
8 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼾𖽔𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽐𖼮𖽷𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼟𖽡 ,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ．
10 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾏 𖼨𖽑𖽢𖾐 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏,
𖼑𖽜 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼡𖽑𖽔𖾐, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘.
11 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑,
𖼟𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽡𖾑 𖼾𖽘𖾏 𖼙𖽠𖾏, 𖼺𖽱𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼞𖽘𖾐 𖼑𖽻𖾏 𖼺𖽘𖾏,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼀𖽘𖾏 𖼊𖽔𖾐． （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜！
𖼞𖽪𖾑 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾏,
𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐, 𖼺𖽱𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐.
13 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼗𖽙 𖽃𖽡𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼚𖽙𖾏.”
