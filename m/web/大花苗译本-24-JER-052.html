（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 24：18 - 25：7）
52
𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼦𖽔𖾐 𖼄𖽙𖾏 𖼊𖽑𖽔𖾑 , 𖼟𖽪𖾏 𖼖𖽡 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖽐𖼷𖽑𖽹𖾏. 2 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑. 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 , 𖼸𖽙 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏, 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼳𖽠𖾑 𖽐𖼷𖽜𖾑 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏. 4 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼖𖽪𖾑 𖼟𖽻𖾏 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼟𖽻𖾏, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖼙𖽻 𖼮𖽪𖾐 𖼾𖽡 , 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖽐𖼊𖽻𖾏 𖼙𖽻 .
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼺𖽘𖾏
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼙𖽻 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖽃𖽡𖾑 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑. 6 𖼖𖽪𖾑 𖼚𖽻𖾐 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼯𖽢𖾐, 𖽐𖼚𖽙 𖼙𖽻 𖼅𖽔𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐. 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼑𖽻𖾏 𖼋𖽘. 7 𖼙𖽻 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼁𖽘𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼙𖽻 𖼏𖽘 𖼀𖽜𖾑, 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖽐𖼊𖽻𖾏 𖼙𖽻 𖼷𖽷𖾏 𖼖𖽘𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼺𖽘𖾑 𖽃𖽡𖾑. 8 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼏𖽙 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖼖𖽠𖾑 𖼯𖽢 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼞𖽑𖽔𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏. 9 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽡𖽝𖾏, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼖𖽡𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼙𖽻 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏.
𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽻𖾏 𖼞𖽳𖾏
10 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽡𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼐𖽑𖽙𖾏 𖼺𖽘𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏. 11 𖼎𖽺𖾏 𖼞𖽳𖾏 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼺𖽘𖾏, 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼋𖽻𖾏 𖼎𖽪𖾑 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏, 𖼞𖽱𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼮𖽡𖽝𖾏, 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔 𖼙𖽠𖾏.

12 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏, 𖼖𖽪𖾑 𖼀𖽜𖾐 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼟𖽻𖾏, 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽪𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽡𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐, 13 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖼟𖽻 , 𖼎𖽺𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐, 𖼐𖽱𖾏 𖼞𖽑𖽔𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏. 14 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽃𖽡𖾑, 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 . 15 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽪𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼄𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾏 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼙𖽻 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼊𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼚𖽻𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 16 𖼈𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽪𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾏 𖽃𖽡𖾑 𖼮𖽡𖽝𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏, 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖼯𖽡𖽙 𖼡𖽻𖾐.
𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏
17 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼋𖽻𖾏, 𖼊𖽑𖽡𖽝𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽜𖾐 𖼖𖽪𖾑 𖼋𖽺𖾏, 𖽐𖼎𖽙 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏, 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼋𖽻𖾏 𖽃𖽡𖾑 𖼨𖽑𖽹𖾑 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏 𖼋𖽘𖾏， 18 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼀𖽑𖽻𖾏、𖼊𖽪𖾑 𖼞𖽻𖾏 𖼀𖽑𖽻𖾏、𖼊𖽪𖾑 𖼷𖽑𖽳𖾏 𖼟𖽪𖾏 𖼷𖽑𖽳𖾏 𖽃𖽔𖾏 𖼐𖽑𖽨𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐、𖼞𖽑𖽙𖾑 𖼈𖽳𖾐、𖽃𖽔𖾑 𖼯𖽱𖾏 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑、 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏、 19 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼊𖽡 𖽐𖼮𖽪、𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐、𖼞𖽑𖽙𖾑 𖼈𖽳𖾐、𖼀𖽑𖽻𖾏、𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐、𖽃𖽔𖾑 𖼯𖽱𖾏 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑、𖼨𖽑𖽻𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑, 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼞𖽪𖾐 𖽃𖽔𖾏、 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏, 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 20 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼋𖽻𖾏 𖼟𖽪𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏, 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼋𖽪 𖼑𖽨𖾏 𖼋𖽻𖾏, 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽻𖾏 𖼐𖽡𖾑, 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼙𖽪𖾏. 21 𖼊𖽪𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼐𖽡𖾑 𖼺𖽡𖽝𖾐 𖼟𖽻𖾏 𖼾𖽡𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼊𖽔𖾐 𖼚𖽻𖾐 𖽐𖼊𖽡𖾏, 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼚𖽙 , 𖽃𖽡𖾐 𖼞𖽑𖽡𖾏 𖼉𖽜𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏. 22 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼘𖽙𖾏 𖼋𖽻𖾏, 𖼺𖽡𖽝𖾐 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼋𖽻𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖼅𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾐 𖼊𖽑𖽡𖽝𖾐 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼊𖽔𖾐, 𖼊𖽪𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼸𖽙𖾏 𖼊𖽪𖾑 𖼐𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐, 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑. 23 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼅𖽔𖾏 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼚𖽻𖾏 𖼗𖽪𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑, 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽪𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑.
