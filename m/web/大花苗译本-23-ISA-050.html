50
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼞𖽪𖾑 𖽐𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼺𖽘𖾏,
𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽺 𖼅𖽡 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼮𖽷𖾏 𖼮𖽵𖾑 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑?
𖼅𖽡 𖼊𖽻𖾏 𖼋𖽺 , 𖼟𖽪𖾏 𖼑𖽜 𖼅𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼅𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼊𖽻𖾏 𖽐𖼎𖽻𖾏 𖼺𖽘𖾏, 𖼟𖽪𖾏 𖼑𖽜 𖼅𖽡 𖼾𖽔𖾏.
2 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽔𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽺𖾑?
𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼘𖽪?
𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽂𖽪𖾏 𖼖𖽔𖾏 , 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼡𖽑𖽔𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼞𖽪𖾑 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏,
𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽻𖾐 𖼎𖽺𖾏 𖼮𖽵𖾏 𖼗𖽱𖾏 , 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼋𖽔 𖼙𖽠𖾏.
3 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽑𖽡𖽠𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑,
𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖽃𖽔𖾏 𖼛𖽔𖾏 𖼟𖽪𖾏 𖼈𖽳𖾑 𖽐𖼀𖽙 .”
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼑𖽡𖾏 𖼑𖽻 𖼸𖽙𖾏
4 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖽃𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 ,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏 𖽃𖽡𖾑．
𖼮𖽵𖾑 𖼷𖽑𖽔𖾏 𖼊𖽡𖾏 𖼟𖽡𖾏 𖼮𖽘𖾏 𖼯𖽷𖾏 𖼞𖽪𖾑, 𖼮𖽘𖾏 𖼯𖽷𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼸𖽠𖾏 𖼑𖽻 , 𖼺𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
5 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽜𖾐 𖼊𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼳𖽠𖾑 𖽐𖼷𖽜𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐.
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 𖼊𖽡𖾑 𖼢𖽻𖾏, 𖼞𖽪𖾑 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽙𖾐 𖼞𖽪𖾑 𖼚𖽻𖾐 𖽃𖽔𖾏 𖼸𖽷𖾏, 𖼞𖽪𖾑 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼚𖽙𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑 𖽐𖼊𖽙𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏.
 
7 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏．
𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼼𖽔𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼏𖽔𖾏,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
8 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽺𖾐 𖼊𖽡𖾏 𖼈𖽳𖾏 𖼞𖽪𖾑 ,
𖼡𖽔 𖼋𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼖𖽺𖾑?
𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏,
𖼡𖽔 𖼋𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐 𖼖𖽺𖾑?
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑.
9 𖼑𖽔𖾏 𖼘𖽪 ! 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑,
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼨𖽑𖽢𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼖𖽺𖾑?
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾏 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼀𖽑𖽙𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼊𖽻𖾏 𖼀𖽡𖾑 𖼞𖽘𖾐 𖼮𖽷𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏.
 
10 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼗𖽪𖾏,
𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏,
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏,
𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
11 𖼑𖽔𖾏 𖼘𖽪!𖼅𖽡 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐,
𖼯𖽡𖽙𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼾𖽨𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼎𖽻𖾐 𖽐𖼎𖽙 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽠𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼯𖽡𖽙𖾏,
𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽱𖾏 𖼎𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
