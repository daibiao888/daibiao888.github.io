𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏.
101
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼨𖽑𖽪𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡！
2 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼟𖽡 𖼯𖽡𖽹𖾏 𖼋𖽱𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
3 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼚𖽘𖾐, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 , 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼘𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 .
4 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 (“𖼑𖽪𖾏 𖽐𖼚𖽪𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼊𖽪𖾐 𖽐𖼚𖽪𖾐”) 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐.
5 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼐𖽱𖾏 𖼼𖽷𖾏 𖼼𖽙 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼖𖽻𖾐 𖼊𖽪𖾏 𖼮𖽘𖾑 𖼺𖽘𖾏,
𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼗𖽱𖾏 𖽃𖽔𖾏, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼯𖽔𖾏 𖼐𖽱𖾏 .
6 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑.
7 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
8 𖼞𖽪𖾑 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼟𖽙𖾏 𖼊𖽪𖾏 𖼮𖽘𖾑 𖼺𖽘𖾏,
𖼻𖽻𖾏 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼚𖽙 𖼙𖽻 𖼖𖽻𖾐 𖼺𖽘𖾏.
