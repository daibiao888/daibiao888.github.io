“𖼟𖽡 𖼮𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼑𖽨𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼮𖽑𖽷𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖾏 𖼁𖽜𖾏 𖼞𖽡𖾑 𖼙𖽠𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽤𖾐 𖼐𖽱𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 . 2 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼮𖽔𖾑 𖼡𖽔𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼚𖽡𖾐 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏 , 𖼽𖽔𖾑 𖼮𖽤𖾐 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 , 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼋𖽔𖾏 𖽐𖼎𖽑𖽦𖾑 , 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 . 3 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼛𖽙𖾏 𖼙𖽠𖾏 𖼀𖽻𖾐 𖼙𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽜𖾏 𖼖𖽔𖾑、𖼡𖽑𖽙𖾑 𖼯𖽢 𖼮𖽑𖽙𖾏、𖼦𖽡𖾐 𖼞𖽳𖾑 𖼋𖽪 𖼡𖽔𖾏 𖼳𖽷𖾏 , 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 , 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 . 4 𖼟𖽡 𖼮𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽨𖾏 𖼢𖽻 𖽐𖼎𖽙𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼮𖽤𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏 .
5 𖽃𖽔 𖼁𖽙𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽘𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼐𖽑𖽘𖾑 , 𖽃𖽔𖾑 𖼾𖽠 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽘𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼮𖽑𖽙𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼳𖽪𖾐 𖼋𖽡𖽝 .
6 𖼟𖽡 𖼮𖽔𖾑 𖼅𖽻𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼁𖽙𖾏 𖼉𖽳𖾏 𖼐𖽻, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 , 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼅𖽔𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼐𖽻 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼅𖽔𖾏 𖼡𖽳𖾏 𖼐𖽻 , 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼐𖽻 𖼁𖽔 𖼐𖽡𖽙𖾐 𖼣𖽔𖾑 𖼣𖽔𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼡𖽳𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼞𖽑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼊𖽑𖽡𖽝𖾐 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼙𖽠𖾏. 7 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽙𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼺𖽘𖾏 , 𖼮𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼊𖽜𖾑 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 .
8 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 , 𖽐𖼞𖽔𖾏 𖼏𖽪𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼽𖽔𖾑 𖼙𖽻𖾏 𖼮𖽡𖽝𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼀𖽻𖾐 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 , 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼟𖽡 𖼾𖽡𖾏 .
9 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼮𖽘𖾏 𖼮𖽘𖾑 𖼯𖽡𖽙 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽳𖾐 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼽𖽔𖾑 𖼺𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼁𖽡𖽝 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏. 10 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼑𖽨𖾏、𖼑𖽜𖾏 𖼖𖽔𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 . 11 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼚𖽻𖾐 𖼾𖽘𖾐 𖼊𖽑𖽡𖽝𖾐 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏 𖽃𖽔𖾏 .
