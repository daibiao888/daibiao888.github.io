𖼀𖽡𖾐 𖼺𖽔𖾐 𖼖𖽡𖽝 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼦𖽙 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼀𖽳 , 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 , 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼀𖽻𖾐 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 , 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 .
2 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼋𖽔𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼼𖽙 𖼐𖽡𖾑 𖽃𖽡𖾑 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼺𖽔𖾐 𖼖𖽡𖽝 𖼊𖽑𖽡𖽝𖾐 𖼽𖽔𖾏 𖼦𖽙 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼀𖽳 𖽃𖽡𖾐 𖼸𖽻 𖼨𖽑𖽪𖾏 𖼋𖽔𖾏 . 3 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼳𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼡𖽑𖽙𖾑 𖼯𖽡𖽙𖾏 𖼋𖽔𖾐 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖽃𖽡𖾑 . 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼷𖽑𖽔𖾏 𖼊𖽡𖾏 𖼟𖽡𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 . 4 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽡𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼼𖽙 , 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 , 5 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼯𖽡𖽙𖾏 𖼋𖽔𖾐 𖽃𖽡𖾑 , 𖼊𖽪𖾏 𖼀𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼷𖽻𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 .” 6 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼞𖽳𖾑 𖽃𖽔𖾑 𖼾𖽠 𖽃𖽔 𖼁𖽙𖾏, 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼎𖽺𖾏 𖼑𖽜 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼋𖽘𖾏 .” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖽐𖼎𖽳𖾏 𖼺𖽘𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽔𖾏 . 7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 , 𖼷𖽻𖾏 𖽃𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼺𖽷𖾐 .
𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐
（𖼋𖽠𖾏 26：1 - 37）
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽡𖾑 , 𖼎𖽑𖽻𖾏 𖼟𖽻𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼗𖽙𖾏 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 , 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼅𖽘𖾏 𖽐𖼎𖽙 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖼡𖽠𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 , 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼳𖽡𖾑 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼎𖽻𖾏 . 9 𖼷𖽑𖽔𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼾𖽡𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼇𖽘𖾑 𖼚𖽻𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼟𖽪𖾏 𖽐𖼊𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 . 10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽙𖾏 𖼀𖽜𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼛𖽔𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 , 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼀𖽜𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖽃𖽡𖾑 𖼷𖽑𖽔𖾏 𖼛𖽔𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 . 11 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼐𖽡𖾑 𖼛𖽔𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼇𖽺𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑 , 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖽃𖽡𖾑 , 𖼛𖽔𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 . 12 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽪𖾏 𖼡𖽑𖽙𖾑 𖼇𖽺𖾑 , 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽪𖾏 𖼡𖽑𖽙𖾑 𖼇𖽺𖾑 , 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐． 13 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼋𖽪𖾏 𖽐𖼡𖽜𖾏 𖼞𖽪𖾐 , 𖼟𖽙𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 . 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼊𖽡𖽝𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐.
