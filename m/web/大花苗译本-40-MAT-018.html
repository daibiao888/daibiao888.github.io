（𖼞𖽑𖽙𖾐 9：33-37；𖼖𖽪𖾏 9：46-48）
18
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 ： “𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐？” 2 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐， 3 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏，𖼅𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙． 4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼡𖽔 𖼋𖽱𖾏 𖼾𖽨𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼐𖽡𖾑，𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼯𖽡𖽙 ． 5 𖼡𖽔 𖼋𖽱𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼷𖽹𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ， 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏． 6 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏 ， 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼢𖽪𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖼚𖽘𖾐，𖼗𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼊𖽙𖾐，𖼎𖽺𖾏 𖼖𖽔𖾑 𖼻𖽻𖾏 𖼺𖽷𖾐． 7 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏,𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼯𖽡𖽙 ， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏． 8 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏，𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼾𖽔𖾏， 𖼽𖽔𖾑 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏．𖼟𖽡 𖼟𖽪𖾏 𖼋𖽱𖾐 𖼋𖽱𖾐 𖼋𖽡𖾏 𖼋𖽱𖾐 𖼋𖽱𖾐 𖼊𖽠𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏，𖼎𖽺𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏，𖼊𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼯𖽘𖾏 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏． 9 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼾𖽔𖾏，𖼽𖽔𖾑 𖼞𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏．𖼟𖽡 𖼯𖽠 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾑，𖼎𖽺𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼷𖽷𖾏 𖼞𖽡 𖼑𖽡𖽻 𖼯𖽘𖾏 𖼋𖽠．”
𖼾𖽘𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼗𖽠𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏

21 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼀𖽡𖾐 𖼊𖽳 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼟𖽻𖾐 𖼖𖽺𖾑？ 𖼸𖽙 𖼳𖽤𖾏 𖼟𖽻𖾏 𖼋𖽡𖽝𖾏 𖼷𖽻𖾏 𖼋𖽘𖾏？” 22 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏，𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼸𖽙 𖼳𖽤𖾏 𖼟𖽻𖾏 𖼊𖽔𖾐，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼋𖽪𖾏 𖼳𖽤𖾏 𖼟𖽻𖾏． 23 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼮𖽔𖾑 𖽐𖼮𖽷𖾏． 24 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽔𖾑 𖽃𖽡𖾑，𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼏𖽷𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 （“𖽃𖽡𖾐 𖼉𖽻𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳” 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼚𖽻𖾐 𖼉𖽻𖾏 𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐 𖼞𖽡𖾏） 𖽐𖼮𖽷𖾏 𖼋𖽔𖾏． 25 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼁𖽻𖾏 ， 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏，𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼋𖽺 𖼺𖽘𖾏 𖼁𖽻𖾏． 26 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：‘ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖽐𖼷𖽜𖾏 𖼋𖽙𖾏 𖼞𖽪𖾑 ， 𖼞𖽪𖾑 𖼸𖽠𖾏 𖼁𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽻𖾏 𖼟𖽡．’ 27 𖼡𖽑𖽜𖾑 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼏𖽷𖾏． 28 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏，𖽐𖼮𖽷𖾐 𖼏𖽻 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔，𖼟𖽪𖾏 𖼏𖽷𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼛𖽡𖾏 𖼑𖽡𖽝𖾏 𖽐𖼮𖽷𖾏（𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 “𖼊𖽡𖾏 𖼐𖽔𖾏 𖼖𖽨”，𖼘𖽙 𖼄𖽔𖾐 𖼑𖽡𖽝𖾏 𖼷𖽑𖽷𖾐），𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼊𖽔𖾑 𖼐𖽱𖾏 𖼺𖽹𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖼚𖽘𖾐 𖼐𖽑𖽡𖾏：‘ 𖼟𖽡 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽷𖾏 𖼞𖽪𖾑 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑 𖼁𖽻𖾏 𖼞𖽪𖾑．’ 29 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 𖼖𖽔𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ： ‘ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖽐𖼷𖽜𖾏 𖼋𖽙𖾏 𖼞𖽪𖾑 ，𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 ．’ 30 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼺𖽘𖾏，𖼋𖽙𖾏 𖼐𖽱𖾏 𖼞𖽑𖽔𖾏 𖼁𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽷𖾏． 31 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 𖼅𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼅𖽻𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑． 32 𖼮𖽵𖾑 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏，𖼐𖽑𖽡𖾏：‘ 𖼟𖽡 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑 ，𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽷𖾏 𖽃𖽡𖾑 𖼯𖽔𖾏 𖼺𖽘𖾏， 33 𖼟𖽡 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽡 𖼽𖽺 𖼗𖽔，𖼺𖽱𖾏 𖼞𖽪𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽡 𖼘𖽪𖾏？’ 34 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏，𖼋𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼞𖽑𖽔𖾏 𖽐𖼮𖽷𖾏 𖼁𖽻𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏． 35 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼯𖽔𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼾𖽔𖾏，𖼞𖽪𖾑 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏，𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼅𖽡 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏．”
