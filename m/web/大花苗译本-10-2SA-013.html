𖼣𖽔𖾏 𖼐𖽺𖾏 𖽐𖼎𖽙 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐
13
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼞𖽱𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏, 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼣𖽔𖾏 𖼐𖽺𖾏 𖽃𖽺𖾐 𖼏𖽻 𖼐𖽱𖾏. 2 𖼣𖽔𖾏 𖼐𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖽐𖼊𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏, 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼟𖽪𖾏 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 𖼺𖽷𖾐, 𖼣𖽔𖾏 𖼐𖽺𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽡 𖼟𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏. 3 𖼣𖽔𖾏 𖼐𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔 , 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽙 𖼐𖽔 𖼊𖽔, 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼳𖽷𖾏 𖼄𖽡𖾐 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐． 𖼷𖽷𖾑 𖼽𖽙 𖼐𖽔 𖼊𖽔 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝. 4 𖼐𖽱𖾏 𖼑𖽪 𖼣𖽔𖾏 𖼐𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼘𖽪！ 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼐𖽔𖾑 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼾𖽔 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼖𖽺𖾑? 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑.” 𖼣𖽔𖾏 𖼐𖽺𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼏𖽻 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼐𖽔𖾑.” 5 𖼽𖽙 𖼐𖽔 𖼊𖽔 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼋𖽔𖾏 𖼑𖽔𖾏 𖼟𖽡 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼋𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼑𖽻𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼻𖽻𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼗𖽙𖾏 𖼑𖽻𖾏.” 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼣𖽔𖾏 𖼐𖽺𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼸𖽱𖾏 𖼸𖽘𖾏． 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽢𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼋𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼮𖽡𖽙𖾏 𖼣𖽔𖾏 𖼳𖽡𖽝𖾐, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼗𖽙𖾏 𖼑𖽻𖾏.”
7 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼚𖽙 𖼟𖽻, 𖽐𖼎𖽙 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼣𖽔𖾏 𖼐𖽺𖾏 𖽐𖼞𖽔𖾏, 𖼑𖽜 𖼐𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏.” 8 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼣𖽔𖾏 𖼐𖽺𖾏 𖽐𖼞𖽔𖾏, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼣𖽔𖾏 𖼐𖽺𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼮𖽑𖽡𖽠𖾏． 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼳𖽡𖽝𖾐, 𖼎𖽺𖾏 𖽐𖼊𖽑𖽜𖾏 𖼟𖽪𖾏 𖼳𖽦𖾑 . 9 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼀𖽑𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼈𖽡𖽝𖾏 𖼣𖽔𖾏 𖼐𖽺𖾏𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻𖾏, 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼙𖽠𖾏 !” 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 10 𖼣𖽔𖾏 𖼐𖽺𖾏 𖽐𖼎𖽙 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼯𖽡𖽙𖾏 𖼳𖽡𖽝𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼀𖽱𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼗𖽙𖾏 𖼑𖽻𖾏.” 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑, 𖼯𖽡𖽙𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽤𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼀𖽱𖾏, 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼣𖽔𖾏 𖼐𖽺𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑. 11 𖼞𖽑𖽡𖽠𖾏 𖼳𖽡𖽝𖾐 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼑𖽻𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼊𖽔𖾑 𖼮𖽡𖽝𖾏 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼐𖽑𖽡𖾏: “ 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼘𖽪！ 𖼟𖽡 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼀𖽱𖾏.” 12 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼘𖽪！ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼞𖽪𖾑！ 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼚𖽘𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑. 13 𖼟𖽡 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏, 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼯𖽡𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼖𖽺𖾑? 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽔𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔. 𖼮𖽔𖾑 𖼖𖽡𖾏 𖼯𖽡𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼎𖽳𖾏 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼺𖽙𖾏 𖼟𖽡.” 14 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼣𖽔𖾏 𖼐𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼗𖽪𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾐 𖼊𖽪𖾑 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
