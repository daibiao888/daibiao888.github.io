𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼦𖽡𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼳𖽡𖽝𖾏 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼘𖽔𖾑𖽑 , 𖼺𖽔𖾐 𖼊𖽡𖾐 𖼑𖽡𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼮𖽢𖾑 𖼄𖽡𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼞𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 .
2 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 , 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾏 𖼣𖽹𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼀𖽳 · 𖼽𖽔𖾏 𖽂𖽺 , 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 .” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼣𖽹𖾏 𖼙𖽻 . 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽷𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼅𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼞𖽑𖽔𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖼮𖽔 𖼽𖽔𖾑 𖽃𖽔𖾐 𖼷𖽷𖾐 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 , 𖼯𖽡𖾏 𖼽𖽔𖾑 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼣𖽹𖾏 𖼙𖽻 , 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼚𖽑𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼯𖽠 .” 4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼊𖽔𖾐 , 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼣𖽹𖾏 𖼙𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼏𖽘 𖼀𖽜𖾑 . 5 𖼣𖽹𖾏 𖼙𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼚𖽻𖾏 𖼗𖽜𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑 𖼖𖽡 , 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖽃𖽔𖾑 𖼼𖽘 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼾𖽘𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 .
