𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼷𖽹𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 10：1 - 19）
12
𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼅𖽻𖾏 𖼛𖽔𖾏𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼙𖽻 , 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 2 𖼐𖽡 𖼀𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼐𖽙𖾑𖽑 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 . 3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼮𖽘𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏, 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 4 “𖼟𖽡 𖼉𖽺𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐, 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼳𖽷𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡.” 5 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏, 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑.” 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏.
𖼦𖽡𖾐 𖼸𖽙𖾏 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑
6 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏, 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼅𖽔𖾏 𖼋𖽪 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏? 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼊𖽡𖾐 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑.” 7 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽔𖾑 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑, 𖼎𖽑𖽻𖾏 𖼻𖽻𖾏 𖼗𖽪𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏.”
𖼑𖽻 𖼸𖽙𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑
8 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼘𖽑𖽙𖾐, 𖼟𖽪𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼦𖽡𖾏 𖼊𖽳𖾏， 9 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼉𖽺𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼳𖽷𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏, 𖼅𖽡 𖼅𖽔𖾏 𖼋𖽪 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏.” 10 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏, 𖼟𖽡 𖼉𖽺𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼳𖽷𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏． 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡 𖼐𖽡𖾑: ‘ 𖼞𖽪𖾑 𖼊𖽔𖾑 𖽃𖽔𖾏 𖼯𖽱𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏, 𖽐𖼊𖽪 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼖𖽪𖾑 𖼚𖽔𖾑. 11 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖽐𖼚𖽠𖾏 𖼅𖽡, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽦𖾏 𖼖𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖼅𖽡！’ ”
12 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏, 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼅𖽡 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑， 𖼸𖽙 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼋𖽘. 13 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽹𖾑 𖼊𖽡𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏， 14 𖼸𖽙𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼉𖽺𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽃𖽡𖾑！ 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖽐𖼚𖽠𖾏 𖼅𖽡, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽦𖾏 𖼖𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖼅𖽡！” 15 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼸𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼳𖽷𖾏 𖼘𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼽𖽔𖾐 𖽐𖼎𖽙 𖼐𖽡 𖼀𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑.
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖽐𖼷𖽜𖾑
