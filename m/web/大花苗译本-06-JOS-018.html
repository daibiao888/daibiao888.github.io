𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽷𖾏 𖼘𖽙 , 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 , 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼏𖽻 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏 . 2 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼳𖽡𖾏 , 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 3 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖽐𖼎𖽙𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : “𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡 , 𖼅𖽡 𖼊𖽳𖾑𖽑 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏 𖼊𖽻𖾏 , 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑 ? 4 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 𖼽𖽔𖾑 𖼺𖽹𖾑 𖼋𖽠𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼺𖽻𖾏 𖽐𖼎𖽑𖽹𖾏 , 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 𖽐𖼮𖽡𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 . 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼳𖽤𖾏 𖽐𖼀𖽠 , 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 , 𖼽𖽙 𖼺𖽳 𖼾𖽡𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 . 6 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼳𖽤𖾏 𖽐𖼀𖽠 , 𖼺𖽻𖾏 𖽐𖼎𖽑𖽹𖾏 𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 . 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽜 𖼅𖽡 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠． 7 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼀𖽠 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 . 𖼮𖽢𖾑 𖼊𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼖𖽨 𖼀𖽡𖽝𖾏 𖼳𖽡𖾏 𖽐𖼎𖽙 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖽐𖼊𖽘𖾏 𖼳𖽡𖾏 , 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽡𖾑 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼋𖽘 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 .”
