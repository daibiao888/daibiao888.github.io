28
“𖼑𖽡𖽝𖾏 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖽠𖾏 ,
𖼞𖽪𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽡𖽺𖾏.
2 𖼖𖽑𖽻𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼮𖽡𖽠𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼋𖽻𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼷𖽑𖽷𖾑 𖼾𖽘𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏,
𖼎𖽺𖾏 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼋𖽻𖾏 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 ,
𖼸𖽙 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 .
4 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼷𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼋𖽻𖾏 𖼖𖽑𖽻𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 ,
𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 , 𖼮𖽘𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼅𖽻𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼗𖽙𖾏 .
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐 , 𖼸𖽠𖾏 𖼋𖽠𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 ,
𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼺𖽱𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼦𖽡𖾏 𖽐𖼷𖽜𖾑 𖼗𖽙𖾏 .
6 𖼊𖽡𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 , 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑 𖽐𖼷𖽔𖾐 ,
𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼯𖽷𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼞𖽪𖾐 .
7 𖼚𖽘𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼋𖽻𖾏 𖼖𖽑𖽻𖾏 ,
𖼙𖽡𖽝𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 .
8 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼯𖽪𖾏 𖼷𖽠𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 ,
𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼮𖽡𖽻𖾏 𖼉𖽺𖾏 .
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼸𖽻 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼼𖽔𖾏 ,
𖼦𖽡𖾏 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏 .
10 𖼐𖽡𖽙𖾐 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽐𖼚𖽙 𖼸𖽻 𖼋𖽠𖾏 𖼟𖽱𖾏 𖽃𖽻𖾐 ,
𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 .
11 𖼐𖽱𖾏 𖽐𖼮𖽠𖾏 𖼘𖽙𖾏 𖼮𖽡𖽝𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼮𖽑𖽻𖾏 𖼙𖽻𖾏 ,
𖼟𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 .
 
12 𖼈𖽡𖽝𖾏 , 𖼽𖽔𖾑 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 ?
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑 ?
13 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼡𖽜𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 ,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 .
14 𖽃𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼐𖽑𖽡𖾏 : 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 ,
𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼐𖽑𖽡𖾏 : 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 .
15 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼅𖽔𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖽐𖼡𖽜𖾏 .
16 𖽃𖽳 𖼇𖽺𖾑 𖼞𖽪𖾐 𖽐𖼎𖽙 𖼄𖽔𖾐 𖼐𖽻𖾐 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,
𖽐𖼎𖽙 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑 𖽐𖼷𖽔𖾐, 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼮𖽔𖾏 .
17 𖼞𖽪𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼮𖽔𖾏 ,
𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽃𖽔𖾏 , 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖼘𖽑𖽪𖾏 .
18 𖼳𖽗𖾑 𖼦𖽪 、 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼎𖽻𖾏 𖽐𖼎𖽷𖾑, 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼮𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 ,
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏 𖼮𖽑𖽡𖽻𖾏 𖼇𖽔𖾑 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 (“𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 ” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽡𖽝𖾐”).
19 𖼞𖽪𖾐 𖼳𖽷 𖼑𖽡𖾏 𖼽𖽱𖾏 𖼟𖽪𖾏 𖼉𖽘𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼦𖽡𖾏 𖽐𖼮𖽔𖾏 ,
𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼮𖽔𖾏 .
 
20 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏 𖼖𖽺𖾑 ?
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑 ?
21 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼖𖽔𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 ,
𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼺𖽱𖾑 𖼑𖽡𖾏 .
22 𖼖𖽻𖾐 𖼷𖽑𖽪𖾐 𖼊𖽑𖽡𖽝𖾐 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏 : 𖼀𖽡𖾐 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼊𖽪𖾑 𖽐𖼷𖽷𖾏 𖼐𖽡𖾑 .
 
23 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼎𖽺𖾏 𖼀𖽻𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 .
24 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼸𖽙 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏 ,
𖼑𖽔𖾏 𖽐𖼷𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 ，
25 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼮𖽢𖾏 𖼊𖽡𖾏 𖼏𖽵 𖼐𖽑𖽤𖾑 𖼳𖽷𖾐 ,
𖼎𖽺𖾏 𖽐𖼮𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽻𖾐 .
