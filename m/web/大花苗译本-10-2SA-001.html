𖼺𖽻𖾐 𖼘𖽙 𖼋𖽔 𖼗𖽠𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼊𖽻𖾏 𖼊𖽔𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽷𖾐 𖼗𖽙, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽡𖾐 𖼞𖽳 𖼖𖽔𖾑 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽔𖾐 𖼐𖽑𖽪𖾐. 2 𖼸𖽙 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼖𖽪𖾑 𖼾𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏, 𖽐𖼀𖽑𖽻𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏 𖼎𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑. 3 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏?” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽪𖾑 𖼾𖽡 𖼀𖽜𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 .” 4 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽜𖾑 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑， 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 ?” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “ 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼏𖽘 𖼀𖽜𖾑, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼋𖽔 𖼙𖽠𖾏, 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼖𖽔𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.” 5 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽪 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼋𖽔 𖼙𖽠𖾏 𖼖𖽺𖾑?” 6 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼸𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡 𖼀𖽙𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼞𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼊𖽪𖾑 𖼄𖽑𖽪𖾑, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽐𖼎𖽙 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑. 7 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑, 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑. 8 𖼐𖽱𖾏 𖼑𖽪 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏? 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐. 9 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼷𖽑𖽜𖾑 𖼟𖽡𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼐𖽑𖽙𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼊𖽔𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑, 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽔. 10 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼢𖽻 𖼙𖽠𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏, 𖼞𖽪𖾑 𖼘𖽑𖽱𖾑 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼁𖽻 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 ，𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑.”
11 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼮𖽑𖽷𖾏 𖼾𖽨𖾏 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏. 12 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏, 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼸𖽙 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼟𖽪𖾏 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼊𖽑𖽡𖽝𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼐𖽔𖾑. 13 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽪 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐.” 14 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼖𖽺𖾑?” 15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼨𖽑𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏.” 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏. 16 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼋𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼐𖽑𖽙𖾏 𖼺𖽘𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑.”
𖼑𖽜 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐
17 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽜 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖽃𖽔𖾏 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼐𖽡𖾑， 18 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼟𖽙𖾏 𖼄𖽔𖾐 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 𖼐𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 𖼐𖽡𖾑 𖽐𖼷𖽷𖾏 “𖼐𖽜𖾑𖽑 𖽐𖼞𖽻𖾏”, 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏《 𖼽𖽔𖾐 𖼳𖽔 𖽃𖽳𖾐 𖽐𖼊𖽠𖾑 》𖼮𖽡𖽝𖾏. 19 𖽐𖼞𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪！ 𖼟𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏！
𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼋𖽔 𖼙𖽠𖾏！
20 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼎𖽻𖾏 𖼄𖽻𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼳𖽷 𖼮𖽡𖾑 𖼖𖽪𖽝 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖽃𖽡𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏.
 
21 𖼮𖽡𖾑 𖼖𖽡 𖼀𖽙𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽻𖾐 𖼑𖽘！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼑𖽡𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏,
𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼚𖽜𖾑 𖼛𖽙𖾏.
 
22 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼐𖽜𖾑𖽑 𖼈𖽙𖾏 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽐𖼚𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑 𖼦𖽡𖾐 𖼑𖽡𖾏 𖼡𖽜𖾐,
𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼄𖽑𖽪𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼀𖽑𖽺𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼑𖽡𖾏 𖼛𖽙𖾏 𖼯𖽡𖾏 𖼦𖽡𖾐 𖼳𖽻𖾏 𖽐𖼊𖽡𖾏.
 
23 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼯𖽡𖽝𖾏 𖼺𖽷𖾐 𖽃𖽡𖾑, 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏.
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼇𖽺𖾐,
𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼇𖽺𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾑 𖼚𖽘𖾑, 𖼋𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐.
 
24 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪！ 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏！
𖼐𖽱𖾏 𖼙𖽔 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼟𖽙𖾏 𖼅𖽡 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼅𖽔𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏.
 
25 𖼊𖽪𖾐 𖼋𖽻 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼢𖽻 𖼙𖽠𖾏！
𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏！
26 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼘𖽪 ! 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐！
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼙𖽔 𖼟𖽡 𖼊𖽔𖾐 𖼋𖽡𖽝！
𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼢𖽳 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼟𖽪𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼑𖽡𖾏 𖽐𖼮𖽑𖽠𖾏.
 
27 𖼊𖽪𖾐 𖼋𖽻 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼢𖽻 𖼙𖽠𖾏！
𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏！
