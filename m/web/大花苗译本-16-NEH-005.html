𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽤𖾐 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏, 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. 2 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼊𖽻𖾏 𖼡𖽻𖾐 𖼡𖽜𖾏 𖼊𖽪𖾑 𖼳𖽦𖾑.” 3 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖼸𖽺𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏、 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 ,𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼡𖽻𖾐 𖼡𖽜𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏.” 4 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖼷𖽹𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼗𖽙𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖽐𖼎𖽙 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼑𖽡𖾏 𖼳𖽳𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 5 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑, 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼀𖽡𖾐 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼦𖽡𖾏 𖼺𖽱𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼟𖽙𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽜𖾑, 𖼀𖽡𖾐 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖼋𖽘𖾏, 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏、 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼗𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏.”
𖼐𖽡 𖼳𖽡𖾑 𖼄𖽡𖾐 𖼐𖽙𖾑𖽑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼮𖽱𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐
6 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝. 7 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼖𖽔𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏.” 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 8 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼅𖽔𖾏 𖼎𖽙𖾑 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏, 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼺𖽘𖾏 𖽃𖽡𖾑, 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼋𖽺 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 , 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼅𖽔𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐. 9 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:“ 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏, 𖼅𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼀𖽡𖾐. 10 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 , 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑,𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼡𖽻𖾐 𖼡𖽜𖾏 𖼷𖽹𖾏 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼯𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼺𖽘𖾏 . 11 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼞𖽑𖽻𖾑 𖼅𖽡 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏、 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏、 𖼉𖽘𖾏 𖼞𖽗𖾐 𖼖𖽗、 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏、 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾑、 𖼡𖽻𖾐 𖼖𖽻𖾏、 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐、 𖽐𖼎𖽙 𖼛𖽙𖾏, 𖽃𖽡𖾐 𖼀𖽔𖾏 𖽐𖼀𖽠 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽙𖾑 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.” 12 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽙𖾑 𖼁𖽻𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖼟𖽡 𖼗𖽪𖾏 𖽃𖽔𖾏.” 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼋𖽔𖾏, 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽵 𖼗𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏. 13 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼡𖽘𖾐 𖼮𖽑𖽙𖾏 𖼐𖽑𖽡𖾏 : “𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 , 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 ，𖼇𖽳𖾑 𖼺𖽘𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼸𖽙 𖼟𖽪𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏.” 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼐𖽑𖽡𖾏 : “𖽃𖽔𖾏 𖼄𖽞!” 𖼎𖽺𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏.
