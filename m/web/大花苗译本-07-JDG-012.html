𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱, 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼨𖽑𖽪𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼅𖽻𖾏 𖼖𖽺𖾑? 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼞𖽔𖾏.” 2 𖼽𖽦𖾑 𖼇𖽪 𖼊𖽑𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖼅𖽡 𖼋𖽔𖾏, 𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽔𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏. 3 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼋𖽔𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽦𖾑 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼅𖽡 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 𖼖𖽺𖾑?” 4 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱, 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝, 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽔𖾐.” 5 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏, 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖾏 𖼅𖽻𖾏, 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼋𖽡𖾏 𖼅𖽻𖾏.” 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼾𖽙?” 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼾𖽙.” 6 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼐𖽑𖽡𖾏 𖼳𖽷𖾏 𖼀𖽙𖾑 𖼖𖽡𖽝.” 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽙𖾏 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼳𖽘𖾐 𖼗𖽪, 𖼐𖽑𖽡𖾏 𖽃𖽔𖾏 𖼳𖽡𖾑 𖼀𖽙𖾑 𖼖𖽡𖽝 , 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏 𖼊𖽔𖾏 𖼎𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏. 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽔𖾏 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐. 7 𖼽𖽦𖾑 𖼇𖽪 𖼊𖽑𖽔𖾑 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼋𖽻𖾏 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏． 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼋𖽔 𖼺𖽘𖾏, 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻.
𖼽𖽡𖾑 𖼀𖽡𖾐 𖼷𖽔𖾏、𖼽𖽡𖾑 𖼖𖽪𖽝、𖼽𖽔𖾏 𖼊𖽪𖽝𖾏
8 𖼽𖽦𖾑 𖼇𖽪 𖼊𖽑𖽔𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼅𖽔𖾏 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼀𖽡𖾐 𖼷𖽔𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼋𖽻 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏. 9 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐, 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖽐𖼷𖽑𖽹𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼷𖽑𖽹𖾏 𖼞𖽑𖽔𖾏 𖼡𖽔𖾏 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖼑𖽜 𖼐𖽱𖾏 𖽐𖼀𖽙𖾏 𖼊𖽪𖾐 𖼎𖽑𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽙𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼐𖽤𖾐. 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼋𖽻 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼊𖽻𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏. 10 𖼽𖽡𖾑 𖼀𖽡𖾐 𖼷𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏, 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟.
11 𖼽𖽡𖾑 𖼀𖽡𖾐 𖼷𖽔𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽪𖽝 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼋𖽻 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼳𖽡𖽻𖾏. 12 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽪𖽝 𖼋𖽔 𖼺𖽘𖾏, 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼊𖽡𖾐 𖼽𖽔𖾏 𖼽𖽔𖾐 𖼖𖽪𖽝.
13 𖼽𖽡𖾑 𖼖𖽪𖽝 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼅𖽔𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽪𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼳𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼊𖽪𖽝𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼋𖽻 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏. 14 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐, 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑, 𖼯𖽡𖽜𖾏 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼋𖽪𖾏 𖼡𖽔𖾑 𖼯𖽱𖾏 𖼀𖽡𖾏 𖼊𖽪𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑. 𖼽𖽔𖾏 𖼊𖽪𖽝𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼋𖽻 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼊𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏. 15 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽪𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼊𖽪𖽝𖾏 𖼋𖽔 𖼺𖽘𖾏, 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽪𖽝𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐.
