𖼘𖽑𖽪𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏.

𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
“𖼀𖽡𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏.” 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼙𖽔 .
2 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！
𖼀𖽡𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼉𖽺𖾏 𖽐𖼚𖽙 .
 
3 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏,
𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼦𖽡 𖼋𖽳 𖼊𖽔𖾐 𖼋𖽡𖽝 .
4 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽔𖾏 𖼳𖽡𖾏 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏, 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
𖼸𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“ 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏”) 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏.
5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏,
𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼾𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑.
 
6 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏！
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏！
7 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖽐𖼚𖽙 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼚𖽙 𖼟𖽻 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏！
8 𖼑𖽜 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼎𖽙 𖼽𖽺 𖼗𖽔 𖼊𖽔𖾐, 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
“𖼊𖽻𖾏 𖼅𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐！”
9 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼊𖽔𖾐,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏！
