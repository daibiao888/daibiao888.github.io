𖼮𖽑𖽡𖽻𖾏 𖼗𖽠𖾏 𖽃𖽡𖾐 𖼷𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻, 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼤𖽜𖾏 𖼳𖽻𖾐 , 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐, 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽑𖽤𖾏 𖽐𖼞𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼐𖽤𖾐.” 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏, 2 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽔𖾑 𖼳𖽪𖾐 𖼐𖽱𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼗𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏, 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼷𖽙𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐.” 3 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽡𖾐 𖼟𖽻𖾐 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼟𖽡𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼺𖽙𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏.” 4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼅𖽻𖾏 𖼯𖽡𖽻𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼋𖽪𖾏 𖼞𖽡𖾑 𖼋𖽱𖾏 𖼄𖽔𖾐 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖼡𖽔𖾑 𖽐𖼷𖽘𖾏 𖼡𖽑𖽹𖾐 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾐 𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼋𖽪 𖼡𖽔𖾑 𖽐𖼷𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽡𖽝𖾏. 5 𖼮𖽡𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾐 𖼟𖽙𖾏 𖼯𖽡, 𖼮𖽙𖾏 𖼞𖽡𖾑 𖼋𖽱𖾏 𖼏𖽘 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐, 𖼄𖽔𖾐 𖼳𖽻𖾐 𖼟𖽪𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼤𖽜𖾏 𖼺𖽷𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼤𖽜𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼚𖽻𖾏 𖼮𖽡𖽝𖾏 𖼊𖽑𖽡𖽝𖾐 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏、𖼞𖽗𖾐 𖼖𖽗 𖼉𖽘𖾏 𖼷𖽷𖾑 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏. 6 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔𖾏 𖼋𖽱𖾏 𖽃𖽔𖾏 𖼘𖽪?” 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽪𖾏 𖼊𖽑𖽡 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼈𖽻𖾑 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽃𖽔𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼗𖽔 𖼙𖽠𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏. 7 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼎𖽙 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼅𖽡 𖼺𖽘𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼳𖽙𖾏.” 8 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼊𖽑𖽡𖽝𖾐 𖼚𖽔𖾑 𖼀𖽑𖽔𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏． 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼊𖽑𖽗𖾐 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼖𖽪𖾑 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼷𖽔𖾏.
