𖼗𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙𖾏 𖽐𖼚𖽪𖾏
21
𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽱𖾏 𖽃𖽻𖾐,
𖼯𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖽐𖼚𖽱𖾏 𖼅𖽻𖾏.
2 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏,
𖼮𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼮𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
3 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼷𖽹𖾏, 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 .
4 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼙𖽙𖾏 𖽐𖼊𖽪𖾏、𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏、𖼊𖽪𖾐 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼾𖽔𖾏.
5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽻𖾐 𖽐𖼡𖽔𖾏 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼷𖽹𖾑 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏．
6 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐,
𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼋𖽔 , 𖼟𖽪𖾏 𖼦𖽘𖾑 𖼀𖽙𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾐 𖼅𖽻 𖼷𖽑𖽔𖾐 𖼗𖽙 .
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼾𖽨𖾏 𖼮𖽑𖽡𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼸𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 .
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼖𖽡𖾏 𖽐𖼞𖽑𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
9 𖼁𖽻𖾏 𖼏𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼏𖽪𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼞𖽪𖾐,
𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼐𖽡𖽙𖾐.
10 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖽐𖼚𖽪𖾏 𖼸𖽳 ,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼼𖽷𖾏 𖼼𖽙 .
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼗𖽔 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 .
12 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏,
𖼐𖽱𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽐𖼀𖽑𖽻𖾏 𖽐𖼀𖽙 , 𖼸𖽙 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔 .
13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽠𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼑𖽡𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 ,
𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼑𖽻 𖼊𖽡𖾐.
14 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼺𖽘𖾏, 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼮𖽑𖽙𖾏, 𖼸𖽠𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
15 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑， 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼈𖽡𖽝𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 .
16 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼦𖽡 𖼸𖽱.
17 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏；
𖼦𖽠𖾑 𖼉𖽳 𖼮𖽡𖽠𖾑 𖽐𖼎𖽙 𖼛𖽙𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 .
18 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼎𖽙𖾑 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼘𖽑𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
19 𖼁𖽻𖾏 𖼏𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 ,
𖼦𖽡𖾐 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼐𖽡𖽙𖾐.
20 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏, 𖼮𖽷𖾏 𖼮𖽡𖽝𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼎𖽙 𖼛𖽙𖾏,
𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼡𖽔𖾏 𖼊𖽘𖾑 𖼋𖽱𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼡𖽔𖾏 𖼊𖽘𖾑 𖼋𖽱𖾏 𖼎𖽑𖽔𖾑 𖼋𖽘𖾏 𖼺𖽘𖾏.
21 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽐𖼎𖽙 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏,
𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑、𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽐𖼎𖽙 𖽐𖼚𖽻 𖼮𖽡𖾏 .
22 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼊𖽪𖾐 𖼋𖽻 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 ,
𖼊𖽑𖽠𖾐 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑.
23 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼎𖽙 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏,
𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼾𖽨𖾏 𖼦𖽡𖾐 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 .
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽙𖾏 𖼎𖽺𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 “𖼙𖽙𖾏 𖽐𖼊𖽪𖾏”,
𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼙𖽙𖾏 𖽐𖼊𖽪𖾏、𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 .
25 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽜 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼺𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 .
26 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼦𖽠𖾑 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽻𖾏,
𖼉𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼗𖽔 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏.
27 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐,
𖽂𖽪𖾏 𖽂𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼋𖽔𖾏 𖼖𖽑𖽘𖾑 𖼯𖽡𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏.
28 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 ,
𖼮𖽔𖾑 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼺𖽘𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽑𖽡𖾏, 𖼋𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 .
29 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
30 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳、𖼀𖽻𖾐 𖼊𖽘𖾐、
𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
31 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏.
𖼈𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
