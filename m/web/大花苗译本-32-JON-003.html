𖼽𖽙 𖼐𖽔 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼚𖽔𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏
3
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼽𖽙 𖼐𖽔 𖼐𖽑𖽡𖾏: 2 “ 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 ! 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼙𖽻 𖽃𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏.” 3 𖼽𖽙 𖼐𖽔 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 . 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝（𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝”） , 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼞𖽡𖾑. 4 𖼽𖽙 𖼐𖽔 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻, 𖼅𖽻𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼞𖽡𖾑 , 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼎𖽺𖾏 𖼋𖽙𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐, 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼙𖽻 𖼽𖽔𖾑 𖼼𖽠𖾏 𖼢𖽻 𖼋𖽘𖾏 !” 5 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽜 𖼸𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑, 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼸𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐, 𖼽𖽔𖾑 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖾏 𖽐𖼀𖽔𖾏 𖼮𖽡𖽝𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐.
6 𖼋𖽡𖾏 𖼄𖽻𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼸𖽙𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖼛𖽠𖾏 𖼺𖽘𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽻𖾏, 𖼎𖽺𖾏 𖽐𖼀𖽔𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐, 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼮𖽑𖽻𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐. 7 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽐𖼷𖽻𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼙𖽻 𖼐𖽑𖽡𖾏: “𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 ， 𖼑𖽨𖾏 𖼮𖽑𖽷𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻𖾏 𖽐𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼦𖽻𖾏 𖽃𖽻𖾐. 8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼽𖽔𖾑 𖽐𖼀𖽔𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏, 𖽐𖼎𖽻𖾏 𖼺𖽘𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳. 9 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽠𖾏 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼛𖽙𖾏 𖼋𖽔𖾏.”
10 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏, 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
