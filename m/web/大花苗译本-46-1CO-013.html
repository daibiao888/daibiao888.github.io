𖽐𖼮𖽑𖽠𖾏
13
𖼞𖽪𖾑 𖼮𖽔𖾑 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼗𖽪𖾏，𖼈𖽡𖽝𖾏 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼮𖽑𖽠𖾏，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽐𖼎𖽔𖾏 𖼡𖽠𖾏 、𖽐𖼎𖽔𖾏 𖽐𖼀𖽔𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐． 2 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏，𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐， 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼀𖽻𖾐 𖼟𖽳，𖼎𖽺𖾏 𖼅𖽔𖾏 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏，𖼟𖽙𖾏 𖼞𖽪𖾑 𖼸𖽠𖾏 𖼮𖽑𖽹𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐， 𖼈𖽡𖽝𖾏 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼮𖽑𖽠𖾏，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏． 3 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏 𖽐𖼀𖽘𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏，𖼎𖽺𖾏 𖼾𖽨𖾏 𖼇𖽳𖾑 𖼾𖽨𖾏 𖼄𖽔𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏， 𖼈𖽡𖽝𖾏 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼮𖽑𖽠𖾏，𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏．
4 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖼸𖽙 𖼡𖽘𖾐，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼑𖽔𖾏 𖼙𖽳𖾏，𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐，𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏， 𖼦𖽡𖾐 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐 𖼾𖽨𖾏 ． 5 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏，𖼦𖽡𖾐 𖽐𖼎𖽑𖽦𖾑 𖼾𖽨𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，𖼦𖽡𖾐 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐， 𖼦𖽡𖾐 𖼮𖽔𖾑 𖼺𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏． 6 𖼦𖽡𖾐 𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖼮𖽔𖾑 𖼙𖽔 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐． 7 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽐𖼷𖽜𖾏 𖼯𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼊𖽪𖾏 𖽐𖼮𖽜， 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏，𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏．
8 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽘𖾏． 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏，𖽐𖼷𖽙𖾑 𖼙𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏，𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼗𖽪𖾏，𖽐𖼷𖽙𖾑 𖼙𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏， 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼷𖽙𖾑 𖼙𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏． 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼀𖽻𖾐 𖼅𖽔𖾏 𖽐𖼚𖽜𖾏，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼅𖽔𖾏 𖽐𖼚𖽜𖾏． 10 𖼋𖽙𖾏 𖼸𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼷𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼚𖽜𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏． 11 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏， 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏，𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏，𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏．𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽡𖽝𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽘，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼗𖽔𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼑𖽪𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏． 12 𖼀𖽡𖾐 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼑𖽔𖾏 𖼚𖽑𖽪𖾏，𖼁𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑，𖼸𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽱𖾐 𖼚𖽑𖽪𖾏 𖼁𖽙𖾏 𖼚𖽑𖽪𖾏．𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼅𖽔𖾏 𖽐𖼚𖽜𖾏，𖼸𖽙 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼀𖽻𖾐 𖼋𖽘 𖽐𖼮𖽢𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼀𖽻𖾐 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐． 13 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽐𖼮𖽜、𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏、𖼅𖽔𖾏 𖽐𖼮𖽑𖽠𖾏， 𖼷𖽷𖾐 𖼼𖽘𖾐 𖼐𖽡𖾑， 𖼼𖽘𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏．
