𖼖𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽑𖽘𖾑 𖼁𖽠𖾏
10
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽘𖾑 𖽐𖼷𖽱𖾏,
𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
𖽃𖽔𖾑 𖼖𖽪𖾐 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼎𖽺𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐；
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼛𖽙 𖼻𖽻𖾏,
𖼎𖽺𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏.
2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼊𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 ,
𖽐𖼊𖽪𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼁𖽠 .
 
3 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏:
“𖼀𖽡𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽠𖾏 𖼑𖽜 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?”
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼺𖽱𖾏 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏
𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼟𖽱𖾏 𖼖𖽡𖾏 𖼉𖽻𖾏.
 
5 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼀𖽳 · 𖼽𖽔𖾏 𖽂𖽺 𖼑𖽡𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏,
𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼖𖽔𖾏 𖼑𖽜 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼁𖽠 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼐𖽱𖾏 .
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼯𖽡𖽙𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼳𖽪 ,
𖼯𖽡𖽙𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏, 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖽝 𖼀𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 (𖼽𖽦𖾑 𖼖𖽪𖽝 𖼀𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏),
𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 , 𖼐𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼺𖽱𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽻𖾐 𖼑𖽡𖾏 𖼀𖽑𖽔𖾏 𖽐𖼀𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
8 𖼀𖽳 · 𖼽𖽔𖾏 𖽂𖽺 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐,
𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐,
𖼞𖽡𖾑 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼦𖽡𖾏 𖼇𖽔𖾑.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽑𖽡𖾏: “𖽐𖼀𖽙 𖼈𖽳𖾑 𖼀𖽡𖾐 ！”
𖽐𖼎𖽙 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼯𖽱𖾏 𖼎𖽻𖾐 𖼐𖽑𖽡𖾏: “𖼢𖽻 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑！”
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏
9 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 !
𖼟𖽡 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏 ,
𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼙𖽔 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑.
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 ,
𖼋𖽡𖽝𖾏 𖼾𖽙 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼦𖽡𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼖𖽠𖾑 𖼯𖽢𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?
10 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼾𖽨𖾏 𖽃𖽔𖾐 𖼼𖽘𖾐 𖼾𖽔𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐,
𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
11 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔 𖼑𖽡𖽝 𖼑𖽨𖾏 𖼟𖽪𖾏 𖼑𖽻 𖼗𖽪𖾏, 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏 𖼯𖽪𖾏 𖼡𖽻𖾐,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼚𖽑𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼛𖽙 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽐𖼡𖽪𖾏 𖼮𖽹𖾑 (“𖽐𖼡𖽪𖾏 𖼮𖽹𖾑”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼊𖽻𖾏 𖼯𖽡𖽜𖾏”),
𖼽𖽨 𖼊𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐,
𖼽𖽔𖾐 𖼞𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽹𖾑 𖼊𖽡𖾐.
12 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼯𖽡𖽙 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼎𖽺𖾏 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏, 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽑𖽳𖾏 𖼊𖽡𖾐 𖼡𖽔𖾏 𖼁𖽙𖾏 .
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼋𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏, 𖼟𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼺𖽱𖾏 𖼑𖽘 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 .
 
13 𖼅𖽡 𖼯𖽡𖽙𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼳𖽻𖾏 𖼤𖽜𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
