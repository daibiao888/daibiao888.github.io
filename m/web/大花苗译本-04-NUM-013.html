𖼄𖽙𖾑 𖼳𖽡𖾑 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏
（𖼎𖽺𖾏 1：19 - 33）
13
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼡𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑， 𖽃𖽡𖾐 𖼳𖽡𖾏 𖼽𖽔𖾑 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 .” 3 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼎𖽑𖽻𖾏 𖼀𖽔𖾑 𖼖𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏 :
𖼖𖽨 𖼀𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼞𖽑𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽔𖾑 𖼄𖽪𖾐 𖼽𖽔𖾏 .
5 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼦𖽙 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽔𖾑 𖼇𖽔 .
6 𖼽𖽨 𖼊𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼇𖽪 𖼐𖽡 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼖𖽳 .
7 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼮𖽢𖾑 .
8 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼐𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐 .
9 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼇𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽑𖽔 𖼊𖽑𖽡 .
10 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼺𖽙𖾐 𖼊𖽡𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼮𖽢𖾑 𖼊𖽡𖽝 .
11 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼺𖽪𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼮𖽢𖾑 𖼊𖽡𖾐 .
12 𖼊𖽗𖾏 𖼳𖽡𖾏 𖽃𖽡𖾐 , 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼖𖽡𖾏 .
13 𖼽𖽔𖾏 𖼳𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼖𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼊𖽑𖽡𖽝 .
14 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼇𖽪 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼐𖽔 𖼀𖽡𖾐 .
15 𖼮𖽢𖾑 𖼊𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼮𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼮𖽨𖾏 𖼖𖽡𖾏 .
16 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 . 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼨𖽑𖽪𖾏 𖼐𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐 𖽃𖽔𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 .
17 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐， 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏， 18 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼮𖽤𖾐 ， 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼋𖽡𖽝𖾏 𖼋𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 , 𖼋𖽡𖽝𖾏 𖼮𖽡𖽻𖾐 𖼖𖽔𖾏 𖼯𖽠 ， 19 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽡𖽝𖾏 𖼻𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 ， 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼋𖽡𖽝𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼊𖽜𖾑 𖼖𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼏𖽔𖾏 ? 20 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼋𖽡𖽝𖾏 𖼛𖽙 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼛𖽙 ， 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 ． 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼚𖽔𖾐 , 𖼞𖽑𖽡𖽠𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼯𖽡𖽙𖾏 𖼗𖽙 .” 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼁𖽱𖾏 𖼳𖽦𖾑 .
