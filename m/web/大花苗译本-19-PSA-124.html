𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽑𖽡𖾏：
𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐,
2 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼊𖽱𖾑 𖼀𖽡𖾐,
3 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖼯𖽡𖽝𖾏 𖼙𖽠𖾏 𖼙𖽠𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
4 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖼮𖽑𖽡𖽻𖾏 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼳𖽔𖾐,
𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼞𖽑𖽔𖾏 𖼈𖽳𖾑 𖼀𖽡𖾐 𖼺𖽘𖾏，
5 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙
𖼞𖽑𖽔𖾏 𖼈𖽳𖾑 𖼀𖽡𖾐 𖼺𖽘𖾏.
 
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏！
𖼐𖽱𖾏 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖽐𖼡𖽹𖾏 𖼻𖽻𖾑 𖼸𖽔 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏 (“𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼊𖽪𖾑 𖼐𖽑𖽡𖽝𖾑 ”).
7 𖼀𖽡𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽢𖾐 𖼐𖽻 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼛𖽡𖾏 𖼋𖽠𖾏,
𖼖𖽪𖾑 𖼸𖽙𖾏 𖽐𖼚𖽔 𖼙𖽠𖾏, 𖼀𖽡𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏.
8 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏,
𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏.
