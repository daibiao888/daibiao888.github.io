3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼀𖽳 𖼊𖽔𖾏 𖼐𖽡 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼄𖽞 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏 𖽐𖼞𖽔𖾏， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖽃𖽡𖾑， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖼽𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼛𖽙𖾏 𖼐𖽔 𖼊𖽔 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 ， 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼋𖽔𖾏， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽻𖾏 𖽐𖼮𖽻 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑， 𖼘𖽑𖽡𖾑 𖼛𖽙𖾏 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏． 4 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠𖾏 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏：“ 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼽𖽔𖾑 𖽐𖼞𖽡𖾏 𖼋𖽡𖾏 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼐𖽡𖾑？ 5 𖼋𖽡𖾏 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼋𖽺 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼛𖽡𖾏 𖼑𖽡𖽝𖾏 𖽐𖼀𖽘𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏．” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼎𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏． 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼯𖽔𖾏 𖼐𖽱𖾏， 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼐𖽱𖾏， 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 ， 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝． 7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏， 𖼮𖽔𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼯𖽪𖾏， 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼦𖽡𖾐 𖼳𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐． 8 𖼖𖽔𖾑 𖼁𖽙𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼐𖽱𖾏 𖼚𖽜𖾑 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽱𖾏． 9 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾏 𖼋𖽻𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼡𖽑𖽔𖾐 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼁𖽙𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏， 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏．”
𖼽𖽨 𖼊𖽔𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 26：14-16；𖼖𖽪𖾏 22：3-6）
10 𖼮𖽢𖾑 𖼖𖽡𖽙𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐， 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 ， 𖼽𖽔𖾑 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏 𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼎𖽺𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼻𖽻𖾏 𖼯𖽡𖽹𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑．
𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏
（𖼊𖽑𖽹𖾏 26：17-25；𖼖𖽪𖾏 22：7-13，21-23；𖼽𖽙 13：21-30）
12 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 𖽃𖽡𖾑 𖼐𖽑𖽪𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑， 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏， 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼽𖽔𖾑 𖼀𖽡𖾐 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏，𖼑𖽜 𖼟𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑 𖼄𖽑𖽙𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽻𖾏 𖼖𖽺𖾑？ ”
13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼷𖽷𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼅𖽻𖾏， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼚𖽔𖾏 𖼙𖽻 𖼽𖽔𖾑 𖽐𖼮𖽷𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽳𖾑 𖽃𖽡𖾐 𖼣𖽑𖽪𖾐 𖽃𖽻𖾐， 𖼅𖽡 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐． 14 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏， 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼾𖽡𖾏 𖽃𖽡𖾑 𖽐𖼞𖽔𖾏 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏：‘ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼐𖽑𖽡𖾏 ：𖽐𖼞𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 ， 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼑𖽻𖾏 𖼡𖽜𖾑 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏？’ 15 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼊𖽑𖽘𖾐， 𖼮𖽑𖽤𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼊𖽜𖾐 𖼻𖽻𖾏 𖼋𖽘𖾏， 𖼅𖽡 𖼎𖽺𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 ．” 16 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻， 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽷𖾐 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼄𖽑𖽙𖾏 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏．
17 𖼸𖽙 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 ， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏． 18 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼽𖽔𖾑 𖼋𖽺 𖼞𖽪𖾑．” 19 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽳𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼋𖽡𖽝𖾏 𖼾𖽙 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 ？” 20 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾏 𖼖𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐，𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼳𖽡𖽝𖾐 𖼚𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑 𖽃𖽡𖾑． 21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽺 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏， 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽨 𖼐𖽱𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼖𖽔𖾑 𖼻𖽻𖾏 𖼺𖽷𖾐．”
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼮𖽵𖾑 𖼉𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑
（𖼊𖽑𖽹𖾏 26：31-35；𖼖𖽪𖾏 22：31-34；𖼽𖽙 13：36-38）
22 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼳𖽡𖽝𖾐 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼺𖽘𖾏，𖽐𖼀𖽹𖾏 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ，𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼷𖽹𖾏 𖼑𖽻𖾏， 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑．” 23 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽑𖽙𖾑 𖼈𖽳𖾐 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼮𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏． 24 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽘𖾑， 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼮𖽑𖽦𖾐， 𖼟𖽪𖾏 𖼘𖽑𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏． 25 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ， 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼦𖽻𖾏 𖼋𖽡𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼐𖽡𖾑， 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼦𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽦𖾐．” 26 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ，𖼚𖽔𖾏 𖼞𖽗𖾐 𖼖𖽗 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼙𖽠𖾏．
𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼀𖽡𖾐 𖼊𖽳 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 26：31-35；𖼖𖽪𖾏 22：31-34；𖼽𖽙 13：36-38）
27 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼢𖽻 ， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼐𖽑𖽡𖾏：
‘ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏，
𖼾𖽘𖾏 𖼎𖽺𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏．’
28 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖽃𖽡𖾑， 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖽐𖼊𖽡𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏．” 29 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼢𖽻， 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼢𖽻．” 30 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡， 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾑， 𖼡𖽹𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼡𖽔𖾏 𖽃𖽔𖾐 𖼼𖽘𖾏， 𖼟𖽡 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼟𖽻𖾐 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼞𖽪𖾑．” 31 𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖼊𖽳 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼐𖽑𖽡𖾏：“ 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼋𖽔， 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼟𖽡．” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
（𖼊𖽑𖽹𖾏 26：36-46；𖼖𖽪𖾏 22：39-46）
32 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼞𖽑𖽳 𖼳𖽡𖾑 𖼄𖽔𖾐 𖼐𖽡， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑， 𖼋𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏．” 33 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼊𖽳、 𖼽𖽔𖾐 𖼞𖽙、 𖼽𖽙 𖼦𖽗 𖽐𖼎𖽙 𖼅𖽻𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼊𖽳𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏， 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝， 34 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏 𖼮𖽑𖽡𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔 𖼯𖽡𖽙𖾏， 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼯𖽷𖾏 𖼻𖽙𖾑．” 35 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽹𖾏 𖼀𖽡 𖼯𖽠 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 ， 𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼯𖽪𖾏， 𖼟𖽙𖾏 𖼯𖽡𖽹𖾏 𖼐𖽡𖾑 𖼮𖽑𖽡𖽻𖾏 𖼞𖽪𖾑 𖼙𖽠𖾏．” 36 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔𖾑 𖼀𖽔𖾑 ， 𖼉𖽺𖾏 𖼘𖽪！ 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏， 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽑𖽙𖾐 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏，𖼯𖽔 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏．” 37 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽷𖾐 𖼗𖽙， 𖼁𖽙𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽻𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“ 𖼳𖽡𖾑 𖼄𖽞 ，𖼟𖽡 𖼊𖽻𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 𖼘𖽪𖾏？ 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼯𖽷𖾏 𖼻𖽙𖾑 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏？ 38 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽷𖾏 𖼻𖽙𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏， 𖽐𖼮𖽑𖽹𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑．𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖽻𖾏 𖼯𖽡𖽙𖾏， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼯𖽻 𖼋𖽘𖾏．” 39 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐． 40 𖼺𖽷𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽻𖾏 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏． 41 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖼺𖽷𖾐 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼀𖽱𖾏 𖼳𖽙𖾏 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐 𖼘𖽪𖾏？ 𖼷𖽻𖾏 𖼋𖽘𖾏， 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏！ 𖼅𖽡 𖼑𖽔𖾏， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼊𖽻𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏． 42 𖼳𖽠𖾑 𖼗𖽙𖾏，𖼀𖽡𖾐 𖼅𖽻𖾏 𖼽𖽔𖾐！ 𖼑𖽔𖾏 𖼘𖽪！ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽺 𖼞𖽪𖾑 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．”
