𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼟𖽻𖾏 𖼼𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
（𖼋𖽠𖾏 20：1 - 17）
5
𖼄𖽙𖾑 𖼳𖽡𖾑 𖼨𖽑𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪 ! 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑、𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾏 𖼇𖽺𖾏 𖼅𖽡 , 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽻 , 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 . 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼦𖽙 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 . 3 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖽃𖽔𖾏 , 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖼺𖽷𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 . 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼚𖽑𖽪𖾏 𖼊𖽱𖾐 𖼚𖽑𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 . 5 ( 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽑𖽡𖽝𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼅𖽡 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼮𖽑𖽹𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 , 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 ).
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 .
7 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
8 𖼟𖽡 𖼦𖽡𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 , 𖼺𖽱𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏、𖼀𖽡𖾑 𖼊𖽡𖾐、𖼼𖽘𖾏 𖼚𖽘𖾐 𖽐𖼚𖽡𖽝 𖼊𖽡𖾐、𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 . 9 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼋𖽡𖾏 𖼼𖽘𖾏 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 . 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼾𖽱 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 , 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼸𖽙 𖼊𖽪𖾐 , 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾐 𖼚𖽻𖾐 𖼷𖽑𖽷𖾏 . 10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 , 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽑𖽡𖽝𖾐 𖼷𖽑𖽷𖾏 .
11 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼨𖽑𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼨𖽑𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 .
12 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏. 13 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖽃𖽔𖾏 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏， 14 𖼈𖽡𖽝𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 . 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏、𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏、𖼑𖽨𖾏、𖼑𖽜𖾏 𖼖𖽔𖾑、𖼯𖽦𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼡𖽑𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 , 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼳𖽙𖾏 . 15 𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼎𖽙 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼟𖽪𖾏 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 .
16 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 , 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 , 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 .
17 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 .
18 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 .
19 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼺𖽘𖾐 𖽃𖽔𖾏 𖼸𖽜 .
20 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙 .

（𖼋𖽠𖾏 20：18 - 21）
22 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽡𖾑 , 𖼋𖽠𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼗𖽪𖾏 . 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼺𖽻𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼮𖽡𖽝𖾏 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏 .
23 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽪𖾑 𖼎𖽻𖾐 , 𖼅𖽡 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏 , 𖼅𖽡 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼗𖽻 𖼋𖽔𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑， 24 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 : 𖼑𖽔𖾏 𖼘𖽪 ! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼢𖽳 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼑𖽔𖾏 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏． 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖼺𖽷𖾐 . 25 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽔𖾑 𖼷𖽑𖽷𖾑 𖼀𖽡𖾐 𖼺𖽘𖾏 , 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖽐𖼚𖽪𖾐 𖼋𖽔 𖼘𖽪 ? 𖼮𖽔𖾑 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 , 𖼯𖽡𖾏 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 . 26 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖽃𖽡𖾑 , 𖼙𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼐𖽑𖽙𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼺𖽱𖾏 𖼀𖽡𖾐 𖼐𖽙𖾑𖽑 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑 ? 27 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 , 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 , 𖼄𖽔𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼸𖽙𖾏 𖽃𖽔𖾏 .
28 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽑𖽔𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏． 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏： 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼗𖽪𖾏 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽙 . 29 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 , 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 . 30 𖼟𖽡 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼦𖽡𖾑 ! 31 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 , 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏、𖼼𖽘𖾏 𖼻𖽡𖽝𖾑、𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 , 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 , 𖼟𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽡𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 . 32 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖽃𖽡𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 . 33 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡, 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 , 𖼟𖽙𖾏 𖼅𖽡 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼳𖽪𖾏 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 .
