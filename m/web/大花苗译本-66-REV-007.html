5 𖼽𖽨 𖼊𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼘𖽙𖾏 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
𖼖𖽨 𖼀𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
𖼮𖽢𖾑 𖼊𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
6 𖼽𖽔𖾏 𖼳𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
7 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
𖼖𖽡𖾏 𖽂𖽺𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
8 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
𖼽𖽙 𖼺𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐．
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼋𖽔𖾏 𖼷𖽻𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
9 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼞𖽪𖾑 𖼑𖽔𖾏， 𖼁𖽙𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼾𖽱 𖼋𖽘 𖼋𖽻𖾏，𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐、𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 、𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏、𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼗𖽪𖾏 𖼋𖽔𖾏，𖼳𖽠𖾑 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼎𖽙 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐， 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼺𖽱𖾐 𖼺𖽱𖾐， 10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：
“ 𖼯𖽔 𖼅𖽔𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑、
𖼾𖽘𖾏 𖼁𖽜𖾏 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ！”
11 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼗𖽻 𖼊𖽑𖽡𖽝𖾐 𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼟𖽻𖾏 𖼯𖽪𖾏，𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 12 𖼐𖽑𖽡𖾏 ：
“𖽃𖽔𖾏 𖼄𖽞！𖼊𖽻𖾏 𖼅𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔、 𖽐𖼚𖽻 𖼮𖽡𖾏、 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳、
𖽐𖼮𖽢𖾏 𖼙𖽔、 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏、 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 、
𖼛𖽘𖾏 𖼼𖽙𖾏，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏．𖽃𖽔𖾏 𖼄𖽞！”
13 𖼊𖽪𖾐 𖼗𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼑𖽪 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏？” 14 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪 ！𖼟𖽡 𖼀𖽻𖾐 𖼯𖽡𖽙．” 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏． 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑，𖽐𖼷𖽑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 𖽐𖼷𖽑𖽷𖾏 𖼗𖽡𖽝𖾏 𖼋𖽘𖾏．
