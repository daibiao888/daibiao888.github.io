𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽔𖾏 𖼞𖽳𖾑， 𖼑𖽜 𖼾𖽔𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖽐𖼎𖽙 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼯𖽔𖾏 𖼾𖽔𖾏． 2 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾏 𖼯𖽡𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽑𖽵𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽔𖾏 𖼏𖽻 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼡𖽑𖽹𖾐 𖽐𖼊𖽠𖾏 ， 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼐𖽱𖾏 𖼑𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼯𖽔𖾏 𖼾𖽔𖾏． 4 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼊𖽻𖾏 𖼋𖽻𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽹𖾑， 𖼺𖽱𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽃𖽡𖾐 𖼼𖽘𖾐．
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏， 𖼮𖽡𖾑 𖼊𖽪 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏， 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏， 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼺𖽹𖾑 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：
“𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐，
𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼅𖽔𖾏 𖼟𖽡．” 𖽃𖽡𖾑 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏．
6 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼐𖽑𖽡𖾏：
“ 𖼟𖽡 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼄𖽳 𖼮𖽡𖾑 𖼳𖽡𖾐 𖼊𖽳 𖼑𖽡𖾏 𖼼𖽘𖾏，
𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏．”
7 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼺𖽷𖾐， 𖼐𖽱𖾏 𖼙𖽔 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑， 𖼎𖽺𖾏 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼎𖽙 𖽐𖼊𖽜 𖼽𖽔𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼇𖽳𖾑 𖼛𖽡𖾏 𖼋𖽔 𖼺𖽘𖾏 𖽃𖽡𖾑， 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 ， 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼑𖽻 𖼊𖽡𖾐． 8 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼈𖽡𖽝𖾏， 𖼑𖽜 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼎𖽺𖾏 𖼾𖽻 𖼑𖽻 𖼸𖽙𖾏． 9 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏， 𖼎𖽺𖾏 𖼑𖽜 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏， 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏． 10 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼄𖽳 𖼮𖽡𖾑 𖼳𖽡𖾐 𖼊𖽳𖼑𖽡𖾏 𖼼𖽘𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏．
𖼽𖽔𖾑 𖼘𖽑𖽙𖾐 𖼊𖽡𖽝𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐
11 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼄𖽳 𖼮𖽡𖾑 𖼳𖽡𖾐 𖼊𖽳 𖼀𖽡𖾐 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐， 𖼈𖽡𖽝𖾏 𖼚𖽠𖾏 𖼐𖽑𖽡𖾏 𖼊𖽔𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽻 𖼦𖽡𖾐 𖼁𖽺𖾏 𖼅𖽻𖾏． 12 𖼑𖽔𖾏 𖼅𖽡 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼗𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼋𖽡𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼋𖽘𖾏， 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽡𖽝𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼺𖽷𖾐， 𖼅𖽡 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖼊𖽔𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽻𖾏 𖼡𖽜𖾑 𖽐𖼡𖽑𖽔𖾐． 13 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖼦𖽡𖾐 𖼳𖽦𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼤𖽔𖾑 𖼤𖽔𖾏 𖼊𖽔𖾐． 14 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽡𖽝𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼑𖽻𖾏 𖼡𖽜𖾑 𖽐𖼡𖽑𖽔𖾐， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 ，𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽙𖾐， 𖼸𖽠𖾏 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼮𖽷𖾑．
