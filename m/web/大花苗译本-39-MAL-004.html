𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼊𖽡𖾏 𖼈𖽳𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼟𖽪𖾏 𖼚𖽻𖾑 𖼯𖽡 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽻𖾏 𖽐𖼡𖽜𖾐, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼷𖽑𖽷𖾑 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖽃𖽔𖾑 𖼯𖽤𖾐 𖽃𖽔𖾑 𖼯𖽡 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼷𖽷𖾐 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐. 2 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼐𖽱𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼐𖽱𖾏 𖼮𖽡𖾏 (“𖼐𖽱𖾏 𖼮𖽡𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖽃𖽔𖾑 𖼊𖽡𖾏”)𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼞𖽑𖽙𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐． 𖼅𖽡 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼚𖽑𖽡𖽝𖾏 𖼚𖽜𖾑𖽑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼛𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼯𖽦𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 . 3 𖼅𖽡 𖼽𖽔𖾑 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼏𖽵 𖼮𖽡𖽝𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼷𖽑𖽹𖾏 𖼡𖽘𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽠𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼞𖽑𖽻𖾑 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼻𖽙𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
4 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼦𖽙 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖽐𖼎𖽙 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏.
5 𖼑𖽔𖾏 𖼘𖽪! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖽃𖽺𖾐 𖽐𖼮𖽑𖽹𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼅𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑． 6 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼉𖽺𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏, 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽔𖾏 𖼉𖽺𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼸𖽙𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.”
