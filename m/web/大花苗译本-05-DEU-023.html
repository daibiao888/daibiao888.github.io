3 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼮𖽑𖽡𖽻𖾏 𖼗𖽠𖾏 𖼟𖽻𖾏 𖼷𖽑𖽷𖾏 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 . 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼯𖽡𖽙𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖽐𖼎𖽙 𖽃𖽻𖾐 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼸𖽱𖾏 𖼷𖽹𖾏 𖼅𖽡 , 𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼄𖽡𖾐 𖼺𖽙𖾐 𖼀𖽙𖾑 𖼊𖽔𖾏 𖼄𖽡𖾐 𖼟𖽪𖾏 𖼀𖽑𖽡𖾐 𖼊𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾐 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼖𖽔 , 𖼋𖽔𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼅𖽡 . 5 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼀𖽔𖾑 𖼖𖽔 , 𖼟𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼅𖽳𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 . 6 𖼟𖽡 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖽐𖼎𖽙 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 .

𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼏𖽙 𖼾𖽡 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏
9 𖼟𖽡 𖼊𖽑𖽱𖾏 𖼏𖽙 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 , 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼟𖽪𖾏 𖼚𖽡𖾐 .
10 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼮𖽑𖽻𖾏 𖼗𖽻𖾏 𖼮𖽘𖾑 , 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖼾𖽡 𖼙𖽠𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼾𖽡 . 11 𖼸𖽙 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 𖼙𖽠𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼾𖽡 .

𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
15 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼺𖽘𖾏, 𖼀𖽜𖾑 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 . 16 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼚𖽙 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼽𖽔𖾑 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼺𖽹𖾑 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼐𖽱𖾏 .
17 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽐𖼞𖽔𖾏 𖼁𖽠 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽔𖾑 𖼾𖽠 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽐𖼞𖽔𖾏 𖼁𖽠 𖽃𖽔𖾑 𖼾𖽠 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 (“𖽃𖽔𖾑 𖼾𖽠 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼚𖽡𖾑” ) . 18 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼊𖽻𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼊𖽻𖾏 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼏𖽵 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼼𖽘𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼳𖽪𖾐 .
19 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽜𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼡𖽻𖾐 𖼡𖽜𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 . 20 𖼡𖽜𖾑 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼳𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 , 𖼮𖽔𖾑 𖼡𖽜𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 , 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 .
𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼏𖽵
21 𖼟𖽡 𖼮𖽔𖾑 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 , 𖼽𖽔𖾑 𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼏𖽵 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽔𖾑 𖼊𖽳𖾑𖽑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼾𖽱 𖼟𖽡 , 𖼟𖽡 𖼦𖽡𖾐 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼋𖽘𖾏 . 22 𖼟𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 . 23 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 , 𖼟𖽪𖾏 𖼟𖽡 𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼏𖽵 𖼐𖽑𖽡𖾏 𖼽𖽔𖾑 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 ， 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼏𖽵 𖼗𖽪𖾏 𖽃𖽡𖾑 , 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 .
24 𖼟𖽡 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 , 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼑𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼟𖽙𖾏 𖼮𖽻𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 . 25 𖼟𖽡 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼞𖽙𖾏 𖼡𖽻𖾐 , 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼊𖽪𖾏 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼖𖽡𖽝𖾐 𖼤𖽜𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖽐𖼮𖽳 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐 𖽃𖽡𖾑.
