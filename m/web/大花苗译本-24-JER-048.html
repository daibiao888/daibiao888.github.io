𖼐𖽑𖽡𖾏 𖼸𖽙 𖼄𖽙𖾑 𖼽𖽔𖾏 , 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:
“𖼐𖽡 𖼀𖽙𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽑𖽡 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏.
𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼷𖽷𖾏 “𖼄𖽡𖾐 𖼺𖽷𖾑 𖼞𖽔𖾐 𖽂𖽪”) 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏.
2 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔．
𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼳𖽷𖾏 𖼀𖽺𖾐 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼦𖽺𖾏 𖼐𖽱𖾏, 𖼐𖽑𖽡𖾏:
“𖼋𖽔𖾏 𖼘𖽪！ 𖼀𖽡𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏, 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼊𖽡𖽝𖾑 𖼞𖽪𖽙.
𖼄𖽔𖾐 𖼊𖽳 𖼄𖽡𖽝 𖼘𖽪 ! 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼮𖽦 𖼊𖽙𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐.
𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼟𖽡.
3 𖼎𖽑𖽻𖾏 𖼦𖽙 𖼘𖽙 𖼐𖽡𖽝𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏,𖼐𖽑𖽡𖾏:
𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏! 𖼊𖽻𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏 !
4 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏！
𖼐𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑.
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖾏 𖼖𖽪𖾐 𖼳𖽡𖾑 𖽃𖽔𖾑 𖼼𖽘 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼐𖽡𖽝𖾑 𖼷𖽷𖾏 𖼖𖽘𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼦𖽙 𖼘𖽙 𖼐𖽡𖽝𖾏 𖼖𖽪𖾑 𖼡𖽘𖾐 𖽃𖽔 𖼼𖽘, 𖼐𖽙𖾑𖽑 𖼳𖽘𖾐 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏.
6 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼏𖽘 𖼀𖽜𖾑, 𖼾𖽨𖾏 𖼮𖽨𖾏 𖼾𖽨𖾏 𖼳𖽦𖾑,
𖼟𖽙𖾏 𖼅𖽡 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼞𖽑𖽻𖾐 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏.
7 𖼟𖽡 𖼑𖽜 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼾𖽨𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏．
𖼮𖽡𖾑 𖼄𖽙（“𖼮𖽡𖾑 𖼄𖽙 ” 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺） 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼟𖽪𖾏 𖼺𖽙𖾏𖼮𖽡𖾑 𖼄𖽙, 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏.
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽡𖾏．
𖼎𖽻𖾐 𖼦𖽘𖾏 𖼗𖽱𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏.
 
9 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼎𖽻𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼽𖽘𖾏 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼽𖽔𖾑 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐. 10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼘𖽙𖾏 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏．
𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼙𖽻 𖼊𖽻𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐
11 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽡𖾑 𖼗𖽙𖾏, 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽔𖾐,
𖼺𖽱𖾏 𖼮𖽡𖽠𖾑 𖽐𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽐𖼚𖽙 𖼦𖽡𖾐 𖼙𖽔 𖼊𖽻𖾏 𖼩𖽪𖾏,
𖼦𖽡𖾐 𖼙𖽔 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼉𖽺𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏．
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼺𖽷𖾐,
𖽐𖼷𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼚𖽑𖽡𖾏.
12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼑𖽔𖾏 𖼘𖽪𖾏! 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏, 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼷𖽷𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽡𖾑𖽑 𖼮𖽡𖽠𖾑 𖽃𖽡𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼘𖽡𖾑𖽑 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼞𖽑𖽔𖾏 𖼘𖽡𖾑𖽑 𖼟𖽙𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼺𖽘𖾏, 𖽐𖼊𖽻𖾏 𖽐𖼮𖽻 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼣𖽑𖽪𖾐 𖼺𖽘𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 13 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼮𖽡𖾑 𖼄𖽙 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼺𖽱𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼁𖽠 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐. 14 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽑𖽡𖾏：
𖼀𖽡𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼋𖽻, 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽺𖾐 𖼮𖽑𖽜 𖽃𖽡𖾑 𖼖𖽺𖾑？
15 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼙𖽻,
