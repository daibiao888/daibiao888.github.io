𖼽𖽙 𖼀𖽳 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 :
2 “𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 ,
𖼎𖽺𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼞𖽪𖾑 .
3 𖼷𖽑𖽜𖾑 𖼮𖽑𖽙 𖼯𖽔𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼚𖽙𖾏 𖼦𖽻！
4 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼾𖽙 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽺𖾑 ?
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽡𖾐 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖼘𖽪 ?
5 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 ,
𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼀𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 .
6 𖼷𖽑𖽔𖾏 𖼟𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽐𖼊𖽱𖾏 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 ,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 .
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐,
𖼊𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 ?
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼊𖽪𖾐 𖼘𖽑𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 .
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 ,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾑 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼮𖽘𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼉𖽳𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽨𖾏 𖼨𖽑𖽳𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽻𖾐 𖼉𖽳𖾏 .
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽷𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼊𖽪𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼮𖽡𖽻𖾐 𖼺𖽱𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏.
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼢𖽜𖾏 𖽐𖼎𖽙 𖼢𖽜𖾏 𖽐𖼎𖽔 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 ,
𖼎𖽺𖾏 𖼑𖽜 𖼖𖽡𖾑 𖼘𖽪𖾏 𖼳𖽘𖾐 𖼙𖽔 𖼚𖽙𖾏 .
13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 ,
𖽃𖽡𖾐 𖽐𖼮𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 .
14 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : 𖼀𖽜𖾑 𖼀𖽡𖾐 𖼙𖽠𖾏 !
𖼀𖽡𖾐 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼀𖽻𖾐 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 .
15 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 ?
𖼀𖽡𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼖𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼖𖽺𖾑 ?
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪 ?
16 𖼑𖽔𖾏 𖼘𖽪 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼻𖽡𖽝𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 ,
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼡𖽔𖾏 𖼞𖽪𖾑 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .
 
17 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼸𖽠𖾏 𖼋𖽔 ,
𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼸𖽠𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽺𖾑 ?
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 , 𖼞𖽑𖽡𖽠𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ?
18 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼺𖽱𖾏 𖽃𖽔𖾑 𖽐𖼊𖽻𖾏 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 ,
𖼺𖽱𖾏 𖼦𖽡𖾑 𖼁𖽘𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼷𖽑𖽔𖾐 𖼙𖽠𖾏 𖼖𖽺𖾑 ?
19 𖼅𖽡 𖼐𖽑𖽡𖾏 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼳𖽻𖾏 𖼦𖽡𖾏 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽝𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 ,
𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 , 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 .
20 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏 ,
𖼾𖽨𖾏 𖼦𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 .
21 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏 , 𖼐𖽱𖾏 𖼋𖽡𖽝𖾏 𖼡𖽔𖾑 𖼸𖽠𖾏 𖼞𖽳𖾑 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼘𖽪 ?
22 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽠𖾏 𖼨𖽑𖽢𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 ,
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼞𖽑𖽡𖽠𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼖𖽺𖾑 ?
23 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽔，𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼺𖽷𖾐 ,
𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼺𖽷𖾐 .
24 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼊𖽑𖽻𖾐 𖼄𖽡𖾏 𖼀𖽪𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼺𖽷𖾐 , 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼽𖽘𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾐 𖼐𖽻𖾏 𖼺𖽷𖾐 .
25 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽔 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼑𖽻𖾏 𖽃𖽦𖾐 ,
𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽑𖽡𖽝𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼑𖽡𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼦𖽘𖾏.
26 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽙 ,
𖼞𖽘𖾑 𖼞𖽘𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 .
 
27 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼅𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 ,
