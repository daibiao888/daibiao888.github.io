𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏

23
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠.
2 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼳𖽙𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐,
𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏.
3 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼖𖽡𖽾 𖼯𖽷𖾏 𖼗𖽙𖾏,
𖼑𖽜 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼷𖽷𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏.
 
4 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼦𖽘𖾑 𖼊𖽙𖾐 𖼦𖽘𖾏 𖼡𖽑𖽠𖾏 𖼟𖽪𖾏 𖼚𖽪𖾐 𖼈𖽡𖽝𖾏,
𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐,
𖼟𖽡 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏, 𖼟𖽡 𖼊𖽪𖾑 𖼡𖽠𖾏, 𖼖𖽔𖾏 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼞𖽪𖾑.
 
5 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏,
𖼟𖽡 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼅𖽳𖾏 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏.
6 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼷𖽑𖽷𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐 𖽐𖼎𖽙 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼖𖽠𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖽐𖼚𖽙, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
