24
𖼊𖽡𖾐 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼊𖽡𖾐,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽡𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼁𖽡𖽝.
2 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼊𖽡𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼎𖽻𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑.
 
3 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐?
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑?
4 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼚𖽠𖾐, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖼦𖽡𖾐 𖼚𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼏𖽵 𖼗𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏.
5 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼎𖽺𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳.
6 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡 𖼚𖽑𖽪𖾏 𖽃𖽡𖾑. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
7 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼗𖽙𖾏！
𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏! 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐！
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽃𖽡𖾑 𖼖𖽔𖾑 𖼁𖽺𖾏 𖼋𖽔𖾏 𖼋𖽘!
8 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏！
9 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼗𖽙𖾏！
𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏! 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏！
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼖𖽔𖾑 𖼁𖽺𖾏 𖼋𖽔𖾏 𖼋𖽘!
10 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏！ （𖼳𖽡𖾏 𖼖𖽔𖾑）
