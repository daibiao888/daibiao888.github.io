𖼽𖽔𖾑 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖽐𖼎𖽙 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑, 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼑𖽻 .
 
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏,
𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑,
𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏.
6 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼊𖽪𖾐 𖼘𖽑𖽔𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏，
7 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼦𖽡𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑.
8 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐,
𖼟𖽪𖾏 𖼷𖽑𖽷𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏, 𖼟𖽪𖾏 𖼷𖽑𖽷𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝 𖼊𖽱𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
 
9 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼯𖽡𖽙𖾏 𖼏𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐, 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼐𖽑𖽜𖾑,
𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏, 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼑𖽡𖾏 𖼡𖽜𖾐 𖼙𖽠𖾏.
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼻𖽙𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽔𖾏.
11 𖼎𖽺𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏
𖽐𖼎𖽙 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏.
12 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 , 𖼐𖽡𖽙𖾐 𖼺𖽙𖾐 𖼣𖽔𖾑 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑.
13 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼇𖽺𖾐 𖼺𖽘𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖾏 𖼅𖽻𖾏,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 .
14 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐,
𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼎𖽑𖽻𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏, 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
15 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼟𖽙𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽐𖼀𖽑𖽔𖾐 𖼺𖽘𖾏,
𖼄𖽔𖾐 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽻𖾏, 𖽃𖽻𖾐 𖼮𖽡𖽻𖾐 𖼺𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖼊𖽙𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 .
16 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽃𖽻𖾐 𖼎𖽑𖽻𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙 ,
𖼟𖽙𖾏 𖽃𖽻𖾐 𖼺𖽱𖾏 𖼖𖽪𖾑 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏, 𖽐𖼚𖽱𖾏 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 .
 
17 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 ,

18 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼙𖽱 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 .
19 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 :
“𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼖𖽺𖾑? ”
20 𖼐𖽱𖾏 𖼙𖽔 𖽐𖼊𖽻𖾏 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏, 𖼟𖽙𖾏 𖽃𖽻𖾐 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐,
𖼋𖽡𖽝𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾑 𖼸𖽠𖾏 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼘𖽪?

21 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
