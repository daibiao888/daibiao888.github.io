7 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐. 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼊𖽙𖾑 𖼽𖽡𖾏, 𖼟𖽪𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽃𖽔𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑.
8 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽪 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼄𖽡𖾐 𖼖𖽳 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼄𖽑𖽪𖾑 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏？ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪𖾏 𖽃𖽺𖾐 𖼏𖽠𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏.” 9 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼖𖽔𖾑 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼊𖽔𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽙𖾑 𖼖𖽡 𖼽𖽔𖾏 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑, 𖼎𖽑𖽻𖾏 𖽐𖼊𖽻𖾐 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼮𖽡𖽝𖾏, 𖼟𖽡 𖼯𖽔 𖼽𖽔𖾑 𖼯𖽡𖾏 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼅𖽻𖾏， 𖼋𖽠𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏.” 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽡𖾑！ 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑.”
𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖼩𖽻𖾏
10 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼺𖽻𖾐 𖼘𖽙, 𖼀𖽜𖾑 𖼸𖽙 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼮𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 11 𖼽𖽔𖾏 𖼮𖽡 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼮𖽡 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼖𖽺𖾑? 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼅𖽡 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏:
