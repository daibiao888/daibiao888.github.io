𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖽐𖼮𖽡𖽙𖾏 𖽃𖽔𖾐 𖼑𖽘
𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽐𖼊𖽜 𖼀𖽻𖾐 𖼟𖽳
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 3：1 -15）
1
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 — 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 .
2 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼟𖽪𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼊𖽑𖽡𖽝𖾐 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼋𖽔𖾏 . 3 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖽝𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 𖼙𖽠𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽃𖽔𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 . 4 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 · 𖼽𖽦𖾑 𖼖𖽡 𖼮𖽑𖽹𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼑𖽜 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 , 5 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖼋𖽻𖾏 𖼟𖽪𖾏 𖼦𖽪𖾏 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖽂𖽪𖾑 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽡𖾐 𖼺𖽔𖾐 𖼖𖽡𖽝 𖽃𖽔𖾏 𖽃𖽡𖾑 , 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼀𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 . 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼅𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 6 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽐𖼮𖽡𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽔𖾑 𖼏𖽘 𖼋𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏 𖼯𖽦𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 .
7 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖽐𖼊𖽜 𖼋𖽪 𖼳𖽷𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 .” 8 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼙𖽔 𖼺𖽘𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ! 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 . 10 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼟𖽳 , 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼁𖽺𖾏 𖼗𖽙𖾏 , 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 , 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼨𖽑𖽢𖾐 𖽐𖼎𖽑𖽹𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑 ?” 11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼷𖽑𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽡 𖼅𖽔𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼊𖽜 𖼅𖽔𖾏 𖼊𖽻𖾏 、 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾐 、 𖽐𖼚𖽻 𖼮𖽡𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼖𖽻𖾐 𖼊𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼟𖽡 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼗𖽡𖾏 , 𖼮𖽔𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽙𖾏 𖼻𖽻𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽔𖾐 . 12 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽻𖾏 𖼟𖽡, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 、 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 、 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼟𖽡. 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 , 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 .”
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 10：26 - 29）
