𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼘𖽑𖽙𖾐. 2 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼮𖽹𖾑 𖼗𖽙𖾏 𖼑𖽻𖾏 𖼋𖽘 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼡𖽻𖾐 𖼗𖽙𖾏.” 3 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽹𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏:‘ 𖼅𖽡 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼋𖽔𖾏, 𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼚𖽑𖽪𖾏 𖼋𖽘𖾏.’ 4 𖼟𖽡 𖼮𖽔𖾑 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼷𖽔𖾑 𖼞𖽱𖾑 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼑𖽜 𖼟𖽡 𖼅𖽔𖾏 𖼡𖽻𖾐； 5 𖼟𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏: ‘𖼅𖽡 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽔𖾏, 𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼚𖽑𖽪𖾏.’ ” 6 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼟𖽡𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼅𖽔𖾏 𖼷𖽔 𖼞𖽱𖾑 𖼺𖽷𖾐?” 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏:“ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼑𖽪 𖼸𖽙 𖼀𖽡𖾐 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼑𖽜𖾏 𖼮𖽘𖾏, 𖼑𖽪 𖼐𖽑𖽡𖾏: 𖼅𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼋𖽡𖽝𖾏 𖼡𖽑𖽔𖾑 𖼐𖽡𖽙𖾐 𖼺𖽷𖾐 𖼘𖽪? 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼡𖽑𖽔𖾑 𖼅𖽔𖾏 𖼞𖽱𖾑 𖼊𖽑𖽡𖽝𖾐 𖼘𖽪? 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼑𖽪 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏, 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼋𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼖𖽺𖾑?” 8 𖼽𖽨 𖼊𖽔𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼊𖽠𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏， 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼟𖽡, 𖽐𖼎𖽙 𖼣𖽔𖾑 𖼽𖽻𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽔. 9 𖼞𖽪𖾑 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾑, 𖼟𖽡 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼋𖽡𖾏 𖼖𖽠𖾑 𖼾𖽱, 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼞𖽪𖾑 𖼯𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼎𖽷𖾏 𖼾𖽔𖾏 𖼐𖽡𖾑. 10 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽔𖾑 𖼊𖽳𖾑𖽑, 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖼊𖽪𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏.”
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼵𖽪𖾐 𖼳𖽤𖾏、𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏、𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏、𖼄𖽪 𖼽𖽙、𖽃𖽔 𖼛𖽡𖾏 𖼳𖽪𖾏、𖽃𖽔 𖼛𖽡𖾏 𖼷𖽷𖾑 𖼞𖽑𖽡𖽠𖾐, 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽃𖽡𖾑 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼮𖽡𖽝𖾏, 𖽃𖽔𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼅𖽻 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑. 12 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽡𖽻𖾐 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼑𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼅𖽡 𖼋𖽡𖾏 𖼇𖽜𖾑 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽡𖽙𖾏 𖼅𖽻 𖼊𖽑𖽡𖽝𖾐, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼯𖽡𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏 𖼐𖽔𖾑. 13 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼷𖽔𖾑 𖼞𖽱𖾑, 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑. 14 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐, 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽻𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏. 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼺𖽘𖾏 𖼖𖽡𖾏 𖼖𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽻𖾏 𖼋𖽘𖾏！”
𖼽𖽙 𖼺𖽳 𖼑𖽜 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏
15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖽃𖽡𖾑, 𖼮𖽡𖽻𖾐 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼑𖽡𖽝𖾏 𖼊𖽑𖽡𖽝𖾐, 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐, 𖼳𖽠𖾑 𖼊𖽠𖾏 𖽐𖼡𖽜𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡, 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼽𖽙 𖼺𖽳 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
16 𖼽𖽙 𖼺𖽳 𖼁𖽙𖾏 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽔𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼨𖽑𖽳𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼯𖽦𖾏, 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽙 𖼡𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼳𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏.” 17 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼺𖽳 𖽐𖼞𖽔𖾏. 18 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼨𖽑𖽳𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼺𖽳 𖽐𖼞𖽔𖾏, 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼐𖽑𖽡𖾏: “ 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽻𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼖𖽪𖾑 𖼇𖽜𖾑 𖼊𖽜𖾑 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖼦𖽺𖾏 𖼀𖽡𖾐, 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑, 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼀𖽡𖾐 𖼑𖽜𖾏 𖼖𖽔𖾑.” 19 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼛𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏.
20 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪！𖼟𖽻𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼋𖽔𖾏, 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼋𖽔𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼊𖽔𖾐. 21 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼀𖽡𖾐 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼳𖽙𖾏 𖼀𖽱𖾏 𖼋𖽘𖾏, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼡𖽑𖽜𖾐 𖼇𖽜𖾑, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙, 𖼀𖽡𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼊𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼷𖽻𖾏 𖼺𖽷𖾐, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼐𖽡𖽙𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼇𖽜𖾑 𖽐𖼎𖽙𖾏, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼋𖽘𖾏. 22 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼑𖽡𖽝𖾏 𖼊𖽑𖽡𖽝𖾐 𖼋𖽔𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽐𖼚𖽙 𖼇𖽜𖾑 𖼮𖽡𖽝𖾏.” 23 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽡𖾑 𖼎𖽻𖾏 𖼅𖽡 𖽐𖼚𖽙 𖼇𖽜𖾑 𖼮𖽡𖽝𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼞𖽪𖾑 𖼺𖽪𖾐 𖼙𖽔 𖼷𖽹𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏.” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 24 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼺𖽳 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏, 𖼄𖽔𖾐 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽔𖾑 𖼊𖽠𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼡𖽜𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽪𖾐 𖼑𖽜𖾏 𖼖𖽔𖾑. 25 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼋𖽡𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖽃𖽡𖾑, 𖼋𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼑𖽻𖾏 𖼳𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼗𖽙𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼑𖽻𖾏 𖼉𖽔𖾏.
