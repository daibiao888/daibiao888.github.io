𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑、 𖼽𖽙 𖼊𖽑𖽗𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐、 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽡𖾏 𖼳𖽱 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽡𖾐 𖼮𖽢𖾑 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ,𖼈𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 . 2 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼾𖽡𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾏 𖼖𖽔 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏.” 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼮𖽑𖽹𖾏 𖼟𖽡𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽻 , 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖽃𖽔𖾏 𖼼𖽙 𖽃𖽡𖾐 𖼼𖽘𖾐.
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐 𖼳𖽷𖾑 𖼽𖽔𖾏 · 𖼽𖽔𖾐 𖼳𖽪 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼸𖽙 𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽻𖾐 𖼡𖽔𖾑 𖼸𖽷 𖼑𖽡𖾏 𖼟𖽱𖾏 𖽃𖽻𖾐, 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽻𖾏 𖽐𖼊𖽻𖾐 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼇𖽘𖾑 𖼅𖽻𖾏 𖼷𖽹𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 ， 4 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼽𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽡𖾏 𖼳𖽱 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐,𖽃𖽔𖾐 𖼖𖽑𖽘𖾐 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 , 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏. 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼖𖽔 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 , 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽺𖾏 𖼟𖽡， 6 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 , 𖼮𖽑𖽙 𖼐𖽱𖾏, 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼐𖽱𖾏, 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼮𖽘𖾏 𖼟𖽙𖾏 𖼊𖽑𖽔𖾑 𖼀𖽡𖾐 𖼖𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：
𖼋𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑.
8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 ,
𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼑𖽡𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼳𖽱 ,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖽐𖼚𖽙, 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼞𖽪𖽙 𖼊𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏.
9 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 ,
𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐．
𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜,
