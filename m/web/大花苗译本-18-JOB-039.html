11 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼘𖽑𖽙𖾐 𖼎𖽺𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 ?
𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼄𖽔𖾐 𖼟𖽡 𖼑𖽪𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼘𖽪 ?
12 𖼋𖽡𖽝𖾏 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼟𖽡 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼨𖽑𖽹𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 ?

29 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼚𖽺𖾏 𖼑𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 ,

30 𖼐𖽱𖾏 𖼊𖽪𖾑 𖼣𖽔𖾏 𖼣𖽔𖾏 𖼖𖽔𖾏 𖽐𖼷𖽹𖾏 𖽐𖼮𖽑𖽘𖾑 ,

𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 .”
