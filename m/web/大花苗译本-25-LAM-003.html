𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼐𖽑𖽡𖾏 𖼽𖽔𖾑 𖽐𖼮𖽷𖾐 𖼮𖽵𖾑 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐
3
𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼙𖽔 𖼟𖽱𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏.
2 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼦𖽡𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
3 𖼐𖽱𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏
𖼯𖽪 𖼯𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑.
4 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽐𖼎𖽙 𖽐𖼡𖽹𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡,
𖼐𖽱𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 ( “𖼖𖽙𖾑 𖼺𖽘𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼸𖽺𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐” ) 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏.
5 𖼐𖽱𖾏 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 ,
𖼎𖽑𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖽐𖼎𖽙 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼞𖽪𖾑 .
6 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
 
7 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼉𖽘𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏,
𖼐𖽱𖾏 𖼊𖽑𖽡𖽝𖾐 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽑𖽤𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑.
8 𖼞𖽪𖾑 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽜 𖼟𖽪𖾏 𖼮𖽨𖾏,
𖼐𖽱𖾏 𖼖𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
9 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼙𖽔 𖼸𖽻, 𖽐𖼎𖽳𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼮𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼖𖽡𖾐 𖽐𖼞𖽑𖽡𖾐 𖼖𖽡𖾏 𖽐𖼞𖽑𖽻𖾏.
 
10 𖼐𖽱𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼚𖽹𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼯𖽡𖽝𖾏 𖼋𖽙𖾏 𖼞𖽪𖾑 ,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐.
11 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼎𖽙𖾑 𖼇𖽳𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏,
𖼄𖽔𖾐 𖼞𖽪𖾑 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼙𖽔 𖼟𖽡𖾏 𖼸𖽡𖽝𖾏 𖼊𖽙𖾏.
 
12 𖼐𖽱𖾏 𖽐𖼊𖽘𖾏 𖼐𖽑𖽜𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼀𖽙𖾑.
13 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖽐𖼚𖽙 𖼊𖽪𖾑 𖼈𖽙𖾏 𖼀𖽙𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽐𖼀𖽱𖾏.
 
14 𖼞𖽪𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼚𖽙𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼨𖽑𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑.
15 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽻𖾏 𖼄𖽡𖾏 𖼇𖽪𖾑 𖼟𖽪𖾏 𖼮𖽻𖾏.
16 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼐𖽑𖽡𖽝𖾑,
𖼎𖽑𖽻𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖽐𖼀𖽙 𖼮𖽡𖽝𖾏.
 
17 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼡𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼟𖽪𖾏 𖼚𖽡𖾐,
𖼞𖽪𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
18 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼯𖽻 𖼼𖽙 ,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽐𖼮𖽷𖾏 𖼊𖽜𖾑!”
 
19 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖽐𖼎𖽙 𖼳𖽔𖾏 𖽐𖼮𖽡 ，
𖼟𖽪𖾏 𖼺𖽱𖾏 𖼄𖽡𖾏 𖼇𖽪𖾑 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼮𖽷𖾐 𖼟𖽪𖾏 𖽃𖽦𖾐.
20 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 ,
𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼊𖽳𖾏.
 
21 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 ,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 ．
22 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽪𖾏,
𖼐𖽱𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽘．
23 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼟𖽪𖾏 𖼮𖽑𖽦𖾐 𖼯𖽡𖽙 ,
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼋𖽡𖽝 .
24 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼀𖽠,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 .”
 
25 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖽃𖽡𖾑,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 .
26 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 .
27 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼚𖽑𖽙𖾏 𖼖𖽡𖾑 𖼡𖽔𖾐,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 .
28 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼗𖽪𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽑𖽨𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽔𖾑.
29 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼻𖽡𖽝𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼸𖽠𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏.
30 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼸𖽷𖾏,
𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏.
 
31 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
32 𖼮𖽵𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽳𖾏 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼺𖽘𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐.
33 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖽃𖽦𖾐,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏.
 
34 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼞𖽱𖾏 𖼮𖽡𖽝𖾏 𖼯𖽪𖾏 (“𖼯𖽪𖾏 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖼸𖽹𖾏 ” )𖼎𖽻𖾏 𖼡𖽘𖾐 𖼊𖽠𖾏,
35 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
36 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽪𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼎𖽙𖾑 𖼺𖽘𖾏 𖼋𖽡𖽝 𖼮𖽑𖽵𖾑,
𖼋𖽡𖾏 𖼐𖽡𖾑 𖼮𖽵𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼏𖽻 .
 
37 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼦𖽡𖾐 𖼏𖽵 𖼮𖽡𖽝𖾏,
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼊𖽡𖽝𖾑 𖼎𖽺𖾏 𖼊𖽡𖽝𖾑 𖼖𖽺𖾑?
38 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽐𖼎𖽙 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼖𖽺𖾑?
39 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼖𖽺𖾑?
 
40 𖼀𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼻𖽻𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼾𖽨𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏,
𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
41 𖼀𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝
𖼮𖽘𖾏 𖼋𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
42 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼷𖽜𖾑 𖼟𖽡 ,
𖼟𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽔𖾏 .
 
43 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖼊𖽠𖾏 𖽐𖼎𖽙𖾐, 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼀𖽡𖾐 ,
𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏.
44 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼚𖽪𖾐 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼾𖽨𖾏 𖼮𖽡𖽝𖾏,
𖼟𖽙𖾏 𖼀𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑.
45 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐
𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏.
 
46 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾐.
47 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖽐𖼎𖽙 𖼡𖽑𖽙𖾑 𖼘𖽑𖽻𖾑 ,
𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 , 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼀𖽡𖾐.
48 𖼑𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 ,
𖼞𖽪𖾑 𖼞𖽔𖾏 𖼅𖽔 𖽐𖼚𖽱𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐.
 
49 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏,
𖽐𖼚𖽱𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽪𖾏.
50 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽻𖾏 𖼮𖽳𖾐,
𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽔𖾏 .
51 𖼑𖽜 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼙𖽻 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐.
52 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽡𖾑 𖼖𖽠𖾑 𖼯𖽡𖽻𖾏 𖼞𖽪𖾑 ,
𖼺𖽱𖾏 𖼖𖽠𖾑 𖼐𖽻 𖼯𖽡𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
53 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽻𖾐 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑．
54 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽻𖾐 𖽐𖼚𖽱𖾏 𖼮𖽑𖽡𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼛𖽙𖾏 𖼳𖽦𖾑 𖼋𖽘𖾏 !”
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼮𖽵𖾑 𖼚𖽠𖾏 𖼮𖽨𖾏
55 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！
𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏.
56 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽡 𖼖𖽔𖾏 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼚𖽠𖾏 𖼮𖽨𖾏 :
“𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽠𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽻 .”
57 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 ,
𖼟𖽡 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏!”
 
58 𖼮𖽵𖾑 𖼘𖽪！ 𖼟𖽡 𖼨𖽑𖽢𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖼟𖽡 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑.
59 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼁𖽙𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼞𖽡𖾑.
60 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼞𖽪𖾑, 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼦𖽺𖾏 𖼞𖽪𖾑,
𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏.
 
61 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼞𖽪𖾑 ,
𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑,
