20 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼊𖽑𖽹𖾏 𖼊𖽪𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐，𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼋𖽔𖾏，𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼽𖽔𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏． 21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ： “𖼟𖽡 𖼽𖽔𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏？” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ：“ 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼐𖽡𖾑，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼼𖽻𖾏 𖼟𖽡 𖼺𖽘𖾑 𖼺𖽷𖾏，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼼𖽻𖾏 𖼟𖽡 𖼺𖽘𖾑 𖼇𖽳𖾏．” 22 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 ： “ 𖼅𖽡 𖽐𖼊𖽜 𖽃𖽔𖾑 𖼳𖽷𖾏 𖼅𖽡 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐，𖼋𖽡𖽝𖾏 𖼅𖽡 𖼸𖽠𖾏 𖼦𖽻𖾏 𖼞𖽑𖽙𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼦𖽻𖾏 ？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼸𖽠𖾏 𖼯𖽡𖽙”． 23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 ： “ 𖼞𖽑𖽙𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼦𖽻𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼦𖽻𖾏 𖼯𖽡𖽙𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼼𖽻𖾏 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏 𖼼𖽻𖾏 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡 𖼇𖽳𖾏，𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼑𖽜 𖼡𖽔 𖼋𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏．” 24 𖼟𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙 𖽃𖽡𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏． 25 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼀𖽻𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼞𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏． 26 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼡𖽔 𖼋𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼅𖽡． 27 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼅𖽡 𖼡𖽑𖽜𖾑． 28 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼋𖽔𖾏，𖼦𖽡𖾐 𖼾𖽙 𖼽𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏，𖼐𖽱𖾏 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ， 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼳𖽦𖾑，𖼅𖽔𖾏 𖼮𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽔𖾐 𖼄𖽱𖾏
（𖼞𖽑𖽙𖾐 10：46-52；𖼖𖽪𖾏 18：35-43）
29 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐． 30 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖽐𖼊𖽪 𖼞𖽡𖾑，𖼐𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽡𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑，𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪 ！𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏， 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼀𖽡𖾐！” 31 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾏 ， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 ，𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼀𖽡𖾐！” 32 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼳𖽠𖾑 𖽐𖼎𖽙𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏，𖽐𖼎𖽙 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽪𖾑 𖼑𖽜 𖼅𖽡 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？” 33 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼀𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼸𖽠𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑．” 34 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼺𖽔𖾏 𖼺𖽔𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔，𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏．
