𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼁𖽻𖾏
22
“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼑𖽡𖽝 𖼑𖽨𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 , 𖼟𖽪𖾏 𖼋𖽺 𖼺𖽘𖾏 , 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼎𖽑𖽻𖾏 𖼀𖽜𖾐 𖼋𖽪 𖼑𖽨𖾏 𖼁𖽻𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼑𖽨𖾏 , 𖼚𖽻𖾐 𖼋𖽪 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖼁𖽻𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 . 2 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖽐𖼮𖽷𖾐 𖼸𖽜 𖽐𖼮𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖽐𖼞𖽔𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼸𖽜 𖽐𖼊𖽻𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽔 𖼺𖽘𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 . 3 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼋𖽘 , 𖼽𖽔𖾑 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 . 𖼮𖽔𖾑 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼸𖽜 , 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼁𖽻𖾏． 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼋𖽺 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 , 𖼁𖽻𖾏 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝 . 4 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼑𖽡𖽝 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽨𖾏、𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽜𖾏 𖼖𖽔𖾑、𖼡𖽑𖽙𖾑 𖼯𖽢 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏、𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼺𖽷𖾐 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽻𖾏 𖽃𖽔𖾐 𖽐𖼀𖽠 . ”
5 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼙𖽡𖽝𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼾𖽨 𖼯𖽦𖾏 , 𖼯𖽔𖾏 𖼟𖽪𖾏 𖼯𖽦𖾏 𖼅𖽻𖾏 𖼑𖽻𖾏 𖼗𖽔 𖼡𖽻𖾐 , 𖼯𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼾𖽨𖾏 𖽐𖼎𖽹𖾏 𖼙𖽡𖽝𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼁𖽻𖾏 . ”
6 “𖼮𖽔𖾑 𖼮𖽡𖾑 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 , 𖼊𖽻𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼗𖽔 𖼡𖽔𖾑 𖼷𖽻𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼷𖽱𖾏 𖼮𖽡𖽝𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽐𖼮𖽳 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼉𖽘𖾏 𖼡𖽻𖾐 , 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽻𖾏. ”
7 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼄𖽔𖾐 𖼑𖽡𖽝𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼎𖽻𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼑𖽔𖾏 𖼻𖽙𖾑 , 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼏𖽻 𖼸𖽜 𖼑𖽡𖽝 𖼙𖽠𖾏, 𖼮𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼸𖽜 𖼋𖽘𖾏 , 𖼷𖽷𖾑 𖼸𖽜 𖼽𖽔𖾑 𖼁𖽻𖾏 𖽃𖽔𖾐 𖽐𖼀𖽠 , 8 𖼮𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼸𖽜 , 𖼷𖽷𖾑 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐（“𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏”） , 𖼖𖽔𖾑 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 . ”
9 “𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼾𖽔𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽜 𖼑𖽨𖾏、𖼑𖽜 𖼑𖽜𖾏 𖼖𖽔𖾑、𖼑𖽜 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏、𖼑𖽜 𖼮𖽑𖽙𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼗𖽠𖾏 , 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝 , 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 （“𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏”） , 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼏𖽵 𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 , 𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖼁𖽻𖾏 𖽃𖽔𖾐 𖽐𖼀𖽠 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 . ”
10 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼄𖽔𖾐 𖼑𖽜𖾏 𖼖𖽔𖾑 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽨𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼯𖽦𖾏 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼑𖽔𖾏 𖼻𖽙𖾑 , 𖼯𖽦𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼋𖽔 𖼗𖽠𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽻𖾏 𖼄𖽙𖾐 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼙𖽠𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽙𖾏 , 11 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼏𖽵 𖼗𖽪𖾏 , 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼞𖽑𖽡𖽠𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 , 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪 𖼉𖽪𖾏 𖼉𖽪𖾏 , 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼁𖽻𖾏 . 12 𖼯𖽦𖾏 𖼮𖽔𖾑 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼊𖽻𖾏 𖼑𖽡𖽝 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼮𖽵𖾑； 13 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼺𖽘𖾏 , 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼗𖽙 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽷𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑 𖼦𖽡𖾏 𖼅𖽪𖾏 𖼁𖽻𖾏 . ”
14 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼗𖽙𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼋𖽔 𖼺𖽘𖾏 , 𖼯𖽦𖾏 𖼮𖽵𖾑 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽻𖾏； 15 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼁𖽻𖾏； 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼞𖽻𖾏 ( “𖼞𖽻𖾏”𖼟𖽪𖾏 “𖼡𖽑𖽜𖾐 𖽐𖼡𖽜𖾏 𖼼𖽙”) 𖼗𖽙𖾏 𖼊𖽔𖾐 , 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼁𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼊𖽜𖾑 𖼐𖽔𖾑 . ”

25 “𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 , 𖼟𖽡 𖼮𖽔𖾑 𖼷𖽹𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼮𖽙𖾏 𖽐𖼮𖽷𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼳𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 . 26 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼦𖽡𖾐 𖽐𖼷𖽜𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏； 27 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼮𖽑𖽙𖾏 𖼐𖽡𖾑 𖼈𖽳𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼊𖽔𖾐 , 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼈𖽳𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 , 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼈𖽳𖾑 𖼀𖽱𖾏 𖼖𖽺𖾑 ? 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽡𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼅𖽳𖾏 . ”
