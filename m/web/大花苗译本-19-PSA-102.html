3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼺𖽱𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼯𖽡 𖼮𖽡𖽝𖾏.
4 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼄𖽙𖾐, 𖼺𖽱𖾏 𖽐𖼡𖽜𖾐 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼺𖽘𖾏.
5 𖼑𖽜 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏 𖼐𖽔𖾑 ,
𖼞𖽪𖾑 𖽐𖼡𖽹𖾏 𖼻𖽡𖽝𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏.
6 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼤𖽪𖾏 𖼑𖽻𖾏 𖽐𖼀𖽳𖾏 ,
𖼞𖽪𖾑 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼑𖽡𖾏 𖼊𖽔𖾑 𖼞𖽙𖾏 𖼞𖽙𖾏 .
7 𖼞𖽪𖾑 𖼯𖽷𖾏 𖼻𖽙𖾑 𖼦𖽡𖾐 𖼀𖽱𖾏,
𖼞𖽪𖾑 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼎𖽪𖾏 𖽐𖼞𖽔𖾏 𖼊𖽪𖾑 𖼀𖽡𖾏 𖼮𖽑𖽡𖽙𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼗𖽔 .
8 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼞𖽪𖾑 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼏𖽵 𖼗𖽪𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏.
9 𖼞𖽪𖾑 𖼑𖽻𖾏 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼮𖽑𖽻𖾑, 𖼺𖽱𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼦𖽻𖾏 𖽃𖽡𖾑 𖼳𖽡𖽠𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼎𖽻𖾏 𖼉𖽺𖾏.
10 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼟𖽡 𖼳𖽪𖾐 𖽐𖼎𖽙 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼟𖽡 𖼮𖽘𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐, 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼺𖽘𖾏.
11 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼺𖽱𖾏 𖼡𖽔𖾏 𖽐𖼷𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼖𖽔𖾑 𖽐𖼷𖽜𖾑,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼺𖽱𖾏 𖽐𖼡𖽜𖾐 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 .
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐𖼳𖽡 𖼣𖽔𖾑
12 𖼮𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼷𖽷𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏,
𖼟𖽡 𖽐𖼷𖽷𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼸𖽙 𖼉𖽻𖾏 𖼷𖽑𖽷𖾏.
13 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼳𖽡 𖼣𖽔𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼳𖽡 𖼣𖽔𖾑 ,
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼸𖽙 𖼋𖽘𖾏.
14 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼳𖽡 𖼣𖽔𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐,

15 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏,

16 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼮𖽡𖽝𖾏,

17 𖼐𖽱𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼑𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖽐𖼮𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
 
18 𖼋𖽡𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼮𖽡𖽝𖾏,
𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.

𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐，
20 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼞𖽱𖾏 𖼮𖽡𖽝𖾏 𖼐𖽱𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏,
𖼽𖽔𖾑 𖼷𖽷𖾐 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 𖽃𖽡𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑，

𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏，
22 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙

23 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼯𖽻 𖼼𖽙 𖼺𖽘𖾏,
