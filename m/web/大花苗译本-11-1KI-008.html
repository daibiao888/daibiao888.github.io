（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 5：2 - 6：2）
8
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼗𖽻 𖽐𖼎𖽙 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑, 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏, 𖼎𖽑𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼺𖽘𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏. 2 𖼽𖽡𖾑 𖼊𖽑𖽔𖾑 𖼐𖽡𖽝𖾏 𖼖𖽑𖽡𖾏, 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼳𖽤𖾏 𖼖𖽑𖽡𖾏, 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼗𖽻 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏, 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼎𖽺𖾏 𖼞𖽱𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏. 4 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼨𖽑𖽹𖾑 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽡𖽙𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏. 5 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖽃𖽡𖾐 𖼸𖽻 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼖𖽑𖽘𖾑 𖼑𖽨𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑, 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼾𖽱 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏. 6 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼞𖽱𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽤𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙, 𖼟𖽪𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼑𖽡𖾏 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐. 7 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖽐𖼊𖽑𖽔𖾑 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑, 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽐𖼎𖽙 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏. 8 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖽐𖼊𖽡𖾑, 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼡𖽠𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖼡𖽻𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏, 𖼸𖽙 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼺𖽷𖾐. 9 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽐𖼚𖽙 𖼯𖽔 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼊𖽔𖾐, 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐 𖼦𖽙 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑． 𖼋𖽠𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐. 10 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏； 11 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏.
𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏
12 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼐𖽑𖽡𖾏, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼚𖽙.
13 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼋𖽘𖾏,
𖽃𖽔𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 .”

27 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼑𖽔𖾏 𖼘𖽪！ 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏, 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐, 𖼦𖽔𖾏 𖼐𖽔𖾑 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽱𖾏 𖼐𖽡𖾑 𖼘𖽪? 28 𖼯𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽻 𖼡𖽑𖽜𖾑 𖼑𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼎𖽙 𖽐𖼊𖽜 𖼽𖽔𖾑, 𖼑𖽻 𖼡𖽑𖽜𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜. 29 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑, 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼏𖽵 𖽃𖽔𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏. 30 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼑𖽻 𖼊𖽡𖾐, 𖼑𖽻 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼯𖽔𖾏 𖼾𖽔𖾏.
31 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼼𖽷𖾏 𖼼𖽙, 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼏𖽵 𖼗𖽪𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏, 𖼐𖽱𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼏𖽘 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼏𖽵 𖼗𖽪𖾏， 32 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽻 𖼊𖽡𖾐, 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏, 𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏, 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼋𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼅𖽔𖾏 𖼞𖽡𖾑, 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏.
