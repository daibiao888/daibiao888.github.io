7
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼟𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳, 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼋𖽔𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏. 2 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽔 𖼊𖽗𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽻 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖽐𖼚𖽙, 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼯𖽡𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖽐𖼚𖽙.” 3 𖼐𖽔 𖼊𖽗𖾑 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼯𖽪𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐.” 4 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼐𖽑𖽡𖾏: 5 “𖼟𖽡 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼖𖽔𖾏? 6 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼗𖽙𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼊𖽔𖾐, 𖼮𖽑𖽹𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑. 7 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼙𖽔 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼙𖽔 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽡𖾐 𖼳𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼊𖽪𖾐 𖼋𖽻, 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽔𖾑 𖼊𖽪𖾑 𖼾𖽨 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏: 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼑𖽜 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽺𖾑? 18 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ， 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏? 𖼞𖽪𖾑 𖼾𖽡𖾏 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏? 𖼟𖽡 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼏𖽔𖾏 𖼐𖽡𖾑? 19 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼺𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐, 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼎𖽺𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏. 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼮𖽷𖾐 𖽃𖽡𖾑 𖼘𖽪? 20 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼺𖽷𖾐 𖼘𖽪? 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏. 21 𖼟𖽡 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖾑, 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐, 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼐𖽔𖾑, 𖼖𖽔𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏. 22 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼟𖽡 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼙𖽔 𖼐𖽙𖾑𖽑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖽐𖼮𖽔𖾏 𖼋𖽻𖾏 𖼟𖽡, 𖼋𖽠𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 23 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖽐𖼮𖽔𖾏 𖼋𖽻𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼺𖽷𖾐 𖼘𖽪? 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼐𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑, 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺, 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐. 24 𖼟𖽡 𖼙𖽔 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝, 𖽃𖽔𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 25 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼏𖽵 𖼎𖽻𖾏 𖼡𖽑𖽜𖾑 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖽃𖽡𖾑, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏， 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏. 26 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽑𖽡𖾏: 𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼾𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏. 27 𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖼡𖽑𖽜𖾐 𖼡𖽑𖽳𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼟𖽡 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼮𖽙𖾏 𖼺𖽡𖽝𖾐 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑. 28 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽡 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼟𖽡 𖼎𖽺𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏. 29 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾐 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏, 𖼟𖽙𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼷𖽷𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼏𖽵 𖼮𖽡𖽝𖾏, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏！”
