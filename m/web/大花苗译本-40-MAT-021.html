（𖼞𖽑𖽙𖾐 11：1-11；𖼖𖽪𖾏 19：28-40；𖼽𖽙 12：12-19）
21
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐，𖼺𖽪𖾐 𖽐𖼮𖽡𖽻𖾏 𖼀𖽳 𖼇𖽔 𖼮𖽑𖽡 𖼟𖽪𖾏 𖼞𖽗𖾐 𖼖𖽗 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏， 2 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏：“ 𖼅𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡 𖼋𖽡𖾏，𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼡𖽑𖽹𖾐 𖼮𖽡𖽝𖾏，𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼊𖽑𖽡𖽝𖾐，𖼅𖽡 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼮𖽤𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑． 3 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：‘ 𖼮𖽵𖾑 𖼖𖽔𖾑 𖼽𖽔𖾑 𖼐𖽱𖾏 ．’ 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼚𖽠𖾏 𖼮𖽤𖾐 𖼋𖽔𖾏．” 4 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏：
5 “𖼽𖽔𖾑 𖽐𖼎𖽙 𖼳𖽡 𖼣𖽔𖾑𖼖𖽡𖾑 𖼉𖽪𖾏 （𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏“𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 ”）𖼐𖽑𖽡𖾏：
‘ 𖼑𖽔𖾏 𖼘𖽪! 𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼟𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑 !
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼄𖽔𖾏，𖼐𖽱𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 ,
𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑.’ ”
6 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏． 7 𖼮𖽤𖾐 𖼑𖽜𖾏 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼗𖽙𖾏 ，𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽙𖾏 𖼀𖽔𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼯𖽡𖽜𖾏 ． 8 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼚𖽠𖾏 𖼮𖽑𖽙𖾏 𖼀𖽔𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑，𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖽐𖼊𖽙𖾑 𖽃𖽔𖾑 𖼯𖽡 𖽐𖼊𖽻𖾏 𖼗𖽙𖾏 𖼀𖽔𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑， 9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：
“𖼦𖽙 𖼺𖽔𖾐 𖼐𖽔 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏！
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼮𖽵𖾑 𖽐𖼷𖽷𖾏 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏 ！
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐， 𖼦𖽙 𖼺𖽔𖾐 𖼐𖽔．”
10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ， 𖼖𖽪𖾑 𖼙𖽻 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏 𖼋𖽘𖾏， 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？” 11 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼙𖽻．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏
（𖼞𖽑𖽙𖾐 11：15-19；𖼖𖽪𖾏 19：45-48；𖼽𖽙 2：13-22）
12 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏，𖼖𖽠𖾑 𖼊𖽪𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼋𖽺 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏，𖼊𖽑𖽠𖾐 𖽐𖼷𖽜𖾑 𖼞𖽡 𖼏𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽪𖾑 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽺 𖽐𖼡𖽔𖾐 𖼞𖽡 𖼏𖽻𖾏 𖼟𖽪𖾏 𖼼𖽻𖾏， 13 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：
‘ 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，
𖼈𖽡𖽝𖾏 𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼸𖽜 𖼺𖽘𖾏．’”
14 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼅𖽔𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 ， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽻𖾏 𖼺𖽘𖾏． 15 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑，𖼎𖽺𖾏 𖼁𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 ： “𖼦𖽙 𖼺𖽔𖾐 𖼐𖽔 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏！” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝， 16 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼋𖽡𖽝𖾏 𖼟𖽡 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼾𖽙 𖼯𖽡𖽙𖾏， 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼐𖽑𖽡𖾏：‘ 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼎𖽙 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼋𖽠𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼨𖽑𖽪𖾑 𖼁𖽜 ’， 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼟𖽡𖽠 𖼺𖽷𖾐？” 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏，𖼅𖽻𖾏 𖼚𖽔𖾏 𖼀𖽳 𖼊𖽔𖾏 𖼐𖽡 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼀𖽱𖾏 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽔𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏
（𖼞𖽑𖽙𖾐 11：12-14；20-24）
18 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼺𖽷𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼙𖽻，𖼐𖽱𖾏 𖼮𖽑𖽹𖾐 𖼋𖽘， 19 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼐𖽡𖽙𖾐 𖼸𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾑，𖼎𖽺𖾏 𖼅𖽻𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 ，𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼯𖽔 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼚𖽻𖾏 𖼊𖽔𖾐， 𖼐𖽱𖾏 𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽳 𖼟𖽻𖾏 𖼐𖽔𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼟𖽡 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐．” 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏． 20 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏：“𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼡𖽑𖽔𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏．” 21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜，𖼦𖽡𖾐 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼷𖽑𖽠𖾑 𖽐𖼊𖽱𖾏， 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾐，𖼅𖽡 𖼮𖽔𖾑 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏：‘ 𖼮𖽑𖽹𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼀𖽻𖾐 𖼁𖽻𖾏 𖼚𖽔𖾏 𖼳𖽡𖽠𖾑 𖼙𖽠𖾏！’ 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼯𖽡𖽙𖾏． 22 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏， 𖼮𖽔𖾑 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼋𖽡𖽝，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜．”
𖼑𖽪 𖼡𖽔 𖼋𖽱𖾏 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼞𖽑𖽙𖾐 11：27-33；𖼖𖽪𖾏 20：1-8）
23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽷𖾏 𖼖𖽘𖾑， 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼗𖽻 𖼋𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼻𖽡𖽝𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑？ 𖼡𖽔 𖼋𖽱𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼟𖽡？” 24 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼑𖽪 𖼅𖽡 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 ， 𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑． 25 𖼽𖽙 𖼦𖽗 𖼑𖽡𖾏 𖽐𖼷𖽔𖾑 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏？𖼟𖽪𖾏 𖼋𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏？𖼖𖽔𖾏 𖼋𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽺𖾑？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏：‘ 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏 ’， 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：‘ 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼐𖽱𖾏 𖼖𖽺𖾑？’ 26 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏：‘ 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 ’， 𖼀𖽡𖾐 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼮𖽜 𖼽𖽙 𖼦𖽗 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐．” 27 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑．”

28 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽐𖼊𖽱𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏？ 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐．𖼐𖽱𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼐𖽑𖽡𖾏：‘ 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼘𖽪！𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏．’ 29 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：‘ 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼅𖽻𖾏 ’， 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼐𖽱𖾏 𖼚𖽠𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐， 𖼎𖽺𖾏 𖼅𖽻𖾏． 30 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼖𖽔𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼯𖽡𖽙𖾏．𖼊𖽪𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼐𖽑𖽡𖾏：‘ 𖼉𖽺𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼅𖽻𖾏，’ 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏． 31 𖼅𖽡 𖽐𖼊𖽱𖾏 ，𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼐𖽡𖾑 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼗𖽪𖾏 𖼘𖽪？ ” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼽𖽔𖾑 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼅𖽡 𖼺𖽪𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙． 32 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼦𖽗 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼅𖽡 𖼻𖽙𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳（𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼡𖽑𖽔𖾏 𖼇𖽺𖾏 𖼅𖽡 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ”），𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼐𖽱𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖽐𖼮𖽜 𖼐𖽱𖾏． 𖼅𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼖𖽔𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼚𖽠𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼐𖽱𖾏．”
