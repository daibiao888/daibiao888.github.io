𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑、𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝、 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑
14
“𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽳𖾏，𖼅𖽡 𖽐𖼮𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽜 𖼞𖽪𖾑 𖼊𖽑𖽡𖽝𖾐． 2 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼮𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏，𖼞𖽪𖾑 𖽐𖼷𖽙𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼋𖽘𖾏．𖼞𖽪𖾑 𖼅𖽻𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼅𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏． 3 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼅𖽻𖾏 𖼑𖽜 𖼅𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼊𖽡𖽝𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼅𖽡 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏． 4 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏， 𖼸𖽙𖾏 𖼞𖽡𖾑 𖽃𖽡𖾑 ， 𖼅𖽡 𖼀𖽻𖾐 𖼯𖽡𖽙．” 5 𖼊𖽙𖾑 𖼄𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼀𖽡𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖽃𖽡𖾑 𖼖𖽺𖾑？” 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑、𖼟𖽪𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝、𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑，𖼮𖽔𖾑 𖼦𖽡𖾐 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 7 𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 ，𖼯𖽡𖾏 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏．𖼋𖽳 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐， 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏，𖼎𖽺𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼋𖽘𖾏．” 8 𖼇𖽺𖾑 𖼖𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼢𖽳 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼑𖽔𖾏， 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼋𖽘𖾏．” 9 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼇𖽺𖾑 𖼖𖽡，𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼺𖽷𖾐 𖼖𖽔𖾏？𖼷𖽷𖾑 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑，𖼟𖽪𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏，𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏 ‘𖼢𖽳 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼑𖽔𖾏 ’？ 10 𖼋𖽡𖽝𖾏 𖼟𖽡 𖽐𖼮𖽜 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽐𖼚𖽙，𖼉𖽺𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙？𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏，𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏，𖼟𖽪𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼯𖽡𖽙𖾏． 11 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼞𖽪𖾑， 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽐𖼚𖽙， 𖼉𖽺𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙，𖼮𖽔𖾑 𖼦𖽡𖾐 𖽐𖼮𖽜， 𖼅𖽡 𖼁𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽜． 12 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼞𖽪𖾑，𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏，𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽡𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼙𖽠𖾏． 13 𖼅𖽡 𖼻𖽡𖽝𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼊𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏，𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼯𖽡𖽙𖾏，𖼟𖽙𖾏 𖼉𖽺𖾏 𖼑𖽜 𖼊𖽪𖾐 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏． 14 𖼅𖽡 𖼻𖽡𖽝𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖽐𖼊𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 ，𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼯𖽡𖽙𖾏．”
