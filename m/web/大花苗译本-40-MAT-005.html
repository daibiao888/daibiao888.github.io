5
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽙𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐， 𖼐𖽱𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏，𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐， 2 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：
（𖼖𖽪𖾏 6：20-23）
3 “𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽡𖽝．
4 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼾𖽔𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼎𖽳𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐．
5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼄𖽔𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐．
6 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏．
7 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼗𖽔 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐，
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐．
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽡𖽝．
11 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼅𖽡， 𖼟𖽡𖾏 𖼅𖽡， 𖼐𖽑𖽡𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼅𖽡，𖼅𖽡 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏． 12 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏． 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼅𖽡 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏．”
𖽐𖼮𖽳𖾑 𖽐𖼎𖽙 𖼎𖽻𖾐 𖼮𖽡𖾏
（𖼞𖽑𖽙𖾐 9：50；𖼖𖽪𖾏 14：34-35）
13 “ 𖼅𖽡 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼮𖽳𖾑． 𖽐𖼮𖽳𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑，𖼖𖽔𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼟𖽙𖾏 𖼎𖽺𖾏 𖼡𖽘𖾐 𖼖𖽺𖾑？ 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼷𖽷𖾐， 𖼮𖽔 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼗𖽔𖾏 𖼋𖽠𖾏 𖼻𖽻𖾑，𖼯𖽔𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽪𖾏． 14 𖼅𖽡 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼎𖽻𖾐 𖼮𖽡𖾏． 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 ． 15 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐， 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼡𖽘𖾐 𖼊𖽱𖾏， 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼷𖽜𖾏，𖼟𖽙𖾏 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏． 16 𖼅𖽡 𖼎𖽻𖾐 𖼮𖽡𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽔𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼅𖽡 𖼻𖽻𖾏 𖼑𖽪𖾏， 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏．”
𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
17 “ 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼞𖽪𖾑 𖼗𖽙𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾑，𖼞𖽪𖾑 𖼗𖽙𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐，𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑． 18 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼸𖽙 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏， 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑． 19 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼋𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼡𖽔 𖼋𖽱𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐， 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖽃𖽡𖾑． 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽙𖾏 𖽃𖽔𖾏， 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽙𖾏 𖽃𖽔𖾏， 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐． 20 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼅𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖽐𖼎𖽙 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙．”
𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐
21 “ 𖼅𖽡 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼙𖽔 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼡𖽑𖽔𖾐 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：‘𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 ． ’ 22 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼡𖽔 𖼋𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 ， 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏． 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼐𖽑𖽡𖾏 ：“ 𖼖𖽔𖾑 𖼮𖽢𖾑 ” 𖽃𖽡𖾑（“ 𖼖𖽔𖾑 𖼮𖽢𖾑 ” 𖼟𖽪𖾏 𖼽𖽨𖼊𖽑𖽹𖾏𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼖𖽑𖽱𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏）， 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼑𖽪𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏． 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼐𖽑𖽡𖾏： “ 𖼄𖽙 𖼖𖽡𖾏 ” 𖽃𖽡𖾑（ “𖼄𖽙 𖼖𖽡𖾏” 𖼟𖽪𖾏 𖼽𖽨𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼷𖽹𖾐 𖼏𖽔）， 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼚𖽔𖾏 𖽐𖼷𖽷𖾏 𖼞𖽡 𖼑𖽡𖽻 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏． 23 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ， 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼟𖽡 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘， 𖼎𖽺𖾏 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏， 24 𖼟𖽡 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼟𖽡 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼦𖽡 𖼯𖽡𖾏， 𖼎𖽺𖾏 𖼺𖽪𖾐 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏， 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼋𖽔𖾏 𖼖𖽑𖽘𖾑 𖼟𖽡 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑． 25 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽻𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼺𖽷𖾐， 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏 ， 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼟𖽡 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼟𖽡 𖼚𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏， 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼟𖽡 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑，𖼟𖽡 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏． 26 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡， 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖼁𖽻𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏，𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏．”
𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏
27 “ 𖼅𖽡 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏：‘ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 ．’ 28 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼁𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏， 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼋𖽘𖾏 ． 29 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼖𖽡 𖼺𖽷𖾏 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼋𖽘𖾏， 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽳𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼀𖽑𖽔𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏，𖼟𖽪𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽔𖾏 𖼚𖽔𖾏 𖽐𖼷𖽷𖾏 𖼞𖽡 𖼑𖽡𖽻． 30 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼋𖽘𖾏， 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽷𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼀𖽑𖽔𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏，𖼟𖽪𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽔𖾏 𖼚𖽔𖾏 𖽐𖼷𖽷𖾏 𖼞𖽡 𖼑𖽡𖽻．”
𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼐𖽤𖾐
（𖼊𖽑𖽹𖾏 19：9；𖼞𖽑𖽙𖾐 10：11-12；𖼖𖽪𖾏 16：18）
31 “ 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏：‘ 𖼡𖽔 𖼋𖽱𖾏 𖼮𖽔𖾑 𖽐𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 ，𖼽𖽔𖾑 𖼄𖽔𖾐 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼐𖽤𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏．’ 32 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼐𖽤𖾐，𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽜 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 ， 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 ， 𖼡𖽔 𖼖𖽡𖾏 𖼋𖽱𖾏 𖼎𖽺𖾏 𖼷𖽙𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼈𖽻𖾑 𖽐𖼎𖽻𖾏 𖼺𖽘𖾏， 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 ．”
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵
33 “ 𖼅𖽡 𖼎𖽺𖾏 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼡𖽑𖽔𖾐 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：‘ 𖼗𖽪𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏，𖼅𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 𖼊𖽘𖾏，𖼐𖽡𖽙𖾐 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏．’ 34 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼏𖽵 𖼗𖽪𖾏 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐． 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼏𖽵 𖼗𖽪𖾏 ，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑． 35 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼏𖽵 𖼗𖽪𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼋𖽙𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐． 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼏𖽵 𖼗𖽪𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖼙𖽻． 36 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽵 𖼗𖽪𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖽃𖽡𖾐 𖼷𖽙𖾏 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼚𖽪𖾐 𖼺𖽘𖾏． 37 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏： 𖼾𖽙 𖼟𖽪𖾏 𖼾𖽙， 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼐𖽑𖽡𖾏，𖼟𖽪𖾏 𖼋𖽔𖾏 𖼉𖽺𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏．”
𖼽𖽔𖾑 𖽐𖼎𖽙 𖼗𖽔 𖼻𖽻𖾏
（𖼖𖽪𖾏 6：29-30）
38 “ 𖼅𖽡 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏：‘ 𖽃𖽔𖾑 𖼅𖽔 𖼁𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔， 𖼐𖽡𖽝𖾑𖽑 𖼁𖽻𖾏 𖼐𖽡𖽝𖾑𖽑．’ 39 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐．𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼟𖽡 𖼺𖽘𖾑 𖼚𖽑𖽪𖾏 𖼖𖽡 𖼺𖽷𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼺𖽘𖾑 𖼖𖽡 𖼇𖽳𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏． 40 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼟𖽡，𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙，𖼯𖽔𖾏 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼊𖽑𖽡𖽝𖾐． 41 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼊𖽔𖾑 𖼟𖽡 𖼅𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽢𖾏 𖼞𖽡𖾑， 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼅𖽻𖾏 𖽃𖽔𖾐 𖽐𖼮𖽢𖾏． 42 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼊𖽜 ， 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏， 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼡𖽜𖾑 𖼷𖽹𖾏 ， 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏．”
𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑
（𖼖𖽪𖾏 6：27-28，32-36）
43 “ 𖼅𖽡 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏：‘ 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙，𖼽𖽔𖾑 𖼳𖽪𖾐 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑．’ 44 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑， 𖼽𖽔𖾑 𖼘𖽑𖽪𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡𖾏 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏． 45 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽷𖾑， 𖼑𖽘 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳， 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳． 46 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼮𖽑𖽠𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼅𖽡 𖼊𖽔𖾐， 𖼅𖽡 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑？𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼘𖽪？ 47 𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼑𖽪 𖼅𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐， 𖼅𖽡 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼘𖽪？𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼘𖽪𖾏？ 48 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐．”
