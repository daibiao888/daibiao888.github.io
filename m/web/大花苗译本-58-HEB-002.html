𖼽𖽔𖾑 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼐𖽑𖽙𖾑
2
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼀𖽡𖾐 𖼽𖽔𖾑 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼙𖽔 𖼐𖽑𖽙𖾑，𖽐𖼮𖽑𖽹𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖽐𖼚𖽱𖾏 𖼸𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼎𖽺𖾏 𖼛𖽙𖾏 𖼙𖽠𖾏． 2 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝，𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏． 3 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼑𖽔𖾏 𖼳𖽷𖾐 𖼺𖽘𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼋𖽻𖾏 𖼖𖽺𖾑？ 𖼋𖽡𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼐𖽡𖾑 𖼺𖽪𖾐 𖼁𖽱 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼡𖽑𖽔𖾐，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼎𖽻𖾏 𖼀𖽡𖾐． 4 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏，𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑、𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏，𖼊𖽑𖽡𖽝𖾐 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼄𖽔𖾐 𖼎𖽻𖾏 ，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏．
5 “ 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏” 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼯𖽵𖾏． 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼙𖽔 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼐𖽑𖽡𖾏：
“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼟𖽡 𖼎𖽺𖾏 𖼞𖽻𖾏 𖼮𖽳𖾐 𖼐𖽱𖾏？
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼟𖽡 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼐𖽱𖾏？
7 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏，
𖼟𖽡 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑，
𖼟𖽡 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏．
8 𖼟𖽙𖾏 𖼯𖽢 𖼋𖽡𖾏 𖼟𖽻𖾏 𖼮𖽘𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼊𖽠𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐．”
𖼟𖽙𖾏 𖼯𖽢 𖼋𖽡𖾏 𖼟𖽻𖾏 𖼮𖽘𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼯𖽵𖾏， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼯𖽵𖾏． 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼁𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼯𖽵𖾏 𖼺𖽷𖾐． 9 𖼀𖽡𖾐 𖼮𖽔 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼳𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖽃𖽡𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏， 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽳𖾏，𖼑𖽜 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖽝𖾏 𖼋𖽔 𖽐𖼷𖽑𖽡𖽝𖾐．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼚𖽑𖽡𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏
10 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼯𖽵𖾏、 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑， 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼚𖽙， 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼞𖽡𖾑 𖼯𖽡𖽙𖾏． 11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽡𖾑， 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼚𖽑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐． 𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑， 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏， 12 𖼐𖽑𖽡𖾏：
“ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏，
𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡．”
13 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
“ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽜 𖼻𖽡𖽝𖾏 𖼐𖽱𖾏．”
𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
“𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑．”
