6 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼷𖽹𖾏 𖼊𖽻𖾏 𖼟𖽡 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏,

7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑,

8 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 ,

9 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏！
𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼯𖽻 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏！
10 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑,

12 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼡𖽔𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐！

14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽡,

20 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼟𖽱𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐,
