5 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼉𖽺𖾏 𖼄𖽔𖾐 𖼮𖽑𖽡 𖼊𖽪𖽝𖾏 ， 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼄𖽔𖾐 𖼮𖽑𖽡 𖼊𖽪𖽝𖾏 𖼺𖽘𖾏，𖼯𖽡𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼅𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑． 6 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼷𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼞𖽪𖾑 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏，𖼅𖽡 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼙𖽠𖾏． 7 𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼐𖽔𖾑 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼅𖽡 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼊𖽔𖾐，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼯𖽪𖾏 𖼞𖽪𖾑，𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼷𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏． 8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽪 𖼺𖽙𖾐 ，𖼋𖽙𖾏 𖼸𖽙 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐． 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼈𖽡𖽝𖾏．
