23
𖼀𖽡𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏:
𖼽𖽦𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐,
𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽜𖾑 𖼛𖽙𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽙𖾏,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:
2 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
𖼐𖽱𖾏 𖼗𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽙.
3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏, 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾑,
𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽔𖾑 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏，
4 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼑𖽡𖾏 𖼮𖽡𖾏,
𖼺𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐,
𖼺𖽱𖾏 𖼑𖽘 𖼗𖽙𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼑𖽡𖾏 𖼮𖽑𖽘𖾑 𖽐𖼊𖽪𖾏,
𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼘𖽑𖽘𖾑 𖼋𖽠𖾏 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼅𖽔𖾏.
5 𖼞𖽪𖾑 𖼾𖽡𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑,
𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏．
𖼊𖽪𖾑 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼐𖽡𖾑 , 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽐𖼊𖽱𖾏 𖼡𖽑𖽙𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏,
𖼋𖽡𖽝𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼖𖽺𖾑?
6 𖼈𖽡𖽝𖾏 𖼡𖽔𖾑 𖼯𖽱𖾏 𖼸𖽜 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏.
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼖𖽑𖽻𖾏 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼮𖽤𖾏 𖼄𖽑𖽪𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏.”
𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼅𖽡
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 11：10 - 47）
