𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 𖼊𖽻𖾏 , 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼳𖽠𖾑 𖼁𖽱𖾏 , 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖽃𖽔𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 , 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖽐𖼮𖽡𖽻𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏； 2 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼸𖽙 𖼖𖽪𖾏 𖼺𖽷𖾑 ,𖼎𖽺𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼮𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 , 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽪； 3 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼖𖽡𖾏 𖼊𖽑𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼀𖽳 · 𖼦𖽙 𖼖𖽪𖽝 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼋𖽻 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 , 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼺𖽳 , 𖼟𖽪𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 .
4 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 、𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 .
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽻𖾏
5 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼞𖽡𖾏 , 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 , 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 : 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 , 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽪 · 𖼽𖽔𖾏 𖼊𖽔 𖼸𖽙 𖼀𖽳 · 𖼦𖽙 𖼖𖽪𖽝 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼸𖽷, 6 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖽐𖼮𖽡𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽡 𖼄𖽡𖾐 𖼊𖽑𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 , 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼮𖽑𖽙𖾑 𖼸𖽙 𖼊𖽑𖽔𖾑 𖼐𖽔 · 𖼳𖽷𖾏 𖼘𖽙 , 𖼎𖽺𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽔𖾐 𖼐𖽙 𖼦𖽔𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐； 7 𖼎𖽑𖽻𖾏 𖼽𖽔𖾐 𖼐𖽙 𖼦𖽔𖾐 𖽐𖼡𖽜𖾏 𖼸𖽙 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽪 , 𖼎𖽺𖾏 𖼸𖽙 𖼐𖽔 𖼖𖽔𖾑 , 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 , 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 . 8 𖼎𖽑𖽻𖾏 𖼊𖽑𖽔𖾑 𖼀𖽑𖽪𖾐 𖼽𖽔𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 , 𖼸𖽙 𖼞𖽑𖽔𖾑 𖼐𖽔 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 . 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 . 9 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼙𖽻 𖼊𖽑𖽡𖽝𖾐 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼙𖽻 , 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼯𖽡𖽙𖾏 . 10 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼺𖽳 , 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼑𖽻𖾏 𖽃𖽦𖾐 , 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 .
