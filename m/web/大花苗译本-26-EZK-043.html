6 𖼞𖽪𖾑 𖼐𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡𖾏 𖼊𖽘𖾑. 7 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽘𖾐 𖼊𖽠𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼷𖽠𖾐, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏. 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼎𖽑𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑. 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑 𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽔𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑 𖼀𖽔𖾐 𖽃𖽔 𖼛𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽔𖾑 𖼡𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾐 𖼮𖽑𖽤 𖼊𖽔𖾐. 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼖𖽻𖾐 𖼊𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 9 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
10 𖼟𖽡𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖽐𖼊𖽡𖾑 𖼖𖽪𖾑. 11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏, 𖼼𖽘𖾏 𖼼𖽔𖾏, 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏、 𖼼𖽘𖾏、 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏. 12 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼚𖽜𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑. 𖼑𖽔𖾏 𖼘𖽪！ 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑.”
