𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽑𖽹𖾏 𖼮𖽑𖽡𖽠𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑, 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼊𖽡𖾐 𖼺𖽷𖾑 𖽐𖼎𖽙 𖼳𖽪𖾑 𖽃𖽳𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼮𖽡𖾑 𖼖𖽔𖾑 𖽃𖽳𖾐. 2 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽡𖾑 𖼖𖽔𖾑 𖽃𖽳𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑, 𖼮𖽡𖾑 𖼖𖽔𖾑 𖽃𖽳𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼨𖽑𖽳𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏. 3 𖼈𖽡𖽝𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼎𖽑𖽦𖾑 𖼋𖽔 𖼖𖽔𖾏! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽙𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼈𖽻𖾑 𖼋𖽘.” 4 𖼈𖽡𖽝𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼺𖽷𖾐, 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼘𖽪！ 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽡 𖼊𖽪𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏? 5 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: ‘𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑 ’ 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼟𖽪𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏: ‘𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽻𖾏 ’. 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼯𖽡𖽙𖾏.” 6 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖽐𖼚𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼟𖽡 𖼋𖽘𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏. 7 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼄𖽔𖾐 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼎𖽙𖾑 𖼁𖽻𖾏 𖼐𖽱𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐, 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏. 𖼟𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼁𖽻𖾏 𖼐𖽱𖾏, 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐, 𖼟𖽡 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽙𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 .”
8 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽜𖾑 𖼋𖽔𖾏, 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝. 9 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼨𖽑𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼋𖽔𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼘𖽪? 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 , 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼚𖽙, 𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏.” 10 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼘𖽪?” 11 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽑𖽡𖽝𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼐𖽤𖾐 𖼊𖽔𖾏 𖼞𖽪𖾑. 12 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼋𖽡𖽝 , 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼑𖽡𖽝𖾏, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼐𖽤𖾐. 13 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼸𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏, 𖼟𖽡 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: ‘𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽹𖾐 𖼅𖽻𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏. ’ ” 14 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼄𖽔𖾐 𖼑𖽨𖾏、𖼮𖽑𖽷𖾏、𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏. 15 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼞𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐.” 16 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏, 𖽃𖽔𖾏 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼙𖽱 𖽐𖼀𖽙 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 ( “𖼊𖽑𖽱𖾏 𖼸𖽘𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽃𖽔𖾑 𖼅𖽔”), 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼋𖽘𖾏.” 17 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏. 18 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽜 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏, 𖼟𖽙𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼑𖽡𖾏 𖼾𖽡𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼘𖽘𖾑𖽑 𖼊𖽪𖾐 𖼇𖽳𖾏.
