𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼮𖽵𖾑 𖼎𖽺𖾏 𖼺𖽹𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼷𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼅𖽻𖾏，𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏． 2 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼡𖽻𖾐 𖼖𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼯𖽠 𖼊𖽔𖾐 𖼋𖽡𖽝．𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ， 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑，𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼅𖽻𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏． 3 𖼅𖽡 𖼅𖽻𖾏 𖼦𖽡𖾑！𖼞𖽪𖾑 𖼷𖽷𖾐 𖼅𖽡 𖼋𖽠𖾏 𖼅𖽻𖾏，𖼺𖽱𖾏 𖼺𖽱𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼀𖽙𖾏 𖼄𖽑𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐． 4 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼐𖽑𖽘𖾐 𖼑𖽡𖽝𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼞𖽑𖽻𖾏，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪． 5 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼖𖽡 𖼋𖽱𖾏 𖽐𖼞𖽔𖾏，𖼺𖽪𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏：‘𖼊𖽻𖾏 𖼅𖽪𖾏 𖼾𖽡𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼋𖽪！’ 6 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪， 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼊𖽜，𖼎𖽺𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼾𖽡𖾏 𖽃𖽡𖾑 ，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏， 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼅𖽡． 7 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼾𖽡𖾏 𖽃𖽡𖾑，𖼑𖽻𖾏 𖼦𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼊𖽻𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼾𖽡𖾏 𖼐𖽡𖾑 𖼮𖽑𖽹𖾏 𖼚𖽔𖾏 𖼾𖽡𖾏 𖽃𖽡𖾑． 8 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻 𖼡𖽔𖾏 𖼳𖽷𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼅𖽡，𖼊𖽜𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼑𖽻𖾏， 𖼅𖽡 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 ． 9 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼊𖽪𖾐 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：‘𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖽐𖼮𖽡𖽻𖾏 𖼉𖽺𖾏 𖼅𖽡 𖼋𖽘𖾏．’ 10 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻 𖼡𖽔𖾏 𖼳𖽷𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼅𖽡，𖼅𖽡 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏， 11 𖼐𖽑𖽡𖾏：‘𖼅𖽡 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼀𖽡𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏，𖼀𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼳𖽙𖾏 𖼺𖽘𖾏．𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏 ，𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏！’ 12 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼸𖽙 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑， 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖼊𖽻𖾏， 𖼖𖽡 𖼟𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 ．
𖼷𖽷𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻
（𖼊𖽑𖽹𖾏 11：20-24）
13 “𖼞𖽙𖾑 𖼖𖽔𖾑 𖼳𖽱𖾏 𖼘𖽪！𖼟𖽡 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 ， 𖼀𖽳 𖼺𖽹 𖼊𖽔𖾏 𖼘𖽪！ 𖼟𖽡 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏，𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼙𖽻 𖽃𖽔𖾏，𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼙𖽻，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽙𖾑 𖽐𖼀𖽔𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖽐𖼀𖽑𖽻𖾏 𖼮𖽑𖽻𖾑 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼋𖽘𖾏． 14 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽃𖽡𖾑， 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼊𖽻𖾏 ，𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼺𖽷𖾐． 15 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾 𖼙𖽻 𖼘𖽪！
𖼟𖽡 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼸𖽙 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼘𖽪𖾏？
𖼟𖽡 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 ． ”
16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼅𖽡 𖼟𖽪𖾏 𖼑𖽻 𖼞𖽪𖾑，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼗𖽔𖾏 𖼅𖽡 𖼺𖽘𖾏 𖼟𖽪𖾏 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏， 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏，𖼟𖽪𖾏 𖼗𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼋𖽔𖾏 𖽃𖽡𖾑．”
𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼺𖽷𖾑 𖼗𖽙
17 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑，𖼟𖽡𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼚𖽙𖾏 𖽃𖽔𖾏 𖽐𖼷𖽑𖽷𖾏 𖼺𖽷𖾐 𖼗𖽙， 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼀𖽡𖾐 𖼻𖽡𖽝𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏，𖼊𖽪𖾑 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼖𖽔𖾏 𖼑𖽻 𖼀𖽡𖾐 𖼯𖽵𖾏．” 18 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼺𖽔𖾐 𖼊𖽔 ，𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽻𖾐 𖼗𖽙，𖼺𖽱𖾏 𖼺𖽙𖾐 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼯𖽦𖾏． 19 𖼞𖽪𖾑 𖼙𖽔 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼎𖽻𖾏 𖼅𖽡，𖼟𖽙𖾏 𖼅𖽡 𖼸𖽠𖾏 𖼯𖽪𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼞𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼚𖽡𖽝𖾏，𖼎𖽺𖾏 𖼳𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏．𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼸𖽠𖾏 𖼟𖽡𖾏 𖼅𖽡． 20 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼖𖽡𖽾 𖼑𖽻 𖼅𖽡 𖼗𖽪𖾏 𖼎𖽺𖾏 𖼙𖽔，𖼽𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼷𖽷𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼮𖽡𖽝𖾏 𖼮𖽑𖽔𖾏 𖼙𖽔．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽔𖾏 𖼙𖽔 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏
（𖼊𖽑𖽹𖾏 11：25-27；13：16-17）
21 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 ， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽑𖽡𖾏：“ 𖼉𖽺𖾏 𖼘𖽪！ 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼮𖽵𖾑 ！𖼞𖽪𖾑 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼟𖽡，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼯𖽡𖾏 𖼟𖽡 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 ，𖼈𖽡𖽝𖾏 𖼎𖽺𖾏 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼎𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏． 𖼉𖽺𖾏 𖼘𖽪！ 𖼾𖽙 𖼯𖽡𖽙𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 22 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼯𖽵𖾏，𖼋𖽠𖾏 𖼉𖽺𖾏 𖼺𖽘𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏，𖼋𖽠𖾏 𖼊𖽪𖾐 𖼺𖽘𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾐 𖼡𖽑𖽳𖾏 𖼺𖽘𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏．”
23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼗𖽙𖾏 𖼐𖽑𖽡𖾏 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼡𖽑𖽳𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏：“𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼙𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏 ！ 24 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖽃𖽔 𖼊𖽑𖽻𖾏 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 ，𖽐𖼮𖽑𖽠𖾏 𖼁𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽙𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽙𖾏 ， 𖽐𖼮𖽑𖽠𖾏 𖼐𖽑𖽙𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽑𖽙𖾑，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑 ．”
𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏
25 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑，𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 ，𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 ， 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ，𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏？” 26 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼺𖽻𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏？𖼟𖽡 𖼟𖽡𖽠 𖼖𖽡𖾏 𖼯𖽤𖾏？” 27 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜， 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜， 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜，𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜，𖽐𖼮𖽑𖽠𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙 𖼺𖽱𖾏 𖼾𖽨𖾏．” 28 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼾𖽙，𖼟𖽡𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏．” 29 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼞𖽡𖾑，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼼𖽷𖾏 𖼼𖽙 𖼘𖽪？”
