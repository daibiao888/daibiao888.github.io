12
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼘𖽑𖽘𖾑 𖼾𖽨 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼊𖽪𖾐 , 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 , 𖼺𖽱𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼡𖽔𖾏 𖼖𖽑𖽡𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 . 3 𖼾𖽡𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 . 4 𖽃𖽔 𖼁𖽙𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 . 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼦𖽡𖾐 𖼀𖽪𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 . 5 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼾𖽨 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 , 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽔𖾐 𖼋𖽪 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 , 𖼺𖽱𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼡𖽔𖾏 𖼖𖽑𖽡𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 ；𖼳𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 .
6 𖼀𖽪𖾑 𖼋𖽘𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼊𖽪𖾐 , 𖼟𖽪𖾏 𖼑𖽜 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖽃𖽡𖾐 𖼋𖽪𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖽐𖼡𖽔𖾐 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽡𖾐 𖼋𖽪 𖽐𖼡𖽔𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 , 𖼯𖽡𖽙𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽔 𖼛𖽻𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 . 7 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼑𖽜 𖼐𖽱𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 , 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽐𖼮𖽑𖽘𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏 . 𖼊𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 . 8 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼖𖽑𖽘𖾑 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼾𖽘𖾏 𖼁𖽜𖾏 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽐𖼡𖽔𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾐 𖼋𖽪 𖼣𖽔𖾑 𖼣𖽔𖾏 𖽐𖼡𖽔𖾐 , 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 . 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏 .”
