𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖽐𖼮𖽔𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 , 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐!” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 .” 2 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼊𖽪𖾐, 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼊𖽪𖾐, 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏, 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼟𖽡 𖽃𖽡𖾑, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼖𖽑𖽘𖾑 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 .” 3 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 , 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽔𖾐, 𖼀𖽑𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏. 4 𖼸𖽙 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼋𖽘 , 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑. 5 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖽐𖼎𖽙 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽙𖾏, 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑 .” 6 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼄𖽔𖾐 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼎𖽷𖾏, 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽦𖾏； 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏. 7 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾑 𖼈𖽹𖾐!” 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 .” 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽔𖾏, 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽐𖼎𖽙 𖼋𖽠𖾏 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼋𖽘 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼘𖽪?” 8 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏.” 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼎𖽺𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏.
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽳𖾏 𖽃𖽡𖾑 𖼋𖽘𖾏, 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘 , 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼋𖽠𖾏 𖼻𖽻𖾏 𖼮𖽡𖽝𖾏, 𖼡𖽑𖽹𖾐 𖼐𖽱𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽔𖾐 , 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼋𖽠𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑. 10 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼗𖽙𖾏, 𖼽𖽔𖾑 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐. 11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐!𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐!” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 .” 12 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏, 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽺𖾏 𖼐𖽱𖾏！ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽘𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼷𖽷𖾐 𖼟𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼊𖽪𖾐 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 .” 13 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾑 𖼙𖽻𖾏 𖼾𖽘𖾏, 𖽃𖽔𖾐 𖼋𖽪 𖼞𖽪𖾐 𖼚𖽑𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼯𖽡 𖽐𖼊𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽡𖽝𖾏； 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼅𖽻𖾏 𖼚𖽠𖾏 𖼊𖽔𖾑 𖼾𖽘𖾏 𖼗𖽙𖾏, 𖼘𖽪𖾑𖽑 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑. 14 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼊𖽡𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼖𖽳 ”( “ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼖𖽳”𖼟𖽪𖾏 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 ” )． 𖼸𖽙 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏.”
