𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐
𖼞𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 “ 𖼮𖽢𖾑 𖼊𖽑𖽳 ” 𖼢𖽜𖾏 𖽐𖼎𖽔𖾏 .
84
𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！
𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏．
2 𖼞𖽪𖾑 𖼖𖽡𖽾 𖼮𖽳𖾐 𖼉𖽳 𖽐𖼮𖽑𖽠𖾏 𖼮𖽑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
 
3 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼏𖽘 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽻 𖼀𖽡𖾏 𖼮𖽑𖽡𖽙𖾑 𖼑𖽜 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼐𖽻 𖼙𖽡𖽝𖾏 𖼞𖽡𖾏 𖼘𖽪𖾏 𖼑𖽜 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼉𖽳𖾏 𖼟𖽪𖾏 𖼁𖽔 𖼣𖽔𖾑 𖼣𖽔𖾏.
4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖽐𖼚𖽙 , 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 . （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼟𖽡 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 𖽃𖽡𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏！
6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔, 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼐𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼷𖽠𖾑𖽑,
𖼎𖽺𖾏 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐 𖼑𖽘 𖼑𖽡𖾏 𖼅𖽳𖾏 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼁𖽘𖾏 𖽃𖽻𖾐.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏, 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
 
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏！
𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻！ （𖼳𖽡𖾏 𖼖𖽔𖾑）
9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼟𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼑𖽡𖾏 𖼚𖽑𖽪𖾏.
 
10 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐, 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽡𖽙𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼐𖽑𖽪𖾐,
𖼯𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼻𖽙𖾑 𖽃𖽔 𖼛𖽻𖾏,
𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼚𖽙.
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐, 𖼟𖽪𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼗𖽙𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖽐𖼚𖽻 𖼮𖽡𖾏．
𖼐𖽱𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
12 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏．
