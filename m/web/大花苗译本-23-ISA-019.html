19
𖼦𖽔𖾑 𖼗𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼑𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐 :
 
𖼑𖽔𖾏 𖼘𖽪! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼇𖽺𖾏,
𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼣𖽹 𖼮𖽡 ,
𖼣𖽹 𖼮𖽡 𖼑𖽡𖾏 𖼁𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏,
𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼾𖽘𖾏 𖼋𖽘 𖽐𖼮𖽢𖾏.
2 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼙𖽳𖾐 𖼷𖽱𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽐𖼊𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏,
𖼼𖽷𖾏 𖼼𖽙 𖽐𖼊𖽻𖾏 𖼼𖽷𖾏 𖼼𖽙 ,
𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖽐𖼊𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼉𖽺𖾏,
𖼖𖽪𖾐 𖼞𖽪𖽙 𖼐𖽡𖾑 𖽐𖼊𖽻𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖽃𖽡𖾑.
3 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽑𖽪𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑．
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼊𖽺𖾐 𖼁𖽠𖾏 𖽐𖼎𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼅𖽻𖾏 𖼗𖽙𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐.
4 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼄𖽔𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽪𖾏 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼽𖽔𖾑 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
5 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘 𖽐𖼮𖽢𖾏,
𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼽𖽔𖾑 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏.
6 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼮𖽵𖾏 𖼗𖽱𖾏,
𖼣𖽹 𖼮𖽡 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼯𖽠 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼺𖽘𖾏．
𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽳𖾏 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏.
7 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽔𖾐 𖼟𖽱𖾐 𖼑𖽡𖾏 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏,
𖼊𖽑𖽡𖽝𖾐 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡．
𖼡𖽻𖾐 𖼖𖽻𖾏 𖼊𖽻𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼙𖽠𖾏, 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏.
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽢𖾐 𖽐𖼀𖽳𖾏 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽐𖼊𖽻𖾏 𖽐𖼀𖽳𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼮𖽑𖽡𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼗𖽔𖾏 𖼸𖽙𖾏 𖼎𖽻𖾏 𖽃𖽻𖾐 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽻 𖼼𖽙 .
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾏 𖼨𖽳𖾑𖽑 𖼅𖽘𖾏 𖼮𖽑𖽔𖾐 𖼺𖽙𖾑
𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽙𖾏 𖽐𖼊𖽻𖾐 𖼚𖽠𖾐 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
10 𖼞𖽪𖽙 𖼊𖽪𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖽐𖼡𖽜𖾏,𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏.
 
11 𖼺𖽙𖾐 𖼣𖽔𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼡𖽔 𖼏𖽔 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼇𖽔 𖼘𖽻𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽡𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔.
𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼘𖽔𖾑𖽑,
𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏.”
12 𖼟𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 𖼏𖽵 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐, 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 𖼋𖽘𖾏 !
13 𖼺𖽙𖾐 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 ,
𖼐𖽙 𖼇𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽻 𖼮𖽑𖽵𖾑．
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼡𖽔𖾏 𖼳𖽡𖾏 𖼞𖽪𖾐 𖽐𖼞𖽔𖾐 𖼊𖽠𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖽃𖽡𖾑,
𖼟𖽙𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼞𖽡𖾑 .
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼟𖽙𖾏 𖼣𖽹 𖼮𖽡 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏,
𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼢𖽻 𖼮𖽡𖽠𖾑 𖽃𖽺𖾐 𖽐𖼊𖽔𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼢𖽻 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼐𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
15 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼡𖽔𖾑 𖽐𖼷𖽘𖾏,
𖽃𖽔𖾑 𖼯𖽡 𖼺𖽱𖾐 𖼺𖽱𖾐 𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐, 𖼼𖽙 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑.

𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼮𖽵𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏
