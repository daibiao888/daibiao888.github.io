𖼖𖽔𖾑 𖼮𖽡𖽝 𖼑𖽔𖾏 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼻𖽷𖾏, 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼄𖽔𖾐 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼦𖽡𖾐 𖼾𖽙 𖼯𖽡𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽔 𖼺𖽘𖾏.” 2 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼮𖽡𖽝, 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼟𖽙𖾏 𖼟𖽡 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽵𖾑 𖼘𖽪?” 3 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼐𖽑𖽡𖾏: “𖼅𖽔𖾏 𖼞𖽪𖾑 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼀𖽑𖽡 𖼖𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑 , 𖼟𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 (“𖼊𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏”).” 4 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼀𖽑𖽡 𖼖𖽔𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖽃𖽔𖾏 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏, 𖼽𖽔𖾐 𖼞𖽙 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏. 5 𖼀𖽑𖽡 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏 𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼳𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐. 6 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼐𖽑𖽡𖾏: “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽑𖽔𖾑 𖼞𖽡𖾑, 𖼎𖽺𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏, 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐.”𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼊𖽗𖾏(“𖼊𖽗𖾏” 𖼟𖽪𖾏 “𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼾𖽙”). 7 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼀𖽑𖽡 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽤𖾑, 𖼎𖽺𖾏 𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼳𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼊𖽑𖽡𖽝𖾐. 8 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼖𖽹𖾐 𖼻𖽷𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐（“𖽃𖽺𖾐 𖼘𖽑𖽙𖾐”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼸𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ”）, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 (“𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏” 𖼟𖽪𖾏 “𖼦𖽡𖾏 𖽐𖼊𖽜𖾏”).
9 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼳𖽡 𖼀𖽑𖽔𖾏 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽔𖾏 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏. 10 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼳𖽡 𖼀𖽑𖽔𖾏 𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼳𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐. 11 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼻𖽻𖾏 𖼞𖽡𖾑 𖼙𖽻 .” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼮𖽢𖾑 𖼊𖽳 (“𖼮𖽢𖾑 𖼊𖽳” 𖼟𖽪𖾏 “𖼻𖽻𖾏 𖼞𖽡𖾑 𖼙𖽻”). 12 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼳𖽡 𖼀𖽑𖽔𖾏 𖼎𖽺𖾏 𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼳𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽃𖽡𖾑. 13 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔 𖼁𖽙𖾏 𖼊𖽪𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼳𖽳 (“𖼽𖽔𖾏 𖼳𖽳”𖼟𖽪𖾏 “𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏”).
14 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼤𖽜𖾏 𖼳𖽻𖾐, 𖼖𖽨 𖼀𖽡𖽝𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼇𖽘𖾐 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼇𖽪𖽾𖾑 𖼮𖽑𖽡 , 𖼯𖽡𖽙𖾏 𖼗𖽙 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏． 𖼖𖽔𖾑 𖼮𖽡𖽝 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙 𖼟𖽙𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼟𖽡 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼇𖽘𖾑 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼇𖽪𖽾𖾑 𖼮𖽑𖽡 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼇𖽺𖾐 𖼞𖽪𖾑.” 15 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽔𖾏 𖼞𖽪𖾑 𖼈𖽻𖾑 𖼊𖽪𖾏 𖼺𖽙𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼑𖽪𖾏 𖼺𖽷𖾐 𖼘𖽪𖾏？ 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼞𖽪𖾑 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼇𖽪𖽾𖾑 𖼮𖽑𖽡 𖼺𖽷𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏 ?” 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼐𖽑𖽡𖾏: “𖼑𖽜 𖼟𖽡 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼇𖽪𖽾𖾑 𖼮𖽑𖽡 , 𖼄𖽑𖽙𖾏 𖼐𖽔𖾑 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼀𖽱𖾏.” 16 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼽𖽔𖾐 𖼞𖽙 𖼎𖽑𖽻𖾏 𖽐𖼎𖽹𖾏 𖼙𖽡𖽝𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼀𖽱𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼇𖽪𖽾𖾑 𖼮𖽑𖽡 𖼅𖽔𖾏 𖼟𖽡 𖼋𖽘𖾏.” 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑 𖼽𖽔𖾐 𖼞𖽙 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏. 17 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 , 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑, 𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼳𖽻𖾏 𖼀𖽜𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽃𖽡𖾑. 18 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼈𖽻𖾑.” 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 (“𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑”𖼟𖽪𖾏 “𖽐𖼡𖽜𖾏 𖼼𖽙”). 19 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽤𖾑, 𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼳𖽻𖾏 𖼚𖽻𖾏 𖼗𖽜𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽃𖽡𖾑. 20 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼈𖽻𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼞𖽪𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼚𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐.” 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 (“𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝” 𖼟𖽪𖾏 “𖼐𖽑𖽨𖾑 𖼙𖽪𖾏”). 21 𖽃𖽔𖾑 𖼡𖽜𖾑 𖼗𖽙 𖼎𖽺𖾏 𖼳𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽑𖽹𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼊𖽡𖾐 𖼐𖽔.
22 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽻𖾏 𖼮𖽳𖾐 𖼖𖽔𖾑 𖼮𖽡𖽝 , 𖼑𖽻 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼋𖽘𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏. 23 𖼖𖽔𖾑 𖼮𖽡𖽝 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖼐𖽑𖽡𖾏: “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.” 24 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 (“𖼽𖽙 𖼺𖽳”𖼟𖽪𖾏 “𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐”), 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐.”
𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼊𖽡𖾏 𖼏𖽵 𖽐𖼡𖽜𖾏 𖼼𖽙
25 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼳𖽻𖾏 𖼽𖽙 𖼺𖽳 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽡𖾐 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 26 𖼮𖽑𖽙 𖼟𖽡 𖼄𖽔𖾐 𖼐𖽤𖾐 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏： 𖼞𖽪𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼟𖽡 𖼖𖽔𖾏 𖼀𖽻𖾐 𖼯𖽡𖽙.” 27 𖼖𖽔𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏, 𖼮𖽑𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼮𖽔𖾑 𖼺𖽙𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼐𖽔𖾑.” 28 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼟𖽡 𖽐𖼡𖽜𖾏 𖼼𖽙, 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡.” 29 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼟𖽡 𖼯𖽦𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼟𖽡 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼯𖽡𖽙. 30 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽔𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼅𖽔𖾏 𖽃𖽺𖾐 𖼯𖽠, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼳𖽤𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼊𖽠𖾏 𖼐𖽜𖾑 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 . 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼡𖽔𖾏 𖼊𖽘𖾑 𖼋𖽱 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼾𖽨𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼘𖽪?” 31 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼘𖽪? 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼄𖽔𖾐 𖼋𖽪 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼈𖽡𖽝𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼯𖽪𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼾𖽨 𖼟𖽡 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐. 32 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼟𖽡 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐, 𖼄𖽔𖾐 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼸𖽺𖾏 、𖼡𖽑𖽙𖾑 𖽐𖼚𖽻 𖽐𖼎𖽙 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼚𖽪𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼮𖽑𖽷𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼚𖽻、𖼡𖽑𖽙𖾑 𖼸𖽺𖾏 𖼺𖽹𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼞𖽪𖾑 𖽐𖼡𖽜𖾏 𖼼𖽙. 33 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼟𖽡 𖼋𖽔𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖽐𖼡𖽜𖾏 𖼼𖽙, 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼋𖽡𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽺𖾏 𖼦𖽡𖾐 𖽐𖼚𖽻 𖽃𖽡𖾑, 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼚𖽪𖾐 𖽃𖽡𖾑, 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽡𖽝 𖼗𖽙𖾏. 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼸𖽠𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼗𖽙𖾏.” 34 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼻𖽻𖾏 𖼯𖽡𖽙𖾏, 𖼞𖽪𖾑 𖼯𖽪𖾏 𖼸𖽙𖾏 𖼟𖽡 𖼗𖽪𖾏 𖽃𖽔𖾏.” 35 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼚𖽻 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼞𖽘𖾐, 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻、𖼡𖽑𖽙𖾑 𖼸𖽺𖾏 𖽃𖽡𖾑、𖽃𖽔𖾑 𖼑𖽡𖽝 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻、𖼟𖽪𖾏 𖼸𖽺𖾏、𖼟𖽪𖾏 𖼚𖽻𖾐 𖼚𖽠𖾐、𖼊𖽑𖽡𖽝𖾐 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼚𖽪𖾐, 𖼞𖽑𖽔𖾏 𖼺𖽹𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼅𖽡 𖼾𖽨 . 36 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼾𖽨 𖼡𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼚𖽡𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼞𖽡𖾑 , 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽨 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼳𖽙𖾏.
