𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼎𖽑𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼋𖽔𖾏 𖼸𖽙 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏． 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾑 𖽐𖼊𖽪, 𖼽𖽔𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏； 2 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘, 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾑 𖼏𖽘 𖼘𖽪! 𖽃𖽔𖾑 𖼏𖽘 𖼘𖽪! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼾𖽡𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖽃𖽡𖾑, 𖼊𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾏 𖼇𖽔𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼷𖽑𖽘𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾏 𖼇𖽔𖾑.” 3 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼚𖽔𖾐 𖼼𖽘𖾏, 𖼐𖽑𖽡𖾏: “𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏, 𖽃𖽔𖾑 𖼏𖽘 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼑𖽡𖾏 𖼮𖽑𖽻𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽢𖾑𖽑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏.” 4 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼎𖽻𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖼳𖽤𖾐 𖼋𖽡𖾏 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼡𖽑𖽳𖾏 𖼐𖽑𖽡𖾏: “𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏!” 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼞𖽻𖾏 𖼺𖽷𖾐 𖼗𖽙. 5 𖽃𖽔𖾑 𖼏𖽘 𖼖𖽔𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖽃𖽔𖾑 𖼏𖽘 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼑𖽡𖾏 𖼮𖽑𖽻𖾑 𖽐𖼮𖽢𖾑𖽑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼸𖽪𖾏 𖼋𖽘𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑. 6 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼅𖽳𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼻𖽻𖾏 𖼙𖽠𖾏 ．” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐. 7 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏, 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡.” 8 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼮𖽔𖾑 𖼄𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼁𖽺𖾏 𖼅𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼦𖽻𖾏 𖽃𖽻𖾐. 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼦𖽻𖾏 𖽃𖽻𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽷𖾐 𖼗𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼅𖽻𖾏.” 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼞𖽡𖾑 𖼺𖽷𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼺𖽷𖾐 𖼙𖽠𖾏.
