4 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼦𖽔𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼳𖽻𖾏 𖼙𖽠𖾏,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼷𖽷 𖼟𖽻𖾏 (“𖼷𖽷 𖼟𖽻𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼳𖽻𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐”) 𖼑𖽻𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
 
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼚𖽑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼷𖽷𖾑 𖼷𖽷 𖽃𖽡𖾐 𖼼𖽘𖾐.
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐,

6 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏 𖼑𖽡𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑、 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏、 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼟𖽳,

17 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼎𖽻 ,

18 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽱𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖽃𖽡𖾑,
𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽔𖾑 𖼾𖽱 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼙𖽪𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽔𖾑 𖼾𖽱 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?”
19 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳,
𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏,
𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼎𖽑𖽵𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽻 𖼊𖽻𖾏.
20 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽡𖾑！
𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 ,
𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽑𖽹𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼊𖽪𖾑 𖼘𖽑𖽻𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼚𖽙𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖽃𖽡𖾐 𖼸𖽙𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏.
21 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏.
𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼚𖽔𖾑 𖼯𖽵𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼟𖽡𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼅𖽔𖾏 𖼼𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼉𖽺𖾏.
22 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼀𖽡𖾐 𖼯𖽡𖽙 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽡𖾑,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼀𖽡𖾐 .
 
23 𖼟𖽡 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖼋𖽻 𖼙𖽠𖾏,
𖼦𖽡𖾐 𖼸𖽠𖾏 𖼡𖽑𖽹𖾐 𖼞𖽑𖽙𖾑 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽘𖾏 𖼮𖽡𖽝𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐．
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼊𖽻𖾏 𖼇𖽺𖾐 𖼺𖽘𖾏,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼖𖽔𖾏 𖼸𖽠𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 .
24 𖽐𖼚𖽙 𖼙𖽻 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼄𖽙𖾐 𖼋𖽘 .”
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼋𖽘 .
