3
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏、𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼊𖽱𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼯𖽜𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 、𖽃𖽔𖾏 𖼀𖽡𖾐 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽃𖽡𖾑． 2 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼸𖽙 𖼡𖽘𖾐 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼸𖽙 𖼡𖽘𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐． 3 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 ， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏． 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽐𖼞𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽷𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 5 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑， 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼸𖽙 𖼡𖽘𖾐， 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏． 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖼊𖽪𖾐，𖼟𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼾𖽡𖾏． 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼺𖽡𖽝𖾐 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼷𖽻𖾑 𖼙𖽔 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾐 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐．
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼐𖽑𖽡𖾏：
“𖼄𖽑𖽙𖾏 𖼑𖽔𖾏 ，𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼗𖽪𖾏，
8 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 ，
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼊𖽻𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐．
9 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑，𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼞𖽪𖾑，
𖼑𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽐𖼊𖽪𖾐 𖽐𖼎𖽹𖾏．
10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼷𖽑𖽷𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑，
𖼐𖽑𖽡𖾏： 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔，
𖼎𖽺𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑．
11 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼏𖽵 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏：
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏．”
12 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏， 𖽐𖼮𖽑𖽹𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜，𖼇𖽳𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼺𖽘𖾏． 13 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽪𖾑 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 “ 𖼄𖽑𖽙𖾏 𖼑𖽔𖾏 ” 𖼺𖽷𖾐， 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼞𖽑𖽻𖾑 𖼨𖽑𖽳𖾏， 𖽐𖼮𖽑𖽹𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼾𖽔𖾏 𖼮𖽑𖽵𖾑， 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼼𖽔𖾏． 14 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽑𖽔𖾐 𖼁𖽱 𖼅𖽔𖾏， 𖼻𖽙𖾑 𖼏𖽔𖾏 𖽐𖼎𖽜𖾏 𖼟𖽪𖾏 𖼸𖽙 𖼡𖽘𖾐， 𖼯𖽡𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼊𖽪 𖽐𖼚𖽙 𖼅𖽔𖾏 𖽐𖼀𖽠 𖼋𖽘𖾏． 15 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼐𖽑𖽡𖾏：
“ 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏， 𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼗𖽪𖾏，
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏．”
16 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾑 𖼐𖽱𖾏 𖼗𖽪𖾏， 𖼎𖽺𖾏 𖽐𖼷𖽜𖾑 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼘𖽪？ 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼘𖽪？ 17 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼘𖽪？ 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽔 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑 𖼘𖽪？ 18 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼡𖽔 𖼋𖽱𖾏 𖼏𖽵 𖼗𖽪𖾏， 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼘𖽪？ 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼸𖽙𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼘𖽪？ 19 𖼑𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼾𖽱 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼐𖽔𖾑．
