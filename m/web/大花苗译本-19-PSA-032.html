𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽵𖾑 𖼯𖽔𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏.
32
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼾𖽔𖾏 𖼊𖽻𖾏 𖽐𖼀𖽙 𖼮𖽡𖽝𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏！
2 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
 
3 𖼞𖽪𖾑 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼾𖽔𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼺𖽘𖾏.
4 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼞𖽪𖾑 𖽃𖽻𖾐 𖽐𖼮𖽑𖽘𖾏 𖼋𖽘𖾏 𖼋𖽘𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
5 𖼞𖽪𖾑 𖼯𖽜𖾏 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡,
𖼦𖽡𖾐 𖼈𖽺𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏．
𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽜𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.”
𖼟𖽡 𖼎𖽺𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽪𖾑 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼸𖽙 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑.
7 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑,
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖼎𖽑𖽻𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
8 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡, 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼽𖽔𖾑 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑, 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼟𖽡.
9 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼑𖽜𖾏 𖽐𖼎𖽙 𖼑𖽜𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖼺𖽻𖾑 𖼺𖽻𖾑 𖼮𖽡𖽝𖾏,
𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼯𖽵𖾏 𖼋𖽻 𖼐𖽱𖾏.
 
10 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽡𖽻𖾐 𖼊𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖼯𖽔 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼐𖽱𖾏.
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼘𖽪! 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼘𖽪! 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽺𖾐 𖼙𖽔.
