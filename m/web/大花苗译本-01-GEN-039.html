𖼽𖽙 𖼺𖽳 𖼊𖽻𖾏 𖼨𖽑𖽳𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡. 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽙𖾑 𖼊𖽑𖽡 𖼇𖽔 𖼐𖽡𖽙𖾐 𖼇𖽔 𖼘𖽻𖾐 𖽐𖼚𖽙 𖼟𖽻, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 — 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑, 𖼎𖽑𖽻𖾏 𖼸𖽻𖾏 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼽𖽙 𖼺𖽳 𖼮𖽡𖽝𖾏. 2 𖼽𖽙 𖼺𖽳 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽐𖼞𖽔𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐, 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼑𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼎𖽳𖾏 𖼎𖽪𖾑 . 3 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐, 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼎𖽳𖾏 𖼎𖽪𖾑 . 4 𖼽𖽙 𖼺𖽳 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏, 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏, 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏. 5 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼄𖽔𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼽𖽙 𖼺𖽳 𖼞𖽳𖾑 𖼯𖽵𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼑𖽜 𖼽𖽙 𖼺𖽳, 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼾𖽡𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑； 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏. 6 𖼀𖽙𖾑 𖼊𖽑𖽡 𖼇𖽔 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼞𖽑𖽔𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼽𖽙 𖼺𖽳 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼉𖽔𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼑𖽻𖾏, 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑. 𖼽𖽙 𖼺𖽳 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼎𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝.
𖼀𖽙𖾑 𖼊𖽑𖽡 𖼇𖽔 𖼐𖽤𖾐 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼽𖽙 𖼺𖽳
7 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼀𖽙𖾑 𖼊𖽑𖽡 𖼇𖽔 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼺𖽘𖾏 𖽐𖼮𖽑𖽠𖾏 𖼎𖽻𖾏 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼀𖽱𖾏 .” 8 𖼽𖽙 𖼺𖽳 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼐𖽤𖾐 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪！ 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼞𖽑𖽔𖾏 𖼇𖽳𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑. 9 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼾𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼯𖽔 𖼅𖽔𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼊𖽔𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪？” 10 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼐𖽱𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖽐𖼎𖽙 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽙 𖼺𖽳 𖼦𖽡𖾐 𖽐𖼡𖽔 𖼑𖽻 𖼐𖽱𖾏, 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏.
11 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐, 𖼽𖽙 𖼺𖽳 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑. 12 𖽃𖽔 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼊𖽔𖾑 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼮𖽡𖽝𖾏, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼀𖽱𖾏 .” 𖼽𖽙 𖼺𖽳 𖼘𖽱𖾑𖽑 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼺𖽘𖾏, 𖼏𖽘 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼙𖽠𖾏. 13 𖽃𖽔 𖼁𖽙𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼺𖽘𖾏 𖼏𖽘 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼙𖽠𖾏. 14 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼾𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽙, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼑𖽔𖾏! 𖼞𖽪𖾑 𖼈𖽻𖾑 𖼨𖽑𖽳𖾏 𖼷𖽷𖾑 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼁𖽺𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼀𖽡𖾐 𖽐𖼞𖽔𖾏, 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼀𖽡𖾐, 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼀𖽱𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏. 15 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼋𖽡𖽝 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼏𖽘 𖼋𖽠𖾏 𖼻𖽻𖾑 𖼙𖽠𖾏. 16 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽑𖽡𖽠𖾏 𖼽𖽙 𖼺𖽳 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼳𖽻𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼮𖽡𖽝𖾏, 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖽐𖼞𖽔𖾏. 17 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼷𖽷𖾑 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼨𖽑𖽳𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼐𖽡𖾑, 𖼁𖽺𖾏 𖼋𖽔𖾏 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼞𖽪𖾑. 18 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼏𖽘 𖼋𖽠𖾏 𖼻𖽻𖾑 𖼙𖽠𖾏.
