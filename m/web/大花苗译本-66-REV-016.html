𖼀𖽜𖾐 𖼼𖽘𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔
16
𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏，𖽐𖼎𖽙 𖼳𖽤𖾏 𖼗𖽜𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼅𖽻𖾏， 𖼞𖽑𖽡𖽠𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐．”
2 𖼺𖽪𖾐 𖼁𖽱 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏， 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼷𖽑𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼎𖽺𖾏 𖼮𖽵𖾏 𖼗𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼑𖽡𖾏 𖼺𖽻𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼑𖽡𖾏 𖽐𖼀𖽠𖾏 𖼼𖽘𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑．
3 𖽃𖽔𖾏 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑，𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑，𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽔 𖼊𖽘𖾏 𖼋𖽘𖾏．
4 𖼷𖽷𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐，𖽃𖽻𖾐 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑 𖼋𖽘𖾏． 5 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽻𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：
“𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐，
𖼟𖽡 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼋𖽡𖽝．
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖽐𖼚𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑，
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼄𖽔𖾐 𖽐𖼮𖽑𖽘𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽻𖾏
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏．”
7 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖽃𖽔𖾑 𖼏𖽘 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽑𖽡𖾏：
“ 𖼾𖽙 𖼯𖽡𖽙𖾏，𖼮𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼘𖽪！
𖼟𖽡 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼯𖽡𖽙𖾏！”
8 𖼚𖽻𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐，𖼟𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐． 9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼮𖽑𖽘𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼞𖽪𖾐 𖼷𖽑𖽡𖽝𖾐，𖼎𖽺𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻， 𖼦𖽡𖾐 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
