𖼦𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼐𖽡 𖼳𖽡𖾑 𖼄𖽡𖾐 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 : 𖼽𖽔𖾏 𖼊𖽔 𖼳𖽡𖽫𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽡𖽻𖾏 𖼮𖽡𖾑 𖼺𖽷𖾑 𖼖𖽨 𖼖𖽑𖽡𖾏 (𖼟𖽪𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼖𖽪𖾑 𖼯𖽢𖾐 𖼖𖽑𖽡𖾏, 𖼖𖽑𖽡𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼖𖽪𖾑 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽙 𖼖𖽪𖾑 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽑𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐), 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 𖼳𖽪𖾑 𖼳𖽔𖾑. 2 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽔𖾐 𖼐𖽔 𖼐𖽡, 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼋𖽔𖾏, 𖼞𖽪𖾑 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼸𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼗𖽙 𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏. 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 : “𖼸𖽻𖾏 𖼟𖽪𖾏 𖼺𖽷𖾐 𖼗𖽙 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑, 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑.”
4 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼐𖽡𖽝𖾑, 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖼯𖽠 𖼐𖽑𖽪𖾐, 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽪𖾐 𖼡𖽜𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼐𖽑𖽡𖾏 : 5 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖽃𖽡𖾑 𖼘𖽪！ 𖼟𖽡 𖼻𖽙𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 , 𖼻𖽙𖾑 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑． 6 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼡𖽑𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 , 𖼟𖽡 𖼡𖽑𖽜𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽜 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽜𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼯𖽜𖾏 𖼀𖽡𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡, 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼋𖽘𖾏. 7 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝,𖼦𖽡𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑、 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖽐𖼎𖽙 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑. 8 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏: 𖼅𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼅𖽡 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼎𖽻𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐. 9 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼮𖽔𖾑 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑, 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑, 𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔 𖽐𖼊𖽪 𖽐𖼊𖽪𖾏 𖽃𖽔 𖽐𖼊𖽪 𖼊𖽡𖾐 𖼈𖽡𖽝𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑. 10 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏, 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽻𖾏, 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑. 11 𖼮𖽵𖾑 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼄𖽑𖽙𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 , 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏.” 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼮𖽡𖽠𖾑.
