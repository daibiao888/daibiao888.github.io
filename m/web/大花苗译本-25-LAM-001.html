1
𖽃𖽺𖾐! 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼾𖽨𖾏 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖼖𖽺𖾑 !
𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏,
𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼖𖽜𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖽃𖽡𖾑,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑.
 
2 𖼐𖽱𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼐𖽡𖽝𖾑, 𖼞𖽔𖾏 𖼅𖽔 𖽐𖼚𖽱𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼚𖽑𖽪𖾏,
𖼐𖽡𖽙𖾐 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼐𖽱𖾏.
𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏,
𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 .
 
3 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽻𖾏 𖼷𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼇𖽔𖾑 𖼷𖽻𖾏, 𖼎𖽺𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏.
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼐𖽱𖾏, 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏 𖼖𖽠𖾑 𖼯𖽢 𖼐𖽱𖾏 𖼋𖽘𖾏.
 
4 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼳𖽡 𖼣𖽔𖾑 , 𖼑𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼎𖽺𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐,
𖼐𖽱𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼐𖽱𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏,
𖼐𖽱𖾏 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 𖼮𖽑𖽡𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽔𖾏 𖼑𖽻𖾏 𖽐𖼷𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏.
 
5 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑,
𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽔𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐,
𖼐𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏.
 
6 𖼳𖽡 𖼣𖽔𖾑 𖼙𖽻 ( “𖼙𖽻”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 ”)𖼑𖽡𖾏 𖼚𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽑𖽔𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏.
𖼐𖽱𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼷𖽹𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼎𖽘𖾏 𖽐𖼡𖽜𖾐,
𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼏𖽘 𖼀𖽜𖾑 .
 
7 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼐𖽡𖽙𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼳𖽔𖾏 𖽐𖼮𖽡 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑.
𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽨𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏,
𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏.
 
8 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑, 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼐𖽱𖾏 ,
𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽔𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏 𖼑𖽡𖾏 𖼡𖽜𖾐.
 
9 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽪𖾑 𖼡𖽘𖾐 𖼮𖽑𖽙𖾏,
𖼐𖽱𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼊𖽱𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼦𖽡𖾐 𖼮𖽑𖽜 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼐𖽱𖾏. 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽔𖾏 𖼞𖽪𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽺𖾐 𖽐𖼷𖽱𖾏.”
 
10 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,
𖼐𖽱𖾏 𖼁𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑,
𖼐𖽑𖽡𖾏 𖼸𖽙 𖼸𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑, 𖼟𖽡 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼚𖽙.
 
11 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏, 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼡𖽜𖾑 𖼑𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼘𖽑𖽪𖾑 𖼡𖽻𖾐 𖼡𖽜𖾏, 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼾𖽨𖾏 𖼳𖽦𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽹𖾐 𖼑𖽔𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.”
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏
