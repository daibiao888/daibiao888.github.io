𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼖𖽨 𖼀𖽡𖽝𖾏 、 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 、 𖼖𖽡𖾏 𖽂𖽺𖾏 、 𖼽𖽨 𖼊𖽔𖾏 、 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 、 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 、 2 𖼊𖽗𖾏 、 𖼽𖽙 𖼺𖽳 、 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 、 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 、 𖼮𖽢𖾑 𖼊𖽳 、 𖼽𖽔𖾏 𖼳𖽳 .
𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
3 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽃𖽳𖾐 、 𖽃𖽳 𖼐𖽔 、 𖼳𖽷𖾏 𖼖𖽔𖾑 , 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽪𖾑 𖼽𖽔𖾏 𖽐𖼷𖽑𖽹𖾏 𖼑𖽜 𖼽𖽨 𖼊𖽔𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼯𖽡𖽙 . 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖽃𖽳𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼋𖽔 𖼙𖽠𖾏 . 4 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽪𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 , 𖼑𖽜 𖼽𖽨 𖼊𖽔𖾏 𖼅𖽔𖾏 𖼇𖽔 𖼖𖽳 𖼺𖽷𖾑 𖽐𖼎𖽙 𖼳𖽡𖽝𖾏 𖼖𖽔𖾑 . 𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼖𖽜𖾐 . 5 𖼇𖽔 𖼖𖽳 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼺𖽷𖾑 𖼖𖽪𖽝 、 𖼦𖽔𖾐 𖼄𖽪𖾐 𖼖𖽳 . 6 𖼳𖽡𖽝𖾏 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽡𖽾𖾑 𖼖𖽡𖾏 、 𖼽𖽡𖾐 𖼊𖽑𖽔𖾏 、 𖼳𖽡𖾑 𖼄𖽔𖾏 、 𖼮𖽢 𖼞𖽙 、 𖼊𖽔𖾏 𖼖𖽔𖾑 ( “𖼊𖽔𖾏 𖼖𖽔𖾑” 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 4:31 𖼟𖽪𖾏 “𖼊𖽔 𖼊𖽔𖾏”) , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼀𖽜𖾐 𖼖𖽜𖾐 . 7 𖼮𖽢𖾑 𖼄𖽡𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼞𖽔𖾑 , 𖼷𖽷𖾑 𖼽𖽔𖾏 𖼞𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐, 𖼚𖽑𖽙𖾏 𖼏𖽻 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 . 8 𖼽𖽡𖾐 𖼊𖽑𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾐 .
𖼳𖽡𖾑 𖼺𖽷𖾑 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
9 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼺𖽷𖾑 𖼖𖽪𖽝 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼖𖽔𖾑 𖼄𖽡𖽝 、 𖼖𖽔 、 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽹𖾏 . 10 𖼖𖽔 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼐𖽔 𖼊𖽔 , 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼐𖽔 𖼊𖽔 𖼅𖽔𖾏 𖼐𖽔 𖼳𖽪𖽝𖾏 . 𖼐𖽔 𖼳𖽪𖽝𖾏 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 11 𖼐𖽔 𖼳𖽪𖽝𖾏 𖼅𖽔𖾏 𖼺𖽔𖾐 𖼄𖽞 , 𖼺𖽔𖾐 𖼄𖽞 𖼅𖽔𖾏 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 . 12 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼅𖽔𖾏 𖽃𖽙 𖼀𖽡𖾐 𖼊𖽳 , 𖽃𖽙 𖼀𖽡𖾐 𖼊𖽳 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼳𖽡𖾑 . 13 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼊𖽪𖾐 𖽃𖽔𖾑 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔 , 𖼊𖽪𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽔 𖼊𖽔 、 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼄𖽡𖾐 𖼽𖽔𖾏 ( “𖼳𖽷𖾏 𖼄𖽡𖾐 𖼽𖽔𖾏 ” 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖽃𖽡𖾐 16:9 𖼟𖽪𖾏“𖼳𖽔𖾑 𖼄𖽔𖾐 ” ) 、 14 𖼚𖽻𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼐𖽔 𖼊𖽑𖽗𖾐 𖼐𖽡𖽝 、 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽹𖾏 、 15 𖼚𖽻𖾏 𖼗𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼳𖽡𖽝𖾑 、 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 . 16 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔 𖼟𖽪𖾏 𖼳𖽡𖾐 𖼖𖽪𖾐 𖼽𖽔𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼞𖽹𖾑 . 𖼳𖽡𖾐 𖼖𖽪𖾐 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼳𖽺𖾑 、 𖼽𖽙 𖼽𖽔 、 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼦𖽳， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 . 17 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼞𖽹𖾑 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 , 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾏 𖼊𖽑𖽡𖽝 .
𖼮𖽢𖾑 𖼖𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
