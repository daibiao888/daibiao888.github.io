𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏,
𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼋𖽘𖾏.
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽡𖽾 𖼗𖽙𖾏 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽡𖽾 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼗𖽙𖾏”).”
 
17 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏.
18 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼋𖽔𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑,
𖼟𖽡 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐,
𖼟𖽡 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼑𖽡𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏．
19 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽡𖽻𖾐 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼑𖽡𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼾𖽨 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼯𖽷𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 ．
𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼊𖽪𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.”
 
20 𖼅𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏,
𖼎𖽑𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼽𖽔𖾐 𖼞𖽙 ．”
𖼅𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽔𖾐 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏.
21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽔𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏,
𖼐𖽱𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽙𖾏 𖽃𖽻𖾐 𖼎𖽑𖽻𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖽐𖼀𖽑𖽔𖾐 𖼺𖽘𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽃𖽻𖾐 𖼎𖽺𖾏 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
22 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 !”
