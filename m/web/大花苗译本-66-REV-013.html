5 𖼊𖽪𖾑 𖼼𖽘𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 、𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼯𖽦𖾏， 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏， 𖼟𖽙𖾏 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼊𖽻𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖽃𖽔𖾐 𖼖𖽑𖽡𖾏． 6 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼎𖽺𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ，𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 ，𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏． 7 𖼎𖽺𖾏 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏，𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏 ，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼯𖽦𖾏，𖼟𖽙𖾏 𖼯𖽵𖾏 𖼺𖽘𖾏 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏、 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 、𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼗𖽪𖾏 、𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐． 8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐、 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼁𖽱𖾏 𖼷𖽷𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼗𖽙𖾏 ， 𖽐𖼷𖽷𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼮𖽡𖽝𖾏，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼐𖽡𖾑．
