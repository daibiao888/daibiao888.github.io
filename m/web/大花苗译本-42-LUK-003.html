𖼽𖽙 𖼦𖽗 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏
（𖼊𖽑𖽹𖾏 3：1-12；𖼞𖽑𖽙𖾐 1：1-8；𖼽𖽙 1：19-28）
3
𖼞𖽹𖾑 𖼺𖽔𖾐 𖼊𖽑𖽡 𖼀𖽡𖾐 𖼖𖽨 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑，𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼀𖽞𖾐 𖼊𖽨𖾑 · 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏， 𖼳𖽡𖾑 𖼖𖽡𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼅𖽻 𖽃𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼐𖽱𖾏 𖼞𖽱𖾑 𖼇𖽺𖾑 𖼖𖽡 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼅𖽻 𖽃𖽔𖾏 𖼽𖽡𖾐 𖼊𖽑𖽪𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼊𖽑𖽳 𖼖𖽔𖾑 𖼞𖽑𖽙𖾐 𖼐𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼡𖽑𖽙𖾏 𖼐𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼖𖽱𖾐 𖼺𖽔𖾐 𖼐𖽡𖽝 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼅𖽻 𖽃𖽔𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼖𖽡 𖼐𖽡𖼞𖽡𖾏 𖼄𖽱𖾏， 2 𖼽𖽔𖾏 𖼐𖽔 𖽐𖼎𖽙 𖼞𖽹𖾑 𖼽𖽔𖾏 𖼇𖽔 𖽃𖽔𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏．𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 ， 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼦𖽗 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼗𖽙𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏． 3 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼚𖽻𖾑 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼊𖽻𖾏 𖽐𖼷𖽔𖾑，𖼟𖽙𖾏 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏． 4 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：
“𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：
‘𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽵𖾑 𖼞𖽡𖾑，
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼟𖽙𖾏 𖽐𖼞𖽳𖾏．
5 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼟𖽱𖾐 𖼦𖽘𖾑 𖼽𖽔𖾑 𖼷𖽑𖽪𖾐 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼋𖽡𖽝𖾏，
𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼳𖽻𖾐 𖼽𖽔𖾑 𖼊𖽻𖾑𖽑 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏，
𖼞𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼞𖽑𖽡𖾐 𖼖𖽡𖾏 𖽐𖼞𖽑𖽻𖾏 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼨𖽑𖽢𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏，
𖼞𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡𖾏 𖼀𖽡𖾐 𖽐𖼞𖽡𖾐 𖼀𖽡𖾐 𖽐𖼞𖽱𖾐 𖼽𖽔𖾑 𖼨𖽑𖽢𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏．
6 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏．’”
7 𖼽𖽙 𖼦𖽗 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽱𖾏 𖽐𖼷𖽔𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏： “𖼅𖽡 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼐𖽡𖾑，𖼡𖽔 𖼋𖽱𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼖𖽺𖾑？ 8 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻． 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽑𖽡𖾏：‘ 𖼀𖽡𖾐 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼉𖽺𖾏 ．’ 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼋𖽡𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼳𖽠𖾑 𖼋𖽠𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏． 9 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼊𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽐𖼊𖽻𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏， 𖼷𖽑𖽔𖾏 𖼇𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽙𖾑 𖼗𖽙𖾏 ， 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼯𖽘𖾏 𖼋𖽠．” 10 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？” 11 𖼽𖽙 𖼦𖽗 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼮𖽑𖽙𖾏，𖼽𖽔𖾑 𖼄𖽔𖾐 𖼇𖽺𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼅𖽔𖾏 ，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏，𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑．” 12 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼋𖽔𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼷𖽔𖾑，𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏，𖼀𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？” 13 𖼽𖽙 𖼦𖽗 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼳𖽻𖾏 𖼊𖽔𖾐，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼳𖽻𖾏 ．” 14 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？” 𖼽𖽙 𖼦𖽗 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼼𖽠𖾏 𖼀𖽑𖽡𖽝𖾏 𖼡𖽔 𖼖𖽡𖾏 𖼋𖽱𖾏 ，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼾𖽨𖾏 𖼅𖽔𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼯𖽡𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐．”
𖼽𖽙 𖼦𖽗 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
