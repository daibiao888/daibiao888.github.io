（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 7：23 - 51）
4
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼷𖽷𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼋𖽻𖾏 , 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼇𖽘𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼺𖽡𖽝𖾐 𖼟𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 2 𖼎𖽺𖾏 𖼘𖽡𖾑𖽑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏 , 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼘𖽙𖾏 , 𖼺𖽡𖽝𖾐 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼇𖽘𖾑 𖼟𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖽐𖼮𖽔𖾏 𖽃𖽡𖾐 𖼞𖽑𖽡𖾏 𖼉𖽜𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 3 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖼅𖽔𖾏 𖼑𖽨𖾏 𖼼𖽘𖾏 (“𖼑𖽨𖾏 ”𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “𖼞𖽡𖾑 𖽐𖼀𖽔𖾐”), 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼋𖽪𖾏 𖼑𖽨𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖽐𖼞𖽑𖽳𖾏 , 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼘𖽡𖾑𖽑 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏 𖽃𖽡𖾑 𖼘𖽡𖾑𖽑 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏. 4 𖼅𖽔𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼋𖽪 𖼑𖽨𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖼮𖽹𖾑 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏 , 𖼷𖽷𖾐 𖼋𖽪 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 , 𖼷𖽷𖾐 𖼋𖽪 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 , 𖼷𖽷𖾐 𖼋𖽪 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 , 𖼷𖽷𖾐 𖼋𖽪 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼊𖽔𖾐 . 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽨𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖼑𖽨𖾏 𖽃𖽔𖾑 𖽐𖼷𖽘𖾏 𖼊𖽱𖾐 𖼎𖽻𖾏 𖽐𖼚𖽙 . 5 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏 𖼊𖽔𖾐 𖼚𖽻𖾐 𖽐𖼊𖽡𖾏 , 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼋𖽺𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏 , 𖽐𖼊𖽡𖾏 𖼋𖽻𖾏 𖼷𖽷𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽔𖾏 𖼊𖽑𖽳 𖽃𖽻𖾐 . 6 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼟𖽻𖾏 𖼗𖽪𖾏 𖼋𖽺𖾏 : 𖼀𖽜𖾐 𖼖𖽪𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 , 𖼀𖽜𖾐 𖼖𖽪𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 . 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼗𖽙 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼽𖽔𖾑 𖽐𖼷𖽔𖾑 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 ； 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 .
7 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼙𖽔 𖼏𖽵 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼟𖽻𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼞𖽪𖾐 , 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 : 𖼀𖽜𖾐 𖼖𖽪𖾐 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼺𖽷𖾏 , 𖼀𖽜𖾐 𖼖𖽪𖾐 𖼐𖽡𖽙𖾐 𖼺𖽘𖾐 𖼇𖽳𖾏 . 8 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼟𖽻𖾏 𖼸𖽱𖾏 𖼞𖽡 𖼏𖽻𖾏 ,𖼮𖽙𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 : 𖼀𖽜𖾐 𖼸𖽱𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼺𖽷𖾏 , 𖼀𖽜𖾐 𖼸𖽱𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼇𖽳𖾏 . 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽪𖾏 𖼞𖽑𖽙𖾐 𖼞𖽪𖾐 .
