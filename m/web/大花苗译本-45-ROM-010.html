5 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑，𖼄𖽙𖾑 𖼳𖽡𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 ，𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏．” 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼉𖽺𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ：“ 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏：‘ 𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏？’” 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏， 𖼅𖽻𖾏 𖼨𖽑𖽳𖾏 𖼮𖽡𖾑 𖼊𖽪 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏． 7 “𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 ：‘𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏？’” 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 ，𖼅𖽻𖾏 𖼨𖽑𖽳𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏 ． 8 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑 ？𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：
“ 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼐𖽡𖾑 𖼡𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼚𖽡𖾐，
𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘，𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼊𖽑𖽡𖽝𖾐．”
𖼟𖽪𖾏 𖼐𖽑𖽡𖾏， 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽑𖽔𖾐． 9 𖼟𖽡 𖼮𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼯𖽜𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼮𖽵𖾑，𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏，𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏． 10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼯𖽜𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏． 11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼻𖽡𖽝𖾏 𖼐𖽱𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏．” 12 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼇𖽺𖾐 𖼡𖽔𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽔 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽵𖾑 𖼊𖽔𖾐，𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼐𖽱𖾏，𖼐𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 13 𖽃𖽡𖾑 𖼟𖽪𖾏：“ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼨𖽑𖽪𖾏 𖼮𖽵𖾑 𖽐𖼷𖽷𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏．”
14 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼮𖽜 𖼐𖽱𖾏 𖼺𖽷𖾐，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼐𖽱𖾏 𖼘𖽪 ？ 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 ，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖽐𖼮𖽜 𖼐𖽱𖾏 𖼖𖽺𖾑？𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼐𖽙𖾑𖽑 𖼘𖽪？ 15 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼅𖽻 𖼺𖽷𖾐，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼡𖽑𖽔𖾐 𖼖𖽺𖾑 ？𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼻𖽻𖾏， 𖼡𖽑𖽔𖾐 𖼻𖽻𖾏 𖼄𖽻𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽠𖾏 𖼐𖽜𖾑 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝！” 16 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾏 𖼯𖽪𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼄𖽻𖾏 𖼻𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼮𖽜 𖼖𖽺𖾑 ？” 17 𖼑𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼀𖽻𖾐 ，𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼑𖽻 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼗𖽙𖾏 𖼯𖽡𖽙𖾏，𖼑𖽻 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼗𖽙𖾏 𖼯𖽡𖽙𖾏． 18 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽪 ，𖼋𖽡𖽝𖾏 𖼾𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼐𖽙𖾑𖽑 𖼺𖽷𖾐？ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏：
“𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽘𖾐 𖼗𖽪 𖼡𖽑𖽔𖾐 𖽐𖼷𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾏 𖼋𖽻𖾏，

19 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼑𖽪 ，𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐？𖼄𖽙𖾑 𖼳𖽡𖾑 𖼺𖽪𖾐 𖼐𖽑𖽡𖾏：
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽡𖾑， 𖼊𖽱𖾑 𖼅𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐，
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔，𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 ．”
20 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼚𖽔𖾐 𖼐𖽑𖽡𖾏 ：
“𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 𖽃𖽡𖾑，𖼞𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼢𖽳 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ．”
21 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼳𖽤𖾐 𖽐𖼞𖽡𖽠𖾏 𖼋𖽡𖾏 𖼨𖽑𖽪𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖽃𖽡𖾑．”
