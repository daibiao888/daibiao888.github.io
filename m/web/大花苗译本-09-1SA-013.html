𖼺𖽻𖾐 𖼘𖽙 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑. 2 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 𖼸𖽙𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽡𖽙𖾐 𖼄𖽡 𖼄𖽙 𖽐𖼎𖽙 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼸𖽙𖾏 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼑𖽡𖾏 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏, 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽹𖾑 𖼳𖽙𖾏 𖽃𖽡𖾑, 𖼺𖽻𖾐 𖼘𖽙 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏. 3 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖽐𖼊𖽻𖾏 𖼮𖽢𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽪𖾑 𖼾𖽡 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑， 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏. 𖼺𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐, 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏: 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑. 4 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼊𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽪𖾑 𖼾𖽡 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑， 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼳𖽪𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼎𖽺𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼺𖽻𖾐 𖼘𖽙 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼮𖽡 𖼮𖽢.
