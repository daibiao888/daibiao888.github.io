𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏

25
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼟𖽡.
2 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼟𖽡,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑.
3 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼯𖽔 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽪𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
 
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑,
𖼄𖽔𖾐 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑！
5 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑,
𖼞𖽪𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼋𖽙𖾏 𖼟𖽡.
 
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼎𖽙 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼗𖽙𖾏 𖽃𖽔𖾐 𖼅𖽔𖾏 𖼋𖽘.
7 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼺𖽷𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼟𖽡 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐, 𖼸𖽙𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑.
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼋𖽡𖽝,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏.
9 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜,
𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
10 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖽃𖽡𖾑,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏.
 
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼯𖽔𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖼺𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽃𖽺𖾐 𖼐𖽑𖽤𖾑.
12 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼺𖽹𖾑.
13 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪,
𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐.
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼾𖽨𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
15 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼡𖽪𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏.
 
16 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖼚𖽔𖾏 𖼞𖽪𖾑, 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼗𖽔 𖼎𖽺𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏.
17 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔.
18 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽔𖾏 𖼞𖽪𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 、 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽔𖾑,
𖼯𖽔𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼾𖽔𖾏.
 
19 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝.
20 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡.
21 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽡.
 
22 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏.
