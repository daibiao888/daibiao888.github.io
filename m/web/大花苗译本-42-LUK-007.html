𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼎𖽺𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾． 2 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ，𖼐𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼋𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐，𖼄𖽙𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔． 3 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽪𖾏，𖼎𖽺𖾏 𖽐𖼮𖽢𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼗𖽻，𖼅𖽻𖾏 𖽐𖼊𖽜 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽔𖾏 𖼞𖽑𖽙𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑． 4 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼎𖽑𖽻𖾏 𖼼𖽙 𖽐𖼊𖽜 𖼐𖽑𖽡𖾏：“𖼐𖽱𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑， 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼀𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐，𖼐𖽱𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼀𖽡𖾐．” 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏，𖼡𖽔𖾏 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼚𖽡𖾐，𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼎𖽺𖾏 𖽐𖼮𖽢𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪 ！𖼦𖽡𖾐 𖼅𖽪𖾏 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏，𖼟𖽡 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏． 7 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡，𖼯𖽔 𖼽𖽔𖾑 𖼟𖽡 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼊𖽔𖾐，𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼻𖽻𖾏 𖼋𖽘𖾏． 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽪𖾑，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽵𖾏，𖽐𖼎𖽙 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 ：‘𖼅𖽻𖾏’ 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏，𖽐𖼎𖽙 𖼷𖽷𖾑 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏： ‘𖼋𖽔𖾏 ’𖼐𖽱𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏，𖽐𖼎𖽙 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏：‘ 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑’，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏．” 9 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽙𖾑 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 ，𖼎𖽺𖾏 𖼏𖽘𖾏 𖼯𖽡𖽙𖾏，𖼦𖽡𖾐 𖼋𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽪𖾑 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏．” 10 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼺𖽷𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽔𖾏，𖼁𖽙𖾏 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖼻𖽻𖾏 𖼋𖽘𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼊𖽪𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙
11 𖼮𖽑𖽡𖽻𖾏 𖼗𖽠𖾏 𖽃𖽡𖾐 𖼷𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼐𖽔 𖼽𖽡𖽾𖾑， 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏 ． 12 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽱𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏，𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼘𖽑𖽘𖾑 𖼾𖽨 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐，𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏． 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼅𖽻𖾏 𖼺𖽘𖾏 𖼊𖽪𖾐 𖼋𖽔． 13 𖼮𖽵𖾑 𖼁𖽙𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼐𖽡𖾑 ，𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼡𖽑𖽔𖾐 𖼐𖽡𖽝𖾑 𖼊𖽡𖾐．” 14 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼀𖽔𖾐 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼎𖽺𖾏 𖼳𖽠𖾑 𖽐𖼎𖽙𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐， 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏！” 15 𖼷𖽷𖾑 𖼊𖽪𖾐 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐． 16 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼋𖽘𖾏！” 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 ：“𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼋𖽘𖾏！” 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑，𖽐𖼎𖽙 𖼟𖽻𖾏 𖼯𖽪𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏．

18 𖼽𖽙 𖼦𖽗 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽳𖾏 𖼽𖽙 𖼦𖽗． 19 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏，𖼷𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽵𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖼘𖽪？𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽷𖾐 𖼘𖽪？” 20 𖽃𖽔𖾏 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗 𖼷𖽷𖾐 𖼀𖽡𖾐 𖼋𖽔𖾏 𖼑𖽪 𖼟𖽡：‘𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖼘𖽪？𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽷𖾐 𖼘𖽪？’” 21 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼊𖽪𖾐 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 ，𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔，𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼚𖽪𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑． 22 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “𖼅𖽡 𖼅𖽻𖾏 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽑𖽙𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽙 𖼦𖽗，𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑，𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼸𖽠𖾏 𖼅𖽻𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼗𖽘𖾐 𖼸𖽠𖾏 𖼐𖽑𖽙𖾑 𖼗𖽪𖾏，𖼊𖽪𖾐 𖼋𖽔𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑 𖼄𖽻𖾏 𖼻𖽻𖾏． 23 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼢𖽻 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼽𖽙 𖼦𖽗 𖼷𖽷𖾐 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼽𖽙 𖼦𖽗 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏，𖼟𖽪𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪？𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖼟𖽪𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼖𖽺𖾑？ 25 𖼅𖽡 𖼋𖽠𖾏 𖼅𖽻𖾏 ，𖼋𖽡𖽝 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪？𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼯𖽪𖾏 𖽐𖼀𖽔 𖼘𖽪？𖼑𖽔𖾏 𖼘𖽪 ！𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼟𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏，𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻． 26 𖼅𖽡 𖼋𖽠𖾏 𖼅𖽻𖾏 ， 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？𖼟𖽪𖾏 𖼑𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼖𖽔𖾏？𖼾𖽙 𖼯𖽡𖽙𖾏， 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ， 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐． 27 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼺𖽻𖾏 𖼐𖽑𖽡𖾏：
‘ 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼨𖽑𖽳𖾏 𖼟𖽡 𖽐𖼊𖽡𖾏，
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼊𖽑𖽔𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼟𖽡 𖼞𖽡𖾑 ’， 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏．
28 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼘𖽑𖽘𖾑 𖼾𖽨，𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼽𖽙 𖼦𖽗，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖽃𖽡𖾑 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏．” 29 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏，𖼟𖽪𖾏 𖼙𖽔 𖼊𖽻𖾏 𖼽𖽙 𖼦𖽗 𖽐𖼷𖽔𖾑，𖼐𖽑𖽙𖾑 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳． 30 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼊𖽻𖾏 𖼽𖽙 𖼦𖽗 𖽐𖼷𖽔𖾑，𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏．
31 𖼮𖽵𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽐𖼮𖽔𖾏 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽺𖾑？𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪？ 32 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：
