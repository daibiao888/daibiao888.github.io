𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼟𖽻𖾏 𖼼𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
（𖼋𖽠𖾏 34：1 - 10）
10
“𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾏 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 : 𖼟𖽡 𖼽𖽔𖾑 𖼸𖽻 𖼋𖽠𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 , 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽡𖾑 , 𖽐𖼮𖽡𖾏 𖼎𖽻𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼼𖽪𖾏 𖼋𖽠𖾏 . 2 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼟𖽡 𖼮𖽻𖾏 𖽐𖼮𖽻 𖼺𖽘𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼊𖽠𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖼐𖽡𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖼐𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖽐𖼚𖽙 𖼼𖽪𖾏. 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼼𖽪𖾏 , 𖼎𖽺𖾏 𖼸𖽻 𖼋𖽠𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 , 𖽐𖼎𖽙 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 , 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼐𖽡𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏 . 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽑𖽡𖽠𖾏 𖼟𖽻𖾏 𖼼𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 , 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽡𖾑 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼺𖽻𖾏 , 𖼺𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖼐𖽡𖾑 , 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏 . 5 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖽐𖼡𖽜𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 , 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼺𖽷𖾐 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 .
6 ( 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼀𖽡𖾐 𖼘𖽙 𖼀𖽡𖾐 𖼐𖽡 · 𖼽𖽔𖾏 𖼞𖽔𖾑 (“𖼀𖽡𖾐 𖼘𖽙 𖼀𖽡𖾐 𖼐𖽡 · 𖼽𖽔𖾏 𖼞𖽔𖾑” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “𖼽𖽔𖾏 𖼞𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐”) 𖼳𖽠𖾑 𖼊𖽠𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼖𖽔𖾑 , 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼋𖽔 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑． 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼙𖽻 𖼋𖽪𖾏 . 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼳𖽠𖾑 𖼊𖽠𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼞𖽪 𖼞𖽙𖾑 𖼊𖽔𖾏 , 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪 𖼞𖽙𖾑 𖼊𖽔𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼽𖽙 𖼀𖽔𖾑 𖼊𖽑𖽔𖾑 . 8 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽑𖽡𖽠𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼇𖽺𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼞𖽱𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 , 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 , 𖼻𖽡𖽝𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 . 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼀𖽠 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼏𖽵 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 ) .
