𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖽐𖼊𖽠𖾑
𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼊𖽻𖾏 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐
1
𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼽𖽔𖾏 𖼐𖽔 𖼊𖽑𖽪 𖼙𖽻 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼳𖽡𖾑 𖼖𖽳 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼗𖽪𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐. 2 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐. 3 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏, 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼀𖽜𖾐 𖼖𖽑𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐.
4 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼐𖽑𖽡𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
5 “𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼄𖽔𖾐 𖼟𖽡 𖼷𖽷𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐, 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼋𖽘,
𖼟𖽡 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐, 𖼞𖽪𖾑 𖼇𖽺𖾐 𖼟𖽡 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏.
𖼞𖽪𖾑 𖼷𖽷𖾐 𖼟𖽡 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 (“𖼷𖽑𖽔𖾏 𖼞𖽪𖽙” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐”)𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼋𖽘.”
6 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽺𖾐! 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼺𖽷𖾐.”
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼺𖽷𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼟𖽡 𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏,
𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏.
8 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐,
𖼽𖽔𖾑 𖼮𖽨𖾏 𖼟𖽡.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪!
𖼞𖽪𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘.
10 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼑𖽔𖾏 𖼺𖽹𖾑 𖼮𖽘𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼚𖽙𖾐 𖼺𖽘𖾏、 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑、 𖼖𖽻𖾐 𖼺𖽘𖾏、 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼎𖽺𖾏 𖼯𖽡𖽙.”
𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼯𖽡 𖽃𖽔𖾑 𖼯𖽡 𖼷𖽷𖾑 𖼞𖽑𖽡𖽠𖾐
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼎𖽺𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐, 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏?” 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼯𖽡 𖽃𖽔𖾑 𖼯𖽡 𖼷𖽷𖾑 𖼞𖽑𖽡𖽠𖾐.” 12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑.”
𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏.” 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼀𖽑𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽑𖽙𖾐 𖽐𖼀𖽻𖾏, 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼘𖽡𖾑𖽑 𖽐𖼮𖽢𖾑𖽑.”
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑. 15 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻, 𖼞𖽑𖽡 𖼙𖽻 𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽻.” 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 16 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼟𖽪𖾏 𖼑𖽜 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏、𖼷𖽑𖽷𖾑 𖼳𖽤𖾐 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺、𖼛𖽪𖾏 𖼇𖽻𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏.
