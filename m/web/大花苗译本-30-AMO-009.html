𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑
9
𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽔 𖼏𖽘 , 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼮𖽷𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐, 𖼟𖽙𖾏 𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏 𖽃𖽔𖾏 𖼼𖽙 ,
𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼀𖽻𖾐 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾑 𖼇𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏.
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 .
2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼸𖽙 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼮𖽤𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽤𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 .
3 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽺𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼄𖽡𖾐 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼯𖽡𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼚𖽺𖾏 𖼚𖽔𖾏 𖼡𖽘𖾐 𖼳𖽡𖽠𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽙𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
4 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽔𖾏 𖼙𖽠𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽙𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼺𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼦𖽡𖾐 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.”
 
5 𖼮𖽵𖾑 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼊𖽡𖾐, 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼾𖽘𖾏 𖼙𖽠𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏.
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼐𖽡 𖼘𖽙 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏 𖽐𖼀𖽻𖾏 𖼘𖽑𖽙𖾐,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼖𖽻𖾏 𖽐𖼡𖽜𖾏 𖼙𖽠𖾏.
6 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼟𖽙𖾏 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
 
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪 !
𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼅𖽡 𖼺𖽱𖾏 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽺𖾑?
𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼨𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐,
𖼨𖽑𖽳𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼮𖽢𖾑 𖼇𖽺𖾑 𖼊𖽑𖽙 ,
𖼨𖽑𖽳𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼮𖽡 𖽃𖽳𖾐 𖼖𖽺𖾑?
8 𖼑𖽔𖾏 𖼘𖽪! 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽔
𖼳𖽹𖾐 𖼑𖽔𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼐𖽡𖾑 ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏 𖼖𖽻𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼮𖽘𖾑 𖼺𖽘𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
9 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾐 𖼟𖽪𖾏 𖼷𖽹𖾑,
𖼄𖽔𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼺𖽱𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼈𖽘𖾐 𖼮𖽑𖽻𖾏 𖼮𖽑𖽻𖾏 𖼡𖽻𖾐,
𖼈𖽡𖽝𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
10 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼐𖽑𖽡𖾏:
‘ 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽠𖾑 𖼦𖽡𖾐 𖼯𖽢 𖼀𖽡𖾐 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼀𖽡𖾐 .’
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.”
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼑𖽡𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏,
11 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼢𖽻 𖼙𖽠𖾏,
𖽐𖼷𖽷𖾑 𖼨𖽑𖽢𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 ,
𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏,
𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏, 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
12 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼳𖽙𖾏 𖽃𖽡𖾑
𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
 
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪 ! 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏,
𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖼀𖽳𖾐 𖼮𖽘𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏,
