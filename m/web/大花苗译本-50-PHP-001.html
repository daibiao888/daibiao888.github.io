𖼮𖽡𖾑 𖼊𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾑 𖼀𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼊𖽑𖽡 𖼄𖽙𖾑 𖼊𖽑𖽹𖾏 ，𖼺𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽵𖾐 𖼑𖽪𖾏，𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼇𖽺𖾑 𖼖𖽡 𖼀𖽡𖾐 ， 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼊𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼚𖽙． 2 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼋𖽡𖽝𖾏 𖼋𖽪，𖼎𖽑𖽻𖾏 𖼀𖽡𖾐 𖼉𖽺𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼚𖽔𖾏 𖼅𖽡．
𖼀𖽻𖾐 𖼘𖽙 𖼘𖽑𖽪𖾐 𖼸𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
