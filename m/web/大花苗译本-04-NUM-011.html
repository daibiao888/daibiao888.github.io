𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽐𖼎𖽙𖾑 𖼸𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏． 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼋𖽘𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼯𖽡 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼯𖽡 𖼟𖽪𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽐𖼚𖽜𖾏 . 2 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽝𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏𖼄𖽙𖾑 𖼳𖽡𖾑 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏 . 3 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼊𖽑𖽔𖾑 𖼀𖽡𖾏 𖼖𖽔𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼯𖽡 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 .
4 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼳𖽠𖾑 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽠𖾑 𖼉𖽳 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼎𖽺𖾏 𖼐𖽡𖽝𖾑 𖼢𖽔𖾏 𖼐𖽑𖽡𖾏 : “𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽐𖼡𖽹𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼑𖽻𖾏 𖼖𖽺𖾑 ? 5 𖼀𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼷𖽹𖾏 𖼊𖽻𖾏 𖽐𖼀𖽳𖾏 𖼑𖽻𖾏 , 𖼎𖽺𖾏 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼞𖽡𖾑 𖼛𖽡𖾐、𖼞𖽡𖾑 𖽐𖼀𖽔𖾐 𖼡𖽘𖾐 𖼄𖽪𖾑、𖼞𖽡𖾑 𖼎𖽑𖽔𖾏、𖼛𖽙𖾏、𖼢𖽜𖾏 . 6 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼺𖽘𖾏 𖼋𖽘𖾏 , 𖼋𖽠𖾏 𖼋𖽡𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼐𖽡𖾑 𖼺𖽘𖾏 , 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 .”
7 𖼋𖽡𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼐𖽡𖾑 𖼺𖽱𖾏 𖽃𖽔 𖼛𖽡𖾏 𖼻𖽻𖾐 𖼷𖽑𖽱𖾐 , 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 （“𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼛𖽙𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼦𖽘𖾏 ”） . 8 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 , 𖼳𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼄𖽔𖾐 𖼐𖽔 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼢𖽪𖾏 𖼡𖽜𖾑 𖼢𖽪𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼯𖽡𖽙𖾏 𖼎𖽦𖾑𖽑 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼦𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼀𖽑𖽻𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼳𖽡𖽝𖾐 , 𖽐𖼷𖽑𖽡𖽝𖾐 𖼦𖽘𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖽐𖼊𖽜𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼦𖽘𖾏 . 9 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾑 𖼙𖽱 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼄𖽔𖾐 𖼐𖽔 𖼖𖽔𖾏 𖽐𖼎𖽙 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 .
𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐
