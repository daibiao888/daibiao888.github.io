26 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽳𖾑𖽑 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑, 𖼽𖽡𖾐 𖼦𖽪𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼏𖽘 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼁𖽠𖾏 𖽃𖽡𖾑, 𖼀𖽜𖾑 𖼸𖽙 𖼳𖽡𖾑 𖼽𖽡𖾑 𖼖𖽔𖾑. 27 𖼸𖽙 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽐𖼡𖽜𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐, 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼨𖽑𖽳𖾏 𖼞𖽡𖾑, 28 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼋𖽔𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼅𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽘𖾏.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏, 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼚𖽔𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏, 𖼦𖽡𖾐 𖼮𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖾏 𖼙𖽠𖾏. 29 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏. 30 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽵𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼊𖽻𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏.
