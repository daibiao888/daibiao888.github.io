14
𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏,
𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏.
2 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐.
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖼟𖽙𖾏 𖼊𖽪𖾑 𖼡𖽠𖾏 𖽐𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼾𖽨𖾏.
4 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽨𖾏, 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏；
𖼡𖽻𖾐 𖼖𖽻𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽨𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑.
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼗𖽔 𖽐𖼎𖽑𖽦𖾑 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐, 𖼖𖽡 𖼟𖽻𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 .
7 𖼟𖽡 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼀𖽻𖾐 𖼟𖽳.
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏，𖼐𖽱𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖼡𖽔 𖼏𖽔 , 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 ( 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “ 𖼾𖽨𖾏 𖼮𖽑𖽵𖾑 𖼾𖽨𖾏” ).
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼇𖽜𖾏 ( 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼞𖽑𖽡𖽠𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼘𖽑𖽪𖾑 𖼎𖽙𖾑 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔” ).
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖽃𖽺𖾐 𖼙𖽔 𖽃𖽡𖾑 𖼷𖽑𖽡𖽝𖾏.
10 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏, 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼊𖽔𖾐；
𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼚𖽙𖾏, 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼊𖽻𖾏.
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽻 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏.
12 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼸𖽙𖾐 𖼞𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏.
13 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼙𖽔 𖼚𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖼸𖽠𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼊𖽳𖾏；
𖼙𖽔 𖼚𖽙𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏.
14 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼾𖽨𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏；
𖼻𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏.
15 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼊𖽪𖾏 𖽐𖼮𖽜 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼷𖽑𖽔𖾏 𖼏𖽔𖾏 𖼑𖽔𖾏 𖼞𖽡𖾑 .
16 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼎𖽺𖾏 𖼀𖽜𖾑 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼾𖽨𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼾𖽨𖾏.
17 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔 𖼏𖽔 ；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽪𖾐.
18 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼊𖽻𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏, 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽔𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑.
19 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼻𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽃𖽔 𖼛𖽻𖾏．
20 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏, 𖼼𖽷𖾏 𖼼𖽙 𖼊𖽪𖾏 𖼳𖽪𖾐 𖼐𖽱𖾏 ；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼽𖽺 𖼗𖽔 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐.
21 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼼𖽷𖾏 𖼼𖽙 , 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼾𖽔𖾏；
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
22 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼅𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼘𖽪?
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 .
23 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼘𖽑𖽻𖾐 𖽐𖼡𖽔𖾏, 𖼊𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏.

27 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐,

28 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏,

29 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼷𖽹𖾑 , 𖼢𖽳 𖼋𖽠𖾏 𖼾𖽨𖾏 𖼡𖽔 𖼏𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.

32 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽠𖾐 𖼢𖽻 ,
