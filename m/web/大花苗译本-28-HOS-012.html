𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽐𖼎𖽙𖼽𖽨 𖼊𖽔𖾏
12
𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼑𖽻𖾏 𖼮𖽢𖾏,
𖼎𖽺𖾏 𖼖𖽠𖾑 𖼸𖽙𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼮𖽢𖾏,
𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑 𖽐𖼎𖽙 𖽐𖼚𖽪𖾏 𖼸𖽳,
𖽐𖼎𖽙 𖼽𖽔𖾏 𖼳𖽪 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖼄𖽔𖾐 𖼛𖽙𖾏 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 .
 
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙𖼽𖽨 𖼊𖽔𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 ,
𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏 .
3 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼷𖽹𖾏 𖼊𖽔𖾑 𖼮𖽡𖽝𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖽃𖽔 𖼙𖽻𖾏 𖼊𖽠𖾏,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾑 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡 𖽐𖼮𖽔 𖼛𖽘𖾏 𖼼𖽙𖾏.
4 𖼐𖽱𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼦𖽡 𖽐𖼮𖽔 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏,
𖼐𖽡𖽝𖾑 𖼼𖽷𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼡𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽜 .
𖼐𖽡𖽙𖾐 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖽐𖼮𖽷𖾐 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏．
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏.
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼻𖽙𖾑 𖼏𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏、 𖼖𖽠𖾑 𖼞𖽡𖾑， 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽂𖽪𖾏 𖽂𖽪𖾏 𖽃𖽔𖾏 𖼟𖽙𖾏 𖼮𖽵𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐
7 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏, 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼸𖽱𖾏 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏,
𖽃𖽺𖾐 𖽐𖼡𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑.
8 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏, 𖼊𖽻𖾏 𖼋𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼾𖽔𖾏.”
9 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 ,
𖼺𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.
 
10 𖼞𖽪𖾑 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼎𖽺𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼚𖽔𖾐 𖼼𖽘𖾏,
𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 .
11 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾏 𖼊𖽜𖾑.
𖼐𖽡𖽙𖾐 𖼮𖽡 𖼮𖽢 𖼖𖽑𖽘𖾑 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼟𖽱𖾏 𖼖𖽡𖾏 𖼉𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼚𖽻𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐.
12 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽜 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽤𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽤𖾐 𖼑𖽔𖾏 𖼞𖽻𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏.
13 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼨𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏 ,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼎𖽺𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
14 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼟𖽙𖾏 𖼮𖽵𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼐𖽱𖾏 𖼮𖽵𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼊𖽑𖽱𖾏 𖼷𖽑𖽘𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 .
