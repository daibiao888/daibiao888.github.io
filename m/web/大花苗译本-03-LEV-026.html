𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽠𖾏
（𖼎𖽺𖾏 7：12 - 24；28：1 - 14）
26
“𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼅𖽡 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼁𖽠𖾏 𖼼𖽘𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼁𖽠𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼁𖽠𖾏 𖼼𖽘𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼷𖽑𖽙𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 , 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 2 𖼅𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 ,𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .”
𖼻𖽙𖾑 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼅𖽳𖾏
3 “𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 , 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 4 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼟𖽙𖾏 𖼑𖽘 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼅𖽡 , 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽔𖾏 𖼋𖽠𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 , 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 . 5 𖼅𖽡 𖽐𖼊𖽻𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼚𖽡𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏， 𖼚𖽡𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼽𖽔𖾑 𖼚𖽡𖾏 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼮𖽘𖾑 . 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 . 6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼅𖽡 𖼢𖽻 𖼀𖽱𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽠𖾏 𖼅𖽡． 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼎𖽑𖽻𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽔 𖼋𖽘𖾏 𖼺𖽘𖾏 , 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼦𖽡𖾐 𖽐𖼮𖽡𖽻𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 . 7 𖼅𖽡 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 . 8 𖼅𖽡 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 . 9 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽻𖾏 𖼮𖽳𖾐 𖼅𖽡 , 𖼟𖽙𖾏 𖼅𖽡 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼏𖽵 . 10 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼡𖽻𖾐 𖽃𖽔𖾏 𖼗𖽻𖾏 , 𖼎𖽺𖾏 𖼑𖽜 𖼡𖽻𖾐 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼑𖽡𖾏 𖼡𖽻𖾐 𖽃𖽔𖾏 𖼗𖽻𖾏 . 11 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 , 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼳𖽪𖾐 𖼅𖽡 . 12 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 . 13 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼙𖽔 𖼄𖽔𖾐 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼟𖽙𖾏 𖼅𖽡 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽜𖾑； 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼖𖽙𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼺𖽘𖾏 , 𖼟𖽙𖾏 𖼅𖽡 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼮𖽨𖾏 𖼀𖽡𖾏 𖼮𖽨𖾑 𖼅𖽻𖾏.”
𖼦𖽡𖾐 𖼻𖽙𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔

14 “𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 , 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 , 15 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 , 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 , 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 16 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼅𖽡 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼑𖽪𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡 , 𖼄𖽙𖾐 𖼡𖽔 𖼺𖽻、𖼄𖽙𖾐 𖼚𖽪𖾐 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 . 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼀𖽳𖾐 𖼮𖽘𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼯𖽡𖽙 𖽃𖽡𖾑 . 17 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼷𖽜𖾑 𖼚𖽑𖽪𖾏 𖼎𖽻𖾏 𖼅𖽡 , 𖼅𖽡 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐． 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼅𖽡 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼅𖽡； 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 , 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼏𖽘 𖼀𖽜𖾑 . 18 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼺𖽷𖾐 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼅𖽡 𖼾𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼳𖽤𖾏 𖽐𖼀𖽠 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼅𖽡 . 19 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼅𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼙𖽙𖾏 𖽐𖼊𖽪𖾏， 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼡𖽑𖽔𖾑 𖼺𖽱𖾏 𖼖𖽑𖽻𖾏， 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽐𖼞𖽡 𖼺𖽱𖾏 𖼋𖽻𖾏 . 20 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼋𖽠𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼋𖽘𖾏 , 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 .
21 𖼅𖽡 𖼮𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐 , 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼅𖽡 𖼾𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼳𖽤𖾏 𖽐𖼀𖽠 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼅𖽡 . 22 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼦𖽔𖾏 𖼑𖽻𖾏 𖼅𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 , 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼅𖽡 𖼯𖽦𖾏 , 𖼟𖽙𖾏 𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽙𖾑 𖼯𖽠 , 𖼅𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 .
