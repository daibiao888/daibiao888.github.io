5 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼾𖽨𖾏 𖼋𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖽐𖼚𖽙 𖼟𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏， 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼾𖽨𖾏． 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼐𖽡𖽙𖾐 𖼅𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼘𖽪？𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙，𖼅𖽡 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏． 6 𖼞𖽪𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼀𖽻𖾐 ，𖼀𖽡𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏． 7 𖼀𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 ； 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼳𖽠𖾏 𖼋𖽻𖾏 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏， 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ， 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼳𖽠𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼈𖽡𖽝𖾏． 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝 ， 𖼮𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐． 9 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼅𖽡 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾑， 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝． 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽪𖾏： 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏． 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼅𖽡， 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼅𖽡， 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 ， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐．
11 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼺𖽷𖾐： 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 ，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑， 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏，𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 ， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐． 12 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼡𖽑𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏， 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼑𖽪 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼄𖽻𖾏 ．
13 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽪 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪．
14 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏、 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏、 𖼊𖽑𖽡𖽝𖾐 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼑𖽡𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏， 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐．
