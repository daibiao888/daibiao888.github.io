𖼡𖽑𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏
（𖼋𖽠𖾏 34：11 - 16）
7
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 , 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 𖼟𖽪𖾏 𖼦𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼞𖽳 𖼮𖽢𖾑 𖼺𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼀𖽡𖾐 𖼖𖽡𖾏 𖼳𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼳𖽡𖾑 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 , 𖼟𖽪𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾐 𖼟𖽡 𖼋𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽡 . 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 , 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽻𖾑 𖼅𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 3 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽑𖽻𖾐 𖼟𖽙 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼅𖽡 𖽐𖼷𖽑𖽹𖾏 𖼡𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼊𖽪𖾐 𖼷𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 . 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼇𖽳𖾑 𖼮𖽵𖾑 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼮𖽵𖾑 , 𖼅𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼙𖽠𖾏, 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼅𖽡 , 𖼎𖽺𖾏 𖼞𖽑𖽜 𖼑𖽜 𖼖𖽻𖾐 𖼅𖽡 𖼺𖽘𖾏 . 5 𖼅𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 : 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼁𖽠𖾏 , 𖽐𖼊𖽙𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽠𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 . 6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 , 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 .
7 “𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼅𖽡 𖼐𖽔𖾑 , 𖼺𖽹𖾑 𖼞𖽑𖽡𖽠𖾏 𖼅𖽡 , 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼯𖽠 . 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼮𖽑𖽠𖾏 𖼅𖽡 , 𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼗𖽪𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼐𖽑𖽡𖾏 , 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼎𖽑𖽻𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼅𖽡 𖼀𖽜𖾐 𖼛𖽡𖾏 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 . 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 𖼺𖽘𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 , 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 , 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽑𖽡𖽝𖾐 𖼷𖽑𖽷𖾏 . 10 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 . 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼳𖽪𖾐 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽳𖾑𖽑 𖼮𖽡𖽝𖾏 . 11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏、𖼼𖽘𖾏 𖼻𖽡𖽝𖾑、𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡 .”
𖼻𖽙𖾑 𖼸𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
（𖼎𖽺𖾏 28：1 - 14）
12 “𖼅𖽡 𖼮𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑 , 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼻𖽙𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼺𖽘𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 . 13 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 , 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 , 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 , 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼡𖽑𖽙𖾑 𖼘𖽑𖽘𖾑 𖼾𖽨 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖼋𖽠𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼡𖽻𖾐 𖼖𖽻𖾏 、 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖽐𖼎𖽙 𖼛𖽙𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 , 𖼾𖽘𖾏 𖼁𖽜𖾏 . 14 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼅𖽡 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔 𖼾𖽠 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏 , 𖼯𖽦𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼨𖽑𖽳𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 . 15 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼇𖽳𖾑 𖼟𖽡 𖼺𖽘𖾏 , 𖼣𖽹 𖼮𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼄𖽙𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐 𖽃𖽡𖾑 , 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼮𖽑𖽨𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼮𖽔𖾑 𖼮𖽑𖽨𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼟𖽡 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼊𖽔𖾐 . 16 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 , 𖼟𖽡 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 , 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼟𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼸𖽙𖾏 .
