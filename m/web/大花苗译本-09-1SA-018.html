𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖽐𖼮𖽑𖽠𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏
18
𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼗𖽪𖾏 𖼺𖽘𖾏, 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾏 𖼺𖽱𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏. 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖽐𖼮𖽑𖽠𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼺𖽱𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑. 2 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼺𖽻𖾐 𖼘𖽙 𖼞𖽑𖽻𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼞𖽔𖾏. 3 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖽐𖼮𖽑𖽠𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑. 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏. 4 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼚𖽠𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏、𖽃𖽔𖾑 𖼏𖽦𖾏、𖼊𖽪𖾑 𖼐𖽜𖾑𖽑、𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏. 5 𖼺𖽻𖾐 𖼘𖽙 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏, 𖼐𖽱𖾏 𖼊𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐. 𖼺𖽻𖾐 𖼘𖽙 𖼺𖽹𖾑 𖼮𖽘𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽱𖾏.
𖼺𖽻𖾐 𖼘𖽙 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏
6 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼊𖽻𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼷𖽷𖾑 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽷𖾐 𖼗𖽙, 𖽃𖽔 𖼁𖽙𖾏 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖽃𖽺𖾐 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖽐𖼊𖽻𖾏 𖽐𖼎𖽔𖾏, 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏, 𖼙𖽔 𖼷𖽹𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏. 7 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔 𖼁𖽙𖾏 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏, 𖼐𖽑𖽡𖾏:
“𖼺𖽻𖾐 𖼘𖽙 𖼊𖽔𖾏 𖼋𖽔 𖼷𖽑𖽡𖽝𖾐 𖼷𖽑𖽡𖽝𖾐,
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽔𖾏 𖼋𖽔 𖼉𖽻𖾏 𖼉𖽻𖾏.”
8 𖼺𖽻𖾐 𖼘𖽙 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼦𖽡𖾐 𖼙𖽔 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑, 𖼐𖽑𖽡𖾏: “𖼄𖽔𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼄𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 𖼷𖽑𖽡𖽝𖾐 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼯𖽔 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽜𖾑 𖼋𖽘.” 9 𖼎𖽑𖽻𖾏 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼳𖽠𖾑 𖼁𖽱𖾏, 𖼺𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏.
10 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐, 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽃𖽔𖾏 𖼮𖽡𖾑, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼀𖽪 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽑𖽡𖾏. 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏, 𖼺𖽻𖾐 𖼘𖽙 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼄𖽑𖽪𖾑. 11 𖼺𖽻𖾐 𖼘𖽙 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖼄𖽑𖽪𖾑 𖼇𖽜𖾏 𖼇𖽜𖾏 𖽃𖽡𖾐 𖼗𖽔𖾏, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽙𖾏 𖼊𖽙𖾏, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼮𖽦𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼮𖽑𖽤. 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼀𖽜𖾑 𖽐𖼮𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼟𖽻𖾐. 12 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼙𖽠𖾏, 𖼇𖽳𖾑 𖼺𖽻𖾐 𖼘𖽙 𖼺𖽘𖾏 𖼋𖽘𖾏. 13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽻𖾐 𖼘𖽙 𖼟𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼾𖽨𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼏𖽙 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼁𖽺𖾏 𖼗𖽙𖾏. 14 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽑𖽡𖽝𖾐. 15 𖼺𖽻𖾐 𖼘𖽙 𖼁𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏. 16 𖼈𖽡𖽝𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖽐𖼮𖽑𖽠𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼁𖽺𖾏 𖼗𖽙𖾏.
𖼺𖽻𖾐 𖼘𖽙 𖼏𖽵 𖼄𖽡𖾐 𖼖𖽔𖾑 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏
