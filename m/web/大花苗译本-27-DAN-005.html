𖼀𖽳 𖼳𖽔𖾑 𖼺𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏, 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽡𖾑 𖼼𖽻𖾏 𖼊𖽱𖾐 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑. 2 𖼀𖽳 𖼳𖽔𖾑 𖼺𖽔𖾐 𖽃𖽺𖾐 𖼙𖽔 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼞𖽪𖾐 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 (“𖼉𖽺𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐”) 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼋𖽔𖾐, 𖼻𖽻𖾏 𖼟𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐、 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖾏 𖼣𖽔𖾏 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏、 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑. 3 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏, 𖼟𖽪𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 𖼯𖽡𖽙𖾏 𖼋𖽔𖾐, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐、 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖾏 𖼣𖽔𖾏 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑. 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐、 𖼑𖽡𖽝𖾏、 𖼋𖽻𖾏、 𖼗𖽻𖾏、 𖼋𖽠𖾏、 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽔𖾏 𖽃𖽡𖾑.
