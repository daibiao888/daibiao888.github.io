（𖼞𖽑𖽙𖾐 3：13-19；𖼖𖽪𖾏 6：12-16）
10
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽠𖾏 𖼯𖽵𖾏 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏，𖼎𖽺𖾏 𖼸𖽠𖾏 𖼞𖽑𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐． 2 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼄𖽞𖼎𖽺𖾏 𖽐𖼷𖽷𖾏 𖼀𖽡𖾐 𖼊𖽳，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼊𖽳 𖼖𖽡𖽝， 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼊𖽑𖽹𖾏 𖼊𖽪𖾐𖼽𖽔𖾐 𖼞𖽙𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑𖼽𖽙 𖼦𖽗， 3 𖼇𖽺𖾑 𖼖𖽡 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼊𖽙𖾑 𖼘𖽙 𖼄𖽹𖾐， 𖼊𖽙𖾑 𖼄𖽔𖾐， 𖽐𖼎𖽙 𖼄𖽔𖾐 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏， 𖼽𖽔𖾏 𖼖𖽳 𖼇𖽺𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙𖽐𖼎𖽙 𖼊𖽔 𖼊𖽑𖽹𖾏 ， 4 𖼇𖽞𖾏 𖼵𖽪𖽺𖾏 𖼊𖽗𖾐（“𖼇𖽞𖾏 𖼵𖽪𖽺𖾏 𖼊𖽗𖾐 ” 𖽃𖽡𖾐 𖼸𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼘𖽙 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏） 𖼳𖽡𖾑 𖼄𖽞 ，𖽐𖼎𖽙 𖼮𖽢𖾑 𖼖𖽡𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽨 𖼊𖽔𖾏，𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼷𖽷𖾐 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼅𖽻𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏
（𖼞𖽑𖽙𖾐 6：7-13；𖼖𖽪𖾏 9：1-6）
5 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼷𖽷𖾐 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏，𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 ，𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 ，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽻， 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏， 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼙𖽠𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 7 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼡𖽑𖽔𖾐 𖼷𖽷𖾏 𖼖𖽘𖾑，𖼐𖽑𖽡𖾏 ：‘ 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏！’ 8 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐，𖼽𖽔𖾑 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽙𖾏 𖼨𖽑𖽪𖾑 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏， 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏．𖼅𖽡 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼊𖽻𖾏 𖼗𖽙𖾏 ， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏． 9 𖼅𖽡 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾐、𖼑𖽡𖽝𖾏、𖼋𖽻𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑， 10 𖼅𖽻𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼮𖽑𖽙𖾏 𖼄𖽡𖾐 𖼋𖽱𖾐，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼞𖽑𖽻𖾏 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼊𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼯𖽡𖽙𖾏． 11 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼡𖽔𖾏 𖼳𖽷𖾏 ， 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼼𖽙𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽪 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏， 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼅𖽻𖾏． 12 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪． 13 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 ， 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏， 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏，𖼋𖽡𖽝𖾏 𖼋𖽪 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽙𖾑 𖼛𖽔𖾏 𖼅𖽡． 14 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼅𖽡， 𖼦𖽡𖾐 𖼑𖽻 𖼅𖽡 𖼗𖽪𖾏，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼇𖽳𖾑 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 ， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼅𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽘𖾏． 15 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ， 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽃𖽡𖾑， 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖽐𖼎𖽙 𖽃𖽳 𖼄𖽙 𖼖𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 ，𖼳𖽷𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 ．”
𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔

16 “𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼅𖽡 𖼅𖽻𖾏，𖼺𖽱𖾏 𖼺𖽱𖾏 𖼾𖽘𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼀𖽙𖾏 𖼄𖽑𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼚𖽠𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐，𖼻𖽻𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐． 17 𖼅𖽡 𖼽𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼑𖽪𖾏 𖼨𖽑𖽢𖾐 𖼅𖽡，𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽐𖼊𖽻𖾏 𖼅𖽡． 18 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽔𖾐，𖼽𖽔𖾑 𖼊𖽻𖾏 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽡𖾑． 19 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼅𖽡 𖼺𖽘𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑，𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼊𖽳𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏．𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏， 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡． 20 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏，𖼟𖽪𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼑𖽡𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼐𖽡𖽙𖾐 𖼅𖽡 𖽐𖼚𖽙 𖼐𖽑𖽡𖾏． 21 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏，𖼅𖽔𖾏 𖼟𖽪𖾏 𖼉𖽺𖾏 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼊𖽪𖾐 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽔，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐，𖼟𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔． 22 𖼅𖽡 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼳𖽪𖾐 𖼅𖽡．𖼮𖽔 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼷𖽜𖾏 𖼸𖽙 𖼡𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏． 23 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽡𖾏 𖼅𖽡， 𖼅𖽡 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑． 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝𖼙𖽻 𖼅𖽡 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼺𖽷𖾐，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏． ”
𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏
24 𖼊𖽪𖾐 𖼟𖽡𖽠𖾏 𖽐𖼊𖽠𖾑 𖼦𖽡𖾐 𖼺𖽡𖽝𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏，𖼡𖽑𖽜𖾑 𖼦𖽡𖾐 𖼺𖽡𖽝𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏 𖼮𖽵𖾑． 25 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼟𖽡𖽠𖾏 𖽐𖼊𖽠𖾑 𖼡𖽑𖽙𖾑 𖽐𖼮𖽷𖾐 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 ， 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽜𖾑 𖼡𖽑𖽙𖾑 𖽐𖼮𖽷𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐，𖼋𖽡𖾏 𖼞𖽡𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼋𖽘𖾏．𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖽃𖽔𖾏 ‘ 𖼀𖽡𖽝 𖼳𖽡𖾑 𖼀𖽑𖽪 ’（“𖼀𖽡𖽝 𖼳𖽡𖾑 𖼀𖽑𖽪 ”𖼟𖽪𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏），𖼐𖽱𖾏 𖼽𖽔𖾑 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼖𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏．”
26 “ 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼮𖽡𖽝𖾏，𖼟𖽪𖾏 𖼦𖽡𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐． 27 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼅𖽡 𖼐𖽡𖽙𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽑𖽷𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖼏𖽪𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏． 28 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ， 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽙𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐，𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼛𖽡𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼖𖽻𖾐 𖼎𖽻𖾏 𖽐𖼷𖽷𖾏 𖼞𖽡 𖼑𖽡𖽻 𖽃𖽡𖾑． 29 𖽃𖽔𖾐 𖼋𖽪 𖼀𖽡𖾏 𖼮𖽑𖽡𖽙𖾑， 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼋𖽺 𖽃𖽡𖾐 𖼇𖽺𖾏 𖼑𖽡𖽝𖾏 𖼘𖽪？ 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏，𖽃𖽡𖾐 𖼋𖽪 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐． 30 𖼟𖽪𖾏 𖼅𖽡 𖼚𖽻𖾐 𖼇𖽻𖾏，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖼙𖽔 𖼾𖽱 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏． 31 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼀𖽡𖾏 𖼮𖽑𖽡𖽙𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐．”
𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼯𖽜𖾏 𖼮𖽡𖾑 𖼊𖽪
（𖼖𖽪𖾏 12：8-9）
32 “ 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼯𖽜𖾏 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽜𖾏 𖼐𖽱𖾏， 33 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼐𖽱𖾏． ”
𖼎𖽷𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐
（𖼖𖽪𖾏 12：51-53；14：26-27）
34 “ 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼗𖽙𖾏， 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐，𖼞𖽪𖾑 𖼗𖽙𖾏， 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏，𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏． 35 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼗𖽙𖾏，
𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐，
𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐 ，
𖼟𖽪𖾏 𖼐𖽤𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼁𖽙𖾐 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐，
36 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．
37 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑， 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏； 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑，𖼦𖽡𖾐 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 ． 38 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼎𖽷𖾏 𖼾𖽨𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖽃𖽡𖾑，𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏． 39 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑，𖼽𖽔𖾑 𖼛𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼛𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏，𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑．”
𖼽𖽔𖾑 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
（𖼞𖽑𖽙𖾐 9：41）
40 “ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼯𖽡𖽙𖾏 𖼅𖽡，𖼟𖽪𖾏 𖼷𖽹𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏．𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑，𖼟𖽪𖾏 𖼷𖽹𖾏 𖼯𖽡𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼋𖽔𖾏． 41 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽐𖼷𖽷𖾏， 𖼷𖽹𖾏 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑，𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏， 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽐𖼷𖽷𖾏，𖼷𖽹𖾏 𖼯𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ，𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏． 42 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼡𖽔 𖼋𖽱𖾏 𖼑𖽜 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼷𖽷𖾏， 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼞𖽑𖽙𖾐 𖼈𖽳𖾐 𖽃𖽻𖾐 𖼸𖽡𖽝 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼦𖽻𖾏，𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ，𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏．”
