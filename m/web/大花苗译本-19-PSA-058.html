𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼘𖽪！ 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏, 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼘𖽪？
𖼅𖽡 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼸𖽙𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼘𖽪?
2 𖼦𖽡𖾐 𖼾𖽙！ 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼅𖽡 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼅𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑 𖽐𖼚𖽪𖾏 𖼸𖽳 .
 
3 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽂𖽪𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼷𖽹𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽔𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽂𖽪𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼷𖽹𖾏 𖼅𖽻𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼞𖽡𖾑 .
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽙𖾏 𖼋𖽻𖾏 𖼚𖽠𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖼑𖽡𖾏 𖼋𖽙𖾏 𖼋𖽻𖾏 𖼚𖽠𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖼡𖽔𖾐 𖼗𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖽐𖼮𖽠𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼮𖽡𖽝𖾏，
5 𖼦𖽡𖾐 𖼑𖽻 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐,
𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼸𖽪𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏, 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼑𖽻 .
 
6 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼮𖽻𖾏 𖽐𖼮𖽻 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼊𖽪𖾑 𖼐𖽑𖽡𖽝𖾑；
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽻𖾏 𖼺𖽘𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼟𖽻𖾏．
7 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖽐𖼚𖽱𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽘𖾏 𖼐𖽑𖽜𖾑 𖼮𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼈𖽙𖾏 𖼺𖽱𖾏 𖼊𖽻𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏.
8 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼷𖽷 𖼢𖽻𖾏 𖼡𖽳𖾏 𖼾𖽘𖾏 𖼺𖽘𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐.
9 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼚𖽻𖾑 𖼋𖽠𖾏, 𖼖𖽪𖾑 𖼀𖽑𖽻𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼳𖽙𖾑 𖼺𖽷𖾐,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼷𖽑𖽔𖾐 𖼙𖽠𖾏.
 
10 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼎𖽺𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑 𖽐𖼷𖽔𖾑 𖼊𖽠𖾏.
11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ,
𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼑𖽪𖾏．”
