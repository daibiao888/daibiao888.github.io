40 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏,

43 𖼐𖽱𖾏 𖼯𖽪 𖼯𖽪𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡 𖼋𖽙 𖼙𖽠𖾏.
44 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽱𖾏 𖼐𖽑𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼐𖽡𖽝𖾑 𖽐𖼊𖽜 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,

𖼸𖽙𖾏 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼷𖽻𖾏 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.

48 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 ,
