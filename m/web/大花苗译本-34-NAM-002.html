𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼘𖽪 ! 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖽃𖽡𖾑 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼟𖽡 ,
𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 , 𖼻𖽙𖾑 𖼞𖽡𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏,
𖼟𖽙𖾏 𖼖𖽪𖾑 𖼚𖽔𖾑 𖽃𖽺𖾐 𖼋𖽻𖾏, 𖼮𖽡𖽻𖾐 𖼮𖽡𖽻𖾐 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏.
 
2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 ,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖽃𖽡𖾑, 𖼙𖽔 𖼟𖽙𖾏 𖼽𖽔𖾑 𖼞𖽙 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 ,
𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏.
3 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐,
𖼊𖽪𖾐 𖼏𖽙𖾏 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐,
𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑,
𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼋𖽻𖾏 𖼖𖽑𖽻𖾏 𖼮𖽡𖾏 𖼙𖽪 𖼙𖽪 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏,
𖽃𖽔𖾏 𖼞𖽙𖾏 𖼄𖽑𖽪𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽑𖽪𖾑 𖽃𖽔𖾏 𖼖𖽔𖾏 𖼇𖽜𖾏 𖼇𖽜𖾏 𖼮𖽘𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏.
4 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼏𖽘 𖼷𖽹𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖼏𖽘 𖼅𖽻𖾏 𖼏𖽘 𖼗𖽙𖾏 ,
𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼺𖽱𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾐,
𖼏𖽘 𖼇𖽺𖾏 𖼺𖽱𖾏 𖼺𖽙𖾐 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼯𖽦𖾏 .
5 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼛𖽡𖾐 𖼦𖽡𖾏 𖼛𖽠𖾏,
𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼇𖽺𖾏,
𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
6 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼡𖽙𖾏 𖼖𖽪𖾑 𖽃𖽔𖾑 𖼟𖽱𖾐 𖼑𖽡𖾏 𖽃𖽻𖾐 ,
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼖𖽔𖾏 𖽐𖼊𖽱𖾑𖽑 𖼈𖽳𖾑 𖼺𖽘𖾏.
7 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽔𖾏 𖼙𖽠𖾏.
𖼖𖽪𖾑 𖼟𖽻 𖽃𖽡𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼀𖽔𖾏 𖽐𖼎𖽙𖾐 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 𖼺𖽱𖾏 𖽐𖼡𖽔𖾐.
𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏.
8 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼗𖽙𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼺𖽱𖾏 𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼦𖽡 𖼸𖽱 ,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼞𖽑𖽔𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼳𖽠𖾑 𖽐𖼎𖽙𖾏! 𖼳𖽠𖾑 𖽐𖼎𖽙𖾏 !”
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼑𖽔𖾏 𖼋𖽘𖾏.
9 𖼅𖽡 𖼦𖽔𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼦𖽡𖾑！
𖼅𖽡 𖼦𖽔𖾏 𖼞𖽪𖾐 𖼦𖽡𖾑！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼮𖽷𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐,
𖼻𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐.
 
10 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑、 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏、 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏, 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼇𖽻𖾑 𖼯𖽡𖽻 𖼦𖽡𖾏 𖽐𖼊𖽑𖽱𖾏,
𖽃𖽡𖾐 𖼮𖽡𖾏 𖼙𖽱 𖼄𖽙𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖼚𖽑𖽪𖾏 𖼞𖽑𖽔𖾏 𖼚𖽑𖽡𖾏 𖼼𖽘𖾏.
11 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼾𖽨 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼡𖽔𖾐 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐、 𖽃𖽔𖾐 𖼑𖽡𖽝 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐、 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽃𖽔𖾏 𖼇𖽜𖾏,
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼘𖽪?
12 𖼡𖽔𖾐 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼑𖽜 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼑𖽜 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼋𖽙𖾏 𖼋𖽔 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏,
𖼞𖽑𖽡𖽠𖾏 𖽐𖼡𖽹𖾏 𖼻𖽻𖾏 𖼸𖽔 𖼷𖽑𖽠𖾏 𖼀𖽪𖾑 𖼐𖽱𖾏 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑、
𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼐𖽱𖾏 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑 𖼖𖽘𖾑 .
13 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼚𖽑𖽡𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏, 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼡𖽙𖾏 𖼖𖽻𖾐 𖼟𖽡 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼟𖽡 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏, 𖼟𖽡 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏.”
