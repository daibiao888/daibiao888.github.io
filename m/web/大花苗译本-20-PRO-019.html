19
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼡𖽔 𖼏𖽔 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑.
2 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽙𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 , 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾏 𖼡𖽔𖾑 𖼏𖽔𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏 , 𖼯𖽠 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏.
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 , 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
4 𖼅𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼯𖽡𖾏 𖼽𖽺 𖼗𖽔 𖼮𖽑𖽨𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼯𖽡𖾏 𖼽𖽺 𖼗𖽔 𖼀𖽜𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .
5 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐；
𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼀𖽜𖾑 𖼦𖽡𖾐 𖼛𖽡𖾏.
6 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼊𖽜 𖼅𖽳𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽪𖾏 𖼺𖽘𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖽐𖼡𖽔 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 .
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏, 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼊𖽪𖾏 𖼳𖽪𖾐 𖼐𖽱𖾏 ,
𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼀𖽜𖾑 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼚𖽡𖾐,
𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼻𖽻𖾏 𖼗𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼀𖽜𖾑 𖼙𖽠𖾏.
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖽐𖼮𖽑𖽠𖾏 𖼮𖽑𖽔𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼀𖽻𖾐 𖼊𖽘𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼮𖽑𖽵𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐；
𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 .
10 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼟𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼦𖽡 𖽐𖼷𖽷；
𖽃𖽔𖾏 𖼐𖽡𖾑， 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼯𖽡𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏?
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼇𖽘𖾑 𖼦𖽡𖾐 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐；
𖼯𖽔𖾏 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽠𖾏, 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .
12 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖽃𖽡𖾑 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏,
𖼐𖽱𖾏 𖼅𖽳𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼙𖽱 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔 𖽐𖼚𖽻𖾏 𖽐𖼡𖽜𖾐.
13 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 , 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼐𖽤𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼋𖽻𖾏, 𖼺𖽱𖾏 𖼑𖽘 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏 .
14 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾑 𖼀𖽑𖽱𖾐 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼎𖽻𖾏；
𖼮𖽔 𖼅𖽔𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼎𖽻𖾏 .
15 𖽐𖼞𖽜𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽱𖾏 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽑𖽹𖾐 𖽃𖽔𖾑 𖼚𖽘𖾐.
16 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 .
17 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫；
𖼐𖽱𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏.
18 𖼷𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼺𖽷𖾐, 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼟𖽡 𖼊𖽪𖾐,
𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼛𖽙𖾏 𖼋𖽔 .
19 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐,
𖼟𖽡 𖼮𖽔𖾑 𖼮𖽨𖾏 𖼐𖽱𖾏, 𖼯𖽡𖾏 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼟𖽻𖾐 𖼮𖽨𖾏 𖼐𖽱𖾏.
20 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐, 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖼟𖽙𖾏 𖼟𖽡 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐,
𖼮𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏.
22 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 ，𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽳𖾐 𖼉𖽳；
𖼊𖽪𖾐 𖼳𖽻𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑.
23 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑；
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐, 𖼦𖽡𖾐 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 .
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼚𖽙 𖼋𖽺𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼯𖽡𖽙𖾏 𖼗𖽙 𖼎𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼊𖽜𖾑 𖼊𖽪𖾏 𖽐𖼞𖽜𖾏 𖼞𖽑𖽡𖽠𖾏.
