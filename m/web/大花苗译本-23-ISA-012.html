𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏
12
𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡．
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏,
𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼾𖽘𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼟𖽡 𖼎𖽺𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏.
 
2 𖼑𖽔𖾏 𖼘𖽪! 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼮𖽑𖽹𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼟𖽪𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏.
 
3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖽃𖽺𖾐 𖼙𖽔 𖼊𖽻𖾏 𖽃𖽻𖾐.”
4 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏:
“𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏,
𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐.
 
5 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑.
6 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼘𖽪 ! 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .”
