𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼘𖽪!
𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖽐𖼷𖽷𖾏,
2-3 𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼷𖽙𖾏 𖼺𖽙𖾑 ,

𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏,
𖼷𖽑𖽔𖾏 𖼄𖽑𖽙𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝．
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏.
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼚𖽙𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏.
 
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼋𖽡𖽝，
𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐 𖼋𖽡𖽝 .
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼀𖽻𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 .
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼘𖽑𖽘𖾑 𖼻𖽻𖾏 𖼺𖽱𖾏 𖽐𖼡𖽜𖾐,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏, 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏 ,
𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
8 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐,
𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
 
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼑𖽔𖾏 𖼘𖽪！ 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖼑𖽔𖾏 𖼘𖽪！ 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏.
10 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼞𖽪𖾐, 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼑𖽨𖾏 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐,
𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼚𖽜𖾑 𖼋𖽘𖾏.
11 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏,
𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐.
 
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼺𖽱𖾐 𖼺𖽱𖾐,

13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽙 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽐𖼚𖽙 ,

14 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏, 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐,
