10
𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡：“ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼾𖽘𖾐，𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏，𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽡𖾏 𖽐𖼷𖽜𖾑 𖼅𖽻𖾏，𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽜，𖼟𖽪𖾏 𖼺𖽷𖾑 𖼈𖽳𖾐 𖼺𖽷𖾏 𖼸𖽜． 2 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏，𖼟𖽪𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏． 3 𖼊𖽪𖾐 𖼻𖽙𖾑 𖽃𖽔 𖼛𖽻𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏，𖼾𖽘𖾏 𖼎𖽺𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼾𖽘𖾏 𖽐𖼷𖽷𖾏，𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼾𖽘𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏． 4 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼾𖽘𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏，𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼑𖽻 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏，𖼎𖽺𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏． 5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼑𖽡𖽻𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼅𖽻𖾏， 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼀𖽜𖾑 𖼙𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼑𖽡𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏．” 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏．
7 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼟𖽪𖾏 𖼾𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏． 8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼸𖽜，𖼟𖽪𖾏 𖼺𖽷𖾑 𖼈𖽳𖾐 𖼺𖽷𖾏 𖼸𖽜， 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼦𖽡𖾐 𖼑𖽻 𖼐𖽱𖾏 𖼸𖽻𖾏． 9 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼛𖽻𖾏，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼊𖽻𖾏 𖽐𖼡𖽜𖾐 𖼑𖽻𖾏． 10 𖼸𖽜 𖼋𖽔𖾏，𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽡𖽝、 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 、𖼽𖽔𖾑 𖼖𖽻𖾐；𖼞𖽪𖾑 𖼋𖽔𖾏， 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑，𖼎𖽺𖾏 𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽱𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝．
11 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏，𖼻𖽻𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼑𖽜 𖼾𖽘𖾏 𖼗𖽔𖾏 𖼳𖽦𖾑． 12 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏，𖼦𖽡𖾐 𖼾𖽙 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏，𖼾𖽘𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖼁𖽡𖽝，𖼐𖽱𖾏 𖼁𖽙𖾏 𖼄𖽑𖽘𖾏 𖼋𖽔𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼾𖽘𖾏 𖼺𖽘𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼙𖽠𖾏，𖼊𖽳𖾑 𖼘𖽳𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼾𖽘𖾏，𖼖𖽠𖾑 𖼟𖽙𖾏 𖼾𖽘𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏． 13 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼙𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏 𖼊𖽔𖾐，𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽻𖾐 𖼾𖽘𖾏． 14 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏，𖼞𖽪𖾑 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼾𖽘𖾏，𖼞𖽪𖾑 𖼾𖽘𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑． 15 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼉𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼉𖽺𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼾𖽘𖾏 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑． 16 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼾𖽘𖾏，𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼾𖽘𖾐 𖼐𖽡𖾑，𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖽐𖼀𖽙𖾐，𖼺𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏． 17 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼞𖽪𖾑 𖼉𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑 𖼺𖽘𖾏，𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙． 18 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼦𖽔𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼙𖽠𖾏， 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏；𖼞𖽪𖾑 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼞𖽑𖽡𖽠𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼗𖽙．𖼋𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑．”
