𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼷𖽷𖾐, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼞𖽡𖾑. 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖽃𖽡𖾑, 𖼇𖽺𖾏 𖼽𖽔𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏.”
𖼮𖽨𖾏 𖼮𖽵𖾑 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 𖼢𖽳 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏
2 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼋𖽻𖾏 𖼖𖽺𖾑? 𖼐𖽱𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼋𖽻 𖼖𖽺𖾑? 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼷𖽑𖽷𖾑 𖼮𖽡𖽺𖾏 𖼞𖽪𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽑𖽔𖾏 𖽐𖼊𖽻𖾐. 3 𖼐𖽱𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼮𖽡𖽺𖾏 𖼑𖽡𖽝𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑, 𖼽𖽔𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽐𖼮𖽔𖾏 𖼮𖽡𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 4 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼷𖽹𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏.
𖼖𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼷𖽜𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼇𖽳𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏
5 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼅𖽡, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐． 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏, 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖼐𖽜𖾐, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏, 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖽐𖼡𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼡𖽜𖾏 𖼼𖽙, 𖼸𖽺𖾏 𖼯𖽵𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏, 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐, 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑. 6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏.
𖼖𖽑𖽘𖾑 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠
7 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖼎𖽑𖽻𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏, 𖼅𖽡 𖼯𖽪 𖼯𖽪𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏. 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼅𖽡, 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼑𖽪 𖼞𖽪𖾑, 𖼀𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼖𖽺𖾑? 8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼦𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽡𖽝 𖽃𖽡𖾑 𖼘𖽪? 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑． 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏: 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼦𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼖𖽺𖾑? 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽐𖼀𖽠𖾏 𖽃𖽡𖾑 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏, 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏. 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽡𖾐 𖼞𖽪𖽙 𖼙𖽱 𖼙𖽱 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼞𖽪𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑, 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼎𖽺𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑. 10 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼺𖽘𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼼𖽪, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐, 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖽐𖼮𖽔𖾏 𖼞𖽪𖾑, 𖼋𖽡𖽝𖾏 𖼞𖽪𖾑 𖼸𖽠𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽜 𖼅𖽡 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏 𖽐𖼮𖽢𖾑𖽑 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼋𖽻𖾏. 11 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼅𖽡 𖼖𖽔𖾏 𖼷𖽷𖾑 𖼷𖽷( “𖼷𖽷𖾑 𖼷𖽷”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 ”), 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼟𖽡𖾏 𖼅𖽡 𖼡𖽻𖾐 𖼖𖽻𖾏． 𖼅𖽡 𖽐𖼎𖽹𖾏 𖼙𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼳𖽦𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼛𖽡𖾏 𖼀𖽻𖾐 𖼙𖽠𖾏. 12 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑.”
