10 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼀𖽪𖾑 𖼋𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼅𖽡 , 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼅𖽳𖾏, 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼺𖽷𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑. 11 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼅𖽔𖾏 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼦𖽡𖾐 𖼾𖽙 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼖𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏.𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 12 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼞𖽪𖾑 , 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼅𖽡. 13 𖼅𖽡 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑, 𖼮𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏. 14 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖼞𖽪𖾑, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 , 𖼄𖽔𖾐 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼅𖽡 𖼚𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼗𖽙𖾏. 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖽃𖽡𖾑. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
