115
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼀𖽡𖾐,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼀𖽡𖾐,
𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏！
 
2 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼯𖽪𖾏 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏:
“𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?”
3 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼟𖽪𖾏 𖼸𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽺𖾐 𖼙𖽔 𖽃𖽡𖾑 𖽃𖽔𖾏.
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐、𖼑𖽡𖽝𖾏,
𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏，
5 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼅𖽔𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽔𖾏，
6 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽻 ,
𖼅𖽔𖾏 𖽃𖽔𖾑 𖽐𖼀𖽱 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽘𖾐，
7 𖼅𖽔𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏,
𖼅𖽔𖾏 𖼊𖽠𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼅𖽻 ,
𖼅𖽔𖾏 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐.
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼁𖽠𖾏 𖼖𖽔𖾏 𖼺𖽱𖾏 𖼁𖽠𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼋𖽡𖾏 𖼁𖽠𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.
 
9 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫；
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼅𖽡 (“𖼅𖽡”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼐𖽱𖾏 𖼸𖽻𖾏”), 𖼟𖽪𖾏 𖼅𖽡 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐.
10 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼾𖽡𖾏 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼅𖽡 , 𖼟𖽪𖾏 𖼅𖽡 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐.
11 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼅𖽡 , 𖼟𖽪𖾏 𖼅𖽡 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐.
 
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾐 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼀𖽡𖾐,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐,
𖼽𖽔𖾑 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏,
𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼾𖽡𖾏.
13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼘𖽑𖽙𖾐 𖼳𖽻𖾐,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
 
14 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼅𖽡 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼟𖽙𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼅𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
15 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏！
16 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼊𖽡𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐.
17 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽦 𖽐𖼊𖽪𖾐 𖼊𖽙𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏.
18 𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼎𖽑𖽻𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
 
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫!
