𖼇𖽔 𖼘𖽻𖾐 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼮𖽢𖾑 𖼺𖽔𖾐 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐. 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“ 𖼑𖽔𖾏 𖼘𖽪𖾏 ! 𖼅𖽔𖾏 𖽃𖽻𖾐 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖽐𖼚𖽔𖾏 𖼋𖽔𖾏, 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙 ,
𖼽𖽔𖾑 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏 𖼮𖽑𖽡𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑,
𖼊𖽑𖽡𖽝𖾐 𖼙𖽻 𖽐𖼎𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼙𖽻 𖽃𖽡𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏.
3 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼏𖽘 𖼑𖽡𖾏 𖼳𖽘𖾐,
𖽐𖼎𖽙 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼏𖽘 𖼟𖽡𖾏 𖼨𖽑𖽪 𖼨𖽑𖽪 , 𖼀𖽡𖾑 𖼘𖽑𖽙𖾐 𖼦𖽡 𖽐𖼚𖽙 𖼟𖽡𖾏 𖼨𖽑𖽻 𖼨𖽑𖽻 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽡𖾏 𖼯𖽻 𖼼𖽙,
𖼦𖽡𖾐 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏.
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽔𖾐 𖽐𖼮𖽡𖽻𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏 ,
𖼋𖽠𖾏 𖼺𖽘𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙、𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖽃𖽡𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼟𖽪𖾏 𖼮𖽢𖾑 𖼇𖽺𖾑 𖼊𖽑𖽙 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑.
5 𖼮𖽢𖾑 𖼺𖽔𖾐 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼮𖽢𖾏 𖼺𖽘𖾏,
𖼽𖽔𖾏 𖼳𖽷 𖼮𖽡𖾑 𖼖𖽪𖽝 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏,
𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼘𖽪！
𖼟𖽡 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼷𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑, 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼖𖽺𖾑?
𖼟𖽡 𖼽𖽔𖾑 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼐𖽘𖾐 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼮𖽡𖽝𖾏, 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼺𖽻𖾑 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙.
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼽𖽔𖾏 𖼳𖽷 𖼮𖽡𖾑 𖼖𖽪𖽝 , 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑,
