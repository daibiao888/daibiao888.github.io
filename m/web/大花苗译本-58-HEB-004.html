𖽐𖼮𖽜 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏

𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼗𖽪𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏， 𖼯𖽪𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼺𖽷𖾐 𖼖𖽡𖾏 𖼯𖽡𖾏， 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐 𖽐𖼮𖽑𖽹𖾏， 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼊𖽻𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏． 2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼐𖽑𖽙𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜 𖽐𖼎𖽙 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼐𖽑𖽙𖾑 𖽃𖽡𖾑 𖼦𖽡 𖽐𖼷𖽔． 3 𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼋𖽘𖾏 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏：
“𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼏𖽵 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏：
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏！”
𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼼𖽙 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏， 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼁𖽱𖾏 𖼷𖽷𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼗𖽙𖾏 𖼸𖽙𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽑𖽔𖾏 𖼷𖽷𖾏 𖼻𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏． 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑， 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：“𖼐𖽡𖽙𖾐 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽙 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏．” 5 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏：“ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏．” 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽺𖾏 𖼅𖽻𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼄𖽻𖾏 𖼻𖽻𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏， 7 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼗𖽡𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽃𖽔𖾑 𖼡𖽜𖾐 ，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼊𖽡𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐， 𖼡𖽜𖾑 𖼷𖽹𖾏𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼊𖽠𖾑 𖼐𖽑𖽡𖾏 𖽃𖽔𖾏 “𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 ”， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑：
“ 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 ，𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼗𖽪𖾏，
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏．”
8 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼋𖽘𖾏， 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽡𖾏 𖼏𖽵 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼋𖽘𖾏． 9 𖼑𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼗𖽙𖾏， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼋𖽡𖽝 𖼊𖽔𖾑 𖼋𖽡𖽝 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏， 𖼟𖽪𖾏 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏． 10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 ， 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼼𖽙 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼼𖽙 𖽃𖽡𖾐 𖼼𖽘𖾐． 11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽃𖽡𖾑， 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽻 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼸𖽙𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼢𖽻 𖼙𖽠𖾏．
