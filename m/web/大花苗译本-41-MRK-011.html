𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐
（𖼊𖽑𖽹𖾏 21：1-11；𖼖𖽪𖾏 19：28-40；𖼽𖽙 12：12-19）
11
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐， 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼀𖽳 𖼇𖽔 𖼮𖽑𖽡 𖽐𖼎𖽙 𖼀𖽳 𖼊𖽔𖾏 𖼐𖽡，𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽗𖾐 𖼖𖽗 𖼇𖽻𖾑 𖼎𖽻𖾐 ， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 ， 2 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼼𖽙𖾏 𖼀𖽡 𖼋𖽡𖾏， 𖽂𖽪𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼮𖽡𖽝𖾏， 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼙𖽔 𖼯𖽡𖽜𖾏 𖼺𖽷𖾐， 𖼅𖽡 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼎𖽺𖾏 𖼮𖽤𖾐 𖼋𖽔𖾏． 3 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏：‘𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑？’ 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏：‘ 𖼮𖽵𖾑 𖼖𖽔𖾑 𖼽𖽔𖾑 𖼏𖽻 𖼐𖽱𖾏．𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖼸𖽠𖾏 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼮𖽤𖾐 𖼐𖽱𖾏 𖼺𖽷𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑．’ ” 4 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖼎𖽺𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑， 𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖼢𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼚𖽠𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏． 5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼅𖽔𖾏 𖼟𖽪𖾏 𖼑𖽪 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼚𖽠𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑．” 6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽤𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏． 7 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼮𖽤𖾐 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼄𖽔𖾐 𖼾𖽨𖾏 𖼮𖽑𖽙𖾏 𖼀𖽔𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽜𖾏 𖼐𖽱𖾏． 8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼚𖽠𖾏 𖼮𖽑𖽙𖾏 𖼀𖽔𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽙𖾑 𖽃𖽔𖾑 𖽐𖼷𖽔 𖽐𖼊𖽻𖾏 𖼀𖽔𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑， 9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：
“ 𖼦𖽙 𖼺𖽔𖾐 𖼐𖽔！
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼮𖽵𖾑 𖽐𖼷𖽷𖾏 𖼋𖽔𖾏， 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏！
10 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔！
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼦𖽙 𖼺𖽔𖾐 𖼐𖽔！”
11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐， 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏， 𖼑𖽔𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏． 𖼖𖽔𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 ，𖼚𖽔𖾏 𖼀𖽳 𖼊𖽔𖾏 𖼐𖽡𖼙𖽠𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽔𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏
（𖼊𖽑𖽹𖾏 21：18-19）
12 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼀𖽳 𖼊𖽔𖾏 𖼐𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽹𖾐 𖼋𖽘． 13 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐， 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼚𖽻𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼯𖽔 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼚𖽻𖾏 𖼊𖽔𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑． 14 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽳 𖼟𖽻𖾏 𖼐𖽔𖾑，𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼖𖽪𖾐．” 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼖𖽔𖾏 𖼐𖽑𖽙𖾑 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏
（𖼊𖽑𖽹𖾏 21：12-17；𖼖𖽪𖾏 19：45-48；𖼽𖽙 2：13-22）
15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐． 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏，𖼎𖽺𖾏 𖼖𖽠𖾑 𖼊𖽪𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼋𖽺 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏，𖼎𖽺𖾏 𖼊𖽑𖽠𖾐 𖽐𖼷𖽜𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽪𖾑 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼑𖽡𖾏 𖼸𖽱𖾏 𖼞𖽡 𖼏𖽻𖾏， 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽺 𖽐𖼡𖽔𖾐 𖼑𖽡𖾏 𖼡𖽔𖾑 𖼋𖽙 𖼟𖽪𖾏 𖼼𖽻𖾏． 16 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼮𖽑𖽡𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏． 17 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：
‘ 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 ’ 𖼖𖽺𖾑？
𖼈𖽡𖽝𖾏 𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖽐𖼞𖽔𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼸𖽜 𖼺𖽘𖾏．”
18 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑， 𖼎𖽺𖾏 𖽐𖼊𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼛𖽙𖾏 𖼋𖽔， 𖼈𖽡𖽝𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏． 19 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼋𖽘𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼙𖽻 𖼙𖽠𖾏．
（𖼊𖽑𖽹𖾏 21：20-22）
20 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼎𖽺𖾏 𖼁𖽙𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽪𖾏 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏． 21 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 ， 𖼟𖽡 𖼑𖽔𖾏！ 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽔𖾏 𖽃𖽡𖾑 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏．” 22 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 23 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼮𖽑𖽹𖾏 𖼙𖽠𖾏 ，𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼚𖽙 𖼳𖽡𖽠𖾑 𖼙𖽠𖾏！’ 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼷𖽑𖽠𖾑 𖽐𖼊𖽱𖾏， 𖼮𖽔 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏， 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏， 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑． 24 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏， 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏． 25 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏， 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽔𖾑 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼅𖽡，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽔𖾏 𖼐𖽱𖾏， 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖼯𖽔𖾏 𖼅𖽡 𖼾𖽔𖾏． 26 𖼅𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼯𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽔𖾏， 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽔𖾏 𖼅𖽡 𖼾𖽔𖾏．”
𖼑𖽪 𖼡𖽔 𖼋𖽱𖾏 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 21：23-27；𖼖𖽪𖾏 20：1-8）
27 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐． 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙， 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼗𖽻 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐， 28 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼻𖽡𖽝𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑， 𖼡𖽔 𖼋𖽱𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼟𖽡，𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑？” 29 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽪 𖼅𖽡 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏， 𖼅𖽡 𖼊𖽡𖾐 𖼞𖽪𖾑， 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑． 30 𖼽𖽙 𖼦𖽗 𖼑𖽡𖾏 𖽐𖼷𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏 ， 𖼖𖽔𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏？ 𖼅𖽡 𖼊𖽡𖾐 𖼞𖽪𖾑．” 31 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼐𖽑𖽡𖾏：“ 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏 ：‘ 𖼋𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏’， 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏： ‘𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼐𖽱𖾏 ？ ’ 32 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏 ：‘𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 ’， 𖼀𖽡𖾐 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾏 𖼯𖽜𖾏 𖼽𖽙 𖼦𖽗 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐． 33 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼊𖽡𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏： “ 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑．”
