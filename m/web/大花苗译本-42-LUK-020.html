20
𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 ，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼡𖽑𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑， 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼗𖽻 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐， 2 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐，𖼟𖽡 𖼻𖽡𖽝𖾏 𖼎𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 ？ 𖼡𖽔 𖼋𖽱𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼟𖽡 ？” 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼑𖽪 𖼅𖽡 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏，𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑． 4 𖼽𖽙 𖼦𖽗 𖼐𖽱𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾑，𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏 𖼖𖽔𖾏 ？𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼘𖽪 ？ ” 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：‘ 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏 ’， 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏：‘𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼐𖽱𖾏 𖼖𖽺𖾑 ？’ 6 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：‘𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏’， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼋𖽔， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽜 𖼽𖽙 𖼦𖽗 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐．” 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏．” 8 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑．”
𖼊𖽪𖾐𖼖𖽑𖽘𖾏𖼊𖽪𖾐𖼀𖽻𖾐𖽐𖼊𖽠𖾏𖼊𖽪𖾐𖼗𖽻𖽐𖼮𖽔𖾏𖼑𖽪𖼽𖽦𖾑𖼺𖽪𖾑
𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽘𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏
（𖼊𖽑𖽹𖾏 21：33-46；𖼞𖽑𖽙𖾐 12：1-12）
9 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏，𖼡𖽑𖽠𖾑 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽘𖾏 𖽃𖽔𖾏，𖼎𖽺𖾏 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏． 10 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼐𖽱𖾏 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼖𖽪𖾑 𖼉𖽘𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽪𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖽃𖽔𖾏 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼳𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏．𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼺𖽷𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏． 11 𖼉𖽘𖾏 𖼮𖽵𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑 𖼅𖽻𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏，𖼎𖽺𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼊𖽑𖽡𖽝𖾐，𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼺𖽷𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏． 12 𖼉𖽘𖾏 𖼮𖽵𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼷𖽷𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖼡𖽑𖽜𖾑 𖼅𖽻𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 ， 𖼊𖽑𖽠𖾐 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏． 13 𖼉𖽘𖾏 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼅𖽻𖾏，𖼸𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽠𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼐𖽱𖾏． ’ 14 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 ， 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽘𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏，𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼐𖽑𖽡𖾏：‘𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽡𖾑 ， 𖼀𖽡𖾐 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏，𖼟𖽙𖾏 𖼀𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏．’ 15 𖼎𖽺𖾏 𖼊𖽑𖽠𖾐 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼋𖽠𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼙𖽠𖾏， 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏．𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑，𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽺𖾑 ？ 16 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽘𖾏 𖼐𖽡𖾑，𖼄𖽔𖾐 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． ” 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽙𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑！” 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：
‘𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏，
𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼞𖽪𖾐 𖽐𖼞𖽔𖾏 𖼊𖽠𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼮𖽡𖽙𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑．’
𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ？ 18 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼀𖽻𖾐 𖼷𖽠𖾏 𖼮𖽡𖽙𖾑 𖽃𖽔𖾏 𖼈𖽳𖾐 𖽃𖽡𖾑 𖼦𖽡𖾏 𖼇𖽔𖾑 ，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼺𖽘𖾏， 𖼮𖽡𖽙𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐 𖽃𖽡𖾑 𖼀𖽻𖾐 𖼷𖽠𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔 𖼮𖽡𖾑，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼗𖽱𖾏 𖼺𖽘𖾏．”

19 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 ，𖼀𖽻𖾐 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏，𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖽐𖼮𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼮𖽑𖽹𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏．
𖼊𖽪𖾐𖽃𖽔𖾏𖼉𖽘𖾏𖼟𖽪𖾏𖽐𖼚𖽪𖾏
20 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼊𖽔𖾐 𖼡𖽔 𖼅𖽔 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼎𖽺𖾏 𖼷𖽷𖾐 𖼸𖽜 𖼟𖽪𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼡𖽘𖾐 𖼄𖽙𖾐， 𖼻𖽻𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 ． 21 𖼸𖽻𖾏 𖼸𖽜 𖼟𖽪𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖽃𖽡𖾑 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏，𖼀𖽡𖾐 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼾𖽙 𖼯𖽡𖽙𖾏，𖼦𖽡𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽑𖽪𖾏，𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼡𖽑𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏． 22 𖼀𖽡𖾐 𖼊𖽪𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐，𖼋𖽡𖽝𖾏 𖼊𖽪𖾏 𖼢𖽻 𖼖𖽔𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼢𖽻？” 23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽔𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖼮𖽑𖽵𖾑，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： 24 “ 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏．𖼊𖽪𖾑 𖼼𖽘𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼊𖽪𖾑 𖽐𖼷𖽷𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼁𖽡𖽝？”𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 ：“𖼟𖽪𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐 𖼁𖽡𖽝．” 25 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼞𖽹𖾑 𖼺𖽔𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．” 26 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐， 𖼞𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼡𖽘𖾐 𖼗𖽪𖾏，𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ，𖼸𖽻𖾏 𖼎𖽺𖾏 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏．

𖼑𖽪𖾏 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙
（𖼊𖽑𖽹𖾏 22：23-33；𖼞𖽑𖽙𖾐 12：18-27）
27 𖼺𖽔𖾐 𖼊𖽪𖾑 𖼞𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽑𖽡𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 ， 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼋𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏： 28 “ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏， 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼑𖽜 𖼀𖽡𖾐 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：‘𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽙𖾏 𖼐𖽤𖾐 𖼊𖽘𖾏 ，𖼦𖽡𖾐 𖼑𖽡𖽝 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼺𖽷𖾐 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼷𖽙𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐，𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼷𖽔𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏． ’ 29 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏，𖼷𖽷𖾑 𖼘𖽑𖽙𖾐 𖼷𖽙𖾏 𖼐𖽤𖾐，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼋𖽔 𖼙𖽠𖾏． 30 𖼷𖽷𖾑 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼷𖽙𖾏 𖼐𖽱𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼋𖽔 𖼙𖽠𖾏． 31 𖼷𖽷𖾐 𖼗𖽜𖾐 𖽃𖽡𖾐 𖼗𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼷𖽙𖾏 𖼐𖽱𖾏， 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼊𖽪𖾏 𖼙𖽔 𖼷𖽙𖾏 𖼐𖽱𖾏，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼋𖽔 𖼙𖽠𖾏． 32 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼖𖽔𖾏 𖼋𖽔 𖼙𖽠𖾏． 33 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼖𖽺𖾑 ？ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼞𖽑𖽡𖽝𖾐 𖼊𖽪𖾏 𖼙𖽔 𖼷𖽙𖾏 𖼐𖽱𖾏．” 34 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼖𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑， 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽙𖾏 𖼡𖽔𖾏， 35 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑，𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙，𖼟𖽪𖾏 𖼦𖽡𖾐 𖼷𖽙𖾏 𖼦𖽡𖾐 𖼡𖽔𖾏． 36 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼋𖽔，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐； 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼋𖽘𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐． 37 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙，𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐 《𖽐𖼊𖽠𖾑 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 》𖼡𖽑𖽳𖾏 𖽐𖼎𖽑𖽹𖾏 𖼋𖽘𖾏 ，𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 38 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏．” 39 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏．” 40 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼑𖽪 𖼐𖽱𖾏 𖼋𖽪 𖼳𖽷𖾏．
𖼮𖽡𖾑 𖼊𖽪 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏
（𖼊𖽑𖽹𖾏 22：41-46；𖼞𖽑𖽙𖾐 12：35-37）
41 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼖𖽺𖾑 ？ 42 𖼐𖽡𖽙𖾐 《𖽐𖼊𖽠𖾑 𖽐𖼞𖽻𖾏》 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏：
‘ 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏：
𖼟𖽡 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏，
43 𖼋𖽙𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐．’
44 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽵𖾑，𖼐𖽱𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼖𖽺𖾑？”
𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏
（𖼊𖽑𖽹𖾏 23：1-36；𖼞𖽑𖽙𖾐 12：38-40）
45 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽻 𖽃𖽡𖾑 ，𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏： 46 “ 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏， 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏， 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽻𖾏，𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐， 𖼑𖽻𖾏 𖼉𖽔𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑． 47 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼺𖽘𖾏， 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏．𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝！”
