𖼽𖽡𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖽐𖼮𖽡𖽙𖾏
𖼞𖽪𖾐 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼡𖽜𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 36：22 - 23）
1
𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏𖼞𖽪𖾐 𖼖𖽡𖽝 𖼁𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 , 𖼎𖽺𖾏 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖾐 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼡𖽜𖾏 𖽐𖼊𖽠𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼞𖽪𖽙 𖼙𖽱 𖼐𖽑𖽡𖾏:
2 “𖼀𖽙𖾑 𖼺𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖾐 𖼖𖽡𖽝 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ： 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼄𖽔𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏 , 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼑𖽜 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 3 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 ,𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 . 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 , 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐. 4 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 , 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐、 𖼑𖽡𖽝𖾏、 𖼚𖽘𖾐 𖼐𖽜𖾐 , 𖽐𖼎𖽙 𖼯𖽦𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 , 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏.”
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏、 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏、 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼅𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼎𖽑𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏、 𖼞𖽪𖾐、 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏、 𖼯𖽦𖾏、 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏(“𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 ”),𖼎𖽺𖾏 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼊𖽑𖽡𖽝𖾐. 7 𖼞𖽪𖾐 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼦𖽔𖾏 𖼗𖽙𖾏, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼞𖽔𖾏 𖼮𖽡𖽝𖾏. 8 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏𖼞𖽪𖾐 𖼖𖽡𖽝 𖼮𖽙𖾏 𖼟𖽙𖾏𖼄𖽡𖾐 𖼊𖽑𖽡 𖼖𖽡𖾏 𖼊𖽔 𖼟𖽪𖾏 𖼞𖽳𖾑 𖽐𖼞𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐, 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼳𖽳 𖼀𖽔𖾑 𖼺𖽔𖾐. 9 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐:𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑 𖼞𖽪𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼗𖽪𖾏, 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑 𖼑𖽡𖽝𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽪𖾐, 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼯𖽢𖾏 𖼋𖽪𖾏， 10 𖼞𖽑𖽙𖾐 𖼞𖽪𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼗𖽪𖾏, 𖼞𖽑𖽙𖾐 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼟𖽻𖾏 𖼗𖽪𖾏, 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼮𖽘𖾏. 11 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼮𖽘𖾑. 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼎𖽑𖽻𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽡𖾑, 𖼳𖽳 𖼀𖽔𖾑 𖼺𖽔𖾐 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏.
