𖼄𖽙𖾑 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼞𖽪𖾑 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 , 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 : 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼢𖽳 𖼎𖽻𖾏 𖼟𖽡 .” 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽪 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼋𖽪 𖼳𖽷𖾏 ?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽪𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 .” 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖼗𖽔𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 .” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 , 𖼮𖽡𖾑 𖼀𖽜𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼏𖽘 𖼀𖽜𖾑 . 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼳𖽤𖾐 𖼋𖽠𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼗𖽙𖾏 𖼯𖽡𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼷𖽘𖾏 , 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 . 5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏、𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏、𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼋𖽘𖾏 .”
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼳𖽤𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼮𖽑𖽙𖾏 .” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼮𖽑𖽙𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 , 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼚𖽑𖽡𖾏 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏 𖼺𖽘𖾏 , 𖼚𖽠𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼀𖽪𖾏 . 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼎𖽺𖾏 𖼳𖽤𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼮𖽑𖽙𖾏 .” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼮𖽑𖽙𖾏 , 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼮𖽑𖽙𖾏 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼻𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 , 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 . 8 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼟𖽡 𖼗𖽪𖾏 , 𖼦𖽡𖾐 𖽐𖼮𖽜 𖽃𖽡𖾐 𖼋𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽜 𖼊𖽪𖾑 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼐𖽡𖾑 . 9 𖽃𖽔𖾐 𖼋𖽪 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽐𖼮𖽜 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼟𖽡 𖼗𖽪𖾏 , 𖼟𖽡 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼦𖽺𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖽃𖽻𖾐 𖼗𖽙𖾏 , 𖼘𖽑𖽡𖾑 𖼎𖽻𖾏 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 , 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽺𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽘𖾑 .”
