𖼦𖽔𖾐 𖼀𖽔𖾑 𖼞𖽪 𖼊𖽻𖾏 𖼮𖽵𖾑 𖼡𖽑𖽜𖾐 𖼡𖽑𖽳𖾏
2
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 ,
𖼳𖽠𖾑 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼑𖽔𖾏 ,
𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏,
𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼛𖽠𖾏 𖼑𖽪𖾏 .
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
“𖼄𖽔𖾐 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐 𖼐𖽡𖾑 𖼺𖽻𖾏 𖼟𖽙𖾏 𖽐𖼎𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼮𖽡𖽝𖾏,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼟𖽡𖽠 𖽃𖽡𖾑 𖼖𖽡 𖼟𖽻𖾏 𖼟𖽡𖽠 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼏𖽘 𖼖𖽔𖾏 𖼻𖽻𖾏 𖼟𖽡𖽠”).
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼏𖽵 ,
𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼽𖽔𖾑 𖼸𖽪𖾏 𖼋𖽘𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼮𖽑𖽵𖾑.
𖼊𖽔𖾑 𖼊𖽳𖾑𖽑 𖼈𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼺𖽷𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼊𖽔𖾑 𖼊𖽳𖾑𖽑.”
𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏
4 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼾𖽨𖾏, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼮𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼎𖽑𖽻𖾏 𖽐𖼮𖽜 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼊𖽔𖾐.
5 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼮𖽡𖽠𖾑 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏、 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏,
𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼀𖽜𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼐𖽱𖾏 𖼺𖽱𖾏 𖼋𖽔 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼦𖽡 𖼸𖽱 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐,
𖼨𖽑𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼁𖽡𖽝 .
6 𖼋𖽡𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼚𖽙𖾏 𖼐𖽱𖾏, 𖼯𖽡𖽙𖾏 𖽐𖼞𖽻𖾏 𖼡𖽔𖾏 𖼚𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼗𖽪𖾏 𖼀𖽡 𖼋𖽻 𖼟𖽪𖾏 𖼷𖽠𖾑𖽑, 𖼐𖽑𖽡𖾏:
“𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏! 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼟𖽡 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼾𖽨𖾏 𖼁𖽡𖽝 ,
𖼮𖽡𖽻𖾐 𖼮𖽡𖽻𖾐 𖼞𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼸𖽺𖾏 𖼮𖽡𖽝𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼺𖽷𖾐 𖼖𖽺𖾑?”
7 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡𖾏 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼯𖽷𖾏 𖼳𖽠𖾑,
𖼟𖽡 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽔𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
8 𖼑𖽜 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐,
𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑 𖽐𖼚𖽪𖾏 𖼎𖽻𖾏 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽻 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼟𖽡 .
 
9 𖼑𖽜 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼮𖽷𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑,
𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼳𖽻𖾏 𖼉𖽳𖾏, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏.
10 𖼟𖽡 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏,
𖼟𖽙𖾏 𖼟𖽡 𖼾𖽡𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼾𖽨𖾏 𖼦𖽺𖾏 𖼾𖽨𖾏 𖼳𖽦𖾑.
11 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼮𖽑𖽤 𖼽𖽔𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾑,
𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼑𖽡𖾏 𖼡𖽔𖾏 𖼎𖽪𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾏 𖼽𖽔𖾑 𖼊𖽡𖾐.
12 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼙𖽻,
𖼎𖽑𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼙𖽻 𖼼𖽙𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 !
13 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏, 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽑𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼚𖽔𖾏 𖼯𖽢𖾏 𖼙𖽠𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾐 𖼾𖽙 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖼯𖽤𖾏?
14 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼯𖽜𖾏 𖽐𖼎𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼚𖽻 𖼮𖽡𖾏 , 𖼽𖽔𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼷𖽑𖽔𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
 
15 𖼄𖽔𖾐 𖼮𖽡𖽠𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽻𖾏, 𖼎𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽻𖾏 𖼢𖽻, 𖼟𖽙𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼊𖽡𖾐 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 !
16 𖼟𖽡 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 ,
𖼟𖽡 𖼖𖽔𖾏 𖼦𖽻𖾏 ! 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐, 𖼽𖽔𖾑 𖼮𖽡𖾑 𖼸𖽙 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
17 𖼑𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖽐𖼎𖽙 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏,
𖼽𖽔𖾑 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼟𖽡 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑,
𖽃𖽔𖾏 𖼑𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑 𖽐𖼚𖽪𖾏 𖼎𖽻𖾏 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽻 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑.
 
18 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖽃𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽳𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽺𖾑?
𖼘𖽡𖾑𖽑 𖼁𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽡𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼊𖽺𖾐 𖼁𖽠𖾏 𖼡𖽔 𖼏𖽔 𖼐𖽡𖾑 , 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽺𖾑?
19 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽠𖾏 𖼸𖽻 𖼞𖽳𖾏 𖼐𖽑𖽡𖾏 “𖼯𖽷𖾏 𖼳𖽠𖾑”,
𖽐𖼎𖽙 𖼁𖽠𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼐𖽑𖽡𖾏： “𖼳𖽠𖾑 𖼗𖽙𖾏！” 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！
𖼊𖽪𖾑 𖼐𖽡𖾑 𖼸𖽠𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼑𖽔𖾏 𖼘𖽪! 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐 𖼊𖽔𖾐, 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼳𖽦𖾑 𖼀𖽘𖾏 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐.
 
20 𖼯𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏.
