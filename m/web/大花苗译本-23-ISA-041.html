16 𖼟𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼨𖽑𖽢𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼮𖽢𖾏 𖼽𖽔𖾑 𖼷𖽑𖽔𖾐 𖼙𖽠𖾏,
𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽔𖾐 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏.
𖼈𖽡𖽝𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏.
 
17 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽻𖾐 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼟𖽪𖾏 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏, 𖽃𖽔𖾑 𖽐𖼚𖽺𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡．
𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼞𖽪𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏.
18 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼮𖽢𖾏 𖼡𖽑𖽜𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐,
𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼦𖽘𖾑 𖼡𖽑𖽜𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼁𖽘𖾏 𖽃𖽻𖾐,
𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼷𖽠𖾑𖽑 𖽃𖽻𖾐.
19 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼯𖽡𖽙 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏、
𖼷𖽻𖾏 𖼮𖽢 𖽐𖼊𖽻𖾏、 𖼷𖽷𖾑 𖼣𖽑𖽪𖾐 𖼉𖽜 𖽐𖼎𖽙 𖼞𖽗𖾐 𖼖𖽗 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏．
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽔𖾏 𖼯𖽡𖽙 𖼮𖽡𖽝𖾏 𖼊𖽑𖽪𖾑、 𖼦𖽡𖾑 𖼑𖽡𖽻𖾏、
𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 .
20 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼎𖽺𖾏 𖼀𖽻𖾐,
𖼎𖽺𖾏 𖽐𖼊𖽱𖾏 ，𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼯𖽡𖽙𖾏,
𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽡𖾑 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏.”
𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑
21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼽𖽔𖾑 𖼮𖽑𖽳𖾑 𖼋𖽠𖾏 𖼅𖽡 𖼑𖽪𖾏 𖼗𖽙𖾏.”
𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼋𖽠𖾏 𖼅𖽡 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼞𖽡𖾑 .
22 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏,
𖼅𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏,
𖼟𖽙𖾏 𖼀𖽡𖾐 𖼻𖽻𖾏 𖼻𖽻𖾏 𖽐𖼊𖽱𖾏 , 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼑𖽪𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼄𖽔𖾐 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐.
23 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼟𖽙𖾏 𖽐𖼎𖽑𖽹𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼑𖽪𖾏,
𖼟𖽙𖾏 𖼀𖽡𖾐 𖼀𖽻𖾐 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
𖼅𖽡 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔,
𖼟𖽙𖾏 𖼀𖽡𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏, 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏．
24 𖼑𖽔𖾏 𖼘𖽪! 𖼅𖽡 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 ,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽔𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼅𖽡 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖼳𖽪𖾐 𖼯𖽡𖽙 .
 
25 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼋𖽔𖾏 , 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼯𖽪𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼯𖽪𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
26 𖼡𖽔 𖼋𖽱𖾏 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖼡𖽑𖽳𖾏 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼀𖽻𖾐 𖼖𖽺𖾑?
𖼡𖽔 𖼋𖽱𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏 “𖼐𖽱𖾏 𖼦𖽡𖾐 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼖𖽺𖾑?”
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼡𖽑𖽳𖾏 𖽐𖼎𖽑𖽹𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼅𖽡 𖼗𖽪𖾏 .
27 𖼞𖽪𖾑 𖼺𖽪𖾐 𖼁𖽱 𖽐𖼎𖽙 𖼳𖽡 𖼣𖽔𖾑 𖼐𖽑𖽡𖾏:
“𖼑𖽔𖾏 𖼘𖽪！𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑!” 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼻𖽻𖾏 𖼄𖽻𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 .
28 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼞𖽪𖾑 𖼑𖽪 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼸𖽠𖾏 𖼊𖽡𖾐 𖽃𖽡𖾐 𖼖𖽙𖾏.
29 𖼑𖽔𖾏 𖼘𖽪！ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽠𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖼘𖽡𖾑𖽑 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽢𖾏,
𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼯𖽡𖽙𖾏.”
