𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜 𖼯𖽻 𖼚𖽑𖽔𖾏，𖼅𖽡 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏，𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼦𖽡𖾏 𖼺𖽱𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏． 2 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽻𖾏 𖼢𖽻；𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏，𖼯𖽔 𖼑𖽻𖾏 𖼻𖽻𖾐 𖽐𖼷𖽔𖾐 𖼊𖽔𖾐． 3 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻𖾏；𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽠𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼋𖽘𖾏． 4 𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏，𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼀𖽡𖾑 𖼙𖽻 𖼘𖽪𖾏？ 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼳𖽠𖾑 𖼏𖽔𖾏，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽻𖾏 𖼢𖽻，𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼅𖽔𖾏 𖼞𖽳𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼟𖽪𖾏 𖼏𖽔𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼟𖽪𖾏 𖼏𖽔𖾏． 5 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑；𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼊𖽪𖾏 𖼦𖽡𖾏 𖼺𖽱𖾏．𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 ． 6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼮𖽵𖾑 𖼻𖽙𖾑．𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼮𖽵𖾑 𖼑𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽑𖽔𖾏 𖼙𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏；𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻𖾏， 𖼑𖽜 𖼮𖽵𖾑 𖼦𖽡𖾐 𖼑𖽻𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼙𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 7 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼋𖽔． 8 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼯𖽡𖽝𖾏， 𖼟𖽪𖾏 𖼑𖽜 𖼮𖽵𖾑 𖼯𖽡𖽝𖾏，𖼮𖽔𖾑 𖼋𖽔 𖼟𖽪𖾏 𖼑𖽜 𖼮𖽵𖾑 𖼋𖽔．𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼯𖽡𖽝𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼋𖽔，𖼀𖽡𖾐 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼁𖽡𖽝． 9 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼮𖽡𖾑 𖼊𖽪 𖼋𖽔 𖼙𖽠𖾏，𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼋𖽘𖾏，𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑． 10 𖼟𖽡 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑， 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽠𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼘𖽪𖾏？ 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼘𖽪？ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏． 11 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：
“ 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏，𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼏𖽵 𖼗𖽪𖾏，
𖼉𖽻𖾏 𖼉𖽻𖾏 𖼇𖽻𖾑 𖼯𖽡𖽻 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑；
𖼉𖽻𖾏 𖼉𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ．”
