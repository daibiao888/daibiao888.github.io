𖼊𖽑𖽡𖽝𖾐 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 、 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
3 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼑𖽜 𖼾𖽨𖾏 𖼨𖽑𖽢𖾏 𖼮𖽪𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑,

4 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼮𖽵𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏,
𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑,

5 𖼽𖽔𖾏 𖼳𖽷 𖼮𖽡𖾑 𖼖𖽪𖽝 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼮𖽢𖾑 𖼺𖽔𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼽𖽡𖾑 𖼞𖽳 𖼖𖽪𖽝 𖼑𖽜 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
𖼮𖽢𖾑 𖼺𖽔𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼽𖽔𖾏 𖼳𖽷 𖼮𖽡𖾑 𖼖𖽪𖽝 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏.
6 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽘𖾑 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏 (“𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽘𖾑 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏 ”𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “𖼊𖽪𖾐 𖼚𖽻𖾐”) 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼳𖽷 𖼊𖽑𖽪 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏.
7 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽐𖼮𖽑𖽘𖾑,
𖽐𖼎𖽙 𖼊𖽪𖾑 𖼐𖽑𖽡𖽝𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖼳𖽪𖾐,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏, 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼺𖽱𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑,
𖼽𖽡𖾑 𖼞𖽳 𖼖𖽪𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐.
8 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼮𖽪𖾐 𖼾𖽡 ,
𖼟𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼏𖽙 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼯𖽵𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏.
𖼳𖽡 𖼣𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼮𖽨𖾏 𖽐𖼀𖽘𖾏
9 𖼳𖽡 𖼣𖽔𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 (“ 𖼖𖽡𖾑 𖼉𖽪𖾏 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 ”) 𖼘𖽪 ! 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼙𖽔 𖼚𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪! 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏.
𖼑𖽔𖾏 𖼘𖽪 ! 𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼟𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑 !
𖼐𖽱𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼄𖽔𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑, 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 .
10 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 ,
𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 ,
𖼊𖽪𖾑 𖼐𖽑𖽜𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐,
𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼐𖽡𖾑 𖼞𖽳𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽡𖾑,
𖼎𖽑𖽻𖾏 𖼽𖽨𖾏 𖼇𖽔 𖼖𖽔𖾑 𖼊𖽡𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼞𖽳𖾑 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏.
𖼮𖽵𖾑 𖼏𖽵 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏
11 𖼳𖽡 𖼣𖽔𖾑 𖼘𖽪 ! 𖼞𖽪𖾑 𖼑𖽜 𖽐𖼮𖽑𖽘𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖽃𖽡𖾑,

13 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽔𖾏 𖼐𖽑𖽜𖾑 𖼟𖽪𖾏 𖽐𖼊𖽘𖾏,

𖼳𖽡 𖼣𖽔𖾑 𖼘𖽪 ! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐,
𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼳𖽡𖾑 𖼖𖽔 ( “𖼳𖽡𖾑 𖼖𖽔”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼽𖽔𖾐 𖽂𖽔 ” )𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐,
𖼟𖽙𖾏 𖼟𖽡 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏.
 
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼢𖽳 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼐𖽱𖾏 𖼊𖽪𖾑 𖼈𖽙𖾏 𖼽𖽔𖾑 𖼀𖽙𖾑 𖼋𖽠 𖼗𖽙𖾏 𖼺𖽱𖾏 𖼺𖽙𖾐 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼯𖽦𖾏.
𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐,
𖼯𖽡𖽜𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖼅𖽻𖾏 𖼗𖽙𖾏 .
15 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼗𖽔𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼞𖽡𖾏 𖼘𖽪𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼦𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖽐𖼮𖽑𖽘𖾑,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼀𖽪𖾑 𖽐𖼮𖽑𖽘𖾑.
 
16 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐,𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼘𖽙𖾏 𖼎𖽻𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑,
𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 ( 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏”).
17 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 !
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 !
