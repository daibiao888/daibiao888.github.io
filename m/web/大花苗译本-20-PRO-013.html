𖼗𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖽐𖼚𖽪𖾏
13
𖼊𖽪𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽻 𖼉𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼗𖽔 𖼦𖽡𖾐 𖼑𖽻 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 .
2 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼑𖽡𖾏 𖽐𖼀𖽜𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳.
3 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼊𖽻𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼘𖽑𖽙𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏.
4 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽜 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽻𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏.
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼳𖽪𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼅𖽔𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
6 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼀𖽑𖽻𖾏 𖽐𖼀𖽙 .
7 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼊𖽻𖾏.
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽪𖾑 𖼎𖽙𖾑 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽙𖾑𖽑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽠𖾏 𖼮𖽑𖽙 .
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏， 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼖𖽔𖾑 𖼋𖽔 .
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑, 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
11 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼚𖽱𖾏 𖽐𖼞𖽘𖾏 𖼖𖽑𖽘𖾐 𖼊𖽻𖾏 𖼗𖽙𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽡 𖼟𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽻𖾐 𖽐𖼡𖽔𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
12 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼊𖽻𖾏, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏；
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏 𖼸𖽙 , 𖼺𖽱𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑.
13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐,
𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼋𖽔 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏.
15 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼀𖽡𖾐 𖽐𖼞𖽡𖾐 𖼀𖽡𖾐 𖽐𖼞𖽱𖾐 𖼮𖽷𖾑 𖼅𖽻𖾏.
16 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽔𖾏 𖼑𖽪𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼾𖽨𖾏 𖼡𖽔 𖼏𖽔.
17 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼞𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
18 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏.
19 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼡𖽘𖾐 𖼄𖽪𖾑,
𖼀𖽜𖾑 𖼛𖽡𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐, 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼳𖽪𖾐.
20 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ；
𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠.
21 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼖𖽠𖾑 𖼊𖽪𖾐 𖼾𖽔𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
22 𖼻𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼷𖽷𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑.
