18
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐𖼐𖽑𖽡𖾏: 2 “𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏.” 3 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏,𖼸𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖼀𖽔𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐. 4 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼞𖽙𖾏 𖼀𖽔𖾐 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖽙𖾑 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼞𖽙𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼀𖽔𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐, 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼻𖽻𖾏, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 .
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑,𖼐𖽑𖽡𖾏: 6 “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼐𖽡𖾑 𖽐𖼮𖽨𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼖𖽺𖾑? 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼘𖽪 ! 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼞𖽙𖾏 𖼐𖽡𖽙𖾐 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 . 7 𖼞𖽪𖾑 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼐𖽑𖽡𖾏 𖼸𖽙 𖽃𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼞𖽪𖽙,𖼐𖽑𖽡𖾏: 𖼽𖽔𖾑 𖼚𖽙𖾐 𖼺𖽘𖾏、𖼎𖽑𖽔𖾑 𖼺𖽘𖾏、𖼖𖽻𖾐 𖼺𖽘𖾏. 8 𖽃𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,𖼞𖽪𖾑 𖼎𖽺𖾏 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,𖼦𖽡𖾐 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽘. 9 𖼞𖽪𖾑 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼐𖽑𖽡𖾏 𖼸𖽙 𖽃𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼞𖽪𖽙 𖼐𖽑𖽡𖾏, 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏, 𖼯𖽡𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏. 10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,𖼦𖽡𖾐 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽘 .
𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽨 𖼊𖽔𖾏𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼾𖽔𖾏 𖼐𖽔𖾑 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔
11 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐𖼐𖽑𖽡𖾏: 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽐𖼊𖽻𖾏 𖼅𖽡, 𖼏𖽵 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼅𖽡 , 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽙𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼅𖽡 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏. 12 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 . 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼼𖽔𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏.”
13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼅𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽪 ,
𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽙𖾑𖽑 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏(“𖼖𖽡𖾑 𖼉𖽪𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑”) 𖽃𖽔𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽪𖾏 𖼸𖽳 .
14 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼑𖽡𖾏 𖽐𖼀𖽪𖾏, 𖼐𖽡𖽙𖾐 𖽐𖼎𖽘𖾏 𖼯𖽢𖾏 𖼑𖽡𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼾𖽘𖾏 𖼺𖽘𖾏 𖼖𖽺𖾑?
𖽃𖽻𖾐 𖼸𖽡𖽝 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖽐𖼚𖽱𖾏 𖼗𖽙𖾏，𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏 𖼖𖽺𖾑?
15 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏,
𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼎𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑 𖼦𖽡𖾏 𖼚𖽠𖾏 𖼢𖽻 ,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽢𖾏.
16 𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼷𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏,
𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽙𖾏,
