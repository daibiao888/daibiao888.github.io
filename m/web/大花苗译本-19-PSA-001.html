𖽐𖼊𖽠𖾑 𖽐𖼞𖽻𖾏 𖽃𖽡𖾐 𖼑𖽘
𖼻𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏
1
𖼦𖽡𖾐 𖼸𖽙𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑,
𖼦𖽡𖾐 𖼳𖽠𖾑 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑,
𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐,
2 𖼯𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑,
𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼮𖽳𖾐 𖽐𖼊𖽱𖾏 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑,
𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
3 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏, 𖼯𖽡𖽙 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽔𖾐 𖼟𖽱𖾐 𖽃𖽻𖾐,
𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐,
𖽃𖽔 𖽐𖼚𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡. 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑.
 
4 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑,
𖼟𖽪𖾏 𖼺𖽱𖾏 𖼦𖽡𖾑 𖼁𖽘𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏.
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽃𖽡𖾑, 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏,
𖼊𖽪𖾐 𖼾𖽔𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑,
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏.
