𖼞𖽑𖽻𖾑 𖼸𖽻𖾏 𖽐𖼎𖽻𖾏 𖼾𖽔𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼮𖽵𖾑
14
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ! 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼟𖽡 𖼟𖽪𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 𖼙𖽠𖾏.
2 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏:
“𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏 ,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼘𖽪𖾑𖽑 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 .
3 𖼽𖽔𖾏 𖼳𖽪 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐,
𖼀𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼯𖽡𖽝𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:
𖼟𖽡 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐.”
 
4 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 ,
𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏.
5 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼙𖽱 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 ,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏 𖼋𖽠𖾏 𖼁𖽘𖾏,
𖼺𖽱𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖽐𖼊𖽻𖾏 𖼗𖽔 𖽃𖽔𖾑 𖼯𖽤𖾐.
6 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼽𖽔𖾑 𖼘𖽑𖽘𖾑 𖼟𖽪𖾏 𖽐𖼊𖽡𖾑,
𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼞𖽗𖾐 𖼖𖽗 𖽐𖼊𖽻𖾏,
𖼐𖽱𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼺𖽱𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏.
7 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 , 𖼟𖽙𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼘𖽑𖽘𖾑 𖼘𖽑𖽙𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼘𖽑𖽘𖾑 𖽐𖼷𖽱𖾏 𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏.
𖼐𖽱𖾏 𖼻𖽻𖾏 𖽐𖼷𖽷𖾏 𖼺𖽱𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼮𖽡𖽠𖾑 .
8 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖽐𖼎𖽙 𖼊𖽺𖾐 𖼁𖽠𖾏 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪?”
𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽡𖾐 𖼐𖽱𖾏, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽻𖾏 𖼮𖽳𖾐 𖼐𖽱𖾏.
𖼞𖽪𖾑 𖼺𖽱𖾏 𖼇𖽘𖾐 𖼊𖽑𖽪𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾏 𖼟𖽡𖾏 𖽐𖼷𖽔𖾐 𖼄𖽡𖾐 𖼺𖽡𖽝𖾐,
𖼟𖽡 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼗𖽙𖾏 .
 
9 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳？ 𖼸𖽠𖾏 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 ,
𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 ？ 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 .
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼊𖽪𖾐 𖼾𖽔𖾏 𖼯𖽡𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 .
