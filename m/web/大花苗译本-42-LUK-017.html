𖼯𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡
（𖼊𖽑𖽹𖾏 18：6-7，21-22；𖼞𖽑𖽙𖾐 9：42）
17
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼢𖽻 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼯𖽡𖽙𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏． 2 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼡𖽑𖽹𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼢𖽪𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼦𖽡𖾑 𖼚𖽘𖾐，𖼗𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑，𖼎𖽺𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼣𖽔𖾏 𖼾𖽻𖾏 𖼐𖽡𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽻𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻． 3 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏．𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏，𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏，𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 ，𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽔𖾏 𖼐𖽱𖾏． 4 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼳𖽤𖾏 𖼟𖽻𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡，𖼳𖽤𖾏 𖼟𖽻𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼋𖽘𖾏’， 𖼟𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽔𖾏 𖼐𖽱𖾏．”
𖼸𖽻𖾏𖼟𖽪𖾏𖼟𖽙𖾏𖼊𖽜𖾑𖽃𖽔𖾏𖼾𖽔𖾏𖽃𖽡𖾑𖼅𖽔𖾏𖼳𖽻𖾏𖽐𖼚𖽔 𖼋𖽘𖾏
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜
5 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼄𖽻𖾏 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏：“ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼮𖽵𖾑 𖽐𖼀𖽘𖾏 𖼊𖽑𖽡𖽝𖾐 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜．” 6 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼮𖽘𖾑 𖼻𖽻𖾐 𖼛𖽙𖾐，𖼟𖽪𖾏 𖽐𖼎𖽙 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼀𖽡𖾐 𖼞𖽘𖾐 𖼻𖽻𖾐 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼽𖽔𖾑 𖼚𖽙𖾐 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 ，𖼯𖽡𖽙 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 ’，𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼅𖽡．
𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼑𖽡𖾏 𖼙𖽻 𖼋𖽪𖾏
7 𖼅𖽡 𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼾𖽨 𖼾𖽘𖾏，𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼎𖽑𖽻𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏， 𖼟𖽡 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼞𖽑𖽜 𖼑𖽜 𖼗𖽙𖾏 𖼼𖽻𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼦𖽡𖾑 ？’ 8 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼄𖽑𖽙𖾏，𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 𖼋𖽔𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑，𖼋𖽙𖾏 𖼞𖽪𖾑 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼋𖽘𖾏，𖼟𖽡 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 ’ 𖼖𖽺𖾑？ 9 𖼡𖽑𖽜𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏，𖼐𖽱𖾏 𖼮𖽵𖾑 𖼋𖽡𖽝𖾏 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼐𖽱𖾏 𖼘𖽪？ 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼋𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖽃𖽡𖾑，𖼯𖽔 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：‘𖼀𖽡𖾐 𖼟𖽪𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙，𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼐𖽡𖾑，𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏．’ ”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼟𖽻𖾏 𖼗𖽜𖾏 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏
11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐，𖼮𖽑𖽡𖽻𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏． 12 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼼𖽙𖾏， 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼗𖽜𖾏 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏 𖼋𖽔𖾏 𖼺𖽘𖾑 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼸𖽱𖾏 𖼐𖽱𖾏，𖼳𖽠𖾑 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐， 13 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼽𖽦𖾑 𖼺𖽪𖾑 ，𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 ，𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼀𖽡𖾐！” 14 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽙𖾏 𖼋𖽘𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “𖼅𖽡 𖼅𖽻𖾏 𖼄𖽔𖾐 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏．” 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏． 15 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼋𖽔𖾏 𖼮𖽘𖾏 𖼺𖽘𖾏 𖼨𖽑𖽪𖾏 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 16 𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼐𖽱𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝．𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐． 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽙𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼗𖽜𖾏 𖼘𖽪 ？𖼸𖽻𖾏 𖼯𖽢𖾏 𖼗𖽜𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼘𖽪？ 18 𖼋𖽠𖾏 𖼷𖽷𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽷𖾐 𖼋𖽔𖾏，𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪𖾏？” 19 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“ 𖼳𖽠𖾑 𖼗𖽙𖾏 ，𖼅𖽻𖾏 𖼦𖽡𖾑 ，𖼟𖽡 𖽐𖼮𖽜 𖼮𖽨𖾏 𖼟𖽡 𖼋𖽘𖾏．”
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏
（𖼊𖽑𖽹𖾏 24：23-28，37-41）
20 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪：“ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼊𖽑𖽻𖾑 𖼋𖽱 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏？”𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 ，𖼦𖽡𖾐 𖼾𖽙 𖽃𖽔𖾑 𖼅𖽔 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏． 21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏：‘𖼑𖽔𖾏 𖼘𖽪！ 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽑𖽡𖾏 ： 𖼑𖽔𖾏 𖼘𖽪 ！𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼉𖽺𖾏，’ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼐𖽡𖽙𖾐 𖼅𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐．” 22 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏，𖼟𖽪𖾏 𖼅𖽡 𖽐𖼮𖽑𖽠𖾏 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏． 23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏：‘𖼑𖽔𖾏 𖼘𖽪！ 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼉𖽺𖾏， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽑𖽡𖾏 ： 𖼑𖽔𖾏 𖼘𖽪 ！𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑，’ 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐！ 24 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏，𖼺𖽱𖾏 𖼺𖽱𖾏 𖼺𖽙𖾐 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼯𖽦𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖽐𖼊𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼯𖽦𖾏 𖼮𖽡𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖽐𖼊𖽪𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏． 25 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼺𖽪𖾐 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼎𖽺𖾏 𖼊𖽻𖾏 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏． 26 𖼐𖽙 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏． 27 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏，𖼷𖽙𖾏 𖼐𖽤𖾐 𖼡𖽔𖾏 𖽐𖼷𖽑𖽹𖾏，𖼸𖽙 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼐𖽙 𖼽𖽔𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽙𖾏 ，𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏． 28 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼘𖽙 𖼊𖽳 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖼦𖽻𖾏， 𖼅𖽔𖾏 𖼋𖽺 , 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐， 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏． 29 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼘𖽙 𖼊𖽳 𖼋𖽠𖾏 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐， 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽐𖼎𖽙 𖼖𖽨 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼗𖽙𖾏，𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏． 30 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 31 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼏𖽪𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖼏𖽪𖾏，𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏 𖼞𖽑𖽡𖽠𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏． 32 𖼅𖽡 𖼽𖽔𖾑 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖼘𖽙 𖼊𖽳 𖼑𖽡𖾏 𖼐𖽤𖾐． 33 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑． 34 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏，𖼄𖽑𖽙𖾏 𖽃𖽡𖾑 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽱𖾏 𖽃𖽡𖾐 𖼸𖽱𖾏 𖼸𖽘𖾏，𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼙𖽠𖾏 𖼇𖽳𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏． 35 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼢𖽪𖾏 𖼡𖽜𖾑，𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼙𖽠𖾏 𖼇𖽳𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏． 36 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼙𖽡𖽝𖾏 ，𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼙𖽠𖾏 ，𖼇𖽳𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏 ．” 37 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼘𖽪？”𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽻𖾑 𖼋𖽔 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏， 𖼚𖽘𖾑 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏．”
