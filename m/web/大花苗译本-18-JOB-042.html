𖼽𖽙 𖼀𖽳 𖼯𖽜𖾏 𖼾𖽔𖾏
42
𖼽𖽙 𖼀𖽳 𖼊𖽡𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 :
2 “𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼉𖽻𖾏 𖼑𖽪𖾏 ,
𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 .
3 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼟𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼖𖽺𖾑 ?
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 ,
𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐.
4 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼞𖽪𖾑 𖼑𖽪 𖼟𖽡 , 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 .
5 𖼞𖽪𖾑 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼟𖽡 ,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 𖼟𖽡 .
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼾𖽨𖾏 ,
𖼐𖽡𖽙𖾐 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼮𖽑𖽻𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻.”
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼽𖽙 𖼀𖽳
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼽𖽙 𖼀𖽳 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽑𖽡 𖼄𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 , 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼽𖽙 𖼀𖽳 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽙 . 8 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽤𖾐 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼽𖽙 𖼀𖽳 𖼉𖽺𖾏 𖽃𖽡𖾑 , 𖼑𖽜 𖼾𖽨𖾏 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼽𖽙 𖼀𖽳 𖼎𖽺𖾏 𖼑𖽜 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 , 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼅𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼚𖽘𖾐 𖼯𖽡𖽙𖾏 𖼅𖽡 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 , 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼽𖽙 𖼀𖽳 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽙 .” 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼊𖽑𖽡 𖼄𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 、 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾐 𖼖𖽳 𖼊𖽔 、 𖼐𖽔 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽙𖾐 𖼇𖽔 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼽𖽙 𖼀𖽳 .
10 𖼽𖽙 𖼀𖽳 𖼑𖽜 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼽𖽙 𖼀𖽳 𖼎𖽑𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽙𖾑 𖼗𖽙𖾏 ( “𖽃𖽦𖾐 𖽐𖼀𖽜𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼊𖽻𖾏 𖼦𖽔𖾏”) , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾐 𖽐𖼀𖽠 . 11 𖼽𖽙 𖼀𖽳 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼞𖽱𖾏 𖼻𖽷𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼡𖽑𖽙𖾑 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼞𖽑𖽔𖾏 𖼋𖽔𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 , 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼑𖽻𖾏 𖼉𖽔𖾏 , 𖼎𖽺𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼖𖽔𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼐𖽱𖾏 , 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼑𖽡𖽝𖾏 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼁𖽻 𖼞𖽪𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 . 12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼽𖽙 𖼀𖽳 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 , 𖼐𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 , 𖼚𖽻𖾏 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏 𖼘𖽙 𖼊𖽑𖽙𖾐 , 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖽐𖼞𖽡𖽠𖾏 𖼑𖽨𖾏 , 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽜𖾏 𖼖𖽔𖾑 . 13 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 , 𖼷𖽷𖾐 𖼖𖽜𖾐 𖽐𖼷𖽑𖽹𖾏 . 14 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖽐𖼷𖽷𖾏 𖼽𖽦𖾑 𖼄𖽡𖾐 𖼄𖽔𖾐 , 𖽐𖼷𖽑𖽹𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖽐𖼷𖽷𖾏 𖼮𖽡𖾑 𖼳𖽡𖾐 𖼽𖽔𖾏 , 𖼷𖽷𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽑𖽹𖾏 𖽐𖼷𖽷𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼦𖽔𖾐 𖼀𖽑𖽪 . 15 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 , 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼽𖽙 𖼀𖽳 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼖𖽡𖾏 𖽃𖽡𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 . 16 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼽𖽙 𖼀𖽳 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 , 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼘𖽑𖽔𖾑 , 𖼸𖽙 𖼚𖽔𖾏 𖼚𖽻𖾐 𖼷𖽑𖽷𖾏 . 17 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽙 𖼀𖽳 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏 , 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼀𖽪𖾑 𖼋𖽘𖾏 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼙𖽠𖾏 .
