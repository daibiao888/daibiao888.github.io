𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼢𖽳 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 , 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖽃𖽳𖾐 𖼐𖽔 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼡𖽻𖾐 , 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽔 𖼏𖽵 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 , 𖼁𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 . 2 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽻𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 , 𖼖𖽪𖾐 𖽃𖽔𖾐 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽃𖽔𖾐 𖼁𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 . 3 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼊𖽠𖾏 𖽐𖼞𖽔𖾏 , 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖽐𖼊𖽡𖾑 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼇𖽘𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼮𖽑𖽡𖽠𖾏 𖼇𖽘𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼇𖽘𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽡𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 . 4 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑 𖼡𖽘𖾐 𖽃𖽔𖾏 𖽐𖼮𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 , 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼇𖽘𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 , 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼟𖽪𖾏 𖼘𖽙𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑. 5 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 , 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼊𖽑𖽪𖾑 𖼀𖽔𖾐 𖼘𖽙𖾏, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼘𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼳𖽪𖾏 𖼡𖽻𖾏, 𖼎𖽺𖾏 𖼞𖽳𖾏 𖽐𖼊𖽻𖾏 𖼺𖽱𖾐 𖼺𖽱𖾐 𖼼𖽘𖾏 𖼊𖽑𖽡𖽝𖾐 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼼𖽘𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏. 6 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼘𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽤 , 𖼟𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽺𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 . 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼀𖽔𖾑 𖽂𖽔𖾐 𖼽𖽡𖾑 𖼑𖽡𖾏 𖼞𖽪𖾐 . 7 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼘𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼊𖽑𖽡𖽝𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼊𖽻𖾏 、 𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏 、 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 、 𖽃𖽔 𖼛𖽻𖾏 , 𖽃𖽔𖾐 𖼋𖽜𖾏 𖼮𖽑𖽤 𖼞𖽳𖾏 𖼅𖽔𖾏 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 . 8 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼇𖽘𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 , 𖼇𖽘𖾑 𖼟𖽪𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼎𖽺𖾏 𖼘𖽙𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖼚𖽻𖾏 𖼀𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 . 9 𖼺𖽙𖾑 𖼖𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 . 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼖𖽔𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼘𖽙𖾏.
𖼷𖽷𖾏 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳
10 𖼐𖽡𖽙𖾐 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑, 𖼸𖽻 𖼞𖽳𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 , 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏 . 11 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼷𖽷𖾑 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼐𖽡𖾑 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼊𖽡𖾏 𖽐𖼊𖽡𖾑 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼺𖽘𖾑 𖼐𖽡𖾑 𖼖𖽪𖾑 𖼮𖽑𖽤 , 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼺𖽘𖾑 𖽃𖽡𖾑 𖼖𖽔𖾏 𖽐𖼊𖽡𖾑 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼦𖽡𖾏 𖼊𖽡𖾏 . 12 𖼷𖽷𖾑 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼊𖽡𖾏 𖽐𖼊𖽡𖾑 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼺𖽘𖾑 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼮𖽑𖽤 , 𖼊𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖽐𖼊𖽡𖾑 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼦𖽡𖾏 𖼊𖽡𖾏 . 13 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖽐𖼊𖽑𖽔𖾑 𖽃𖽔𖾑 𖼊𖽡𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼚𖽑𖽪𖾏 𖼊𖽱𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 . 14 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼀𖽑𖽜𖾑 、 𖼺𖽙𖾑 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑 、 𖼺𖽙𖾑 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼗𖽙𖾏 𖽐𖼊𖽙𖾏 𖼀𖽑𖽙𖾑 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽺𖾏 , 𖼳𖽡𖾑 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼎𖽻𖾏 .
𖼷𖽷𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 7：15 - 22）
15 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼷𖽑𖽙𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 , 𖼺𖽡𖽝𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼷𖽑𖽔𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼀𖽜𖾑 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 16 𖼎𖽺𖾏 𖼸𖽙𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽐𖼚𖽙 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼼𖽘𖾏 𖽃𖽔𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 , 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖼎𖽺𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽪𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 , 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 . 17 𖼷𖽑𖽙𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖽐𖼮𖽔𖾏 𖽐𖼞𖽔𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼺𖽷𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼇𖽳𖾏 , 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼺𖽷𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 , 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼇𖽳𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 .
