2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼯𖽡𖽙𖾏,

5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐,
