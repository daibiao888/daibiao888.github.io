（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 2：1 - 18）
5
𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼖𖽔, 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼮𖽑𖽠𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼐𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽜𖾑 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼷𖽷𖾐 𖼡𖽑𖽜𖾑 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏. 2 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼳𖽡𖾑 𖼖𖽔, 𖼐𖽑𖽡𖾏: 3 “𖼟𖽡 𖼀𖽻𖾐 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽜 𖼚𖽻𖾐 𖼺𖽘𖾐 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽠𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐. 4 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔. 5 𖼞𖽪𖾑 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼷𖽹𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑, 𖼐𖽡𖽙𖾐 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼊𖽙𖾑 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏, 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼏𖽵 , 𖼄𖽔𖾐 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐, 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖽐𖼊𖽙𖾑 𖼈𖽳𖾐 𖽐𖼊𖽙𖾑 𖽐𖼊𖽻𖾏.”
7 𖼳𖽡𖾑 𖼖𖽔 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼐𖽑𖽡𖾏: “𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 , 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼐𖽡𖾑 .” 8 𖼳𖽡𖾑 𖼖𖽔 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞, 𖼐𖽑𖽡𖾏: “𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏, 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽐𖼎𖽙 𖼊𖽑𖽪𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏. 9 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼨𖽑𖽹𖾑 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑, 𖼡𖽑𖽹𖾐 𖽃𖽔𖾏 𖼡𖽔𖾑 𖽐𖼊𖽠, 𖽐𖼊𖽘𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽳𖾏 𖽃𖽡𖾑. 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼚𖽠𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑, 𖼟𖽡 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼯𖽡𖽙𖾏 𖼅𖽻, 𖼟𖽡 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏, 𖼄𖽔𖾐 𖼡𖽻𖾐 𖼡𖽜𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏.” 10 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼳𖽡𖾑 𖼖𖽔 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼡𖽑𖽙𖾑 𖼽𖽔𖾑 𖽃𖽡𖾑, 𖼄𖽔𖾐 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽐𖼎𖽙 𖼊𖽑𖽪𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏； 11 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼄𖽔𖾐 𖽃𖽔𖾐 𖼉𖽻𖾏 𖼞𖽙𖾑 𖽃𖽳𖾐 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼳𖽻𖾐, 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼞𖽙𖾑 𖽃𖽳𖾐 𖼛𖽙𖾏 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼖𖽔, 𖽃𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏． 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼷𖽑𖽔𖾏 𖼳𖽡𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼖𖽔. 12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼙𖽔 𖼏𖽵 𖼮𖽡𖽝𖾏, 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞． 𖼳𖽡𖾑 𖼖𖽔 𖽐𖼎𖽙 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏 ，𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏.
𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏
