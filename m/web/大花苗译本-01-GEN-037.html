𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑
37
𖼽𖽔𖾐 𖼞𖽙 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐, 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑. 2 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐:
𖼽𖽙 𖼺𖽳 𖼟𖽻𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡 𖽃𖽡𖾐 𖼸𖽻 𖼾𖽨 𖼯𖽦𖾏, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖼀𖽑𖽡 𖼖𖽔𖾑、 𖼳𖽡 𖼀𖽑𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼅𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏． 𖼽𖽙 𖼺𖽳 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐. 3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼮𖽑𖽠𖾏 𖼽𖽙 𖼺𖽳 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽪𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏 𖽃𖽡𖾑 𖼘𖽘𖾑𖽑 𖼾𖽨 𖼐𖽔𖾑, 𖼐𖽱𖾏 𖼑𖽜 𖼽𖽙 𖼺𖽳 𖽃𖽔𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼮𖽑𖽙𖾏 𖼘𖽪𖾑𖽑 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏. 4 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼮𖽑𖽠𖾏 𖼽𖽙 𖼺𖽳 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼎𖽺𖾏 𖼳𖽪𖾐 𖼽𖽙 𖼺𖽳, 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏.
5 𖼽𖽙 𖼺𖽳 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼳𖽪𖾐 𖼐𖽱𖾏. 6 𖼽𖽙 𖼺𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽢𖾏 𖼅𖽡 𖼑𖽻 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏： 7 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼡𖽑𖽹𖾐 𖽃𖽔𖾏 𖼷𖽻𖾏 𖼡𖽻𖾐, 𖽃𖽔𖾏 𖼷𖽻𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽹𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏, 𖽃𖽔𖾏 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽹𖾐 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼞𖽑𖽡 𖼙𖽻 𖽃𖽔𖾏 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽹𖾐 𖽃𖽡𖾑 𖼛𖽪𖾏 𖼇𖽻𖾑.” 8 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪? 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼽𖽔𖾑 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼀𖽡𖾐 𖼘𖽪?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼗𖽪𖾏, 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼳𖽪𖾐 𖼐𖽱𖾏.
9 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑, 𖼖𖽔𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡, 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑, 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼁𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐、 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖽐𖼎𖽙 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐, 𖼋𖽔𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑.” 10 𖼽𖽙 𖼺𖽳 𖼄𖽔𖾐 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼎𖽺𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼋𖽪 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽔𖾏 𖼯𖽤𖾏?” 11 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼐𖽱𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏.
12 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼙𖽠𖾏, 𖼅𖽻𖾏 𖼾𖽨 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐. 13 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼎𖽙 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼾𖽨 𖼾𖽘𖾏 𖼙𖽠𖾏 𖼘𖽪? 𖼟𖽡 𖼋𖽔𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑.” 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑.” 14 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪, 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪, 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼗𖽙 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 𖼖𖽪𖾑 𖼦𖽘𖾑, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼙𖽠𖾏. 15 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽷𖾐 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖾏 𖼁𖽜𖾏 𖼙𖽠𖾏, 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼎𖽑𖽦𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏?” 16 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡, 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼾𖽨 𖼾𖽘𖾏?” 17 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼊𖽙𖾑 𖼊𖽑𖽗𖾑 ”． 𖼽𖽙 𖼺𖽳 𖼎𖽺𖾏 𖼖𖽠𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼊𖽙𖾑 𖼊𖽑𖽗𖾑. 18 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏, 𖼷𖽪𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼺𖽷𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑, 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼋𖽔. 19 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼑𖽔𖾏! 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼋𖽘𖾏. 20 𖼋𖽔𖾏 ！ 𖼀𖽡𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼐𖽑𖽙𖾏 𖼺𖽘𖾏, 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑 𖼺𖽘𖾏, 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: 𖼅𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏． 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼋𖽡𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖽃𖽡𖾑 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏?”
