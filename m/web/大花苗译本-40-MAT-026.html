26
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜，𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏： 2 “ 𖼅𖽡 𖼀𖽻𖾐，𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖽃𖽔𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼋𖽺 ， 𖼎𖽺𖾏 𖽐𖼮𖽦𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐．” 3 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽪𖾐 𖼗𖽻，𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼞𖽹𖾑 𖼽𖽔𖾏 𖼇𖽔 𖽐𖼚𖽙 𖼟𖽻 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏． 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏，𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽐𖼎𖽑𖽦𖾑 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏． 5 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖼷𖽪𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼊𖽡𖾐，𖽐𖼮𖽑𖽹𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽑𖽙𖾑 𖼁𖽔𖾏 𖼏𖽙．”
𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼮𖽑𖽠𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼞𖽑𖽙𖾐 14：3-9；𖼽𖽙 12：1-8）
6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼀𖽳 𖼊𖽔𖾏 𖼐𖽡 𖼳𖽡𖾑 𖼄𖽞 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏 𖽐𖼞𖽔𖾏， 7 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 ，𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖼽𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼛𖽙𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼋𖽔𖾏 ， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼘𖽡𖾑𖽑 𖼛𖽙𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏． 8 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 ： “ 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼞𖽡𖾏 𖼺𖽘𖾏！ 9 𖼋𖽡𖾏 𖼛𖽙𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼋𖽺 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼑𖽡𖽝𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏．” 10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏： “ 𖼡𖽔𖾐 𖼞𖽳𖾑 𖼊𖽡𖾐 ， 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼮𖽑𖽙 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖾑 𖼘𖽪？𖼐𖽱𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼻𖽻𖾏 𖼑𖽪𖾏． 11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽪𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐． 12 𖼑𖽡𖾏 𖼘𖽡𖾑𖽑 𖼋𖽡𖾏 𖼛𖽙𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽱𖾏． 13 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼋𖽻𖾏，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼡𖽑𖽔𖾐 𖼊𖽪𖾑 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 ，𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏． ”
𖼽𖽨 𖼊𖽔𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼞𖽑𖽙𖾐 14：10-11；𖼖𖽪𖾏 22：3-6）
14 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼮𖽢𖾑 𖼖𖽡𖽙𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽨 𖼊𖽔𖾏，𖼐𖽱𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐，𖼅𖽻𖾏 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼐𖽑𖽡𖾏： 15 “𖼞𖽪𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼅𖽡，𖼅𖽡 𖼯𖽪𖾏 𖼄𖽔𖾐 𖼀𖽡𖾑 𖼯𖽠 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑？” 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼙𖽡 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏． 16 𖼎𖽑𖽻𖾏 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 ， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼻𖽻𖾏 𖼯𖽡𖽹𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏
（𖼞𖽑𖽙𖾐 14：12-21；𖼖𖽪𖾏 22：7-14，21-23；𖼽𖽙 13：21-30）

36 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼞𖽑𖽳 𖼳𖽡𖾑 𖼄𖽔𖾐 𖼐𖽡，𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑，𖼋𖽙𖾏 𖼞𖽪𖾑 𖼚𖽔𖾏 𖼀𖽡 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏．” 37 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼊𖽑𖽹𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼙𖽠𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽳𖾏，𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝， 38 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ， 𖼸𖽙 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔，𖼅𖽡 𖽐𖼎𖽙𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑 𖼯𖽷𖾏 𖼻𖽙𖾑．” 39 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼏𖽔𖾏， 𖼚𖽑𖽪𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 ， 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼉𖽺𖾏 𖼘𖽪！𖼮𖽔𖾑 𖼟𖽪𖾏 𖼯𖽪𖾏 ，𖼟𖽙𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾑 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼮𖽑𖽡𖽻𖾏 𖼞𖽪𖾑 𖼙𖽠𖾏．𖽃𖽡𖾑 𖼈𖽡𖽝𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 ．” 40 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽷𖾐 𖼗𖽙𖾏 𖼁𖽙𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽻𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 ，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏 ： “𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼯𖽷𖾏 𖼻𖽙𖾑 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏？ 41 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽷𖾏 𖼻𖽙𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏， 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑．𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖼐𖽑𖽡𖽻𖾏 𖼯𖽡𖽙𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼯𖽻 𖼋𖽘𖾏．” 42 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼅𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼉𖽺𖾏 𖼘𖽪！𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼐𖽡𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼞𖽪𖾑，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽪𖾑 𖼦𖽻𖾏，𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑．” 43 𖼐𖽱𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼚𖽑𖽔𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 . 44 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 ，𖼷𖽷𖾐 𖼟𖽻𖾐 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼺𖽱𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐． 45 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼀𖽱𖾏 𖼳𖽙𖾏 𖼺𖽷𖾐 𖼘𖽪𖾏？𖼑𖽔𖾏 𖼘𖽪！ 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼊𖽻𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽡𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏． 46 𖼳𖽠𖾑 𖼗𖽙𖾏， 𖼀𖽡𖾐 𖼅𖽻 𖼽𖽔𖾐！𖼑𖽔𖾏 𖼘𖽪！𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽺 𖼞𖽪𖾑 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．”
