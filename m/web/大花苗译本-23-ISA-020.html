𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽔𖾐 𖽃𖽳𖾐 𖼞𖽺𖾑 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼳𖽷 𖼊𖽑𖽪 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼽𖽔𖾏 𖼳𖽷 𖼊𖽑𖽪 , 𖼦𖽔𖾏 𖼳𖽪𖾏 𖼊𖽻𖾏 𖼙𖽻. 2 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “ 𖼚𖽠𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼖𖽪𖾑 𖼚𖽔𖾑 , 𖼎𖽺𖾏 𖼚𖽠𖾏 𖼞𖽑𖽻𖾏 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼺𖽘𖾏.” 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏, 𖼯𖽡𖽙𖾏 𖼊𖽠𖾏 𖼊𖽡𖾏 𖼮𖽢𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑. 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏、 𖼯𖽡𖽙𖾏 𖼊𖽠𖾏 𖼊𖽡𖾏 𖼮𖽢𖾏 𖼅𖽻𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 𖼞𖽡𖾑, 𖼋𖽡𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖽐𖼎𖽙 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾐 𖼳𖽷. 4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏, 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼗𖽻𖾏 𖼘𖽑𖽔𖾏, 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏 𖼯𖽡𖽙𖾏 𖼊𖽠𖾏 𖼊𖽡𖾏 𖼮𖽢𖾏, 𖼢𖽳 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼊𖽡𖾐, 𖼟𖽙𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏. 5 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾐 𖼳𖽷 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖽃𖽡𖾑, 𖼣𖽹 𖼮𖽡 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼷𖽻𖾑 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
