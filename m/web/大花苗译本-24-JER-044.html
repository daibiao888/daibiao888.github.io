11 “𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼷𖽜𖾑 𖼚𖽑𖽪𖾏 𖼮𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼅𖽡, 𖼟𖽙𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. 12 𖼸𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼋𖽘 𖽐𖼮𖽢𖾏, 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾑 𖼊𖽡𖾏, 𖼊𖽻𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏, 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖽐𖼮𖽷𖾐 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼋𖽔 𖼙𖽠𖾏, 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏、𖼮𖽡𖽠𖾏 𖼏𖽘𖾏、𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 、𖼊𖽱𖾑 𖼷𖽑𖽷𖾑. 13 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏、𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏、𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼞𖽡𖾏, 𖼨𖽑𖽢𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐, 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽢𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑. 14 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑, 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙.”
