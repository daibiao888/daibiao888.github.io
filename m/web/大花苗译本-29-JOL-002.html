2
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼷𖽑𖽙𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼎𖽙𖾑 .
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼇𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏,
𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼋𖽘𖾏.
2 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏、 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏 ,
𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼊𖽔𖾐、 𖼚𖽪𖾐 𖼖𖽡𖾐 𖽐𖼮𖽡𖽝𖾐,
𖼺𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼎𖽻𖾐 𖼩𖽜𖾏.
𖼅𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻𖾏 𖼷𖽷𖾑 𖼷𖽷 𖼎𖽺𖾏 𖼘𖽑𖽙𖾐 𖼎𖽺𖾏 𖼋𖽻𖾏,
𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼗𖽙𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑,
𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼉𖽻𖾏 𖼷𖽑𖽷𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏.
 
3 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽻𖾐 ,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼸𖽙 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽱𖾏 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽔𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼸𖽠𖾏 𖽐𖼚𖽺𖾏 𖼋𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
 
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼑𖽜𖾏,
𖼑𖽜𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼏𖽘 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾏 𖼏𖽘 𖼖𖽡𖾏 𖼯𖽤𖾏 .
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽐𖼮𖽷𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼚𖽑𖽡𖽝𖾏 𖼚𖽜𖾑𖽑,
𖼺𖽱𖾏 𖼖𖽪𖾑 𖼀𖽑𖽪𖾐 𖼼𖽙𖾐 𖼑𖽡𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 ,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖽃𖽔𖾏 𖽐𖼊𖽻𖾏 𖼳𖽻𖾐 𖼑𖽡𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 ,
𖼺𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼏𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏, 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 .
6 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽡𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏,
𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼞𖽑𖽔𖾏 𖼚𖽑𖽡𖾏 𖼺𖽘𖾏.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽘 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼋𖽻 , 𖼺𖽱𖾏 𖼏𖽙 𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 .
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼅𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼞𖽡𖾑, 𖼦𖽡𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 .
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼦𖽡𖾐 𖼊𖽑𖽠𖾐 𖼦𖽡𖾏 𖽐𖼮𖽵𖾏,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼅𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼞𖽡𖾑 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏,
𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖽐𖼮𖽵 𖼚𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 ,
𖼦𖽡𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏.
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽵 𖼁𖽺𖾏 𖼙𖽻 ,
𖼚𖽑𖽡𖽝𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 ,
𖽐𖼞𖽘 𖽐𖼮𖽡𖾏 𖽐𖼞𖽔𖾏,
𖼁𖽺𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼸𖽜 .
 
10 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 ,
𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼷𖽻𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏 ,
𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 .
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼸𖽻𖾏 𖼏𖽙 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐,
𖼐𖽱𖾏 𖼏𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖽃𖽡𖾑.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏,
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼋𖽻𖾏 𖼖𖽺𖾑?
𖼞𖽑𖽻𖾑 𖼟𖽙𖾏 𖽐𖼎𖽻𖾏 𖼾𖽔𖾏 𖼚𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼡𖽜𖾑、 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏、 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑 .”
13 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼦𖽡𖾐 𖼾𖽙 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼅𖽡 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼅𖽳𖾏, 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐,
𖼦𖽡𖾐 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼅𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼎𖽺𖾏 𖼸𖽠𖾏 𖼎𖽙𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏,
𖼦𖽡𖾐 𖼮𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑.
14 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽱𖾏 𖼎𖽙𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼷𖽷𖾐 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼮𖽡𖽝𖾏,
𖼟𖽪𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
𖼊𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼾𖽨𖾏
15 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐,

17 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,

“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏,

18 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼳𖽙𖾑,
𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏.
19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽻 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼎𖽻𖾏 𖼅𖽡 , 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖽐𖼎𖽙 𖼛𖽙𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐,

𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏.
20 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼡𖽔𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼚𖽡𖾐,
𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽠𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏.
𖽃𖽡𖾐 𖼸𖽻 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 ,
𖽃𖽡𖾑 𖼸𖽻 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖽐𖼷𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼗𖽱𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼮𖽵𖾏 𖼑𖽡𖽻𖾏 𖼸𖽙 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏.”
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏
“ 21 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼘𖽪! 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏,
𖼽𖽔𖾑 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼋𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
22 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼘𖽪！𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏，

𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐,
𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼖𖽔𖾏 𖼷𖽷𖾏 𖼟𖽡𖾏 𖽃𖽔𖾏 𖽐𖼡𖽑𖽻𖾏.
 
23 𖼳𖽡 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪! 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼄𖽔𖾐 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐 𖼑𖽘 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼎𖽻𖾏 𖼅𖽡 ,
𖼑𖽜 𖼅𖽡 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼑𖽘 𖼗𖽙𖾏, 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐 𖼑𖽘 ,
𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼑𖽘 , 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
 
24 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼡𖽻𖾐,
𖼚𖽘𖾐 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼮𖽡𖽠𖾑 𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼛𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖽐𖼎𖽙 𖼛𖽙𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 .
25 𖼞𖽪𖾑 𖼸𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼙𖽘𖾑 𖼙𖽘𖾏 𖼷𖽷 𖼷𖽷、 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼷𖽷 𖼷𖽷、 𖼙𖽘𖾑 𖼙𖽘𖾏 𖼷𖽷 𖼟𖽻𖾏、 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼷𖽷 𖼟𖽻𖾏,

“ 28 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼖𖽡𖽾 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖽃𖽡𖾑, 𖼅𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏,
𖼅𖽡 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼁𖽙𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏,
𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏.
29 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼖𖽡𖽾 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼎𖽙 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑.
30 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼢𖽳 𖼊𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑, 𖼅𖽔𖾏 𖽐𖼮𖽑𖽘𖾑、 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏、 𖼅𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏. 31 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖼺𖽱𖾏 𖽐𖼮𖽑𖽘𖾑, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏,𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏.”
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖼮𖽵𖾑 𖽐𖼷𖽷𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏 𖽐𖼀𖽘𖾏
“ 32 𖼸𖽙 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏, 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏.”
