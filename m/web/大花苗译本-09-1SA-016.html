𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽡 𖼑𖽜 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑? 𖼟𖽡 𖼄𖽔𖾐 𖼛𖽙𖾏 𖽐𖼊𖽡𖾏 𖼟𖽙𖾏 𖼀𖽪𖾑 𖼊𖽪𖾑 𖼞𖽪𖾐, 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼏𖽵 𖼟𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼗𖽙 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.” 2 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼅𖽻𖾏 𖼖𖽺𖾑? 𖼺𖽻𖾐 𖼘𖽙 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼐𖽙𖾑𖽑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽔𖾏 𖼞𖽪𖾑.” 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽤𖾐 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔 𖼑𖽡𖽝 𖼑𖽨𖾏 𖼅𖽻𖾏, 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 3 𖼟𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼟𖽙𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏. 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡, 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼐𖽱𖾏.” 4 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏, 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 , 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼗𖽻 𖽐𖼮𖽑𖽹𖾏 𖼟𖽡𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏. 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼟𖽪𖾏 𖼑𖽜 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽔𖾏 𖼖𖽔𖾏?” 5 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽪𖾏 𖼑𖽜 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽔𖾏 𖼯𖽡𖽙. 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼾𖽨𖾏, 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏.” 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼾𖽨𖾏, 𖼮𖽘𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏.
𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼚𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏
6 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑, 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏: 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐. 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼳𖽪𖾏 𖼢𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼺𖽹𖾑 𖼐𖽱𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼳𖽪𖾏 𖼢𖽻𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼑𖽔𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐.” 8 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼟𖽙𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽔 𖼊𖽔 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼺𖽹𖾑 𖼐𖽱𖾏.” 9 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼳𖽔𖾑 𖼄𖽔𖾐 𖼎𖽑𖽻𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏, 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼺𖽹𖾑 𖼐𖽱𖾏.” 10 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼨𖽑𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖼎𖽑𖽻𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏, 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼐𖽡𖾑 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽹𖾑 𖽃𖽡𖾑.” 11 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼞𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑 𖼖𖽔𖾏 ?” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼺𖽷𖾐, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼾𖽨 𖼾𖽘𖾏 𖼙𖽠𖾏.” 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏, 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼗𖽙𖾏, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏.” 12 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏． 𖼐𖽱𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐, 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼢𖽻𖾏, 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼻𖽻𖾏 𖽐𖼎𖽻. 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏, 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼚𖽜𖾑 𖼐𖽱𖾏.” 13 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼚𖽜𖾑, 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼚𖽜𖾑 𖼐𖽱𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼳𖽠𖾑 𖼁𖽱𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐. 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼖𖽔𖾑 𖼄𖽔𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏.

14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼇𖽳𖾑 𖼺𖽻𖾐 𖼘𖽙 𖼺𖽘𖾏, 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼟𖽡𖾏 𖼐𖽱𖾏. 15 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼟𖽡𖾏 𖼟𖽡. 16 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼸𖽠𖾏 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼋𖽔𖾏, 𖼋𖽙𖾏 𖼟𖽙𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼉𖽺𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏.” 17 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼸𖽠𖾏 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏, 𖼨𖽑𖽳𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑.” 18 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼙𖽔 𖼁𖽙𖾏 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖼙𖽻 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼻𖽻𖾏. 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼚𖽔𖾐, 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳, 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼻𖽻𖾏 𖽐𖼎𖽻, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐.” 19 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽻𖾐 𖼘𖽙 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼳𖽡𖾑 ,𖼐𖽑𖽡𖾏: “𖽐𖼮𖽢𖾏 𖼟𖽡 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼾𖽘𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑.” 20 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼀𖽡𖾑 𖼯𖽠 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐, 𖼊𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖼘𖽑𖽱𖾏 𖼮𖽡𖽠𖾑, 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼋𖽪 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏, 𖼮𖽹𖾑 𖼎𖽻𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙. 21 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐. 𖼺𖽻𖾐 𖼘𖽙 𖽃𖽺𖾐 𖼙𖽔 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏. 22 𖼺𖽻𖾐 𖼘𖽙 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽢𖾏 𖼟𖽡 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽻𖾏 𖼅𖽳𖾏 𖼋𖽘𖾏.” 23 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼢𖽜𖾏 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖽐𖼎𖽷𖾑, 𖼺𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼙𖽠𖾏.
