𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏
4 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼮𖽡 𖼚𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼎𖽑𖽻𖾏 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑 𖼸𖽙 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 , 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 5 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 𖽃𖽡𖾑 , 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 , 6 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼑𖽔𖾏 𖼞𖽡𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 , 𖼦𖽡𖾐 𖼾𖽙 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼊𖽪𖾏 𖼞𖽡𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 . 7 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼑𖽔𖾏 𖼞𖽡𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 , 𖼯𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 .”
