8 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 , 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐,
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼡𖽔 𖼋𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼡𖽙𖾏 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏 𖼖𖽺𖾑 ?
9 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼑𖽡𖾏 𖼮𖽑𖽙𖾏 ,
𖼯𖽡𖽙𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖼀𖽑𖽙𖾑 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 ,
10 𖼑𖽜 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽐𖼚𖽜𖾏 ,
𖼎𖽺𖾏 𖽃𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼎𖽪𖾑 .
11 𖼐𖽑𖽡𖾏 : 𖼟𖽡 𖼮𖽔𖾑 𖼸𖽙 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼊𖽔𖾐 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏,
𖼟𖽡 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼸𖽙 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼯𖽡𖽙𖾏 𖼦𖽡𖾑 .
 
12 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼗𖽙𖾏 𖼋𖽡𖽝𖾏 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼑𖽡𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 ,
𖼟𖽙𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼑𖽡𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼀𖽻𖾐 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 ,
13 𖼟𖽙𖾏 𖼋𖽡𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼐𖽡𖾑 𖼊𖽔𖾑 𖼮𖽡𖽝𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 ,
𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼘𖽪 ?
14 𖼑𖽜 𖼋𖽡𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼐𖽡𖾑，𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼚𖽑𖽡𖾏 𖼺𖽱𖾏 𖼸𖽺𖾏 𖼖𖽪𖾑 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖼎𖽻𖾏 ,
𖼉𖽻𖾏 𖼮𖽘𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼺𖽱𖾏 𖼮𖽑𖽙𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 .
