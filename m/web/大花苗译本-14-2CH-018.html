𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 22： 1 - 28）
18
𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼊𖽻𖾏 𖽃𖽺𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 , 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼦𖽔𖾐 𖽃𖽔𖾏 𖼮𖽑𖽻𖾐 𖼟𖽙𖾏 . 2 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼳𖽡𖽻𖾏 , 𖼐𖽱𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 . 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽜 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼑𖽨𖾏 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 𖼞𖽑𖽻𖾑 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼾𖽨𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏𖼖𖽔𖾑 𖼄𖽙 . 3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽪 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖽝𖾏 𖼟𖽡 𖼯𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙 𖼖𖽺𖾑 ?” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽡 𖼉𖽺𖾏 , 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 .”
4 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼷𖽑𖽜𖾑 𖼟𖽡 𖼺𖽪𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 .” 5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼦𖽡 𖼸𖽱 , 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙 𖼋𖽡𖽝𖾏 𖼅𖽻𖾏 𖼢𖽻 𖼖𖽔𖾏 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼢𖽻 ?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “ 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼢𖽻 𖼯𖽡𖽙𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .” 6 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽑𖽡𖾏 : “𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼡𖽑𖽔𖾑 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼖𖽺𖾑 ?” 7 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽑𖽡𖾏 : “𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽷𖾐 , 𖼟𖽪𖾏 𖼽𖽡𖽾𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼄𖽡𖾐 𖼞𖽹𖾑 𖼽𖽔𖾐 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼐𖽱𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 , 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽔𖾐 .” 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽑𖽡𖾏 : “𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 .” 8 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏 𖼋𖽔𖾏 , 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼅𖽻𖾏 𖼨𖽑𖽪𖾏 𖼽𖽡𖽾𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼄𖽡𖾐 𖼞𖽹𖾑 𖼽𖽔𖾐 𖼋𖽔𖾏 .” 9 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 , 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏, 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼐𖽑𖽘𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽻𖾏 , 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 . 10 𖼮𖽡𖾑 𖼐𖽔 𖼐𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼋𖽪 𖼞𖽪𖾐 𖼖𖽑𖽻𖾏, 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼐𖽡𖾑 𖽐𖼎𖽻𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼸𖽙 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 .” 11 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙 𖼢𖽻 𖼯𖽡𖽙𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .”

23 𖼮𖽡𖾑 𖼐𖽔 𖼐𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼋𖽔𖾏 , 𖽐𖼊𖽻𖾏 𖼄𖽡𖾐 𖼞𖽹𖾑 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼚𖽑𖽪𖾏 , 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼖𖽺𖾑 ?” 24 𖼄𖽡𖾐 𖼞𖽹𖾑 𖼽𖽔𖾐 𖼐𖽑𖽡𖾏 : “𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏 .” 25 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼄𖽔𖾐 𖼄𖽡𖾐 𖼞𖽹𖾑 𖼽𖽔𖾐 𖼨𖽑𖽳𖾏 𖼺𖽷𖾐 𖼅𖽻 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼽𖽔𖾏 𖼄𖽞 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 , 𖼐𖽑𖽡𖾏 : 26 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼄𖽔𖾐 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼺𖽘𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 , 𖼑𖽻𖾏 𖼦𖽡𖾐 𖼮𖽻𖾏 𖼦𖽻𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 , 𖼋𖽙𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽪 𖼺𖽷𖾐 𖼗𖽙 .” 27 𖼄𖽡𖾐 𖼞𖽹𖾑 𖼽𖽔𖾐 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼋𖽡𖽝𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽪 𖼺𖽷𖾐 𖼗𖽙𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏 .” 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “ 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪 ! 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 ！”
𖼽𖽔𖾏 𖼦𖽔𖾐 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼋𖽔 𖼙𖽠𖾏
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 22： 29 - 35）
28 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 , 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙 𖼙𖽠𖾏 𖼋𖽘𖾏 . 29 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼘𖽑𖽪𖾑 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 , 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽑𖽘𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽙𖾏 .” 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪𖾑𖽑 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 .
30 𖼺𖽪𖾐 𖼁𖽱 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼐𖽑𖽡𖾏 : “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼋𖽻𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼘𖽑𖽙𖾐 𖼳𖽻𖾐 , 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 , 𖼮𖽔𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 𖼊𖽔𖾐 .” 31 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 , 𖼷𖽷𖾑 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖽃𖽡𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏 . 32 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼋𖽘𖾏 . 33 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼀𖽙𖾑 𖼐𖽜𖾑𖽑 , 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼀𖽙𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼷𖽷𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽘𖾐 . 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 :“ 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏 , 𖼟𖽡 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼗𖽙 , 𖼨𖽑𖽹𖾑 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼦𖽡𖾑!” 34 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖽐𖼮𖽷𖾏 𖽃𖽺𖾐 𖽐𖼊𖽻𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽜𖾏 𖼑𖽡𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼸𖽙 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 . 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼖𖽔𖾑 𖽐𖼷𖽜𖾑 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏 .
