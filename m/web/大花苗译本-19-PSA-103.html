𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫；
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑.
2 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏.
3 𖼐𖽱𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐.
4 𖼐𖽱𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼟𖽡 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖽃𖽔𖾏 𖼟𖽡 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑.
5 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼟𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽐𖼮𖽑𖽠𖾏 𖼽𖽔𖾑 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 ,
𖼟𖽙𖾏 𖼟𖽡 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼚𖽘𖾑 𖼗𖽻𖾏 𖼋𖽘𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼘𖽑𖽔𖾏.
 
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖼑𖽜 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖽃𖽡𖾑 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏.
7 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏.
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼅𖽔𖾏 𖼅𖽳𖾏,
𖼦𖽡𖾐 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏.
9 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼖𖽔𖾏 𖼡𖽑𖽔𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐.
10 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼀𖽡𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼯𖽡𖽙𖾏 𖼀𖽡𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼀𖽡𖾐 𖼾𖽔𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼀𖽡𖾐.
11 𖽐𖼊𖽪𖾏 𖼡𖽔𖾏 𖼊𖽡𖾐 𖼺𖽡𖽝𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 ,
𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼖𖽔𖾏 𖼘𖽑𖽙𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏！
12 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼡𖽔𖾏 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼚𖽡𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 ,
𖼐𖽱𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼾𖽔𖾏 𖼖𖽔𖾏 𖼡𖽔𖾏 𖼀𖽡𖾐 𖼚𖽡𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏！
13 𖼉𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏！
14 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽻𖾐 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼼𖽘𖾏,
𖽐𖼮𖽡𖽙𖾏 𖼸𖽙 𖼀𖽡𖾐 𖼟𖽪𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼊𖽔𖾐.
 
15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼺𖽱𖾏 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐,
𖼐𖽱𖾏 𖼘𖽑𖽘𖾑 𖽐𖼷𖽱𖾏 𖼺𖽱𖾏 𖼁𖽘𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏.
16 𖼏𖽻 𖼮𖽢𖾏 𖽂𖽪𖾏 𖼷𖽑𖽔𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘 ,
𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽪𖾐 𖼐𖽡𖽙𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏 .
17 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏,
𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼊𖽪𖾐 𖼘𖽑𖽔𖾏.
18 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑.
 
19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼮𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 (“ 𖼛𖽘𖾏 𖼼𖽙𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼞𖽪𖽙”) 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 .
20 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
21 𖼅𖽡 𖽃𖽔𖾏 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 ,
𖽃𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
22 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
