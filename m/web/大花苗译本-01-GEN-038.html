𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐
38
𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼽𖽨 𖼊𖽔𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼺𖽘𖾏 𖽐𖼡𖽜𖾏 𖼙𖽠𖾏, 𖼸𖽙 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽔𖾏 𖼊𖽪𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼳𖽡𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏. 2 𖼽𖽨 𖼊𖽔𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏, 𖼎𖽺𖾏 𖼷𖽙𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏. 3 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖽃𖽳𖾐. 4 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽤𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼊𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖽃𖽳 𖼐𖽔. 5 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼎𖽺𖾏 𖼊𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼳𖽷𖾏 𖼖𖽔𖾑． 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽻𖾏 𖼳𖽷𖾏 𖼖𖽔𖾑 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽡𖾑 𖼳𖽡.
6 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖽃𖽳𖾐 𖼷𖽙𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐. 7 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖽃𖽳𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼋𖽔 𖼙𖽠𖾏. 8 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖽃𖽳 𖼐𖽔 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼀𖽱𖾏,𖼸𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽱𖾑 𖼑𖽡𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼷𖽙𖾏 𖼐𖽱𖾏, 𖼑𖽜 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏.” 9 𖽃𖽳 𖼐𖽔 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽪𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖼁𖽡𖽝, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏. 10 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽳 𖼐𖽔 𖽃𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼋𖽔 𖼙𖽠𖾏. 11 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼖𖽔𖾑 𖼎𖽺𖾏 𖼋𖽔, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽔𖾑 𖼅𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼐𖽤𖾐 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼞𖽔𖾏 𖼻𖽙𖾑 𖼟𖽡 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼙𖽻 𖼋𖽪𖾏 , 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼳𖽷𖾏 𖼖𖽔𖾑 𖼘𖽑𖽙𖾐.” 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼞𖽔𖾏 𖼗𖽠𖾏.
12 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏, 𖼽𖽨 𖼊𖽔𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼋𖽔 𖼙𖽠𖾏, 𖼽𖽨 𖼊𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽡𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 𖼽𖽔𖾏 𖼊𖽪𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽡𖾑 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽡 𖼐𖽔, 𖽐𖼮𖽡𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼖𖽻𖾑 𖼚𖽻𖾐 𖼾𖽘𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑. 13 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼾𖽠𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼊𖽑𖽡 𖼐𖽔 𖼅𖽻𖾏 𖼖𖽻𖾑 𖼚𖽻𖾐 𖼾𖽘𖾐 𖼙𖽠𖾏.” 14 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼖𖽔𖾑 𖼘𖽑𖽙𖾐 𖼊𖽡𖽝𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽘𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼷𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐, 𖼎𖽺𖾏 𖼚𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏, 𖼘𖽑𖽪𖾑 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑, 𖼞𖽑𖽡𖽠𖾏 𖼛𖽔𖾏 𖽐𖼷𖽔𖾑 𖼚𖽑𖽪𖾏 𖽐𖼀𖽙 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏, 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼐𖽔 𖼽𖽡𖽾𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻, 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼊𖽑𖽡 𖼐𖽔. 15 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏, 𖼸𖽙𖾏 𖼞𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼀𖽙 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏. 16 𖼽𖽨 𖼊𖽔𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽔𖾏! 𖼯𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼀𖽱𖾏.” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼐𖽤𖾐. 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼐𖽑𖽡𖾏: “ 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼀𖽱𖾏, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼘𖽪?” 17 𖼽𖽨 𖼊𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖽐𖼀𖽙𖾏 𖼮𖽑𖽷𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼮𖽤𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏, 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡.” 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼺𖽘𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼟𖽡 𖼯𖽪𖾏 𖼄𖽔𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼸𖽺𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼮𖽡𖽝𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏.” 18 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼖𖽔𖾑 𖼄𖽔𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽪𖾏?” 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼖𖽪𖾑 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼸𖽙𖾏 𖼘𖽑𖽘𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼖𖽪𖾑 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏, 𖽐𖼎𖽙 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏.” 𖼽𖽨 𖼊𖽔𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏. 19 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼞𖽑𖽡𖽠𖾏 𖼛𖽔𖾏 𖽐𖼷𖽔𖾑 𖼚𖽑𖽪𖾏 𖼺𖽘𖾏, 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽑𖽘𖾑 𖼐𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼮𖽑𖽙𖾏.
20 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼊𖽪𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽡𖾑 𖼖𖽔𖾑, 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏 𖼅𖽻𖾏, 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼗𖽙𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽱𖾏. 21 𖼎𖽺𖾏 𖼑𖽪 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖽐𖼞𖽔𖾏 𖼁𖽠 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼐𖽔 𖼽𖽡𖽾𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 𖽃𖽡𖾑 𖽐𖼞𖽔𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼅𖽔𖾏 𖽐𖼞𖽔𖾏 𖼁𖽠 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏.” 22 𖼐𖽱𖾏 𖼺𖽷𖾐 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽱𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼞𖽔𖾏 𖼁𖽠 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏.” 23 𖼽𖽨 𖼊𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼺𖽘𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏 𖼐𖽡𖾑 𖼅𖽻𖾏 𖼋𖽘𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽱𖾏, 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼅𖽻, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽡 𖼞𖽪𖾑 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.”
𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖽐𖼷𖽹𖾐
24 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼖𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽨 𖼊𖽔𖾏, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼖𖽔𖾑 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼑𖽪𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏, 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏.” 𖼽𖽨 𖼊𖽔𖾏 𖼐𖽑𖽡𖾏:“ 𖼮𖽤𖾐 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏.” 25 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼮𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼁𖽡𖽝, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼯𖽡𖽙𖾏.”𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:“ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼯𖽜𖾏 𖽃𖽡𖾐 𖼯𖽜𖾏, 𖼖𖽪𖾑 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼸𖽙𖾏 𖼘𖽑𖽘𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼐𖽡𖾑, 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼁𖽡𖽝?” 26 𖼽𖽨 𖼊𖽔𖾏 𖼾𖽨𖾏 𖼯𖽜𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼳𖽷𖾏 𖼖𖽔𖾑.” 𖼎𖽑𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼽𖽨 𖼊𖽔𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏 𖼋𖽘𖾏.
27 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼇𖽺𖾏 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼋𖽘, 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼟𖽪𖾏 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖽃𖽔𖾑 𖽐𖼷𖽹𖾐. 28 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽪𖾐 𖼳𖽤𖾐 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼗𖽙𖾏, 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼷𖽹𖾏 𖼊𖽪𖾐 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼷𖽙𖾏 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼘𖽑𖽙𖾐.” 29 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼞𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏, 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼊𖽪𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽔𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼘𖽪?” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼇𖽔 𖼖𖽳 𖼺𖽷𖾑 (“𖼇𖽔 𖼖𖽳 𖼺𖽷𖾑” 𖼟𖽪𖾏 “𖼦𖽔𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏”). 30 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼐𖽱𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽹𖾐 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖽃𖽡𖾑, 𖼳𖽻𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼊𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼳𖽡𖽝𖾏 𖼖𖽔𖾑 (“𖼳𖽡𖽝𖾏 𖼖𖽔𖾐”𖼟𖽪𖾏 “𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐”).
