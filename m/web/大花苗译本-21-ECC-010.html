10
𖼄𖽙𖾑 𖼄𖽙𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼚𖽜𖾑 𖼅𖽔𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼗𖽱𖾏,
𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼡𖽔 𖼏𖽔 𖼸𖽠𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖽐𖼚𖽻 𖼮𖽡𖾏.
2 𖼊𖽪𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽹𖾑 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏,
𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽹𖾑 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏.
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼋𖽙𖾏 𖼞𖽡𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼡𖽔 𖼏𖽔 ,
𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 .
4 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼟𖽡 ,
𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼄𖽔𖾏 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏 𖼑𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
5 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼺𖽱𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏，
6 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐,
𖼊𖽪𖾐 𖼅𖽔𖾐 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼢𖽜𖾏.
7 𖼞𖽪𖾑 𖼙𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽜𖾑 𖼯𖽡𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏,
𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽪𖾐 𖽐𖼊𖽠𖾏 𖽐𖼷𖽷𖾏 𖼯𖽡𖾏 𖼺𖽱𖾏 𖼡𖽑𖽜𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖼘𖽑𖽻𖾑 𖽃𖽡𖾑, 𖼾𖽨𖾏 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼙𖽠𖾏,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 𖽃𖽡𖾑, 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼐𖽱𖾏.
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽻 𖽃𖽔𖾑 𖼈𖽳𖾐 ( “𖼸𖽻” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼦𖽡𖾏 𖼚𖽙𖾑” ) 𖽃𖽡𖾑, 𖼏𖽻 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽑𖽔𖾏 𖼋𖽠𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼑𖽪𖾏 𖼦𖽡𖾐 𖼚𖽑𖽡𖽝𖾐.
10 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼖𖽑𖽻𖾏 𖼦𖽡𖾐 𖽐𖼮𖽳𖾏 𖼋𖽘𖾏, 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼨𖽑𖽙𖾑 𖼟𖽙𖾏 𖽐𖼮𖽳𖾏, 𖼽𖽔𖾑 𖼮𖽡𖽻𖾐 𖽐𖼞𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
11 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽃𖽔𖾏 𖼐𖽜𖾐 𖼺𖽷𖾐, 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼮𖽔𖾑 𖼺𖽪𖾐 𖽐𖼮𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏,
𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏.
𖼊𖽪𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼅𖽳𖾏
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼅𖽳𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼾𖽨𖾏.
13 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽪𖾐 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼡𖽔 𖼏𖽔 ,
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼙𖽙𖾏 𖽐𖼊𖽪𖾏.
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏．
𖼐𖽱𖾏 𖼋𖽔 𖼙𖽠𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼸𖽠𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼖𖽺𖾑?
15 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼡𖽔 𖼏𖽔, 𖼐𖽱𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼟𖽙𖾏 𖼾𖽨𖾏 𖼯𖽡𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻 , 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐.
 
16 𖼞𖽪𖽙 𖼘𖽪！ 𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏,
𖼟𖽡 𖽐𖼀𖽙𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼳𖽠𖾑 𖼊𖽔𖾐 𖽐𖼷𖽙𖾑 𖼷𖽹𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼟𖽡 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！
17 𖼞𖽪𖽙 𖼘𖽪！ 𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏,
𖼟𖽡 𖽐𖼀𖽙𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽐𖼷𖽷𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼦𖽡𖾐 𖼑𖽜 𖼢𖽻 𖼮𖽡𖽠𖾑,
𖼟𖽡 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
𖼦𖽔𖾑 𖼗𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼞𖽜
18 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽜 , 𖽃𖽔𖾏 𖼎𖽪𖾏 𖽐𖼞𖽔𖾏 𖼎𖽺𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼞𖽜𖾏, 𖽐𖼞𖽔𖾏 𖼎𖽺𖾏 𖼮𖽑𖽻𖾏 𖼙𖽻𖾏.
19 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏, 𖼟𖽪𖾏 𖼑𖽜 𖼙𖽔 𖼚𖽙𖾏,
𖼮𖽡𖽠𖾑 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼺𖽘𖾏 𖼉𖽻𖾏 𖼑𖽪𖾏.
20 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼀𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽡𖾑 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 .
