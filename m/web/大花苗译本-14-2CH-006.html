𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 8：12 - 21）
6
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽑𖽡𖾏 :
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 .
2 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼋𖽘𖾏 ,
𖽃𖽔𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 .”
3 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽑𖽪𖾏 𖼗𖽙𖾏 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼖𖽔𖾏 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 . 4 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏 ,𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼡𖽑𖽙𖾑 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 , 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏 . 5 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 , 𖼷𖽑𖽙𖾏 𖼷𖽷𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 , 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼺𖽹𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 6 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 ,𖼎𖽺𖾏 𖼺𖽹𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 .”
7 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽔 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 . 8 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏 : 𖼟𖽡 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼷𖽑𖽙𖾏 𖼷𖽷𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 , 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼻𖽻𖾏 . 9 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 , 𖼯𖽔 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼘𖽘𖾑𖽑 𖼾𖽨 , 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 .
10 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼙𖽔 𖼏𖽵 𖼋𖽘𖾏 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏 , 𖼎𖽺𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼋𖽘𖾏. 11 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽡𖽝𖾏 , 𖽐𖼚𖽙 𖼼𖽪𖾏 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 𖼟𖽪𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑.”
