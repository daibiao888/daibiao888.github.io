4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽔𖾐 𖼋𖽡𖽝,

6 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏,

7 𖼐𖽱𖾏 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏,

8 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽔𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏！
9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼅𖽔𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼋𖽘,
𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼷𖽑𖽙𖾏, 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏．
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏,
𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼚𖽔𖾏 𖼯𖽢𖾏.
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏,
