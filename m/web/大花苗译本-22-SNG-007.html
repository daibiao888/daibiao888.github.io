4 𖼟𖽡 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖽐𖼷𖽑𖽱𖾏 𖼟𖽻𖾏,
𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼺𖽱𖾏 𖼳𖽡𖾑 𖼳𖽷𖾏 𖼀𖽺𖾐 𖼀𖽔𖾑 𖼊𖽑𖽳 · 𖼖𖽔𖾑 𖼀𖽡𖾐 𖽃𖽔 𖼛𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽻𖾐,

5 𖼟𖽡 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼺𖽱𖾏 𖼮𖽢𖾑 𖼄𖽡 𖼇𖽻𖾑 𖼎𖽻𖾐,

6 𖼖𖽜𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏, 𖼟𖽡 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼋𖽡𖽝 !
