𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 , 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻 . 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖼽𖽔𖾏 𖼺𖽔𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼞𖽪𖽙 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼳𖽡𖽻𖾏 . 2 𖼽𖽔𖾏 𖼺𖽔𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 3 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼏𖽘 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 , 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼁𖽠 , 𖽐𖼊𖽙𖾑 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐, 4 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑； 5 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼋𖽠𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 (“𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼼𖽘𖾏”), 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 . 6 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽪𖾐 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 . 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼀𖽡𖾑 𖼯𖽠 𖼳𖽡𖽻𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼎𖽻𖾏 𖼐𖽱𖾏 . 7 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼋𖽡𖾏 𖼙𖽻 𖼐𖽡𖾑 , 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 、 𖽐𖼀𖽙 𖽐𖼊𖽑𖽘𖾐 、𖼊𖽡𖾏 𖽃𖽔 𖼛𖽻𖾏 、 𖽃𖽔𖾏 𖼎𖽪𖾑 𖽃𖽔 𖼛𖽻𖾏 . 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼀𖽡𖾐 𖼺𖽷𖾐 , 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽔𖾑 . 𖼀𖽡𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼀𖽡𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪 .” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼙𖽻 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 . 8 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼸𖽻𖾏 𖼏𖽙 , 𖼽𖽨 𖼊𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼄𖽑𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼉𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖽐𖼎𖽙 𖼀𖽙𖾑 𖼐𖽑𖽜𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼾𖽡𖾏 𖼉𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 . 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 .

9 𖼅𖽔𖾏 𖼞𖽪𖾐 𖼳𖽷 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖽝𖾏 𖼖𖽔𖾑 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼉𖽻𖾏 𖼏𖽙 , 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼗𖽪𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 , 𖼋𖽠𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼸𖽙 𖼋𖽘𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼳𖽔𖾑 . 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼷𖽹𖾏 𖽐𖼮𖽷𖾏 , 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼑𖽡𖾏 𖼳𖽡𖾐 𖼇𖽔 𖼖𖽪𖾑 𖼦𖽘𖾑 , 𖼊𖽱𖾐 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼊𖽑𖽔𖾑 𖽐𖼮𖽷𖾏 . 11 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼯𖽔 𖼅𖽔𖾏 𖼟𖽡 𖼸𖽠𖾏 𖽐𖼀𖽘𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏 , 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 . 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 , 𖼻𖽡𖽝𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼸𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼐𖽡𖾑 . 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼟𖽡 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼟𖽡 .”
12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼊𖽻𖾏 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏 . 13 𖼽𖽔𖾏 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼏𖽙 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽔𖾑 𖽃𖽳𖾐 . 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽻 𖼊𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 , 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼏𖽙 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 . 14 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖽐𖼚𖽔 𖼮𖽡𖾑 𖼖𖽔𖾑 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼙𖽻 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖽐𖼮𖽑𖽹𖾏 . 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼦𖽔𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 15 𖼎𖽺𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖽐𖼞𖽔𖾏 𖼯𖽦𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼘𖽙 𖼊𖽑𖽙𖾐 , 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏 .
