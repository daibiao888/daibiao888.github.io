𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 ,
𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏！
2 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑，
3 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼸𖽱𖾏 𖼸𖽘𖾏.
4 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼀𖽱𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏,
5 𖼋𖽙𖾏 𖼸𖽙 𖼞𖽪𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖽐𖼞𖽔𖾏 𖼐𖽡𖽙𖾐.”
 
6 𖼀𖽡𖾐 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼊𖽑𖽔𖾑 ,
𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖼋𖽘𖾏 (𖼽𖽔𖾏 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 13: 6 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼮𖽡𖾑 𖼖𖽡𖽝 · 𖼽𖽦𖾑 𖼖𖽡𖽾”).
7 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼡𖽔𖾑 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏.
 
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 ,
𖽐𖼎𖽙 𖼟𖽡 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏！
9 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼐𖽑𖽘𖾑 𖼮𖽡𖽝𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏！
10 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽑𖽪𖾏 𖼙𖽠𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼟𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑！
 
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼏𖽵 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 ,
𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 , 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼅𖽔𖾏 𖼗𖽙 𖽃𖽡𖾑
𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑.
12 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼮𖽔𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖽐𖼎𖽙 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑.”
 
13 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽹𖾑 𖼊𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 ,
𖽃𖽺𖾐 𖼙𖽔 𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐，
14 𖼐𖽑𖽡𖾏: “𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽔𖾑 .
15 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼟𖽙𖾏 𖼡𖽻𖾐 𖼡𖽜𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼟𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏 .
16 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼐𖽑𖽘𖾑 𖼮𖽡𖽝𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏！
17 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐,
𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼘𖽑𖽘𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼞𖽪𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 ,
𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖾏．
18 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼀𖽔𖾏 𖼐𖽑𖽘𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪 .”
