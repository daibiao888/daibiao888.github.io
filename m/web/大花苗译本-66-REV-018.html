𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼼𖽠𖾏 𖼢𖽻 𖼙𖽠𖾏
18
𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐， 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏，𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼮𖽡𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜． 2 𖼐𖽱𖾏 𖼊𖽑𖽱𖾏 𖼼𖽙 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：
“𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼢𖽻 𖼙𖽠𖾏 𖼋𖽘𖾏！ 𖼢𖽻 𖼙𖽠𖾏 𖼋𖽘𖾏！
𖼑𖽱𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼐𖽡𖽙𖾐 ，
𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼖𖽡𖽾 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼑𖽱𖾏 𖼡𖽑𖽙𖾑，
𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼑𖽡𖾏 𖼉𖽳𖾏 ，
𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐．
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼦𖽻𖾏 𖼐𖽱𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐，
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏；
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼷𖽑𖽷𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝．”
4 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽑𖽡𖾏：
“𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！𖼅𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏，
𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼋𖽡𖾏 𖼾𖽔𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽐𖼀𖽠𖾏，
𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔．
5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼮𖽳𖾐 𖼸𖽷𖾏 𖽐𖼊𖽪𖾏，
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖽃𖽔𖾏， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼊𖽱𖾏 𖼸𖽙 𖼋𖽘𖾏．
6 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏， 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏，
𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾐 𖽐𖼀𖽠 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏，
𖼯𖽡𖽙𖾏 𖼞𖽑𖽙𖾐 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽡𖽠𖾏 𖼮𖽡𖽠𖾑 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾐 𖽐𖼀𖽠 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏．
7 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼾𖽨𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼾𖽨𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼷𖽑𖽷𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑，
𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏：
‘ 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐，
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏，
𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐．
8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼊𖽔𖾐，𖼐𖽱𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏，
𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏、 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐、 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏．
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐．”
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝
9 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 、𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼷𖽑𖽷𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑，𖼁𖽙𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽜𖾏 𖼐𖽱𖾏 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏； 10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏， 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼐𖽑𖽡𖾏：
“𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！ 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，
𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 ！
𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼊𖽔𖾐，𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．”
11 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽜 𖼐𖽱𖾏 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽘； 12 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾐、𖼑𖽡𖽝𖾏、𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑、𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑、𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙、𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼘𖽑𖽘𖾐 𖽐𖼀𖽔、𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐、𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏、𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖽐𖼷𖽑𖽱𖾏 𖼟𖽻𖾏 𖽃𖽔𖾏、𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽠𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽔𖾏、𖼋𖽻𖾏、𖼖𖽑𖽻𖾏、𖽃𖽔𖾑 𖼈𖽳𖾐 𖼚𖽠𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 ， 13 𖽐𖼎𖽙 𖼀𖽡𖾏 𖼚𖽻𖾏 𖼵𖽪𖾐 𖼞𖽪𖽺𖾏、𖼋𖽻𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏、𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏、𖼛𖽙𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏、𖼵𖽪𖾐 𖼳𖽤𖾏、𖼮𖽡𖽠𖾑、𖼛𖽙𖾏、𖽃𖽔𖾑 𖼋𖽪 𖼳𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏、𖽃𖽔 𖼛𖽡𖾏 𖼳𖽻𖾐、𖼑𖽨𖾏、𖼾𖽘𖾏、𖼑𖽜𖾏 、 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐、𖼡𖽑𖽜𖾑、𖼊𖽜𖾑 𖼐𖽜𖾐．
