𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼘𖽪 !
𖼅𖽡 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 , 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 .
2 𖼷𖽪𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽠𖾏,
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼦𖽡𖾏 𖼑𖽡𖽝 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼺𖽱𖾏 𖼦𖽡𖾏 𖼁𖽘𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡 ,
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼮𖽡𖽻𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏,
𖼅𖽡 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏.
3 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 𖼘𖽪 !
𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 ,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑, 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏.
𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼺𖽔𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏,
𖼽𖽔𖾏 𖼳𖽷 𖼮𖽡𖾑 𖼖𖽪𖽝 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏.
𖼽𖽔𖾏 𖼳𖽷 𖼊𖽑𖽪 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼑𖽻𖾏 𖼳𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏,
𖼽𖽡𖾑 𖼞𖽳 𖼖𖽪𖽝 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼚𖽙𖾐 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼗𖽙 .
5 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼊𖽑𖽡 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪𖾏 𖼳𖽡𖽠𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 !
𖼮𖽢𖾑 𖼐𖽔 、 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼘𖽪 !
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼖𖽔𖾏 𖼟𖽡 , 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽻𖾐 𖼟𖽡 , 𖼸𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏.”
6 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽐𖼎𖽘𖾏 𖽐𖼡𖽜𖾐,
𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐, 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼯𖽦𖾐.
7 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼽𖽨 𖼊𖽔𖾏 𖼾𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼾𖽨 𖼯𖽦𖾏 ,
𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾑 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼳𖽷 𖼮𖽡𖾑 𖼖𖽪𖽝 𖼑𖽡𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙.
 
8 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑, 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏, 𖼦𖽔𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐.
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼏𖽵 𖼗𖽪𖾏：
𖼄𖽙𖾑 𖼽𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽱𖾏 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 ,
𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽱𖾏 𖽃𖽳 𖼄𖽙 𖼖𖽔𖾑 ,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼡𖽜𖾐 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏、 𖼡𖽑𖽙𖾑 𖽐𖼮𖽳𖾑 , 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼚𖽑𖽡𖾏 𖼊𖽡𖾐 𖼡𖽔𖾏 𖼁𖽙𖾏.
𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼞𖽪𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼳𖽙𖾏 𖽐𖼎𖽙𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.”
 
10 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼟𖽪𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑, 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐 𖼾𖽨𖾏,
𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏.
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼢𖽳 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼾𖽔 𖼀𖽡𖾏 𖼮𖽑𖽤𖾏 ,
𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼐𖽱𖾏 .
 
12 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪 !
𖼅𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏.
 
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏,
𖼖𖽻𖾐 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼳𖽪 ,
𖼟𖽙𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼎𖽺𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏.
14 𖽐𖼀𖽙𖾏 𖼯𖽦𖾐, 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 ( “𖼞𖽪𖽙” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “𖼷𖽑𖽔𖾏 𖼮𖽘𖾑”) 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏, 𖼽𖽔𖾑 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,

𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏,
