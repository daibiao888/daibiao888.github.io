𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼮𖽑𖽠𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 26：6-13；𖼞𖽑𖽙𖾐 14：3-9）
12
𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼀𖽳 𖼊𖽔𖾏 𖼐𖽡，𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽙𖾏 𖼖𖽔𖾑 𖼺𖽔𖾐 𖼖𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖽃𖽡𖾑． 2 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼸𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼄𖽔𖾐 𖼊𖽔𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼡𖽑𖽔𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼺𖽔𖾐 𖼖𖽪𖾏． 3 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼷𖽷𖾐 𖼙𖽡𖾏 𖽃𖽔𖾐 𖼸𖽡𖽝𖾐 𖼛𖽙𖾏 𖼐𖽔 𖼊𖽔 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏、 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏、𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑，𖼚𖽜𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏，𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼳𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏，𖼛𖽙𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼀𖽪𖾑 𖽐𖼞𖽔𖾏． 4 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼖𖽡𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽨 𖼊𖽔𖾏，𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑 ，𖼐𖽑𖽡𖾏： 5 “𖼋𖽡𖾏 𖼛𖽙𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼋𖽺 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼛𖽡𖾏 𖼑𖽡𖽝𖾏，𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼖𖽺𖾑？” 6 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼸𖽜，𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖼑𖽡𖽝𖾏，𖼯𖽪 𖼯𖽪𖾏 𖼑𖽡𖽝 𖼖𖽪𖾑 𖽐𖼚𖽙 𖼐𖽑𖽘𖾐 𖽃𖽡𖾑 𖼑𖽡𖽝𖾏． 7 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“𖼯𖽔𖾏 𖼐𖽱𖾏， 𖼐𖽱𖾏 𖼋𖽡𖾏 𖼛𖽙𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼊𖽱𖾏 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼯𖽡𖽙𖾏． 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼅𖽔𖾏 𖼞𖽪𖾑．”
9 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏，𖼦𖽡𖾐 𖼾𖽙 𖼑𖽜 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐， 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼖𖽔𖾑 𖼺𖽔𖾐 𖼖𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖽃𖽡𖾑． 10 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼅𖽡 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼖𖽔𖾑 𖼺𖽔𖾐 𖼖𖽪𖾏 𖼊𖽑𖽡𖽝𖾐． 11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼑𖽜 𖼖𖽔𖾑 𖼺𖽔𖾐 𖼖𖽪𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽐𖼮𖽜 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼙𖽠𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐
（𖼊𖽑𖽹𖾏 21：1-11；𖼞𖽑𖽙𖾐 11：1-11；𖼖𖽪𖾏 19：28-40）
12 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐，𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼋𖽔𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 ，𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐， 13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼺𖽱𖾐 𖼺𖽱𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 ，𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：
“ 𖼦𖽙 𖼺𖽔𖾐 𖼐𖽔，
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼮𖽵𖾑 𖽐𖼷𖽷𖾏 𖼋𖽔𖾏， 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏！”
14 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑，𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼯𖽡𖽜𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：
15 “𖼳𖽡 𖼣𖽔𖾑 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏！
𖼑𖽔𖾏 𖼘𖽪！𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔𖾏 𖼋𖽘，𖼯𖽡𖽜𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑．”
16 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽔𖾐 𖼁𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖽐𖼊𖽱𖾏 𖼸𖽙 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽡𖽝． 17 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐，𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖼖𖽔𖾑 𖼺𖽔𖾐 𖼖𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖽃𖽡𖾑，𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏． 18 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼐𖽡𖾑，𖼎𖽺𖾏 𖼅𖽻𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏． 19 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼑𖽔𖾏， 𖼅𖽡 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙，𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼞𖽑𖽔𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏．”
