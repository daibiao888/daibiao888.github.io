𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽵𖾐 𖽐𖼞𖽔𖾏 ， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼛𖽠𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏，𖼐𖽱𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼗𖽱𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑． 2 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼐𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼟𖽡 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 ？𖼄𖽔𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽵𖾏 𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼚𖽠𖾐 𖽐𖼮𖽑𖽦𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼯𖽵𖾐 𖽐𖼞𖽔𖾏 𖼑𖽪𖾏 𖼋𖽘𖾏 ． ’ 3 𖼷𖽷𖾑 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼮𖽵𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽪𖾑 ，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼑𖽪𖾏，𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏？𖽐𖼮𖽡𖽠𖾏 𖼊𖽡𖾐 𖼖𖽔𖾏 ，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏！ 𖼳𖽻𖾏 𖼡𖽜𖾑 𖼖𖽔𖾏 ，𖼊𖽑𖽱𖾏 𖼸𖽘𖾏！ 4 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼯𖽵𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼑𖽪𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 ，𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏．’ 5 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼏𖽷𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖽐𖼮𖽷𖾏，𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼨𖽑𖽪𖾏 𖼋𖽔𖾏，𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼏𖽷𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼀𖽡𖾑 𖼯𖽠 𖽐𖼮𖽷𖾏？’ 6 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ：‘ 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼊𖽑𖽻𖾐 𖼛𖽙𖾏．’（“𖼊𖽑𖽻𖾐” 𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼀𖽔𖾏 𖼊𖽑𖽳） 𖼊𖽪𖾐 𖼯𖽵𖾐 𖽐𖼞𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：‘𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖽐𖼊𖽠𖾑 𖽐𖼮𖽷𖾏 𖼗𖽙𖾏， 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏，𖼺𖽻𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼊𖽑𖽻𖾐．’ 7 𖼎𖽺𖾏 𖼑𖽪 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼏𖽷𖾏 𖼀𖽡𖾑 𖼯𖽠 𖽐𖼮𖽷𖾏？ 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ：‘𖽃𖽡𖾐 𖼀𖽔𖾏 𖼙𖽪 𖼳𖽻𖾐，’ 𖼊𖽪𖾐 𖼯𖽵𖾐 𖽐𖼞𖽔𖾏 𖼐𖽑𖽡𖾏：‘𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖽐𖼊𖽠𖾑 𖽐𖼮𖽷𖾏 𖼗𖽙𖾏， 𖼺𖽻𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏．’ 8 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼎𖽺𖾏 𖼷𖽻𖾑 𖼷𖽷𖾑 𖼊𖽪𖾐 𖼯𖽵𖾐 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽻𖾏 𖼊𖽘𖾐 ，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼑𖽪𖾏， 𖼀𖽻𖾏 𖼊𖽘𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏． 9 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔，𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼏𖽻 𖼋𖽘𖾏 ， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼷𖽹𖾏 𖼅𖽡 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐． 10 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐，𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 ，𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽑𖽡𖽝𖾐． 11 𖼅𖽡 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼋𖽪𖾏 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼦𖽡𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐， 𖼡𖽔 𖼋𖽱𖾏 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼋𖽪𖾏 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼎𖽻𖾏 𖼅𖽡 𖼯𖽵𖾏 𖼖𖽺𖾑？ 12 𖼅𖽡 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐， 𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼅𖽡 𖼾𖽨𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼺𖽷𖾐 𖼘𖽪？ 13 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼞𖽔𖾏 𖼡𖽑𖽜𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽵𖾑，𖼦𖽡𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼷𖽷𖾑 𖼐𖽡𖾑 𖽐𖼮𖽑𖽠𖾏 𖼷𖽷𖾑 𖼉𖽺𖾏，𖼎𖽺𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼋𖽻𖾏 𖼷𖽷𖾑 𖼐𖽡𖾑， 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼷𖽷𖾑 𖼉𖽺𖾏．𖼅𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼎𖽺𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼊𖽑𖽡𖽝𖾐．”
𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙
（𖼊𖽑𖽹𖾏 5：31-32；11：12-13；𖼞𖽑𖽙𖾐 10：11-12）
14 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽠𖾑 𖼉𖽳 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽙𖾑 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜，𖼎𖽺𖾏 𖼚𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑． 15 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐． 16 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐，𖼸𖽙 𖼽𖽙 𖼦𖽗 𖼊𖽔𖾐 𖼊𖽜𖾑，𖼋𖽳 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼡𖽑𖽔𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼁𖽺𖾏 𖼅𖽻𖾏． 17 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏，𖼖𖽡 𖼟𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼇𖽳𖾏 𖼺𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏． 18 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼐𖽤𖾐 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽺𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 ，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼗𖽔 𖽐𖼎𖽻𖾏 𖼺𖽘𖾏，𖼖𖽔𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏．”
