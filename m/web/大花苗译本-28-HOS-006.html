𖼞𖽑𖽻𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼮𖽵𖾑
6
𖼋𖽔𖾏 ! 𖼀𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽱𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼀𖽡𖾐 , 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏；
𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼄𖽙𖾐, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏.
2 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖽃𖽔𖾐 𖼐𖽑𖽪𖾐, 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼯𖽷𖾏 𖼗𖽙𖾏 ,
𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏,
𖼀𖽡𖾐 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏.
3 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 ,
𖼐𖽱𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼺𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼁𖽙𖾏 𖼞𖽡𖾑 ,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼺𖽱𖾏 𖼑𖽘 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼑𖽘 𖽐𖼊𖽪𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
 
4 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼘𖽪 ! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼯𖽡𖽙𖾏 𖼟𖽡 𖼖𖽺𖾑?
𖼽𖽨 𖼊𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼯𖽡𖽙𖾏 𖼟𖽡 𖼖𖽺𖾑?
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼺𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼦𖽘𖾑 𖼀𖽙𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼙𖽱 𖼟𖽪𖾏 𖽐𖼡𖽑𖽔𖾐 𖼟𖽪𖾏 𖼇𖽺𖾏.
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼗𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼺𖽱𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼢𖽳 𖼋𖽠𖾏.
6 𖼞𖽪𖾑 𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐,𖼦𖽡𖾐 𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 ,
𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 .
 
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼽𖽔𖾏 𖼊𖽘𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 .
8 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼙𖽻 ,
𖼊𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 𖼘𖽙𖾏.
9 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼸𖽜 𖽐𖼚𖽺𖾏 𖼮𖽡𖽝𖾏 𖼋𖽙𖾏,
𖼐𖽡𖽙𖾐 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼐𖽑𖽙𖾏 𖼋𖽔𖾏,
𖼋𖽡𖽝 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼋𖽘𖾏.
10 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼋𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼳𖽪𖾐,
𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽻𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼋𖽘𖾏.
 
11 𖼽𖽨 𖼊𖽔𖾏 𖼘𖽪 ! 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼟𖽡 𖼊𖽡𖾏 𖼏𖽵 𖽃𖽡𖾐 𖼋𖽪 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏.
