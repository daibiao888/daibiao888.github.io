“ 𖼽𖽙 𖼀𖽳 𖼘𖽪！ 𖼷𖽑𖽜𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 ,

4 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼷𖽷𖾏 𖼞𖽪𖾑 ,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼐𖽱𖾏 𖼀𖽘𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 .
5 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼸𖽙 𖼎𖽺𖾏 𖼊𖽡𖾐 𖼞𖽪𖾑 ,

6 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 ,
𖼞𖽪𖾑 𖼖𖽔𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼷𖽷𖾏 𖼊𖽡𖽝𖾑 .
7 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼚𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼟𖽡 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 .
 
8 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏 ,
𖼎𖽺𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼟𖽡 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼐𖽑𖽡𖾏 :
9 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽠𖾏 , 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 ,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 .
10 𖼑𖽔𖾏 𖼘𖽪 ！ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽑𖽦𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 ,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 .
11 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼚𖽑𖽙𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼮𖽡𖽝𖾏 ,
𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 .
 
12 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼊𖽡𖾐 𖼟𖽡 𖼐𖽑𖽡𖾏 , 𖼟𖽡 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 .
13 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼐𖽑𖽡𖾏 :
𖼦𖽡𖾐 𖼞𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏， 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐 𖼖𖽺𖾑 ?
14 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 , 𖽃𖽔𖾐 𖼟𖽻𖾐 ,
𖼈𖽡𖽝𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼦𖽡𖾐 𖼑𖽻 .
15 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 ,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖽐𖼎𖽙 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 ,
16 𖼡𖽑𖽜𖾐 𖼊𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 ,
𖼎𖽑𖽻𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ，
17 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 ,

18 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽝𖾏 , 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 ,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼦𖽡𖾐 𖼋𖽔 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 .
𖼽𖽡𖾑 𖼖𖽡𖾏 𖼦𖽪𖾏 𖼐𖽑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽢𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑

20 𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼳𖽪𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 ,
