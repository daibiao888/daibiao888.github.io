𖼊𖽔𖾏 𖽂𖽺𖾏 𖼮𖽔𖾑 𖼾𖽱 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏, 𖼺𖽹𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 2 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏, 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼷𖽷𖾐 𖼸𖽻, 𖽃𖽡𖾐 𖼸𖽻 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽙 𖼽𖽔 𖼋𖽡𖾏, 𖽃𖽡𖾐 𖼸𖽻 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾐 𖼖𖽪𖾐 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼽𖽔 𖼑𖽡𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼳𖽺𖾑 𖼋𖽡𖾏， 𖽃𖽡𖾐 𖼸𖽻 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼊𖽑𖽹𖾏 𖼋𖽡𖾏. 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼏𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏.” 3 𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼏𖽘 𖼀𖽜𖾑, 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼀𖽡𖾐 𖼋𖽔 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐, 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼋𖽻𖾏 𖼋𖽳𖾏 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼗𖽜𖾏, 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼙𖽻 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐．” 4 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼑𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼻𖽻𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖽃𖽔𖾑 𖽐𖼊𖽪, 𖼡𖽔𖾑 𖼷𖽑𖽡𖽝𖾐 𖼡𖽔𖾏 𖼀𖽔𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼦𖽡𖾐 𖼸𖽙𖾐 𖽃𖽔𖾏 𖼡𖽹𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 5 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽙 𖼽𖽔, 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼳𖽺𖾑 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼊𖽑𖽹𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼟𖽪𖾏 𖼻𖽻𖾏.” 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏. 6 𖼏𖽙 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼸𖽱𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖽐𖼚𖽙 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏. 7 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼢𖽻 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼉𖽻𖾏. 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼎𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼎𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾏 𖼻𖽻𖾏 𖽃𖽡𖾑, 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼐𖽡𖽝𖾑𖽑 𖽃𖽔𖾏 𖼏𖽦𖾏.

9 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖽐𖼮𖽷𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼅𖽡, 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼯𖽡𖽜𖾏 𖼊𖽪𖾑 𖼑𖽜𖾏 𖼼𖽘𖾏 𖽐𖼮𖽡𖽻𖾏 𖼇𖽘𖾑 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼐𖽱𖾏 𖼊𖽪𖾑 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼦𖽡𖾏 𖼚𖽱𖾏 𖼚𖽑𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼯𖽡 𖽐𖼊𖽻𖾏 𖼮𖽡𖽝𖾏, 𖼮𖽘𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼮𖽡𖽝𖾏, 𖼊𖽪𖾑 𖼑𖽜𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽜𖾏 𖼏𖽘 𖼙𖽠𖾏 𖼋𖽘𖾏. 10 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽙 𖼽𖽔 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼮𖽘𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼮𖽡𖽝𖾏.” 11 𖼽𖽙 𖼽𖽔 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼋𖽘𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼺𖽘𖾏, 𖼟𖽙𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼙𖽠𖾏 𖼖𖽺𖾑? 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼋𖽔 𖼐𖽱𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼟𖽻𖾏 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖽝𖾏、𖽃𖽡𖾐 𖼸𖽙𖾐 𖼘𖽑𖽘𖾐 𖼎𖽻𖾏 𖼟𖽡.” 12 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼽𖽙 𖼽𖽔 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖽝𖾏 𖼖𖽡𖾏 𖼖𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼦𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼐𖽙𖾑𖽑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼳𖽺𖾑 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼊𖽑𖽹𖾏 𖼐𖽑𖽡𖾏: “ 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽺𖾏 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖽃𖽡𖾑. 13 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼦𖽺𖾏 𖼐𖽱𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏, 𖼟𖽡 𖼾𖽨𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐. 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼖𖽔𖾏 𖼈𖽺𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.” 14 𖼽𖽙 𖼽𖽔 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼞𖽡𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏.” 𖼽𖽙 𖼽𖽔 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼷𖽷𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼄𖽑𖽪𖾑, 𖼷𖽪𖾑 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼐𖽡𖽙𖾐 𖼇𖽘𖾑 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼙𖽠𖾏 𖼺𖽷𖾐, 𖼎𖽺𖾏 𖼐𖽑𖽙𖾏 𖼊𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐. 15 𖼟𖽻𖾏 𖼗𖽜𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼽𖽙 𖼽𖽔 𖼯𖽡𖽙𖾏 𖼏𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼊𖽔𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼺𖽘𖾏. 16 𖼽𖽙 𖼽𖽔 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐 𖽐𖼎𖽳𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼖𖽠𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐. 17 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐， 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼷𖽱𖾏 𖽃𖽡𖾐 𖼚𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐. 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽘 𖼀𖽜𖾑 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 18 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼯𖽡𖽝𖾏 𖼺𖽷𖾐 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽘𖾑 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼮𖽡𖽝𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼷𖽷𖾐 𖽐𖼷𖽷𖾏 𖼮𖽡𖽝𖾏!” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼊𖽪𖾑 𖽐𖼷𖽷𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖽃𖽔𖾏 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏.
𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼄𖽻𖾏
19 𖼺𖽔𖾐 𖼊𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼄𖽔𖾐 𖼺𖽷𖾑 𖼐𖽑𖽡𖾏: “𖼯𖽪𖾏 𖼞𖽪𖾑 𖼏𖽘 𖼅𖽻𖾏, 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐.” 20 𖼽𖽙 𖼽𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏, 𖼋𖽙𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼋𖽔 𖼺𖽘𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏.” 21 𖼽𖽙 𖼽𖽔 𖽐𖼎𖽙 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.” 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼽𖽙 𖼽𖽔 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑, 𖼎𖽺𖾏 𖼏𖽘 𖼙𖽠𖾏 𖼋𖽘𖾏. 22 𖼺𖽔𖾐 𖼊𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼄𖽔𖾐 𖼺𖽷𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼽𖽔 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼮𖽑𖽙 𖼟𖽡 𖼯𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽘 𖼅𖽻𖾏.” 𖼽𖽙 𖼽𖽔 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼄𖽻𖾏 𖼐𖽡𖾑, 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼡𖽜𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼽𖽔𖾑 𖼏𖽘 𖼅𖽻𖾏 𖼖𖽺𖾑?” 23 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “ 𖼦𖽡𖾐 𖼞𖽳𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼏𖽘 𖼅𖽻𖾏.” 𖼽𖽙 𖼽𖽔 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼏𖽘 𖼅𖽻𖾏.” 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼄𖽔𖾐 𖼺𖽷𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖼏𖽘 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼏𖽘 𖼮𖽑𖽡𖽻𖾏 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏.
