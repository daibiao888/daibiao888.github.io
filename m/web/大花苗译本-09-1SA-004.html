12 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼖𖽜𖾐 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼏𖽘 𖼀𖽜𖾑, 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏, 𖽐𖼀𖽑𖽻𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏 𖼎𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼳𖽷𖾏 𖼘𖽙. 13 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏 𖽃𖽡𖾑, 𖼽𖽡𖾑 𖼖𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏, 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼊𖽳𖾏. 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻 𖼎𖽻𖾏 𖼄𖽻𖾏, 𖽃𖽡𖾐 𖼙𖽻 𖼙𖽱 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏. 14 𖼽𖽡𖾑 𖼖𖽡 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏, 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖽐𖼎𖽙𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾐 𖼐𖽔𖾑?” 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼞𖽑𖽜 𖼑𖽜 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡. 15 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽡𖾑 𖼖𖽡 𖼅𖽔𖾏 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 𖼋𖽘𖾏, 𖽃𖽔𖾑 𖼅𖽔𖾏 𖼡𖽑𖽜𖾐 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏. 16 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼖𖽡 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼗𖽙𖾏 𖼯𖽡𖽙𖾏， 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼀𖽜𖾑 𖼺𖽷𖾐 𖼗𖽙.” 𖼽𖽡𖾑 𖼖𖽡 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐 𖼘𖽪！ 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏?” 17 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼄𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼏𖽘 𖼀𖽜𖾑, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐! 𖼟𖽡 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼦𖽙 𖼇𖽪 𖼐𖽡、𖼇𖽺𖾑 𖼐𖽡 𖼦𖽔𖾐 𖼖𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏.” 18 𖼐𖽱𖾏 𖽂𖽪𖾏 𖼐𖽑𖽡𖾏 𖼸𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏, 𖼽𖽡𖾑 𖼖𖽡 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼖𖽙𖾑 𖼦𖽡𖾑 𖼚𖽘𖾐 𖼋𖽔 𖼙𖽠𖾏. 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽤𖾑. 𖼽𖽡𖾑 𖼖𖽡 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏.
19 𖼽𖽡𖾑 𖼖𖽡 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼇𖽺𖾑 𖼐𖽡 𖼦𖽔𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖼐𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼖𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼄𖽙𖾐, 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑 𖼾𖽨 𖼊𖽪𖾐. 20 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔 𖽃𖽡𖾑, 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏! 𖼟𖽡 𖼳𖽻𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼋𖽘.” 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐. 21 𖼐𖽱𖾏 𖼊𖽡𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽡𖾐 𖼮𖽢𖾑 𖼀𖽙, 𖼐𖽑𖽡𖾏: “𖽐𖼚𖽻 𖼮𖽡𖾏 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼋𖽘𖾏!” 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏, 𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼖𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼐𖽔𖾑. 22 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼚𖽻 𖼮𖽡𖾏 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏.”
