𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽪𖽝 、𖼐𖽔 𖼊𖽔 、𖼽𖽔𖾏 𖼀𖽡𖾐 𖼦𖽪𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼗𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼛𖽪𖾏 𖼇𖽻𖾑 . 2 𖼯𖽔 𖼅𖽔𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼸𖽠𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽔𖾐 , 𖼐𖽱𖾏 𖼸𖽻 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 , 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽔𖾏 .”
3 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 , 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏、𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽻 . 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 , 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 .” 4 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 .
𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼎𖽻𖾐 𖼮𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘 , 𖼸𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖾏 , 𖼷𖽑𖽙𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 . 5 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼅𖽻𖾏 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 6 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖽐𖼮𖽑𖽘𖾑 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼋𖽺𖾏 , 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼦𖽡 𖽐𖼊𖽔 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 . 7 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽠𖾑 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼟𖽡𖽠 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽻 . 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 , 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 .” 8 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼞𖽑𖽡𖽠𖾏 𖽐𖼮𖽑𖽘𖾑 𖼦𖽡 𖽐𖼊𖽔 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼑𖽔𖾏 ! 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽘𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 .”
9 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 、𖼐𖽔 𖼊𖽔 、𖼽𖽔𖾏 𖼀𖽡𖾐 𖼦𖽪𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼗𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼞𖽑𖽔𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 . 10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼀𖽔𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑 𖽐𖼷𖽔𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼨𖽑𖽪𖾑 . 11 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼸𖽻𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏； 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼎𖽺𖾏 𖼦𖽻𖾏 .
𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖽐𖼮𖽡𖾏 𖼎𖽻𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼊𖽑𖽡𖽝𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼺𖽻𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 , 𖼟𖽙𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 .” 13 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 . 14 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼗𖽻 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽙𖾏 , 𖼋𖽙𖾏 𖼸𖽙 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼗𖽙𖾏 , 𖼑𖽔𖾏 𖼘𖽪! 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 、𖼦𖽪𖾏 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 . 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 .”
15 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼅𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼮𖽡𖽝𖾏 . 16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼮𖽳 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼮𖽡𖽝𖾏 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 , 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 . 17 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 , 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼗𖽙𖾏 , 𖼼𖽘𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼯𖽡 . 18 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼄𖽑𖽙𖾏 .
