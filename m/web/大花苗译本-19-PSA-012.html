𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 “𖼾𖽡𖾏 𖼳𖽘𖾐 ” .
12
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘.
2 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑 𖼼𖽷𖾏 𖼼𖽙,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏, 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽡𖽝 𖼺𖽱𖾏 𖼛𖽙𖾏, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼦𖽡𖾏 𖼺𖽱𖾏.
 
3 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼚𖽡𖽝 𖽐𖼎𖽙 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼺𖽘𖾏.
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾐 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏,
𖼀𖽡𖾐 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼾𖽨𖾏 𖼁𖽡𖽝,
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼖𖽺𖾑?”
 
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼊𖽻𖾏 𖼷𖽑𖽪𖾑𖼷𖽡𖽝𖾏
𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼉𖽳 𖽃𖽡𖾑 𖼮𖽡𖽝𖾏.
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼷𖽙𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼷𖽑𖽷𖾑 𖼮𖽡𖽺𖾏 𖼳𖽤𖾏 𖼟𖽻𖾏.
 
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐.
8 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼮𖽑𖽳𖾑 𖼺𖽡𖽝𖾐
𖼯𖽡𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼊𖽠𖾏 𖼊𖽠𖾑 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑.
