𖼊𖽻𖾏 𖼮𖽨𖾏 𖼯𖽡𖽝𖾏 𖼐𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏，𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏.
30
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖼺𖽹𖾑 𖼮𖽘𖾏 𖼮𖽑𖽳𖾑 𖼺𖽡𖽝𖾐 𖼞𖽪𖾑, 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏.
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖼞𖽪𖾑 𖼙𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡, 𖼟𖽡 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏.
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼙𖽔 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼮𖽨𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖼎𖽑𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼙𖽠𖾏 𖽃𖽡𖾑, 𖼮𖽨𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏.
 
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏,
𖼮𖽑𖽔𖾏 𖼙𖽔 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏.
5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽡𖾐 𖽐𖼮𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽔𖾐,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏 𖼈𖽡𖽝𖾏,
𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼙𖽔 𖼚𖽙𖾏.
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪, 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙.”
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼙𖽔 𖼺𖽘𖾏 𖼅𖽳𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼺𖽱𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼟𖽡 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏.
 
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡.
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐, 𖼐𖽑𖽡𖾏:
9 “𖼞𖽪𖾑 𖼊𖽻𖾏 𖼟𖽡𖾏 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑, 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑, 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪?
𖽃𖽘𖾑 𖼡𖽔 𖼗𖽔𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡, 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼘𖽪?
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼊𖽡𖾐, 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑！
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑！”
 
11 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏,
𖼚𖽠𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐, 𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼀𖽔𖾏 𖼐𖽑𖽘𖾑，
12 𖼻𖽻𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖽾 (“𖼖𖽡𖽾”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼚𖽻 𖼮𖽡𖾏”) 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡, 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽘．
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡, 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏！
