2 𖼄𖽔𖾐 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼑𖽡𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 ,
𖼄𖽔𖾐 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏，
3 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼟𖽻𖾏 𖼯𖽪𖾏 𖽐𖼚𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 𖼺𖽱𖾏 𖽃𖽻𖾐,

5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
𖼟𖽡 𖼽𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼟𖽡 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼽𖽔𖾑 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼯𖽡 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
6 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼘𖽑𖽡𖾑 𖽐𖼮𖽢𖾑𖽑 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽡 ,
𖽐𖼎𖽙 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏.
7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 ,

8 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼗𖽙𖾏 𖼸𖽱𖾏 𖼀𖽡𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼢𖽜𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏.
9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼘𖽪， 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐！
𖼑𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼐𖽔𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐, 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼀𖽡𖾐 𖼾𖽔𖾏.

11 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼞𖽱𖾏 𖼮𖽡𖽝𖾏 𖼑𖽡𖾏 𖼢𖽔𖾏 𖽐𖼷𖽘𖾏 𖽐𖼎𖽙𖾑 𖼸𖽙 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,

𖼊𖽑𖽡𖽝𖾐 𖼳𖽤𖾏 𖽐𖼀𖽠 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼟𖽪𖾏 𖼟𖽡 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼎𖽘𖾏 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐,
