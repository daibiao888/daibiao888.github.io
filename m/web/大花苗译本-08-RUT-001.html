1
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖽃𖽡𖾑, 𖼞𖽪𖽙 𖽐𖼚𖽙 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖽐𖼎𖽙 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 2 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼄𖽡𖾐 𖼘𖽳 , 𖼐𖽱𖾏 𖼐𖽤𖾐 𖽐𖼷𖽷𖾏𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 , 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖼖𖽪𖽝, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽷𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 , 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟, 𖼽𖽡𖾑 𖼇𖽔 𖼊𖽑𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐. 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 3 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙𖾏 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖼑𖽡𖾏 𖼈𖽻𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼄𖽡𖾐 𖼘𖽳 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼛𖽡𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐. 4 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼐𖽡𖾑 𖼷𖽙𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽷𖾏 𖽃𖽳 𖽃𖽳𖾐 𖼀𖽔𖾑 , 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽷𖾏 𖼘𖽪𖾏 𖼊𖽳 , 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼳𖽡𖽻𖾏. 5 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼄𖽔𖾐 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼖𖽔𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼛𖽡𖾏 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼈𖽻𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐.
6 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼎𖽑𖽻𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼺𖽷𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼺𖽘𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼇𖽳𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐, 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐. 8 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐 𖼐𖽑𖽡𖾏:“ 𖽃𖽔 𖼖𖽜𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼦𖽡𖾑! 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼯𖽡𖽙𖾏 𖼅𖽡, 𖼺𖽱𖾏 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼯𖽡𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐. 9 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼈𖽻𖾑 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖽐𖼞𖽔𖾏 𖼊𖽻𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼗𖽙𖾏 𖼳𖽦𖾑.” 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏, 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽻𖾐 𖽃𖽔𖾑 𖼸𖽷𖾏 𖼦𖽡𖾐 𖼇𖽺𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼯𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑， 10 𖼐𖽑𖽡𖾏:“ 𖼦𖽡𖾐 𖼾𖽙！ 𖼀𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽡𖾐 𖼸𖽻 𖼎𖽙𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙.” 11 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖼐𖽑𖽡𖾏:“ 𖼞𖽪𖾑 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪! 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼦𖽡𖾑! 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼖𖽺𖾑? 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼅𖽡 𖼈𖽻𖾑 𖼺𖽷𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏? 12 𖼞𖽪𖾑 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪! 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼦𖽡𖾑! 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼈𖽻𖾑 𖼋𖽘𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽑𖽡𖾏, 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾑 𖼅𖽔𖾏 𖼈𖽻𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐, 13 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼋𖽙𖾏 𖼋𖽻𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼘𖽑𖽙𖾐 𖼖𖽺𖾑? 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼋𖽙𖾏 𖼋𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼡𖽔𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪? 𖼞𖽪𖾑 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪! 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏, 𖼞𖽪𖾑 𖼳𖽦𖾑 𖽃𖽦𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼅𖽡, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼗𖽙𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼞𖽪𖾑.” 14 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐 𖼎𖽺𖾏 𖼯𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑, 𖽃𖽳 𖽃𖽳𖾐 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼁𖽙𖾐 𖼡𖽑𖽻𖾐 𖽃𖽔𖾑 𖼸𖽷𖾏 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼘𖽪𖾏 𖼊𖽳 𖼁𖽔 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼁𖽙𖾐 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼁𖽙𖾐 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 .
15 𖼐𖽔 𖽃𖽳 𖼄𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼟𖽡 𖽃𖽔𖾏 𖼻𖽷𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼻𖽷𖾏 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼦𖽡𖾑 !” 16 𖼘𖽪𖾏 𖼊𖽳 𖼐𖽑𖽡𖾏:
“𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼞𖽪𖾑 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐.
𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏,
𖼞𖽪𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏,
