45
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 𖼻𖽻𖾏 𖼗𖽪𖾏,
𖼞𖽪𖾑 𖼑𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽡𖽠 𖼞𖽪𖾑 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏,
𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼸𖽠𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼊𖽔𖾑 𖽐𖼚𖽘𖾏.
 
2 𖼟𖽡 𖼻𖽻𖾏 𖽐𖼎𖽻 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽔𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
3 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼎𖽹𖾏 𖼚𖽔𖾑 𖼎𖽑𖽹𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏,
𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼚𖽔𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
4 𖼑𖽜 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝、𖼳𖽻𖾐 𖼢𖽜、𖼖𖽠𖾑 𖼞𖽡𖾑, 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖽙𖾐 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼳𖽠𖾏,
𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼽𖽔𖾑 𖼡𖽑𖽳𖾏 𖼋𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏.
5 𖼟𖽡 𖼊𖽪𖾑 𖼈𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼮𖽳𖾏, 𖼮𖽑𖽡𖽙𖾐 𖼏𖽻 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼟𖽡 𖼀𖽡𖾑 𖼊𖽡𖾐.
 
6 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏,
𖼟𖽡 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑.
 
7 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑, 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼚𖽜𖾑 𖼟𖽡,
𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼚𖽜𖾑 𖼟𖽡 𖼽𖽺 𖼗𖽔.
8 𖼟𖽡 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼅𖽔𖾏 𖼄𖽪 𖼽𖽙、𖼮𖽑𖽹 𖼳𖽤𖾏、𖼵𖽪𖾐 𖼞𖽪𖽺𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼦𖽘𖾏,
𖼐𖽡𖽙𖾐 𖼟𖽻 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼷𖽑𖽱𖾏 𖼟𖽻𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖽐𖼚𖽙 𖼅𖽔𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽷𖾑 𖼺𖽙𖾑 𖼑𖽡𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑, 𖼟𖽙𖾏 𖼟𖽡 𖼙𖽔 𖼚𖽙𖾏.
9 𖼟𖽡 𖽐𖼀𖽙𖾏 𖼣𖽔𖾏 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏,
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼐𖽑𖽘𖾑 𖽐𖼊𖽻𖾏 𖽃𖽳 𖼇𖽺𖾑 𖼞𖽪𖾐 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼺𖽘𖾑 𖼺𖽷𖾏.
 
10 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻, 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖼽𖽔𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻！
𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏，
11 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼉𖽳 𖼟𖽪𖾏 𖼟𖽡 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽵𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼐𖽱𖾏.
12 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 ( “𖼖𖽡𖾑 𖼉𖽪𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏”) 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼺𖽘𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼊𖽜 𖼅𖽳𖾏.
 
13 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼚𖽙 𖼟𖽻 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼐𖽱𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼞𖽪𖾐 𖼳𖽡𖾑 𖼯𖽡𖽙𖾏.
14 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼁𖽘𖾏 𖼎𖽻𖾏, 𖼊𖽻𖾏 𖼨𖽑𖽳𖾏 𖼚𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖽐𖼞𖽻𖾏 𖽐𖼮𖽡𖽻 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼗𖽔 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽳𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻.
 
16 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐,
𖼟𖽡 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
17 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽷𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏.
