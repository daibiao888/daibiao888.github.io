𖼽𖽨 𖼊𖽔𖾏 𖼺𖽻𖾏 𖼋𖽔
（𖼞𖽑𖽙𖾐 15：1-2；𖼽𖽙 18：28-32）
27
𖼸𖽙 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼅𖽡 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼗𖽻，𖼦𖽡𖾏 𖼊𖽳𖾏 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽔． 2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼚𖽔𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑．
3 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑， 𖼽𖽨 𖼊𖽔𖾏 𖼟𖽪𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽻𖾏 𖼊𖽡𖾏 𖼾𖽔𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏，𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼑𖽜，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼙𖽡 𖼑𖽡𖽝𖾏 𖼎𖽙𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼗𖽻 ， 𖼐𖽑𖽡𖾏： 4 “ 𖼞𖽪𖾑 𖼋𖽺 𖼊𖽜𖾑 𖼐𖽜𖾐（“𖼊𖽜𖾑 𖼐𖽜𖾐” 𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼮𖽑𖽘𖾑”） 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏，𖼟𖽪𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼋𖽘𖾏 ．” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 ：“𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼞𖽳𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼾𖽨𖾏 𖽃𖽔𖾏．” 5 𖼽𖽨 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼑𖽡𖽝𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼺𖽘𖾏，𖼋𖽠𖾏 𖼅𖽻𖾏 𖼺𖽻𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏． 6 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼷𖽑𖽷𖾑 𖼐𖽡𖾑， 𖼐𖽑𖽡𖾏 ： “𖼑𖽡𖽝𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽘𖾑 𖽐𖼡𖽜𖾏，𖼀𖽡𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼞𖽔𖾏 𖼼𖽪𖾏 𖽐𖼚𖽙．” 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏，𖼄𖽔𖾐 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼊𖽡𖾐，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽱𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼞𖽪𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐． 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 “𖼊𖽡𖾐 𖽐𖼮𖽑𖽘𖾑 ”． 9 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼗𖽪𖾏， 𖼐𖽑𖽡𖾏：“𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼙𖽡 𖼑𖽡𖽝𖾏，𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼸𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽺 𖼐𖽡𖾑 𖼑𖽡𖽝𖾏 𖽐𖼡𖽜𖾏， 10 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼊𖽡𖾐 ， 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼮𖽵𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏． ”
𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼳𖽹𖾐 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼞𖽑𖽙𖾐 15：2-5；𖼖𖽪𖾏 23：3-5；𖼽𖽙 18：33-38）
11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼳𖽠𖾑 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 ？”𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 ：“𖼟𖽡 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽙．” 12 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼗𖽻 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼐𖽱𖾏 ， 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐 𖼋𖽪 𖼡𖽔𖾏 𖼳𖽷𖾏 ． 13 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼐𖽑𖽙𖾑 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼛𖽠𖾏 𖼑𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 ？” 14 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏，𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝．
（𖼞𖽑𖽙𖾐 15：6-15；𖼖𖽪𖾏 23：13-25；𖼽𖽙 18：39-19：16）
15 𖼷𖽑𖽔𖾏 𖼟𖽻𖾏 𖼸𖽙 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼐𖽡𖾑，𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏 𖼊𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑，𖼮𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 16 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑． 17 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖽃𖽡𖾑，𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼽𖽔𖾑 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼎𖽻𖾏 𖼅𖽡？𖼟𖽪𖾏 𖼽𖽔𖾑 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑𖼖𖽔𖾏？ 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼮𖽡𖾑 𖼊𖽪 ？” 18 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼋𖽔𖾏． 19 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼐𖽤𖾐 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏，𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽳𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，𖼑𖽜 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼊𖽔𖾐 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖽃𖽦𖾐． 20 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼗𖽻 𖼙𖽳𖾐 𖼷𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑𖼺𖽘𖾏，𖼊𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑． 21 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑，𖼅𖽡 𖼽𖽔𖾑 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼎𖽻𖾏 𖼅𖽡？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑．” 22 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼷𖽷𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼮𖽡𖾑 𖼊𖽪 ，𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽑𖽡𖾏：“𖽐𖼮𖽦𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐！” 23 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽑𖽡𖾏：“𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼖𖽺𖾑？ 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼺𖽘𖾏 𖼐𖽔𖾑？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖽐𖼮𖽦𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐！” 24 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽔𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙，𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽙𖾑 𖽐𖼷𖽑𖽙𖾑 𖼑𖽪𖾏，𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽑𖽳𖾑 𖽃𖽻𖾐 𖼗𖽙 𖽐𖼷𖽔𖾑 𖼋𖽡𖾏， 𖼐𖽑𖽡𖾏：“𖽐𖼚𖽱𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼐𖽡𖾑 𖽐𖼮𖽑𖽘𖾑， 𖼾𖽔𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 ，𖼅𖽡 𖼾𖽨𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑．” 25 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼐𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑．” 26 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼎𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽦𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐．

45 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼮𖽳， 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼞𖽑𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼺𖽘𖾏 𖼸𖽙 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼐𖽑𖽪𖾐 𖼢𖽹𖾐． 46 𖼸𖽙 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼐𖽑𖽪𖾐 𖼢𖽹𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼽𖽡𖾑 𖼖𖽡！ 𖼽𖽡𖾑 𖼖𖽡！𖼖𖽔𖾑 𖼄𖽔𖾐 𖼺𖽔𖾐 𖼀𖽔𖾏 𖼞𖽙 𖼊𖽔𖾏 𖼐𖽡？” 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ！𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏！𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏？” 47 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼉𖽺𖾏 𖽃𖽡𖾑，𖼅𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏．” 48 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽘 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼳𖽡𖽠𖾑 𖼚𖽻𖾐，𖼚𖽻𖾏 𖼎𖽻𖾏 𖼮𖽡𖽠𖾑 𖼦𖽡𖾐 𖼡𖽻𖾐，𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽻𖾏． 49 𖼈𖽡𖽝𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏： “𖼋𖽙𖾏 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏，𖼀𖽡𖾐 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼋𖽔𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽔𖾏．” 50 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏，𖼊𖽪𖾏 𖼀𖽘𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏．
