𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼀𖽻𖾐 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏、𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼽𖽙 𖼦𖽗， 2 （𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑，𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 ）， 3 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼽𖽨 𖼊𖽑𖽹𖾏， 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏． 4 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏， 5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼳𖽱 𖼮𖽢𖾑， 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼺𖽳． 6 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼞𖽡𖾑 𖼚𖽑𖽔𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪．𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖽐𖼮𖽳．
7 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐，𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼄𖽔𖾐 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽻𖾏．” 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻 𖼅𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼙𖽠𖾏． 9 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼽𖽔𖾑 𖽃𖽻𖾐 𖼦𖽻𖾏 𖼖𖽺𖾑？” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽨𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏． 10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏：“ 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽳𖾏，𖼎𖽺𖾏 𖼀𖽻𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 ：‘ 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼄𖽔𖾐 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽻𖾏 ’ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏，𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼊𖽜，𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼎𖽻𖾏 𖼟𖽡．” 11 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑！ 𖼟𖽡 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽑𖽻𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐，𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼎𖽺𖾏 𖼊𖽙𖾐，𖼟𖽡 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼊𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼖𖽺𖾑？ 12 𖼟𖽡 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼺𖽷𖾐 𖼖𖽔𖾏？𖼐𖽱𖾏 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼀𖽡𖾐，𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼅𖽡 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼯𖽦𖾏 𖼊𖽪𖾏 𖼦𖽻𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖾑 𖽃𖽻𖾐．” 13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏：“ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽻𖾏 𖼋𖽡𖾏 𖽃𖽻𖾐 𖼐𖽡𖾑，𖼸𖽠𖾏 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼺𖽷𖾐， 14 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏，𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏．𖽃𖽻𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏，𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐， 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼸𖽙 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏．” 15 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑！ 𖼄𖽔𖾐 𖼋𖽡𖾏 𖽃𖽻𖾐 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑，𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑．”
