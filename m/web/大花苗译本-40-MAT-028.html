𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙
（𖼞𖽑𖽙𖾐 16：1-10；𖼖𖽪𖾏 24：1-12；𖼽𖽙 20：1-10）
28
𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼖𖽔𖾑 𖼋𖽘𖾏，𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼺𖽘𖾏 𖽐𖼊𖽪𖾏， 𖼄𖽙 𖼊𖽔𖾏 𖼖𖽔𖾑𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽡𖾑，𖼋𖽔𖾏 𖼑𖽔𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏． 2 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏，𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏，𖼦𖽡𖾏 𖼚𖽙𖾑 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼺𖽘𖾏，𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼳𖽔𖾐． 3 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼺𖽱𖾏 𖼺𖽙𖾐 𖼀𖽡𖾏 𖼗𖽺𖾏 𖽐𖼊𖽘𖾏，𖼮𖽑𖽙𖾏 𖼚𖽠𖾐 𖼺𖽱𖾏 𖽐𖼀𖽪𖾏． 4 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏，𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼟𖽪𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐． 5 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏！𖼞𖽪𖾑 𖼀𖽻𖾐 𖼅𖽡 𖽐𖼎𖽑𖽦𖾑 𖼷𖽷𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽦𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐． 6 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼋𖽘𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏．𖼅𖽡 𖼋𖽔𖾏 𖼑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖽃𖽡𖾑． 7 𖼎𖽺𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏，𖼐𖽑𖽡𖾏 ：‘𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼋𖽘𖾏．𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖽐𖼊𖽡𖾏 𖼚𖽔𖾏𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 ’．𖼑𖽔𖾏 𖼘𖽪！𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼋𖽘𖾏．” 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾐 𖼙𖽪𖾐 𖼞𖽡𖾏 𖼙𖽔 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼊𖽑𖽻𖾏 𖼯𖽪𖾏 𖽐𖼮𖽑𖽹𖾏，𖼊𖽑𖽻𖾏 𖼯𖽪𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼏𖽘 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏． 9 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖽃𖽡𖾑，𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼐𖽑𖽡𖾏：“𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪 ！” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼁𖽔 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏． 10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏！𖼅𖽡 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏，𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼞𖽪𖾑．”
𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑
11 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼅𖽡 𖼅𖽻𖾏 𖽃𖽡𖾑，𖼑𖽔𖾏 𖼘𖽪 ！ 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼦𖽡𖾑 𖽐𖼷𖽘𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻，𖼄𖽔𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼁𖽙𖾏 𖽃𖽡𖾑，𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 ． 12 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼗𖽻 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼐𖽑𖽡𖾏： 13 “𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：‘ 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖼑𖽡𖽝 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼑𖽜𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏．’ 14 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼅𖽔𖾏 𖼀𖽡𖾐 𖼞𖽑𖽻𖾑 𖼐𖽱𖾏， 𖼡𖽑𖽠𖾑 𖼅𖽡 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏．” 15 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼊𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽘，𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏．𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏
（𖼞𖽑𖽙𖾐 16：14-18；𖼖𖽪𖾏 24：36-49；𖼽𖽙 20：19-23；𖼎𖽻𖾏 1：6-8）
16 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏，𖽐𖼮𖽡𖽻𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼋𖽘𖾏． 17 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼷𖽑𖽠𖾑 𖽐𖼊𖽱𖾏． 18 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏． 19 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏，𖼞𖽑𖽻𖾑 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏，𖼎𖽺𖾏 𖼻𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 、𖼊𖽪𖾐 、𖼳𖽟𖾏 𖼖𖽡𖽾 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 20 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽙𖾑．𖼑𖽔𖾏 𖼘𖽪！𖼞𖽪𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽑𖽳𖾐 𖼷𖽑𖽷𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐，𖼸𖽙 𖼟𖽪𖾏 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏．”
