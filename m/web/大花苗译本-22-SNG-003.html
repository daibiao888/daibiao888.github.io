3
𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼸𖽘𖾏 𖼮𖽑𖽡𖽠𖾏,
𖽐𖼎𖽑𖽦𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏, 𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏.
2 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼮𖽑𖽙𖾑 𖼅𖽻𖾏 𖼮𖽑𖽙𖾑 𖼗𖽙𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼙𖽻 ,
𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼇𖽘𖾑,
𖽐𖼎𖽑𖽦𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑.”
𖼞𖽪𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏,𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏.
3 𖽐𖼚𖽙 𖼙𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡 𖽐𖼮𖽡 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼋𖽔𖾏 𖽐𖼮𖽷𖾐 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 :
“𖼋𖽡𖽝𖾏 𖼅𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑?”
4 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖽐𖼮𖽷𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑.
𖼞𖽪𖾑 𖼊𖽔𖾑 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏,
𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼞𖽔𖾏,
𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼖𖽜𖾑 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏.
5 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪！
𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼷𖽹𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼮𖽑𖽙, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼯𖽷𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑, 𖼋𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐(𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽪𖾏”).
𖼷𖽷𖾐 𖼸𖽙𖾐
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 :
6 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏, 𖼼𖽘𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏,
𖼎𖽑𖽻𖾏 𖼄𖽪 𖼽𖽙 𖽐𖼎𖽙 𖼵𖽪𖾐 𖼳𖽤𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼚𖽜𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽔𖾑 ?
7 𖼑𖽔𖾏 𖼘𖽪! 𖼟𖽪𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽡𖾏 𖼖𖽪𖾑 𖽐𖼮𖽑𖽘𖾏 𖼟𖽡𖾏,
𖼚𖽻𖾐 𖼺𖽘𖾐 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖼋𖽻 ,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼚𖽙 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼋𖽻.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽔𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏, 𖽃𖽺𖾐 𖼸𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 ,
𖼖𖽪𖾑 𖼚𖽔𖾑 𖼎𖽑𖽹𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏.
9 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖽐𖼊𖽻𖾏
𖼑𖽜 𖼾𖽨𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼮𖽑𖽘𖾏 𖼟𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐.
10 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖽐𖼮𖽑𖽘𖾏 𖼟𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏,
𖽐𖼮𖽑𖽘𖾏 𖼟𖽡𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖽃𖽔𖾏,
𖼮𖽑𖽡𖽠𖾏 𖼼𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼀𖽔𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏.
11 𖼳𖽡 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪 !
𖼅𖽡 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼖𖽡𖾏 𖼇𖽻𖾏 𖽐𖼊𖽻𖾏 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼷𖽙𖾏 𖼐𖽤𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼯𖽡𖽙𖾏.
