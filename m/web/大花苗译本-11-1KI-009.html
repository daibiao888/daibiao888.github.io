9
𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻, 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽑𖽠𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏 𖼋𖽘𖾏， 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖼢𖽳 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞, 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼀𖽡𖽝𖾏 𖼢𖽳 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐. 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽜 𖼽𖽔𖾑 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼞𖽑𖽔𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏. 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼐𖽡𖾑 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑. 4 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼾𖽻 𖼸𖽙𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝 𖽐𖼎𖽙 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡, 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖽐𖼎𖽙 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏， 5 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝, 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼞𖽪𖾑 𖼏𖽵 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽪𖾏 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐. 6 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼅𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼎𖽙𖾑 𖼺𖽷𖾏 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖽐𖼎𖽙 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡, 𖼅𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺， 7 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼾𖽨𖾏 𖽐𖼷𖽷𖾏 𖼇𖽺𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏, 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼚𖽙𖾏, 𖼊𖽻𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏. 8 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼈𖽡𖽝𖾏, 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼽𖽔𖾑 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏, 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏, 𖼐𖽑𖽡𖾏: ‘ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽡𖽝𖾐 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑?’ 9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: ‘𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼅𖽻𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺, 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼐𖽱𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.’ ”
𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼺𖽘𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼖𖽔
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 8：1 - 2）
10 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻, 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼟𖽻 𖽐𖼞𖽔𖾏 𖼐𖽡𖾑, 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑. 11 ( 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼖𖽔 𖼙𖽔 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼡𖽑𖽙𖾑 𖼽𖽔𖾑 𖽃𖽡𖾑, 𖼺𖽘𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏、 𖽐𖼊𖽻𖾏 𖼊𖽑𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾐 ), 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡 𖼖𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼖𖽔. 12 𖼳𖽡𖾑 𖼖𖽔 𖼎𖽑𖽻𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐, 𖼐𖽑𖽡𖾏: 13 “𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼘𖽪！ 𖼙𖽻 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼙𖽻 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?” 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼙𖽻 𖼐𖽡𖾑 𖼊𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼮𖽢𖾑 𖼀𖽪𖾏 𖼖𖽳, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏. 14 𖼳𖽡𖾑 𖼖𖽔 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 𖼞𖽪𖾐 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏.
