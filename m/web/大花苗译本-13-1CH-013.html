𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼦𖽡𖾏 𖼊𖽳𖾏 . 2 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼮𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 , 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽡𖾏 𖼯𖽡𖾏 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼁𖽙𖾏 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽔𖾏 , 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼯𖽦𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 . 3 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽹𖾑 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼙𖽙𖾏 𖼚𖽔𖾏 𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .” 4 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼖𖽔𖾏 𖼐𖽑𖽡𖾏 : “𖼻𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 , 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼖𖽔𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 .”
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼨𖽑𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼚𖽙 𖼳𖽡𖾑 𖼦𖽳 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 𖼨𖽑𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 , 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 · 𖼽𖽦𖾑 𖼖𖽡𖽾 𖼨𖽑𖽹𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼋𖽔𖾏 . 6 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 , 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 · 𖼽𖽦𖾑 𖼖𖽡𖽾 , 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼨𖽑𖽹𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼋𖽔𖾏 , 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼊𖽡𖾏 𖽐𖼷𖽷𖾏 . 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽔 𖼊𖽔 𖽐𖼞𖽔𖾏 𖼞𖽱𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 , 𖽂𖽪𖾑 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼽𖽙 𖼖𖽠𖾑 𖼖𖽪𖾑 𖼀𖽑𖽪𖾐 𖼼𖽙𖾐 . 8 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 、 𖼢𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼺𖽙𖾑 、 𖽐𖼎𖽔𖾏 、 𖽐𖼎𖽔𖾏 𖽐𖼀𖽔𖾏 、 𖼞𖽡𖾑 𖼊𖽳𖾏 𖼷𖽻𖾑 𖼖𖽔, 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 .
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼮𖽡𖾑 𖼊𖽪𖽝𖾏 ( 𖼮𖽡𖾑 𖼊𖽪𖽝𖾏 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖽃𖽔𖾐 6 : 6 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽔 𖼞𖽺𖾑 ) 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼑𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼷𖽠𖾐 𖼛𖽔𖾏 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖼏𖽘𖾏 𖽐𖼎𖽘 𖼚𖽑𖽡𖽝𖾏 𖼚𖽜𖾑𖽑 ”) 𖽂𖽪𖾑 𖼺𖽔𖾐 𖼎𖽺𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼊𖽠𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 . 10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖽂𖽪𖾑 𖼺𖽔𖾐 , 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼊𖽠𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖼋𖽔 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 . 11 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼮𖽵 𖼊𖽔𖾏”) 𖽂𖽪𖾑 𖼺𖽔𖾐 𖼺𖽘𖾏 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽹𖾏 𖼊𖽳𖾏 , 𖼎𖽺𖾏 𖼊𖽡𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼀𖽑𖽡𖾐 𖼖𖽡𖽝 𖼺𖽷𖾑 · 𖽂𖽪𖾑 𖼺𖽔𖾐 , 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 . 12 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼐𖽑𖽡𖾏 : “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼽𖽔𖾑 𖼨𖽑𖽹𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼖𖽺𖾑 ?” 13 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼦𖽡𖾐 𖼨𖽑𖽹𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑， 𖼎𖽺𖾏 𖼨𖽑𖽹𖾑 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽳 𖼀𖽡𖽝 · 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 . 14 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽳 𖼀𖽡𖽝 · 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖽐𖼞𖽔𖾏 𖼷𖽷𖾐 𖼖𖽑𖽡𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖽃𖽳 𖼀𖽡𖽝 · 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼑𖽡𖾏 𖼾𖽡𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 .
