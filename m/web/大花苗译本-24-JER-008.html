8
“ 𖼸𖽙 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏, 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏, 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏, 𖼊𖽪𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖼊𖽑𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 2 𖼗𖽔𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐、 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖽐𖼎𖽙 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖽃𖽡𖾑、𖼡𖽑𖽙𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽃𖽡𖾑、𖼡𖽑𖽙𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 𖽃𖽡𖾑、𖼡𖽑𖽙𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑、𖼡𖽑𖽙𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽃𖽡𖾑, 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼮𖽡𖽝𖾏、𖼦𖽡𖾐 𖼎𖽺𖾏 𖼊𖽱𖾏 𖼺𖽘𖾏, 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏. 3 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼐𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼯𖽪𖾏 𖼺𖽹𖾑 𖼞𖽑𖽡𖽠𖾏 𖼋𖽔 𖼦𖽡𖾐 𖼺𖽹𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
4 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼢𖽻, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽙𖾑 𖼙𖽠𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼋𖽘𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
5 𖼸𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖾑 , 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼗𖽡𖾏 𖼖𖽺𖾑?
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐.
6 𖼞𖽪𖾑 𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳．
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽻𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼐𖽑𖽡𖾏:
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼋𖽪 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼐𖽔𖾑 𖼖𖽺𖾑?
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼏𖽘 𖼎𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼞𖽡𖾑,
𖼺𖽱𖾏 𖼑𖽜𖾏 𖽐𖼮𖽵 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏.
7 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾑 𖼁𖽔𖾏 𖼑𖽻𖾏 𖽐𖼀𖽳𖾏 𖼀𖽻𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼗𖽙𖾏、𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏,
𖽐𖼡𖽔𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 、𖼐𖽻 𖼙𖽡𖽝𖾏 𖼞𖽡𖾏 𖼘𖽪𖾏 𖽐𖼎𖽙 𖼚𖽘𖾑 𖼑𖽻𖾏 𖽐𖼀𖽳 𖼖𖽔𖾏 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼗𖽙𖾏,
𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑.
 
8 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏: 𖼀𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳?
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 .
𖼑𖽔𖾏 𖼘𖽪𖾏! 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼐𖽱𖾏 𖼊𖽔𖾑 𖽐𖼚𖽘𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼺𖽻𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑.
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏、𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏、𖼊𖽻𖾏 𖼄𖽔𖾐.
𖼑𖽔𖾏 𖼘𖽪𖾏！ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼖𖽔𖾏 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏,
𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑?
10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽤𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐, 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼦𖽠𖾑 𖼉𖽳 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑,
𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐.
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼞𖽑𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐, 𖼐𖽑𖽡𖾏:
𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽘𖾏! 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽘𖾏 !
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 .
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼳𖽪𖾐, 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼢𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼢𖽻 .
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽻 𖼙𖽠𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
13 “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 ,
𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 ,
𖽃𖽔 𖽐𖼚𖽻𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡．
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑,
𖼽𖽔𖾑 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
14 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼖𖽺𖾑?
𖼀𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱, 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼏𖽔𖾏,
𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼛𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼛𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏,
𖼎𖽺𖾏 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼮𖽷𖾐 𖽃𖽦𖾐 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼦𖽻𖾏,
𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
15 𖼀𖽡𖾐 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 ,
𖼈𖽡𖽝𖾏 𖼊𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
𖼋𖽙𖾏 𖼑𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏,
𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙, 𖼊𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼋𖽘𖾏.
 
16 𖼎𖽑𖽻𖾏 𖼊𖽗𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼑𖽜𖾏 𖼷𖽑𖽔𖾐 𖽃𖽔𖾑 𖽐𖼀𖽱 ,
𖼐𖽱𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼢𖽔𖾏 𖼟𖽡𖾏 𖼉𖽺𖾏 𖼀𖽡 𖼟𖽡𖽝 ,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽪𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 .
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 ,
𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑,
𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑, 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽙 .
17 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼚𖽔𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼖𖽠𖾑 𖼋𖽠𖾏,
𖼀𖽡𖾑 𖼐𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼅𖽡 ,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼑𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏
18 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼾𖽨𖾏 𖼸𖽠𖾏 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼾𖽨𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽐𖼮𖽑𖽡𖾏 𖼉𖽪𖾏 𖼡𖽔 𖼏𖽔 .
19 𖼑𖽻 𖼘𖽪! 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖼮𖽨𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼋𖽔𖾏, 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼳𖽡 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏 ?”
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:“𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏,
𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑, 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼖𖽺𖾑?
20 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼖𖽔𖾏 𖼋𖽘𖾏 𖼋𖽘𖾏,
𖼀𖽡𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼺𖽷𖾐！”
21 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏: “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼄𖽙𖾐,
𖼞𖽪𖾑 𖼖𖽔𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐,
𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾐 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽔𖾑 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏.
 
22 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼵𖽪𖾐 𖼳𖽤𖾏 𖼖𖽺𖾑?
𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼞𖽑𖽙𖾏 𖽐𖼷𖽘𖾏 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼖𖽺𖾑?”
