1
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖼡𖽑𖽜𖾐 𖼡𖽑𖽳𖾏，𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏， 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏 𖼋𖽘𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽜𖾑. 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼽𖽙 𖼦𖽗． 2 𖼽𖽙 𖼦𖽗 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏 𖼞𖽑𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏． 3 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽡𖽠 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼐𖽡𖾑， 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾑 𖼎𖽺𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽃𖽡𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．
4 𖼽𖽙 𖼦𖽗 𖼺𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼳𖽡𖾏 𖼽𖽔𖾏．𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐、𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐、𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 ， 𖽐𖼎𖽙 𖼎𖽑𖽻𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏， 5 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖽃𖽡𖾑，𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽪𖾐 𖼯𖽡𖽝𖾏 𖼗𖽙，𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪，𖼄𖽔𖾐 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖼎𖽻𖾏 𖼅𖽡．
𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼀𖽡𖾐， 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼮𖽑𖽘𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼀𖽡𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽙𖾏， 6 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏， 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 ． 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 、 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏，𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏．𖽃𖽔𖾏 𖼄𖽞！
7 𖼑𖽔𖾏 𖼘𖽪！𖼐𖽱𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏，
𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼅𖽔 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏，
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼊𖽑𖽡𖽝𖾐，
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼐𖽔𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏．
𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝．𖽃𖽔𖾏 𖼄𖽞！
8 𖼮𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼖𖽔𖾑 𖼇𖽔，𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽙 𖼄𖽞 𖼞𖽔（“𖽃𖽔𖾑 𖼖𖽔𖾑 𖼇𖽔” 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼖𖽔𖽃𖽔𖾑 𖼅𖽔 𖽐𖼊𖽠𖾑 𖼖𖽪𖾑 𖼟𖽪𖾏 𖼁𖽱𖾏， “𖽃𖽙 𖼄𖽞 𖼞𖽔” 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏），𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐、𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐、𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐， 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏．”
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼦𖽗 𖼁𖽙𖾏
9 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼽𖽙 𖼦𖽗，𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔、 𖼑𖽡𖾏 𖼞𖽪𖽙、 𖼑𖽡𖾏 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽔𖾏 𖽐𖼀𖽠 𖽃𖽡𖾑，𖼞𖽪𖾑 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼐𖽔𖾑，𖼙𖽔 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽔 𖼄𖽙𖾑． 10 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼐𖽑𖽪𖾐，𖼞𖽪𖾑 𖼊𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑（𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼳𖽟𖾏 𖼖𖽡𖽾 𖽐𖼚𖽙”），𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 𖼺𖽱𖾏 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏， 11 𖼐𖽑𖽡𖾏：“𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼎𖽻𖾏 𖽐𖼊𖽠𖾑 𖼮𖽡𖽝𖾏，𖼺𖽘𖾏 𖼎𖽻𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽪 𖼺𖽙𖾐、 𖼺𖽷𖾏 𖼄𖽺𖾐 𖼐𖽔、𖼀𖽡𖽝 𖼮𖽢𖾑 𖼄𖽙𖾑、 𖼊𖽑𖽪𖽺𖾑 𖼽𖽔𖾐 𖼊𖽑𖽪𖽺𖾑 𖼖𖽔、 𖼺𖽔𖾐 𖼊𖽡、𖼇𖽺𖾑 𖼖𖽔𖾑 𖼊𖽑𖽡𖽝 𖼇𖽺𖾑𖽐𖼎𖽙 𖼖𖽻𖾐 𖼊𖽡𖾐 𖼮𖽢𖾑．”
12 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼀𖽡𖾏 𖼙𖽜𖾏 𖼗𖽙𖾏，𖼽𖽔𖾑 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏．𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖼋𖽘𖾏 ，𖼎𖽺𖾏 𖼁𖽙𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖽃𖽔𖾏． 13 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐，𖼐𖽑𖽘𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖼯𖽢 𖼊𖽠𖾏，𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖽃𖽡𖾐 𖼸𖽙𖾐 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 𖼞𖽪𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐． 14 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖽐𖼎𖽙 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐，𖼚𖽠𖾐 𖼺𖽱𖾏 𖼚𖽻𖾐 𖼾𖽘𖾐 ，𖼺𖽱𖾏 𖽐𖼀𖽪𖾏， 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏， 15 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼮𖽡𖾏 𖼺𖽱𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼮𖽡𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼷𖽑𖽷𖾏 𖼳𖽱𖾑． 16 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼯𖽡𖽙𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐，𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽘𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡𖽝𖾐 𖽐𖼮𖽳𖾏，𖼐𖽱𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼮𖽡𖾏 𖼺𖽱𖾏 𖼋𖽘 𖼐𖽑𖽪𖾐 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼮𖽡𖾏 ．
17 𖼞𖽪𖾑 𖽂𖽪𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏， 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼀𖽑𖽻𖾏 𖽐𖼀𖽙 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼺𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐．𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼀𖽔𖾐 𖼞𖽪𖾑 ，𖼐𖽑𖽡𖾏：“ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏！𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 ， 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐， 18 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏．𖼞𖽪𖾑 𖼙𖽔 𖼋𖽔 𖼋𖽘𖾏， 𖼑𖽔𖾏 𖼘𖽪！𖼞𖽪𖾑 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼯𖽡𖽙 𖼺𖽷𖾐，𖼽𖽔𖾑 𖼯𖽡𖽝𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏；𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼋𖽔 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼚𖽺𖾏 𖽐𖼮𖽦𖾏 𖼀𖽑𖽪𖾑． 19 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏，𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼑𖽪𖾏，𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏， 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽡𖽝𖾏． 20 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼯𖽡𖽙𖾏 𖽐𖼎𖽙 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼞𖽪𖾐 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：𖼳𖽤𖾏 𖼗𖽪𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 ，𖼳𖽤𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 ． ”
