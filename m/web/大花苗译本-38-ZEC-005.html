𖼞𖽪𖾑 𖼎𖽺𖾏 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼽𖽘𖾏. 2 𖼐𖽱𖾏 𖼑𖽪 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏?” 𖼞𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼟𖽪𖾏 𖼽𖽘𖾏, 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼇𖽘𖾑 𖼟𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏.” 3 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼸𖽜 𖽃𖽡𖾑 𖼖𖽔𖾑 𖼸𖽙𖾏 𖼛𖽔𖾏 𖽐𖼊𖽠𖾑 𖼺𖽘𖾑 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽃𖽡𖾑 𖼖𖽔𖾑 𖼸𖽙𖾏 𖼛𖽔𖾏 𖽐𖼊𖽠𖾑 𖼺𖽘𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽻𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏. 4 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼟𖽙𖾏 𖼛𖽔𖾏 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼸𖽜 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏, 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏, 𖼖𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏、 𖽐𖼊𖽻𖾏、 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏．”
𖼁𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏
5 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖼋𖽪 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏?” 6 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽪𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏(𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼽𖽡𖾑 𖼇𖽔”).” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.” 7 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼷𖽑𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼘𖽙𖾏 𖼟𖽪𖾏 𖼈𖽳𖾑 𖼊𖽻𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖽃𖽡𖾑 𖼉𖽺𖾏 𖽐𖼚𖽙. 8 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖽃𖽡𖾑 𖽐𖼚𖽙, 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼷𖽑𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼘𖽙𖾐 𖼟𖽪𖾏 𖼈𖽳𖾑 𖽃𖽡𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖽃𖽡𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏. 9 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼮𖽢𖾏, 𖼽𖽘𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖽃𖽔𖾑 𖼊𖽡𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼁𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽳𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼮𖽘𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐. 10 𖼞𖽪𖾑 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽳𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?” 11 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏𖼳𖽷𖾏 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼑𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼷𖽑𖽙𖾏 𖼷𖽷𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏, 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏.”
