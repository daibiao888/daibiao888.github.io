𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑
（𖼞𖽑𖽙𖾐 9：2-13；𖼖𖽪𖾏 9：28-36）
17
𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼊𖽳、𖼽𖽔𖾐 𖼞𖽙𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼽𖽙 𖼦𖽗，𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐． 2 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼐𖽱𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼮𖽡𖾏 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 ，𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼚𖽠𖾐 𖼨𖽑𖽪𖾑 𖼮𖽡𖾏． 3 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏，𖼅𖽔𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 ， 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏． 4 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪 ！𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼟𖽡 𖼮𖽔𖾑 𖼯𖽪𖾏，𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼷𖽑𖽙𖾏 𖼷𖽷𖾐 𖼖𖽪𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐，𖽃𖽡𖾐 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐，𖽃𖽡𖾐 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐，𖽃𖽡𖾐 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼐𖽡𖽙𖾐．” 5 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼚𖽙𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽙 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏， 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽱𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ， 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼐𖽱𖾏．” 6 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽙𖾑， 𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽡𖽝𖾏，𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝． 7 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏， 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏．” 8 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏，𖼮𖽔𖾑 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐．
9 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽙𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼡𖽔 𖼋𖽱𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙． 10 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖾏，𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏， 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼋𖽔𖾏？” 11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼗𖽙𖾏 ，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏． 12 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼙𖽔 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏，𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏， 𖼎𖽺𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏．𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 ，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽦𖾐．” 13 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗𖼐𖽑𖽡𖾏 ．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼊𖽑𖽡 𖼗𖽻𖾏
（𖼞𖽑𖽙𖾐 9：14-29；𖼖𖽪𖾏 9：37-43）
14 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼐𖽑𖽡𖾏： 15 “ 𖼮𖽵𖾑 𖼘𖽪！𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑 𖼷𖽔𖾑 𖼊𖽪𖾐 ， 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼊𖽑𖽡 𖼗𖽻𖾏 𖽃𖽦𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼀𖽡𖾑 𖼯𖽠 𖼟𖽻𖾐 𖼢𖽻 𖼚𖽔𖾏 𖼯𖽘𖾏 𖼋𖽠 𖼀𖽻𖾐 𖼚𖽔𖾏 𖽃𖽻𖾐． 16 𖼞𖽪𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼞𖽑𖽙𖾏 𖼐𖽱𖾏．” 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖽃𖽺𖾐 ！𖼷𖽑𖽷𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑 ， 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼸𖽙 𖼊𖽑𖽻𖾑 𖼋𖽱 𖼖𖽺𖾑？𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼷𖽜𖾏 𖼅𖽡 𖼸𖽙 𖼊𖽑𖽻𖾑 𖼋𖽱 𖼖𖽺𖾑？𖼨𖽑𖽳𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑．” 18 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽔𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼚𖽘𖾐 𖽃𖽡𖾑，𖼀𖽡𖾑 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ，𖼎𖽑𖽻𖾏 𖼖𖽡𖾏 𖽃𖽡𖾑 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏． 19 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼊𖽪𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐 𖽃𖽡𖾑？” 20 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏 𖼯𖽠 𖼐𖽔𖾑．𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼅𖽡 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼮𖽘𖾑 𖼻𖽻𖾐 𖼛𖽙𖾐 𖽃𖽔 𖼛𖽡𖾏，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏：‘ 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽡𖾑 𖼮𖽑𖽹𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼉𖽺𖾏 ’， 𖼖𖽪𖾑 𖼎𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽑𖽹𖾏 𖼙𖽠𖾏， 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏． 21 𖼐𖽑𖽡𖾏 𖼸𖽙 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼐𖽡𖾑，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼈𖽡𖽝𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼯𖽡𖽝𖾏 𖼗𖽙
（𖼞𖽑𖽙𖾐 9：30-32；𖼖𖽪𖾏 9：43-45）
22 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖾏 ． 23 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏，𖼸𖽙 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙．” 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖼊𖽳𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽪𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏
24 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾，𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼋𖽡𖽝𖾏 𖼊𖽪𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏？” 25 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“𖼊𖽪𖾏 𖼯𖽡𖽙𖾏．” 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽪𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼳𖽡𖾑 𖼄𖽞 ，𖼟𖽡 𖽐𖼊𖽱𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏？𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖽐𖼎𖽙 𖼡𖽔 𖼋𖽱𖾏 𖼳𖽻𖾏 𖼞𖽡𖾑 𖼳𖽳𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽳𖾏 𖼘𖽪？ 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼾𖽨𖾏 𖼊𖽪𖾐 𖼖𖽔𖾏？𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪？” 26 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“𖼟𖽪𖾏 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼮𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼊𖽪𖾐 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼊𖽪𖾏 𖼳𖽳𖾏 ． 27 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼙𖽔，𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 𖼗𖽔𖾏 𖼞𖽻𖾐 𖼮𖽢𖾐 𖽐𖼀𖽳𖾏， 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼮𖽢𖾐 𖼊𖽻𖾏 𖽃𖽡𖾑 ，𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖼡𖽑𖽜𖾐 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏，𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼑𖽡𖽝𖾏 ，𖼯𖽡𖽙𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖽃𖽔𖾏 𖼟𖽡 𖼞𖽪𖾑 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏．”
