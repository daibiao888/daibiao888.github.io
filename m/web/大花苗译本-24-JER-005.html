4 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏, 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼋𖽡𖽝 ,
𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑.
5 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑,
𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑．
𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐, 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼄𖽔𖾐 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼖𖽙𖾑 𖼺𖽘𖾏,
𖼀𖽡𖾏 𖽐𖼊𖽡𖾏 𖼛𖽡𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏.
 
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐, 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔 𖼺𖽘𖾏,
𖼊𖽡𖾐 𖼯𖽢𖾏 𖼊𖽳𖾐 𖼘𖽳𖾐 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼊𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼊𖽪𖾑 𖼇𖽜𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖼙𖽻 𖽐𖼚𖽺𖾏 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽠𖾏 𖼙𖽻, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼺𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
 
7 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼯𖽔𖾏 𖼟𖽡 𖼖𖽺𖾑？
𖼟𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏,
𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼏𖽵 𖼗𖽪𖾏．
𖼞𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼟𖽙𖾏 𖼮𖽻𖾏,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏,
𖽃𖽔𖾏 𖼡𖽔𖾏 𖼸𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖽐𖼞𖽔𖾏.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼀𖽪𖾐 𖼮𖽻𖾏 𖼺𖽘𖾏, 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼀𖽜𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽑𖽔𖾏,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼢𖽔𖾏 𖼟𖽡𖾏 𖼉𖽺𖾏 𖼀𖽡 𖼟𖽡𖽝 .
9 “𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼖𖽺𖾑?
𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑”?
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
10 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏,
𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼋𖽘𖾏 𖼺𖽘𖾏,
𖼮𖽔 𖼽𖽔𖾑 𖼷𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼺𖽘𖾏 𖼊𖽔𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼺𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼁𖽡𖽝 .

18 𖼟𖽪𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼅𖽡 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖼺𖽘𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼉𖽔𖾏 𖼾𖽨𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼮𖽵𖾑 𖼅𖽳𖾏
19 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼀𖽡𖾐, 𖼟𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏, 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼇𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼞𖽪𖾑”), 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺, 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼅𖽡 𖼁𖽡𖽝 𖽃𖽡𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐．
20 𖼅𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏,
𖼎𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏, 𖼐𖽑𖽡𖾏:
21 “𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼘𖽪！
𖼅𖽡 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼑𖽔𖾏,
𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼦𖽡𖾐 𖼑𖽻,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼑𖽻 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑.
22 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 𖼖𖽺𖾑?
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼅𖽡 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼺𖽷𖾐 𖼖𖽔𖾏 ?
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐,

𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏 𖼈𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏．
23 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼼𖽔𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏,

24 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏,

𖼐𖽱𖾏 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼄𖽔𖾐 𖼑𖽘, 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐 𖼑𖽘 𖽐𖼊𖽪𖾏 𖼳𖽙𖾑 𖼑𖽘 ,

25 𖼅𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽙𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼎𖽙𖾑 𖼇𖽳𖾑 𖼅𖽡 𖼺𖽘𖾏,
𖼅𖽡 𖼾𖽔𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
26 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖼑𖽡𖽝 𖼑𖽔𖾏, 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼮𖽢𖾐 𖼐𖽻 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽑𖽔𖾑 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽔𖾏 𖼟𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
27 𖽐𖼚𖽙 𖼚𖽱𖾐 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼐𖽻 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏,

28 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼛𖽙 𖼟𖽡𖾏 𖼮𖽡𖾏 𖼮𖽦𖾑𖽑, 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼮𖽑𖽡𖽻𖾏 𖼮𖽑𖽤𖾑,
𖼦𖽡𖾐 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,

29 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼖𖽺𖾑?
𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑?
