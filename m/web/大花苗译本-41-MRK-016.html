𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙
（𖼊𖽑𖽹𖾏 28：1-8； 𖼖𖽪𖾏 24：1-12；𖼽𖽙 20：1-10）
16
𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖼄𖽙 𖼊𖽔𖾏 𖼖𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏， 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏， 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼘𖽙 𖼄𖽡𖾐， 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏，𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽜𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑． 2 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝，𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏， 3 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼀𖽡𖾐 𖼦𖽡𖾏 𖼚𖽙𖾑 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼡𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖽃𖽡𖾑 𖼖𖽺𖾑 ？ ” 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼑𖽔𖾏 ，𖼁𖽙𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏！ 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼮𖽡𖽙𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝． 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏， 𖼎𖽺𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏， 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 ． 6 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏， 𖼅𖽡 𖽐𖼎𖽑𖽦𖾑 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽦𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼑𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏． 7 𖼅𖽡 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼊𖽳： 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖽐𖼊𖽡𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏， 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽱𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡．” 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖼞𖽑𖽜 𖼑𖽜 𖼏𖽘 𖼙𖽠𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼏𖽘𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 ， 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼡𖽔 𖼋𖽱𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽔𖾑．（𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖽐𖼊𖽠𖾑 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 9𖼽𖽘𖾑—20 𖼽𖽘𖾑）
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼑𖽔𖾏
（𖼊𖽑𖽹𖾏 28：9-10；𖼽𖽙 20：11-18）
9 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 ， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼋𖽘𖾏， 𖼺𖽪𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙 𖼎𖽻𖾏𖼄𖽙 𖼊𖽔𖾏 𖼖𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽔𖾏， 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼙𖽔 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 ． 10 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼊𖽑𖽻𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼐𖽡𖽝𖾑 𖼷𖽷𖾏 𖼖𖽘𖾑． 11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼋𖽘𖾏， 𖼟𖽪𖾏𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏，𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜． 12 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼼𖽙𖾏， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽑𖽡𖾏 𖼼𖽘𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 13 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏， 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜．
（𖼊𖽑𖽹𖾏 28：16-20；𖼖𖽪𖾏 24：36-49；𖼽𖽙 20：19-23；𖼎𖽻𖾏 1：6-8）
14 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜， 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑． 15 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “𖼅𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾏 𖼋𖽻𖾏， 𖼡𖽑𖽔𖾐 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻． 16 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽔𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏 ， 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽡𖾏 𖼾𖽔𖾏． 17 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐， 𖼐𖽑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼗𖽪𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐， 18 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼯𖽡𖽻𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏， 𖼮𖽔𖾑 𖼦𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽠𖾏 𖼐𖽱𖾏， 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼻𖽻𖾏 𖼋𖽘𖾏．”
（𖼖𖽪𖾏 24：50-53；𖼎𖽻𖾏 1：9-11）
