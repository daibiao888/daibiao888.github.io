20 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼟𖽪𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖼊𖽡𖽝𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼅𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼞𖽙𖾑 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 , 𖼽𖽔𖾑 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼮𖽢𖾑 𖼮𖽡𖾑 𖼄𖽡𖾐 𖼳𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼀𖽳 𖼖𖽔𖾑 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽡𖾑 , 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼐𖽱𖾏 . 21 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏 : “𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽔𖾏 𖼋𖽪 𖼞𖽳𖾑 𖽃𖽔𖾏 𖼳𖽷𖾏 ？𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼋𖽔𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼟𖽡 , 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽳𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽻𖾐 𖼟𖽡 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 .” 22 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼎𖽙𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼇𖽳𖾑 𖼐𖽱𖾏 , 𖼘𖽪𖾑𖽑 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 , 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽡 𖼞𖽙𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 , 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼸𖽙 𖼄𖽡𖾐 𖼮𖽡 𖼊𖽙𖾑 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 23 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼀𖽙𖾑 𖼐𖽜𖾑𖽑 𖼀𖽙𖾑 𖼏𖽻 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏 , 𖼟𖽡 𖼮𖽤𖾐 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼦𖽡𖾑!” 24 𖼐𖽱𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼮𖽤𖾐 𖼐𖽱𖾏 𖽐𖼡𖽜𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 , 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽡𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 , 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏 , 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾑 𖽐𖼷𖽘𖾏 𖼺𖽘𖾏 . 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽡𖽝𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽜 𖼐𖽱𖾏 𖼐𖽡𖽝𖾑 𖼮𖽑𖽡𖾏. 25 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼑𖽜 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽡𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔 𖼾𖽠 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽡𖾏 , 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼮𖽑𖽡𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼮𖽡𖽝𖾏 . 𖼋𖽡𖾏 𖽐𖼞𖽻𖾏 𖼐𖽡𖾑 𖼺𖽻𖾏 𖼎𖽻𖾏 《𖽐𖼊𖽠𖾑 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐》 𖼮𖽡𖽝𖾏 . 26 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏 , 𖽐𖼎𖽙 𖼻𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 , 27 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼁𖽱𖾏 𖼸𖽙 𖼺𖽻𖾏 , 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏《 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏》 𖼮𖽡𖽝𖾏 .
