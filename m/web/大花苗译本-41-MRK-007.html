𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼸𖽪𖾏
（𖼊𖽑𖽹𖾏 15：1-20）
7
𖼅𖽔𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏， 𖼎𖽑𖽻𖾏𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼋𖽔𖾏，𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏， 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼷𖽔𖾑 𖼺𖽷𖾐 𖼑𖽻𖾏 𖼉𖽔𖾏． 3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼟𖽪𖾏 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼸𖽪𖾏，𖼮𖽔𖾑 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼷𖽔𖾑 𖼋𖽡𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏． 4 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼗𖽙，𖼮𖽔𖾑 𖼦𖽡𖾐 𖽐𖼷𖽔𖾑 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽙𖾑，𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽔𖾑 𖼞𖽑𖽙𖾐 𖼈𖽳𖾐、𖽐𖼷𖽔𖾑 𖼡𖽑𖽠𖾏、 𖽐𖼷𖽔𖾑 𖼞𖽑𖽙𖾐 𖼋𖽻𖾏 𖽐𖼎𖽙 𖼸𖽘𖾏 𖼮𖽑𖽡𖽠𖾏． 5 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼐𖽑𖽡𖾏： “ 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏，𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼖𖽺𖾑 ？” 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏， 𖼅𖽡 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖼺𖽻𖾏：
‘ 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼞𖽪𖾑，
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼡𖽔𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝．
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏，𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐，
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙．’
8 𖼅𖽡 𖼗𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼺𖽘𖾏， 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽪𖾏．” 9 𖼎𖽺𖾏 𖽐𖼎𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼗𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼺𖽘𖾏 ， 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼾𖽨𖾏 𖼸𖽪𖾏． 10 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏：‘𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 ，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼺𖽘𖾏．’ 11 𖼅𖽡 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼐𖽑𖽡𖾏：‘ 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏，𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡， 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼞𖽳 𖽃𖽳𖾐 𖼀𖽔𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏 （“𖼞𖽳 𖽃𖽳𖾐 𖼀𖽔𖾐” 𖼟𖽪𖾏“ 𖼖𖽑𖽘𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏”）．’ 12 𖼅𖽡 𖼎𖽺𖾏 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏． 13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼅𖽡 𖼯𖽡𖽙𖾏 𖼅𖽡 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼷𖽹𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑， 𖼎𖽺𖾏 𖼗𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼺𖽘𖾏．𖼅𖽡 𖼎𖽺𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑．”
𖼳𖽪𖾏 𖼢𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐
14 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏, 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏． 15 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼨𖽑𖽪𖾏 𖼢𖽻𖾏 𖼁𖽺𖾏 𖼁𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽙 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽙 𖼋𖽠𖾏 𖼗𖽙𖾏 ， 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． 16 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼑𖽻，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻．” 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽳𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏， 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏． 18 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼺𖽷𖾐 𖼖𖽔𖾏？ 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼨𖽑𖽪𖾏 𖼢𖽻𖾏 𖼁𖽺𖾏 𖼁𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽙 ，𖼦𖽡𖾐 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ？ 19 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼁𖽺𖾏 𖼁𖽻 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐， 𖼟𖽪𖾏 𖼁𖽻 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼁𖽻 𖼡𖽑𖽙𖾑 𖼮𖽑𖽡𖾑．” 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏： 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼯𖽡𖽙𖾏． 20 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏： “ 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼉𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽙，𖼎𖽺𖾏 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． 21 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐， 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏、 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 、 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼺𖽘𖾐 𖽃𖽔𖾏 𖼸𖽜、 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏、 22 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏、𖼉𖽳 𖼗𖽔 𖼁𖽡𖽝、𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏、 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽢𖾐 𖽃𖽔𖾑 𖼅𖽔、 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐、 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 、𖼙𖽙𖾏 𖽐𖼊𖽪𖾏、𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔 𖼏𖽔． 23 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼎𖽺𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．”
𖼇𖽺𖾑 𖼐𖽡 𖼮𖽡𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏
（𖼊𖽑𖽹𖾏 15：21-28）
24 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏， 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙、𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏， 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 ，𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖽐𖼚𖽺𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏． 25 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏，𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼄𖽡𖽝𖾑 𖽐𖼷𖽑𖽹𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼯𖽵𖾏，𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏， 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽠𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐． 26 𖼖𖽔𖾑 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼖𖽡𖾏 𖼐𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐（“𖼳𖽡𖾑 𖼖𖽡𖾏 𖼐𖽡”𖼟𖽪𖾏 “𖼳𖽡𖾑 𖼖𖽔”）， 𖼟𖽪𖾏 𖼳𖽱𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼇𖽺𖾑 𖼐𖽡 𖼮𖽡𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐， 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼟𖽙𖾏 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼐𖽱𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 ． 27 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ： “𖼺𖽪𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏，𖼄𖽔𖾐 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼡𖽜𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼚𖽡𖾑 𖼑𖽻𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏．” 28 𖼖𖽔𖾑 𖼁𖽙𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼾𖽙 𖼯𖽡𖽙𖾏， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽪𖾑 𖼚𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽘𖾐 𖼞𖽡 𖼏𖽻𖾏，𖼖𖽔𖾏 𖼑𖽻𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼗𖽙．” 29 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “𖼑𖽜 𖼟𖽡 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑， 𖼟𖽡 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼦𖽡𖾑， 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼇𖽳𖾑 𖼟𖽡 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏．” 30 𖼖𖽔𖾑 𖼁𖽙𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖽐𖼮𖽡𖽻𖾏 𖽐𖼞𖽔𖾏，𖼎𖽺𖾏 𖼁𖽙𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼸𖽱𖾏 𖼸𖽘𖾏， 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏．
𖼞𖽑𖽙𖾏 𖽃𖽔𖾑 𖼗𖽘𖾐 𖽐𖼎𖽙 𖼅𖽜 𖼅𖽜
31 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏， 𖼮𖽑𖽡𖽻𖾏 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 ，𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼀𖽙𖾑 𖼖𖽡𖾏𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏， 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼳𖽡𖽠𖾑． 32 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖽃𖽔𖾐 𖼗𖽘𖾐 𖽃𖽔𖾑 𖽐𖼚𖽺𖾏 𖼅𖽜 𖼅𖽜 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑． 33 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼇𖽳𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏， 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐， 𖼮𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏， 𖼎𖽺𖾏 𖽐𖼊𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖽻𖾐 𖼚𖽜𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏， 34 𖼑𖽔𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏 ， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ：“ 𖼽𖽡𖾑 𖼇𖽔 𖼊𖽔𖾏．” 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 ： “𖼡𖽑𖽜𖾐 𖼦𖽡𖾑．” 35 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼎𖽺𖾏 𖼡𖽑𖽜𖾐 𖼋𖽘， 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼖𖽔𖾏 𖼮𖽙𖾏 𖼺𖽘𖾐， 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽑𖽹𖾏 𖼋𖽘𖾏． 36 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼐𖽑𖽡𖾏． 37 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏：“𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏， 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼗𖽘𖾐 𖼸𖽠𖾏 𖼐𖽙𖾑𖽑 𖼗𖽪𖾏， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽜 𖼅𖽜 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏．”
