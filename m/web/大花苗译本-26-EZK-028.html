𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 2 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖽐𖼎𖽙 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏, 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,

𖼟𖽡 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽡𖾏 𖼈𖽡𖽝𖾏,
𖼟𖽡 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾐, 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
3 𖼑𖽔𖾏 𖼘𖽪! 𖼟𖽡 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 ,
𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼈𖽺𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼟𖽡.
4 𖼟𖽡 𖼻𖽡𖽝𖾏 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼟𖽳 𖼊𖽻𖾏 𖼋𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,

𖼟𖽡 𖼎𖽺𖾏 𖼑𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏.”
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐,
𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳, 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽑𖽔𖾏 𖽐𖼚𖽔 𖼟𖽡 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼊𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑,
𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑,
𖼟𖽡 𖼽𖽔𖾑 𖼋𖽔 𖼎𖽻𖾏 𖼳𖽡𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽔𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
9 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼋𖽡𖽝𖾏 𖼟𖽡 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏” 𖼺𖽷𖾐 𖼖𖽺𖾑?
𖼋𖽡𖽝 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼟𖽡 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾐, 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
10 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼋𖽔 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
𖼐𖽑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼑𖽜 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 12 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼑𖽜 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：
𖼟𖽡 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏，𖼟𖽪𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼾𖽻 (“𖼼𖽘𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼾𖽻” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼷𖽷𖾏 𖼼𖽘𖾏” ),
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏, 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽐𖼎𖽻.
13 𖼟𖽡 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖼉𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖽐𖼀𖽔𖾏 𖼐𖽑𖽘𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,
𖼟𖽪𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐、𖼽𖽱𖾏 𖼟𖽪𖾏 𖼉𖽘𖾏、 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼮𖽡𖽾𖾑 𖽐𖼞𖽘𖾑、
𖽃𖽔𖾑 𖼈𖽳𖾐 𖼳𖽪𖽺𖾐 𖼷𖽑𖽘𖾑 𖼽𖽱𖾏、 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼄𖽔𖾐 𖼐𖽻𖾐 𖼖𖽡𖽝𖾐、 𖼀𖽡𖾏 𖼽𖽱𖾏、
𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑 𖽐𖼷𖽔𖾐、 𖼽𖽱𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾐、 𖼽𖽱𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖽐𖼎𖽙 𖼞𖽪𖾐,
𖼎𖽺𖾏 𖼅𖽔𖾏 𖼖𖽪𖾑 𖽐𖼎𖽔𖾏 𖼖𖽡𖾑 𖼘𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
