𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
97
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼙𖽔 𖼚𖽙𖾏！
2 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽙 𖼷𖽻𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼎𖽙 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏.
3 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼾𖽱𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏,
𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑.
4 𖼐𖽱𖾏 𖼺𖽙𖾐 𖽃𖽔𖾏 𖼀𖽡𖾏 𖼯𖽦𖾏 𖼮𖽡𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 .
5 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼺𖽱𖾏 𖼮𖽡𖽝𖾐 𖼾𖽘𖾏 𖼺𖽘𖾏.
 
6 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .
7 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼉𖽻𖾏 𖼮𖽘𖾑 𖼮𖽵𖾑 𖼳𖽺 𖼘𖽪！𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼐𖽱𖾏 .
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼳𖽡 𖼣𖽔𖾑 𖼐𖽑𖽙𖾑 𖼟𖽡 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼙𖽔 ,
𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽻 ( “𖼙𖽻”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖽐𖼷𖽑𖽹𖾏 ”) 𖼎𖽺𖾏 𖼙𖽔 𖼚𖽙𖾏.
9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐, 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼋𖽻𖾏,
𖼟𖽡 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 , 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼉𖽻𖾏 𖼮𖽵𖾑 𖼮𖽵𖾑 𖼳𖽺 .
 
10 𖼅𖽡 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼐𖽱𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽡𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
11 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼅𖽔𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
12 𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼚𖽙𖾏,
𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖽃𖽡𖾑.
