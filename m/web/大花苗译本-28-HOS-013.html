13
𖽃𖽔 𖼊𖽑𖽻𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽂𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏,
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼀𖽔𖾑 𖼖𖽡 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.
2 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖼎𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼾𖽔𖾏,
𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼘𖽡𖾑𖽑 𖼷𖽷𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏,
𖼟𖽪𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼗𖽙𖾏 𖽃𖽔𖾏,
𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔𖾏.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏:
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼁𖽠 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼡𖽑𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼦𖽘𖾑 𖼀𖽙𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼙𖽱 𖼟𖽪𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖽐𖼡𖽑𖽔𖾐 𖼙𖽠𖾏,
𖼺𖽱𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏 𖼦𖽡𖾑 𖼁𖽘𖾏 𖼡𖽻𖾐 𖼊𖽻𖾏 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼷𖽑𖽔𖾐 𖼙𖽠𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼀𖽘𖾏 𖼋𖽠𖾏 𖼽𖽘𖾏 𖼋𖽠𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏.
𖼑𖽜 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼮𖽵𖾑
4 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼢𖽻𖾏, 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺,
𖼋𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽨𖾏 𖼮𖽵𖾑 .
5 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖽃𖽡𖾑 𖼯𖽜𖾏 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 .
 
6 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼊𖽪𖾑 𖼾𖽨 , 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏 ,
𖼊𖽻𖾏 𖼑𖽻𖾏 𖼮𖽻𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏,
𖼎𖽺𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 .
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼇𖽜𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 .
8 𖼞𖽪𖾑 𖽐𖼮𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼚𖽹𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼺𖽘𖾏,
𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐 (“𖼖𖽪𖾑 𖽐𖼎𖽙𖾐” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “𖽃𖽔𖾑 𖼺𖽡𖽝𖾐”),
𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖽃𖽔𖾐 𖼑𖽡𖽝 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑
9 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ! 𖼟𖽡 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼷𖽜𖾑 𖼊𖽠𖾏 𖼞𖽪𖾑 , 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼊𖽠𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼟𖽡 𖽃𖽡𖾑.
10 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼟𖽡 !
𖼟𖽡 𖼙𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽜 𖼞𖽪𖾑 𖼷𖽑𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐.”
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼮𖽵𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
11 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽻𖾏 𖼟𖽡 ,
𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼄𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏.
 
12 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼮𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏.
13 𖽃𖽔 𖼁𖽙𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼑𖽡𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳,
𖼸𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏,𖼉𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐.
14 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼚𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏.
𖼛𖽙𖾏 𖼋𖽔𖾏 𖼘𖽪! 𖼟𖽡 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼘𖽪! 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼚𖽺𖾏 𖼙𖽠𖾏.
 
15 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼘𖽑𖽘𖾑 𖽐𖼷𖽱𖾏 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼋𖽔𖾏,
𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼮𖽢𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼷𖽑𖽔𖾐 𖼋𖽔𖾏,
