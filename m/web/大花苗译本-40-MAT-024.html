24
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏，𖼇𖽳𖾑 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼡𖽑𖽳𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏． 2 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼅𖽡 𖼁𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼘𖽪𖾏？𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼷𖽷𖾐 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼮𖽡𖽝𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼎𖽑𖽔𖾑．”
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏
（𖼞𖽑𖽙𖾐 13：3-13；𖼖𖽪𖾏 21：7-19）
3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼞𖽗𖾐 𖼖𖽗 𖼇𖽻𖾑 𖼎𖽻𖾐，𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼋𖽔𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽑𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐， 𖼊𖽑𖽻𖾑 𖼋𖽱 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑？𖼟𖽡 𖼎𖽺𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪？” 4 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 ，𖽐𖼮𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽵𖾑 𖼅𖽡． 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪 ’ ，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐． 6 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏，𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 ， 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏， 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼾𖽙 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏． 7 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼉𖽺𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏，𖼞𖽪𖽙 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖽙 𖼉𖽺𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏， 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽡𖽻𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼮𖽷𖾑 𖼑𖽡𖽝𖾏 𖼮𖽷𖾏 𖼳𖽡𖽻𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏． 8 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼳𖽠𖾑 𖼁𖽱𖾏． 9 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼳𖽻𖾏 𖽐𖼚𖽔，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼅𖽡．𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼳𖽪𖾐 𖼅𖽡． 10 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼽𖽔𖾑 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻，𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼉𖽻 𖼷𖽻𖾏，𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼳𖽪𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏． 11 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑，𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐． 12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼮𖽑𖽨𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼸𖽡𖽝𖾏． 13 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼷𖽜𖾏 𖼸𖽙 𖼡𖽘𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏． 14 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖽐𖼷𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾏 𖼋𖽻𖾏 ，𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐，𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼽𖽔𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏．”
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐

15 “𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 、𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 𖽃𖽡𖾑 𖼳𖽠𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏， 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 𖼙𖽔 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼟𖽡𖽠 𖼑𖽘 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏． 16 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽑𖽹𖾏 ， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐． 17 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖼏𖽪𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏． 18 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽷𖾐 𖼗𖽙 𖼞𖽑𖽡𖽠𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 ． 19 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑，𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑，𖼊𖽑𖽡𖽝𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏． 20 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏， 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼀𖽜𖾑 𖽃𖽡𖾑，𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏． 21 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼮𖽷𖾑 𖼮𖽑𖽡𖽠𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖼷𖽷𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑，𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼅𖽔𖾏． 22 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼨𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽹𖾑 𖼊𖽻𖾏，𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽻𖾏 𖼖𖽪𖾑 𖼊𖽘𖾏． 23 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑， 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏： “ 𖼑𖽔𖾏 𖼘𖽪 ！𖼮𖽡𖾑 𖼊𖽪 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 ’， 𖼡𖽑𖽙𖾑 𖼯𖽢 ‘ 𖼮𖽡𖾑 𖼊𖽪 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 ’， 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼐𖽱𖾏． 24 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 ， 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑，𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼮𖽑𖽵𖾑，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽹𖾑 𖼊𖽻𖾏 ，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼮𖽑𖽵𖾑 𖼊𖽻𖾏 𖼺𖽘𖾏． 25 𖼑𖽔𖾏 𖼘𖽪！𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼺𖽪𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼋𖽘𖾏． 26 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏：‘ 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼮𖽡𖾑 𖼊𖽪 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 ’， 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 ：‘ 𖼑𖽔𖾏 𖼘𖽪！ 𖼮𖽡𖾑 𖼊𖽪 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼮𖽑𖽤𖾏 𖽐𖼞𖽔𖾏 ’， 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜． 27 𖼺𖽙𖾐 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼯𖽦𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼮𖽡𖾏 𖼗𖽙𖾏，𖼮𖽡𖾏 𖼸𖽙 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 28 𖼡𖽻𖾑 𖼋𖽔 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏，𖼚𖽘𖾑 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏． ”
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏
（𖼞𖽑𖽙𖾐 13：24-27；𖼖𖽪𖾏 21：25-28）
29 “𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，
𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏，
𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼦𖽡𖾐 𖼮𖽡𖾏，
𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽻𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏，

𖼽𖽔𖾑 𖼯𖽷𖾏 𖼻𖽙𖾑
（𖼞𖽑𖽙𖾐 13：28-31；𖼖𖽪𖾏 21：29-33）
32 “ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽻 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏，𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼘𖽑𖽘𖾑 𖽃𖽔𖾑 𖽐𖼷𖽔 𖼘𖽑𖽘𖾑 𖽃𖽔 𖽐𖼚𖽻𖾏，𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼳𖽙𖾑 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏． 33 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 ，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽙𖾏 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼐𖽡𖾑，𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼗𖽙𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼋𖽘． 34 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼮𖽑𖽡𖽻𖾏 𖼺𖽷𖾐，𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼊𖽡𖽝𖾑． 35 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏 ，𖼞𖽪𖾑 𖼗𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼛𖽙𖾏．”
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐
（𖼞𖽑𖽙𖾐 13：32-37；𖼖𖽪𖾏 17：26-30，34-36）
36 “ 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐， 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐，𖼯𖽔 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼀𖽻𖾐 𖼊𖽔𖾐． 37 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 ， 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽙 𖼽𖽔𖾏𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏． 38 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽠𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼑𖽻𖾏 𖼦𖽻𖾏 𖼷𖽙𖾏 𖼐𖽤𖾐 𖼡𖽔𖾏 𖽐𖼷𖽑𖽹𖾏，𖼸𖽙 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼐𖽙 𖼽𖽔𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏 𖽃𖽡𖾑， 39 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼦𖽡𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖼗𖽙𖾏 𖼸𖽙 𖼋𖽘𖾏，𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽪𖾐 𖼋𖽔 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 40 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼙𖽡𖽝𖾏，𖼖𖽔𖾑 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼙𖽠𖾏，𖼇𖽳𖾑 𖼗𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏． 41 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼢𖽪𖾏 𖼡𖽜𖾑，𖼖𖽔𖾑 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼙𖽠𖾏，𖼇𖽳𖾑 𖼗𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏． 42 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼅𖽡 𖼽𖽔𖾑 𖼯𖽷𖾏 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼅𖽡 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼋𖽱 𖼗𖽙𖾏． 43 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼀𖽻𖾐 𖼯𖽡𖽹𖾏 𖼋𖽱𖾏 𖼅𖽔𖾏 𖼸𖽜 𖼋𖽔𖾏，𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼯𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑，𖼦𖽡𖾐 𖼟𖽙𖾏 𖼸𖽜 𖽐𖼮𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖽐𖼞𖽔𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼑𖽡𖽝，𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼯𖽡𖽙． 44 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼅𖽡 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼽𖽔𖾑 𖼗𖽙𖾏 𖼋𖽘𖾏．”
𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐
（𖼖𖽪𖾏 12：41-48）
45 “𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐，𖼟𖽪𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼺𖽹𖾑 𖼊𖽻𖾏，𖼽𖽔𖾑 𖼐𖽱𖾏 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐， 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼇𖽺𖾐 𖼡𖽻𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽺𖾑？ 46 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼺𖽷𖾐 𖼗𖽙，𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 ， 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏． 47 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼮𖽵𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼑𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜． 48 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽐𖼚𖽪𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼗𖽙𖾏 𖼙𖽡 ’， 49 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼢𖽻 𖼮𖽡𖽠𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏． 50 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙，𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐，𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼮𖽵𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼗𖽙𖾏 ， 51 𖼻𖽻𖾏 𖼻𖽻𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼖𖽠𖾑 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏，𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼟𖽡𖾏 𖼋𖽙𖾏 𖼐𖽡𖽝𖾑𖽑 𖼷𖽷𖾏 𖼞𖽹𖾑．”
