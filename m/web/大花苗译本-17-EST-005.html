𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 , 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼞𖽑𖽡𖽠𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑 𖼐𖽑𖽘𖾑 𖼮𖽡𖽝𖾏 , 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽐𖼚𖽙 , 𖼳𖽠𖾑 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼚𖽙 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 , 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖽐𖼞𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 . 2 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 , 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼞𖽪𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼮𖽡𖾏 𖼀𖽜𖾏 . 3 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼘𖽪 ！ 𖼟𖽡 𖼖𖽔𖾑 𖼽𖽔𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏?𖼟𖽡 𖼽𖽔𖾑 𖽐𖼊𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼟𖽪𖾏 𖼞𖽪𖽙 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 , 𖼊𖽪𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 .” 4 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼐𖽑𖽡𖾏 : “𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 , 𖼷𖽑𖽜𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼨𖽑𖽳𖾏 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 .” 5 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼨𖽑𖽪𖾏 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼋𖽔𖾏 , 𖼟𖽙𖾏 𖼸𖽙𖾏 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼗𖽪𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 .” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽳𖾏 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 . 6 𖼐𖽡𖽙𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖽃𖽡𖾑 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼖𖽔𖾑 𖼽𖽔𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 , 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼊𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼟𖽪𖾏 𖼞𖽪𖽙 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 , 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 .” 7 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼽𖽔𖾑 , 𖼞𖽪𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼊𖽜 ． 8 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼯𖽪𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼽𖽔𖾑 , 𖼯𖽪𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽜 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼷𖽑𖽜𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽡𖾏 𖼷𖽡𖽝𖾏 . 𖼀𖽡 𖼟𖽡 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖼑𖽪 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼟𖽙𖾏 𖽐𖼎𖽑𖽹𖾏 .”
