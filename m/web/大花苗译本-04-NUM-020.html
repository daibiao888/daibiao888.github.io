𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼀𖽔𖾑 𖽐𖼊𖽻𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏
（𖼋𖽠𖾏 17：1 - 7）
20
𖼖𖽪𖾑 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼳𖽱 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 , 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏𖼮𖽢𖾑 𖼊𖽡𖾐 𖼺𖽷𖾑 . 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼣𖽔𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 .
2 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼦𖽻𖾏 , 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼷𖽜𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 . 3 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼙𖽔 𖼋𖽔 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽔 . 4 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼨𖽑𖽳𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼐𖽡𖾑 , 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼞𖽑𖽔𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼖𖽺𖾑 ? 5 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼊𖽔𖾑 𖼀𖽡𖾐 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 , 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽱𖾏 𖼚𖽘𖾐 𖼐𖽡𖾑 𖼖𖽺𖾑 ? 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼀𖽳𖾐 𖼮𖽘𖾑 , 𖼎𖽺𖾏 𖼦𖽡𖾑 𖼅𖽔𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 , 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 , 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼦𖽻𖾏 .”
6 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼇𖽳𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽔 𖼛𖽻𖾏 , 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 8 “𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 , 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼅𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼦𖽡 𖼸𖽱 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽙𖾏 𖼋𖽠𖾏 𖽃𖽻𖾐 𖼗𖽙𖾏 , 𖽃𖽻𖾐 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽦𖾏 𖼦𖽻𖾏 .”
9 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 . 10 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼦𖽡 𖼸𖽱 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 . 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 𖼐𖽡𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 : 𖼀𖽡𖾐 𖼑𖽜 𖼅𖽡 𖼟𖽙𖾏 𖽃𖽻𖾐 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼐𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ?” 11 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼮𖽘𖾏 𖼋𖽡𖾏 , 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏 𖽐𖼊𖽻𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽃𖽔𖾐 𖼟𖽻𖾐 , 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽦𖾏 𖼦𖽻𖾏 𖽃𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏 . 12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽑𖽡𖾏 : “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼞𖽪𖾑 , 𖼦𖽡𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼨𖽑𖽳𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 .” 13 𖼋𖽡𖾏 𖽃𖽻𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼀𖽔𖾑 (“𖼄𖽡𖾐 𖼖𖽡 𖼀𖽔𖾑 ” 𖼟𖽪𖾏 “𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 ”) 𖽃𖽻𖾐 , 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .
𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼦𖽡𖾐 𖽐𖼮𖽠𖾏 𖼞𖽡𖾑 𖼟𖽙𖾏 𖼅𖽻𖾏
