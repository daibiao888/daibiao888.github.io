5 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼨𖽑𖽢𖾏 𖼮𖽪𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐， 6 𖼑𖽜 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼨𖽑𖽢𖾏 𖼮𖽪𖾐 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 、 𖼽𖽡𖾑 𖼊𖽑𖽗𖾐 、 𖼊𖽑𖽡 𖼞𖽙𖾑 𖼽𖽔𖾏 、 7 𖼀𖽳 · 𖼺𖽪𖾏 、 𖼺𖽙𖾑 𖼞𖽙𖾑 、 𖼽𖽔𖾏 𖼊𖽪𖾏 𖼖𖽔 、 8 𖼮𖽢𖾑 𖼊𖽑𖽳 、 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼳𖽔𖾑 、 𖼳𖽡𖾑 𖼇𖽪 、 9 𖼽𖽔𖾏 𖼊𖽙𖾑 𖼖𖽹 𖼽𖽡𖽾𖾑 、 𖼖𖽔𖾑 𖼮𖽡 、 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼮𖽢𖾑 、 10 𖼺𖽙𖾐 𖼖𖽔𖾑 、 𖼽𖽔𖾏 𖼽𖽔𖾐 𖼖𖽪𖽝 、 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 . 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 . 11 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼎𖽺𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 , 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 , 𖼎𖽺𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼡𖽻𖾐 𖼡𖽜𖾏 、 𖼛𖽙𖾏 、 𖼮𖽡𖽠𖾑 . 12 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼄𖽑𖽪𖾑 , 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼙𖽻 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 . 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼖𖽔𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼙𖽠𖾏 .
