𖽃𖽡𖾐 𖼸𖽙𖾐 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
66
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏！
2 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏！
𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏！
3 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏！
𖼑𖽜 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼽𖽔𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼟𖽡.
4 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼟𖽡, 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡,
𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖽐𖼷𖽷𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
5 𖼅𖽡 𖼋𖽔𖾏 𖼑𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏.
6 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽠𖾏 𖼋𖽡𖾏 𖼛𖽡𖾏,
𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼙𖽔 𖼚𖽙𖾏.
7 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏．
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼉𖽻𖾏 𖼞𖽪𖽙,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
8 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏.
9 𖼐𖽱𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼀𖽡𖾐 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼮𖽡𖽝𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖽃𖽔𖾏 𖼼𖽙.
10 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼙𖽔 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼀𖽡𖾐, 𖽐𖼮𖽔𖾏 𖼮𖽡𖽺𖾏 𖼀𖽡𖾐,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖽐𖼮𖽔𖾏 𖼮𖽡𖽺𖾏 𖼑𖽡𖽝𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐.
11 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏,
𖼄𖽔𖾐 𖽐𖼎𖽘𖾏 𖼐𖽑𖽤𖾑 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑.
12 𖼟𖽡 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼀𖽑𖽪𖾐 𖼼𖽙𖾐 𖼖𖽻𖾐 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖼀𖽡𖾐 𖽐𖼮𖽡𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽻𖾐, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏.
13 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼮𖽡𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏, 𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼏𖽵 𖼎𖽻𖾏 𖼟𖽡,
14 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏, 𖽐𖼚𖽙 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏.
15 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼛𖽙 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑,
𖼄𖽔𖾐 𖼞𖽡𖾑 𖼙𖽻𖾏 𖼾𖽘𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼟𖽡,
𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼟𖽡. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
16 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼅𖽡 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼑𖽻 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏.
17 𖼞𖽪𖾑 𖼙𖽔 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼐𖽱𖾏,
𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼷𖽻𖾑 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐.
18 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽔𖾑 𖼐𖽑𖽤𖾑 𖼎𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼮𖽵𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼑𖽻.
19 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏,
𖼐𖽱𖾏 𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
20 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼋𖽡𖽝！
𖼐𖽱𖾏 𖼦𖽡𖾐 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏.
