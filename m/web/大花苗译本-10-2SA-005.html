（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 11：1 - 9；14：1 - 7）
5
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 𖼁𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏, 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼡𖽹𖾏． 2 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡. 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖼙𖽔 𖼏𖽵 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽪𖾑 𖼾𖽨 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 , 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏.” 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼗𖽻 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 𖼁𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏. 4 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏. 5 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼖𖽡 𖼚𖽻𖾏 𖼖𖽑𖽡𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽑𖽡𖽝𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏.
𖼦𖽔𖾏 𖼊𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑
6 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑． 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼁𖽺𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽡 𖼛𖽜、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑, 𖼖𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼋𖽻𖾏 𖼟𖽡 .” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼁𖽺𖾏 𖼅𖽻𖾏. 7 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑, 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼖𖽪𖾑 𖼙𖽻 . 8 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼡𖽔 𖼋𖽱𖾏 𖼮𖽔𖾑 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼟𖽱𖾏 𖽃𖽻𖾐 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼳𖽪𖾐 𖽃𖽡𖾑.” 𖼋𖽳 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑、𖼐𖽔𖾐 𖼄𖽱𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏.” 9 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽐𖼚𖽙, 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼊𖽡𖾏 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻． 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼄𖽡𖾐 𖼘𖽙 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼞𖽡 𖼙𖽻 𖼯𖽪𖾏 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤𖾏 . 10 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼟𖽡𖾏 𖼾𖽙 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐.
11 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼖𖽔 𖼺𖽘𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼋𖽠𖾏、𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐, 𖼑𖽜 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻. 12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽔𖾑, 𖼎𖽺𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏. 13 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼎𖽺𖾏 𖼷𖽙𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼣𖽔𖾏 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏. 14 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼳𖽻𖾏 𖼟𖽪𖾏 : 𖼳𖽔𖾑 𖼄𖽪𖾐 𖼽𖽔𖾏、𖼳𖽙 𖼀𖽔、𖼐𖽔 𖼊𖽗𖾑、𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞、 15 𖼽𖽡𖾏 𖼳𖽢、𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽪𖾑 𖼽𖽔𖾏、𖼐𖽡 𖼇𖽺𖾑、𖼽𖽔𖾐 𖼇𖽺𖾑 𖼽𖽔𖾏、 16 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼄𖽔𖾐、𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼊𖽔𖾏、𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼖𖽡𖽝 .
𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼇𖽺𖾐 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 14：8 - 17）
17 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏. 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 , 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 . 18 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼋𖽘 , 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽡𖾏 𖼇𖽔 𖼽𖽡𖽾𖾑 𖼖𖽪𖾑 𖼦𖽘𖾑 . 19 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼋𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖼘𖽪? 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼘𖽪?” 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼯𖽡𖽙𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.” 20 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼀𖽔𖾑 𖼖𖽡 · 𖼀𖽑𖽡𖾐 𖼖𖽔𖾑 𖼳𖽡𖽾𖾑 , 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼊𖽱𖾑𖽑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽱𖾑𖽑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼀𖽔𖾑 𖼖𖽡 · 𖼀𖽑𖽡𖾐 𖼖𖽔𖾑 𖼳𖽡𖽾𖾑. 21 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽢𖾏 𖼁𖽠𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼮𖽡𖽝𖾏, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼯𖽡𖽙𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 22 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏, 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽡𖾏 𖼇𖽔 𖼽𖽡𖽾𖾑 𖼖𖽪𖾑 𖼦𖽘𖾑 . 23 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼅𖽻𖾏, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼷𖽷𖾑 𖼀𖽡𖾐 𖼞𖽘𖾐 𖼻𖽻𖾐 𖼺𖽘𖾑 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 24 𖼟𖽡 𖽃𖽡𖾐 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼀𖽡𖾐 𖼞𖽘𖾐 𖼻𖽻𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽙 𖼋𖽘𖾏.” 25 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏, 𖽐𖼊𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼀𖽔𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼺𖽳 .
