11 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏、𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 , 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 12 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼮𖽘𖾏 𖼐𖽡𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼳𖽦𖾑 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 . 13 𖼋𖽡𖾏 𖼀𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼳𖽡𖽠𖾏 𖽐𖼮𖽳𖾑 , 𖼐𖽡𖽙𖾐 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽠 𖼺𖽘𖾏 𖽐𖼮𖽳𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 . 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼽𖽔𖾑 𖼳𖽡𖽠𖾏 𖽐𖼮𖽳𖾑 𖼖𖽑𖽘𖾑.
