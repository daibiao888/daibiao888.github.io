𖽃𖽺𖾐! 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼊𖽜 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼑𖽜𖾏, 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼊𖽑𖽡𖽝𖾐 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏,
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼨𖽑𖽪𖾑,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
2 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼗𖽪𖾏,
𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖽐𖼊𖽻𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑.
3 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖼊𖽔𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼖𖽡𖽾.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽂𖽪𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼢𖽻, 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼛𖽙𖾏 𖼋𖽔𖾏.
 
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽐𖼎𖽙 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖼙𖽠𖾏 𖼞𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏,
𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼈𖽡𖽝𖾏,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏 𖽐𖼚𖽺𖾏 𖽐𖼎𖽙𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫
𖼽𖽔𖾑 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏.
5 𖼐𖽻 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼊𖽑𖽔𖾑 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼈𖽳𖾑 𖼣𖽔𖾑 𖼣𖽔𖾏,
𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ．
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,
𖼽𖽔𖾑 𖼮𖽑𖽡𖽻𖾏 𖽐𖼚𖽜𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑.
 
6 “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！ 𖼅𖽡 𖼊𖽱𖾐 𖼊𖽡𖾑 𖼢𖽻𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏. 7 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 𖼑𖽡𖽝𖾏、𖼊𖽪𖾑 𖼁𖽠𖾏 𖼞𖽪𖾐, 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖽝𖾏 𖼟𖽙𖾏 𖼾𖽨𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽙 𖽃𖽡𖾑, 𖼞𖽑𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏.
𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐
8 𖼽𖽔𖾏 𖼳𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏,
𖼅𖽔𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏, 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼏𖽦𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼐𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽹𖾏 𖼙𖽠𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 ,
𖼅𖽔𖾏 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
