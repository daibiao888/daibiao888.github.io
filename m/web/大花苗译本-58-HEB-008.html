8
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽹𖾑， 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼐𖽱𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏， 2 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑， 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝 𖽃𖽡𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏， 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 ， 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏． 3 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏， 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏． 4 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼖𖽑𖽘𖾑 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼋𖽘𖾏． 5 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏， 𖼟𖽪𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽪𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑， 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐， 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼟𖽡 𖽃𖽡𖾑 𖽃𖽔𖾏．” 6 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼻𖽻𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾑， 𖼊𖽪𖾑 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼏𖽵 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼻𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏． 7 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼊𖽪𖾑 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼺𖽻𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖽃𖽡𖾑 𖼋𖽘． 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏：
“ 𖼑𖽔𖾏 𖼘𖽪！𖼮𖽵𖾑 𖼐𖽑𖽡𖾏 ， 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏，
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏
𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼾𖽡𖾏，𖼎𖽺𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐．
9 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽤𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼋𖽡𖾏，
𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐，
𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑．
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏，
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏．𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏．
10 𖼮𖽵𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，
𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 ， 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ：
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐，
𖼎𖽺𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐，
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏．
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼾𖽨𖾏 𖼼𖽷𖾏 𖼼𖽙 𖽐𖼎𖽙 𖼾𖽨𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏， 𖼐𖽑𖽡𖾏： 𖼅𖽡 𖼽𖽔𖾑 𖼯𖽜𖾏 𖽐𖼎𖽑𖽹𖾏 𖼮𖽵𖾑，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼷𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖼸𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼯𖽜𖾏 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑．
12 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼦𖽡𖾐 𖼎𖽺𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏．
13 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼸𖽙 “ 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 ”， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼊𖽪𖾑 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽪𖾐 𖼋𖽘， 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼡𖽪𖾐， 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼗𖽻𖾏 𖽃𖽡𖾑， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏．
