𖼷𖽻𖾑 𖼖𖽔 𖼳𖽡 𖼣𖽔𖾑 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏
𖽃𖽡𖾐 𖼸𖽙𖾐 𖽐𖼞𖽻𖾏，𖼞𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏.
48
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐！
𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
2 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽪𖾑 𖼙𖽻 ,
𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 , 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽺𖾐 𖼙𖽔.
3 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼙𖽻 𖽐𖼚𖽙 𖼟𖽻 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼾𖽨𖾏 𖼢𖽳 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
 
4 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼦𖽡 𖼸𖽱 ,
𖽃𖽡𖾐 𖼸𖽻 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏.
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼛𖽙𖾏 𖼚𖽔𖾐,
𖼞𖽑𖽜 𖼑𖽜 𖼏𖽘 𖼀𖽜𖾑 .
6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖽐𖼎𖽙 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼊𖽔𖾑 𖼮𖽡𖽝𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼊𖽪𖾐 𖼄𖽙𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.
7 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼷𖽑𖽔𖾐 𖽐𖼊𖽱𖾑𖽑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 𖼑𖽡𖾏 𖽐𖼞𖽙𖾏.
8 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼙𖽻 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏 , 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 ,
𖼮𖽳𖾐 𖽐𖼊𖽱𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏.
10 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 , 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏,
𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼖𖽠𖾑 𖼞𖽡𖾑 .
11 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏, 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼙𖽔 ,
𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽻( “𖼙𖽻” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏” ) 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
 
12 𖼅𖽡 𖼽𖽔𖾑 𖼦𖽡 𖽐𖼮𖽡 𖼳𖽡 𖼣𖽔𖾑 ,
𖼚𖽻𖾐 𖼺𖽘𖾐 𖼦𖽡 𖽐𖼮𖽡 , 𖼮𖽔𖾑 𖼾𖽱 𖼙𖽻 𖽐𖼊𖽑𖽘𖾐.
13 𖼑𖽔𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼡𖽻𖾏 𖼙𖽻 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 ,
𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 ,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 𖼡𖽜𖾐.
14 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏 𖽃𖽔𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐, 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 .
