1
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼖𖽡𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘 𖼟𖽻𖾏 𖼯𖽡𖽙𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏， 2 𖼐𖽡𖽙𖾐 𖼷𖽑𖽷𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽘𖾏 𖼐𖽡𖾑， 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏，𖼎𖽺𖾏 𖼺𖽹𖾑 𖼮𖽘𖾏 𖼐𖽱𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑，𖼎𖽺𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜． 3 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼘𖽑𖽙𖾐，𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼋𖽡𖽝， 𖼐𖽱𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼮𖽑𖽳𖾑 𖼮𖽹𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜．𖼐𖽱𖾏 𖽐𖼷𖽔𖾑 𖼨𖽑𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽔𖾏 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽜𖾏 𖼊𖽔𖾑 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏． 4 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏，𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼷𖽷𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖼇𖽔𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽡𖽻𖾏 𖼇𖽔𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾑 𖼋𖽱 𖼙𖽔 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽑𖽡𖾏：
“ 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐，
𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼅𖽔𖾏 𖼟𖽡 ” 𖼖𖽺𖾑 ？
𖼎𖽺𖾏 𖼙𖽔 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼡𖽔 𖼖𖽡𖾏 𖼋𖽱𖾏 𖼐𖽑𖽡𖾏：
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏，
𖼐𖽱𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 ”𖼖𖽺𖾑？
6 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑， 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
“ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼐𖽱𖾏．”
7 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼊𖽪𖾐 𖼷𖽷𖾐， 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：
“𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼮𖽢𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 ，
𖼟𖽙𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 ．”
8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼊𖽪𖾐，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
“𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏，
𖼟𖽡 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏，𖼟𖽪𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳．
9 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳，
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼯𖽡𖽙𖾏 𖼛𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼘𖽑𖽡𖾑 𖼚𖽜𖾑 𖼟𖽡，
𖼖𖽔𖾑 𖼊𖽪𖾑 𖼘𖽑𖽡𖾑 𖼚𖽜𖾑 𖼟𖽡 𖼽𖽺 𖼗𖽔．”
10 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
“𖼮𖽵𖾑 𖼘𖽪！𖼟𖽡 𖼺𖽪𖾐 𖼁𖽱 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏，
𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏．
11 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏，𖼈𖽡𖽝𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖽝𖾏；
𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖽐𖼀𖽑𖽙𖾐 𖼺𖽘𖾏；
12 𖼟𖽡 𖼽𖽔𖾑 𖼦𖽡𖾏 𖼚𖽙𖾑 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐，𖼺𖽱𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏，
𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼺𖽱𖾏 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏，
𖼮𖽔 𖼅𖽔𖾏 𖼟𖽡 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏，
𖼟𖽡 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽘．”
13 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝𖾏 𖼙𖽔 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽑𖽡𖾏：
“𖼟𖽡 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏,
𖼋𖽙𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐.”
14 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏，𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼼𖽙，𖼑𖽜 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏？
