𖼑𖽜 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫
118
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐,
𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽷𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏！
 
2 𖼊𖽻𖾏 𖼅𖽪𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽑𖽡𖾏:
“𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽷𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏！”
3 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼾𖽡𖾏 𖼐𖽑𖽡𖾏:
“𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽷𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏！”
4 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽷𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏！”
 
5 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽱𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑, 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼇𖽘𖾑.
6 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏，
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑?
7 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏.
8 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼻𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
9 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼻𖽡𖽝𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐.
 
10 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏.
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑 , 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏.
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼡𖽔𖾑 𖼄𖽙𖾏 𖼄𖽪𖾑 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑,
𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼇𖽳𖾏 𖼺𖽘𖾏,
𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏.
13 𖼟𖽡 𖼨𖽑𖽪𖾏 𖼛𖽘𖾏 𖼊𖽑𖽠𖾐 𖼞𖽪𖾑 , 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼢𖽻,
𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏.
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 𖽐𖼀𖽘𖾏.
 
15 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏 𖽐𖼎𖽙 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼮𖽘𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
17 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽔, 𖼖𖽔𖾑 𖼯𖽡𖽝𖾏 𖼯𖽡𖽙 𖼺𖽷𖾐,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏.
18 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑,
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼋𖽔 .
 
19 𖼑𖽜 𖼞𖽪𖾑 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼅𖽻𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
20 𖼖𖽪𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔 𖼛𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼅𖽻𖾏！
 
21 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼋𖽘𖾏,
𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏！
 
22 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏,
𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽪𖾐 𖽐𖼞𖽔𖾐 𖼊𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼮𖽡𖽙𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑.
23 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏,
𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑.
24 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼏𖽵 𖼮𖽡𖽝𖾏,
𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑！
25 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏，
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏．
 
26 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏 𖼋𖽔𖾏, 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏，
𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏, 𖼑𖽜 𖼅𖽡 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏．
27 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽱𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐．
𖼅𖽡 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖼡𖽑𖽹𖾐 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 ,
𖼮𖽤𖾐 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼞𖽪𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑.
