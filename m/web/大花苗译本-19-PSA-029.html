29
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼀𖽙𖾏 𖼊𖽪𖾐 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏、𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
2 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼐𖽑𖽘𖾑 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
 
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽻𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 ,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽙𖾏 𖼺𖽙𖾐 𖼢𖽔𖾐,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼺𖽙𖾐 𖼢𖽔𖾐 𖼎𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑.
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏.
 
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏.
6 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽑𖽡𖽝𖾏 𖼺𖽱𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏,
𖼟𖽙𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼖𖽡𖽝 𖼚𖽑𖽡𖽝𖾏 𖼺𖽱𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼡𖽔𖾏 𖼻𖽻𖾏.
 
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽙𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼇𖽺𖾐 𖼺𖽘𖾏.
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 𖼮𖽢𖾑 𖼊𖽡𖾐 𖼺𖽷𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏.
 
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼷𖽹𖾏 𖼳𖽤𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏,
𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼞𖽑𖽔𖾏 𖼉𖽜 𖼟𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼺𖽘𖾏．
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼷𖽻𖾑 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏.
 
10 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽻𖾐,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏.
