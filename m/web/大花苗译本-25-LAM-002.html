𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐, 𖼇𖽺𖾏 𖼋𖽘𖾏 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼳𖽦𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,

13 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼦𖽡 𖽐𖼮𖽔 𖼖𖽺𖾑?
𖼳𖽡 𖼣𖽔𖾑 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼮𖽔𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐, 𖼟𖽙𖾏 𖼻𖽻𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼟𖽡 𖼖𖽺𖾑?
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 𖼘𖽑𖽙𖾐 𖼺𖽱𖾏 𖼳𖽡𖽠𖾑, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼞𖽑𖽙𖾏 𖼟𖽡 𖼖𖽺𖾑?
 
14 𖼟𖽡 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼑𖽜 𖼟𖽡 𖼁𖽙𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼟𖽙𖾏 𖼟𖽡 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 ,
𖼈𖽡𖽝𖾏 𖼡𖽑𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽐𖼎𖽙 𖼅𖽜 𖼡𖽑𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼎𖽻𖾏 𖼟𖽡.
 
15 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 𖽃𖽡𖾑 𖽐𖼀𖽔𖾏 𖼋𖽡𖾏 𖼚𖽙𖾏 𖼟𖽡．
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽪𖾑 𖼙𖽻, 𖼇𖽜𖾏 𖼇𖽜𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽑𖽡𖾏:
“𖼊𖽡𖽝𖾏 𖼾𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽺𖾐 𖼙𖽔 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼘𖽪𖾏?”
 
16 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏, 𖼎𖽺𖾏 𖼋𖽙𖾏 𖼐𖽑𖽡𖽝𖾑 𖼐𖽑𖽡𖾏:
“𖼀𖽡𖾐 𖽐𖼡𖽙𖾏 𖼖𖽻𖾐 𖼐𖽱𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏 𖼸𖽙 𖼋𖽘𖾏!
𖼀𖽡𖾐 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 𖼋𖽘𖾏!”
 
17 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼏𖽵 𖼊𖽡𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑,
𖼸𖽪𖾏 𖼋𖽘𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼏𖽵 𖼊𖽡𖾏 𖼮𖽡𖽝𖾏.
𖼐𖽱𖾏 𖼊𖽑𖽠𖾐 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏,
𖼟𖽙𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖼎𖽻𖾏 𖼟𖽡 ,
𖼟𖽙𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐.
 
18 𖼳𖽡 𖼣𖽔𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽡𖽝𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼮𖽵𖾑 .
𖼳𖽡 𖼣𖽔𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼘𖽪！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐, 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼦𖽡𖾐 𖼳𖽙𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼤𖽔𖾏 𖼤𖽔𖾏 𖽃𖽔 𖼅𖽔 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏.
 
19 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼷𖽑𖽔𖾏 𖼟𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽪𖾑 𖼯𖽡𖽹𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏,
𖼐𖽡𖽙𖾐 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽑𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖽐𖼮𖽢𖾑𖽑 𖽃𖽻𖾐．
𖼟𖽡 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽡𖾏 𖽐𖼷𖽜𖾏 𖼮𖽑𖽹𖾐 𖽐𖼮𖽑𖽡𖾏 𖼉𖽪𖾏 𖼢𖽻 𖼙𖽠𖾏,
𖼟𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼮𖽘𖾏 𖼋𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼮𖽵𖾑 .
 
20 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽔𖾏，
𖼋𖽡𖽝𖾏 𖼟𖽡 𖼙𖽔 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖽃𖽔 𖼁𖽙𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼑𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼘𖽑𖽘𖾑 𖼾𖽨 , 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼁𖽔 𖽃𖽡𖾑 𖼖𖽺𖾑?
𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼮𖽵𖾑 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼊𖽔𖾏 𖼖𖽺𖾑?
 
21 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼢𖽻 𖼋𖽔 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏,
𖼞𖽪𖾑 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼞𖽑𖽔𖾏 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼐𖽑𖽡𖽝𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐，
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
𖼟𖽡 𖼐𖽑𖽙𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏.
 
22 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼦𖽡 𖼸𖽱 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏．
𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼁𖽔 𖼡𖽑𖽙𖾑 𖼊𖽪𖾑 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑,
𖼞𖽡𖾑 𖼀𖽔𖾑 𖼞𖽑𖽔𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
