𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼄𖽙𖾑 𖼽𖽔𖾏
16
𖼅𖽡 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼎𖽑𖽻𖾏 𖼳𖽡𖾑 𖼖𖽔𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏, 𖼺𖽘𖾏 𖽐𖼮𖽡𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼙𖽻 𖼖𖽪𖾑 𖼎𖽻𖾐(“𖼙𖽻” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏”).
2 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 (“𖼖𖽡𖾑 𖼉𖽪𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏”) 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼐𖽺𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏,
𖼽𖽔𖾑 𖼺𖽱𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼮𖽡 , 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼉𖽳𖾏.
 
3 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼋𖽠𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼐𖽡𖽙𖾐 𖼋𖽘 𖼐𖽑𖽪𖾐 𖼺𖽱𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼈𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼏𖽘 𖼀𖽜𖾑 .
4 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼯𖽪𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐,
𖼷𖽑𖽜𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼞𖽻𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼳𖽦𖾑 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏.
 
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼼𖽠𖾏 𖼀𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼺𖽘𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏.
5 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼎𖽑𖽻𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼐𖽡𖽙𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐,
𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏, 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 .
𖼄𖽙𖾑 𖼽𖽔𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
6 𖼀𖽡𖾐 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏,
𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼐𖽙𖾑𖽑 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐 𖼾𖽨𖾏 𖼦𖽡𖾑 𖼚𖽘𖾐 𖽐𖼊𖽪𖾏、 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏、 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼐𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼊𖽔𖾐.
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼄𖽙𖾑 𖼽𖽔𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏．
𖼅𖽡 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼮𖽡 𖽃𖽳𖾐 · 𖼦𖽔𖾐 𖼖𖽡𖽝 𖼳𖽳 𖼑𖽡𖾏 𖼳𖽡𖽝𖾐 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽐𖼷𖽘𖾏 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏,
𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .
 
8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼳𖽷𖾏 𖼀𖽺𖾐 𖼑𖽡𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼄𖽔𖾐 𖼑𖽡𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼞𖽑𖽔𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽃𖽡𖾑,
𖼊𖽪𖾑 𖽃𖽔 𖼯𖽡 𖼐𖽡𖾑 𖼘𖽑𖽙𖾐 𖼸𖽙 𖼽𖽔𖾐 𖼳𖽡𖽝𖾏 𖽐𖼊𖽡𖾑 𖼸𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏,
𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 , 𖼮𖽑𖽡𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼮𖽳𖾑 .
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼄𖽔𖾐 𖼑𖽡𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏,
𖼺𖽱𖾏 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼳𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽝𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
𖼳𖽡𖾑 𖼳𖽷𖾏 𖼀𖽺𖾐 、 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼖𖽡𖾏 𖼘𖽪 !
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽔𖾏 𖼅𖽔 𖽐𖼊𖽙𖾑 𖼟𖽡, 𖼟𖽡 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖽐𖼎𖽙 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏,
𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏 𖽐𖼮𖽦 𖼗𖽠𖾏 𖼋𖽘𖾏.
10 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼛𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼐𖽡𖽙𖾐 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽐𖼚𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏．
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼮𖽡𖽠𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼸𖽹𖾏 𖼮𖽡𖽠𖾑 𖼋𖽘𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏 𖽐𖼮𖽦 𖼊𖽙𖾏 𖼺𖽘𖾏 .
11 𖽃𖽔𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼑𖽜 𖼄𖽙𖾑 𖼽𖽔𖾏 𖽐𖼷𖽘𖾏 𖽐𖼎𖽙𖾑 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼢𖽜𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽐𖼀𖽱𖾏 𖼑𖽜 𖼮𖽡 𖽃𖽳𖾐 · 𖼦𖽔𖾐 𖼖𖽡𖽝 𖼳𖽳 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.
12 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼎𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼖𖽔𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 .
13 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼄𖽙𖾑 𖼽𖽔𖾏 . 14 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 𖼐𖽡𖾑 𖽐𖼚𖽙 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏 𖼑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏, 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 , 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽺𖾐 𖼯𖽠 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 .”
