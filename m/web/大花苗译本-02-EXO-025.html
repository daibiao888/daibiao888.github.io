10 “𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼼𖽪𖾏. 𖽐𖼊𖽡𖾑 𖽃𖽔𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 𖽐𖼊𖽘𖾏 , 𖼇𖽘𖾑 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 𖽐𖼊𖽘𖾏 , 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 𖽐𖼊𖽘𖾏 . 11 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖼡𖽑𖽠𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏 , 𖽐𖼚𖽙 𖼢𖽻𖾏 𖼊𖽪𖾏 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏 , 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼎𖽑𖽹𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏. 12 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼘𖽑𖽡𖾑 𖼚𖽻𖾐 𖼖𖽪𖾐 𖽃𖽔 𖼉𖽙𖾏 𖼞𖽪𖾐 , 𖼊𖽡𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼚𖽻𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 , 𖼺𖽘𖾑 𖼐𖽡𖾑 𖽃𖽔𖾐 𖼖𖽪𖾐 , 𖼺𖽘𖾑 𖽃𖽡𖾑 𖽃𖽔𖾐 𖼖𖽪𖾐 . 13 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 , 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏 . 14 𖼖𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽠𖾏 𖼙𖽡 𖼎𖽻𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼟𖽙𖾏 𖼻𖽻𖾏 𖼞𖽱𖾑 𖼖𖽔𖾑 𖼼𖽪𖾏 . 15 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼐𖽡𖾑 𖼖𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼙𖽡 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔 𖼉𖽙𖾏 𖼮𖽡𖽝𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 . 16 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖽐𖼚𖽙 𖼼𖽪𖾏 𖼮𖽡𖽝𖾏 .” 17 “𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 (“ 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼷𖽑𖽹𖾐 𖼇𖽻𖾑 𖼟𖽪𖾏 𖼯𖽔𖾏 𖼾𖽔𖾏”) , 𖽐𖼊𖽡𖾑 𖽃𖽔𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 𖽐𖼊𖽘𖾏 , 𖼇𖽘𖾑 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 𖽐𖼊𖽘𖾏 . 18 𖼖𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖽐𖼊𖽪𖾏 𖼋𖽠𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼗𖽙𖾏 , 𖼊𖽡𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 . 19 𖼺𖽘𖾑 𖼐𖽡𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 , 𖼺𖽘𖾑 𖽃𖽡𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 , 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼦𖽡𖾏 𖼻𖽡𖽝𖾏 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 . 20 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖽃𖽔𖾑 𖼊𖽡𖾏 , 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 , 𖼽𖽔𖾑 𖼚𖽑𖽪𖾏 𖼊𖽱𖾐 𖼚𖽑𖽪𖾏 , 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 . 21 𖼖𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼼𖽪𖾏 . 22 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼦𖽡𖾏 𖼁𖽙𖾏 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 , 𖼟𖽙𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 .”

23 “𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼸𖽱𖾏 𖼞𖽡 𖼏𖽻𖾏 , 𖽐𖼊𖽡𖾑 𖽃𖽔𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼇𖽘𖾑 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 𖽐𖼊𖽘𖾏 . 24 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏 , 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼎𖽑𖽹𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏. 25 𖼞𖽡 𖼏𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼟𖽪𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼇𖽘𖾑 𖼚𖽻𖾐 𖽐𖼊𖽡𖾏 , 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼞𖽑𖽡 𖼙𖽻 𖼎𖽑𖽹𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏. 26 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼖𖽪𖾐 𖽃𖽔 𖼉𖽙𖾏 𖼞𖽪𖾐 , 𖼊𖽡𖾏 𖼎𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼞𖽡 𖼏𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏. 27 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽡𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼽𖽔𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼟𖽪𖾏 𖼊𖽘𖾑 , 𖼟𖽙𖾏 𖼻𖽻𖾏 𖼙𖽡 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼞𖽡 𖼏𖽻𖾏 . 28 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 𖽃𖽡𖾑 , 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏 , 𖼟𖽙𖾏 𖼻𖽻𖾏 𖼞𖽱𖾑 𖼞𖽡 𖼏𖽻𖾏 . 29 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡 𖼏𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑、𖽃𖽔𖾑 𖼯𖽱𖾏 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑、𖼨𖽑𖽻𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖽐𖼎𖽙 𖼞𖽑𖽙𖾐 , 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖽃𖽔𖾏 . 30 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼞𖽡 𖼏𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼊𖽜𖾐 𖼮𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽡𖽙𖾐 𖼳𖽡𖽝𖾐.”
𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼞𖽪𖾐
（𖼋𖽠𖾏 37：17 - 24）
31 “𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 . 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖽐𖼊𖽻𖾏、𖼞𖽑𖽙𖾐、𖽃𖽔𖾑 𖼖𖽪𖾐、𖽃𖽔𖾏 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼁𖽘𖾏 、𖼁𖽘𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼦𖽡𖾏 𖼊𖽡𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖽐𖼊𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 . 32 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽠𖾏 𖼚𖽻𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼯𖽡 , 𖼺𖽘𖾑 𖼐𖽡𖾑 𖼷𖽷𖾐 𖼯𖽡 , 𖼺𖽘𖾑 𖽃𖽡𖾑 𖼷𖽷𖾐 𖼯𖽡 . 33 𖼺𖽘𖾑 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼖𖽪𖾐 𖼞𖽑𖽙𖾐 , 𖼼𖽘𖾏 𖼺𖽱𖾏 𖼁𖽘𖾏 𖼷𖽷𖾑 𖼞𖽑𖽡𖽠𖾐 , 𖼅𖽔𖾏 𖽃𖽔𖾏 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼁𖽘𖾏、𖼅𖽔𖾏 𖼁𖽘𖾏 , 𖼺𖽘𖾑 𖽃𖽡𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼖𖽪𖾐 𖼞𖽑𖽙𖾐 , 𖼼𖽘𖾏 𖼺𖽱𖾏 𖼁𖽘𖾏 𖼷𖽷𖾑 𖼞𖽑𖽡𖽠𖾐 , 𖼅𖽔𖾏 𖽃𖽔𖾏 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼁𖽘𖾏、𖼅𖽔𖾏 𖼁𖽘𖾏 . 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖽐𖼮𖽡𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼚𖽻𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 . 34 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼽𖽘𖾑 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼖𖽪𖾐 𖼞𖽑𖽙𖾐 , 𖼼𖽘𖾏 𖼺𖽱𖾏 𖼁𖽘𖾏 𖼷𖽷𖾑 𖼞𖽑𖽡𖽠𖾐 , 𖼅𖽔𖾏 𖽃𖽔𖾏 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼁𖽘𖾏 、𖼅𖽔𖾏 𖼁𖽘𖾏 . 35 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼯𖽡 , 𖼅𖽔𖾏 𖽃𖽔𖾏 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼁𖽘𖾏 , 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼯𖽡 𖼦𖽡𖾏 𖼻𖽡𖽝𖾏 𖼦𖽡𖾏 𖼊𖽡𖾏 , 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽠𖾏 𖼚𖽻𖾏 𖼋𖽪 𖽃𖽔𖾑 𖼯𖽡 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 . 36 𖽃𖽔𖾑 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼁𖽘𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼯𖽡 𖼽𖽔𖾑 𖼦𖽡𖾏 𖼻𖽡𖽝𖾏 𖼦𖽡𖾏 𖼊𖽡𖾏 , 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖽐𖼊𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 . 37 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖽃𖽡𖾑 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾏 , 𖼋𖽡𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 , 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼮𖽡𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼐𖽡𖾑 , 𖼟𖽙𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼊𖽱𖾐 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 . 38 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼷𖽑𖽳𖾏 𖽐𖼎𖽙 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖽃𖽔𖾏 . 39 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 , 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖽃𖽡𖾐 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳. 40 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼟𖽡 𖽃𖽡𖾑 𖽃𖽔𖾏 .”
