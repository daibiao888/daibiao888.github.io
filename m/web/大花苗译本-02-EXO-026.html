𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐
（𖼋𖽠𖾏 36：8 - 38）
26
“𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽻𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 . 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 𖽐𖼎𖽙 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖼗𖽙𖾏 𖽃𖽔𖾏 , 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼎𖽻𖾏 . 2 𖼷𖽑𖽔𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼽𖽔𖾑 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼾𖽡𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼇𖽘𖾑 𖼚𖽻𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖽐𖼊𖽻𖾐 𖼽𖽔𖾑 𖽐𖼊𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 . 3 𖼀𖽜𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼛𖽔𖾏 𖼽𖽔𖾑 𖼦𖽡𖾏 𖼊𖽡𖾏 , 𖼀𖽜𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖽃𖽡𖾑 𖼷𖽑𖽔𖾏 𖼛𖽔𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼦𖽡𖾏 𖼊𖽡𖾏 . 4 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼐𖽡𖾑 𖼛𖽔𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼡𖽑𖽙𖾑 𖼇𖽺𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑 𖼎𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖽃𖽡𖾑 ,𖼛𖽔𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 . 5 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽪𖾏 𖼡𖽑𖽙𖾑 𖼇𖽺𖾑 , 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼖𖽪𖾏 𖼡𖽑𖽙𖾑 𖼇𖽺𖾑 , 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 . 6 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼋𖽪𖾏 𖽐𖼡𖽜𖾏 𖼞𖽪𖾐 , 𖼎𖽑𖽻𖾏 𖽐𖼡𖽜𖾏 𖽐𖼡𖽜𖾏 𖼟𖽙𖾏 𖽐𖼊𖽻𖾐 𖼦𖽡𖾏 𖼊𖽡𖾏 , 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼊𖽡𖽝𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 .”
7 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼚𖽻𖾐 𖼮𖽑𖽷𖾏 𖽐𖼊𖽙𖾏 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 , 𖽃𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼞𖽑𖽱𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 . 8 𖼷𖽑𖽔𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼽𖽔𖾑 𖽐𖼊𖽡𖾑 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼇𖽘𖾑 𖼚𖽻𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖽐𖼊𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 . 9 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼀𖽜𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼦𖽡𖾏 𖼊𖽡𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼀𖽑𖽙𖾐 , 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼚𖽻𖾏 𖼛𖽔𖾏 𖼦𖽡𖾏 𖼊𖽡𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼀𖽑𖽙𖾐 , 𖼚𖽻𖾏 𖼛𖽔𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼡𖽳𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 . 10 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼐𖽡𖾑 𖼛𖽔𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽪𖾏 𖼡𖽑𖽙𖾑 𖼇𖽺𖾑 , 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼚𖽑𖽙𖾏 𖽃𖽡𖾑 ,𖼛𖽔𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽪𖾏 𖼡𖽑𖽙𖾑 𖼇𖽺𖾑 . 11 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼋𖽪𖾏 𖽐𖼡𖽜𖾏 𖼋𖽻𖾏 , 𖽐𖼡𖽜𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼇𖽺𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼟𖽙𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼞𖽑𖽱𖾐 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼊𖽔𖾐 . 12 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼞𖽑𖽱𖾐 𖼳𖽙𖾏 𖼚𖽻𖾑 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖼚𖽻𖾑 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖽃𖽔𖾐 𖼡𖽜𖾐 . 13 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼞𖽑𖽱𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽐𖼊𖽡𖾑 𖽃𖽡𖾑 , 𖼺𖽘𖾑 𖼐𖽡𖾑 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼺𖽘𖾑 𖽃𖽡𖾑 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼽𖽔𖾑 𖼚𖽻𖾑 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑 𖽃𖽔𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖽐𖼀𖽙 𖼈𖽳𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 . 14 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼚𖽻𖾏 𖼖𖽡𖽝𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼊𖽑𖽡𖽝𖾐 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼳𖽡𖽠𖾑 𖼚𖽡𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔𖾏 𖼀𖽑𖽙𖾑 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼊𖽔𖾐 𖽃𖽔𖾑 𖼳𖽔𖾐 .”
