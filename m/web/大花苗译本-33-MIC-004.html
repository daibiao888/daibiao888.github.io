𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪
（𖼺𖽹 2：2 - 4）
4
𖽃𖽔𖾑 𖼡𖽜𖾐 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏,
𖼮𖽑𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼇𖽔𖾑 𖼉𖽻𖾏 𖼩𖽜 ,
𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖽐𖼚𖽱𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑 .
2 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼐𖽑𖽡𖾏:
“𖼋𖽔𖾏 𖼘𖽪! 𖼀𖽡𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐,
𖼚𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏．
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾐,
𖼀𖽡𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 .”
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼳𖽡 𖼣𖽔𖾑 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 .”
3 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐,
𖼑𖽜 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼚𖽡𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽑𖽜 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖽝 𖼮𖽑𖽵𖾑.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼇𖽺𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼄𖽑𖽪𖾑 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔𖾐 𖼖𖽡𖽝𖾐.
𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽻𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼉𖽺𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 .
4 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐
𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼻𖽡𖽝𖾏 𖼾𖽨𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏,
𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏
𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼳𖽡 𖼣𖽔𖾑
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖽃𖽡𖾑,
𖼦𖽡 𖼸𖽱 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑,
𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽃𖽡𖾑.
7 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖽃𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏,
𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼎𖽑𖽻𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.”
 
8 𖼟𖽡， 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼼𖽙𖾐 𖼐𖽡𖾑 𖼘𖽪！
𖼳𖽡 𖼣𖽔𖾑 𖼙𖽻 ( “𖼙𖽻”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 ”) 𖼑𖽡𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼘𖽪 !
𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼟𖽡 .
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼑𖽡𖾏 𖼞𖽪𖽙 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼟𖽡 .
𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑
9 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼟𖽡 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑 𖼨𖽑𖽪𖾏 𖼖𖽺𖾑?
𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏?
𖼟𖽡 𖼊𖽪𖾐 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼚𖽙𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼖𖽔𖾏?
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼟𖽡 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼊𖽪𖾐 𖼋𖽘𖾏 𖼖𖽔𖾏?
10 𖼳𖽡 𖼣𖽔𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 (“𖼖𖽡𖾑 𖼉𖽪𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 ”) 𖼘𖽪! 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼀𖽱𖾏 𖼄𖽙𖾐,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼊𖽪𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏,
𖼅𖽻𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏,
𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼎𖽙𖾑 𖼟𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
11 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼟𖽡 , 𖼐𖽑𖽡𖾏:
“𖼊𖽻𖾏 𖼅𖽪𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼊𖽻𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏.”
