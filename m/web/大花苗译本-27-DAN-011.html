𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼊𖽹𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽔𖾏 𖼖𖽡𖾏 𖽂𖽪𖾑 𖼺𖽪𖾐 𖼁𖽱 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼙𖽔 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼖𖽳 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏. 2 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼟𖽡. 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼺𖽷𖾐, 𖼚𖽻𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏. 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼙𖽳𖾐 𖼷𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼳𖽡𖾑 𖼖𖽔 𖼞𖽪𖽙 .
3 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏, 𖼊𖽔𖾑 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏. 4 𖼐𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏, 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽪𖽙 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼐𖽱𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼚𖽙𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽃𖽡𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
