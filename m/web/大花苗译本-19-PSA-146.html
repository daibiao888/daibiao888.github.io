𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 !
 
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫．
2 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫；
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼺𖽷𖾏, 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
 
3 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏.
4 𖼐𖽱𖾏 𖽂𖽪𖾏 𖼊𖽪𖾏 𖼀𖽘𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼁𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼙𖽠𖾏 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽐𖼊𖽱𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼷𖽹𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
 
5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼀𖽘𖾏,
𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏、𖼊𖽡𖾐、𖼳𖽡𖽠𖾑, 𖽐𖼎𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼯𖽢 𖼋𖽡𖾏 𖼟𖽻𖾏 𖼮𖽘𖾑,
𖼐𖽱𖾏 𖼻𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
7 𖼐𖽱𖾏 𖼑𖽜 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖽃𖽡𖾑 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖼄𖽔𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖽃𖽡𖾑.
 
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼚𖽠𖾏 𖼮𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼞𖽱𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼮𖽡𖽝𖾏.
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽜𖾐 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽔 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽘𖾏 𖼳𖽠𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏．
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑, 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼖𖽡𖾐 𖽐𖼞𖽑𖽡𖾐 𖼖𖽡𖾏 𖽐𖼞𖽑𖽻𖾏.
 
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏！
𖼳𖽡 𖼣𖽔𖾑 𖼘𖽪！ 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼉𖽻𖾏 𖼷𖽑𖽷𖾏！
