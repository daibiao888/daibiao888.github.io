13 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼐𖽙𖾑𖽑 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏，𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 “𖼀𖽔𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐”，𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽳 𖼀𖽔𖾑 𖼊𖽔𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏． 14 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼯𖽡𖽝𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏，𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖽐𖼮𖽳．𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼑𖽔𖾏 𖼘𖽪！ 𖼅𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏！” 15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼋𖽠𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏！𖼋𖽠𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏！𖽐𖼮𖽦𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐！” 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽦𖾏 𖼅𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐 𖼘𖽪𖾏？” 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼅𖽡 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼋𖽠𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐 𖼺𖽘𖾏， 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．” 16 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽦𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐．

23 𖼊𖽪𖾐 𖼏𖽙𖾏 𖽐𖼮𖽦𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽙𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼚𖽻𖾐 𖽐𖼀𖽠， 𖽃𖽡𖾐 𖼗𖽜𖾏 𖼏𖽙 𖼊𖽻𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠．𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼄𖽡𖾐 𖼋𖽱𖾐，𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼄𖽡𖾐 𖼋𖽱𖾐 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼞𖽘𖾐 𖼟𖽪𖾏 𖼺𖽠𖾏，𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖽐𖼊𖽙𖾏 𖼸𖽙 𖼀𖽡𖾑 𖼊𖽡𖾐． 24 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽐𖼮𖽑𖽷𖾏 𖼺𖽘𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼊𖽻𖾏．” 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：
“𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼇𖽺𖾐 𖼞𖽪𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏，
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼮𖽑𖽙𖾏 𖼄𖽡𖾐 𖼋𖽱𖾐 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠．”
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼊𖽪𖾐 𖼏𖽙𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 25 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼖𖽔𖾑 𖼞𖽱𖾑，𖼊𖽑𖽡𖽝𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼟𖽪𖾏 𖼞𖽳 𖼘𖽙 𖼀𖽔𖾏 𖼐𖽤𖾐，𖽐𖼎𖽙 𖼄𖽙 𖼊𖽔𖾏 𖼖𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏，𖼳𖽠𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐 𖽃𖽡𖾐 𖼺𖽘𖾐． 26 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼳𖽠𖾑 𖽃𖽡𖾐 𖼺𖽘𖾐，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼐𖽑𖽡𖾏：“ 𖽃𖽔 𖼁𖽙𖾏， 𖼑𖽔𖾏 𖼘𖽪！ 𖼟𖽡 𖼊𖽪𖾐．” 27 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“ 𖼑𖽔𖾏 𖼘𖽪！𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐．” 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼳𖽠𖾑 𖼁𖽱𖾏，𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾑 𖼷𖽹𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏．
