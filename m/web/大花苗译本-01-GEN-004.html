𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼞𖽹𖾑 𖼽𖽡𖾑、𖼽𖽔𖾏 𖼀𖽳

𖼽𖽔𖾏 𖼊𖽘𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼳𖽢𖾏 𖽂𖽔𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏, 𖼳𖽢𖾏 𖽂𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏, 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼞𖽹𖾑 𖼽𖽡𖾑 (“𖼞𖽹𖾑 𖼽𖽡𖾑 𖼟𖽪𖾏 “𖼊𖽻𖾏”), 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐.” 2 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼞𖽹𖾑 𖼽𖽡𖾑 𖼑𖽡𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼽𖽔𖾏 𖼀𖽳. 𖼽𖽔𖾏 𖼀𖽳 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼯𖽦𖾏； 𖼞𖽹𖾑 𖼽𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐. 3 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐, 𖼞𖽹𖾑 𖼽𖽡𖾑 𖼞𖽑𖽡𖽠𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼑𖽡𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫； 4 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼊𖽔𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼾𖽘𖾏 𖽐𖼎𖽙 𖼾𖽘𖾏 𖼑𖽡𖾏 𖼛𖽙𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏． 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼏𖽻 𖼽𖽔𖾏 𖼀𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑, 5 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼑𖽔𖾏 𖼦𖽡𖾏 𖼏𖽻 𖼞𖽹𖾑 𖼽𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑. 𖼞𖽹𖾑 𖼽𖽡𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼚𖽑𖽡𖾏 𖼺𖽘𖾏. 6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽹𖾑 𖼽𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼘𖽪? 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼚𖽑𖽡𖾏 𖼺𖽘𖾏 𖼘𖽪? 7 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏？ 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏, 𖼾𖽔𖾏 𖼎𖽺𖾏 𖼸𖽺𖾏 𖽐𖼚𖽺𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔 𖼛𖽻𖾏． 𖼾𖽔𖾏 𖼽𖽔𖾑 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼟𖽡, 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏.”
𖼞𖽹𖾑 𖼽𖽡𖾑 𖼊𖽔𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏
8 𖼞𖽹𖾑 𖼽𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼽𖽔𖾏 𖼀𖽳 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾐 𖼖𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐.” 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽹𖾑 𖼽𖽡𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼽𖽔𖾏 𖼀𖽳, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏. 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽹𖾑 𖼽𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼽𖽔𖾏 𖼀𖽳 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐！ 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏?” 10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼺𖽘𖾏? 𖼟𖽡 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽐𖼷𖽘𖾏 𖼐𖽑𖽡𖾏． 11 𖼊𖽡𖾐 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼟𖽡 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑． 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏. 12 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐, 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽜 𖼟𖽡 𖼋𖽠𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏； 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽔𖾏 𖽐𖼮𖽡 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐.” 13 𖼞𖽹𖾑 𖼽𖽡𖾑 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏. 14 𖼑𖽔𖾏 𖼘𖽪！ 𖼟𖽡 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼖𖽠𖾑 𖼞𖽪𖾑 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑, 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼟𖽡 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏； 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽔𖾏 𖽐𖼮𖽡 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽷𖾐 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽔𖾏 𖼞𖽪𖾑.” 15 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼟𖽡 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼳𖽤𖾏 𖽐𖼀𖽠.” 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽹𖾑 𖼽𖽡𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽷𖾐 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼊𖽔𖾏 𖼐𖽱𖾏. 16 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼞𖽹𖾑 𖼽𖽡𖾑 𖼇𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏, 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼐𖽙 𖼊𖽳 (“ 𖼐𖽙 𖼊𖽳”𖼟𖽪𖾏 “𖼳𖽔𖾏 𖽐𖼮𖽡”) 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.

17 𖼞𖽹𖾑 𖼽𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼀𖽱𖾏, 𖼐𖽱𖾏 𖼐𖽤𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏, 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼐𖽙. 𖼞𖽹𖾑 𖼽𖽡𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻, 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 𖽐𖼷𖽷𖾏, 𖼊𖽡𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼐𖽙. 18 𖼽𖽡𖾑 𖼐𖽙 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼐𖽔, 𖼽𖽡𖾑 𖼐𖽔 𖼅𖽔𖾏 𖼄𖽡𖾐 𖼦𖽪𖾏 𖼽𖽔𖾐 𖼖𖽡𖾏, 𖼄𖽡𖾐 𖼦𖽪𖾏 𖼽𖽔𖾐 𖼖𖽡𖾏 𖼅𖽔𖾏 𖼄𖽔𖾐 𖼊𖽑𖽪𖾐 𖼺𖽔𖾐 𖼖𖽡𖾏, 𖼄𖽔𖾐 𖼊𖽑𖽪𖾐 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼅𖽔𖾏 𖼖𖽔𖾑 𖼄𖽳. 19 𖼖𖽔𖾑 𖼄𖽳 𖼷𖽙𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼊𖽔𖾏, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽷𖾏 𖼳𖽡𖾐 𖼖𖽔𖾑. 20 𖼽𖽔𖾏 𖼊𖽔𖾏 𖼅𖽔𖾏 𖼽𖽔𖾐 𖼀𖽔, 𖼽𖽔𖾐 𖼀𖽔 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼾𖽨 𖼯𖽦𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐. 21 𖼽𖽔𖾐 𖼀𖽔 𖼑𖽡𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽨 𖼀𖽔； 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼊𖽪𖾐 𖼷𖽑𖽙𖾑 𖼖𖽡𖾑 𖼖𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐. 22 𖼳𖽡𖾐 𖼖𖽔𖾑 𖼎𖽺𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼊𖽑𖽪𖾐 𖼀𖽔·𖼞𖽹𖾑 𖼽𖽡𖾑； 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼋𖽻𖾏 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽳𖾏 𖽃𖽡𖾑. 𖼊𖽑𖽪𖾐 𖼀𖽔·𖼞𖽹𖾑 𖼽𖽡𖾑 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼐𖽔 𖼄𖽔𖾐. 23 𖼖𖽔𖾑 𖼄𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐 𖼐𖽑𖽡𖾏:
“𖼽𖽔𖾏 𖼊𖽔𖾏、𖼳𖽡𖾐 𖼖𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏；
𖼞𖽪𖾑 𖼐𖽤𖾐 𖼻𖽻𖾏 𖼻𖽻𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏；
𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼊𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏；
𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏．
24 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼞𖽹𖾑 𖼽𖽡𖾑, 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼳𖽤𖾏 𖽐𖼀𖽠,
𖼊𖽔𖾏 𖼖𖽔𖾑 𖼄𖽳, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼳𖽤𖾏 𖽐𖼀𖽠.”
25 𖼽𖽔𖾏 𖼊𖽘𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼀𖽱𖾏, 𖼳𖽢𖾏 𖽂𖽔𖾑 𖼎𖽺𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼊𖽡𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼺𖽳 𖼊𖽑𖽳, 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼗𖽙𖾏 𖽐𖼷𖽷𖾑 𖼽𖽔𖾏 𖼀𖽳, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽹𖾑 𖼽𖽡𖾑 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.” 26 𖼺𖽳 𖼊𖽑𖽳 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼐𖽙 𖼺𖽷𖾏. 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 (“𖼨𖽑𖽪𖾏 𖽐𖼊𖽜” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 “𖼖𖽑𖽘𖾑 𖼸𖽔𖾏”) 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏.
