8
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼺𖽱𖾏 𖼞𖽪𖾑 𖼷𖽔𖾑 𖼞𖽱𖾑 ,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼄𖽡𖾏,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 𖽐𖼮𖽷𖾐 𖼟𖽡 , 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼦𖽡𖾐 𖼡𖽑𖽻𖾐,
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼞𖽪𖾑 .
2 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 ,
𖼨𖽑𖽳𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼦𖽻𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏.
3 𖼐𖽱𖾏 𖼳𖽤𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼇𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏,
𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼁𖽔 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏.
4 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪 !
𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼮𖽑𖽙, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼯𖽷𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏, 𖼋𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏, 𖼋𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽪𖾏).
𖼚𖽻𖾏 𖼸𖽙𖾏
𖼨𖽑𖽪𖾑 𖼁𖽜 :
5 𖼖𖽜𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 :
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽻𖾏 𖼀𖽑𖽡𖽾 𖽐𖼞𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼨𖽑𖽪𖾏 𖼯𖽷𖾏 𖼟𖽡 ,
𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼑𖽜 𖼟𖽡 𖽐𖼷𖽜𖾏 𖼄𖽙𖾐,
𖼖𖽜𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼑𖽜 𖼟𖽡 𖽐𖼷𖽜𖾏 𖼄𖽙𖾐.
 
6 𖼮𖽑𖽙 𖼟𖽡 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖼖𖽪𖾑 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖼐𖽱𖾏 𖼺𖽻𖾏,
𖼐𖽑𖽘𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏 𖼺𖽱𖾏 𖼀𖽡𖾏 𖼡𖽔𖾏 𖼟𖽪𖾏 𖼮𖽑𖽡𖽙𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖼺𖽱𖾏 𖼋𖽔 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼐𖽱𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼮𖽡𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼮𖽡𖾏, 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏.
7 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏, 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼊𖽙𖾑 𖼋𖽔 ,
𖼛𖽡𖾏 𖽃𖽻𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽑𖽪𖾐 𖼺𖽘𖾏,
𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼽𖽔𖾑 𖼘𖽪𖾑𖽑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏,
𖼖𖽔𖾑 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.
 
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼅𖽡 :
8 𖼀𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑 ,
𖼐𖽱𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼄𖽡𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼘𖽑𖽙𖾐 𖼺𖽷𖾐,
𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏,
𖼀𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ?
9 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼮𖽑𖽤 ,
𖼀𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼳𖽔𖾑 𖼷𖽑𖽙𖾏 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼑𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼛𖽻𖾏,
𖼀𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼐𖽱𖾏 .
 
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐:
10 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼮𖽑𖽤 ,
𖼞𖽪𖾑 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼄𖽡𖾏 𖼺𖽱𖾏 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼳𖽔𖾐.
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖽃𖽡𖾐 𖼼𖽘𖾐.
𖼈𖽻𖾑 𖼮𖽑𖽦𖾐：
11 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽡𖽙𖾐 𖼀𖽔𖾑 𖼖𖽡 · 𖼦𖽔𖾐 𖼄𖽞 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 ,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑,
𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏.
12 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼘𖽪 ! 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼟𖽪𖾏 𖼟𖽡 𖼁𖽡𖽝 ,
𖽃𖽔𖾐 𖼀𖽔𖾏 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖽃𖽡𖾑.
 
13 𖼟𖽡 𖼖𖽜𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖼐𖽡𖾑,
𖼽𖽺 𖼗𖽔 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼑𖽻 𖼟𖽡 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼐𖽙𖾑𖽑 𖼊𖽑𖽡𖽝𖾐.
 
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 :
14 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼘𖽪! 𖼮𖽑𖽙 𖼟𖽡 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏 !
𖼺𖽱𖾏 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏 𖼺𖽱𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼷𖽹𖾏， 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽵𖾏 𖼦𖽘𖾏 𖽃𖽡𖾑.
