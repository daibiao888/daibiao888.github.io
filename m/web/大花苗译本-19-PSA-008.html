𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏 ． 𖼯𖽡𖽙𖾏 “𖼮𖽢𖾑 𖼊𖽑𖽳” 𖼢𖽜𖾏 𖽐𖼎𖽔𖾏 .
8
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼘𖽪！
𖼟𖽡 𖽐𖼷𖽷𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏!
 
𖼟𖽡 𖼟𖽙𖾏 𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼢𖽳 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏.
2 𖼟𖽡 𖼑𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽪𖾏 𖼊𖽔𖾐,
𖼎𖽑𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼎𖽙 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼟𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼼𖽷𖾏 𖼁𖽻𖾏 𖽃𖽡𖾑 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼗𖽪𖾏.
 
3 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏,
𖼊𖽑𖽡𖽝𖾐 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽜𖾐 𖼟𖽡𖾏 𖼅𖽔𖾏 𖼡𖽘𖾑𖽑 𖼅𖽔𖾏 𖼢𖽔𖾏.
4 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖼞𖽻𖾏 𖼮𖽳𖾐 𖼐𖽱𖾏!
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼐𖽱𖾏!
 
5 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ( “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐”) 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏,
𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼚𖽔𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑.
6-8 𖼟𖽡 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏,
𖼟𖽙𖾏 𖼯𖽢 𖼋𖽡𖾏 𖼟𖽻𖾏 𖼮𖽘𖾑, 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽨𖾏 𖼮𖽑𖽷𖾏、
𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏、𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏、𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖽠𖾑、
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖽐𖼚𖽙 𖼅𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼞𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼡𖽘𖾐 𖼊𖽠𖾏.
 
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼘𖽪！
𖼟𖽡 𖽐𖼷𖽷𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏!
