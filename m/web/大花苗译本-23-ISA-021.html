𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼦𖽡 𖼋𖽙
21
𖼦𖽔𖾑 𖼗𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼑𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐 :
 
𖼅𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏, 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽡 𖽐𖼮𖽑𖽹𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼑𖽡𖾏 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 , 𖼦𖽡 𖽐𖼀𖽱𖾏 𖼦𖽡𖾏 𖽐𖼀𖽠𖾏 𖼷𖽑𖽔𖾐 𖼮𖽑𖽡𖽻𖾏.
2 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐 𖼁𖽙𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 :
“𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐.
𖼽𖽡𖾑 𖼖𖽗 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏，
𖼄𖽔𖾐 𖼊𖽹𖾏 𖼘𖽪！ 𖼟𖽻𖾏 𖼙𖽻𖾐 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏，
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏 𖽐𖼮𖽦 𖼺𖽘𖾏 .”
3 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼄𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼊𖽔𖾑 𖼏𖽔𖾏 𖼮𖽡𖽝𖾏,
𖼺𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼾𖽨 𖼊𖽪𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼞𖽪𖾑 𖼄𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼸𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽻 ,
𖼞𖽪𖾑 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼸𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽔𖾏.
4 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼏𖽘𖾏 𖼡𖽑𖽙𖾐 𖼡𖽑𖽙𖾐, 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼞𖽪𖾑 ,
𖼦𖽡𖾏 𖼻𖽻𖾏 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼉𖽳， 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏.
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏,
𖼊𖽑𖽔𖾑 𖼮𖽡𖽝𖾏 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐 𖼟𖽪𖾏 𖼀𖽔𖾏 𖼡𖽘𖾐 𖼮𖽨𖾐, 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼎𖽺𖾏 𖼦𖽻𖾏.
𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼘𖽪！ 𖼅𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 ,
𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐.
6 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼟𖽡 𖼅𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 ,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
7 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 ,
𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖼋𖽔𖾏,
𖼎𖽺𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 , 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼘𖽙 𖼊𖽑𖽙𖾐,
𖼯𖽡𖾏 𖼽𖽔𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏.”
8 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼺𖽱𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏, 𖼐𖽑𖽡𖾏 :
“𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽪𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼟𖽡𖾏 𖼦𖽡𖾏 𖼊𖽪𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 ,
𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼙𖽱 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 .
9 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏
𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖼋𖽔𖾏.”
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼢𖽻 𖼙𖽠𖾏 𖼋𖽘𖾏！ 𖼢𖽻 𖼙𖽠𖾏 𖼋𖽘𖾏！
𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽵𖾑 𖼳𖽺 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼺𖽘𖾏.”
10 𖼞𖽪𖾑 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼟𖽪𖾏 𖼏𖽻 𖽐𖼊𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽔 𖼛𖽡𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏 𖼘𖽪！
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼎𖽑𖽻𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏，𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼋𖽘𖾏.
𖼦𖽔𖾑 𖼗𖽔𖾏𖼊𖽪 𖼄𖽔𖾐 𖼑𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐
11 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼊𖽪 𖼄𖽔𖾐 𖼑𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐：
𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼎𖽑𖽻𖾏 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼨𖽑𖽪𖾏 𖼑𖽪 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
“𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼘𖽪！ 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 ?
𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼘𖽪！𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 ?”
12 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼐𖽑𖽡𖾏: “𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏, 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏．
𖼅𖽡 𖼮𖽔𖾑 𖼑𖽪 𖼯𖽡𖾏 𖼑𖽪 𖼢𖽻 𖼋𖽘𖾏,
𖼯𖽪𖾏 𖼟𖽙𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼋𖽔𖾏 𖼑𖽪 .”
𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽔𖾑 𖼀𖽳
13 𖼦𖽔𖾑 𖼗𖽔𖾏 𖽃𖽔𖾑 𖼖𖽔𖾑 𖼀𖽳 𖼑𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐 :
𖼊𖽡𖾐 𖼊𖽗𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖼘𖽪,
𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽙𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼖𖽔𖾑 𖼀𖽳 𖼑𖽡𖾏 𖼡𖽔𖾏 𖼻𖽻𖾏 𖽐𖼚𖽙 .
14 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽑𖽡 𖼄𖽔𖾐 𖼘𖽪 !
𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖽃𖽻𖾐 𖼋𖽔𖾏, 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖽃𖽡𖾑, 𖼯𖽡𖽙𖾏 𖼳𖽡𖽝𖾐 𖼋𖽔𖾏 𖼸𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑.
15 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽘 𖽐𖼚𖽺𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼟𖽪𖾏 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼐𖽘𖾐,
𖼊𖽑𖽡𖽝𖾐 𖼈𖽙𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽜𖾑𖽑 𖽐𖼎𖽙 𖽐𖼮𖽷𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
