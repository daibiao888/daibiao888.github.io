𖽃𖽔𖾑 𖼄𖽙𖾑 𖼺𖽷𖾑 𖽐𖼊𖽠𖾑
1
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽃𖽔𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏, 𖼊𖽑𖽡 𖼞𖽙𖾑 𖼽𖽔𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼊𖽻𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 .
2 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏,
𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐,
𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼑𖽡𖾏 𖽐𖼎𖽘𖾏 𖽐𖼡𖽜𖾐 𖼽𖽔𖾑 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖼮𖽢𖾑 𖼄𖽡𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼽𖽔𖾑 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 .”
𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: “𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼯𖽪 𖼯𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼐𖽑𖽡𖽝𖾑 𖼞𖽡𖾏 𖼮𖽢𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼡𖽻𖾐 𖼟𖽡𖾏 𖽐𖼊𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 .
4 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 𖼑𖽡𖾏 𖼾𖽡𖾏,
𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼀𖽡𖽝 · 𖼦𖽔𖾐 𖼊𖽔 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑.
5 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼎𖽪𖾑 𖽃𖽔 𖼛𖽻𖾏,
𖼖𖽻𖾐 𖼺𖽘𖾏 𖼽𖽔𖾏 𖽂𖽺 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏
𖽐𖼎𖽙 𖼀𖽳 · 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.
𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼮𖽡 𖽃𖽳𖾐 .
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼮𖽢𖾑 𖼺𖽔𖾐 𖼯𖽪 𖼯𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼞𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 .
7 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 ,
𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑.
8 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼳𖽷 𖼊𖽑𖽪
𖽐𖼎𖽙 𖼽𖽔𖾏 𖼳𖽷 𖼮𖽡𖾑 𖼖𖽪𖽝 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼽𖽡𖾑 𖼞𖽳 𖼖𖽪𖽝 .
𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : “𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼯𖽪 𖼯𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼞𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏.
10 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 ,
𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑.”
𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼷𖽷𖾐 𖼚𖽻𖾐 𖼟𖽻𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼖𖽠𖾑 𖼾𖽨𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐,
𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼷𖽷𖾐 𖼅𖽔𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐.
12 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼊𖽑𖽡 𖼄𖽔𖾏 ,
𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 .”
𖼽𖽔𖾏 𖼄𖽞 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼗𖽡𖾏 𖼐𖽡𖾑 : “𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽪 𖼯𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽑𖽺𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑,
𖼑𖽜 𖼟𖽪𖾏 𖼎𖽳𖾏 𖼇𖽘𖾑 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐.
14 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏,
𖼐𖽡𖽙𖾐 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
