𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼞𖽻𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
13
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼖𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑? 𖼖𖽔𖾑 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼖𖽔𖾏?
𖼟𖽡 𖽐𖼀𖽙 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
2 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼮𖽔𖾑 𖼺𖽙𖾏, 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏, 𖼖𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼳𖽠𖾑 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼞𖽪𖾑, 𖼖𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
 
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑, 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼁𖽙𖾏 𖼞𖽡𖾑, 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼀𖽱𖾏 𖼸𖽙 𖼋𖽔 𖼙𖽠𖾏 .
4 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼳𖽠𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.”
𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼢𖽻 𖽃𖽡𖾑 𖼙𖽔 𖼚𖽙𖾏.
 
5 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼑𖽜 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼙𖽔 𖼚𖽙𖾏.
6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑.
