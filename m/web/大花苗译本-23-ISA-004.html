𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼳𖽤𖾏 𖼗𖽜𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽔𖾑 𖽃𖽡𖾐 𖼗𖽜𖾐 𖽃𖽔𖾑 𖼾𖽠 𖼮𖽡𖽝𖾏, 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼑𖽻𖾏 𖼀𖽡𖾐 𖼾𖽨𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏,𖼐𖽑𖽘𖾑 𖼀𖽡𖾐 𖼽𖽨𖾏 𖼮𖽑𖽙𖾏, 𖼈𖽡𖽝𖾏 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼯𖽪𖾏 𖼀𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,𖼟𖽙𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼀𖽡𖾐 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.”
2 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼘𖽑𖽘𖾑 𖼟𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼻𖽻𖾐 𖼷𖽷𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼙𖽠𖾏 𖼢𖽳 𖼋𖽠𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏. 3 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 , 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖽐𖼷𖽷𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑. 4 𖼮𖽵𖾑 𖼎𖽑𖽻𖾏 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽐𖼎𖽙 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑, 𖽐𖼷𖽔𖾑 𖼺𖽘𖾏 𖼳𖽡 𖼣𖽔𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏, 𖼎𖽺𖾏 𖽐𖼷𖽔𖾑 𖼨𖽑𖽪𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼚𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑. 5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽻𖾐 , 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼟𖽙𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼅𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼎𖽙 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏, 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼁𖽙𖾏 𖼮𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼀𖽡𖾑 𖼳𖽔𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑. 6 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼳𖽙𖾏, 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼚𖽺𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏, 𖼑𖽘 𖼮𖽢𖾏 𖼘𖽑𖽙𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
