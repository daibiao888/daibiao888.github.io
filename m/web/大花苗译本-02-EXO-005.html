5
𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 .” 2 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽻 𖼐𖽱𖾏 𖼗𖽪𖾏 , 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼖𖽺𖾑 ? 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 !” 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼢𖽳 𖼎𖽻𖾏 𖼀𖽡𖾐 , 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 , 𖼅𖽻𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼞𖽡𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼞𖽡𖾏、𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾐 .” 4 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 , 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼇𖽳𖾑 𖼑𖽪𖾏 𖼺𖽘𖾏 𖼖𖽺𖾑 ? 𖼅𖽡 𖼅𖽻𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼅𖽡 𖽐𖼎𖽘𖾏 !” 5 𖼇𖽔 𖼘𖽻𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼑𖽔𖾏 𖼘𖽪！ 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 . 𖼅𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽙𖾏 𖼮𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼼𖽙 !”
6 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑， 𖼇𖽔 𖼘𖽻𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼼𖽙 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼐𖽑𖽡𖾏 : 7 “𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼄𖽔𖾐 𖽐𖼡𖽜𖾐 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼑𖽡 𖼋𖽺𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖽐𖼡𖽜𖾐 . 8 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾏 𖼑𖽡 𖼋𖽺𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼯𖽠 , 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 , 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽠 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏 , 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼢𖽔𖾏 𖼐𖽑𖽡𖾏 : 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 9 𖼅𖽡 𖼄𖽔𖾐 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽑𖽤𖾑 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 , 𖼦𖽡𖾐 𖼑𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼮𖽑𖽵𖾑 .”
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼼𖽙 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼄𖽔𖾐 𖽐𖼡𖽜𖾐 𖼎𖽻𖾏 𖼅𖽡 . 11 𖼅𖽡 𖼾𖽨𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖽐𖼡𖽜𖾐 , 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 ! 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼼𖽙 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽠 .” 12 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 , 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖽐𖼊𖽻𖾏 𖼡𖽻𖾐 𖼗𖽙𖾏 𖽃𖽔𖾏 𖽐𖼡𖽜𖾐 . 13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼼𖽙 𖼯𖽪 𖼯𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼋𖽘𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼼𖽙 , 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼡𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.” 14 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽳𖾑 𖼼𖽙 𖽐𖼊𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼗𖽙𖾏 𖼨𖽑𖽳𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖽃𖽔 𖼑𖽘 𖼄𖽑𖽙𖾏 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼑𖽡 𖼋𖽺𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏 𖼅𖽡 𖼼𖽙 𖼖𖽺𖾑 ?
