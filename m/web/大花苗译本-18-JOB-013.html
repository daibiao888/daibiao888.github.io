𖼽𖽙 𖼀𖽳 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏
13
“𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼙𖽔 𖼁𖽙𖾏 ,
𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼙𖽔 𖼐𖽑𖽙𖾑 , 𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 .
2 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼀𖽻𖾐 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼀𖽻𖾐 𖼯𖽡𖽙 ,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼅𖽡 .
3 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼞𖽪𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼞𖽡𖾑 .
4 𖼅𖽡 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼞𖽑𖽙𖾏 𖽐𖼷𖽘𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 .
5 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐 ,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ！
6 𖼮𖽑𖽙 𖼅𖽡 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼐𖽑𖽡𖾏 ,
𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼗𖽪 .
7 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼖𖽔𖾏 ?
𖼑𖽜 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼖𖽔𖾏 ?
8 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼚𖽑𖽪𖾏 𖼘𖽪𖾏 ?
𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼗𖽪 𖼖𖽔𖾏 ?
9 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼋𖽠𖾏 𖼅𖽡 𖼗𖽙 , 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼘𖽪 ?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼅𖽡 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼮𖽑𖽵𖾑 𖼐𖽱𖾏 𖼘𖽪𖾏 ?
10 𖼅𖽡 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 ,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼅𖽡 .
11 𖼐𖽱𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼅𖽡 𖽐𖼮𖽑𖽹𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 𖼘𖽪 ?
𖼐𖽱𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼘𖽪 ?
12 𖼅𖽡 𖼑𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽙𖾏 𖼮𖽑𖽻𖾑 𖼑𖽡𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼗𖽪𖾏 ,
𖼅𖽡 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽪𖾏 𖼏𖽔𖾏 , 𖼟𖽪𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼞𖽙𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 .
 
13 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐 , 𖼯𖽔𖾏 𖼞𖽪𖾑 ！
𖼮𖽙𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼞𖽪𖾑 𖼞𖽑𖽔𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 .
14 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖽐𖼡𖽹𖾏 𖼮𖽘𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼐𖽑𖽡𖽝𖾑 ,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .
15 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽔𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 (𖼡𖽑𖽙𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼞𖽪𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼋𖽘𖾏”) ,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 .
16 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼊𖽠𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 .
17 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 ,
𖼟𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼅𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 .
18 𖼑𖽔𖾏 𖼘𖽪 ！𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 ,
𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
19 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 ,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼯𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼊𖽪𖾏 𖼀𖽘𖾏 𖼋𖽔 𖼙𖽠𖾏 .
20 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼼𖽘𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏 .
21 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼳𖽻𖾏 𖼺𖽷𖾐 𖼙𖽠𖾏 , 𖼡𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼞𖽪𖾑 .
22 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼟𖽡 𖽂𖽪𖾏 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼊𖽡𖾐 ,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼟𖽡 𖼎𖽺𖾏 𖼊𖽡𖾐 𖼞𖽪𖾑 .
23 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽐𖼎𖽙 𖼾𖽔𖾏 , 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽺𖾑 ?
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖽐𖼎𖽙 𖼾𖽔𖾏 .
24 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼖𖽺𖾑 ?
25 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖽃𖽔 𖽐𖼚𖽻𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖽃𖽔𖾏 𖼼𖽙 𖽃𖽡𖾑 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼖𖽔𖾏 ?
𖼽𖽔𖾑 𖼖𖽠𖾑 𖽃𖽔𖾏 𖽐𖼊𖽻𖾏 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏 𖽃𖽡𖾑 𖼐𖽔𖾑 𖼖𖽔𖾏 ?
26 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼨𖽑𖽢𖾐 𖼞𖽪𖾑 ,
