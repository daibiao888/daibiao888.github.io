𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽻𖾐 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼮𖽨𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏
10
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐?
𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼚𖽺𖾏 𖽐𖼎𖽙𖾏?
2 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼞𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼖𖽠𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼊𖽻𖾏 𖽃𖽡𖾑.
 
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼉𖽳 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼎𖽺𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼐𖽱𖾏 ( 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “ 𖼐𖽱𖾏 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼉𖽳 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐, 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫”).
4 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼯𖽡𖽙𖾏 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖼦𖽡𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽐𖼚𖽙 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
 
5 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼎𖽳𖾏 𖼎𖽪𖾑,
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖽐𖼚𖽜𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼁𖽙𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
6 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙,
𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽑𖽳𖾐 𖼷𖽑𖽷𖾏 𖼦𖽡𖾐 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔.
 
7 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼀𖽪𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏、𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑、𖼸𖽺𖾏 𖼯𖽵𖾏,
𖼊𖽪𖾑 𖼡𖽘𖾐 𖽃𖽔𖾏 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼋𖽻𖾏 𖼚𖽠𖾏、 𖽐𖼚𖽪𖾏 𖼸𖽳.
8 𖼐𖽱𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼼𖽙𖾏 𖼋𖽙𖾏,
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼻𖽡𖽝𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽡𖽻𖾏.
9 𖼐𖽱𖾏 𖽐𖼚𖽺𖾏 𖽐𖼎𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏, 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼼𖽻𖾏 𖼯𖽠𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑,
𖼐𖽱𖾏 𖽐𖼚𖽺𖾏 𖼮𖽡𖽝𖾏, 𖼖𖽔𖾑 𖼦𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼙𖽠𖾏,
𖼐𖽱𖾏 𖼮𖽤𖾐 𖼸𖽙𖾏, 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼦𖽔𖾏 𖼙𖽠𖾏.
 
10 𖼐𖽱𖾏 𖼸𖽺𖾏 𖽐𖼊𖽔𖾏 𖽐𖼚𖽺𖾏 𖽐𖼎𖽙𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼻𖽡𖽝𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽡𖽻𖾏 𖼎𖽺𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾑 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
11 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏: 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼐𖽱𖾏 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏.
 
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪， 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏！ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪，
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽘𖾏 𖼋𖽡𖾏！ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐.
13 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑？
 
14 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼙𖽔 𖼑𖽔𖾏 𖼋𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼟𖽡 𖼞𖽑𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐．
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼻𖽡𖽝𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽡𖽻𖾏 𖼄𖽔𖾐 𖼾𖽨𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡,
𖼟𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼀𖽘𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏.
 
15 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼊𖽻𖾏 𖼚𖽑𖽘𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏.
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏,
𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽪𖾏 𖼮𖽘𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏.
 
17 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼐𖽱𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏, 𖼟𖽡 𖽐𖼷𖽙𖾑 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 ( 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼀𖽻𖾐 𖼋𖽘”),
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，
18 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽐𖼀𖽘𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
