𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐
21
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑, 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼑𖽜 𖼺𖽔𖾐 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑. 2 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏，𖼺𖽔𖾐 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏, 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏, 𖼎𖽺𖾏 𖼑𖽜 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐. 3 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼊𖽡𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼳𖽻𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐. 4 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼾𖽡𖾏 𖼐𖽑𖽪𖾐, 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼸𖽙𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏. 5 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏. 6 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼐𖽑𖽡𖾏: “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼚𖽙𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽙𖾑𖽑 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼙𖽔 𖼚𖽙𖾏.” 7 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼺𖽪𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼽𖽔𖾑 𖼘𖽘𖾑𖽑 𖼾𖽨 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼘𖽪? 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼳𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐.”
8 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼘𖽑𖽙𖾐 𖼷𖽷𖾏 𖼖𖽘𖾑, 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼄𖽡𖾏 𖼺𖽘𖾏. 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼇𖽳𖾑 𖼄𖽡𖾏, 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏.
𖼳𖽢𖾏 𖼮𖽢 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼊𖽻𖾏 𖼖𖽠𖾑
9 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽢𖾏 𖼮𖽢 𖼑𖽜 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼳𖽻𖾏 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐. 10 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼄𖽔𖾐 𖼖𖽜𖾑 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼐𖽡𖾑 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽜𖾑 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼷𖽹𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏.” 11 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽜 𖼐𖽱𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 𖽃𖽺𖾐 𖼊𖽳𖾏. 12 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼊𖽳𖾏． 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏, 𖼟𖽡 𖼖𖽔𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼑𖽻 𖼸𖽙𖾏； 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏. 13 𖼐𖽑𖽡𖾏 𖼸𖽙 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐, 𖼞𖽪𖾑 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼷𖽑𖽙𖾏 𖼊𖽡𖽝𖾑 𖽃𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼟𖽡 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼯𖽡𖽙.”
14 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼞𖽑𖽡𖽠𖾏 𖼳𖽡𖽝𖾐 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼘𖽑𖽱𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽻𖾐, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼳𖽢𖾏 𖼮𖽢, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏, 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖼺𖽘𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼙𖽠𖾏. 𖼳𖽢𖾏 𖼮𖽢 𖼅𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼐𖽡𖽙𖾐 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏. 15 𖼖𖽪𖾑 𖼘𖽑𖽱𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽻𖾐 𖼦𖽻𖾏 𖼋𖽘𖾏 𖼋𖽘𖾏, 𖼳𖽢𖾏 𖼮𖽢 𖼇𖽳𖾑 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽡𖽝𖾏， 16 𖼾𖽨𖾏 𖼎𖽺𖾏 𖼙𖽠𖾏 𖼚𖽡𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼀𖽙𖾑 𖽃𖽡𖾐 𖼐𖽜𖾑𖽑 𖼈𖽙𖾏, 𖼊𖽱𖾐 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼼𖽻𖾏, 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔 .” 𖼎𖽺𖾏 𖼊𖽱𖾐 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼼𖽻𖾏, 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑 𖼋𖽘𖾏. 17 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼋𖽘𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾏 𖼳𖽢𖾏 𖼮𖽢 𖼐𖽑𖽡𖾏: “𖼳𖽢𖾏 𖼮𖽢, 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼘𖽪? 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼋𖽘𖾏. 18 𖼳𖽠𖾑 𖼗𖽙𖾏！ 𖼅𖽻𖾏 𖼁𖽔 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼊𖽡𖽝𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.” 19 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼳𖽢𖾏 𖼮𖽢 𖽃𖽔𖾑 𖼅𖽔 𖼁𖽙𖾏 𖼞𖽡𖾑, 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐, 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼖𖽪𖾑 𖼘𖽑𖽱𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽐𖼊𖽡𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼀𖽪𖾑, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼦𖽻𖾏.
20 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼐𖽡𖽙𖾐, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼘𖽑𖽙𖾐 𖼷𖽷𖾏 𖼖𖽘𖾑, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏, 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼀𖽙𖾑 𖼐𖽜𖾑𖽑. 21 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼀𖽔𖾑 𖼖𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼷𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐.
𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏
22 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼇𖽺𖾑 𖼞𖽙 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐. 23 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼏𖽵 𖼗𖽪𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏． 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼯𖽡𖽙𖾏 𖼟𖽡, 𖼟𖽡 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑.” 24 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼯𖽪𖾏 𖼏𖽵 𖼗𖽪𖾏.”
25 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼦𖽔𖾏 𖼳𖽪𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼙𖽠𖾏, 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳. 26 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼐𖽑𖽡𖾏: “𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐, 𖼟𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼋𖽡𖽝𖾏 𖼞𖽪𖾑 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏.” 27 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼄𖽔𖾐 𖼮𖽑𖽷𖾏 𖽐𖼎𖽙 𖼑𖽨𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳, 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏. 28 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼞𖽑𖽡𖽠𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼮𖽡𖽺𖾏 𖼾𖽨 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏. 29 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼑𖽪 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼐𖽡𖾑 𖼮𖽡𖽺𖾏 𖼾𖽨 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐, 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼐𖽔𖾑?” 30 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼐𖽡𖾑, 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽠𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖾑 .” 31 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼊𖽡𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑 ( “𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑”𖼟𖽪𖾏 “𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼏𖽵 𖼗𖽪𖾏 ” ), 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼏𖽵 𖼗𖽪𖾏 . 32 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼇𖽺𖾑 𖼞𖽙 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽠𖾏. 33 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽡𖽙𖾐 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑 𖼯𖽡𖽙 𖽃𖽡𖾐 𖼇𖽘𖾐 𖼙𖽱𖾑 𖼙𖽱 , 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 — 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏. 34 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽡𖽙𖾐 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏.
