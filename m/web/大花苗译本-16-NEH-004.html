𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼷𖽑𖽔𖾑 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼀𖽡𖾐 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻, 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼚𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. 2 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼊𖽑𖽡𖽝𖾐𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼏𖽙 𖼐𖽑𖽡𖾏 : “𖼸𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽻 𖼼𖽙 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑? 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼖𖽔𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼖𖽔𖾑 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼊𖽔𖾐 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼖𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼙𖽔 𖼷𖽑𖽷𖾑 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼖𖽔𖾏 𖼯𖽤𖾏?” 3 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽙𖾑 𖼀𖽡𖾐 𖼽𖽔𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼐𖽑𖽡𖾏 : 𖽃𖽔𖾑 𖼋𖽜𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽢𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾑, 𖼞𖽡𖾑 𖼋𖽱𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼊𖽜𖾑 𖼖𖽔𖾏 𖼖𖽔𖾑 𖼯𖽪𖾏 𖼢𖽻 𖼙𖽠𖾏. 4 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪! 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽻 𖼊𖽡𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽃𖽡𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏. 5 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼇𖽳𖾏 𖼺𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐. 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼀𖽡𖾐 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻, 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼦𖽡𖾏 𖼷𖽔𖾏 𖼮𖽡𖽝𖾏 , 𖼺𖽡𖽝𖾐 𖼸𖽙 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏.
𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏
7 𖼷𖽑𖽔𖾑 𖼀𖽔𖾑 𖼖𖽔𖾑、 𖼊𖽙𖾑 𖼀𖽡𖾐 𖼽𖽔𖾐、 𖽃𖽔𖾑 𖼖𖽔𖾑 𖼀𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐、 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐、 𖼽𖽔𖾏 𖼳𖽷 𖼊𖽑𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼀𖽡𖾐 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 , 𖼁𖽱𖾏 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝. 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽡𖾐 𖼸𖽻𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼖𖽔𖾑 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼟𖽙𖾏 𖽐𖼚𖽙 𖼙𖽻 𖼳𖽠𖾑 𖽐𖼷𖽑𖽙𖾑. 9 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽔𖾑, 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑, 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼯𖽷𖾏 𖼻𖽙𖾑.
10 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 :
“𖽃𖽔 𖼗𖽔𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽱𖾑 𖽃𖽡𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼯𖽻 𖼼𖽙 𖼋𖽘𖾏.

11 𖼀𖽡𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼷𖽪𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐, 𖼦𖽡𖾐 𖼁𖽙𖾏, 𖼀𖽡𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼟𖽙𖾏 𖼼𖽙 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏.” 12 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽡𖾑, 𖼟𖽻𖾏 𖼟𖽻𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑.” 13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 𖽃𖽡𖾑, 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏、𖼯𖽡𖽙𖾏 𖼄𖽑𖽪𖾑、𖼯𖽡𖽙𖾏 𖼐𖽜𖾑𖽑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼡𖽑𖽘𖾏 𖽃𖽡𖾑. 14 𖼞𖽪𖾑 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼺𖽘𖾏, 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏、 𖼉𖽻 𖼷𖽻𖾏 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖽃𖽡𖾑, 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏、 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏、 𖼐𖽤𖾐、𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏.”
