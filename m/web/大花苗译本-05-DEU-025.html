𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼛𖽠𖾏 𖼑𖽪𖾐
25
“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 , 𖼋𖽔𖾏 𖼑𖽻 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 , 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼽𖽔𖾑 𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼅𖽔𖾏 𖼞𖽡𖾑 , 𖼊𖽡𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 . 2 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼮𖽔𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼏𖽻 𖽐𖼊𖽻𖾏 , 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽔𖾏 𖼏𖽻 𖽐𖼊𖽻𖾏 . 3 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼮𖽔 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖽐𖼚𖽠𖾏 𖼊𖽔𖾐 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼊𖽻𖾏 , 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼊𖽻𖾏, 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼋𖽘𖾏 .
4 𖼑𖽨𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏 𖼯𖽪𖾏 𖼡𖽻𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽱𖾐 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 .
𖼑𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
5 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐 , 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽔𖾑 𖼋𖽔 𖼙𖽠𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼺𖽷𖾐 , 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽔 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼡𖽔𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼸𖽙𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼙𖽻 𖼋𖽪𖾏 , 𖼷𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏 . 6 𖽃𖽔 𖼁𖽙𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽔 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼇𖽳𖾏 𖼺𖽘𖾏． 7 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼷𖽙𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼑𖽡𖾏 𖼐𖽤𖾐 , 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼽𖽔𖾑 𖼚𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼊𖽪𖾐 𖼗𖽻 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖼈𖽻𖾑 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼷𖽷𖾏 , 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼙𖽻 𖼋𖽪𖾏 . 8 𖽐𖼚𖽙 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼗𖽻 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼑𖽪 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖼮𖽔𖾑 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽙𖾏 𖼐𖽱𖾏， 9 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼑𖽡𖾏 𖼐𖽤𖾐 , 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽪𖾐 𖼗𖽻 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼊𖽠𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 , 𖼚𖽠𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼞𖽑𖽻𖾏 𖼺𖽘𖾏 , 𖽐𖼊𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 , 𖼐𖽑𖽡𖾏 : 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 . 10 𖼐𖽡𖽙𖾐𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “ 𖼾𖽡𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼞𖽑𖽻𖾏 ” .
11 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 , 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼋𖽔𖾏 , 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼊𖽔𖾑 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼉𖽜𖾏， 12 𖼽𖽔𖾑 𖼮𖽷𖾐 𖼊𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼐𖽱𖾏 .
