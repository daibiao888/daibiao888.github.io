5
𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐．
2 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏,
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼀𖽻𖾐 𖼟𖽳 .
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽱𖾏 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏,
𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽡𖽝 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼛𖽙𖾏，
4 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖽃𖽦𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼄𖽡𖾏 𖼇𖽪𖾑,
𖽐𖼮𖽳𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡𖽝𖾐 𖽐𖼮𖽳𖾏 .
5 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽔 ,
𖼐𖽱𖾏 𖼷𖽑𖽔𖾏 𖼏𖽔𖾏 𖼞𖽡𖾑 𖼷𖽠𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏，
6 𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼖𖽡 𖼟𖽻𖾏 𖼚𖽑𖽡𖾏 𖼦𖽡𖾐 𖼏𖽔𖾏, 𖼾𖽨𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐.
 
7 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼘𖽪！ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏．
8 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼚𖽡𖾐,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏；
9 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼄𖽔𖾐 𖼟𖽡 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼊𖽔𖾐 𖼋𖽡𖽝 ；
10 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼚𖽱𖾏 𖽐𖼞𖽘𖾏 𖼖𖽑𖽘𖾐 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽔𖾏.
11 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽘𖾏, 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼖𖽻𖾏 𖼡𖽑𖽔𖾑,
𖼟𖽡 𖼎𖽺𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
12 𖼐𖽑𖽡𖾏 : “ 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼳𖽪𖾐 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 ,
𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏，
13 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 .
14 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡 𖼸𖽱 ,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.”
15 𖼟𖽡 𖼽𖽔𖾑 𖼦𖽻𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼑𖽡𖾏 𖽃𖽻𖾐,
𖼦𖽻𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼷𖽑𖽠𖾏.
16 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼷𖽑𖽠𖾏 𖼋𖽡𖽝𖾏 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼙𖽠𖾏？
𖼟𖽡 𖼸𖽙𖾏 𖼟𖽱𖾏 𖽃𖽻𖾐 𖼋𖽡𖽝𖾏 𖼯𖽪𖾏 𖼟𖽙𖾏 𖽐𖼚𖽱𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏？
17 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼺𖽙𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼯𖽡𖽙𖾏.
18 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼷𖽑𖽠𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼽𖽔𖾑 𖽃𖽺𖾐 𖼙𖽔 𖼟𖽡 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼷𖽙𖾏.
19 𖼐𖽱𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼷𖽹𖾏 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏,
𖽃𖽔𖾑 𖼑𖽡𖽝 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 .
20 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏？
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼁𖽔 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐？
 
21 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽙𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏,
𖼐𖽱𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼸𖽙𖾏 𖼞𖽡𖾑 .
22 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼄𖽔𖾐 𖼊𖽻𖾏,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼺𖽱𖾏 𖼖𖽑𖽔𖾏 𖽐𖼊𖽠𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
23 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼎𖽺𖾏 𖼛𖽙𖾏 𖼋𖽔 ,
𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼡𖽔 𖼏𖽔 𖼐𖽔𖾑 𖼅𖽻𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼞𖽡𖾑 𖼺𖽘𖾏.
