𖼳𖽔𖾐、𖼦𖽗、𖼽𖽔𖾐 𖼇𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 1：5 - 23）
10
𖼐𖽙 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽔𖾐、𖼦𖽗、𖼽𖽔𖾐 𖼇𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐. 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼊𖽪𖾐.
2 𖼽𖽔𖾐 𖼇𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼞𖽙𖾑 𖼄𖽡𖽝、𖼄𖽔𖾐 𖼞𖽙、𖼄𖽔𖾐 𖼊𖽹𖾏、𖼽𖽔𖾐 𖽂𖽔、𖼊𖽑𖽪𖾐 𖼀𖽔𖾑、𖼄𖽡𖾐 𖼳𖽳、𖼊𖽑𖽡 𖼖𖽔𖾑. 3 𖼞𖽙𖾑 𖼄𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽷 𖼮𖽡𖾑 𖼐𖽔、 𖼖𖽡𖾏 𖼇𖽔、𖼊𖽑𖽙 𖼮𖽢𖾑 𖼄𖽔𖾐. 4 𖼽𖽔𖾐 𖽂𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑、𖼊𖽑𖽔𖾑 𖼳𖽷𖾑、𖼮𖽡𖾑 𖼊𖽑𖽡、𖼊𖽙𖾑 𖼊𖽗𖾑 . 5 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏、𖼳𖽡𖽠𖾑 𖼎𖽻𖾐, 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏、𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐、𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.
6 𖼦𖽗 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼞𖽪𖾐 𖼳𖽷、𖼣𖽹 𖼮𖽡、𖼇𖽪、𖼮𖽢𖾑 𖼐𖽔. 7 𖼞𖽪𖾐 𖼳𖽷 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽔𖾑、𖼦𖽔𖾐 𖼇𖽺𖾑 𖼖𖽔𖾑、𖼺𖽔𖾐 𖼇𖽪 𖼊𖽑𖽔𖾑、𖼖𖽔𖾑 𖼄𖽔𖾐、𖼺𖽔𖾐 𖼇𖽪 𖼊𖽑𖽡 𖼮𖽢𖾑. 𖼖𖽔𖾑 𖼄𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑、𖼊𖽡𖾐 𖼊𖽗𖾏. 8 𖼞𖽪𖾐 𖼳𖽷 𖼅𖽔𖾏 𖼐𖽡 𖼖𖽪, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼊𖽪𖾐 𖼋𖽻. 9 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼡𖽹𖾏. 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼺𖽱𖾏 𖼐𖽡 𖼖𖽪 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼡𖽹𖾏.” 10 𖼐𖽱𖾏 𖼞𖽪𖽙 𖼳𖽠𖾑 𖼁𖽱𖾏 𖼟𖽪𖾏 𖼀𖽔𖾑 𖼀𖽡𖽝、𖼽𖽡𖾑 𖼖𖽡、𖼽𖽔𖾏 𖼮𖽢、𖼮𖽢 𖼐𖽡, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽷𖾏 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 11 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼳𖽪, 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏、𖼖𖽡𖾏 𖼦𖽙 𖼀𖽳、𖼮𖽢𖾑 𖼖𖽔𖾑, 12 𖽐𖼎𖽙 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏、𖼮𖽢𖾑 𖼖𖽔𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼖𖽡𖾏 𖼳𖽡𖽝𖾑, 𖼖𖽪𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
13 𖼣𖽹 𖼮𖽡 𖼅𖽔𖾏 𖼖𖽪𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽔𖾏 𖼐𖽔 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼖𖽡𖾏 𖼦𖽔𖾐 𖼀𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼐𖽔 𖼇𖽪 𖼊𖽑𖽪𖾐 𖼳𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐、 14 𖼀𖽑𖽔 𖼺𖽷𖾑 𖼖𖽪𖾐 𖼳𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼮𖽢𖾑 𖼺𖽷𖾑 𖼖𖽪𖾏 𖼳𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼮𖽢𖾑 𖼇𖽺𖾑 𖼊𖽑𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐； 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼺𖽷𖾑 𖼖𖽪𖾏 𖼳𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
15 𖼮𖽢𖾑 𖼐𖽔 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼦𖽳, 16 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼞𖽳 𖼮𖽢𖾑 𖼺𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、 17 𖼳𖽡𖾑 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽔𖾏 𖼮𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼳𖽡𖾑 𖼐𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐、 18 𖼽𖽔𖾏 𖽂𖽔𖾐 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼳𖽡𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼦𖽔𖾐 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼮𖽢𖾑 𖼐𖽔 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼞𖽑𖽔𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 19 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽔𖾑 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼮𖽡𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼺𖽔𖾐, 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐、𖽃𖽳 𖼄𖽙 𖼖𖽔𖾑、𖼽𖽔 𖼄𖽔𖾐、𖼳𖽡𖾐 𖼀𖽡𖽝 𖼸𖽙𖾏 𖼞𖽡𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽔𖾑 𖼳𖽔𖾑. 20 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽗 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏、𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏, 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖼞𖽪𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.
