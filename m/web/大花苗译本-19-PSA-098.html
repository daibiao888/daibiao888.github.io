98
𖼅𖽡 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼮𖽑𖽦𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑,
𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏.
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏,
𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 .
3 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏,
𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏 𖼞𖽑𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏.
 
4 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼽𖽔𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑, 𖼙𖽔 𖼛𖽙𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏！
5 𖼽𖽔𖾑 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼎𖽑𖽻𖾏 𖼢𖽜𖾏 𖽐𖼎𖽙 𖼳𖽘𖾐 𖽐𖼞𖽻𖾐 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 .
6 𖼯𖽡𖽙𖾏 𖼞𖽡𖾑 𖼊𖽳𖾏 𖽐𖼎𖽙 𖼞𖽪𖾐 𖼳𖽘𖾐,
𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏！
 
7 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼙𖽔 𖼚𖽙𖾏 𖽐𖼷𖽑𖽷𖾏 𖼳𖽱𖾑,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽪𖾏 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐.
8 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖽐𖼀𖽔𖾏 𖼋𖽡𖾏 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽡𖾐 𖼸𖽻𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏.
9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼸𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏.
