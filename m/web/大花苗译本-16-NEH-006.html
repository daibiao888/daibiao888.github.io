𖼷𖽑𖽔𖾑 𖼀𖽔𖾑 𖼖𖽔𖾑、 𖼊𖽙𖾑 𖼀𖽡𖾐 𖼽𖽔𖾐、 𖽃𖽔𖾑 𖼖𖽔𖾑 𖼀𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖾑 𖼳𖽔𖾏, 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼞𖽪𖾑 𖼨𖽑𖽢𖾏 𖼊𖽡𖽝𖾑 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 , 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 (𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼊𖽡𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼺𖽷𖾐 ). 2 𖼷𖽑𖽔𖾑 𖼀𖽔𖾑 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼳𖽔𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑, 𖼐𖽑𖽡𖾏 : “𖼷𖽑𖽜𖾑 𖼟𖽡 𖼋𖽔𖾏, 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼐𖽙 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼼𖽙𖾏 𖼁𖽙𖾏 𖼚𖽑𖽪𖾏.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼦𖽺𖾏 𖼞𖽪𖾑. 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ,𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼦𖽡𖾐 𖽐𖼮𖽻 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏, 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼳𖽙𖾏 𖼼𖽙 𖼮𖽡𖽝𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼅𖽡 𖼖𖽺𖾑?” 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼚𖽻𖾐 𖼟𖽻𖾐 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏. 5 𖼷𖽑𖽔𖾑 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽜𖾐 𖼟𖽻𖾐 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑, 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼄𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼘𖽙𖾏 𖼮𖽡𖽝𖾏. 6 𖽐𖼚𖽙 𖼄𖽻𖾏 𖼺𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏, 𖼮𖽢𖾑 𖼳𖽷𖾑 𖼄𖽙 (𖼐𖽡𖽙𖾐 2：19 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼮𖽡𖾑 𖼳𖽔𖾏” ) 𖼙𖽔 𖼐𖽑𖽡𖾏, 𖼟𖽡 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖽐𖼷𖽜𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼷𖽑𖽙𖾏 𖼷𖽱𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻, 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 7 𖼟𖽡 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽑𖽡𖾏 : “𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼋𖽔𖾏, 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼦𖽡 𖽐𖼷𖽷 𖼦𖽡𖾏 𖼊𖽳𖾏.” 8 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼐𖽡𖾑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐, 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼊𖽔𖾐.” 9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽐𖼮𖽑𖽹𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼯𖽻 𖼼𖽙 , 𖼸𖽙 𖼟𖽪𖾏 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼦𖽡𖾐 𖼮𖽑𖽜.” 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪! 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
10 𖼞𖽪𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽡𖾐 𖼳𖽡𖾑 𖼊𖽔𖾏 𖼀𖽡𖽝 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖼊𖽡𖾏 𖼖𖽹 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐𖼳𖽷𖾏 𖼄𖽔𖾐 𖼽𖽔𖾐 𖽐𖼞𖽔𖾏, 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼡𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼻𖽻𖾑, 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼁𖽙𖾏 𖼚𖽑𖽪𖾏, 𖼡𖽙𖾏 𖼎𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾑 𖼋𖽔𖾏 𖼐𖽑𖽙𖾏 𖼟𖽡 , 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏 𖼐𖽑𖽙𖾏 𖼟𖽡.” 11 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 : “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼏𖽘 𖼀𖽜𖾑 𖼖𖽺𖾑? 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼖𖽺𖾑? 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼁𖽺𖾏 𖼅𖽻𖾏.” 12 𖼞𖽪𖾑 𖼑𖽔𖾏 𖽐𖼎𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼷𖽷𖾐 𖼐𖽱𖾏, 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼊𖽱𖾑 𖼞𖽪𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏𖼊𖽙𖾑 𖼀𖽡𖾐 𖼽𖽔𖾐 𖽐𖼎𖽙 𖼷𖽑𖽔𖾑 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏. 13 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼐𖽱𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽹𖾏, 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽻𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑. 14 “𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪! 𖼊𖽙𖾑 𖼀𖽡𖾐 𖼽𖽔𖾐、 𖼷𖽑𖽔𖾑 𖼀𖽔𖾑 𖼖𖽔𖾑、𖽃𖽔 𖼁𖽙𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽙 𖼽𖽔𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐, 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽹𖾏, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼮𖽡𖽙𖾏 𖼮𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽡𖾑.”
