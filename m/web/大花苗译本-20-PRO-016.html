16
𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼸𖽠𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
2 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼅𖽔𖾏 𖼑𖽔𖾏 𖼺𖽱𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏；
𖼮𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼮𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
3 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽐𖼊𖽱𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑.
4 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼷𖽷𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼖𖽔𖾏 𖼑𖽜 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼷𖽷𖾏 𖼯𖽡𖽙𖾏.
5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐；
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼀𖽜𖾑 𖼦𖽡𖾐 𖼛𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐.
6 𖼑𖽜 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼎𖽙𖾑 ；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼀𖽜𖾑 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐.
7 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼙𖽔 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏.
8 𖼅𖽔𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼯𖽠 𖼅𖽔𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼷𖽑𖽔𖾏 𖼏𖽔𖾏 𖼞𖽡𖾑 .
10 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐,
𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏.
11 𖼊𖽪𖾑 𖼞𖽡𖾏 𖽐𖼎𖽙 𖼸𖽱𖾏 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼁𖽡𖽝 ,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼯𖽡𖽙𖾏.
12 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼳𖽪𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖽙 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼗𖽙𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏.
13 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 ；
𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏.
14 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏.
15 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼮𖽑𖽦𖾐 𖼟𖽡𖾏 𖼚𖽙𖾏 𖽃𖽔𖾏 𖽐𖼷𖽑𖽷𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑；
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼺𖽱𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼑𖽘 .
16 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼊𖽻𖾏 𖼞𖽪𖾐, 𖼺𖽹𖾑 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼺𖽹𖾑 𖼑𖽡𖽝𖾏.
17 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑, 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑.
18 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏；
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽻𖾏 𖼢𖽻 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏.
19 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 ,
𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼇𖽺𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 .
20 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏；
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
21 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐；
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 , 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖽐𖼞𖽡𖽝𖾏.
22 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 .
23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖽐𖼞𖽡𖽝𖾏.
24 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽱𖾏 𖼄𖽪𖾑 𖽐𖼞𖽔𖾏,
𖼟𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑, 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏.
25 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼸𖽙𖾐 𖼞𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽙 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔 .
26 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼟𖽪𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼋𖽠𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖽃𖽔𖾏 𖼊𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏.
27 𖼡𖽔𖾑 𖼯𖽱𖾏 𖼸𖽜 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼐𖽱𖾏 𖼗𖽪𖾏 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾐.
 
28 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼙𖽳𖾐 𖼷𖽱𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼺𖽹𖾑 𖼗𖽪𖾏 𖼟𖽙𖾏 𖼻𖽻𖾏 𖼽𖽺 𖼗𖽔 𖼷𖽑𖽡𖽝𖾏 𖼇𖽳𖾑 𖼷𖽑𖽡𖽝𖾏,
29 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼮𖽑𖽵𖾑 𖼚𖽡𖾐 𖼼𖽷𖾏 𖼼𖽙 ,
𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏.
30 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽜𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼮𖽡𖽝𖾏, 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼖𖽡𖾐 𖼉𖽳𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏, 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏.
31 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼷𖽑𖽡𖽝𖾏 𖼗𖽡𖽝𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼑𖽡𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑,
𖼐𖽡𖽙𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏.
32 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾐 𖼋𖽻 ；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼙𖽻 .
