𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏
6
“ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽪𖾑 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽺𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 ， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 ， 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼅𖽡 𖼊𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏． 2 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ，𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱， 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏， 𖼽𖽔𖾑 𖼷𖽪𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽻𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏． 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏． 3 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼇𖽳𖾏 𖼀𖽻𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽔𖾑 𖼳𖽷𖾏， 4 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼟𖽡 𖼉𖽺𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖼳𖽹𖾐 𖼑𖽔𖾏 ，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡．”
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
（𖼖𖽪𖾏 11：2-4）
5 “ 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 𖼳𖽠𖾑 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼎𖽙 𖼞𖽡𖾑 𖼳𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏， 𖼽𖽔𖾑 𖼷𖽪𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏． 6 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏， 𖼎𖽺𖾏 𖼡𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼮𖽡𖽝𖾏，𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 ， 𖼟𖽡 𖼉𖽺𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡． 7 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽪 𖼯𖽪𖾏 𖼎𖽙𖾑 𖼐𖽑𖽡𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼑𖽻 𖼊𖽡𖾐． 8 𖼮𖽔𖾑 𖼖𖽡𖾏 𖼯𖽡𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼺𖽷𖾐， 𖼅𖽡 𖼉𖽺𖾏 𖼀𖽻𖾐 𖼅𖽡 𖼽𖽔𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏．
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：
𖼀𖽡𖾐 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 ，
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑．
10 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼞𖽪𖽙 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏，
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐，
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏．
11 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽻𖾏， 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐，
12 𖼯𖽔𖾏 𖼀𖽡𖾐 𖼾𖽔𖾏，
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼙𖽔 𖼯𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼀𖽡𖾐．
13 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽐𖼮𖽷𖾐 𖼏𖽻 （“𖽐𖼮𖽷𖾐 𖼏𖽻 ”𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼁𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 ”） 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑，
𖼮𖽨𖾏 𖼀𖽡𖾐 𖼚𖽠𖾏 𖼛𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾐．
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙、𖼛𖽘𖾏 𖼼𖽙𖾏、𖽐𖼚𖽻 𖼮𖽡𖾏，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼟𖽡 𖼁𖽡𖽝，
𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏．𖽃𖽔𖾏 𖼄𖽞！
𖼯𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖽠𖾏 𖼾𖽔𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑
14 “ 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼯𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖽠𖾏 𖼾𖽔𖾏， 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽔𖾏 𖼅𖽡 𖼮𖽑𖽡𖽠𖾏 𖼾𖽔𖾏． 15 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼯𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖽠𖾏 𖼾𖽔𖾏， 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼯𖽔𖾏 𖼅𖽡 𖼮𖽑𖽡𖽠𖾏 𖼾𖽔𖾏 𖼯𖽡𖽙𖾏． 16 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽪𖾏 𖼡𖽜𖾑， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼯𖽡𖽙𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽳𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼮𖽷𖾑 𖼑𖽔𖾏， 𖼽𖽔𖾑 𖼷𖽪𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑． 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏． 17 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽪𖾏 𖼡𖽜𖾑， 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 、𖽐𖼷𖽔𖾑 𖼚𖽑𖽪𖾏， 18 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽡 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑， 𖼯𖽔 𖼽𖽔𖾑 𖼟𖽡 𖼉𖽺𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 ， 𖼟𖽡 𖼉𖽺𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡．”

24 “ 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽵𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼷𖽷𖾑 𖼐𖽡𖾑 𖽐𖼮𖽑𖽠𖾏 𖼷𖽷𖾑 𖼉𖽺𖾏， 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼷𖽷𖾑 𖼉𖽺𖾏． 𖼅𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖼎𖽺𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼊𖽑𖽡𖽝𖾐． 25 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼅𖽡 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼊𖽳𖾏，𖼽𖽔𖾑 𖼑𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏，𖼽𖽔𖾑 𖼦𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼽𖽔𖾑 𖼐𖽑𖽘𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏．𖼅𖽡 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼘𖽪？ 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼘𖽪？ 26 𖼅𖽡 𖼑𖽔𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼡𖽻𖾐，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼡𖽻𖾐 𖼡𖽜𖾏 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼼𖽪 𖼮𖽡𖽝𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼅𖽡 𖼦𖽡𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼘𖽪 ？ 27 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼮𖽡𖽻𖾐 𖼐𖽡𖽙𖾐 𖼗𖽡𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼘𖽪？ 28 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼅𖽡 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏 𖼖𖽺𖾑 ？ 𖼟𖽡 𖽐𖼊𖽱𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼘𖽑𖽘𖾑 𖼘𖽑𖽙𖾐，𖼋𖽡𖾏 𖼁𖽘𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼦𖽡𖾐 𖽐𖼊𖽙𖾏 𖽐𖼊𖽻𖾐． 29 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽃𖽺𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽃𖽡𖾑，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽘𖾑 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼸𖽻𖾏 𖼁𖽘𖾏 𖼐𖽡𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼺𖽷𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏． 30 𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼯𖽠 𖼐𖽡𖾑 𖼘𖽪！ 𖽐𖼡𖽜𖾐 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼐𖽡𖽙𖾐 𖼺𖽷𖾐， 𖼀𖽡 𖼟𖽡 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼑𖽡𖾏 𖼻𖽻𖾏 𖼼𖽘𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽷𖾐，𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼅𖽡 𖼖𖽺𖾑！ 31 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏，𖼐𖽑𖽡𖾏：‘𖼀𖽡𖾐 𖼖𖽔𖾑 𖼑𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？𖼀𖽡𖾐 𖼖𖽔𖾑 𖼦𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？𖼀𖽡𖾐 𖼖𖽔𖾑 𖼐𖽑𖽘𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏？’ 32 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖽐𖼎𖽑𖽦𖾑 𖼋𖽡𖾏 𖼐𖽡𖾑，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼏𖽻 𖼐𖽡𖾑， 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼀𖽻𖾐 𖼯𖽡𖽙． 33 𖼅𖽡 𖼺𖽪𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼞𖽪𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖼐𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼮𖽑𖽨𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜． 34 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽳𖾏 𖼀𖽡 𖼟𖽡， 𖼀𖽡 𖼟𖽡 𖼅𖽔𖾏 𖼀𖽡 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽳𖾏，𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽔𖾑 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼯𖽡𖾏 𖼷𖽻𖾏 𖼋𖽘𖾏．”
