𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽔 𖼳𖽢 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏, 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼮𖽪𖾐 𖼾𖽡. 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽔 𖼳𖽢 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡.” 2 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽔 𖼳𖽢 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼮𖽔𖾑 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼞𖽳𖾏 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼖𖽡 𖼺𖽷𖾏 𖼺𖽘𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏.” 3 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼗𖽻 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽜𖾑 𖼟𖽡 𖽐𖼷𖽜𖾏 𖼯𖽔𖾏 𖼀𖽡𖾐 𖼳𖽤𖾐 𖼐𖽑𖽪𖾐, 𖼋𖽙𖾏 𖼀𖽡𖾐 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽨𖾏 𖼀𖽡𖾐, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼟𖽡.” 4 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽡𖽙𖾐, 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽻, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑. 5 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼎𖽑𖽻𖾏 𖽐𖼎𖽹𖾏 𖼙𖽡𖽝𖾏 𖼖𖽠𖾑 𖼑𖽨𖾏 𖼺𖽷𖾐 𖼗𖽙, 𖼑𖽪 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽝𖾑?” 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼄𖽔𖾐 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏.
𖼺𖽻𖾐 𖼘𖽙 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐
6 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽙𖾑𖽑 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽡𖽾 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐. 7 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼞𖽡𖽠𖾏 𖼑𖽨𖾏 𖼀𖽑𖽔𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼷𖽑𖽹𖾏, 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑, 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼸𖽙𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼊𖽑𖽡𖽝𖾐 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼀𖽑𖽔𖾏 𖼐𖽱𖾏 𖼑𖽨𖾏 𖼺𖽘𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐. 8 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼺𖽳 𖼮𖽔𖾑 𖼾𖽱 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼉𖽻𖾏, 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼉𖽻𖾏. 9 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼊𖽡𖾐 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: 𖼀𖽡 𖼟𖽡 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼮𖽳, 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏.” 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝 . 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼀𖽡 𖼟𖽡 𖼀𖽡𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼅𖽡, 𖼯𖽔𖾏 𖼅𖽡 𖼸𖽙𖾏 𖼅𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼯𖽡𖽙𖾏 𖼀𖽡𖾐.” 11 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐, 𖼺𖽻𖾐 𖼘𖽙 𖼇𖽺𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼷𖽷𖾐 𖼸𖽻, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽙 𖼾𖽡, 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙 𖼚𖽔𖾏 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼮𖽳, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼏𖽻 𖼐𖽑𖽙𖾏 𖽃𖽡𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼏𖽘 𖼀𖽜𖾑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏.
12 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼀𖽡𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑? 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼀𖽡𖾐 𖼻𖽻𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼺𖽘𖾏.” 13 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏: “𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.” 14 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡 𖼮𖽢, 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖽙.” 15 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡 𖼮𖽢 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 . 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑，𖽃𖽺𖾐 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝.
