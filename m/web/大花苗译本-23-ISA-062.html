𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏
62
𖼞𖽪𖾑 𖼑𖽜 𖼳𖽡 𖼣𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏,
𖼑𖽜 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 ,
𖼋𖽙𖾏 𖼸𖽙 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼮𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼐𖽱𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼺𖽱𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼮𖽡𖾏 𖼙𖽪 .
2 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏．
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼷𖽷𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐,
𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼊𖽡𖾏 𖼯𖽡𖽙𖾏.
3 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏,
𖼐𖽡𖽙𖾐 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖽐𖼚𖽡𖽝𖾏 𖼋𖽡 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑.
4 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏”,
𖼟𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “𖼊𖽡𖾐 𖼡𖽔𖾏 𖼁𖽙𖾏”．
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼙𖽔 𖽃𖽡𖾑”,
𖼟𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼈𖽻𖾑”．
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼙𖽔 𖼟𖽡 ,
𖼟𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼐𖽱𖾏 .
5 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼷𖽙𖾏 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 ,
𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏(“𖼖𖽡𖾑 𖼉𖽪𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼊𖽪𖾐”) 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼷𖽙𖾏 𖼟𖽡 .
𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽤𖾐 𖼮𖽑𖽦𖾐,
𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼟𖽡 .
 
6 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪 !
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏.
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾑 𖼘𖽪 , 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽐𖼎𖽙𖾏,
7 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽙𖾏 𖼮𖽡𖽝𖾏,
𖼋𖽙𖾏 𖼸𖽙 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ,
𖼟𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽻𖾑 𖼖𖽔 .
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼟𖽡 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 ,
𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼦𖽻𖾏 𖼟𖽡 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑.
9 𖼮𖽔 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼑𖽻𖾏, 𖼎𖽺𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼦𖽻𖾏.”
𖼉𖽻𖾏 𖼳𖽡𖾏 𖼽𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽵𖾑 𖼞𖽡𖾑
10 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 ，𖼎𖽑𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏,
𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼅𖽡 𖼽𖽔𖾑 𖼨𖽑𖽢𖾏 𖼨𖽳𖾑𖽑 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼺𖽘𖾏,
𖼑𖽜 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
11 𖼑𖽔𖾏 𖼘𖽪! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏 :
“𖽐𖼎𖽙 𖼳𖽡 𖼣𖽔𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 (“𖼖𖽡𖾑 𖼉𖽪𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏”)𖼐𖽑𖽡𖾏 :
𖼑𖽔𖾏 𖼘𖽪! 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼟𖽡 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏.
𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.”
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 “ 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 ”, 𖽃𖽔𖾏 “ 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽔𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 ”,
𖼟𖽡 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏”, 𖽃𖽔𖾏 “𖼙𖽻 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏”.
