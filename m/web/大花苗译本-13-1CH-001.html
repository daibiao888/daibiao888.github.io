34 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 . 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 . 35 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 、 𖼖𖽨 𖽃𖽳𖾐 、 𖼽𖽦𖾑 𖽂𖽪𖾑 𖼳𖽷𖾑 、 𖼽𖽔𖾐 𖼖𖽔 、 𖼞𖽑𖽙𖾐 𖼖𖽔𖾑 . 36 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼊𖽑𖽡 𖼄𖽔𖾏、 𖽃𖽔𖾑 𖼄𖽙 、 𖼳𖽡𖾐 𖼀𖽙𖾑 、 𖼮𖽢𖾑 𖼊𖽑𖽗𖾐 、 𖼮𖽡𖾑 𖼐𖽔 𖼺𖽷𖾑 、 𖼊𖽑𖽡 𖼐𖽔 、 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 . 37 𖼖𖽨 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽔 𖼦𖽔𖾐 、 𖼳𖽡𖽝𖾏 𖼖𖽔𖾑 、 𖼳𖽔𖾑 𖼄𖽔𖾐 、 𖼄𖽡𖾐 𖼺𖽔𖾐 .
𖼳𖽡𖾑 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
（𖼷𖽷𖾏 36： 20 - 30）
38 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼘𖽙 𖼊𖽑𖽗𖾑 , 𖼳𖽙 𖼀𖽔𖾑 、 𖼮𖽡𖾏 𖼀𖽡𖽝𖾏 、 𖼽𖽔𖾏 𖼐𖽔 、 𖼊𖽡𖾐 𖼳𖽪𖽝𖾏 、 𖼽𖽡𖾐 𖼮𖽑𖽔 、 𖼊𖽡𖾐 𖼳𖽗𖾑 . 39 𖼘𖽙 𖼊𖽑𖽗𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼦𖽙 𖼖𖽡𖾏 、 𖼦𖽙 𖼄𖽔𖾏 . 𖼘𖽙 𖼊𖽑𖽗𖾑 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼊𖽑𖽡 𖼐𖽔 . 40 𖼳𖽙 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼖𖽳 𖽂𖽺 、 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽢 、 𖼽𖽡𖾑 𖼀𖽔𖾑 𖼖𖽪 、 𖼳𖽷𖾏 𖼇𖽺𖾑 、 𖽃𖽔𖾑 𖼐𖽔 . 𖼮𖽡𖾏 𖼀𖽡𖽝𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼽𖽔𖾐 、 𖼽𖽔𖾏 𖼐𖽔 . 41 𖼽𖽔𖾏 𖼐𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼳𖽪𖽝𖾏 . 𖼊𖽡𖾐 𖼳𖽪𖽝𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼦𖽔𖾐 𖼄𖽳 𖼖𖽔 、 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼀𖽔𖾑 、 𖼽𖽡 𖼖𖽔 、 𖼮𖽡𖾑 𖼖𖽔 . 42 𖼽𖽡𖾐 𖼮𖽑𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼀𖽑𖽡 𖼦𖽗𖾐 、 𖼺𖽔𖾐 𖼇𖽔𖾑 、 𖼽𖽔𖾏 𖼞𖽔𖾑 . 𖼊𖽡𖾐 𖼳𖽪𖽝𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽂𖽪𖾑 𖼺𖽷𖾑 、 𖼽𖽔𖾏 𖼖𖽔 .
