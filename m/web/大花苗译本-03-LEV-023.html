𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 , 𖼅𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖽃𖽔𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 .
3 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 , 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼅𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 𖼅𖽡 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 , 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼻𖽙𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 .
4 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 , 𖼟𖽪𖾏 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖽃𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ．
（𖼋𖽠𖾏 12：1 - 20；𖼉𖽪𖾏28：16 - 25；𖼎𖽺𖾏16：1 - 8）
5 𖼖𖽪𖾑 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏 𖼟𖽻𖾏 𖼚𖽻𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼢𖽹𖾐 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏. 6 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼐𖽡𖾑 𖼟𖽻𖾏 𖼀𖽜𖾐 , 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 . 7 𖼐𖽑𖽪𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏； 8 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑， 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 . 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 .”

27 “𖼖𖽪𖾑 𖼳𖽤𖾏 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼟𖽻𖾏 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 . 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 28 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 , 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 , 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼚𖽠𖾏 𖼾𖽔𖾏 . 29 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏 . 30 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼺𖽘𖾏. 31 𖼅𖽡 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 . 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 , 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 . 32 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 . 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 , 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼐𖽡𖾑 𖼺𖽡𖽝𖾐 𖼯𖽢𖾐 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼻𖽻𖾏 𖼷𖽻𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼻𖽻𖾏 𖼷𖽻𖾏 , 𖼽𖽔𖾑 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 .”
𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐
