𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽡𖽠𖾏 𖼋𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑 ?

4 𖼐𖽡𖽙𖾐 𖼞𖽙𖾏 𖽐𖼡𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽪𖾏 𖽐𖼡𖽜𖾐 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑 ,

5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽠𖾑 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖽐𖼊𖽑𖽡𖾏 𖼸𖽜 𖽃𖽡𖾐 𖼼𖽘𖾐，
6 𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼦𖽘𖾑 𖽃𖽔𖾐 𖼟𖽱𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼚𖽑𖽡𖽝𖾐 𖽐𖼚𖽙 ,
𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼡𖽑𖽙𖾑 𖼁𖽘𖾏 𖼡𖽑𖽙𖾏 𖼷𖽔𖾏 𖽐𖼚𖽙 ,
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽙𖾏 𖽐𖼡𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 ,
𖼐𖽡𖽙𖾐 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼦𖽡 𖼸𖽱 .
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽻 𖽐𖼊𖽻𖾏 , 𖼖𖽠𖾑 𖼋𖽠𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 .
 
9 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼨𖽑𖽪𖾏 ,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼚𖽙𖾏 .
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 ,
𖽐𖼊𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼊𖽱𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽐𖼊𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 .
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽙𖾏 𖼞𖽪𖾑 𖼖𖽑𖽔𖾏 𖼐𖽑𖽜𖾑 𖼋𖽻 𖼺𖽘𖾏 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽻𖾏 𖽃𖽦𖾐 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼚𖽠𖾏 𖼺𖽘𖾏 𖼖𖽑𖽔𖾏 𖼺𖽻𖾑 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 .
12 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼺𖽱𖾏 𖼚𖽘𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 ,
𖼊𖽑𖽠𖾐 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 ,
𖼮𖽪𖾐 𖼊𖽡𖽝𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 .
13 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽐𖼀𖽘𖾏 𖼐𖽡𖾑 ,
𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 , 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 .
14 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽵 𖼁𖽺𖾏 𖼋𖽔𖾏 𖼺𖽱𖾏 𖽐𖼮𖽵 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 ,
𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡 𖽐𖼚𖽙.
15 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 ,
𖼖𖽠𖾑 𖼞𖽪𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼺𖽱𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼙𖽠𖾏 ,

16 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼾𖽘𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 ,
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼞𖽪𖾑 .
17 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼮𖽑𖽡𖽙𖾐 𖼞𖽪𖾑 ,
𖼄𖽙𖾐 𖼟𖽪𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖽐𖼮𖽦 , 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼳𖽙𖾏 𖼞𖽡𖽠𖾏 𖼞𖽪𖾑 .
18 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼷𖽷𖾐 ,

19 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽘𖾑 𖽐𖼚𖽙 ,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼺𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼮𖽑𖽻𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
20 𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡 ,
𖼈𖽡𖽝𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼟𖽡 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼞𖽪𖾑 .
21 𖼟𖽡 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 , 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼞𖽪𖾑 .
22 𖼟𖽡 𖼮𖽑𖽳𖾑 𖼞𖽪𖾑 𖼎𖽻𖾏 𖼮𖽢𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼮𖽢𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 ,
