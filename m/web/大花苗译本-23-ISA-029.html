𖼟𖽡 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼟𖽡 𖼗𖽪𖾏 𖼽𖽔𖾑 𖼖𖽙𖾏 𖼘𖽑𖽙𖾐 𖼖𖽙𖾏 𖼳𖽻𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 .
5 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑（“𖼞𖽡𖾑 𖼀𖽔𖾑” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 ”） 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼦𖽡𖾑 𖼁𖽘𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽑𖽔𖾐 𖼽𖽘𖾏 𖼙𖽠𖾏．

6 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼺𖽙𖾐 𖼢𖽔𖾐、 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏、 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐、 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、
𖼊𖽑𖽡𖽝𖾐 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼟𖽪𖾏 𖽐𖼡𖽙𖾏 𖼖𖽻𖾐 , 𖼗𖽙𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 .
7 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼽𖽔𖾏 𖼖𖽡𖾏 𖼽𖽡𖾑 𖼖𖽳 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼽𖽔𖾏 𖼖𖽡𖾏 𖼽𖽡𖾑 𖼖𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑, 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑, 𖼺𖽱𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏.
8 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼑𖽻𖾏 𖼉𖽔𖾏,

𖼡𖽑𖽙𖾑 𖼯𖽢 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽑𖽜𖾏 𖽃𖽻𖾐 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼦𖽻𖾏 𖽃𖽻𖾐,

𖼖𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔
