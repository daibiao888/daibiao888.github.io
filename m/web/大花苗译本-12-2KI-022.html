（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 34： 1 - 2）
22
𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏. 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼽𖽦𖾑 𖼊𖽡𖾐 𖼊𖽔𖾏, 𖼟𖽪𖾏 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼮𖽢𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾏 𖼊𖽔𖾏 𖼽𖽔𖾐 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏. 2 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼖𖽡𖾐 𖼈𖽳𖾐 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏.
𖼊𖽻𖾏 𖼁𖽙𖾏 𖽐𖼊𖽠𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 34：8 - 28）
3 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼄𖽡𖾐 𖼳𖽪𖾑 𖼖𖽔 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐, 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖽐𖼊𖽠𖾑 𖼳𖽔𖾑 𖼇𖽔𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽑𖽡𖾏: 4 “𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼳𖽡𖾑 𖼖𖽳 𖼮𖽢𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼻𖽙𖾑 𖽃𖽔 𖼛𖽻𖾏 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼳𖽻𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼾𖽱 𖼟𖽪𖾏 𖽐𖼎𖽑𖽹𖾏， 5 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽡𖾑, 𖼻𖽻𖾏 𖽐𖼷𖽷𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔， 6 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼋𖽠𖾏 , 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽐𖼊𖽻𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼸𖽻 𖼊𖽡𖽝𖾑 𖼯𖽡𖽙𖾏 𖼅𖽻 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 7 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼐𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖼺𖽙𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝.
