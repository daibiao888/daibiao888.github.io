（𖼋𖽠𖾏 12：1 - 20；𖼖𖽡𖾏 23：5 - 8；𖼉𖽪𖾏 28：16 - 25）
16
𖼟𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼽𖽔𖾏 𖼀𖽡 𖼖𖽑𖽡𖾏, 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼀𖽡 𖼖𖽑𖽡𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 , 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 . 2 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼺𖽹𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑, 𖼎𖽑𖽻𖾏 𖽐𖼀𖽙𖾏 𖼑𖽨𖾐 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼄𖽔𖾐 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 3 𖼟𖽡 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼐𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 ； 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖽐𖼚𖽙 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 , 𖼟𖽪𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 —𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽑𖽜 𖼑𖽜 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 — 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 . 4 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼚𖽻𖾐 𖼺𖽘𖾐 𖽐𖼚𖽙 , 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 . 𖼺𖽪𖾐 𖼁𖽱 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼸𖽙 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 . 5 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽃𖽡𖾑 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑； 6 𖼯𖽔 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼺𖽹𖾑 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 , 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼖𖽔𖾑 𖽐𖼷𖽜𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏 , 𖼖𖽑𖽘𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 . 7 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽹𖾑 , 𖼞𖽑𖽡𖽠𖾏 𖽐𖼡𖽹𖾏 𖼷𖽑𖽡𖽝𖾐 𖼑𖽻𖾏 , 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼽𖽔𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼙𖽠𖾏 . 8 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 , 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 .
𖼳𖽤𖾏 𖼋𖽪𖾏 𖼳𖽤𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐
（𖼋𖽠𖾏 34：22；𖼖𖽡𖾏 23：15 - 21；𖼉𖽪𖾏 28：26 - 31）
9 𖼟𖽡 𖼽𖽔𖾑 𖼮𖽔𖾑 𖼺𖽙𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 , 𖼎𖽑𖽻𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼖𖽡𖽝𖾐 𖼳𖽠𖾑 𖼁𖽱𖾏 𖼤𖽜𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖽐𖼮𖽳 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐 𖽃𖽡𖾑 𖼮𖽔𖾑 𖼺𖽙𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐. 10 𖼟𖽡 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 , 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 , 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼻𖽙𖾑 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼳𖽤𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 . 11 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏、𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼚𖽙 𖼙𖽻 , 𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 , 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽹𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼙𖽔 𖼚𖽙𖾏 . 12 𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼙𖽔 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑 .
𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐
（𖼖𖽡𖾏 23：33 - 34；𖼉𖽪𖾏 29：12 - 39）
13 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼡𖽻𖾐 , 𖼖𖽪𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼮𖽡𖽠𖾑 𖼑𖽡𖾏 𖼮𖽡𖽠𖾑 𖼳𖽻𖾏 𖼮𖽡𖽝𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 . 14 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏、𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼚𖽙 𖼙𖽻, 𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼙𖽔 𖼚𖽙𖾏 . 15 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽹𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽻𖾐 𖼡𖽜𖾏 , 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 , 𖼟𖽡 𖼎𖽺𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝.
16 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼾𖽠 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 , 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼳𖽤𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 , 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 , 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼷𖽷𖾐 𖼟𖽻𖾐 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽹𖾑 𖽃𖽡𖾑 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 . 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 17 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽃𖽡𖾑 , 𖼖𖽑𖽘𖾑 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 .
18 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽃𖽡𖾑 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏． 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖼨𖽑𖽢𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 . 19 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽑𖽪𖾏 𖼯𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼚𖽑𖽡𖾏 𖼇𖽔 𖼙𖽻 𖼺𖽘𖾏 , 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼎𖽙𖾑 𖼡𖽘𖾐 𖼎𖽙𖾏 𖼇𖽻𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼺𖽘𖾏. 20 𖼟𖽡 𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼖𖽠𖾑 𖼞𖽡𖾑 , 𖼮𖽔 𖼅𖽔𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 , 𖼟𖽙𖾏 𖼟𖽡 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 . 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 .
21 𖼟𖽡 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼯𖽡𖽙 𖽐𖼊𖽻𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖽃𖽔𖾏 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 . 22 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼁𖽠𖾏 , 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼳𖽪𖾐 𖼯𖽡𖽙 .”
