𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑
（𖼊𖽑𖽹𖾏 17：1-13；𖼖𖽪𖾏 9：28-36）
2 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼊𖽳、 𖼽𖽔𖾐 𖼞𖽙、 𖼽𖽙 𖼦𖽗 ， 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐， 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑． 3 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼮𖽡𖾏 𖼮𖽑𖽦𖾐 𖼚𖽠𖾐 𖼨𖽑𖽪𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽪𖾐 𖼚𖽻𖾏 𖽐𖼊𖽻𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼚𖽻𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑． 4 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏、 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐， 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏． 5 𖼀𖽡𖾐 𖼊𖽳 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏， 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏！ 𖼯𖽪𖾏 𖼀𖽡𖾐 𖼷𖽑𖽙𖾏 𖼷𖽷𖾐 𖼖𖽪𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐， 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐， 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐，𖽃𖽡𖾐 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼐𖽡𖽙𖾐．” 6 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼟𖽙𖾏 𖼻𖽻𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽹𖾏． 7 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼚𖽙𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽝𖾏， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽐𖼚𖽙 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏， 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼐𖽱𖾏．” 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼞𖽑𖽡 𖼙𖽻，𖼦𖽡𖾐 𖼁𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏， 𖼯𖽔 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 ．

𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼊𖽑𖽡 𖼗𖽻𖾏
