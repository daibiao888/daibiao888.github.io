𖼽𖽙 𖼊𖽑𖽗𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 . 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼳𖽔𖾑 , 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼊𖽪 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 . 2 𖼽𖽙 𖼊𖽑𖽗𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 , 𖼾𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 . 𖼈𖽡𖽝𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼗𖽱𖾏 𖽐𖼚𖽔𖾏 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐. 3 𖼽𖽙 𖼊𖽑𖽗𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼺𖽘𖾑 𖽃𖽔 𖼸𖽷 𖽃𖽔 𖼛𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖽃𖽳 𖼇𖽺𖾑 𖼖𖽳 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 . 4 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼙𖽻 , 𖼐𖽡𖽙𖾐 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽐𖼎𖽙 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 . 5 𖼽𖽙 𖼊𖽑𖽗𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 , 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 𖼑𖽡𖽝𖾏、 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼞𖽙𖾑 𖽃𖽳𖾐 𖼳𖽻𖾐 、 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼞𖽙𖾑 𖽃𖽳𖾐 𖼅𖽙 𖼎𖽻𖾏 𖼽𖽙 𖼊𖽑𖽗𖾐 . 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 、 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 , 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 . 6 𖼽𖽙 𖼊𖽑𖽗𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳𖾏 , 𖼎𖽺𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼾𖽨𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 . 7 𖼽𖽙 𖼊𖽑𖽗𖾐 𖼑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 , 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏《 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏》 𖼮𖽡𖽝𖾏 . 8 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 . 9 𖼽𖽙 𖼊𖽑𖽗𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 , 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻 . 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
