𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
11
𖼞𖽪𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏, 𖼟𖽡 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼐𖽻 𖼽𖽘𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼙𖽠𖾏.
2 𖼑𖽔𖾏 𖼘𖽪！ 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽐𖼊𖽘𖾏 𖼐𖽑𖽜𖾑 𖼞𖽑𖽡𖽠𖾏 𖼈𖽙𖾏 𖽐𖼊𖽡𖾏 𖼮𖽡𖽝𖾏,
𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼀𖽙𖾑 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳.
3 𖼊𖽠𖾏 𖽐𖼞𖽔𖾏 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼡𖽑𖽔𖾑 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑?
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐.
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳,
𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼳𖽪𖾐.
6 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽑𖽔𖾑 𖼸𖽙𖾏,
𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏、𖼖𖽨 𖼦𖽫、𖼮𖽢𖾏 𖼟𖽪𖾏 𖼳𖽙𖾑, 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼐𖽱𖾏 𖽐𖼀𖽠𖾏.
7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏.
