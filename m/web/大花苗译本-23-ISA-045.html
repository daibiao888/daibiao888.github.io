𖼞𖽪𖾐 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼚𖽜𖾑 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏,
𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽠𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏,
𖼟𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼡𖽙𖾏.
𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
2 “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏,
𖼨𖽑𖽢𖾏 𖼋𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼞𖽡𖾐 𖼀𖽡𖾐 𖽐𖼞𖽱𖾐．
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖽐𖼚𖽔 𖽃𖽔 𖼛𖽻𖾏 𖼋𖽻𖾏,
𖼮𖽷𖾐 𖼊𖽪𖾏 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖼎𖽪𖾑 𖼺𖽘𖾏 .
3 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 ,
𖼟𖽙𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽔𖾑 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡 ,
𖼟𖽪𖾏 𖼞𖽪𖾑 — 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
4 𖼑𖽜 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼽𖽔𖾐 𖼞𖽙 ,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼺𖽹𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏,
𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼷𖽷𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡 ,
𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼈𖽡𖽝𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼻𖽻𖾏 𖽐𖼷𖽷𖾏 𖼎𖽻𖾏 𖼟𖽡 .
𖼋𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
5 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼋𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼈𖽡𖽝𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼟𖽡 𖼚𖽔𖾑, 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏．
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼸𖽙 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐, 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
7 𖼞𖽪𖾑 𖼷𖽷𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑, 𖼎𖽺𖾏 𖼷𖽷𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼞𖽪𖾑 𖼺𖽘𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪, 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 — 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
 
8 𖽐𖼊𖽪𖾏 𖼘𖽪 ! 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖽐𖼚𖽙 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏,
𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼘𖽪 !𖼮𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏.
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽠 𖼞𖽘𖾐, 𖼘𖽑𖽘𖾑 𖼋𖽠𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏, 𖼟𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽡𖾐 𖼸𖽻 𖼘𖽑𖽘𖾑 𖼘𖽑𖽙𖾐,
𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 —𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾏 𖼯𖽡𖽙𖾏.
9 𖽃𖽺𖾐! 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼐𖽱𖾏 𖼟𖽪𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖽃𖽔𖾑 𖼷𖽑𖽹𖾏 𖼉𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼷𖽑𖽹𖾏 𖼉𖽔𖾏 𖼊𖽔𖾐,
𖽃𖽘𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼀𖽔𖾐 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏?
𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼟𖽡𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽺𖾐 𖼋𖽡𖾏 𖼘𖽪?
10 𖽃𖽺𖾐! 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏:
𖼟𖽡 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?
𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 (“ 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 ” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽃𖽔 𖼁𖽙𖾏”) 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖼦𖽡 𖽐𖼀𖽱𖾏 𖼄𖽙𖾐 𖼘𖽑𖽘𖾑 𖼾𖽨 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?”
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼞𖽪𖾑 𖽐𖼀𖽙𖾏 𖼊𖽪𖾐 𖼑𖽡𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼑𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽪 𖼞𖽪𖾑 𖼺𖽷𖾐 𖼖𖽔𖾏?
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏, 𖼅𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼺𖽷𖾐 𖼖𖽔𖾏?
12 𖼞𖽪𖾑 𖼷𖽷𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖼎𖽺𖾏 𖼷𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼞𖽪𖾑 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖼊𖽑𖽔𖾑 𖼀𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 ,
𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼏𖽵 𖼊𖽡𖾏 𖼮𖽡𖽝𖾏.
13 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼞𖽪𖾐 𖼖𖽡𖽝 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼐𖽱𖾏”),
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾏 𖼨𖽳𖾑𖽑 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 ,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼙𖽻 ,
𖼚𖽠𖾏 𖼮𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏,
𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽜 𖽐𖼡𖽜𖾏 𖼼𖽙, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼑𖽜 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
 
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
