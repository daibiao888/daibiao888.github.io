𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 、 𖼀𖽔𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏
5
𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽙 𖽃𖽗𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼐𖽑𖽡𖾏:
2 “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼖𖽔𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼗𖽔𖾏 𖼾𖽨𖾏 𖼳𖽦𖾑,
𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
 
3 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 ， 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼘𖽪！ 𖼽𖽔𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 ，
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
 
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖼅𖽻𖾏 𖼎𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 ,
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽐𖼊𖽪𖾏 𖼮𖽑𖽻𖾏 𖼙𖽻𖾏,
𖼚𖽻𖾏 𖽃𖽻𖾐 𖼖𖽔𖾏 𖼀𖽻𖾐 𖼑𖽘.
5 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼚𖽑𖽪𖾏, 𖼎𖽺𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙，
𖼖𖽪𖾑 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽑𖽪𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.
 
6 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼐𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽔𖾑 𖼮𖽢𖾑𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾐 𖼽𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏, 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏,
𖼊𖽪𖾐 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 𖼖𖽔𖾏 𖼮𖽑𖽙𖾑 𖼞𖽡𖾑 𖼅𖽻𖾏 𖼊𖽔𖾐.
7 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼙𖽻 𖼼𖽙𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏! 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏,
𖼋𖽙𖾏 𖼞𖽪𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼑𖽡𖽝𖾏.
8 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽑𖽦𖾑 𖼮𖽵𖾑 𖼳𖽺 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼁𖽔𖾏 𖼏𖽙 𖼎𖽺𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻．
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼚𖽻𖾐 𖼉𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼋𖽡𖽝𖾏 𖼁𖽙𖾏 𖼅𖽔𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾑 𖼄𖽑𖽪𖾑 𖼘𖽪？
9 𖼞𖽪𖾑 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼗𖽔𖾏 𖼾𖽨𖾏 𖼳𖽦𖾑,
𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
 
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼟𖽪𖾏 𖼚𖽠𖾐,
𖼼𖽻𖾏 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽻 𖽃𖽡𖾑,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑,
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏！
11 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐 𖽐𖼎𖽙𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾑 𖼐𖽜𖾑𖽑 𖼑𖽡𖾏 𖼳𖽘𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑,
𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑．
 
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻.
 
12 𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 𖼘𖽪, 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 ! 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 !
𖼟𖽡 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏, 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏, 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 !
𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽙 𖽃𖽗𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼘𖽪, 𖼟𖽡 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏,
𖼦𖽔𖾏 𖼊𖽻𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 !
13 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼡𖽜𖾏 𖼋𖽔𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖼊𖽪𖾐 𖼋𖽻.
14 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖽐𖼡𖽜𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑,
𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼟𖽡．
𖼅𖽔𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼎𖽑𖽻𖾏 𖼄𖽔𖾐 𖼮𖽡 𖽐𖼡𖽜𖾏 𖼋𖽔𖾏,
𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏 𖼾𖽱 𖼖𖽡𖾑 𖼉𖽪𖾏 （𖼡𖽑𖽙𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼊𖽪𖾐 𖽃𖽔𖾏 𖽐𖼊𖽠𖾑”）𖼎𖽑𖽻𖾏 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖽐𖼡𖽜𖾏 𖼋𖽔𖾏.
15 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽔𖾏,
𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏.
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽙𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼏𖽘 𖼚𖽔𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏,
𖼐𖽡𖽙𖾐 𖼖𖽨 𖼀𖽡𖽝𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪, 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
16 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼾𖽘𖾐,
𖼑𖽻 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼘𖽪𖾏 𖼳𖽘𖾐 𖼘𖽪?

24 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼮𖽡𖾑 𖼐𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽡𖾑 𖼀𖽳 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼽𖽔𖾐 𖼽𖽡𖾏 𖼊𖽻𖾏 𖼅𖽳𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔 𖼁𖽙𖾏,

27 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼖𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼊𖽠𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 、𖼢𖽻 𖼚𖽠𖾑、𖽐𖼀𖽑𖽻𖾏 𖽐𖼀𖽙,

29 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼐𖽱𖾏,

30 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽷𖾏 𖼐𖽔𖾑 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼇𖽺𖾐 𖽐𖼎𖽙𖾏 𖼐𖽔𖾑?

31 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼖𖽻𖾐,
