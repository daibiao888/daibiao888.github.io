𖼀𖽔𖾑 𖼖𖽳 𖼨𖽑𖽪𖾏 𖼀𖽔𖾑 𖼖𖽔
22
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖼅𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼄𖽙𖾑 𖼽𖽔𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 , 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 , 𖼊𖽱𖾐 𖼎𖽻𖾏𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 𖼮𖽪𖾐 𖼾𖽡 . 2 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼳𖽡𖾑 𖼀𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼖𖽳 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽘𖾏 . 3 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽔𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽹𖾏， 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏 𖽐𖼮𖽑𖽹𖾏 . 4 𖼄𖽙𖾑 𖼽𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼗𖽻 𖼐𖽑𖽡𖾏 : “𖼐𖽔𖾑 𖼐𖽡𖾑 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼀𖽡𖾐 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 , 𖼺𖽱𖾏 𖼑𖽨𖾏 𖼑𖽻𖾏 𖼋𖽘𖾏 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 .”
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼳𖽡𖾑 𖼀𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼖𖽳 𖽃𖽔𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 5 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼗𖽘𖾏 𖽃𖽔𖾐 𖼟𖽱𖾐 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼀𖽑𖽡𖾐 𖼊𖽙 . 𖼸𖽙 𖼀𖽡𖾐 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼖𖽔 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼼𖽙𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼨𖽑𖽪𖾏 𖼀𖽔𖾑 𖼖𖽔 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼑𖽔𖾏 ! 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼐𖽡𖽙𖾐 𖼀𖽪𖾑 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 , 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽘𖾑 𖽃𖽡𖾑 . 6 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼋𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑 , 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼋𖽔𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼞𖽪𖾑 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏 , 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 . 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼑𖽜 𖼡𖽔 𖼋𖽱𖾏 𖼏𖽵 𖼅𖽳𖾏 , 𖼡𖽔 𖼋𖽱𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼅𖽳𖾏 , 𖼟𖽡 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 , 𖼡𖽔 𖼋𖽱𖾏 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 .”
7 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼗𖽻 𖽐𖼎𖽙 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼗𖽻 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼮𖽔𖾑 𖼳𖽦𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼖𖽔 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼞𖽑𖽔𖾏 𖼀𖽔𖾑 𖼖𖽳 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 . 8 𖼀𖽔𖾑 𖼖𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼄𖽑𖽙𖾏 𖼐𖽔𖾑 𖼅𖽡 𖼀𖽱𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 , 𖼊𖽡𖾐 𖼅𖽡 .” 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼀𖽱𖾏 𖼀𖽔𖾑 𖼖𖽔 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼎𖽙𖾏 𖼋𖽘𖾏 . 9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼀𖽔𖾑 𖼖𖽔 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 ?” 10 𖼀𖽔𖾑 𖼖𖽔 𖼊𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽪𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼀𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼀𖽔𖾑 𖼖𖽳 , 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼐𖽑𖽡𖾏 : 11 𖼑𖽔𖾏 𖼘𖽪 ! 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼐𖽡𖽙𖾐 𖼀𖽪𖾑 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 , 𖼟𖽡 𖼋𖽔𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼞𖽪𖾑 𖼸𖽠𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏 .” 12 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼖𖽔 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 .” 13 𖼀𖽔𖾑 𖼖𖽔 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼖𖽳 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼅𖽻𖾏.” 14 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼖𖽳 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼐𖽑𖽡𖾏 : “𖼀𖽔𖾑 𖼖𖽔 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼋𖽔𖾏 .”
