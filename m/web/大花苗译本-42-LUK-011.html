𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
（𖼊𖽑𖽹𖾏 6：9-15；7：7-11）
11
𖼅𖽔𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 ，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “ 𖼮𖽵𖾑 𖼘𖽪 ！𖽐𖼮𖽢𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼦𖽗 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐．” 2 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：
𖼀𖽡𖾐 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏，
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑，
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡𖾏 𖼞𖽪𖽙 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏，
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐，
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏．
3 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽻𖾏，𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐．
4 𖼯𖽔𖾏 𖼀𖽡𖾐 𖼾𖽔𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼙𖽔 𖼯𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽷𖾏 𖼀𖽡𖾐 𖽐𖼮𖽷𖾏 𖼺𖽘𖾏．
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽐𖼮𖽷𖾐 𖼏𖽻（“𖽐𖼮𖽷𖾐 𖼏𖽻” 𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼁𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 ”） 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑，𖼮𖽨𖾏 𖼀𖽡𖾐 𖼚𖽠𖾏 𖼛𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏．”
𖼮𖽵𖾑𖽃𖽔𖾏𖼞𖽑𖽜𖾑𖼏𖽔𖾏
𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽔𖾑 𖼯𖽪 𖼯𖽪𖾏 𖽐𖼊𖽜 𖼽𖽔𖾑
5 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔，𖼟𖽪𖾏 𖼋𖽘 𖼄𖽑𖽙𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑， 𖼐𖽑𖽡𖾏：‘𖼽𖽺 𖼗𖽔！ 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼷𖽷𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑， 6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑，𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼄𖽔𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼊𖽜𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼑𖽻𖾏．’ 7 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：‘𖼡𖽑𖽔𖾐 𖼮𖽑𖽙 𖼞𖽪𖾑 𖼊𖽡𖾐，𖼞𖽪𖾑 𖽃𖽔 𖼛𖽻𖾏 𖼡𖽙𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏，𖼞𖽪𖾑 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏 𖼸𖽘𖾏 𖼋𖽘，𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡．’ 8 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼽𖽺 𖼗𖽔 𖼐𖽔𖾑 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼚𖽑𖽪𖾏 𖼯𖽪 𖼯𖽪𖾏 𖽐𖼊𖽜 𖼽𖽔𖾑，𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏．” 9 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡：“ 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡，𖼅𖽡 𖽐𖼎𖽑𖽦𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏，𖼅𖽡 𖽐𖼀𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼡𖽑𖽜𖾐． 10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 ，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 ， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼺𖽘𖾏． 11 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼉𖽺𖾏，𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 𖽐𖼊𖽜 𖼽𖽔𖾑 𖼳𖽡𖽝𖾐，𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽺𖾑？𖽐𖼊𖽜 𖼽𖽔𖾑 𖼊𖽔𖾑 𖽐𖼀𖽳𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼘𖽪？ 12 𖽐𖼊𖽜 𖼡𖽳𖾏 𖼡𖽹𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼀𖽡𖾑 𖼞𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼚𖽡𖽝𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼘𖽪？ 13 𖼅𖽡 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏，𖽃𖽔𖾏 𖼐𖽡𖾑 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼄𖽔𖾐 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼘𖽪？”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐
（𖼊𖽑𖽹𖾏 12：22-30；𖼞𖽑𖽙𖾐 3：20-27）
