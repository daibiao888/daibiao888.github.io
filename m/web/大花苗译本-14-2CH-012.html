（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 14： 21 - 31）
12
𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 , 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼺𖽘𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 . 2 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 , 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽷𖾏 𖼺𖽔𖾐 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 3 𖼳𖽷𖾏 𖼺𖽔𖾐 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼗𖽪𖾏 , 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼚𖽻𖾏 𖼉𖽻𖾏 , 𖼖𖽪𖾏 𖼀𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 、 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 、 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽡𖾑 , 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼾𖽱 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 . 4 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 , 𖼎𖽺𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 . 5 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 , 𖼽𖽨 𖼊𖽔𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 , 𖼑𖽜 𖼳𖽷𖾏 𖼺𖽔𖾐 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 . 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼽𖽔𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼅𖽡 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼅𖽡 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼺𖽔𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .” 6 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 , 𖼾𖽨𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼯𖽡𖽙𖾏 .”
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽻𖾐 𖼢𖽜 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼎𖽺𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼽𖽔𖾐 𖼐𖽑𖽡𖾏 : “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 , 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼡𖽜𖾑 𖼷𖽹𖾏𖼳𖽷𖾏 𖼺𖽔𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 . 8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 , 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑 , 𖽐𖼎𖽙 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼋𖽪 𖼇𖽺𖾐 𖼡𖽔𖾏 𖽃𖽔𖾑 𖼳𖽷𖾏 .”
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼺𖽔𖾐 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 , 𖼞𖽑𖽔𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽠𖾏 , 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖽃𖽔𖾏 . 10 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼗𖽙𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖽃𖽔𖾏 𖽃𖽡𖾑 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽃𖽔 𖼛𖽻𖾏 𖽃𖽡𖾑 𖼑𖽔𖾏 𖼻𖽙𖾑 . 11 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽔𖾏 𖼟𖽻𖾏 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 , 𖼏𖽙 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 , 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼺𖽷𖾐 𖼅𖽻 , 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼮𖽡𖽝𖾏 . 12 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽻𖾐 𖼢𖽜 𖽃𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼾𖽘𖾏 𖼙𖽠𖾏 , 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 .
