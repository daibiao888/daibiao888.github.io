2
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼄𖽙𖾑 𖼽𖽔𖾏 𖼯𖽪 𖼯𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼷𖽑𖽷𖾑 𖼟𖽙𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽑𖽻𖾑.
2 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 ,
𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼮𖽢𖾑 𖼖𖽡𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑,
𖼄𖽙𖾑 𖼽𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏、 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏、 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.
3 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼄𖽔𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
𖼽𖽨 𖼊𖽔𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽪 𖼯𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑,
𖼦𖽡𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏,
𖼁𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼸𖽙𖾏 𖽐𖼎𖽙 𖽃𖽡𖾑,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼙𖽠𖾏.
5 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 ,
𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 .”
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽪 𖼯𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼋𖽺 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼑𖽜 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖼞𖽑𖽻𖾏 𖼊𖽔𖾐 𖼋𖽺 𖼺𖽘𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 ,
𖼦𖽔𖾏 𖼙𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼑𖽡𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏,
𖼉𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏,
𖼷𖽪𖾑 𖼷𖽪𖾑 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾑 𖽐𖼊𖽪 ,
𖼀𖽔𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼮𖽡𖽝𖾏, 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼞𖽔𖾏 𖼁𖽠, 𖼦𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼮𖽡𖽠𖾑 .
 
9 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏, 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼈𖽡𖽝𖾏,
𖼞𖽪𖾑 𖼖𖽻𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼚𖽙𖾐 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐.
10 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼨𖽑𖽳𖾏 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏,
𖼟𖽙𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏.
11 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼅𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐,
𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼐𖽔 𖼳𖽡𖾏 𖽃𖽳𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐.
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪 ！ 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
12 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼄𖽔𖾐 𖼮𖽡𖽠𖾑 𖼎𖽻𖾏 𖼐𖽔 𖼳𖽡𖾏 𖽃𖽳𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽻𖾏,
𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏: ‘ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏.’
 
13 𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽹𖾏 𖼅𖽡 ,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖽃𖽔𖾏 𖼷𖽻𖾏 𖼡𖽻𖾐 𖼸𖽹𖾏 𖼮𖽑𖽡𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
14 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼏𖽘 𖼇𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼾𖽨𖾏 𖼮𖽨𖾏 𖼾𖽨𖾏.
