5 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼖𖽔 𖼺𖽳 𖼳𖽠𖾑 𖼊𖽠𖾏 , 𖼮𖽪𖾐 𖼾𖽡 𖼎𖽻𖾏 𖼺𖽪𖾑 𖼞𖽙 . 6 𖼎𖽑𖽻𖾏 𖼺𖽪𖾑 𖼞𖽙 𖼳𖽠𖾑 𖼅𖽻𖾏, 𖼮𖽪𖾐 𖼾𖽡 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼊𖽑𖽘𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 . 7 𖼎𖽑𖽻𖾏𖼽𖽡𖾑 𖼊𖽑𖽘𖾐 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼀𖽡𖾐 · 𖼦𖽔𖾐 𖼳𖽡𖾑 𖼖𖽪 , 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽔𖾑 𖼖𖽡 · 𖼳𖽡𖾐 𖼇𖽺𖾑 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼄𖽡 𖼊𖽙 𖼀𖽘𖾑𖽑 𖼀𖽡 𖼋𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽪𖾐 𖼾𖽡 ． 8 𖼎𖽑𖽻𖾏 𖼀𖽡𖾐 · 𖼦𖽔𖾐 𖼳𖽡𖾑 𖼖𖽪 𖼳𖽠𖾑 𖼅𖽻𖾏, 𖼋𖽡𖾏 𖼳𖽡𖽠𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 , 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼊𖽑𖽘𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼅𖽻𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼞𖽡𖾑 , 𖼮𖽪𖾐 𖼾𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼄𖽔𖾐 𖼖𖽔𖾑 . 9 𖼎𖽑𖽻𖾏 𖼄𖽔𖾐 𖼖𖽔𖾑 𖼳𖽠𖾑 𖼅𖽻𖾏, 𖼗𖽙𖾏 𖼸𖽙 𖼽𖽡𖾑 𖼖𖽡𖽾 , 𖼽𖽡𖾑 𖼖𖽡𖽾 𖼅𖽔𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼁𖽘 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼷𖽠𖾑𖽑 , 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼇𖽘𖾐 𖼺𖽱𖾑 𖼺𖽱𖾐 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼮𖽪𖾐 𖼾𖽡 . 10 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡𖽾 𖼳𖽠𖾑 𖼅𖽻𖾏, 𖼮𖽪𖾐 𖼾𖽡 𖼎𖽻𖾏 𖼳𖽡𖽠𖾑 𖼖𖽡𖽝𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 .
