𖼽𖽙 𖼀𖽳 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 :
2 “ 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼑𖽻 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼋𖽘𖾏 ,
𖼅𖽡 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 .
3 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖼘𖽪𖾏 ?
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽡𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽘𖾏 𖼖𖽺𖾑 ?
4 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼅𖽡 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼖𖽡 𖼉𖽺𖾏 ,
𖼅𖽡 𖼮𖽔𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽷𖾐 ,
𖼞𖽪𖾑 𖼖𖽔𖾏 𖼸𖽠𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼊𖽱𖾑 𖼅𖽡 ,
𖼎𖽺𖾏 𖼇𖽜𖾏 𖼇𖽜𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼎𖽻𖾏 𖼅𖽡 .
5 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼅𖽡 ,
𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏 𖼾𖽘𖾏 𖼺𖽘𖾏 .
𖼽𖽙 𖼀𖽳 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏
6 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼈𖽡𖽝𖾏 , 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼺𖽘𖾏 ,
𖼞𖽪𖾑 𖼮𖽔𖾑 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 , 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼘𖽪 ?
7 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼯𖽻 𖼼𖽙 ,
𖼟𖽡 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 ，
8 𖼟𖽡 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼞𖽪𖾑 , 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖼾𖽔 , 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙 .
9 𖼮𖽵𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼞𖽪𖾑 、 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 ,
𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼞𖽪𖾑 𖼋𖽙𖾏 𖼐𖽑𖽡𖽝𖾑 𖼷𖽷𖾏 𖼞𖽹𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 .
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 ,
𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑 , 𖼦𖽡 𖼸𖽱 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼞𖽪𖾑 .
11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 ,
𖼄𖽔𖾐 𖼞𖽪𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽠𖾏 𖼊𖽠𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .
12 𖼞𖽪𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 , 𖼐𖽱𖾏 𖼖𖽙𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏 ,
𖼊𖽔𖾑 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐 𖼮𖽡𖽝𖾏 𖽐𖼮𖽠𖾏 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼋𖽪 ,
𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼈𖽙𖾏 𖼀𖽙𖾑 .
13 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼀𖽙𖾑 𖼐𖽑𖽜𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼞𖽪𖾑 ,
𖼐𖽱𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼏𖽻𖾏 , 𖼎𖽺𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 ,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼮𖽷𖾐 𖽐𖼮𖽢𖾑𖽑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 .
 
14 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 𖼺𖽘𖾏 , 𖼎𖽺𖾏 𖽐𖼚𖽔 𖼺𖽘𖾏 ,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼋𖽻 𖽃𖽡𖾐 𖼼𖽘𖾐 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖽐𖼮𖽵 𖼎𖽻𖾏 𖼞𖽪𖾑 .
15 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼺𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽐𖼡𖽹𖾏 ,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 .
16 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼞𖽡𖾏 𖽐𖼊𖽔𖾏 ,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖼅𖽔 𖼅𖽔𖾏 𖼋𖽔 𖼑𖽡𖾏 𖼼𖽘𖾏 .
17 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 ,
𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼖𖽔𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼯𖽡𖽙𖾏.
 
18 𖼊𖽡𖾐 𖼘𖽪 ! 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼀𖽙 𖼞𖽪𖾑 𖽐𖼮𖽑𖽘𖾑 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼞𖽪𖾑 𖼐𖽡𖽝𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 ！
19 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼞𖽪𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 ,
𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽠𖾏 .
20 𖼞𖽪𖾑 𖼽𖽺 𖼗𖽔 𖼚𖽙𖾏 𖼞𖽪𖾑 ,
𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼞𖽔𖾏 𖼅𖽔 𖽐𖼚𖽱𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
21 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽠𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽡𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐 ,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙 𖼊𖽡𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 .
22 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼳𖽡𖽻𖾏 ,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼙𖽠𖾏 𖼊𖽑𖽔𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 .”
