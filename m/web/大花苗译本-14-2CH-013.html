13
𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽔𖾐 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ， 2 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 . 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼄𖽡𖾐 𖼞𖽹𖾑 𖼽𖽔𖾏 ( “𖼄𖽡𖾐 𖼞𖽹𖾑 𖼽𖽔𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼄𖽔𖾐 𖼮𖽢𖾑” ), 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽂𖽪𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 .
𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽔𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏
