𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡
14
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 , 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡:
2 “𖼽𖽨 𖼊𖽔𖾏 𖼐𖽡𖽝𖾑 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐, 𖼙𖽻 𖽃𖽔 𖼛𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾏 𖼏𖽔𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼐𖽡𖽝𖾑 𖼮𖽑𖽡𖾏,
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐.
3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖼷𖽷𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽻𖾐,
𖼁𖽙𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽻𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏,
𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼁𖽔 𖼖𖽡𖾑 𖼇𖽻𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏.
4 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼁𖽔 𖼖𖽡𖾑 𖼇𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽘 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼎𖽺𖾏 𖼋𖽠 𖼞𖽘𖾐.
5 𖽐𖼎𖽘𖾏 𖼯𖽢𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼷𖽹𖾏,
𖼨𖽑𖽳𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼷𖽹𖾏 𖼮𖽡𖽝𖾏 𖼖𖽔𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐.
6 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼮𖽢𖾏, 𖼮𖽙𖾏 𖼀𖽘𖾏 𖽐𖼷𖽑𖽔𖾏 𖼺𖽱𖾏 𖼚𖽡𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼡𖽜𖾐, 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼺𖽘𖾏. ”
𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼯𖽜𖾏 𖼾𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼀𖽡𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼛𖽠𖾏 𖼀𖽡𖾐 𖼈𖽡𖽝𖾏,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏.
𖼀𖽡𖾐 𖼟𖽪𖾏 𖼯𖽪 𖼯𖽪𖾏 𖽐𖼷𖽜𖾑 𖼟𖽡 , 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 𖼺𖽘𖾏.
8 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼘𖽪 ! 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼮𖽨𖾏 𖼮𖽵𖾑 𖽃𖽡𖾑 𖼘𖽪!
𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼺𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼊𖽔𖾐,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽪𖾐 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 𖼀𖽱𖾏 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼊𖽔𖾐 𖼖𖽺𖾑?
9 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏,
𖼺𖽱𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽺𖾑?
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼯𖽡𖽙 𖼺𖽷𖾐,
𖼀𖽡𖾐 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼀𖽡𖾐.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼑𖽻 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽘𖾐 𖼅𖽻𖾏 𖽐𖼊𖽘𖾐 𖼗𖽙 ,
𖼦𖽡𖾐 𖼯𖽵𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼖𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏.”
