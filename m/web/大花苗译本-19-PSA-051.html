𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖽐𖼎𖽻𖾏 𖼾𖽔𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙𖼀𖽔 𖼳𖽷𖾏 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏,𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽔 𖼊𖽗𖾑 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 𖼐𖽡𖾑, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
（𖼺𖽔𖾐 𖽃𖽔𖾐 12： 1 -25）
51
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑，
𖼸𖽙𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼷𖽻𖾏 𖼇𖽳𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏．
2 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼷𖽔𖾑 𖽐𖼷𖽑𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼎𖽺𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏．
 
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏,
𖼞𖽪𖾑 𖼾𖽔𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
4 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 𖼋𖽘𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼞𖽪𖾑, 𖼢𖽳 𖼋𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼢𖽳 𖼋𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
5 𖼑𖽔𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽙 𖼳𖽻𖾏 𖼗𖽙𖾏 ,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼾𖽔𖾏 𖽐𖼚𖽙 𖼺𖽜𖾏 𖼞𖽪𖾑 .
 
6 𖼑𖽔𖾏 𖼘𖽪! 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼙𖽔, 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝 ,
𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳.
7 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼐𖽨 𖼳𖽡 𖽐𖼡𖽜𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼷𖽔𖾑 𖽐𖼷𖽑𖽔𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼚𖽠𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼀𖽪𖾏.
8 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑，
𖼟𖽙𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼸𖽺𖾏 𖼖𖽙𖾑 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏.
9 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏,
𖼇𖽳𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.
 
10 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼷𖽷𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏，
𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼖𖽡𖽾 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ( “𖽐𖼮𖽤𖾐 𖽐𖼞𖽳” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼞𖽑𖽙𖾑 𖼏𖽔𖾏”).
11 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼟𖽡 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼙𖽠𖾏.
12 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼺𖽘𖾏 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 .
 
13 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑,
𖼊𖽪𖾐 𖼾𖽔𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼸𖽙𖾏 𖼟𖽡.
14 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 ,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑！
𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 .
 
15 𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾐 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏,
𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡！
16 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼙𖽔 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏,
𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼟𖽡 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼙𖽔 .
17 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 , 𖼟𖽪𖾏 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐.
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐.
 
18 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼟𖽡 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼯𖽡𖽙𖾏 𖼳𖽡 𖼣𖽔𖾑 ,
𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 .
19 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖽃𖽺𖾐 𖼙𖽔 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼊𖽑𖽡𖽝𖾐 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑
𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼙𖽱 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼖𖽑𖽘𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼏𖽘 .
