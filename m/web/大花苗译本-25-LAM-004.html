5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼑𖽻𖾏 𖼻𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖽃𖽡𖾑,

6 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖼑𖽡𖾏 𖼾𖽔𖾏,

12 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,

𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼊𖽪𖾐 𖼗𖽻 𖼟𖽪𖾏 𖼻𖽻𖾏.
 
17 𖼀𖽡𖾐 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖽐𖼀𖽘𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼺𖽘𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼡𖽑𖽙𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐.
 
18 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼖𖽠𖾑 𖼀𖽡𖾐 𖼊𖽠𖾏 𖼐𖽜𖾑 , 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏.
𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖼊𖽡𖾏 𖼈𖽳𖾏,
𖼀𖽡𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼀𖽪𖾑 𖼋𖽘𖾏,
𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏.
 
19 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼀𖽡𖾐, 𖼇𖽺𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾑 𖼚𖽘𖾑 𖼟𖽪𖾏 𖼾𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼖𖽠𖾑 𖼀𖽡𖾐,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽐𖼚𖽺𖾏 𖼋𖽙𖾏，𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼀𖽡𖾐.
20 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 , 𖼺𖽱𖾏 𖼀𖽡𖾐 𖼖𖽪𖾑 𖽃𖽔 𖽐𖼀𖽱 𖼑𖽡𖾏 𖼀𖽘𖾏,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼘𖽑𖽻𖾑 𖼊𖽻𖾏 𖼄𖽔𖾐 𖼮𖽡𖽝𖾏,
𖼀𖽡𖾐 𖼙𖽔 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏:
“𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏.”
 
21 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽂𖽪𖾑 𖼺𖽷𖾑 𖼊𖽡𖾐 𖼘𖽪！ 𖼯𖽔 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼙𖽔 𖼚𖽙𖾏.
𖼈𖽡𖽝𖾏 𖼞𖽑𖽙𖾐 𖽃𖽦𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑,
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼦𖽻𖾏 𖼟𖽙𖾏 𖼢𖽻 , 𖼸𖽙 𖼟𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏.
 
22 𖼳𖽡 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！ 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼷𖽻𖾏 𖼋𖽘𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏.
𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！ 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼗𖽙 .
