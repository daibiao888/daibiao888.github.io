𖼽𖽡𖾑 𖼖𖽡𖾏 𖼦𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽙 𖼀𖽳 𖼊𖽑𖽡𖽝𖾐 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔
（𖼀𖽳 32：1 - 37：24）
32
𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 , 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽙 𖼀𖽳 𖼑𖽔𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖼎𖽺𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐 𖼐𖽱𖾏 . 2 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼀𖽪𖾏 𖼳𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽡𖾑 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐𖼽𖽡𖾑 𖼖𖽡𖾏 𖼦𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽙 𖼀𖽳 , 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽙 𖼀𖽳 𖼾𖽨𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼻𖽻𖾏 , 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 . 3 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽙 𖼀𖽳 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼗𖽙 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼀𖽳 𖼅𖽔𖾏 𖼾𖽔𖾏 . 4 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼦𖽪𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽙 𖼀𖽳 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼎𖽺𖾏 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼘𖽑𖽙𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼾𖽨𖾏 . 5 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼦𖽪𖾏 𖼁𖽙𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼊𖽡𖾐 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 .
𖼽𖽡𖾑 𖼖𖽡𖾏 𖼦𖽪𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏
6 𖼀𖽪𖾏 𖼳𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼦𖽪𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 :
“𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 , 𖼅𖽡 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏 ,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽐𖼮𖽠𖾏 , 𖼦𖽡𖾐 𖼞𖽔𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 .
7 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 : 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼺𖽪𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼘𖽑𖽙𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼯𖽡𖽙𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 .
8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼖𖽡𖽾 ,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼐𖽱𖾏 𖼀𖽘𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 .
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 .
10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 ,
𖼞𖽪𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 .
 
11 𖼑𖽔𖾏 𖼘𖽪 ！𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖼋𖽙𖾏 𖽐𖼎𖽙𖾏 𖼅𖽡 𖼗𖽪𖾏 ,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 𖼅𖽡 𖼗𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 .
12 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏 𖼺𖽡𖽝𖾐 𖼑𖽻 𖼅𖽡 ,
𖼑𖽔𖾏 𖼘𖽪 ！ 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼳𖽠𖾏 𖼽𖽙 𖼀𖽳 ,
𖼸𖽠𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼗𖽪𖾏 .
13 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 : 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼋𖽘𖾏 ,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽠𖾏 𖼳𖽠𖾏 𖼐𖽱𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 .
14 𖼽𖽙 𖼀𖽳 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 ,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼅𖽡 𖼗𖽪𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 .
 
15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏， 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼊𖽡𖾐 ,
𖼦𖽡𖾐 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 .
16 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼊𖽡𖾐 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼋𖽙𖾏 𖽐𖼎𖽙𖾏 𖼖𖽺𖾑 ?
17 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼊𖽡𖾐 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼁𖽠𖾏 𖼗𖽪𖾏 ,
𖼞𖽪𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 .
18 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 ,
𖼞𖽪𖾑 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼞𖽪𖾑 .
19 𖼑𖽔𖾏 𖼘𖽪 , 𖼞𖽪𖾑 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼺𖽱𖾏 𖼘𖽑𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼮𖽡𖽠𖾑 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼛𖽡𖾏 𖼀𖽘𖾏 ,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼘𖽑𖽱𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼇𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 .
20 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼗𖽙𖾏 𖼳𖽦𖾑 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼊𖽡𖾐 ．
21 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽑𖽪𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼻𖽻𖾏 𖼗𖽪𖾏 𖼷𖽻𖾑 𖼚𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 .
22 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼚𖽡𖽝𖾏 ,
𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼇𖽺𖾏 .”
