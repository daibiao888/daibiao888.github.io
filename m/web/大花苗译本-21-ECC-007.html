7
𖼻𖽻𖾏 𖽐𖼷𖽷𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼚𖽜𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼘𖽑𖽘𖾑 𖼾𖽨 .
2 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽔 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
3 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼙𖽔 𖼚𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽳𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
4 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑.
𖼑𖽻 𖼊𖽪𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼑𖽻 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏
5 𖼑𖽻 𖼊𖽪𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 ,
𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼑𖽻 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏.
6 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼚𖽙𖾏,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖼡𖽘𖾐 𖼀𖽑𖽻𖾏 𖼟𖽡𖾏 𖼀𖽡 𖼎𖽺 𖽃𖽡𖾐 𖼼𖽘𖾐.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 .
7 𖼼𖽠𖾏 𖼀𖽑𖽡𖽝𖾏 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼸𖽠𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 .
8 𖼑𖽪𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼑𖽡𖾏 𖼳𖽠𖾑 𖼁𖽱𖾏,
𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏, 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏.
9 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼷𖽹𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼷𖽷𖾏 𖼐𖽡𖽙𖾐 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐.
10 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏：
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽺𖾑?
𖼟𖽡 𖼑𖽪 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
𖼀𖽻𖾐 𖼟𖽳 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏
11 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼯𖽡𖽙𖾏,
𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼺𖽱𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾑 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼮𖽔 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
13 𖼟𖽡 𖼽𖽔𖾑 𖼳𖽹𖾐 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼖𖽡𖾏 𖽐𖼞𖽑𖽻𖾏 𖽃𖽡𖾑,
