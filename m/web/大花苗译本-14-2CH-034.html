𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 . 2 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 , 𖼾𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 , 𖼦𖽡𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 .
3 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 , 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼘𖽑𖽔𖾏 𖼺𖽷𖾐 , 𖼎𖽺𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖼸𖽙 𖼋𖽘𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 , 𖼋𖽡𖽝𖾏 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼁𖽱𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼋𖽠𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 、 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐、 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 , 𖽐𖼎𖽙 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼘𖽡𖾑𖽑 . 4 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼀𖽔𖾑 𖼖𖽡 𖽃𖽔𖾑 𖼏𖽘 , 𖼀𖽑𖽔𖾏 𖼢𖽻 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐(“ 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 ” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “ 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼼𖽘𖾏 ”) , 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼞𖽳𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼘𖽡𖾑𖽑 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖽃𖽔𖾏 𖼮𖽑𖽻𖾑 𖼺𖽘𖾏 , 𖽐𖼀𖽑𖽻𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽠𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏， 5 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼺𖽘𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 . 6 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 、 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 、 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 、 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼷𖽑𖽔𖾏 𖼙𖽪𖾏 𖼙𖽻 , 𖽐𖼎𖽙 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 7 𖼎𖽺𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖼮𖽑𖽻𖾑 , 𖼀𖽑𖽔𖾏 𖼢𖽻 𖼺𖽘𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 , 𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏 .

29 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼗𖽻 𖼋𖽔𖾏 . 30 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 、 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 、 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 、𖽐𖼎𖽙 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼘𖽑𖽙𖾐 𖼳𖽻𖾐 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽠𖾑 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑 , 𖼞𖽑𖽔𖾏 𖼟𖽡𖽠 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻 . 31 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 、 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 、 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏, 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼗𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 . 32 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼎𖽙 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼐𖽡𖾑 , 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 . 33 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 , 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖼳𖽪𖾐 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 , 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖽐𖼚𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 , 𖼎𖽺𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼇𖽳𖾑 .
