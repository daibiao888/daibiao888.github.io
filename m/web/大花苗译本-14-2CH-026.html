（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐14：21 - 22；15：1-7）

19 𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 , 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼷𖽙𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 , 𖼽𖽔𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 . 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼖𖽪𖾑 𖼇𖽻𖾏 𖼐𖽑𖽡𖽝𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼋𖽠𖾏 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 . 20 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾐 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼮𖽵𖾑 𖼎𖽺𖾏 𖼑𖽔𖾏 , 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽑𖽡𖽝𖾑 𖼳𖽠𖾑 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 , 𖼎𖽺𖾏 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼋𖽠𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 , 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼎𖽺𖾏 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼐𖽱𖾏 .
21 𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼋𖽔 .𖼐𖽱𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏 𖼐𖽔𖾑 𖼮𖽡𖽺𖾏 𖼡𖽔𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼟𖽻 𖼮𖽡𖽝𖾏, 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼇𖽺𖾐 𖼡𖽔𖾏 . 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼊𖽑𖽗𖾐 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼑𖽪𖾏 , 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 .
