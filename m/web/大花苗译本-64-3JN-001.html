1 𖼞𖽪𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼗𖽻 𖼐𖽡𖾑 𖼺𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼞𖽹𖾑 𖼽𖽨 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏， 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖽐𖼚𖽙 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 ．
2 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼘𖽪！𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼖𖽔𖾏 𖽐𖼷𖽱𖾏，𖽃𖽔𖾏 𖼮𖽡𖾑 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼛𖽡𖾐 𖽐𖼷𖽱𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐． 3 𖼅𖽔𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼋𖽔𖾏， 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝， 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼑𖽪𖾏，𖼞𖽪𖾑 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝． 4 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼅𖽡 𖼸𖽙𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼑𖽪𖾏， 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼋𖽡𖾏 𖼐𖽡𖾑．
5 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼘𖽪！𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼯𖽡𖽙． 6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏，𖼷𖽑𖽜𖾑 𖼟𖽡 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔， 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏． 7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼮𖽵𖾑 𖽐𖼷𖽷𖾏 𖼋𖽠𖾏 𖼻𖽻𖾑 𖽐𖼎𖽘𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼑𖽡𖾏 𖼋𖽪 𖼡𖽔𖾏 𖼳𖽷𖾏． 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼟𖽙𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼑𖽜 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼑𖽪𖾏．
𖼊𖽨𖾑 𖼊𖽑𖽳 𖼇𖽺𖾑 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼄𖽡𖾐 𖼊𖽨𖾑
9 𖼞𖽪𖾑 𖼙𖽔 𖼺𖽻𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼣𖽔𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼊𖽨𖾑 𖼊𖽑𖽳 𖼇𖽺𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖽃𖽡𖾑，𖼦𖽡𖾐 𖼯𖽪𖾏 𖼷𖽹𖾏 𖼀𖽡𖾐． 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼞𖽪𖾑 𖼮𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 ， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏， 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼐𖽑𖽡𖾏 𖼀𖽡𖾐， 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽷𖾐，𖼐𖽱𖾏 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽹𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 ，𖼎𖽺𖾏 𖼖𖽠𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼙𖽠𖾏．
11 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼘𖽪！ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽻 𖽐𖼚𖽪𖾏， 𖼽𖽔𖾑 𖼾𖽻 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏．𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽡𖽝， 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 12 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼊𖽡𖾐 𖼄𖽡𖾐 𖼊𖽨𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏．𖼀𖽡𖾐 𖼖𖽔𖾏 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼊𖽑𖽡𖽝𖾐，𖼟𖽡 𖼀𖽻𖾐 𖼀𖽡𖾐 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝．
13 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼯𖽡𖽙𖾏 𖼊𖽔𖾑 𖽐𖼚𖽘𖾏 𖽐𖼎𖽙 𖼞𖽔𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼚𖽪𖾐 𖼺𖽻𖾏 𖽐𖼊𖽠𖾑 𖼎𖽻𖾏 𖼟𖽡． 14 𖼞𖽪𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜 𖼑𖽜 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼇𖽺𖾏， 𖼀𖽡𖾐 𖼎𖽺𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼁𖽙𖾏 𖼚𖽑𖽪𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏． 15 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏！ 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽺 𖼗𖽔 𖼑𖽪 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪．𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼸𖽙𖾏 𖽐𖼷𖽷𖾏 𖼑𖽪 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼽𖽺 𖼗𖽔 𖼋𖽡𖽝𖾏 𖼋𖽪．
