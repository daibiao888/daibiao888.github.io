𖼇𖽺𖾑 𖼖𖽡𖾏 𖼄𖽞 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏
4 𖼞𖽪𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡 𖼊𖽑𖽡𖽝𖾐， 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏，𖽐𖼎𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑，𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑． 6 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽔𖾑 𖼮𖽔𖾑𖽑 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼸𖽠𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼀𖽡𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏， 𖼟𖽪𖾏 𖼑𖽜 𖼮𖽡𖾑 𖼊𖽪 𖽃𖽔𖾏． 7 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼘𖽪！𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 ，𖼎𖽺𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖽐𖼎𖽙 𖼎𖽳𖾏 𖽐𖼚𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖽐𖼀𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝．
