23 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪、 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 、𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏， 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼚𖽔𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡． 24 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽳𖾏 𖼚𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪．
