𖼽𖽡𖾐 𖼦𖽪𖾏 𖼋𖽔 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏. 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽺 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽦𖾑 𖼀𖽡𖽾𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽢𖾏 𖼺𖽙𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼖𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽢𖾏 𖼘𖽙 𖼳𖽳 · 𖼦𖽔𖾐 𖼞𖽙𖾑 𖼽𖽡𖽾𖾏. 3 𖼽𖽦𖾑 𖼀𖽡𖽾𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽔𖾏 𖼯𖽢𖾏 𖼀𖽔𖾏 𖼗𖽪𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼖𖽑𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏
4 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑, 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼀𖽡𖾐 𖼊𖽙𖾑 𖼑𖽡𖾏 𖼐𖽤𖾐, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏. 5 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼄𖽔𖾐 𖽐𖼎𖽙 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 𖽐𖼊𖽻𖾏 𖼺𖽱𖾐 𖼺𖽱𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐． 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼑𖽪𖾏. 6 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼮𖽡𖾑 𖼊𖽡𖾐 𖼺𖽷𖾑, 𖼨𖽑𖽪𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽙 𖽃𖽗𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾑 𖼀𖽳 𖼇𖽻𖾑 𖼎𖽻𖾐. 7 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼀𖽡𖽾𖾑 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼖𖽔𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼳𖽪𖽝𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏, 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.” 8 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼮𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏, 𖼟𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏.” 9 𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽻𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼖𖽔𖾑 𖼋𖽺 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.” 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼖𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼊𖽡𖾐 𖼺𖽷𖾑 𖼙𖽠𖾏. 10 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖾑 𖼊𖽡𖾐 𖼺𖽷𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏, 𖼊𖽡𖾐 𖼀𖽙𖾑 𖼖𖽔𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏.
