𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
20
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼟𖽡,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡.
2 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼋𖽔𖾏 𖼮𖽨𖾏 𖽐𖼀𖽘𖾏 𖼟𖽡,
𖼎𖽑𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼋𖽔𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼟𖽡.
3 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼟𖽡 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑. （𖼳𖽡𖾏 𖼖𖽔𖾑）
4 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼡𖽑𖽙𖾑 𖼉𖽳 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡,
𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏.
5 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏,
𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑．
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽐𖼊𖽜.
 
6 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽨𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼐𖽱𖾏,
𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏.
7 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼻𖽡𖽝𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐, 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼻𖽡𖽝𖾏 𖼑𖽜𖾏,
𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼢𖽻 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼙𖽠𖾏,
𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼳𖽠𖾑 𖼟𖽡𖾏 𖼮𖽨𖾏 𖼀𖽡𖾏 𖼮𖽨𖾑.
 
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,
𖼀𖽡𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼀𖽡𖾐！
