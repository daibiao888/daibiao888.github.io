𖼁𖽙𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏
4
𖼷𖽷𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼮𖽘𖾏 𖼞𖽪𖾑 𖼯𖽷𖾏 𖼗𖽙𖾏, 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖼯𖽷𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐. 2 𖼐𖽱𖾏 𖼑𖽪 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏?” 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖽃𖽔𖾏, 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽱 𖼟𖽪𖾏 𖼞𖽪𖾐, 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼅𖽔𖾏 𖼋𖽺𖾏, 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼅𖽔𖾏 𖼳𖽤𖾐 𖼖𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐, 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼡𖽔𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼊𖽙𖾏 𖽐𖼚𖽙. 3 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼇𖽘𖾐 𖼞𖽗𖾐 𖼖𖽗, 𖽃𖽡𖾐 𖼇𖽘𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼋𖽺𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏, 𖽃𖽡𖾐 𖼇𖽘𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼋𖽺𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏.” 4 𖼞𖽪𖾑 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼘𖽪! 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼘𖽪?” 5 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?” 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼘𖽪! 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐.”
6 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼀𖽔𖾑 𖼀𖽳 . 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏, 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖽾, 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼊𖽡𖽝𖾑 𖼑𖽪𖾏. 7 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼘𖽪! 𖼟𖽡 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼀𖽔𖾑 𖼀𖽳 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽡 𖼖𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏. 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽜𖾐 𖼮𖽙𖾏 𖽃𖽔𖾏 𖼎𖽪𖾏 𖽐𖼞𖽔𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏:‘ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏, 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 ( “𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖽃𖽔𖾑 𖼈𖽳𖾐 ” ) 𖼐𖽡𖾑!’ 8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 9 “𖼺𖽙𖾐 𖼘𖽙 𖼀𖽔𖾑 𖼀𖽳 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖼊𖽠𖾏 𖽐𖼞𖽔𖾏, 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽔𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽡𖾏 𖼼𖽙 𖼐𖽡𖾑. 𖼟𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏. 10 𖼡𖽔 𖼋𖽱𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑? 𖼑𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼖𖽺𖾑? 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽙𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼀𖽔𖾑 𖼀𖽳 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼖𖽑𖽔𖾏 𖼞𖽡𖾏 𖼘𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏, 𖼎𖽺𖾏 𖼙𖽔 𖼚𖽙𖾏. 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽔, 𖼑𖽔𖾏 𖽐𖼷𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑.”
𖽃𖽔𖾐 𖼇𖽘𖾐 𖼞𖽗𖾐 𖼖𖽗 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑
11 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽑𖽡𖾏: “𖼖𖽪𖾑 𖽃𖽔𖾏 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼐𖽡𖾑 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 𖽃𖽡𖾑 𖽃𖽔𖾐 𖼇𖽘𖾐 𖼞𖽗𖾐 𖼖𖽗, 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽪 𖼳𖽷𖾏 𖼘𖽪?” 12 𖼞𖽪𖾑 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾐 𖼋𖽪 𖼞𖽗𖾐 𖼖𖽗 𖽃𖽔𖾑 𖼯𖽡 𖼐𖽡𖾑 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼋𖽪𖾏 𖼡𖽔𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼊𖽙𖾏 𖽐𖼚𖽙 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼞𖽪𖾐 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽪 𖼳𖽷𖾏 𖼘𖽪?” 13 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼋𖽪 𖼳𖽷𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?” 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼘𖽪 , 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐.” 14 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 , 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾏 𖼋𖽻𖾏 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼺𖽘𖾑 𖼖𖽡𖾏 𖼊𖽘𖾑.”
