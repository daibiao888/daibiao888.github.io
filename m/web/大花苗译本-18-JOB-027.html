𖼞𖽪𖾑 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼚𖽺𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 .
5 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼅𖽡 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼾𖽙 ,
𖼞𖽪𖾑 𖼸𖽙 𖼋𖽔 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏！
6 𖼞𖽪𖾑 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼺𖽘𖾏 ,
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 , 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼖𖽔𖾏 𖼞𖽪𖾑 .
 
7 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼺𖽱𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼊𖽱𖾑 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 , 𖼊𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 , 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑 ?
9 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼖𖽺𖾑 ?
10 𖼋𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼞𖽑𖽡𖽠𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽃𖽔𖾏 𖼙𖽔 𖼚𖽙𖾏 ,
𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ?
11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 ,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 , 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼈𖽺𖾏 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐 .
12 𖼅𖽡 𖼾𖽨𖾏 𖼖𖽔𖾏 𖼙𖽔 𖼁𖽙𖾏 ,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽡 𖼞𖽑𖽔𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼘𖽪 ?
13 𖽐𖼀𖽠 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼡𖽑𖽙𖾑 𖼏𖽵 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 , 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 .
14 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 ,
𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼊𖽪𖾐 𖼘𖽑𖽔𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼮𖽻𖾏 .
15 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 𖼎𖽺𖾏 𖼊𖽱𖾏 𖼺𖽘𖾏 ,
𖼐𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏 .
16 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽷𖾏 𖼮𖽡𖽝𖾏 𖼺𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐 ,
𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖼺𖽱𖾏 𖽃𖽘𖾑 𖽃𖽔 𖼗𖽔𖾏 .
17 𖼐𖽱𖾏 𖼮𖽔 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽑𖽘𖾑 ,
𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖼇𖽺𖾐 .
18 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼞𖽘𖾐 𖼳𖽡𖽝𖾐 𖼖𖽱𖾑 ,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 ．
19 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 、 𖼻𖽻𖾏 𖼀𖽱𖾏, 𖼈𖽡𖽝𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐,
𖼐𖽱𖾏 𖼡𖽑𖽜𖾐 𖼀𖽑𖽙𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼋𖽘 .
20 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽠𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 ,
𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖽙𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼷𖽑𖽔𖾐 𖼙𖽠𖾏 .
21 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼮𖽢𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼷𖽑𖽔𖾐 𖼽𖽘𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖼎𖽺𖾏 𖼷𖽑𖽔𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼇𖽳𖾑 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 ．
22 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼈𖽙𖾏 𖼀𖽙𖾑 𖼐𖽱𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖽐𖼞𖽡 𖼳𖽡𖾐 ,
𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 .
23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖽐𖼀𖽔𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼊𖽑𖽡𖾏 𖼖𖽔𖾏 ,
