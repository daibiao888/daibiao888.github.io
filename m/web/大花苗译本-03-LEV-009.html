𖼸𖽙 𖼾𖽡𖾏 𖼐𖽑𖽪𖾐 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼨𖽑𖽪𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 , 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼗𖽻 𖼋𖽔𖾏 , 2 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼮𖽤𖾐 𖽐𖼀𖽙𖾏 𖼑𖽨𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏 , 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 . 3 𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽤𖾐 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 , 𖼎𖽺𖾏 𖼮𖽤𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼋𖽪 𖼾𖽘𖾏 𖼁𖽜𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏 𖼯𖽡𖽙𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑， 4 𖼎𖽺𖾏 𖼮𖽤𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 , 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼛𖽙𖾏 , 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 .” 5 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 , 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 . 6 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 .” 7 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼅𖽻𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖼖𖽑𖽘𖾑 𖼟𖽡 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼑𖽜 𖼾𖽨𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 , 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 .”
8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖼊𖽔𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 . 9 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼮𖽑𖽳𖾑 𖽐𖼮𖽑𖽘𖾑 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼚𖽻𖾏 𖼎𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 , 𖼚𖽜𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽻𖾐 𖼋𖽪 𖼞𖽪𖾑 𖼞𖽪𖾐 , 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼮𖽑𖽘𖾑 𖼘𖽑𖽡𖾑 𖼎𖽻𖾏 𖼊𖽠𖾏 𖽃𖽔 𖼏𖽘 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 . 10 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼛𖽙𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼏𖽻𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼛𖽙𖾏 𖼀𖽙𖾏 𖼞𖽑𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘,𖼟𖽪𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏. 11 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖽐𖼡𖽹𖾏 𖽐𖼎𖽙 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 .
