28
𖽃𖽺𖾐! 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼢𖽻 𖼮𖽡𖽠𖾑 𖽐𖼊𖽻𖾏 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼛𖽙 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼐𖽱𖾏 𖼺𖽱𖾏 𖼁𖽘𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼀𖽑𖽙𖾐 𖼋𖽘 ,
𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼢𖽻 𖼮𖽡𖽠𖾑 .
2 𖼑𖽔𖾏 𖼘𖽪！ 𖼮𖽵𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼅𖽔𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼺𖽱𖾏 𖽃𖽡𖾐 𖼼𖽙 𖼑𖽘 𖽃𖽔𖾐 𖼖𖽪𖾐, 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑,
𖼺𖽱𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼗𖽔𖾏 𖽐𖼮𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
3 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼢𖽻 𖼮𖽡𖽠𖾑 𖽐𖼊𖽻𖾏 𖽃𖽡𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼯𖽪𖾏 𖼎𖽻𖾏 𖼡𖽘𖾐 𖼊𖽠𖾏.
4 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼛𖽙 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼺𖽱𖾏 𖼁𖽘𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼀𖽑𖽙𖾐 𖼋𖽘 ,
𖼺𖽱𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐,
𖼦𖽡𖾐 𖼸𖽙 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼺𖽷𖾐 𖼷𖽹𖾏 𖼺𖽪𖾐 𖼳𖽦𖾑, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖽙𖾏 𖼑𖽔𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼐𖽡𖾑 ,
𖽂𖽪𖾏 𖼚𖽡𖾏 𖼷𖽠𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
𖼖𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽪𖾐 𖼮𖽡𖽙𖾐 𖼨𖽑𖽳𖾏
5 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏、 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏，
6 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽃𖽡𖾑 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑,
𖼎𖽺𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼊𖽻𖾏 𖼟𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼙𖽠𖾏.
7 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼎𖽺𖾏 𖼑𖽜 𖼮𖽡𖽠𖾑 𖼟𖽡𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽐𖼊𖽙𖾏, 𖼑𖽜 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐 𖼢𖽻 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼐𖽡𖾑．
𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼺𖽪𖾐 𖼁𖽙𖾏 𖼑𖽜 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐 𖼟𖽡𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽐𖼊𖽙𖾏,
𖼊𖽻𖾏 𖼮𖽡𖽠𖾑 𖽐𖼊𖽠𖾏 𖼮𖽡𖽝𖾏,
𖼑𖽜 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐 𖼢𖽻 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼐𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐,
𖼖𖽡𖾏 𖼅𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐.
 
8 𖼷𖽑𖽔𖾏 𖼸𖽱𖾏 𖼞𖽡 𖼏𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼀𖽪𖾑 𖼋𖽘𖾏 𖽐𖼊𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏.
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼚𖽙𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:
“𖼐𖽱𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼽𖽔𖾑 𖼟𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼘𖽪?
𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼇𖽳𖾑 𖼄𖽡𖾏、 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼇𖽳𖾑 𖼑𖽡𖽝𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
10 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼊𖽑𖽡𖽝𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼊𖽑𖽡𖽝𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑,
𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏, 𖼡𖽑𖽙𖾑 𖼉𖽺𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏.”
 
11 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏: 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙, 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏
𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 .
12 𖼐𖽱𖾏 𖼙𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏,
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼚𖽑𖽔𖾏 𖼊𖽻𖾏 𖼳𖽙𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏．”
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻.
13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏
𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼊𖽑𖽡𖽝𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 , 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼊𖽑𖽡𖽝𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑,
𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏, 𖼡𖽑𖽙𖾑 𖼉𖽺𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏,
𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼈𖽡𖽝𖾏 𖼞𖽡𖾐 𖼮𖽑𖽔𖾐 𖼢𖽻 𖼙𖽠𖾏, 𖼢𖽻 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏,
𖼎𖽺𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼊𖽻𖾏 𖼦𖽡𖾏 𖼚𖽱𖾏 𖼮𖽡𖽝𖾏.
𖼮𖽙𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼎𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑
14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼅𖽡 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏,
𖼟𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼐𖽡𖾑,
𖼽𖽔𖾑 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏.
15 𖼅𖽡 𖼙𖽔 𖼐𖽑𖽡𖾏:
“𖼀𖽡𖾐 𖽐𖼎𖽙 𖼛𖽙𖾏 𖼋𖽔 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖽐𖼎𖽙 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏,
𖼊𖽪𖾑 𖽐𖼚𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏,
𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼈𖽺𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑.”
16 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:
“𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼮𖽙𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽔𖾏 𖼞𖽪𖾐 𖽐𖼞𖽔𖾐 𖼊𖽠𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐,
𖼟𖽪𖾏 𖽃𖽔𖾑 𖼈𖽳𖾏 𖼟𖽪𖾏 𖼙𖽔 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏,
𖼟𖽪𖾏 𖼊𖽠𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏, 𖼞𖽪𖾐 𖽐𖼞𖽔𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽳𖾏.
17 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖽃𖽔𖾏 𖼖𖽑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 ,
𖼯𖽡𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽔𖾏 𖼖𖽑𖽔𖾏 𖼞𖽡𖾏 𖼘𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏.
𖼑𖽘 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽱𖾑𖽑 𖼺𖽘𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏,
𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽪𖾑 𖼮𖽑𖽡𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑.
18 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼛𖽙𖾏 𖼋𖽔𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼊𖽪𖾑 𖽐𖼚𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼏𖽻 𖼐𖽱𖾏 𖼯𖽪𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
19 𖼷𖽑𖽔𖾏 𖼟𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼅𖽡 𖼦𖽔𖾏 𖼙𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼊𖽡𖾏 𖼟𖽡𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏,
𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑.
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏.”
20 𖼸𖽱𖾏 𖼸𖽘𖾏 𖼖𖽪𖾑, 𖼦𖽡𖾐 𖼷𖽻𖾏 𖼳𖽤𖾐 𖼊𖽠𖾏,
𖼡𖽔𖾑 𖼘𖽑𖽱𖾏 𖽐𖼡𖽹𖾏, 𖼦𖽡𖾐 𖼷𖽻𖾏 𖼈𖽳𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 .
21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏, 𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖼀𖽑𖽡𖾐 𖼖𖽔𖾑 𖼳𖽡𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐,
𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼀𖽡𖽝𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 ,
𖼑𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾏,
𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑.
22 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽡 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼊𖽠𖾏 𖼅𖽡 𖽂𖽪𖾏 𖽂𖽪𖾏 𖽐𖼊𖽠𖾏 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼮𖽵𖾑 — 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏,
𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼊𖽪𖾏 𖼮𖽘𖾑 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏.
 
23 𖼅𖽡 𖼽𖽔𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏,
𖼳𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 .
24 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼀𖽳𖾐 𖼮𖽘𖾑 𖼐𖽔𖾑 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑,
𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖼊𖽔𖾐?
𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽠𖾏 𖼊𖽡𖾐 𖼨𖽑𖽹𖾑 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼖𖽺𖾑?
25 𖼐𖽱𖾏 𖼨𖽑𖽹𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 𖼺𖽘𖾏,
𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽔𖾑 𖼀𖽳𖾐 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼳𖽳𖾏,
𖼀𖽳𖾐 𖼙𖽘𖾑 𖼙𖽘𖾏 𖽃𖽔 𖼸𖽔 𖼳𖽳𖾏, 𖼸𖽙𖾏 𖼡𖽔𖾏 𖽐𖼞𖽑𖽳𖾏 𖼯𖽡𖽙 𖼳𖽻𖾐,
𖼐𖽡𖽙𖾐 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙 𖼅𖽙 ,
𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼙𖽡𖽝𖾏 𖼯𖽡𖽙 𖼅𖽙 𖽐𖼷𖽑𖽷𖾐.
26 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 ,
𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏.
 
27 𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼳𖽳𖾏, 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼞𖽡𖾏 𖼮𖽢𖾑 𖼟𖽪𖾏 𖽐𖼮𖽳𖾏 ,
𖼖𖽻𖾐 𖼙𖽘𖾑 𖼙𖽘𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼳𖽳𖾏, 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼀𖽡𖾑 𖼘𖽙𖾐 𖼖𖽻𖾐 ,
𖼈𖽡𖽝𖾏 𖼯𖽡𖽙𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏 𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼳𖽳𖾏,
𖼯𖽡𖽙𖾏 𖼡𖽠𖾏 𖽐𖼊𖽻𖾏 𖼙𖽘𖾑 𖼙𖽘𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼳𖽳𖾏.
28 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼢𖽪𖾏 𖼡𖽜𖾑 𖼢𖽪𖾏 𖼟𖽙𖾏 𖼅𖽙𖾏,
