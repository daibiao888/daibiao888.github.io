15 𖼑𖽻𖾏 𖼮𖽑𖽹𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼄𖽞 · 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“ 𖼽𖽙 𖼦𖽗 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼄𖽞 ，𖼋𖽡𖽝𖾏 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼋𖽡𖾏 𖼐𖽡𖾑？” 𖼀𖽡𖾐 𖼊𖽳 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼾𖽙 𖼯𖽡𖽙𖾏，𖼟𖽡 𖼀𖽻𖾐 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼊𖽪𖾑 𖼞𖽪𖾑 𖼾𖽘𖾏 𖼁𖽜𖾏．” 16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼽𖽙 𖼦𖽗 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼄𖽞， 𖼋𖽡𖽝𖾏 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑？” 𖼀𖽡𖾐 𖼊𖽳 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼾𖽙 𖼯𖽡𖽙𖾏，𖼟𖽡 𖼀𖽻𖾐 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼊𖽪𖾑 𖼾𖽨 𖼞𖽪𖾑 𖼾𖽘𖾏．” 17 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼽𖽙 𖼦𖽗 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼄𖽞， 𖼋𖽡𖽝𖾏 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑？” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼷𖽷𖾐 𖼟𖽻𖾐 “ 𖼋𖽡𖽝𖾏 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑？”， 𖼀𖽡𖾐 𖼊𖽳 𖼎𖽺𖾏 𖼊𖽳𖾏，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼀𖽻𖾐，𖼟𖽡 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼊𖽪𖾑 𖼞𖽪𖾑 𖼾𖽘𖾏． 18 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼺𖽷𖾐，𖼟𖽡 𖼾𖽨𖾏 𖼳𖽦𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏，𖼯𖽔𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼗𖽻𖾏 𖼋𖽘𖾏，𖼟𖽡 𖼽𖽔𖾑 𖼳𖽤𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼳𖽦𖾏 𖼟𖽡 ，𖼨𖽑𖽳𖾏 𖼟𖽡 𖽐𖼮𖽡𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼮𖽡𖽻𖾏．” 19 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐 𖼊𖽳 𖼽𖽔𖾑 𖼋𖽔 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼊𖽳𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐！”
20 𖼀𖽡𖾐 𖼊𖽳 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙，𖼎𖽺𖾏 𖼁𖽙𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼸𖽙𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼋𖽔𖾏，𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼄𖽑𖽙𖾏 ，𖽃𖽡𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐， 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪 ！ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽺 𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？” 𖽃𖽡𖾑． 21 𖼀𖽡𖾐 𖼊𖽳 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏？” 22 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽔𖾏， 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏？ 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐！” 23 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾐 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐， 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼋𖽔．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼋𖽔，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽑𖽡𖾏 ：“𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽔𖾏 ，𖽐𖼎𖽙 𖼟𖽡 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏？”
24 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏，𖼎𖽺𖾏 𖼺𖽻𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑，𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏，𖼀𖽡𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏． 25 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼺𖽷𖾐，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽡𖾐 𖼼𖽘𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏，𖼞𖽪𖾑 𖽐𖼊𖽱𖾏，𖼋𖽡𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏， 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼊𖽡𖾏 𖼋𖽘 𖼦𖽡𖾐 𖼋𖽻𖾏．
