𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼊𖽻𖾏
6
𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪！ 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝. 2 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾐 𖼋𖽪 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖼐𖽡𖽙𖾐.” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼅𖽻𖾏！” 3 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽜𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏.” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼯𖽪𖾏 𖽐𖼎𖽙 𖼅𖽻𖾏.” 4 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏． 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏, 𖼎𖽺𖾏 𖽐𖼊𖽙𖾑 𖽐𖼊𖽻𖾏. 5 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽙𖾑 𖽐𖼊𖽻𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖽃𖽻𖾐 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏， 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪！ 𖽃𖽔𖾑 𖼊𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼗𖽙𖾏 𖼊𖽔𖾐.” 6 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼀𖽻𖾐 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼙𖽠𖾏?” 𖼐𖽱𖾏 𖼡𖽑𖽳𖾏 𖼡𖽑𖽙𖾑 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼑𖽔𖾏, 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖽐𖼊𖽙𖾑 𖼊𖽻𖾏 𖼇𖽔 𖽐𖼊𖽻𖾏, 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽃𖽻𖾐 𖽐𖼚𖽙, 𖽃𖽔𖾑 𖼊𖽻𖾏 𖼎𖽺𖾏 𖽐𖼊𖽘𖾐 𖼗𖽙𖾏 𖼋𖽘𖾏. 7 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 !” 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏.
𖼽𖽔𖾏 𖼖𖽔 𖼏𖽙 𖽐𖼊𖽻𖾏 𖼦𖽡𖾐 𖼳𖽠𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐
8 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼮𖽪𖾐 𖼾𖽡. 9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼡𖽑𖽙𖾑 𖼉𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏.” 10 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼟𖽙𖾏 𖼦𖽡𖾐 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖽃𖽡𖾐 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼊𖽔𖾐. 11 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼋𖽔𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼀𖽘𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼅𖽡 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖼖𖽔𖾏?” 12 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏, 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑, 𖼄𖽔𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼀𖽱𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.” 13 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏, 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏.” 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪!
