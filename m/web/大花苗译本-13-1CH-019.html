19
𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽔 𖼳𖽢 𖼋𖽔 𖼙𖽠𖾏 , 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 2 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼦𖽔𖾐 𖼐𖽺𖾏 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼐𖽔 𖼳𖽢 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼦𖽔𖾐 𖼐𖽺𖾏 .” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼋𖽔 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼐𖽱𖾏 .𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼁𖽙𖾏 𖼦𖽔𖾐 𖼐𖽺𖾏 , 𖼽𖽔𖾑 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼐𖽱𖾏 . 3 𖼈𖽡𖽝𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖽐𖼎𖽙 𖼦𖽔𖾐 𖼐𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼟𖽡 , 𖼟𖽡 𖽐𖼊𖽱𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼖𖽡𖾏 𖼘𖽪𖾏 ? 𖼐𖽱𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 , 𖽐𖼊𖽻𖾏 𖼢𖽻 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼘𖽪 ?” 4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽔𖾐 𖼐𖽺𖾏 𖼄𖽔𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖼮𖽡𖽝𖾏 , 𖼮𖽑𖽪𖾑 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼚𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼮𖽷𖾐 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾐 𖼘𖽑𖽠𖾏 𖼽𖽘𖾑 𖼀𖽡𖾐 𖼊𖽡𖾐 , 𖼟𖽙𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼽𖽘𖾑 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼗𖽙 , 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼯𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏 . 5 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 , 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼸𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 ,𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼚𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼊𖽡𖾑 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼎𖽺𖾏 𖼗𖽙𖾏 .”
𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼽𖽔𖾏 𖼄𖽞 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽔
