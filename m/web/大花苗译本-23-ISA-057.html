𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼛𖽙𖾏 𖼋𖽔𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖼙𖽠𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼳𖽻𖾏 𖼙𖽠𖾏 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
2 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪, 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 (“𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼸𖽘𖾏”) 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼳𖽙𖾏.
3 𖼅𖽡，𖽃𖽔 𖼁𖽙𖾏 𖼸𖽷𖾏 𖼅𖽪𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼘𖽪！
𖽃𖽔𖾑 𖼾𖽠 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼑𖽡𖾏 𖼮𖽘𖾑 ,
𖼽𖽔𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏！
4 𖼅𖽡 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏？
𖼊𖽱𖾐 𖼎𖽻𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼘𖽑𖽡𖽝𖾑 𖽃𖽔𖾑 𖽐𖼚𖽺𖾏 𖼖𖽺𖾑?
𖼅𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼑𖽻 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼮𖽘𖾑 𖼖𖽺𖾑?
5 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽙𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾐 𖼄𖽡𖾐 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽐𖼀𖽜𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖼊𖽠𖾏 𖽐𖼎𖽙𖾐,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼦𖽘𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐、𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼈𖽳𖾐 𖼡𖽑𖽙𖾏 𖼷𖽔𖾏 𖽐𖼚𖽙 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏.
6 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝 𖼡𖽔𖾏 𖽐𖼚𖽱𖾏 𖼅𖽔𖾏 𖼟𖽡 𖽐𖼀𖽠．
𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼡𖽪𖾏 𖼊𖽻𖾏,
𖼟𖽡 𖼖𖽑𖽘𖾑 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼖𖽑𖽘𖾑 𖼺𖽘𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑,
𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽐𖼷𖽜𖾏 𖼯𖽔𖾏 𖼋𖽻 𖼖𖽺𖾑?
7 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼊𖽜𖾐 𖼮𖽡𖽝𖾏 𖼸𖽘𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 .
8 𖼟𖽡 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖽐𖼎𖽙 𖼀𖽔𖾐 𖽃𖽔 𖼛𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼷𖽑𖽙𖾏 𖼮𖽘𖾏 𖼮𖽡𖽝𖾏 𖼟𖽡 𖼊𖽪𖾑 𖼺𖽻𖾏 , 𖼟𖽡 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏 ,
𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏 𖼟𖽡𖾏 𖼋𖽠𖾏 𖽐𖼡𖽹𖾏 𖼙𖽪 , 𖽐𖼮𖽡𖾏 𖼛𖽔𖾏 𖼷𖽱𖾏 𖼸𖽘𖾏 𖼎𖽺𖾏 𖼎𖽳𖾏 𖼇𖽘𖾑 𖼟𖽡 𖼸𖽱𖾏 𖼸𖽘𖾏,
𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽱𖾏 𖼸𖽘𖾏 ,𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼊𖽡𖾐.
9 𖼟𖽡 𖼄𖽔𖾐 𖼛𖽙𖾏 𖼯𖽡𖽙𖾏 𖼸𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 ,
𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼙𖽠𖾏,
𖼸𖽙 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏，
10 𖼟𖽡 𖼑𖽜 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼚𖽡𖾐 𖼐𖽔𖾑 𖼎𖽺𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏：
“𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙”！𖼟𖽡 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏.
 
11 𖼟𖽡 𖽐𖼮𖽑𖽹𖾏 𖼡𖽔 𖼋𖽱𖾏 ？ 𖼑𖽜 𖼡𖽔 𖼋𖽱𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑？
𖼦𖽡𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽺𖾑?
𖼊𖽡𖽝𖾏 𖼾𖽙 𖼞𖽪𖾑 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼐𖽔𖾑 ,
𖼟𖽡 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 𖼖𖽔𖾏?
12 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽳𖾏 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽐𖼎𖽙 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
13 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽝𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑,
𖼮𖽙𖾏 𖼟𖽙𖾏 𖼋𖽡𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼦𖽡 𖼸𖽱 𖽃𖽡𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼟𖽡 .
𖼮𖽢𖾏 𖼽𖽔𖾑 𖼷𖽑𖽔𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾐 𖼖𖽙𖾏 𖼀𖽘𖾏 𖼊𖽔𖾐 𖼷𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽠𖾏.
𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏.
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼐𖽡𖽙𖾐
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼽𖽔𖾑 𖼨𖽑𖽢𖾏 𖼮𖽪𖾐 𖼨𖽑𖽢𖾏 𖼮𖽪𖾐, 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑,
𖼄𖽔𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽠𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏.”
15 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐, 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼷𖽷𖾏 𖼐𖽡𖽙𖾐,
𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼉𖽺𖾏 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐, 𖼳𖽻𖾐 𖼢𖽜 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐,
𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼯𖽷𖾏 𖼳𖽠𖾑,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼯𖽷𖾏 𖼨𖽑𖽪𖾑.
16 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼷𖽷𖾏 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔．
17 𖼑𖽜 𖼐𖽱𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼦𖽠𖾑 𖼉𖽳 𖼗𖽔 𖼁𖽡𖽝 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 ,
𖼞𖽪𖾑 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏,
𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 .
18 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽻𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼐𖽱𖾏 ,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏
𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼊𖽻𖾏 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏.
19 𖼞𖽪𖾑 𖼷𖽷𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼷𖽷𖾏 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑,
𖼋𖽡𖽝𖾏 𖼋𖽪， 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐,
𖼎𖽺𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼈𖽳𖾏,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼐𖽱𖾏．
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
20 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼮𖽦 ,
𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖽃𖽻𖾐 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖽐𖼎𖽙 𖽃𖽘𖾑 𖽐𖼞𖽙𖾏 𖼗𖽙𖾏 .
21 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽪𖾐 𖽐𖼚𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 .”
