5 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼸𖽺𖾏 𖼯𖽵𖾏 ,
𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐,

“𖼟𖽡 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼮𖽑𖽙𖾏 𖼡𖽘𖾐 𖽐𖼊𖽡𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼉𖽻 𖼷𖽻𖾏,

7 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽑𖽙𖾏 𖼅𖽡 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼡𖽜𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑,

8 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼦𖽡 𖼋𖽙 ,

𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .

𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼾𖽨𖾏 𖼟𖽡𖾏 𖼾𖽨𖾏.

12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏,

15 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏,
