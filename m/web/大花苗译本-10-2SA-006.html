𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼷𖽷𖾐 𖼉𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼺𖽹𖾑. 2 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 · 𖼽𖽨 𖼊𖽔𖾏 𖼄𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼨𖽑𖽹𖾑 𖼗𖽙𖾏, 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼐𖽡𖾑 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼊𖽡𖾏 𖼯𖽡𖽙𖾏. 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽱𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽔 𖼊𖽔 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐, 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽔 𖼊𖽔 𖼑𖽡𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽂𖽪𖾑 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼽𖽙 𖼖𖽠𖾑 𖼖𖽪𖾑 𖼀𖽑𖽪𖾐 𖼼𖽙𖾐 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼐𖽡𖾑. 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽱𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼐𖽔 𖼊𖽔 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼽𖽙 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏. 5 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼯𖽡𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼢𖽜𖾏 𖽐𖼎𖽔 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽑𖽪𖾑 𖽃𖽔𖾏 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼚𖽑𖽡𖽝𖾏 𖼙𖽔. 6 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼐𖽔 𖼞𖽹𖾑 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼑𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼷𖽠𖾐 𖼛𖽔𖾏 ( 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼏𖽘𖾏 𖽐𖼎𖽘 𖼚𖽑𖽡𖽝𖾏 𖼚𖽜𖾑𖽑 ” ) 𖽂𖽪𖾑 𖼺𖽔𖾐 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼊𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏. 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖽂𖽪𖾑 𖼺𖽔𖾐, 𖼑𖽜 𖼾𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼋𖽔 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪. 8 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽔𖾏( “𖼊𖽔𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼮𖽵 𖼊𖽔𖾏” ) 𖽂𖽪𖾑 𖼺𖽔𖾐 𖼺𖽘𖾏 𖽃𖽺𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼎𖽺𖾏 𖼊𖽡𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼀𖽑𖽡𖾐 𖼖𖽡𖽝 𖼺𖽷𖾑 · 𖽂𖽪𖾑 𖼺𖽔𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏. 9 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼨𖽑𖽹𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼖𖽺𖾑?” 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼨𖽑𖽹𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻, 𖼈𖽡𖽝𖾏 𖼎𖽺𖾏 𖼨𖽑𖽹𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽳 𖼀𖽡𖽝 · 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏. 11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽳 𖼀𖽡𖽝 · 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖽐𖼞𖽔𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼖𖽑𖽡𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖽃𖽳 𖼀𖽡𖽝 · 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜.
𖼖𖽔𖾑 𖼼𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻
