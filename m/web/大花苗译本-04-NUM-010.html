𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼇𖽳𖾑 𖼳𖽡𖾑 𖼐𖽺𖾐

33 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼞𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 , 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 . 34 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽹𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 , 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 . 35 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏:
“ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 ！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼚𖽻𖾐 𖼺𖽘𖾏 ！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼏𖽘 𖼀𖽜𖾑 ！”
36 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 :
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪， 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽡𖽝𖾐 𖼉𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 ！”
