𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖽐𖼮𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏
8
𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 : 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼮𖽡𖽺𖾏 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑. 2 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾑 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼘𖽪 ! 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽔𖾑 𖼳𖽷𖾏?” 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼮𖽡𖽺𖾏 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑.” 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 (“𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏”𖽐𖼎𖽙 “𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖽃𖽔 𖼙𖽻 𖼦𖽡𖾏 𖼺𖽱𖾏 ) 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.”
3 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼡𖽻𖾑 𖼋𖽔 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏.”
𖼖𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏
4 𖼅𖽡 𖼟𖽪𖾏 𖽐𖼮𖽡𖾏 𖼯𖽪𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏,
𖼎𖽺𖾏 𖼖𖽻𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖽃𖽡𖾑,
𖼽𖽔𖾑 𖼑𖽻 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 .
5 𖼅𖽡 𖼐𖽑𖽡𖾏: “ 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼽𖽔𖾑 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏,
𖼟𖽙𖾏 𖼀𖽡𖾐 𖼻𖽻𖾏 𖼋𖽺 𖼡𖽻𖾐,
𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼽𖽔𖾑 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏,
𖼟𖽙𖾏 𖼀𖽡𖾐 𖼻𖽻𖾏 𖼊𖽑𖽔𖾑 𖼡𖽻𖾐 𖼋𖽺,
𖼋𖽺 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼳𖽜𖾐 𖼊𖽱𖾏,
𖼳𖽻𖾏 𖼑𖽡𖽝𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼎𖽑𖽻𖾏 𖼸𖽱𖾏 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐.
6 𖼻𖽻𖾏 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏,
𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖼞𖽑𖽻𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏,
𖼄𖽔𖾐 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑 𖼋𖽺 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.”
 
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼏𖽵 𖼗𖽪𖾏 , 𖼐𖽑𖽡𖾏:
“𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼸𖽪𖾏 𖼼𖽘𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏.
8 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼖𖽔𖾑 𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏?
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏 𖽐𖼀𖽻𖾏 𖼘𖽑𖽙𖾐,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼀𖽪𖾑 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏 𖽐𖼡𖽜𖾏 𖼙𖽠𖾏.”
9 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼐𖽡𖽙𖾐 𖼋𖽘 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 𖼙𖽠𖾏,
𖼟𖽙𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏.
10 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼅𖽡 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏,
𖼅𖽡 𖽐𖼞𖽻𖾏 𖼨𖽑𖽪𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽔𖾑 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐,
𖼮𖽑𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼮𖽢𖾏,
𖼟𖽙𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏 𖼐𖽡𖾑 𖼺𖽱𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏 𖼾𖽨𖾏 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼊𖽪𖾐,
𖼎𖽑𖽻𖾏 𖼁𖽱𖾏 𖼸𖽙 𖼺𖽻𖾏 𖼺𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.”
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼮𖽵𖾑 𖼗𖽪𖾏
11 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽹𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼳𖽡𖽝𖾐,
𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽻𖾐,
𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 .
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼐𖽡𖾑 𖽐𖼊𖽘𖾐 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽡𖾑,
𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐,
𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 ,
𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏.
 
13 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼟𖽡𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 𖼺𖽘𖾏.
14 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:
‘ 𖼊𖽗𖾏 𖼘𖽪 ! 𖼀𖽡𖾐 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖽃𖽡𖾑 𖼏𖽵 𖼗𖽪𖾏 .’
𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:‘𖼀𖽡𖾐 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼏𖽵 𖼗𖽪𖾏 .’
𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽻 𖼙𖽠𖾏, 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 .”
