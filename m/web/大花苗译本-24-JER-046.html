𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼢𖽻 𖼺𖽘𖾏 𖼋𖽘𖾏,

25 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽙 𖼑𖽡𖾏 𖼽𖽔𖾏 𖼄𖽞 (𖼣𖽹 𖼮𖽡 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐) 𖽐𖼎𖽙 𖼇𖽔 𖼘𖽻𖾐 , 𖼊𖽑𖽡𖽝𖾐 𖼣𖽹 𖼮𖽡 𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼳𖽺, 𖼊𖽑𖽡𖽝𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 , 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑. 26 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼣𖽹 𖼮𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐, 𖽐𖼎𖽙 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾏.” 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
