𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏
11
𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼡𖽻𖾐 𖼡𖽜𖾏 𖼨𖽑𖽢𖾏 𖼎𖽻𖾏 𖽃𖽻𖾐 𖼦𖽡𖾐 𖼇𖽔𖾑，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼗𖽡𖾏，𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏.
2 𖼟𖽡 𖼽𖽔𖾑 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼾𖽡𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐.
3 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼮𖽔𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼑𖽘 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼘𖽑𖽡𖾑 𖽐𖼮𖽢𖾑𖽑 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏．
𖽐𖼊𖽻𖾏 𖼮𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼢𖽻 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏,
𖽐𖼊𖽻𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏, 𖼎𖽺𖾏 𖼊𖽘𖾑 𖽐𖼎𖽙𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏.
4 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼮𖽢𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽳𖾐 𖼮𖽘𖾑 ,
𖼑𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼤𖽜𖾏.
5 𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽢𖾏 𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖼸𖽙𖾐 𖼞𖽡𖾑 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼋𖽔𖾏、 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼘𖽑𖽙𖾐 𖼊𖽡𖽝𖾑；𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏, 𖼟𖽡 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏.
6 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼮𖽘𖾑 , 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼳𖽙𖾏 𖽐𖼎𖽙𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼋𖽱 𖼖𖽔𖾑 𖼽𖽔𖾑 𖼘𖽑𖽙𖾐 𖼻𖽻𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼀𖽳𖾐, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼀𖽳𖾐, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾐 𖼮𖽘𖾏 𖼞𖽑𖽡𖽝𖾐 𖼖𖽔𖾏 𖼻𖽻𖾏．
7 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖽃𖽔𖾑 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 .
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼳𖽡𖽻𖾏, 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼀𖽡𖾑 𖼯𖽠 𖼳𖽡𖽻𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽙𖾏 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼮𖽡𖽻𖾐, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼽𖽔𖾑 𖼗𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 .
𖼘𖽑𖽔𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼳𖽹𖾐 𖼑𖽪 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏
9 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼘𖽪！ 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽡𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖼐𖽡𖽙𖾐 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽔𖾏 𖽃𖽡𖾑, 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽱, 𖽃𖽔𖾑 𖼅𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏； 𖼈𖽡𖽝𖾏 𖼟𖽡𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐, 𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼳𖽹𖾐 𖼑𖽪 𖼟𖽡 .
10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏 𖽐𖼊𖽱𖾏 𖼊𖽳𖾏, 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 .
