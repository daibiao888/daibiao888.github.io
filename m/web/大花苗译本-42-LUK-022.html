𖽐𖼎𖽑𖽦𖾑 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼐𖽑𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 26：1-5；𖼞𖽑𖽙𖾐 14：1-2；𖼽𖽙 11：45-53）
22
𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 ，𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏． 2 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏，𖽐𖼊𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽑𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏．
𖼽𖽨 𖼊𖽔𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑
3 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼺𖽔𖾐 𖼊𖽔 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼮𖽢𖾑 𖼖𖽡𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐，𖼐𖽱𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐． 4 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼊𖽑𖽡𖽝𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 ，𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏． 6 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏， 𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏，𖼽𖽔𖾑 𖼷𖽪𖾑 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 ，𖼄𖽔𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏
（𖼊𖽑𖽹𖾏 26：17-25；𖼞𖽑𖽙𖾐 14：12-21；𖼽𖽙 13：21-30）
7 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏，𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽔𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑 ． 8 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼷𖽷𖾐 𖼀𖽡𖾐 𖼊𖽳 、𖼽𖽙 𖼦𖽗 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼅𖽻𖾏 𖼑𖽜 𖼀𖽡𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼑𖽻𖾏．” 9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏：“ 𖼽𖽔𖾑 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏？” 10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼚𖽙 𖼙𖽻， 𖼽𖽔𖾑 𖽐𖼮𖽷𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽳𖾑 𖽃𖽡𖾐 𖼣𖽑𖽪𖾐 𖽃𖽻𖾐 ，𖼅𖽡 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖽐𖼮𖽡𖽻𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏， 11 𖼎𖽺𖾏 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏：‘𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼑𖽪 𖽐𖼞𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 ？𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼑𖽻𖾏 𖼡𖽜𖾑 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏．’ 12 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼊𖽑𖽘𖾐， 𖼮𖽑𖽤𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼊𖽜𖾐 𖼻𖽻𖾏 𖼋𖽘𖾏，𖼅𖽡 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏．” 13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼅𖽻𖾏 𖽐𖼮𖽷𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼡𖽜𖾑 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏．

35 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼅𖽡 𖼋𖽠𖾏 𖼅𖽻𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼐𖽑𖽘𖾐 𖼑𖽡𖽝𖾏 ，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏，𖼋𖽡𖽝𖾏 𖼅𖽡 𖼯𖽠 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼺𖽷𖾐 ？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “𖼦𖽡𖾐 𖼁𖽙𖾏 𖼯𖽠．” 36 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼐𖽑𖽘𖾐 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼮𖽡𖽝𖾏，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼮𖽡𖽝𖾏，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏，𖼽𖽔𖾑 𖼋𖽺 𖼮𖽑𖽙𖾏 𖼗𖽙𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏． 37 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：‘𖼐𖽱𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐．’ 𖼗𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼸𖽪𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼞𖽳𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼊𖽡𖽝𖾑．” 38 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼑𖽔𖾏 𖼘𖽪！𖼉𖽺𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼏𖽦𖾏．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼷𖽻𖾏 𖼋𖽘𖾏．”
𖼐𖽡𖽙𖾐𖼖𖽪𖾑𖽐𖼊𖽑𖽘𖾐𖼽𖽦𖾑𖼺𖽪𖾑𖼞𖽑𖽻𖾑𖼊𖽪𖾐𖽐𖼎𖽙𖾏
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
39 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼞𖽗𖾐 𖼖𖽗 𖼇𖽻𖾑 𖼎𖽻𖾐，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼖𖽔𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐． 40 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖽐𖼮𖽑𖽹𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑．” 41 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏，𖼅𖽻𖾏 𖼡𖽔𖾏 𖼚𖽡𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽔𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐，𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏， 42 𖼐𖽑𖽡𖾏：“𖼉𖽺𖾏 𖼘𖽪！ 𖼟𖽡 𖼮𖽔𖾑 𖼯𖽪𖾏 𖼯𖽡𖽙𖾏，𖼷𖽑𖽜𖾑 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽑𖽙𖾑 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 ，𖼮𖽔 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑．” 43 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐，𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏，𖼊𖽑𖽡𖽝𖾐 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏． 44 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼎𖽺𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖼐𖽱𖾏 𖽐𖼞𖽘𖾏 𖼖𖽑𖽘𖾐 𖽐𖼚𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽘𖾑 𖽐𖼚𖽙，𖽐𖼚𖽙 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐． 45 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼋𖽘𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏，𖽐𖼮𖽡𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑，𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽳𖾏 𖼊𖽻𖾏 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏， 46 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏？ 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑！”

66 𖽂𖽪𖾏 𖼺𖽘𖾏 𖽐𖼊𖽪𖾏，𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼗𖽻 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑，𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼦𖽡 𖼸𖽱， 𖼨𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼑𖽪𖾏． 67 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼾𖽙 𖼮𖽡𖾑 𖼊𖽪，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜． 68 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼑𖽪 𖼅𖽡，𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐． 69 𖼋𖽳 𖼟𖽻𖾏 𖼐𖽔𖾑 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏．” 70 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼑𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼘𖽪？” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼐𖽑𖽡𖾏，𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏．” 71 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼀𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑？𖽃𖽡𖾑 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏，𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏．”
