𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼊𖽘𖾑 𖼸𖽙 𖼐𖽙 𖼽𖽔𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 1：1- 4）
5
𖼽𖽔𖾏 𖼊𖽘𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐. 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼾𖽨𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏. 2 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼾𖽠 𖽃𖽔 𖼁𖽙𖾏．𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼷𖽷𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
3 𖼽𖽔𖾏 𖼊𖽘𖾑 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼺𖽱𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼎𖽺𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼺𖽳 𖼊𖽑𖽳. 4 𖼽𖽔𖾏 𖼊𖽘𖾑 𖼅𖽔𖾏 𖼺𖽳 𖼊𖽑𖽳 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼾𖽡𖾏 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 5 𖼽𖽔𖾏 𖼊𖽘𖾑 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼯𖽢𖾏 𖼀𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.
6 𖼺𖽳 𖼊𖽑𖽳 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼖𖽡 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼐𖽙 𖼺𖽷𖾏. 7 𖼺𖽳 𖼊𖽑𖽳 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼐𖽙 𖼺𖽷𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼾𖽡𖾏 𖼀𖽔𖾏 𖼖𖽡 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 8 𖼺𖽳 𖼊𖽑𖽳 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼯𖽢𖾏 𖼀𖽔𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.
9 𖼽𖽡𖾑 𖼐𖽙 𖼺𖽷𖾏 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼞𖽹𖾑 𖼐𖽔. 10 𖼽𖽡𖾑 𖼐𖽙 𖼺𖽷𖾏 𖼅𖽔𖾏 𖼞𖽹𖾑 𖼐𖽔 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼾𖽡𖾏 𖼀𖽔𖾏 𖼟𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 11 𖼽𖽡𖾑 𖼐𖽙 𖼺𖽷𖾏 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼯𖽢𖾏 𖼀𖽔𖾏 𖼖𖽡 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.
12 𖼞𖽹𖾑 𖼐𖽔 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼄𖽔𖾐 𖼖𖽳 𖼖𖽡𖽝. 13 𖼞𖽹𖾑 𖼐𖽔 𖼅𖽔𖾏 𖼄𖽔𖾐 𖼖𖽳 𖼖𖽡𖽝 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼾𖽡𖾏 𖼀𖽔𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 14 𖼞𖽹𖾑 𖼐𖽔 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼯𖽢𖾏 𖼀𖽔𖾏 𖼟𖽻𖾏 𖼳𖽡𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.
15 𖼄𖽔𖾐 𖼖𖽳 𖼖𖽡𖽝 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼽𖽔𖾐 𖼖𖽡𖽝. 16 𖼄𖽔𖾐 𖼖𖽳 𖼖𖽡𖽝 𖼅𖽔𖾏 𖼽𖽔𖾐 𖼖𖽡𖽝 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼾𖽡𖾏 𖼀𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 17 𖼄𖽔𖾐 𖼖𖽳 𖼖𖽡𖽝 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼾𖽡𖾏 𖼀𖽔𖾏 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.
18 𖼽𖽔𖾐 𖼖𖽡𖽝 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼐𖽙. 19 𖼽𖽔𖾐 𖼖𖽡𖽝 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼐𖽙 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼾𖽡𖾏 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 20 𖼽𖽔𖾐 𖼖𖽡𖽝 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼯𖽢𖾏 𖼀𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.
21 𖼽𖽡𖾑 𖼐𖽙 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼄𖽔𖾐 𖼊𖽑𖽪𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑. 22 𖼽𖽡𖾑 𖼐𖽙 𖼅𖽔𖾏 𖼄𖽔𖾐 𖼊𖽑𖽪𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 23 𖼽𖽡𖾑 𖼐𖽙 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏. 24 𖼽𖽡𖾑 𖼐𖽙 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼙𖽠𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼋𖽘.
25 𖼄𖽔𖾐 𖼊𖽑𖽪𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼖𖽔𖾑 𖼄𖽳. 26 𖼄𖽔𖾐 𖼊𖽑𖽪𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼖𖽔𖾑 𖼄𖽳 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼳𖽤𖾏 𖼀𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 27 𖼄𖽔𖾐 𖼊𖽑𖽪𖾐 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼯𖽢𖾏 𖼀𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.
28 𖼖𖽔𖾑 𖼄𖽳 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐， 29 𖼊𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼐𖽙 𖼽𖽔𖾏, 𖼐𖽑𖽡𖾏: “𖼷𖽔𖾑 𖼊𖽪𖾐 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼀𖽡𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖽐𖼎𖽙 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼗𖽙𖾏 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼀𖽡𖾐； 𖼋𖽡𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐.” 30 𖼖𖽔𖾑 𖼄𖽳 𖼅𖽔𖾏 𖼐𖽙 𖼽𖽔𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼀𖽜𖾐 𖼀𖽔𖾏 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 31 𖼖𖽔𖾑 𖼄𖽳 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼳𖽤𖾏 𖼀𖽔𖾏 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.
32 𖼐𖽙 𖼽𖽔𖾏 𖼀𖽜𖾐 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼳𖽔𖾐、𖼦𖽗、𖼽𖽔𖾐 𖼇𖽪 𖼋𖽘𖾏.
