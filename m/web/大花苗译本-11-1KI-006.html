𖼁𖽱𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏
6
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏,𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽻𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼳𖽡𖾑 𖼇𖽪 𖼖𖽑𖽡𖾏, 𖼟𖽪𖾏 𖼖𖽪𖾑 𖽃𖽔𖾐 𖼖𖽑𖽡𖾏, 𖼳𖽠𖾑 𖼁𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 2 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏, 𖽐𖼊𖽡𖾑 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼇𖽘𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼺𖽡𖽝𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏. 3 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼡𖽘𖾐 𖽃𖽔𖾏 𖽐𖼮𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼇𖽘𖾑 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐, 𖼇𖽘𖾑 𖼟𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏； 4 𖼎𖽺𖾏 𖼑𖽜 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖼀𖽔𖾐 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼞𖽑𖽙𖾑. 5 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼋𖽜𖾏 𖼮𖽑𖽤, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼷𖽷𖾐 𖼷𖽜𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼞𖽑𖽡𖾏 𖼉𖽜𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑 𖼖𖽪𖾑 𖼮𖽑𖽤 . 6 𖽃𖽡𖾐 𖼷𖽜𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼇𖽘𖾑 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖽃𖽡𖾐 𖼷𖽜𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼇𖽘𖾑 𖼚𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖽃𖽡𖾐 𖼷𖽜𖾏 𖽃𖽔𖾐 𖼳𖽔𖾐 𖼇𖽘𖾑 𖼳𖽤𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏. 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖽐𖼊𖽻𖾏, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼋𖽜𖾏 𖼮𖽑𖽤 𖼦𖽡𖾏 𖼇𖽔𖾑, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼷𖽑𖽻𖾏 𖼊𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼖𖽪𖾑 𖼮𖽑𖽤 . 7 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼸𖽻 𖼗𖽙𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾑 𖼊𖽻𖾏 𖽐𖼎𖽙 𖼮𖽑𖽪𖽺 𖼊𖽑𖽡𖽝𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼖𖽑𖽻𖾏 𖼳𖽘𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽙𖾑𖽑. 8 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏, 𖽃𖽔 𖼛𖽻𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼅𖽔𖾏 𖼊𖽡𖾑 𖼯𖽙𖾏 𖼟𖽪𖾏 𖼞𖽑𖽡𖾏 𖼉𖽜𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖼟𖽙𖾏 𖼻𖽻𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾐 𖼷𖽜𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾐 𖼷𖽜𖾏 𖼻𖽻𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼷𖽷𖾐 𖼷𖽜𖾏. 9 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖼑𖽡𖾏 𖽐𖼊𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼟𖽪𖾏 𖽐𖼮𖽦𖾏 𖼡𖽘𖾐 𖽐𖼊𖽑𖽘𖾐. 10 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽡𖾑, 𖼷𖽑𖽔𖾏 𖼷𖽜𖾏 𖼺𖽡𖽝𖾐 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼎𖽑𖽻𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖽐𖼊𖽻𖾏, 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽤 .
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽑𖽡𖾏: 12 “𖼐𖽑𖽡𖾏 𖼸𖽙 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽙𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼮𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏, 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖽃𖽔𖾏, 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼏𖽵 𖽃𖽡𖾑. 13 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝.”
𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 3：8 - 14）
14 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 15 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼖𖽪𖾑 𖼮𖽑𖽤 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖽐𖼮𖽦𖾏 𖼘𖽙𖾏, 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼋𖽠𖾏 𖼡𖽔𖾏 𖼎𖽪𖾏, 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖽐𖼮𖽦𖾏 𖼘𖽙𖾏, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼊𖽑𖽪𖾑 𖼀𖽔𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼛𖽘𖾐 𖽐𖼞𖽔𖾐. 16 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙, 𖼟𖽪𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑, 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼸𖽙 𖼡𖽔𖾏 𖼎𖽪𖾏, 𖼎𖽑𖽻𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼮𖽡𖽺𖾏. 17 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽃𖽡𖾑 𖼈𖽺𖾏 𖼊𖽑𖽔𖾏 𖽃𖽡𖾐 𖼮𖽑𖽤𖾏 𖽃𖽡𖾑, 𖽐𖼊𖽡𖾑 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏. 18 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽑𖽻𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽐𖼮𖽦𖾏 𖼘𖽙𖾏 𖼮𖽡𖽝𖾏, 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼸𖽻 𖼞𖽳𖾏 𖼞𖽡𖾑 𖽐𖼀𖽔𖾐 𖽐𖼎𖽙 𖼁𖽘𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼋𖽠𖾏. 19 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑, 𖼟𖽙𖾏 𖼻𖽻𖾏 𖼮𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏. 20 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑 𖽐𖼊𖽡𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼇𖽘𖾑 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼘𖽙𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼎𖽻𖾏, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘, 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏. 21 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼘𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 , 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼞𖽪𖾐 𖼮𖽘𖾏 𖼎𖽻𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏, 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏. 22 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽱 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼘𖽙𖾏 𖼮𖽡𖽝𖾏, 𖼘𖽙𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜, 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼖𖽔𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼘𖽙𖾏 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏.
𖼷𖽷𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳
