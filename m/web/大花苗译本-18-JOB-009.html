9
𖼽𖽙 𖼀𖽳 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏:
2 “𖼞𖽪𖾑 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼖𖽺𖾑 ?
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖽐𖼊𖽱𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼞𖽡𖾑 ,
𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽡𖾐 .
4 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 .
𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽺𖾑 ?
5 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽙𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼦𖽡𖾏 𖽐𖼷𖽜𖾑 𖼮𖽑𖽹𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 ,
𖼈𖽡𖽝𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼦𖽡𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 .
6 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 , 𖼇𖽳𖾑 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 ,
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼑𖽡𖾏 𖼊𖽠𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼼𖽙 .
7 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏 , 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏 ,
𖼎𖽺𖾏 𖼘𖽙𖾏 𖼮𖽡𖽝𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 .
8 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼊𖽑𖽔𖾑 𖼀𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 ,
𖼅𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 .
9 𖼐𖽱𖾏 𖼷𖽷𖾏 𖼟𖽻 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼀𖽳 𖼊𖽙𖽪𖾐 、 𖼟𖽻 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼳𖽺𖾑 、 𖼟𖽻 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼄𖽻𖾐 ,
𖽐𖼎𖽙 𖼖𖽪𖾑 𖼟𖽻 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 .
10 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼮𖽔𖾏 𖼋𖽻𖾏 ,
𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼾𖽱 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 .
11 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡𖾏 𖼊𖽘𖾑 , 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 ,
𖼐𖽱𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 .
12 𖼐𖽱𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖽐𖼎𖽳𖾏 𖼋𖽻𖾏 ,
𖼡𖽔 𖼋𖽱𖾏 𖼞𖽔𖾐 𖼑𖽪 𖼐𖽱𖾏 ： 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 ?
𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽜 𖼾𖽔𖾏 𖼐𖽔𖾑
13 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 ,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼖𖽔𖾑 𖼦𖽔𖾐 𖼀𖽳 𖽃𖽡𖾑（“ 𖼖𖽔𖾑 𖼦𖽔𖾐 𖼀𖽳 ” 𖼟𖽪𖾏 𖼙𖽘𖾑 𖼙𖽘𖾏 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖽠𖾑 ） , 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡 𖼯𖽡𖾏 .
14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼞𖽔𖾐 𖼊𖽡𖾐 𖼐𖽱𖾏 ,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼞𖽔𖾐 𖽐𖼎𖽑𖽦𖾑 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼘𖽪 ?
15 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼊𖽡𖾐 𖼐𖽱𖾏 ,
𖼮𖽔 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 .
16 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼐𖽱𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 .
17 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼷𖽑𖽔𖾐 𖼗𖽙𖾏 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑 ,
𖼖𖽡𖾏 𖼅𖽪𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 .
18 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽑𖽱𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼀𖽘𖾏,
𖼎𖽙𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 .
19 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼛𖽘𖾏 𖼼𖽙𖾏 , 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 !
𖼐𖽑𖽡𖾏 𖼸𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 , 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼖𖽺𖾑 ?
20 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏 , 𖼞𖽪𖾑 𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 ,
𖼞𖽪𖾑 𖼮𖽔𖾑 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 , 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 .
21 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 , 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼾𖽨𖾏 ,
𖼞𖽪𖾑 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 .
22 𖼻𖽻𖾏 𖽐𖼎𖽙 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼇𖽺𖾐 , 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 ,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 : 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽱𖾏 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 .
23 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 ,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑.
24 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 ,
𖽐𖼀𖽙 𖼺𖽘𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 .
𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 , 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑 ?
 
25 𖼞𖽪𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼇𖽺𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼏𖽘 𖼇𖽺𖾏 ,
𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 , 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 .
