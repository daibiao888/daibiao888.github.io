𖼺𖽻𖾐 𖼘𖽙 𖼊𖽻𖾏 𖼮𖽨𖾏
10 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼙𖽻， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏，𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼐𖽔 𖼐𖽡 𖼽𖽔𖾏 ，𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼚𖽔𖾐 𖼼𖽘𖾏 𖽐𖼚𖽙 𖼁𖽙𖾏 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼽𖽔𖾏 𖼐𖽔 𖼐𖽡 𖼽𖽔𖾏 ．” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ：“ 𖼮𖽵𖾑 ！ 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑．” 11 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼳𖽠𖾑 𖼗𖽙𖾏，𖼅𖽻𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼟𖽪𖾏𖽐𖼞𖽳𖾏 𖽃𖽡𖾑 ，𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼞𖽔𖾏，𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾏 𖼺𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼺𖽻𖾐 𖼘𖽙 ．𖼑𖽔𖾏 𖼘𖽪 ！ 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑， 12 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼐𖽔 𖼐𖽡 𖼽𖽔𖾏 ，𖼁𖽺𖾏 𖼋𖽔𖾏 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑． 13 𖼽𖽔𖾏 𖼐𖽔 𖼐𖽡 𖼽𖽔𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽙𖾑 𖽃𖽔𖾑 𖼑𖽡𖽝 𖽃𖽔𖾏 𖼁𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏，𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 ． 14 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼽𖽔𖾑 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏．” 15 𖼮𖽵𖾑 𖽐𖼎𖽙𖼽𖽔𖾏 𖼐𖽔 𖼐𖽡 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼯𖽔 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 ，𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑 𖼊𖽻𖾏，𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐，𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼡𖽑𖽔𖾐 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏． 16 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏，𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏．” 17 𖼽𖽔𖾏 𖼐𖽔 𖼐𖽡 𖼽𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼁𖽺𖾏 𖼚𖽔𖾏 𖼾𖽡𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽐𖼞𖽔𖾏，𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼺𖽻𖾐 𖼘𖽙 ，𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼋𖽔𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼑𖽔𖾏 𖽃𖽡𖾑，𖼷𖽷𖾐 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼁𖽙𖾏 𖼞𖽡𖾑，𖼎𖽺𖾏 𖼊𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑．” 18 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼎𖽑𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽃𖽔𖾑 𖼅𖽔， 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼷𖽷𖾏 𖼚𖽑𖽘𖾏 𖼀𖽻𖾐 𖼗𖽙，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 ，𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽐𖼷𖽔𖾑． 19 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏．
