11
𖼮𖽑𖽡𖽻𖾏 𖼗𖽠𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏, 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽠𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼽𖽙 𖼽𖽔 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏． 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼙𖽻𖾏 𖽐𖼊𖽻𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐. 2 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐, 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼀𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼖𖽪𖾑 𖼏𖽪𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑, 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝. 3 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼑𖽪 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏． 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖽝 𖽐𖼷𖽑𖽹𖾏, 𖼟𖽪𖾏 𖼦𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼀𖽔 𖼳𖽷𖾏 𖼀𖽔𖾑.” 4 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼨𖽑𖽳𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼋𖽔𖾏, 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼡𖽔𖾏 𖼖𖽑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼺𖽘𖾏 𖼊𖽔𖾐. 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼋𖽘, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼀𖽱𖾏, 𖼐𖽱𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 5 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏, 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏.”
6 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼽𖽔 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼷𖽷𖾐 𖼟𖽙𖾏𖼦𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑.” 𖼽𖽙 𖼽𖽔 𖼎𖽺𖾏 𖼷𖽷𖾐 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏. 7 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽪 𖼽𖽙 𖼽𖽔 𖼋𖽡𖽝𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐, 𖼎𖽺𖾏 𖼑𖽪 𖼏𖽙 𖼋𖽡𖽝𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐, 𖼎𖽺𖾏 𖼑𖽪 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏. 8 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼷𖽔𖾑 𖼊𖽠𖾏 𖼦𖽡𖾑!” 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼎𖽑𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏. 9 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼸𖽻𖾏 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽃𖽔 𖼛𖽻𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏, 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏. 10 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼅𖽔 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼊𖽡𖾐.” 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽪 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼎𖽑𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼗𖽙, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼘𖽪?” 11 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼖𖽔𖾑 𖼼𖽪𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼏𖽙, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼚𖽙， 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼽𖽙 𖼽𖽔 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑, 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼮𖽪𖾐 𖼾𖽡, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽤𖾐 𖼀𖽱𖾏 𖼘𖽪? 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼏𖽵 𖼗𖽪𖾏: 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑.” 12 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼀𖽡 𖼟𖽡 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼅𖽻𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐, 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐. 13 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼨𖽑𖽪𖾏 𖼟𖽙𖾏 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼋𖽔𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼑𖽻𖾏 𖼦𖽻𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽻𖾏 𖼟𖽪𖾏 𖼢𖽻 𖼮𖽡𖽠𖾑 𖼺𖽘𖾏． 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏， 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼸𖽻𖾏 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼀𖽱𖾏 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏, 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏.
