𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 , 𖼑𖽜 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 .
2 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 3 𖼞𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 、𖼽𖽡𖾑 𖼺𖽔𖾐 、𖼽𖽔𖾐 𖼞𖽙 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 , 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼀𖽻𖾐 𖼺𖽷𖾐 . 4 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖽝𖾏 , 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 5 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 . 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽨𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼅𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼟𖽪𖾏 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼅𖽔𖾏 𖼎𖽙𖾑 𖼅𖽡 . 7 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽨𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 . 8 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 、𖼽𖽡𖾑 𖼺𖽔𖾐 、𖼽𖽔𖾐 𖼞𖽙 𖽃𖽡𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼁𖽺𖾏 𖼅𖽻𖾏 , 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 , 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .” 9 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖼊𖽳𖾏 , 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼗𖽪𖾏 .
