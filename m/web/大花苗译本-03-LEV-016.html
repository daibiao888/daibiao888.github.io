16
𖼽𖽔𖾏 𖼖𖽪𖽝 𖼑𖽡𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽔 𖼙𖽠𖾏. 𖼋𖽔 𖼙𖽠𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “ 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼀𖽑𖽙𖾑 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽺𖾏 𖽐𖼚𖽙 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽡𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼚𖽙 𖼢𖽳 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 . 3 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾑 𖼙𖽻𖾏 𖼾𖽘𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑. 4 𖼽𖽔𖾑 𖼐𖽑𖽘𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 , 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼏𖽷 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖼐𖽑𖽘𖾑 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼸𖽙𖾏 𖼘𖽑𖽘𖾐 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙, 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖽐𖼊𖽻𖾏 𖼖𖽪𖾑 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 , 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏． 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼞𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 𖽐𖼊𖽻𖾏 . 5 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽤𖾐 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏， 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 .
