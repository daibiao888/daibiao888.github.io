𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏

16 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏，𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 ? 𖼞𖽪𖾑 𖼾𖽡𖾏 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 ,𖼟𖽡 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼏𖽔𖾏 𖼐𖽡𖾑 ? 17 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ! 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 , 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼎𖽺𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ! 𖼟𖽡 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 . 18 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽑𖽨𖾏 𖼎𖽻𖾏 𖼡𖽑𖽜𖾑, 𖼞𖽪𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼺𖽷𖾐 𖼘𖽪 ? 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼟𖽡 𖼡𖽑𖽜𖾑 . 19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼟𖽡 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼐𖽔𖾑 , 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 . 20 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼙𖽔 𖼐𖽙𖾑𖽑 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖽐𖼮𖽔𖾏 𖼋𖽻𖾏 𖼟𖽡 , 𖼋𖽠𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏 ,𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 21 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 —𖼟𖽪𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼊𖽔𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼘𖽪 ? 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼐𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 , 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼞𖽪𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 . 22 𖼟𖽡 𖼟𖽙𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽔𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 23 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼏𖽵 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼡𖽑𖽜𖾑 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖽃𖽡𖾑 , 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 , 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏 . 24 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 , 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 : 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 —𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ! 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽡𖾑 ! 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼾𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 . 25 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖼡𖽑𖽜𖾐 𖼡𖽑𖽳𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼮𖽙𖾏 𖼚𖽔𖾐 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 . 26 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽡 𖼙𖽔 𖼏𖽵 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 . 27 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼡𖽑𖽜𖾑 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏, 𖼟𖽙𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼷𖽷𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼟𖽡 𖼙𖽔 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏 ,𖼎𖽺𖾏 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 .”
