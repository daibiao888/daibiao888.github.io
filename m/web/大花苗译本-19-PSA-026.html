26
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙.
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑, 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼞𖽪𖾑,
𖼨𖽑𖽙𖾑 𖽐𖼮𖽔𖾏 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽐𖼎𖽙 𖼮𖽳𖾐 𖽐𖼊𖽱𖾏.
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼞𖽪𖾑 𖼖𖽔𖾏 𖼸𖽙𖾏 𖼟𖽡 𖼞𖽘𖾐 𖼞𖽡𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏.
 
4 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖼗𖽙𖾏.
5 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼦𖽡 𖼸𖽱,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼼𖽻𖾏.
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼖𖽔𖾑 𖽐𖼷𖽔𖾑 𖼋𖽡𖾏 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏,
𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼟𖽡 𖽃𖽔𖾑 𖼏𖽘.
7 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖽐𖼷𖽻𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑.
 
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐
𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖽙𖾐.
9 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼾𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼖𖽻𖾐 𖼺𖽘𖾏.
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳,
𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏.
11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑, 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑！
12 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏,
𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
