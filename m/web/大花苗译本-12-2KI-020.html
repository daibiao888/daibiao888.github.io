𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼄𖽙𖾐 𖼐𖽑𖽤𖾑 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏
（𖼺𖽹 38：1- 8；21 - 22；𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 32：24 - 26）
20
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔. 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼐𖽑𖽡𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼋𖽔 , 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼋𖽘.” 2 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽑𖽪𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 , 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 3 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼑𖽪𖾏, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏.” 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼎𖽺𖾏 𖼐𖽡𖽝𖾑 𖼋𖽘𖾏. 4 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖽐𖼚𖽙 ( “𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 ” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼙𖽻”) 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 5 “𖼟𖽡 𖼺𖽷𖾐 𖼅𖽻, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: ‘ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼞𖽔𖾏 𖼅𖽔, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼟𖽡， 𖼸𖽙 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼟𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼽𖽔𖾏 𖼳𖽪𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏. 𖼞𖽪𖾑 𖼑𖽜 𖼾𖽨𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽔𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑.’ ” 7 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼗𖽙𖾏.” 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏, 𖼘𖽙𖾏 𖼎𖽻𖾏 𖼷𖽑𖽘𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏.
8 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼑𖽪 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼞𖽪𖾑, 𖼸𖽙 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?” 9 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼚𖽔𖾐 𖼼𖽔𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡, 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼟𖽻𖾏 𖼏𖽔𖾏 𖼖𖽔𖾏? 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼑𖽡𖾏 𖼡𖽜𖾐 𖼟𖽻𖾏 𖼏𖽔𖾏 𖼖𖽺𖾑?” 10 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼐𖽑𖽪𖾐 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼟𖽻𖾏 𖼏𖽔𖾏 𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽑𖽪𖾐 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼑𖽡𖾏 𖼡𖽜𖾐 𖼟𖽻𖾏 𖼏𖽔𖾏.” 11 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼐𖽑𖽪𖾐 𖽃𖽔𖾑 𖽐𖼷𖽻𖾏 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖽃𖽡𖾑, 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼟𖽻𖾏 𖼏𖽔𖾏.
（𖼺𖽹 39：1 -8）
12 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽡𖾐 𖼘𖽙 𖼊𖽔 · 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼄𖽙𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏, 𖼎𖽺𖾏 𖼺𖽘𖾏 𖽐𖼊𖽠𖾑 𖼄𖽻𖾏 𖼺𖽘𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏. 13 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼞𖽔𖾏 𖼼𖽪 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖼞𖽪𖾐、 𖼑𖽡𖽝𖾏、 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏、 𖼛𖽙𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼎𖽙 𖼏𖽙 𖽐𖼞𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏, 𖽐𖼎𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏, 𖼞𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏. 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏. 14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏? 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡?” 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼋𖽔𖾏.” 15 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏?” 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏, 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏.”
16 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏， 17 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏, 𖼦𖽡𖾐 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐． 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏. 18 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼀𖽙𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼅𖽔𖾏, 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽐𖼚𖽙 𖽃𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏.” 19 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏! 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼅𖽔𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼘𖽪?”
𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 32： 32 - 33）
20 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽐𖼮𖽡𖽠𖾏 𖼁𖽘𖾏 𖽃𖽻𖾐, 𖽐𖼮𖽡𖽠𖾏 𖼟𖽱𖾏 𖽃𖽻𖾐, 𖼊𖽜𖾑 𖽃𖽻𖾐 𖼚𖽔𖾏 𖼙𖽻, 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏《 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏》𖼮𖽡𖽝𖾏. 21 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏, 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
