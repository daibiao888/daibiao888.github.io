𖼽𖽔𖾏 𖼮𖽡𖾑 𖼀𖽑𖽔𖾏 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏：“ 𖼯𖽪𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼾𖽨𖾏 𖼁𖽡𖽝．” 𖼀𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼊𖽑𖽔𖾑 𖼮𖽔𖾑𖽑 𖼐𖽱𖾏 𖼁𖽡𖽝 𖼐𖽑𖽡𖾏： 2 “ 𖼽𖽔𖾏 𖼮𖽡𖾑 𖼀𖽑𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽑𖽔𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼯𖽡𖽙𖾏． 3 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼻𖽻𖾏，𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽑𖽘𖾑 𖼢𖽔𖾏 ， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼑𖽻 𖼞𖽪𖾑．
