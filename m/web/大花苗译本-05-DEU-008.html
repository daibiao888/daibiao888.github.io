𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 , 𖼟𖽙𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼏𖽵 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖽃𖽡𖾑 . 2 𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖼐𖽡𖾑 , 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼨𖽑𖽙𖾑 𖼟𖽡 , 𖽐𖼮𖽔𖾏 𖼟𖽡 , 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼐𖽑𖽡𖽻𖾏 𖼦𖽡𖾏 𖼐𖽑𖽡𖽻𖾏 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 . 3 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼨𖽑𖽙𖾑 𖼟𖽡 , 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼮𖽑𖽹𖾐 , 𖼞𖽑𖽡𖽠𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼑𖽻𖾏 , 𖼟𖽙𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼊𖽔𖾐 , 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼷𖽑𖽔𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 . 4 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖼐𖽡𖾑 , 𖼟𖽡 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽑𖽘𖾑 𖽐𖼚𖽔 𖼺𖽘𖾏 , 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽃𖽙𖾏 . 5 𖼟𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼟𖽡 , 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽃𖽡𖾑 𖼼𖽘𖾐. 6 𖼟𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 .
