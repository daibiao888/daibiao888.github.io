27 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼟𖽡 , 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼄𖽔𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
28 𖼟𖽡 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏,
𖼟𖽡 𖼊𖽑𖽔𖾑 𖼋𖽡𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼑𖽻𖾏 𖼮𖽻𖾏.
29 𖼟𖽡 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏,
𖼟𖽡 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽘𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼁𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏.
30 𖼟𖽡 𖼮𖽙𖾏 𖼋𖽠𖾏 𖼟𖽡 𖼖𖽡𖽾, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼷𖽷𖾏,
𖼟𖽡 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽑𖽦𖾐.
 
31 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼷𖽷𖾐 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼙𖽔 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽷𖾏 𖼊𖽡𖽝𖾑！
32 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼊𖽡𖾐, 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 ,
𖼐𖽱𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏.
33 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼺𖽷𖾐, 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏！
34 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼊𖽱𖾏！

35 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼞𖽑𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏！
