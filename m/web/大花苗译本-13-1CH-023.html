𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 , 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼀𖽪𖾑 𖼋𖽘𖾏 , 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 .
2 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐、 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 . 3 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼮𖽔𖾑 𖼺𖽙𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼉𖽻𖾏 𖼾𖽡𖾏 𖼷𖽑𖽡𖽝𖾐 . 4 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼉𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼑𖽪𖾏 , 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏， 5 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼻𖽙𖾑 𖽃𖽔 𖼛𖽻𖾏 , 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 𖽐𖼎𖽔 𖼟𖽪𖾏𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 6 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼇𖽺𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽳 𖼳𖽪𖽝𖾏 、 𖼞𖽙𖾑 𖼳𖽢 、 𖼄𖽡𖾐 𖼖𖽔𖾑 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 .
7 𖼞𖽳 𖼳𖽪𖽝𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼅𖽔𖾏 𖼖𖽔𖾑 𖼊𖽗𖾏 𖽐𖼎𖽙 𖼳𖽷𖾏 𖼄𖽺𖾐 . 8 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏𖼽𖽦𖾑 𖼳𖽡𖽝， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼊𖽑𖽗𖾐 𖽐𖼎𖽙 𖼽𖽙 𖽃𖽳𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 . 9 𖼳𖽷𖾏 𖼄𖽺𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼘𖽙 𖼄𖽡 、 𖼦𖽔𖾐 𖼳𖽡𖽫𖾑 、 𖼦𖽔𖾐 𖼖𖽔 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐．𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 . 10 𖼳𖽷𖾏 𖼄𖽺𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼦𖽔𖾐 、 𖼳𖽡𖾏 𖼐𖽔 、 𖼽𖽦𖾑 𖽂𖽪𖾑 𖼳𖽷𖾑 、 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 , 𖼚𖽻𖾐 𖼖𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏𖼳𖽷𖾏 𖼄𖽺𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 . 11 𖼽𖽔𖾐 𖼦𖽔𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 , 𖼳𖽡𖾏 𖼺𖽔𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 . 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖽂𖽪𖾑 𖼳𖽷𖾑 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼮𖽡𖽻𖾐 ,𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽔𖾑 𖼺𖽙𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼊𖽔𖾐 .
