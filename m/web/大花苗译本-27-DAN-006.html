𖼊𖽔𖾏 𖼖𖽡𖾏 𖽂𖽪𖾑 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼗𖽜𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼞𖽪𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜． 2 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖼺𖽹𖾑 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼙𖽘𖾑 𖼙𖽘𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 (𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐), 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠 . 3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼖𖽡𖽾 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,𖽃𖽔𖾏 𖼐𖽡𖾑 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼙𖽘𖾑 𖼙𖽘𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼺𖽹𖾑 𖼐𖽱𖾏 𖼯𖽵𖾏 𖼞𖽪𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 .
4 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼸𖽻𖾏 𖼙𖽘𖾏 𖼙𖽘𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 𖼯𖽵𖾏 𖼞𖽪𖽙 , 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼡𖽘𖾐 𖼛𖽠𖾏 𖼑𖽪𖾏 𖼐𖽱𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽠𖾏 𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏,𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏, 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼾𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽠𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏. 5 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼷𖽷𖾑 𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼡𖽘𖾐, 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼊𖽘𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼡𖽘𖾐.”
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼸𖽻𖾏 𖼙𖽘𖾏 𖼙𖽘𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾐 𖼙𖽳𖾐 𖼦𖽡𖾏 𖼷𖽱𖾏 𖼦𖽡 𖼸𖽱 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽔𖾏 𖼖𖽡𖾏 𖽂𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼉𖽻𖾏 𖼳𖽡𖽻𖾏! 7 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼙𖽘𖾏 𖼙𖽘𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑、 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏、 𖼊𖽪𖾐 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑, 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼦𖽡𖾏 𖼊𖽳𖾏, 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼷𖽹𖾑 𖼗𖽙𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 ， 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖽐𖼚𖽙 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽜𖾏 𖼽𖽔𖾑 𖼊𖽔𖾐， 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼳𖽺 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏,𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽐𖼚𖽙. 8 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！𖼐𖽔𖾑 𖼐𖽡𖾑 𖼮𖽑𖽙 𖼟𖽡 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏, 𖼘𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 , 𖼟𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏, 𖼸𖽙𖾏 𖼄𖽔𖾐 𖼊𖽹𖾏 𖽐𖼎𖽙 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.” 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼊𖽔𖾏 𖼖𖽡𖾏 𖽂𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽙𖾏 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖼎𖽻𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼊𖽠𖾑 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼐𖽡𖾑 .
10 𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼐𖽡𖾑 𖼘𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖼎𖽻𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏, 𖼐𖽱𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼮𖽑𖽤𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼷𖽷𖾐 𖼟𖽻𖾐, 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.
