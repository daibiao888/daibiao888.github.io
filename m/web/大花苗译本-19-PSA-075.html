𖼷𖽻𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐
𖽃𖽡𖾐 𖼸𖽙𖾐 𖽐𖼞𖽻𖾏, 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑” 𖼳𖽘𖾐.
75
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪，𖼀𖽡𖾐 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡， 𖼀𖽡𖾐 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖽃𖽺𖾐 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑.
2 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼏𖽵 𖼮𖽡𖽝𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏.
3 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘,
𖼞𖽪𖾑 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
4 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏！”
𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐！
5 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼅𖽡 𖼊𖽪𖾑 𖼞𖽪𖾐,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽤𖾐 𖼦𖽡𖾑 𖼚𖽘𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏.”
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾐 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐、𖼺𖽘𖾐 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏、
𖼺𖽘𖾐 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼋𖽔𖾏.
7 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽡𖾏 𖼏𖽵,
𖼟𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼦𖽡𖾏 𖼢𖽜𖾏, 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼳𖽠𖾑 𖼺𖽡𖽝𖾐.
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐, 𖼮𖽡𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼦𖽡𖾑 𖽐𖼀𖽔𖾏,
𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖽐𖼚𖽙 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏,
𖼐𖽱𖾏 𖼘𖽑𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼦𖽻𖾏 𖼋𖽡𖾏 𖼮𖽡𖽠𖾑 𖼐𖽡𖾑,
𖼋𖽡𖾏 𖼮𖽡𖽠𖾑 𖼐𖽡𖾑 𖼡𖽔𖾏 𖼷𖽙𖾏 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼦𖽻𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏.
 
9 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏！
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏！
10 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖾐, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽷𖾐 𖼊𖽪𖾏,
𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐.
