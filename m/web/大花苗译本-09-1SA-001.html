1
𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼄𖽔𖾐 𖼺𖽙𖾐 𖼇𖽺𖾑, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼐𖽔. 𖼟𖽪𖾏 𖼺𖽪𖾑 𖼇𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼊𖽑𖽙 𖼦𖽪𖾏 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼘𖽔𖾑𖽑, 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼦𖽪𖾏 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑, 𖼽𖽦𖾑 𖼘𖽙 𖼦𖽗𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐. 2 𖼐𖽱𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽷𖾏 𖼦𖽔𖾐 𖼐𖽔, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽑𖽡 𖼐𖽡 𖼐𖽔． 𖼀𖽑𖽡 𖼐𖽡 𖼐𖽔 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏, 𖼦𖽔𖾐 𖼐𖽔 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏.
3 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼳𖽡𖽻𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼙𖽻 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼘𖽙, 𖼷𖽻𖾑 𖼖𖽔 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼑𖽡𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼦𖽙 𖼇𖽪 𖼐𖽡、𖼇𖽺𖾑 𖼐𖽡 𖼦𖽔𖾐, 𖽃𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏. 4 𖼷𖽑𖽔𖾏 𖼟𖽻𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑, 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼄𖽔𖾐 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼀𖽑𖽡 𖼐𖽡 𖼐𖽔 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼀𖽑𖽡 𖼐𖽡 𖼐𖽔 𖼳𖽻𖾏 𖽃𖽡𖾑. 5 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼦𖽔𖾐 𖼐𖽔 𖼟𖽪𖾏 𖽃𖽔𖾐 𖽐𖼀𖽠 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼦𖽔𖾐 𖼐𖽔 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼦𖽔𖾐 𖼐𖽔 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏. 6 𖼀𖽑𖽡 𖼐𖽡 𖼐𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼦𖽔𖾐 𖼐𖽔 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐, 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐, 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐. 7 𖼷𖽑𖽔𖾏 𖼳𖽡𖽻𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼽𖽡𖾑 𖼖𖽡 𖼮𖽢𖾑 𖼐𖽔 𖼖𖽔𖾏 𖼄𖽔𖾐 𖽃𖽔𖾐 𖽐𖼀𖽠 𖼎𖽻𖾏 𖼦𖽔𖾐 𖼐𖽔, 𖼈𖽡𖽝𖾏 𖼀𖽑𖽡 𖼐𖽡 𖼐𖽔 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽝𖾑 𖼮𖽑𖽡𖾏 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼉𖽔𖾏. 8 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼐𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽔𖾐 𖼐𖽔 𖼘𖽪！ 𖼟𖽡 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼐𖽔𖾑 𖼐𖽡𖽝𖾑, 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼉𖽔𖾏, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼖𖽺𖾑? 𖼅𖽔𖾏 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖼺𖽷𖾐 𖼘𖽪𖾏?”
𖼦𖽔𖾐 𖼐𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼳𖽷𖾏 𖼘𖽙 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼊𖽘𖾏 𖼋𖽘𖾏, 𖼦𖽔𖾐 𖼐𖽔 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽡𖾑 𖼖𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪, 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐. 10 𖼦𖽔𖾐 𖼐𖽔 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐, 𖼎𖽺𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼐𖽡𖽝𖾑, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫， 11 𖼎𖽺𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼮𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼑𖽔𖾏 𖼞𖽻𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼑𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏, 𖼞𖽻𖾏 𖼮𖽳𖾐 𖼦𖽡𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑, 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼮𖽑𖽪𖾑 𖼮𖽑𖽪𖾑 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏.”
12 𖼦𖽔𖾐 𖼐𖽔 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼳𖽙𖾏, 𖼽𖽡𖾑 𖼖𖽡 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏. 13 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽔𖾐 𖼐𖽔 𖼎𖽑𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼊𖽔𖾐, 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼽𖽡𖾑 𖼖𖽡 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼢𖽻 𖼮𖽡𖽠𖾑 𖼋𖽘𖾏. 14 𖼽𖽡𖾑 𖼖𖽡𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼢𖽻 𖼮𖽡𖽠𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑? 𖼟𖽡 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑.” 15 𖼦𖽔𖾐 𖼐𖽔 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼘𖽪！ 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐, 𖼮𖽡𖽠𖾑 𖼸𖽺𖾏 𖼮𖽡𖽠𖾑 𖽐𖼮𖽑𖽦𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼦𖽻𖾏, 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏. 16 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼟𖽡 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼏𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏, 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑.” 17 𖼽𖽡𖾑 𖼖𖽡 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼋𖽡𖽝𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽪 𖼺𖽷𖾐 𖼅𖽻, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽐𖼊𖽜 𖽃𖽡𖾑.” 18 𖼦𖽔𖾐 𖼐𖽔 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏．” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖽃𖽔 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏, 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼋𖽘.
𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐
19 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽠𖾑 𖼗𖽙, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼘𖽘𖾑𖽑 𖼸𖽔𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼖𖽔𖾑 𖼄𖽔𖾐, 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏. 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼐𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼦𖽔𖾐 𖼐𖽔 𖼀𖽱𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽻𖾏 𖼮𖽳𖾐 𖼦𖽔𖾐 𖼐𖽔. 20 𖼦𖽔𖾐 𖼐𖽔 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑, 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼀𖽪𖾑 𖼋𖽘𖾏, 𖼅𖽔𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐, 𖼐𖽑𖽡𖾏: “ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏.”
𖼖𖽑𖽘𖾑 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫
21 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼐𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖽐𖼞𖽔𖾐 𖼾𖽡𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼘𖽙, 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼳𖽡𖽻𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼏𖽵. 22 𖼈𖽡𖽝𖾏 𖼦𖽔𖾐 𖼐𖽔 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼎𖽙 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼐𖽑𖽡𖾏: “𖼋𖽙𖾏 𖼟𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼇𖽳𖾑 𖼄𖽡𖾏 𖼺𖽘𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑.” 23 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼐𖽑𖽡𖾏: “𖼸𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏, 𖼋𖽙𖾏 𖼟𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼇𖽳𖾑 𖼄𖽡𖾏 𖼺𖽘𖾏, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼐𖽱𖾏 𖼗𖽪𖾏．” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏 𖼊𖽪𖾑 𖼾𖽨 𖼣𖽔𖾑 𖼽𖽻𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼄𖽡𖾏. 24 𖼇𖽳𖾑 𖼄𖽡𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼨𖽑𖽳𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼘𖽙, 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼎𖽺𖾏 𖼮𖽤𖾐 𖼷𖽷𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏, 𖽃𖽡𖾐 𖼽𖽡𖾑 𖼇𖽔 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏, 𖽃𖽡𖾐 𖼘𖽑𖽱𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼮𖽡𖽠𖾑, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼳𖽻𖾐 𖼺𖽷𖾐. 25 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏, 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐. 26 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽪𖾑 𖼞𖽔𖾐 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼏𖽵 𖼗𖽪𖾏, 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑, 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏. 27 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏. 28 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏 𖼺𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.”
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
