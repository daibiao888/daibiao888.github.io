11
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏，𖼐𖽱𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏，𖼽𖽔𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼄𖽻𖾏 𖼻𖽻𖾏．
（𖼖𖽪𖾏 7：18-35）
2 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼦𖽗 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 ， 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼅𖽻𖾏， 3 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼗𖽙𖾏 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖼘𖽪？𖼖𖽔𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼋𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽷𖾐 𖼘𖽪？” 4 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏𖼽𖽙 𖼦𖽗，𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽙𖾑𖽑， 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽙𖾏． 5 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑， 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼸𖽠𖾏 𖼅𖽻， 𖼑𖽻𖾐 𖽐𖼮𖽢𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼨𖽑𖽪𖾑， 𖽃𖽔𖾑 𖼗𖽘𖾐 𖼐𖽙𖾑𖽑 𖼗𖽪𖾏， 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏，𖼊𖽪𖾐 𖼳𖽻𖾏 𖼅𖽔𖾏 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 6 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼢𖽻 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗
7 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼽𖽙 𖼦𖽗𖼎𖽺𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏，𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑？ 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖼟𖽪𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼘𖽪？ 8 𖼅𖽡 𖼋𖽠𖾏 𖼅𖽻𖾏，𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼐𖽔𖾑？ 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼯𖽪𖾏 𖽐𖼀𖽔 𖼘𖽪？ 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼯𖽪𖾏 𖽐𖼀𖽔 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻． 9 𖼅𖽡 𖼋𖽠𖾏 𖼅𖽻𖾏 ， 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？𖼟𖽪𖾏 𖼑𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼖𖽔𖾏？𖼾𖽙 𖼯𖽡𖽙𖾏， 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ， 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐． 10 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼺𖽻𖾏 𖼐𖽑𖽡𖾏：
‘ 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼨𖽑𖽳𖾏 𖼟𖽡 𖽐𖼊𖽡𖾏，
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼟𖽡 𖼞𖽡𖾑 ．’
11 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ， 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼡𖽑𖽙𖾑 𖼘𖽘𖾑𖽑 𖼾𖽨，𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼗𖽙𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗 ．𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙，𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏． 12 𖼎𖽑𖽻𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑， 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼦𖽔𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼦𖽔𖾏 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏． ”
13 “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽔𖾏， 𖼸𖽙 𖼽𖽙 𖼦𖽗𖼊𖽔𖾐． 14 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽑𖽡𖽻𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏，𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔𖾏 𖼯𖽡𖽙． 15 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼑𖽻，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻．”
𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜
16 “ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼗𖽙𖾏 𖽐𖼮𖽔𖾏 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑？𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏，𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽺 𖼗𖽔，
17 𖼐𖽑𖽡𖾏：‘𖼀𖽡𖾐 𖼷𖽑𖽙𖾑 𖼖𖽡𖾑 𖼘𖽪𖾏， 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏，
𖼀𖽡𖾐 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐，𖼅𖽡 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼀𖽔𖾏 𖽐𖼎𖽙𖾐．’
18 𖽃𖽡𖾑 𖼟𖽪𖾏𖼽𖽙 𖼦𖽗 𖼋𖽔𖾏，𖼦𖽡𖾐 𖼑𖽻𖾏 𖼦𖽡𖾐 𖼦𖽻𖾏， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：‘ 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏．’ 19 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼋𖽔𖾏，𖼐𖽱𖾏 𖼑𖽻𖾏 𖼐𖽱𖾏 𖼦𖽻𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：‘ 𖼅𖽡 𖼑𖽔𖾏，𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽙𖾑 𖼑𖽻𖾏 𖽐𖼮𖽙𖾑 𖼮𖽡𖽠𖾑，𖼟𖽪𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼽𖽺 𖼗𖽔．’ 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏，𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼋𖽘𖾏．”
𖼷𖽷𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻
（𖼖𖽪𖾏 10：13-15）
20 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼖𖽔𖾏 𖼋𖽡𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽱𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖽃𖽡𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻，𖼐𖽑𖽡𖾏： 21 “𖼞𖽙𖾑 𖼖𖽔𖾑 𖼳𖽱𖾏 𖼘𖽪！𖼟𖽡 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 ！ 𖼀𖽳 𖼺𖽹 𖼊𖽔𖾏 𖼘𖽪！𖼟𖽡 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼙𖽻 𖽃𖽔𖾏，𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙𖽐𖼎𖽙 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼙𖽻，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽙𖾑 𖽐𖼀𖽔𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖽐𖼀𖽑𖽻𖾏 𖼮𖽑𖽻𖾑 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼋𖽘𖾏． 22 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼸𖽙 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 ， 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖽐𖼎𖽙𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼳𖽷𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼅𖽡． 23 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾 𖼙𖽻 𖼘𖽪𖾏！
𖼟𖽡 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼸𖽙 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼘𖽪𖾏？
𖼟𖽡 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼅𖽡 ， 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖼙𖽻， 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼸𖽙 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼺𖽷𖾐． 24 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼸𖽙 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐， 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼳𖽷𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼅𖽡．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽔𖾏 𖼙𖽔 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏
（𖼖𖽪𖾏 10：21-22）
25 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼉𖽺𖾏 𖼘𖽪！ 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼮𖽵𖾑 𖼘𖽪 ！𖼞𖽪𖾑 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼟𖽡！ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼟𖽡 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 ， 𖼈𖽡𖽝𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏． 26 𖼉𖽺𖾏 𖼘𖽪！𖼾𖽙 𖼯𖽡𖽙𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏． 27 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼙𖽔 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑，𖼋𖽠𖾏 𖼉𖽺𖾏 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖼊𖽪𖾐．𖼋𖽠𖾏 𖼺𖽘𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼯𖽪𖾏 𖼢𖽳 𖼡𖽑𖽳𖾏 𖼺𖽘𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖼉𖽺𖾏． 28 𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽑𖽔𖾏 𖼋𖽘𖾏，𖼟𖽪𖾏 𖼎𖽷𖾏 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼳𖽙𖾏． 29 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐，𖼽𖽔𖾑 𖼾𖽻 𖼞𖽪𖾑 𖼼𖽘𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾏 𖽐𖼎𖽙 𖼳𖽻𖾐 𖼢𖽜，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ， 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼳𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑． 30 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼖𖽡 𖼟𖽻𖾏，𖼞𖽪𖾑 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼳𖽷𖾐 𖼯𖽡𖽙．”
