𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏
2
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 ! 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼸𖽱𖾏 𖼸𖽘𖾏 𖽐𖼊𖽱𖾏 𖼮𖽳𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖽐𖼎𖽑𖽦𖾑 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽡𖾑,
𖽂𖽪𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏.
2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽠𖾑 𖼉𖽳 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼳𖽪𖾏 𖼮𖽡𖽝𖾏,
𖼦𖽠𖾑 𖼉𖽳 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼦𖽔𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼅𖽻𖾏 𖼦𖽔𖾏 𖼳𖽪𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼞𖽙𖾏.
3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 ,
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼦𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼺𖽘𖾏,
𖼅𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔.
4 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼅𖽡 , 𖼐𖽑𖽡𖾏:
“𖼀𖽡𖾐 𖼞𖽑𖽔𖾏 𖼦𖽡 𖼋𖽙 𖼙𖽠𖾏 𖼋𖽘𖾏 !
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼀𖽡𖾐 𖼺𖽘𖾏？
𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏.”
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 𖼮𖽤𖾐 𖼖𖽑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 .
𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐
6 𖼐𖽱𖾏 𖼸𖽻𖾏 ( 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “ 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 ”) 𖼐𖽑𖽡𖾏 :
‘ 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 ,
𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼗𖽙𖾏 𖼷𖽠𖾏 𖼀𖽡𖾐.’
7 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏 𖼘𖽪 ! 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 :
‘𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “ 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼡𖽹𖾏 ”)? ’
𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
 
8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼺𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖼎𖽑𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏, 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼙𖽠𖾏.
9 𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼗𖽙𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏,
𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼣𖽔𖾑 𖼾𖽻𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏.
10 𖼅𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼅𖽻𖾏 𖼦𖽡𖾑 !
𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼳𖽙𖾏 𖼐𖽡𖽙𖾐 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼐𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼖𖽻𖾐 ,𖼎𖽺𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
11 𖼯𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼐𖽑𖽡𖾏:
‘ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽡𖽠𖾑 𖼸𖽺𖾏 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐.’
𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐.
𖼏𖽵 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼺𖽱𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏
12 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖼅𖽡 ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏, 𖼺𖽱𖾏 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 (“𖼀𖽙𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 ” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖽐𖼞𖽔𖾏 𖼾𖽘𖾐 ”) 𖼑𖽡𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼖𖽪𖾑 𖽐𖼎𖽘𖾏 𖽐𖼡𖽜𖾐 𖼐𖽱𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏.
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼎𖽺𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏 𖼟𖽙𖾏 𖼘𖽑𖽙𖾐.
