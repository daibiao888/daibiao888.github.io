𖼽𖽔𖾏 𖼖𖽪𖽝 𖼘𖽑𖽡𖾑 𖼑𖽨𖾏 𖼁𖽠𖾏
（𖼎𖽺𖾏 9：6 - 29）
32
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼁𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼊𖽳𖾑𖽑 𖼗𖽡𖾏 𖼦𖽡𖾐 𖽐𖼡𖽜𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼗𖽙𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼳𖽠𖾑 𖼗𖽙𖾏 ! 𖼑𖽜 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽙𖾏 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖽐𖼊𖽡𖾏 𖼯𖽡𖽙𖾏 𖼞𖽡𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 , 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏 𖽐𖼮𖽷𖾐 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 .” 2 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼅𖽻𖾏 𖼚𖽠𖾏 𖼅𖽡 𖼐𖽤𖾐、𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏 𖼞𖽪𖾐 , 𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 .” 3 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽺𖾏 𖼚𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽜𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼎𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏, 𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 . 4 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼗𖽙𖾏 , 𖼘𖽑𖽡𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼁𖽠 , 𖼎𖽑𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑． 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ! 𖼊𖽪𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏.” 5 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼑𖽔𖾏 𖼁𖽙𖾏 , 𖼐𖽡𖽙𖾐 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼁𖽠 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘 , 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼀𖽡 𖼟𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 .” 6 𖼐𖽑𖽪𖾐 𖽃𖽔𖾏 𖼡𖽜𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 , 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 , 𖼎𖽺𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼑𖽻𖾏 𖼦𖽻𖾏 , 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼚𖽑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼇𖽜𖾏 .
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼦𖽡𖾑 ! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼗𖽱𖾏 𖽐𖼚𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏 . 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽺𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼑𖽜 𖼾𖽨𖾏 𖼘𖽑𖽡𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼁𖽠 , 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼐𖽑𖽡𖾏 : 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ! 𖼊𖽪𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 .” 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼑𖽔𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏 . 10 𖼟𖽡 𖼯𖽔𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 , 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼊𖽡𖽝𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 .”

𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼮𖽻𖾏 𖽐𖼮𖽻 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏
