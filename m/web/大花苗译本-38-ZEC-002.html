𖼞𖽪𖾑 𖼎𖽺𖾏 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼖𖽑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏. 2 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏?” 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖽐𖼮𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼑𖽔𖾏 𖼇𖽘𖾑 𖼀𖽡𖾑 𖼯𖽠 , 𖽐𖼊𖽡𖾑 𖼀𖽡𖾑 𖼯𖽠 .” 3 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼅𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼸𖽱𖾏 𖼐𖽱𖾏, 4 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼏𖽘 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏: 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 . 5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼚𖽻𖾐 𖼺𖽘𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 , 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .”
𖼳𖽡 𖼣𖽔𖾑 𖼊𖽻𖾏 𖼛𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖽃𖽺𖾐! 𖽃𖽺𖾐! 𖼞𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼅𖽡 𖼚𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 (“𖼚𖽻𖾐 𖼺𖽘𖾐”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼮𖽢𖾏”), 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼀𖽜𖾑 𖼺𖽷𖾐 𖼗𖽙, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.” 7 𖽃𖽺𖾐! 𖼳𖽡 𖼣𖽔𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 𖼘𖽪 ！ 𖼏𖽘 𖼀𖽜𖾑 𖼦𖽡𖾑 ! 8 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼊𖽻𖾏 𖼋𖽘𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼅𖽡 𖼙𖽠𖾏 𖽃𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼅𖽡, 𖼟𖽪𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼚𖽙 𖼖𖽔𖾑 𖼤𖽔𖾏 𖼤𖽔𖾏 𖽃𖽔 𖼅𖽔 . 9 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼇𖽜𖾏 𖼇𖽜𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏, 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼋𖽘𖾏.
