32 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼳𖽡𖾑 𖼦𖽪𖽾 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾐 , 𖼐𖽡𖽙𖾐 𖼽𖽔𖾐 𖼷𖽔 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 33 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼊𖽑𖽡𖽝𖾐 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 . 34 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼀𖽡𖾐 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽻 𖼺𖽘𖾏, 𖼞𖽑𖽡𖽠𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖽃𖽔𖾑 𖼾𖽠 𖽃𖽔 𖼁𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 , 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 , 𖼦𖽡𖾐 𖼷𖽷𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽡𖽝𖾏 . 35 𖼯𖽔 𖼅𖽔𖾏 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 , 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 . 36 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼐𖽺𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼑𖽡𖾏 𖼽𖽔𖾏 𖼘𖽙 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼙𖽻, 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼗𖽠𖾏 𖼋𖽘𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼊𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 . 37 𖼯𖽔 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼽𖽔𖾐 𖼀𖽙 𖽃𖽔𖾑 𖼟𖽱𖾐 , 𖼊𖽑𖽡𖽝𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼙𖽻, 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼀𖽡𖾐 𖼅𖽻𖾏 𖽃𖽡𖾑 , 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 .”
