𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼐𖽑𖽡𖾏 : 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 3 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 , 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽡𖾑 , 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏. 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
