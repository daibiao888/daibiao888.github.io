𖼞𖽑𖽻𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 16：23 - 33）
96
𖼅𖽡 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼮𖽑𖽦𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
2 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏！
𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏！
3 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏！
𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑！
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 , 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼮𖽵𖾑 𖼳𖽺 𖽃𖽔𖾑 𖼳𖽔𖾐 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏.
5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼁𖽠𖾏 𖼊𖽜𖾑,
𖼮𖽔𖾑 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏.
6 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼎𖽙 𖼚𖽔𖾐 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼎𖽙 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑.
 
7 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼞𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫．
8 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 ,
𖼯𖽡𖽙𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏.
9 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼐𖽑𖽘𖾑 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏．
 
10 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏！
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼼𖽙,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏.”
11 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼗𖽙𖾏 𖼳𖽦𖾑！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼙𖽔 𖼚𖽙𖾏 𖽐𖼷𖽑𖽷𖾏 𖼳𖽱𖾑！
12 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑！
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏.
13 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼋𖽘, 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏.
