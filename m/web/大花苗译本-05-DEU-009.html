4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 : 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼁𖽺𖾏 𖼋𖽔𖾏 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 , 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼐𖽔𖾑 . 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼅𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏, 𖼟𖽪𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽔𖾑 . 5 𖼟𖽡 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼳𖽪𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖼟𖽪𖾏 𖼑𖽜 𖼋𖽡𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼗𖽠𖾏 , 𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 、𖼽𖽡𖾑 𖼺𖽔𖾐 、𖼽𖽔𖾐 𖼞𖽙 𖼏𖽵 𖽃𖽡𖾑 .
（𖼋𖽠𖾏 32：1 - 35）
6 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼻𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 , 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼟𖽡 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏 . 7 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽡𖽙𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 , 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 . 𖼎𖽑𖽻𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 , 𖼸𖽙 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 , 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽐𖼷𖽜𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 8 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼦𖽙 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼅𖽡 , 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼅𖽡 𖼺𖽘𖾏 . 9 𖼞𖽪𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼙𖽠𖾏, 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 , 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏． 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽻𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼄𖽑𖽙𖾏 , 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼉𖽔𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼦𖽻𖾏 𖽃𖽻𖾐 . 10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 , 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼺𖽻𖾏 . 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 , 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 . 11 𖼮𖽑𖽡𖽻𖾏 𖼗𖽠𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼄𖽑𖽙𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 .
