𖽐𖼊𖽠𖾑 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏
𖼉𖽻𖾏 𖼑𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑
1
𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼗𖽪𖾏 .
2 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼐𖽑𖽡𖾏: 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼑𖽡𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 ,
𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 .
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽡𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏,
𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼑𖽡𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏, 𖽐𖼎𖽙 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?
4 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼎𖽺𖾏 𖼗𖽙𖾏 ,
𖼈𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽷𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
5 𖼖𖽪𖾐 𖼐𖽑𖽪𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏, 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼎𖽺𖾏 𖽐𖼷𖽜𖾑,
𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏.
6 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼎𖽻𖾐 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏,
𖼦𖽡𖾐 𖼋𖽡𖾐 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼘𖽙𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏, 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼞𖽘𖾐 𖼙𖽠𖾏.
7 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖽐𖼚𖽱𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑, 𖼳𖽡𖽠𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽪𖾑 ,
𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖽐𖼚𖽱𖾏 𖼗𖽙𖾏, 𖼎𖽺𖾏 𖼎𖽙𖾑 𖽐𖼚𖽱𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼙𖽠𖾏.
8 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼟𖽡𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽘 .
𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼇𖽔𖾑,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼑𖽻, 𖼑𖽻 𖼦𖽡𖾐 𖼷𖽻𖾏.
9 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼅𖽔𖾏, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼙𖽔 𖽃𖽔𖾏, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏．
𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐.
10 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼡𖽑𖽳𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪！𖽃𖽡𖾐 𖼼𖽘𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐．”
𖼋𖽡𖽝 𖼟𖽪𖾏, 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼖𖽡𖾏 𖽃𖽡𖾑 𖼷𖽹𖾏 𖼅𖽔𖾏 𖼋𖽘𖾏, 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽐𖼷𖽙𖾑 𖼙𖽔 𖼅𖽔𖾏 𖼋𖽘 .
11 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼗𖽙𖾏, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏.
𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼮𖽑𖽡𖽻𖾏
12 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏，𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏. 13 𖼞𖽪𖾑 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼡𖽘𖾐 𖽐𖼊𖽪𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏, 𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽱𖾏 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼎𖽘𖾏 𖼐𖽑𖽤𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏. 14 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼑𖽔𖾏 𖼘𖽪! 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑, 𖼟𖽪𖾏 𖼯𖽡𖽻𖾏 𖼮𖽢𖾏 𖼊𖽔𖾐.
15 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖽐𖼞𖽑𖽻𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽡𖽝,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠, 𖼦𖽡𖾏 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼮𖽔𖾑 𖼺𖽙𖾏.
16 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.” 17 𖼞𖽪𖾑 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳, 𖼽𖽔𖾑 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼙𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼡𖽔 𖼏𖽔, 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽻𖾏 𖼮𖽢𖾏 𖼊𖽔𖾐.
18 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳, 𖼎𖽺𖾏 𖼮𖽡𖽻𖾐 𖼅𖽔𖾏 𖽐𖼊𖽱𖾏 𖼊𖽳𖾏,
𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼯𖽡𖾏, 𖼎𖽺𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏.
