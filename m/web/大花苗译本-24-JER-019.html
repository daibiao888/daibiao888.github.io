𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: “𖼟𖽡 𖼅𖽻𖾏 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑,𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽪𖾐 𖼗𖽻 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏. 2 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼐𖽺𖾏 𖼷𖽷𖾐 𖼖𖽪𖾑 𖼦𖽘𖾑, 𖼦𖽔𖾐 𖽃𖽳𖾐 𖼳𖽡𖾑 (“𖼦𖽔𖾐 𖽃𖽳𖾐 𖼳𖽡𖾑 ” 𖼟𖽪𖾏 “𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼉𖽔𖾏”) 𖽃𖽔 𖼛𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡. 3 𖼐𖽑𖽡𖾏: 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐𖼘𖽪! 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑,𖼑𖽔𖾏 𖼘𖽪!𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽙𖾑𖽑 𖽃𖽡𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼊𖽪𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙𖾑. 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏, 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼑𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺, 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐, 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼀𖽪𖾑 𖼋𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑. 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼷𖽱𖾏 𖼀𖽔𖾑 𖼖𖽡 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐, 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽑𖽷𖾑 𖼾𖽨𖾏 𖼊𖽪𖾐, 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼀𖽔𖾑 𖼖𖽡, 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼸𖽙, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏. 6 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽑𖽙 𖼇𖽺𖾑 𖼊𖽑𖽳 𖼡𖽑𖽙𖾑 𖼯𖽢𖾏 𖼳𖽡𖾑 𖼐𖽺𖾏 𖼷𖽷𖾐 𖼖𖽪𖾑 𖼦𖽘𖾑, 𖼎𖽙𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 7 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑, 𖼟𖽙𖾏𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼚𖽔𖾏 𖼯𖽢𖾏 𖼙𖽠𖾏, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐, 𖼢𖽻 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏, 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏. 8 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑, 𖼷𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏, 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏, 𖼽𖽔𖾑 𖼑𖽜 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼊𖽻𖾏,𖼊𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏,𖼎𖽺𖾏 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏 𖼐𖽱𖾏. 9 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,𖼟𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼑𖽻𖾏 𖼾𖽨𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖽐𖼡𖽹𖾏, 𖽐𖼎𖽙 𖼽𖽺 𖼗𖽔 𖽐𖼡𖽹𖾏.
