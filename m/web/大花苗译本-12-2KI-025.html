8 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻𖾏 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏 𖼖𖽪𖾑 𖼀𖽜𖾐 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼳𖽤𖾏, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑, 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐, 9 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻, 𖼎𖽺𖾏 𖼷𖽑𖽷𖾑 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽻 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏. 10 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑, 𖼎𖽺𖾏 𖼎𖽑𖽔𖾑 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻. 11 𖼊𖽑𖽻𖾏 𖼐𖽡𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽪𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽗𖾏, 𖼄𖽔𖾐 𖼙𖽻 𖽐𖼚𖽙 𖼷𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏. 12 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼷𖽷𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾏 𖽃𖽡𖾑 𖼮𖽡𖽝𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽢𖾏 𖼨𖽳𖾑𖽑 𖼉𖽘𖾏 𖼷𖽷𖾐 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏, 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐. 13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼋𖽻𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼖𖽪𖾑 𖼋𖽺𖾏 𖽐𖼎𖽙 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏, 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽔𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼋𖽪 𖼺𖽘𖾏, 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼋𖽻𖾏 𖽃𖽡𖾑 𖼨𖽑𖽹𖾑 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏. 14 𖼞𖽑𖽡𖽠𖾏 𖼀𖽑𖽻𖾏、 𖼚𖽡𖽝𖾑 𖼞𖽻𖾏、 𖼷𖽑𖽳𖾏、 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼋𖽺𖾏 𖼚𖽡𖽝𖾑、 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖼖𖽔𖾏 𖼞𖽑𖽔𖾏 𖼯𖽡𖽙𖾏 𖼙𖽠𖾏， 15 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐、 𖼞𖽑𖽙𖾐， 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼞𖽪𖾐、 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏, 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖼯𖽡𖽙𖾏 𖼙𖽠𖾏. 16 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖽐𖼮𖽡𖾏 𖼋𖽻𖾏, 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽪𖾐 𖼋𖽺𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽻𖾏 𖼐𖽡𖾑, 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼙𖽪𖾏 𖼋𖽘 𖼦𖽡𖾐 𖼋𖽻𖾏. 17 𖼊𖽪𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼐𖽡𖾑 𖼺𖽡𖽝𖾐 𖼟𖽻𖾏 𖼾𖽡𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼡𖽔𖾏 𖼘𖽙𖾏 𖼋𖽻𖾏 , 𖼺𖽡𖽝𖾐 𖼷𖽷𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼋𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼅𖽔𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑, 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖽃𖽔𖾏. 𖼊𖽪𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖽃𖽡𖾑, 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏.
𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼇𖽳𖾑 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏
（𖼽𖽦𖾑 52： 24 - 27）
18 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼳𖽡𖾑 𖼖𖽹 𖼽𖽔𖾐 , 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼳𖽡𖾑 𖼇𖽔𖾑 𖼽𖽔𖾐 𖽐𖼎𖽙 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼻𖽙𖾑 𖽃𖽔 𖼛𖽻𖾏, 19 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼄𖽔𖾐 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼗𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏(“𖼉𖽻 𖼷𖽻𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏”) 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼏𖽙 , 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼙𖽻 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽑𖽪𖾏, 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼙𖽻. 20 𖼐𖽡 𖼀𖽪𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼄𖽔𖾐 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼨𖽑𖽳𖾏 𖼚𖽔𖾏 𖼖𖽡𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 21 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾏 𖼎𖽻𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼖𖽡𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑. 𖽃𖽔𖾏 𖼐𖽡𖾑, 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼇𖽳𖾑 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏.
