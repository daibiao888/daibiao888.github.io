𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼀𖽡𖾐 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽐𖼮𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼳𖽔𖾑． 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼊𖽳 𖼖𖽹 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏！ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼟𖽡 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼦𖽪𖽾 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼳𖽷𖾏 𖼀𖽺𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 . 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼞𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼀𖽡𖾐 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 , 𖼦𖽡𖾐 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽡𖽝𖾏 . 4 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼀𖽡𖾐 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽻 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼗𖽪𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼀𖽡𖾐 𖼦𖽔𖾏 𖼗𖽙𖾏． 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼞𖽙𖾑 𖼀𖽳 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼟𖽪𖾏 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽳𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 . 5 𖼋𖽡𖾏 𖼙𖽻 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼏𖽔𖾏 , 𖼅𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 , 𖼅𖽔𖾏 𖼎𖽪𖾑 𖽃𖽔 𖼛𖽻𖾏． 𖼋𖽠𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 . 6 𖼀𖽡𖾐 𖼞𖽑𖽔𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏 , 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼳𖽡𖾑 𖼳𖽷𖾏 𖼀𖽺𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼦𖽪𖽾 𖽃𖽡𖾐 𖼼𖽘𖾐 , 𖼞𖽑𖽡𖽠𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼾𖽠 𖽃𖽔 𖼁𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼣𖽔𖾑 𖼾𖽻𖾏 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 . 7 𖼯𖽔 𖼅𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 , 𖼞𖽑𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 . 8 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼽𖽔𖾏 𖼐𖽺𖾏 𖼖𖽪𖾐 𖼦𖽘𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼦𖽳 𖼄𖽞 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐， 9 ( 𖼖𖽪𖾑 𖼦𖽳 𖼄𖽞 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑 , 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖽃𖽔𖾏 𖼳𖽡𖾑 𖼖𖽡𖽝 , 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼐𖽡 𖽃𖽳𖾐 ). 10 𖼀𖽡𖾐 𖼦𖽔𖾏 𖼊𖽻𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽻 , 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼺𖽔𖾐 𖼮𖽢𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼊𖽳 𖼖𖽹 , 𖼟𖽪𖾏 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽳𖾐 𖼖𖽪𖾑 𖼞𖽪𖽙 𖽐𖼚𖽙 𖽃𖽡𖾑 𖼙𖽻. 11 ( 𖼖𖽡𖾏 𖼇𖽔 𖼽𖽡𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 , 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽳𖾐 . 𖼐𖽱𖾏 𖼸𖽱𖾏 𖼸𖽘𖾏 𖼟𖽪𖾏 𖼖𖽑𖽻𖾏 𖽃𖽔𖾏 , 𖽐𖼊𖽡𖾑 𖼯𖽢𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼇𖽘𖾑 𖼚𖽻𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 𖽃𖽔𖾏 𖽐𖼮𖽔𖾏 , 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼙𖽻 𖼘𖽪𖾏? )
𖼇𖽺𖾐 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽡𖾐 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖾏 𖽐𖼊𖽘𖾏

12 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼀𖽡𖾐 𖼳𖽪𖾏 𖼊𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏 . 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼐𖽺𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼽𖽔𖾏 𖼘𖽙 𖽃𖽳𖾐 𖼳𖽠𖾑 𖼁𖽱𖾏 , 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 , 𖼊𖽑𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼙𖽻 , 𖼞𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼖𖽨 𖼀𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼊𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐． 13 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼟𖽪𖾏 𖽃𖽳𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 , 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖽃𖽡𖾐 𖽐𖼊𖽘𖾏 𖼳𖽡𖾏 . 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼞𖽙𖾑 𖼀𖽳 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏𖼀𖽔𖾑 𖼳𖽔𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 ,𖼋𖽡𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼇𖽔 𖼽𖽡𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 . 14 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼐𖽡𖽙𖾐 𖼀𖽪𖾑 𖼺𖽘𖾏 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼞𖽙𖾑 𖼀𖽳 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼳𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼄𖽔𖾐 𖼮𖽢𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖽐𖼷𖽷𖾏 𖼨𖽑𖽪𖾏 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼦𖽔𖾐 𖽂𖽙 𖼊𖽑𖽳 · 𖼽𖽔𖾏 𖽃𖽳𖾐（𖼟𖽪𖾏 “𖼽𖽔𖾏 𖽃𖽳𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻”） , 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 . 15 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼄𖽔𖾐 𖼮𖽡． 16 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼐𖽺𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 . 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖽃𖽔𖾏 𖽐𖼚𖽜𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼀𖽙 𖽃𖽔𖾑 𖼟𖽱𖾐 . 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼖𖽨 𖼀𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼊𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏， 17 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼽𖽔𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 , 𖼎𖽑𖽻𖾏 𖼮𖽡𖾑 𖼐𖽡 𖼖𖽡𖽝 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 , 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼮𖽳𖾑 , 𖼊𖽑𖽡𖽝𖾐 𖼀𖽑𖽡𖾐 𖼺𖽷𖾑 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼎𖽻𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼞𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 .
