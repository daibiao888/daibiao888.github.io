𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐
9
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏，𖼐𖽱𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼗𖽙 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼷𖽹𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼋𖽘． 2 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼖𖽔𖾑 𖼀𖽡𖾐，𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼐𖽔𖾑？𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼖𖽔𖾏？𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼐𖽔𖾑？𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼗𖽙 𖼷𖽹𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼋𖽘．” 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼦𖽡𖾐 𖼾𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼾𖽔𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼑𖽪𖾏． 4 𖼷𖽪𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼺𖽷𖾐，𖼀𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼑𖽡𖾏 𖼑𖽪𖾏，𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼋𖽘𖾏． 5 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽪𖾑 𖼎𖽻𖾐．” 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏，𖼎𖽺𖾏 𖽐𖼊𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐，𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖽻𖾐 𖽐𖼮𖽨𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽻𖾏， 𖼚𖽜𖾑 𖼎𖽻𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖽃𖽔𖾑 𖼅𖽔． 7 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏：“ 𖼟𖽡 𖼅𖽻𖾏 𖽐𖼷𖽔𖾑 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼘𖽙 𖼽𖽔𖾏 𖼁𖽘𖾏 𖽃𖽻𖾐．” （𖼳𖽡𖾑 𖼘𖽙 𖼽𖽔𖾏 𖽐𖼷𖽜𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼟𖽪𖾏“ 𖼷𖽷𖾐 𖼋𖽔𖾏”）． 𖼐𖽱𖾏 𖼅𖽻𖾏 𖽐𖼷𖽔𖾑 𖼋𖽘𖾏，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼺𖽷𖾐 𖼗𖽙 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼮𖽡𖾑 𖼋𖽘𖾏． 8 𖼼𖽷𖾏 𖼼𖽙 𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼳𖽻𖾏 𖼡𖽜𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼳𖽻𖾏 𖼡𖽜𖾑 𖽃𖽡𖾑 𖼘𖽪？” 9 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏．”𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼦𖽡𖾐 𖼾𖽙，𖼟𖽪𖾏 𖼺𖽱𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏．”𖼐𖽱𖾏 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏．” 10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏？” 11 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖽐𖼮𖽨𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽻𖾏 𖼚𖽜𖾑 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔，𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼅𖽻𖾏 𖽐𖼷𖽔𖾑 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼘𖽙 𖼽𖽔𖾏 𖼁𖽘𖾏 𖽃𖽻𖾐 ’．𖼞𖽪𖾑 𖼅𖽻𖾏 𖽐𖼷𖽔𖾑 𖼋𖽘𖾏 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑．” 12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏？” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐．”
𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖽐𖼮𖽜
13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽳𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖽃𖽡𖾑 𖼚𖽔𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 14 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽨𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽻𖾏，𖼡𖽑𖽜𖾐 𖼐𖽔𖾐 𖼄𖽱𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏． 15 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽻𖾏 𖼚𖽜𖾑 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔，𖼞𖽪𖾑 𖼅𖽻𖾏 𖽐𖼷𖽔𖾑 𖼋𖽘𖾏 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑．” 16 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔𖾏．”𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑？”𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏． 17 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔，𖼟𖽡 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏？”𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐．”

40 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 ，𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼯𖽡𖽙 𖼐𖽔𖾑？” 41 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐， 𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏，𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽡 𖼐𖽑𖽡𖾏：‘𖼀𖽡𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼯𖽡𖽙𖾏 ’，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼾𖽔𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖽝𖾏 𖼺𖽷𖾐．”
