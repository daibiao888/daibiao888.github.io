16
𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼊𖽪𖾑 𖼞𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼽𖽔𖾑 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏． 2 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽪𖾏 𖼖𖽡𖽝𖾐，𖼅𖽡 𖼐𖽑𖽡𖾏：‘ 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽑𖽘𖾑 𖽐𖼊𖽪𖾏．’ 3 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖽐𖼊𖽪𖾏 𖼖𖽡𖽝𖾐，𖼎𖽺𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼖𖽡𖾐 𖽐𖼮𖽡𖽝𖾐，𖼅𖽡 𖼐𖽑𖽡𖾏：‘ 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼑𖽘 𖼮𖽢𖾏．’ 𖼅𖽡 𖼇𖽺𖾐 𖼊𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽘𖾏 𖼮𖽢𖾏 𖼼𖽘𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼇𖽺𖾐 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼐𖽡𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑． 4 𖼷𖽑𖽷𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼦𖽡𖾐 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼐𖽡𖾑 𖽐𖼊𖽜 𖼑𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑，𖼋𖽠𖾏 𖼽𖽙 𖼐𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼑𖽔𖾏．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏．
𖽃𖽔𖾑 𖼋𖽪 𖼳𖽻𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏
（𖼞𖽑𖽙𖾐 8：14-21）
5 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼀𖽡 𖼋𖽡𖾏，𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼯𖽡𖽙𖾏 𖼳𖽡𖽝𖾐． 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 ： “𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽡𖾑， 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼊𖽪𖾑 𖼞𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏．” 7 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏： “𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼯𖽡𖽙𖾏 𖼳𖽡𖽝𖾐 𖼐𖽔𖾑 𖼁𖽔𖾏．” 8 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼯𖽠 𖼐𖽡𖾑，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼳𖽡𖽝𖾐 𖼖𖽺𖾑？ 9 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼺𖽷𖾐 𖼘𖽪𖾏？ 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼀𖽜𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐，𖼇𖽺𖾐 𖼎𖽻𖾏 𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏，𖼎𖽺𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼮𖽡𖽺𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼘𖽪𖾏？ 10 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼳𖽤𖾏 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐，𖼇𖽺𖾐 𖼎𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼎𖽺𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼯𖽠 𖽃𖽔𖾐 𖼊𖽠𖾐 𖽃𖽔𖾑 𖼁𖽱 𖼘𖽪𖾏？ 11 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽡𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼊𖽪𖾑 𖼞𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏，𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼳𖽡𖽝𖾐，𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼺𖽷𖾐 𖼘𖽪？” 12 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽡𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼳𖽡𖽝𖾐 𖼑𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 ， 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽡𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼊𖽪𖾑 𖼞𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏．
𖼀𖽡𖾐 𖼊𖽳 𖼯𖽜𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪
（𖼞𖽑𖽙𖾐 8：27-30；𖼖𖽪𖾏 9：18-21）
13 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 · 𖼇𖽺𖾑 𖼖𖽡 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏， 𖼐𖽱𖾏 𖼑𖽪 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？” 14 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐．” 15 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？” 16 𖼳𖽡𖾑 𖼄𖽞 ·𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪，𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐．” 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼳𖽡𖾑 𖼄𖽞 𖼀𖽔𖾑 𖼽𖽙 𖼐𖽔，𖼟𖽡 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽙𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼀𖽻𖾐， 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼀𖽻𖾐． 18 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 ， 𖼟𖽡 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽳，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼐𖽡𖾑 𖼮𖽡𖽝𖾏，𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽠𖾏 𖼐𖽱𖾏． 19 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼚𖽺𖾏 𖽐𖼮𖽦𖾏 𖼀𖽑𖽪𖾑 𖼎𖽻𖾏 𖼟𖽡，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖽃𖽡𖾑，𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼯𖽡𖽙，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼚𖽠𖾏 𖼮𖽙𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑，𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼺𖽘𖾏 𖼯𖽡𖽙𖾏．” 20 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
（𖼞𖽑𖽙𖾐 8：31-9：1；𖼖𖽪𖾏 9：22-27）
21 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼁𖽱𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏，𖼾𖽨𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽻，𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽪𖾐 𖼗𖽻、 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏，𖼸𖽙 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙． 22 𖼀𖽡𖾐 𖼊𖽳 𖼊𖽔𖾑 𖼐𖽱𖾏，𖼖𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪，𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑！𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼷𖽠𖾏 𖼟𖽡． ” 23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“𖼺𖽔𖾐 𖼊𖽔 ，𖼑𖽡𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏！𖼟𖽡 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼞𖽪𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖽐𖼊𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪𖾏，𖼯𖽔 𖽐𖼊𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪𖾏 𖼊𖽔𖾐．” 24 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼾𖽨𖾏 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼾𖽨𖾏，𖼽𖽔𖾑 𖼎𖽷𖾏 𖼾𖽨𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐． 25 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏，𖼡𖽔 𖼋𖽱𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼛𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑． 26 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼊𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼛𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏，𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪？𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼋𖽔𖾏 𖼘𖽑𖽪𖾑 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼖𖽺𖾑？ 27 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼚𖽙 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏，𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏． 28 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽑𖽡𖽝𖾏 𖼋𖽔 𖽐𖼷𖽑𖽡𖽝𖾐 𖼺𖽷𖾐，𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏．”
