𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔
（𖽐𖼞𖽻𖾏 18）
22
𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽨𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 2 𖼐𖽑𖽡𖾏:
“ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼷𖽔𖾏, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼾𖽡 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼮𖽵𖾑 ,
3 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏．
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐, 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 ,
𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼷𖽜𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔.
𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼮𖽵𖾑, 𖼟𖽡 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 .
4 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖽃𖽡𖾑,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
 
5 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼑𖽡𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑 ,
𖼖𖽻𖾐 𖼷𖽑𖽪𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖽐𖼚𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏，
6 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼦𖽡𖾏 𖼚𖽱𖾏 𖼞𖽪𖾑 ,
𖼛𖽙𖾏 𖼋𖽔𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 .
 
7 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾐,
𖼞𖽪𖾑 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏.
 
8 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼊𖽡𖾐 𖼎𖽺𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 ,
𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏 𖼊𖽪𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 .
9 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖽐𖼀𖽱 𖼋𖽠𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐,
𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑,
𖽃𖽔𖾑 𖽐𖼮𖽡 𖼊𖽑𖽜𖾏 𖼖𖽔𖾏 𖼎𖽺𖾏 𖼯𖽡𖽻𖾏 𖼋𖽘𖾏.
10 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼚𖽻𖾑 𖼗𖽙𖾏, 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼎𖽺𖾏 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏,
𖼅𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼚𖽪𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
11 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳 𖼽𖽘𖾏 𖽐𖼮𖽡,
𖼐𖽡𖽙𖾐 𖼮𖽢𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼢𖽳 𖼋𖽠𖾏.
12 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱,
𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽔𖾐, 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼟𖽻 𖼑𖽡𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐.
13 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼊𖽑𖽜𖾏 𖼖𖽔𖾏 𖼯𖽡 𖼋𖽘𖾏.
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽙𖾏 𖼺𖽙𖾐 𖼢𖽔𖾏,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏.
15 𖼐𖽱𖾏 𖼀𖽙𖾑 𖼋𖽠𖾏 𖼈𖽙𖾏 𖼗𖽙𖾏, 𖼟𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏,
𖼮𖽡𖾏 𖼋𖽠𖾏 𖼺𖽙𖾐 𖼀𖽡𖾏 𖼗𖽺𖾏 𖽐𖼊𖽘𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾐 𖼘𖽑𖽔𖾐 𖼀𖽡𖾏 𖼘𖽑𖽡𖾏.
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾏 𖼡𖽑𖽔𖾐 𖽂𖽪𖾏 𖼋𖽠𖾏,
𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔 𖽐𖼀𖽱 𖼀𖽘𖾏 𖽂𖽪𖾏 𖼋𖽠𖾏, 𖼡𖽘𖾐 𖼳𖽡𖽠𖾑 𖼎𖽺𖾏 𖼢𖽳 𖼋𖽠𖾏,
𖼊𖽡𖾐 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏 𖼖𖽔𖾏 𖼢𖽳 𖼋𖽠𖾏.
 
17 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼊𖽔𖾑 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏,
𖼄𖽔𖾐 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽤𖾐 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏.
18 𖼐𖽱𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑 .
19 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑,
𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼻𖽡𖽝𖾏.
20 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼇𖽘𖾑,
𖼐𖽱𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼗𖽙𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼞𖽪𖾑 .
 
21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑,
𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
22 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑,
𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
23 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼐𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽑𖽘𖾑 𖼢𖽔𖾏 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼇𖽳𖾑 𖼺𖽘𖾏.

30 𖼞𖽪𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼟𖽡 𖽐𖼮𖽵 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼚𖽙,

42 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,

44 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏,
