𖼐𖽙 𖼽𖽔𖾏 𖼮𖽙𖾏 𖼊𖽪𖾑 𖼖𖽡𖾐 𖽃𖽔𖾐 𖼊𖽑𖽡𖽝𖾐 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐
6 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐, 𖼐𖽙 𖼽𖽔𖾏 𖼡𖽑𖽜𖾐 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏， 7 𖼮𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾑 𖽃𖽔𖾐 𖼅𖽻𖾏； 𖼊𖽪𖾑 𖼖𖽡𖾐 𖽃𖽔𖾐 𖽃𖽡𖾑 𖼽𖽘𖾏 𖼅𖽻𖾏 𖼽𖽘𖾏 𖼗𖽙𖾏, 𖼋𖽙𖾏 𖼸𖽙 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽻𖾐 𖼞𖽑𖽔𖾏 𖽐𖼡𖽑𖽔𖾐 𖼋𖽘. 8 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼟𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽐𖼡𖽔𖾐 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏 𖼽𖽘𖾏 𖼅𖽻𖾏, 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖽃𖽻𖾐 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽐𖼡𖽜𖾏 𖼋𖽘𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐. 9 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼖𖽔𖾏 𖼅𖽔𖾏 𖽃𖽻𖾐, 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼷𖽠𖾏, 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼗𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼐𖽙 𖼽𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽙 𖼽𖽔𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼅𖽻𖾏 𖼷𖽹𖾏 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼙𖽠𖾏. 10 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼋𖽙𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼮𖽙𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏. 11 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽳𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽗𖾐 𖼖𖽗 𖽐𖼊𖽻𖾏 𖽃𖽔 𖽐𖼚𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼚𖽡𖾏 𖼟𖽡𖾏 𖼮𖽑𖽦𖾐 𖼞𖽡𖾑𖽑, 𖼐𖽙 𖼽𖽔𖾏 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽻𖾐 𖽐𖼡𖽜𖾏 𖼋𖽘𖾏. 12 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼋𖽙𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐, 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏, 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼺𖽷𖾐 𖼗𖽙 𖼋𖽘𖾏. 13 𖼸𖽙 𖼐𖽙 𖼽𖽔𖾏 𖼚𖽻𖾏 𖼀𖽔𖾏 𖼖𖽡 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏, 𖼖𖽪𖾑 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐, 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽻𖾐 𖼞𖽑𖽔𖾏 𖽐𖼡𖽑𖽔𖾐 𖼋𖽘𖾏. 𖼐𖽙 𖼽𖽔𖾏 𖼡𖽑𖽜𖾐 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼺𖽘𖾏 𖼑𖽔𖾏, 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖼋𖽘. 14 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽃𖽔𖾐 𖼖𖽑𖽡𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽤𖾏, 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼞𖽑𖽔𖾏 𖽐𖼡𖽑𖽔𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏.
𖼐𖽙 𖼽𖽔𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼋𖽠𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏
