𖼗𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏
26
𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽙𖾑 𖼗𖽙𖾏 𖽐𖼀𖽪𖾏, 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼡𖽻𖾐 𖼑𖽘 𖼗𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏,
𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐, 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏.
2 𖼀𖽡𖾏 𖼮𖽑𖽡𖽙𖾑 𖼽𖽘𖾏 𖼅𖽻𖾏 𖼽𖽘𖾏 𖼗𖽙𖾏, 𖼐𖽻 𖼙𖽡𖽝𖾏 𖼞𖽡𖾏 𖼘𖽪𖾏 𖼽𖽘𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽜𖾑；
𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐, 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽐𖼮𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼗𖽙𖾏 𖼷𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
3 𖼊𖽪𖾑 𖽐𖼚𖽠𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖽐𖼚𖽠𖾏 𖼑𖽜𖾏, 𖼖𖽑𖽔𖾏 𖼺𖽻𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼺𖽻𖾑 𖼑𖽜𖾏 𖼖𖽔𖾑 ；
𖼊𖽪𖾑 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼑𖽜 𖽐𖼊𖽻𖾏 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼢𖽻𖾏,
4 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼊𖽡𖾐 𖼐𖽱𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼟𖽡 𖼺𖽱𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐.
5 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼊𖽡𖾐 𖼐𖽱𖾏 ,
𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
6 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼺𖽘𖾏 𖼄𖽻𖾏,
𖼟𖽪𖾏 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼾𖽨𖾏 𖽐𖼞𖽡𖽠𖾏 𖼊𖽠𖾏, 𖼾𖽨𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ( “𖼾𖽨𖾏 𖼊𖽻𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖼦𖽻𖾏”).
7 𖼊𖽪𖾐 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 𖼅𖽔𖾏,
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 ,
𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼞𖽡𖾏 𖼘𖽪𖾏.
9 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼐𖽡𖽙𖾐 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏,
𖼺𖽱𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼮𖽑𖽡𖽙𖾐 𖼊𖽪𖾐 𖼢𖽻 𖼮𖽡𖽠𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 , 𖼎𖽺𖾏 𖼋𖽠𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 𖽃𖽡𖾑,
𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽙𖾑 𖼐𖽜𖾑𖽑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼀𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽻𖾏 𖼄𖽙𖾐.
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔 𖼏𖽔 , 𖽃𖽔𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏,
𖼺𖽱𖾏 𖼚𖽡𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑.
12 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼙𖽔 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼘𖽪𖾏 ?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏 .
13 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽜 𖼐𖽑𖽡𖾏 : “𖼸𖽙𖾏 𖼞𖽡𖾑 𖼅𖽔𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼖𖽪𖾑 𖼟𖽡𖾏 𖼅𖽔𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽻𖾏.”
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽜 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽜𖾑,
𖼺𖽱𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼊𖽡𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼅𖽻 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖽃𖽡𖾐 𖼼𖽘𖾐.
15 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼚𖽙 𖼋𖽺𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼚𖽑𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
16 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽜 𖼑𖽔𖾏 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳
𖼖𖽔𖾑 𖼊𖽪𖾑 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖼸𖽠𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼸𖽠𖾏 𖼊𖽡𖾐.
17 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽻𖾏 𖼑𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼘𖽙𖾏 𖼏𖽻 ,
𖼟𖽪𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼊𖽔𖾑 𖼚𖽡𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏.
18-19 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼼𖽷𖾏 𖼼𖽙 , 𖼈𖽡𖽝𖾏 𖼐𖽑𖽡𖾏 :
“𖼞𖽪𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼚𖽙𖾏 𖼊𖽔𖾐.”
𖼐𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼩𖽻𖾏 𖼇𖽜𖾏 𖼇𖽜𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏、𖼈𖽙𖾏 𖼟𖽪𖾏 𖽐𖼮𖽳𖾏 𖽐𖼎𖽙 𖼏𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 (“𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖼛𖽙𖾏 𖼋𖽔” ).
20 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽠𖾏 𖼯𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖼋𖽔 ；
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼺𖽹𖾑 𖼗𖽪𖾏, 𖼗𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏.
21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽔 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼙𖽳𖾐 𖼷𖽱𖾏 𖼑𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼺𖽱𖾏 𖼊𖽑𖽜𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡 𖼋𖽠𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑, 𖼺𖽱𖾏 𖼋𖽠𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐.
22 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼺𖽹𖾑 𖼗𖽪𖾏 , 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼺𖽱𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾐 𖼚𖽘𖾐.
23 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾐 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼺𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏.
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼗𖽔, 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼾𖽨𖾏,
𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏.
25 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼐𖽱𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼳𖽪𖾐.
26 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 .
27 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖼘𖽑𖽻𖾑 𖼮𖽡𖽝𖾏, 𖼾𖽨𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼙𖽠𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖽃𖽔𖾑 𖼈𖽳𖾐, 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽙𖾑 𖼦𖽡 𖽐𖼚𖽙 𖼷𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
28 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼳𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽹𖾏 𖼯𖽵𖾏 𖽃𖽡𖾑；
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼚𖽡𖽝 , 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑.
