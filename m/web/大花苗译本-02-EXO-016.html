𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡𖽾 𖼳𖽠𖾑 𖼅𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼖𖽪𖾑 𖽃𖽔𖾐 𖼖𖽑𖽡𖾏 𖼟𖽻𖾏 𖼀𖽜𖾐 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 , 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼖𖽡𖽾 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼳𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 . 2 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 , 3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼷𖽙𖾑 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼀𖽡𖾐 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼀𖽑𖽻𖾏 𖽐𖼡𖽹𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼉𖽔𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽻𖾏．𖼅𖽡 𖼄𖽔𖾐 𖼀𖽡𖾐 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼐𖽡𖾑 , 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼞𖽑𖽔𖾏 𖼮𖽑𖽹𖾐 𖼋𖽔 𖼘𖽪𖾏!”
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼑𖽔𖾏 𖼘𖽪 ! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼮𖽙𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼅𖽡 , 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 , 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼳𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖾏 𖽐𖼀𖽠 , 𖼞𖽪𖾑 𖼻𖽻𖾏 𖽐𖼮𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽙𖾑 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 . 5 𖼸𖽙 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 𖼮𖽡𖽝𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖽻𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖽃𖽡𖾐 𖽐𖼀𖽠 .” 6 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : “𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏 . 7 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 , 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼗𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼋𖽘𖾏 , 𖼀𖽡𖾐 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼅𖽡 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼖𖽺𖾑 ?” 8 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾑 𖼄𖽔𖾐 𖽐𖼡𖽹𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼑𖽻𖾏 , 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼟𖽙𖾏 𖼑𖽻𖾏 𖼮𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼗𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼐𖽱𖾏 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏 . 𖼀𖽡𖾐 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼅𖽡 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 , 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .”
