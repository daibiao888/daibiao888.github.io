3
“𖼟𖽡 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼺𖽔𖾐 𖼊𖽡 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐， 𖼐𖽑𖽡𖾏：‘ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼖𖽡𖽾 𖽐𖼎𖽙 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏，𖼟𖽡 𖼅𖽔𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝． 2 𖼟𖽡 𖼽𖽔𖾑 𖼯𖽷𖾏 𖼨𖽑𖽪𖾑，𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏，𖼈𖽡𖽝𖾏 𖼖𖽔𖾑 𖼋𖽔 𖼋𖽘𖾏 𖽃𖽡𖾑 ，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏， 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏． 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼟𖽡 𖼽𖽔𖾑 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼷𖽹𖾏 𖼊𖽻𖾏、𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏； 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏，𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 ．𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼯𖽷𖾏 𖼨𖽑𖽪𖾑 ，𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼸𖽜 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 ．𖼞𖽪𖾑 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼀𖽻𖾐 ． 4 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼺𖽔𖾐 𖼊𖽡， 𖼟𖽡 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼮𖽑𖽙𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼊𖽻𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼚𖽠𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏 ，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼸𖽷𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 ． 5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏， 𖼽𖽔𖾑 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼞𖽪𖾑 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼇𖽳𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏；𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼯𖽜𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏． 6 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼐𖽑𖽡𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻．’ ”
𖼄𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼇𖽺𖾑 𖼖𖽔𖾑 𖼊𖽑𖽡𖽝 𖼇𖽺𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏
7 “𖼟𖽡 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼇𖽺𖾑 𖼖𖽔𖾑 𖼊𖽑𖽡𖽝 𖼇𖽺𖾑 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐， 𖼐𖽑𖽡𖾏：
‘ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏、𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝、
𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼚𖽺𖾏 𖽐𖼮𖽦𖾏 𖼀𖽑𖽪𖾑，
