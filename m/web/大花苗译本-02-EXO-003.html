𖼄𖽙𖾑 𖼳𖽡𖾑 𖼾𖽨 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽦 𖼊𖽑𖽳 𖼘𖽙 𖼑𖽡𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 , 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 , 𖼐𖽱𖾏 𖼖𖽠𖾑 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽔𖾏 𖼙𖽠𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼦𖽙 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 . 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼎𖽑𖽻𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼢𖽳 𖼎𖽻𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑． 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 , 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼯𖽡 , 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 . 3 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼖𖽔𖾑 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼊𖽺𖾐 𖼗𖽘𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼐𖽡𖾑 , 𖼸𖽻𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼖𖽺𖾑 ?” 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼅𖽻𖾏 𖼑𖽔𖾏 , 𖼎𖽑𖽻𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼄𖽙𖾑 𖼳𖽡𖾑 ! 𖼄𖽙𖾑 𖼳𖽡𖾑 !” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 .” 5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 , 𖼚𖽠𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼞𖽑𖽻𖾏 𖼺𖽘𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .” 6 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏、𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏、𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .” 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
𖼷𖽷𖾐 𖼟𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼅𖽻𖾏 𖼨𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽻𖾏 , 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏, 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏 , 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 . 8 𖼞𖽪𖾑 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 , 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼄𖽡𖾏 𖽐𖼎𖽙 𖼄𖽪𖾑 . 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼦𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼀𖽡𖾐 𖼖𖽡𖾏 𖼳𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼳𖽡𖾑 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 . 9 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏 𖽐𖼎𖽙𖾑 𖼸𖽙 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 , 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼷𖽷𖾐 𖼟𖽡 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 , 𖼟𖽙𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .”
