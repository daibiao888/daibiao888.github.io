12
𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼖𖽳 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼗𖽙𖾏 𖼸𖽙 𖼀𖽘𖾏 𖼐𖽡𖾑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖽐𖼊𖽠𖾑 𖼮𖽡𖽝𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏. 2 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼯𖽡𖽝𖾏 𖼗𖽙, 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏, 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼊𖽻𖾏 𖼳𖽪𖾐. 3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪 , 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼮𖽡𖾏； 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪 𖼺𖽱𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏. 4 𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 𖼘𖽪! 𖼟𖽡 𖼽𖽔𖾑 𖼉𖽺𖾏 𖼮𖽡𖽝𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑, 𖼘𖽙𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼮𖽡𖽝𖾏, 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏. 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼏𖽘 𖼅𖽻𖾏 𖼏𖽘 𖼗𖽙𖾏 , 𖼀𖽻𖾐 𖼟𖽳 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽑𖽨𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
5 𖼞𖽪𖾑 𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 𖼎𖽺𖾏 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼀𖽘𖾑𖽑 𖼐𖽡𖾑, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼀𖽘𖾑𖽑 𖽃𖽡𖾑. 6 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼑𖽪 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼀𖽡𖾑 𖼳𖽔𖾐, 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖽃𖽔𖾑 𖼗𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼖𖽺𖾑?” 7 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼀𖽡𖾑 𖼳𖽔𖾐, 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖽃𖽡𖾑, 𖼮𖽘𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾑 𖼏𖽵 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼸𖽙 𖽃𖽡𖾐 𖼯𖽡𖽹𖾐、 𖽃𖽔𖾐 𖼯𖽡𖽹𖾐、 𖽐𖼊𖽘𖾏 𖼯𖽡𖽹𖾏、 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼋𖽘𖾏.” 8 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼗𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 , 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐,𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪𖾏 ! 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑?” 9 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼊𖽗𖾏 𖼽𖽡𖾐 𖼖𖽡𖾐 𖼘𖽪！ 𖼟𖽡 𖼯𖽔 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼈𖽺𖾏 𖼘𖽙𖾏 𖼮𖽡𖽝𖾏, 𖼋𖽙𖾏 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏. 10 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼾𖽨𖾏, 𖼾𖽨𖾏 𖼚𖽠𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼨𖽑𖽙𖾑 𖽐𖼮𖽔𖾏, 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐, 𖼮𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏. 11 𖼎𖽑𖽻𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐. 12 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽙𖾏 𖼸𖽙 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏. 13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡, 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽙𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 , 𖼟𖽡 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽃𖽔𖾏 𖼟𖽡 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 .”
