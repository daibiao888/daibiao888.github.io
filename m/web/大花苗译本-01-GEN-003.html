3
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏, 𖼯𖽔 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏. 𖼀𖽡𖾑 𖼐𖽘𖾐 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏, 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼉𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽐𖼊𖽻𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼘𖽪𖾏?” 2 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼐𖽑𖽡𖾏: “𖼉𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼊𖽻𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑, 𖼀𖽡𖾐 𖼑𖽻𖾏 𖼢𖽻 𖼯𖽡𖽙𖾏, 3 𖼯𖽔 𖼅𖽔𖾏 𖼉𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼉𖽺𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏: ‘ 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽔𖾏 𖼺𖽔𖾏, 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼋𖽔.’ ” 4 𖼀𖽡𖾑 𖼐𖽘𖾐 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼋𖽔； 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽻𖾐, 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽻𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑, 𖼅𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼋𖽘𖾏, 𖼅𖽡 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽻𖾐 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼮𖽷𖾑.”
𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼮𖽵𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖽃𖽔 𖼁𖽙𖾏 𖼑𖽔𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏, 𖼎𖽺𖾏 𖼚𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼙𖽔, 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳, 𖼐𖽱𖾏 𖼚𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼗𖽙𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼖𖽔𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
7 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼋𖽘𖾏, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏． 𖼅𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼚𖽻𖾏, 𖼑𖽜 𖼾𖽨𖾏 𖼺𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾐. 8 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼷𖽑𖽔𖾐 𖼋𖽔𖾏 𖼮𖽢𖾏 𖼸𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐. 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼐𖽙𖾑𖽑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏, 𖼎𖽺𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖽃𖽡𖾑 𖼋𖽡𖾏 𖼞𖽙𖾏 𖽐𖼊𖽻𖾏 𖼙𖽠𖾏, 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏. 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏?” 10 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼉𖽘𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼟𖽡 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏； 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼚𖽺𖾏 𖼙𖽠𖾏.” 11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏 𖼘𖽪? 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖼑𖽻𖾏 𖼺𖽘𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏?” 12 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼑𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.” 13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼘𖽪?” 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖽃𖽡𖾑 𖼮𖽑𖽵𖾑 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼑𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.”
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏,
𖼇𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏,
𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖽐𖼊𖽻𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑,
𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏 𖼑𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏.
15 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐,
𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐．
𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖽐𖼮𖽻 𖼟𖽡 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼟𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔 𖼙𖽻𖾏 𖼊𖽠𖾏.”
16 𖼎𖽺𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽡𖽻𖾐 𖼮𖽡𖽻𖾐 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼑𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖼟𖽡 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐．
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼟𖽡 𖼈𖽻𖾑,
𖼟𖽡 𖼈𖽻𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼟𖽡.”
17 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼊𖽘𖾑 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽤𖾐 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼺𖽘𖾏,
𖼑𖽻𖾏 𖼺𖽘𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐,
𖼊𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏；
𖼟𖽡 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽡𖾑.
18 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼋𖽔𖾏 𖼋𖽠𖾏 𖼞𖽡𖾑 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏；
𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼑𖽡𖾏 𖼻𖽻𖾐.
19 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼚𖽱𖾏 𖽐𖼞𖽘𖾏 𖼖𖽑𖽘𖾐 𖼀𖽪𖾑 𖼚𖽑𖽪𖾏,𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼊𖽻𖾏 𖼑𖽻𖾏,
𖼸𖽙 𖼟𖽪𖾏 𖼟𖽡 𖼁𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔 𖼗𖽔𖾏.
𖼟𖽡 𖼟𖽪𖾏 𖽃𖽔 𖼗𖽔𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼁𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏.”
20 𖼽𖽔𖾏 𖼊𖽘𖾑 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼳𖽢𖾏 𖽂𖽔𖾑 (“𖼳𖽢𖾏 𖽂𖽔𖾑”𖼟𖽪𖾏 “𖼯𖽡𖽝𖾏 𖼳𖽦𖾑”), 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼑𖽡𖽝𖾏. 21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼽𖽔𖾏 𖼊𖽘𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐, 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽠𖾏 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑.
𖼖𖽠𖾑 𖼋𖽠𖾏 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖼉𖽘𖾏
22 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏:“𖼑𖽔𖾏 𖼘𖽪！ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼦𖽡𖾏 𖼺𖽱𖾏 𖼋𖽘𖾏, 𖼸𖽠𖾏 𖼇𖽺𖾐 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼮𖽷𖾑； 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼅𖽻𖾏 𖼚𖽡𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼑𖽻𖾏, 𖼎𖽺𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏.” 23 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼖𖽠𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖼉𖽘𖾏 𖼙𖽠𖾏, 𖼅𖽻𖾏 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼎𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑. 24 𖼄𖽔𖾐 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼖𖽠𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼷𖽷𖾐 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳（𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼮𖽡𖾑 𖼖𖽪𖾏 𖼀𖽳） 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽘𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼎𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏, 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑.
