2 𖼅𖽡 𖼽𖽔𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼯𖽷𖾏 𖼻𖽙𖾑 𖼙𖽔 𖼅𖽳𖾏. 3 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼀𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏, 𖼸𖽠𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼑𖽜 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼐𖽡𖾑 𖼏𖽻 𖼖𖽑𖽔𖾏 𖼖𖽑𖽻𖾏 𖼡𖽑𖽹𖾐. 4 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼐𖽑𖽡𖾏， 𖼡𖽑𖽔𖾐 𖼟𖽪𖾏 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 . 5 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼮𖽑𖽔𖾏 𖼊𖽔𖾑 𖼏𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼅𖽻𖾏 𖼗𖽙𖾏 . 6 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼽𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼯𖽡𖽙𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖽐𖼮𖽳𖾑 𖽃𖽔𖾏 𖼷𖽷𖾏 𖽐𖼷𖽔, 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
7 𖼊𖽑𖽪𖽺𖾑 𖼮𖽡𖾑 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏，𖼐𖽱𖾏 𖼸𖽠𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ． 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐, 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽵𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏. 8 𖼞𖽪𖾑 𖼷𖽪𖾑 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼛𖽔𖾏 𖼅𖽡, 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼀𖽡𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐. 9 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖽃𖽔𖾑 𖼐𖽡 𖼳𖽡𖾑 𖼄𖽪𖾐 𖽐𖼎𖽙 𖼋𖽔𖾏, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 , 𖼟𖽪𖾏 𖼅𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙 . 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡.
𖼦𖽔𖾑 𖼗𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔
10 𖼽𖽔𖾏 𖼖𖽡𖾐 𖼊𖽔 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼑𖽪 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪 . 𖼀𖽔𖾑 𖼐𖽔 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼷𖽔𖾑 𖽐𖼀𖽠𖾏 𖼄𖽔𖾐 𖼞𖽑𖽙𖾐 𖼖𖽔𖾏 𖼑𖽪 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼊𖽑𖽡𖽝𖾐. ( 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼷𖽷𖾑 𖼄𖽔𖾐 𖼞𖽑𖽙𖾐𖼐𖽡𖾑 , 𖼅𖽡 𖼙𖽔 𖼷𖽹𖾏 𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼋𖽘𖾏； 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼅𖽡 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏.) 11 𖼽𖽦𖾑 𖼺𖽪 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽨 𖼺𖽷𖾑 𖼊𖽪𖾑 𖼎𖽺𖾏 𖼑𖽪 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼊𖽑𖽡𖽝𖾐. 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏, 𖼯𖽔 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖽃𖽔𖾏 𖼑𖽪𖾏 ,𖼎𖽺𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 . 12 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼅𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼟𖽪𖾏 𖼽𖽡𖾐 𖼀𖽔𖾑 𖼇𖽪 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼊𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼑𖽪 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪 . 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼘𖽑𖽪𖾐 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽐𖼚𖽙 ，𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 , 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏, 𖼸𖽠𖾏 𖼳𖽠𖾑 𖼟𖽪𖾏 𖼏𖽔𖾏. 13 𖼐𖽱𖾏 𖼑𖽜 𖼅𖽡 𖽐𖼎𖽙 𖼖𖽻𖾐 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼊𖽑𖽡𖽝𖾐 𖼳𖽡𖾑 𖼖𖽔𖾑 𖼀𖽑𖽙𖾑 𖼖𖽡𖾐𖼑𖽡𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡,𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼚𖽑𖽔𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐， 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼸𖽠𖾏 𖼑𖽜𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏. 14 𖼊𖽪𖾐 𖼞𖽑𖽙𖾏 𖽐𖼷𖽘𖾏 𖼖𖽪𖾏 𖼮𖽢𖾑𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼄𖽔𖾐 𖼑𖽪 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼊𖽑𖽡𖽝𖾐. 15 𖼷𖽑𖽜𖾑 𖼑𖽪 𖼖𖽻𖾐 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡,𖽐𖼎𖽙 𖼐𖽡 𖼇𖽔 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼊𖽑𖽡𖽝𖾐. 16 𖼋𖽡𖾏 𖽐𖼊𖽠𖾑 𖼄𖽻𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼟𖽡𖽠 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼖𖽻𖾐 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽡𖽠 𖼊𖽑𖽡𖽝𖾐； 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽡𖽠 𖽐𖼊𖽠𖾑 𖼄𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼖𖽻𖾐 𖼊𖽡𖾐 𖼮𖽢𖾑𖼎𖽙𖾑 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽡𖾑. 17 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼮𖽡𖾑 𖼀𖽪𖾏 𖼐𖽑𖽡𖾏： “ 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡， 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏．”
18 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼘𖽙 𖼾𖽨𖾏 𖼺𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼊𖽠𖾑 𖼑𖽪 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽪. 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼖𖽑𖽔𖾏 𖼖𖽑𖽻𖾏 𖼡𖽑𖽹𖾐．𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐.
