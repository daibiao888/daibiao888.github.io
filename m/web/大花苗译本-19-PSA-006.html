𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽘𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽷𖾑 𖼺𖽙𖾑, 𖼯𖽡𖽙𖾏“ 𖼾𖽡𖾏 𖼳𖽘𖾐 ”.
6
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼞𖽪𖾑,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽢𖾐 𖼞𖽪𖾑！
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽻 𖼚𖽑𖽔𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽑𖽙𖾏 𖼞𖽪𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏.
3 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼟𖽡 𖼖𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼖𖽺𖾑?
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑！
𖼑𖽜 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑.
5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡?
 
6 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏 𖼐𖽔𖾑 𖼯𖽻 𖼚𖽑𖽔𖾏,
𖼞𖽪𖾑 𖼷𖽑𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽡𖽝𖾑 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼐𖽔𖾑, 𖼎𖽑𖽻𖾏 𖼟𖽪𖾏 𖼸𖽱𖾏 𖼸𖽘𖾏 𖽐𖼊𖽘𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼺𖽘𖾏,
𖼞𖽪𖾑 𖼷𖽑𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽡𖽝𖾑 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼐𖽔𖾑, 𖼟𖽙𖾏 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐 𖼞𖽑𖽔𖾏 𖽐𖼊𖽪𖾐 𖼺𖽘𖾏.
7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏 𖼐𖽔𖾑 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼞𖽡 𖼐𖽑𖽡𖽙𖾏 𖼙𖽠𖾏,
𖼎𖽺𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼐𖽔𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼇𖽔 𖼙𖽻 𖼺𖽘𖾏.
 
8 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖼀𖽜𖾑 𖼞𖽪𖾑 𖼙𖽠𖾏!
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼐𖽡𖽝𖾑 𖽃𖽡𖾑 𖼋𖽘𖾏.
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽻 𖼞𖽪𖾑 𖽐𖼊𖽜 𖼽𖽔𖾑 𖼋𖽘𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
10 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
