2 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽐𖼮𖽔𖾏 𖼮𖽡𖽺𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼙𖽠𖾏 𖽃𖽡𖾑， 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝， 3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼙𖽔 𖼊𖽻𖾏 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 ，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖼗𖽙𖾏． 4 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖽐𖼷𖽜𖾏 𖼟𖽪𖾏 𖼸𖽙 𖼡𖽘𖾐，𖼟𖽙𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠． 5 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼺𖽷𖾐， 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏． 6 𖼯𖽔 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 ， 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼷𖽑𖽠𖾑 𖽐𖼊𖽱𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼷𖽑𖽠𖾑 𖽐𖼊𖽱𖾏，𖼟𖽪𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼟𖽡𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙 𖼉𖽺𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐． 7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽪𖾏 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼮𖽵𖾑 𖼊𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏． 8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐，𖼐𖽡𖽙𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏，𖼖𖽔𖾑 𖼅𖽻𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐．
9 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜， 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼺𖽡𖽝𖾐 𖼷𖽻𖾑 𖼖𖽔． 10 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽻𖾏 𖼦𖽡𖾏 𖼢𖽜𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼙𖽔，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼁𖽘𖾏 𖽐𖼡𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏． 11 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏，𖼮𖽢𖾏 𖼳𖽙𖾑 𖼷𖽑𖽔𖾐 𖼋𖽔𖾏，𖽐𖼡𖽜𖾐 𖼎𖽺𖾏 𖼡𖽑𖽔𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼁𖽘𖾏 𖼖𖽔𖾏 𖽐𖼀𖽑𖽙𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼑𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏，𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼙𖽠𖾏．
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐
