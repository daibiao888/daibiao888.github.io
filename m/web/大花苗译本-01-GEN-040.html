9 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼮𖽡𖽠𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽙 𖼺𖽳, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼇𖽘𖾐 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏. 10 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽢𖾏 𖽃𖽔𖾑 𖼯𖽡, 𖼘𖽘𖾑𖽑 𖽃𖽔𖾑 𖽐𖼷𖽔 𖼷𖽹𖾏 𖼋𖽠𖾏 𖼁𖽘𖾏 𖼋𖽘, 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾏 𖽐𖼡𖽑𖽻𖾏 𖼳𖽦𖾑 𖼋𖽘𖾏. 11 𖼇𖽔 𖼘𖽻𖾐 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼷𖽙𖾏 𖼎𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐, 𖼄𖽔𖾐 𖼞𖽑𖽙𖾐 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏. 12 𖼽𖽙 𖼺𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼷𖽷𖾐 𖼯𖽢𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐. 13 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼉𖽺𖾏 𖽐𖼚𖽙, 𖼇𖽔 𖼘𖽻𖾐 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼟𖽡 𖼋𖽠𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼮𖽡𖾑 𖼞𖽑𖽙𖾐 𖼎𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖽐𖼎𖽙 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼮𖽡𖽠𖾑 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐. 14 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑,𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑, 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼐𖽡𖽙𖾐 𖼇𖽔 𖼘𖽻𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽜 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏, 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼊𖽡𖾐. 15 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼑𖽡𖽝 𖼋𖽺 𖼋𖽔𖾏 𖼊𖽔𖾐, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏.”
16 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼡𖽜𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼻𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼦𖽡𖾐 𖼮𖽳𖾐 𖼊𖽜𖾐 𖼷𖽷𖾐 𖼈𖽘𖾐 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼚𖽠𖾐. 17 𖼖𖽪𖾑 𖼈𖽘𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼇𖽔 𖼘𖽻𖾐 𖽃𖽔𖾏 𖼮𖽡𖽝𖾏, 𖼅𖽔𖾏 𖼐𖽻 𖼽𖽘𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖼈𖽘𖾐 𖽃𖽡𖾑.” 18 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼷𖽷𖾐 𖼖𖽪𖾐 𖼈𖽘𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐, 19 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖽐𖼚𖽙, 𖼇𖽔 𖼘𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽑𖽘𖾐 𖼟𖽡 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼺𖽘𖾏, 𖼄𖽔𖾐 𖼟𖽡 𖼮𖽘𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼮𖽡𖽝𖾏, 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏.”
