𖼦𖽡𖾐 𖼑𖽡𖽝 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏,
𖼽𖽔𖾑 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑,
𖼐𖽱𖾏 𖼟𖽙𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽃𖽔𖾑 𖽐𖼷𖽻𖾏,
𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏.
17 𖼅𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼑𖽻 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼅𖽡 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼐𖽡𖽝𖾑 ,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼐𖽡𖽝𖾑 𖽐𖼚𖽱𖾏 𖼞𖽔𖾏 𖼅𖽔 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼀𖽙𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏.
 
18 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼐𖽑𖽡𖾏：
“ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼾𖽨𖾏, 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑, 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖼘𖽱𖾑𖽑 𖼙𖽠𖾏 𖼋𖽘𖾏.
19 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼙𖽻 ,
𖼞𖽑𖽔𖾏 𖼡𖽙𖾏 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽜𖾐,
𖼽𖽨 𖼊𖽔𖾏 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏,
𖼎𖽺𖾏 𖼦𖽔𖾏 𖼟𖽙𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
𖼸𖽻𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼑𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼮𖽡𖽻𖾐
20 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼋𖽔𖾏,
𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼟𖽡 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽺 𖼗𖽔 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖽃𖽡𖾑,
𖼮𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼗𖽙𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼟𖽡 ,
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽡 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑?
𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽔𖾑 𖼟𖽡 𖼮𖽡𖽝𖾏,
𖼺𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼊𖽪𖾐 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼖𖽺𖾑?
22 𖼟𖽡 𖼮𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽺𖾑?
𖼟𖽡 𖼡𖽘𖾐 𖼮𖽑𖽙𖾏 𖼊𖽻𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏,
𖼟𖽡 𖼡𖽘𖾐 𖽃𖽔 𖼙𖽻𖾏 𖼊𖽠𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐,
𖼟𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼐𖽔𖾑.
23 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼚𖽪𖾐 𖼘𖽪?
𖼊𖽪𖾑 𖼇𖽜𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼖𖽺𖾑?
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏, 𖼅𖽡 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑,
𖼎𖽺𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏 𖼋𖽘𖾏.
24 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼷𖽑𖽔𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼺𖽱𖾏 𖼡𖽔𖾏 𖽐𖼊𖽻𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
25 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼡𖽪𖾏 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏,
𖼟𖽪𖾏 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 .”
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏,
𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 (“𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼊𖽺𖾐 𖼁𖽠𖾏”).
26 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼡𖽑𖽜𖾐 𖼟𖽡 𖼡𖽘𖾐 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏,
𖽐𖼀𖽙 𖼎𖽻𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏,
𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼺𖽱𖾏 𖼚𖽘𖾐.
27 𖼟𖽡 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖽐𖼎𖽘𖾏 𖼯𖽢𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏,
𖼢𖽔𖾏 𖼟𖽡𖾏 𖼉𖽺𖾏 𖼀𖽡 𖼟𖽡𖽝 , 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼑𖽪𖾏 ,
𖼟𖽡 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼞𖽑𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏.
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！ 𖼟𖽡 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏,
𖼟𖽡 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼺𖽷𖾐 𖼖𖽺𖾑?
