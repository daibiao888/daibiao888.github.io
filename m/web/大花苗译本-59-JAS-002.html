𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 ， 𖼅𖽡 𖽐𖼮𖽜 𖼻𖽡𖽝𖾏 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽑𖽪𖾏 𖼯𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． 2 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼙𖽻 𖼑𖽜𖾏 𖼞𖽪𖾐、 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼯𖽪𖾏 𖽐𖼀𖽔， 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼅𖽡 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏，𖼐𖽱𖾏 𖼖𖽔𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼊𖽑𖽡𖽝𖾐． 3 𖼅𖽡 𖼑𖽔𖾏 𖼋𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼯𖽪𖾏 𖽐𖼀𖽔， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼷𖽑𖽜𖾑 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 ” ，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽪𖾏 𖼳𖽻𖾏 𖼐𖽑𖽡𖾏 ：“ 𖼟𖽡 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼉𖽺𖾏”， 𖼡𖽑𖽙𖾑 𖼯𖽢“ 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼡𖽔𖾑 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐 𖽃𖽡𖾑 𖼀𖽡𖾑 𖼊𖽡𖾐．” 4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼅𖽡 𖼑𖽔𖾏 𖼳𖽪𖾏 𖼢𖽻𖾏 𖼯𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼯𖽡𖽙𖾏 𖼅𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼨𖽑𖽢𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪？
5 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 ，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 ， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼺𖽹𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼅𖽔𖾏 𖼊𖽻𖾏， 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖼏𖽵 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼘𖽪？ 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼑𖽔𖾏 𖼦𖽡𖾏 𖼋𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏． 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼅𖽡 、 𖼮𖽤𖾐 𖼅𖽡 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏， 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼐𖽡𖾑 𖼘𖽪 ？ 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽃𖽡𖾑 𖼘𖽪？ 8 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏：“ 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼅𖽡 𖼼𖽷𖾏 𖼼𖽙 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼾𖽨𖾏．” 𖼅𖽡 𖼮𖽔𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼐𖽡𖾑，𖼅𖽡 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼾𖽙 𖼯𖽡𖽙𖾏． 9 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼮𖽔𖾑 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽑𖽪𖾏 𖼯𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏，𖼟𖽪𖾏 𖼊𖽻𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼊𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏． 10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖽐𖼷𖽻𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 ， 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽙𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜． 11 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼁𖽱 𖼐𖽑𖽡𖾏： “𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏．” 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏： “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ．” 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏， 𖼈𖽡𖽝𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼟𖽡 𖼎𖽺𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 ． 12 𖼅𖽡 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 ，𖼯𖽡𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 ． 13 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 ， 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐． 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐．
