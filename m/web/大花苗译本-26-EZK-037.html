𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑. 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼮𖽙𖾏 𖼞𖽪𖾑 𖼎𖽻𖾏 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏, 𖼖𖽪𖾑 𖽐𖼎𖽘𖾏 𖼐𖽡𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽔𖾑 𖼷𖽑𖽘𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐. 2 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡 𖽐𖼮𖽡 𖼎𖽻𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖽃𖽡𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐, 𖼑𖽔𖾏 𖼘𖽪！ 𖼖𖽪𖾑 𖽐𖼎𖽘𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖼑𖽔𖾏 𖼘𖽪！ 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖽃𖽡𖾑 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼊𖽔𖾐 𖼋𖽡𖽝. 3 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼖𖽺𖾑?” 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼀𖽻𖾐 𖼯𖽡𖽙 .” 4 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼎𖽙 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼘𖽪！ 𖼽𖽔𖾑 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏. 5 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑： 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼀𖽘𖾏( “𖼀𖽘𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼖𖽡𖽾”) 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼅𖽡 𖽐𖼚𖽙, 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼋𖽘. 6 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽑𖽡𖽝𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼅𖽡 , 𖼟𖽙𖾏 𖼅𖽡 𖼘𖽘𖾑𖽑 𖽐𖼡𖽹𖾏, 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼡𖽑𖽠𖾑 𖼅𖽡, 𖼟𖽙𖾏 𖼀𖽘𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼅𖽡 𖽐𖼚𖽙, 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼯𖽡𖽝𖾏 𖼋𖽘 , 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.”
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏. 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖽃𖽡𖾑, 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 , 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽻 , 𖼑𖽔𖾏 𖼘𖽪！ 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏. 8 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽠𖾑, 𖼎𖽺𖾏 𖼘𖽘𖾑𖽑 𖽐𖼡𖽹𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼡𖽑𖽠𖾑 𖼎𖽻𖾏 𖼦𖽡𖾐 𖼇𖽔𖾑, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼅𖽔𖾏 𖼀𖽘𖾏 𖼺𖽷𖾐！ 9 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖽐𖼎𖽙 𖼮𖽢𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖼐𖽑𖽡𖾏: 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑： 𖼀𖽘𖾏 𖼮𖽢𖾏 𖼘𖽪！ 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼷𖽑𖽔𖾐 𖼋𖽔𖾏, 𖼷𖽑𖽔𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 .” 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖼀𖽘𖾏 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏, 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼋𖽘 , 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼊𖽡𖽝𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼏𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .
