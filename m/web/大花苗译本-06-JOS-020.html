𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖽃𖽡𖾑 , 𖼑𖽜 𖼾𖽨𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏， 3 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼷𖽪𖾑 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 , 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼗𖽔 𖼋𖽔 𖼙𖽠𖾏 𖼈𖽡𖽝𖾏 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑 , 𖼻𖽻𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 , 𖼋𖽡𖾏 𖼙𖽻 𖼐𖽡𖾑 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼅𖽡 𖼙𖽻 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏 ，𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼁𖽻𖾏 𖼑𖽪𖾏 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑 . 4 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼋𖽡𖾏 𖼙𖽻 𖼐𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽠𖾏, 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 , 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖽐𖼚𖽙 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼗𖽻． 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼚𖽙 𖼙𖽻 , 𖼄𖽔𖾐 𖼡𖽑𖽔 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 . 5 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼁𖽻𖾏 𖼑𖽪𖾏 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑 𖼮𖽔𖾑 𖼖𖽠𖾑 𖼋𖽔𖾏 𖼯𖽢 𖼐𖽱𖾏 , 𖼊𖽪𖾐 𖼗𖽻 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼁𖽻𖾏 𖼑𖽪𖾏 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 , 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼟𖽙𖾏 𖼗𖽔 𖼋𖽔 𖼗𖽠𖾏 𖼈𖽡𖽝𖾏 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 . 6 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 , 𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽻 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 , 𖼋𖽙𖾏 𖼟𖽙𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼋𖽔 𖼙𖽠𖾏 , 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼊𖽻𖾏 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 , 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽜𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 .
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽺𖾐 𖼏𖽵 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 , 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼖𖽡 𖼖𖽡 𖼑𖽡𖾏 𖼮𖽡𖾑 𖼊𖽡𖾐 𖼺𖽷𖾑 , 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 , 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 , 𖼽𖽨 𖼊𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 , 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 ·𖼽𖽔𖾏 𖼀𖽔𖾑 ( 𖼮𖽡𖾑 𖼖𖽡𖽝 ·𖼽𖽔𖾏 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 ) . 8 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏， 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 , 𖼎𖽑𖽻𖾏 𖼖𖽨 𖼀𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼊𖽡𖾏 𖼏𖽵 𖼀𖽡𖾐 𖼳𖽡 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 , 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼊𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼊𖽡𖾏 𖼏𖽵 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙 , 𖼎𖽑𖽻𖾏 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼊𖽡𖾏 𖼏𖽵 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼑𖽡𖾏 𖼞𖽙𖾑 𖼖𖽔 . 9 𖼋𖽡𖾏 𖼙𖽻 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼇𖽺𖾐 𖼊𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽻， 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼷𖽪𖾑 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼻𖽻𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼉𖽺𖾏， 𖼦𖽡𖾐 𖼋𖽔 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼁𖽻𖾏 𖼑𖽪𖾏 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽻 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 .
