𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼮𖽵𖾑 𖼳𖽺 𖼞𖽪𖾐, 𖼺𖽡𖽝𖾐 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼇𖽘𖾑 𖼚𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼷𖽑𖽙𖾏 𖼎𖽻𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑,𖼊𖽪𖾏 𖼖𖽔𖾑 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼇𖽘𖾑. 2 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾏 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏、 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖽐𖼮𖽷𖾏、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼊𖽑𖽡𖽝𖾐 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼞𖽑𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏, 𖼑𖽜 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼡𖽑𖽜𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 . 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏、 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖽐𖼮𖽷𖾏, 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐, 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼊𖽑𖽡𖽝𖾐 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖼋𖽘 , 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼡𖽑𖽜𖾐 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐. 4 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏、 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐、 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼘𖽪! 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡： 5 𖼅𖽡 𖽂𖽪𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼞𖽪𖾐 𖼳𖽘𖾐、 𖼖𖽡𖾑 𖼘𖽪𖾏 𖼳𖽘𖾐、 𖽐𖼎𖽔𖾏 𖼳𖽘𖾐、 𖼢𖽜𖾏 𖼳𖽘𖾐、 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼳𖽘𖾐、 𖼖𖽡𖾑 𖼖𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾐 𖼳𖽘𖾐 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼢𖽜𖾏 𖽐𖼎𖽔 𖼳𖽘𖾐, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏. 6 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽃𖽡𖾑,𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼾𖽱𖾏 𖼺𖽘𖾏.”
