24
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼎𖽺𖾏 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑𖼊𖽔𖾏 𖽂𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼮𖽔𖾑 𖼾𖽱 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. 2 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 𖼽𖽙 𖼽𖽔 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖼟𖽙𖾏 𖽐𖼷𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏, 𖼎𖽑𖽻𖾏 𖼊𖽗𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑, 𖼅𖽻𖾏 𖼮𖽔𖾑 𖼾𖽱 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠.” 3 𖼽𖽙 𖼽𖽔 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽡𖽻𖾐 𖼯𖽠𖾏, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼀𖽔𖾏 𖽐𖼀𖽠, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏, 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑?” 4 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽙 𖼽𖽔 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏． 𖼽𖽙 𖼽𖽔 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 𖼎𖽑𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼮𖽔𖾑 𖼾𖽱 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏. 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖾏 𖼮𖽑𖽡𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏, 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼊𖽳 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼽𖽔𖾏 𖼘𖽙 𖽃𖽳𖾐 𖼮𖽪𖾐 𖼾𖽡, 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖼽𖽔𖾐 𖼳𖽡𖽝𖾏， 6 𖼎𖽺𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖽐𖼎𖽙 𖼊𖽑𖽔𖾑 𖼊𖽑𖽡 · 𖼦𖽙 𖼳𖽷𖾏 𖼊𖽡𖾐, 𖼎𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼊𖽗𖾏 · 𖼽𖽔𖾐 𖼣𖽔𖾑, 𖼮𖽑𖽙𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏, 7 𖼗𖽙𖾏 𖼸𖽙 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑, 𖼊𖽑𖽡𖽝𖾐 𖼳𖽡𖾑 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻, 𖼎𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖽃𖽡𖾑 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑. 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼯𖽢𖾏 𖼖𖽑𖽡𖾏 𖼊𖽑𖽡𖽝𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼐𖽑𖽪𖾐, 𖼎𖽙𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐. 9 𖼽𖽙 𖼽𖽔 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽔𖾑 𖼾𖽱 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖼉𖽻𖾏, 𖼽𖽨 𖼊𖽔𖾏 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼉𖽻𖾏.
