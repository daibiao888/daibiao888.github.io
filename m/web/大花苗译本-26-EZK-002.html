𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏.” 2 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼖𖽡𖽾 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽐𖼚𖽙, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏. 3 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 𖽐𖼷𖽜𖾑 𖼊𖽠𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼖𖽔𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏. 4 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼚𖽑𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏. 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑. 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽻, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼦𖽡𖾐 𖼑𖽻, 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼀𖽻𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐. 6 𖼟𖽡 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼅𖽔𖾏 𖼞𖽡𖾑 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼉𖽡𖽝𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼞𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼚𖽡𖽝𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏. 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏. 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽻 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼦𖽡𖾐 𖼑𖽻 , 𖼟𖽡 𖼮𖽔𖾑 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖼸𖽙 𖼡𖽘𖾐.
8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑, 𖼺𖽱𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡.” 9 𖼞𖽪𖾑 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼑𖽔𖾏 𖼘𖽪！ 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼑𖽘 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑. 10 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼑𖽘 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼡𖽑𖽜𖾐 𖼮𖽡𖽝𖾏, 𖽐𖼚𖽙 𖼢𖽻𖾏 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼊𖽠𖾑, 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏、𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏、 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐.
