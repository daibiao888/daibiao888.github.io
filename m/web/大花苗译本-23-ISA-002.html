𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐
（𖼄𖽡𖾐 4：1 - 3）
2
𖼽𖽔𖾏 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼊𖽻𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐, 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 .
 
2 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏,
𖼮𖽑𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼇𖽔𖾑 𖼉𖽻𖾏 𖼩𖽜 ,
𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼽𖽔𖾑 𖽐𖼚𖽱𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑 .
3 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽡 𖼋𖽔𖾏, 𖼐𖽑𖽡𖾏:
“𖼋𖽔𖾏 𖼘𖽪！ 𖼀𖽡𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐,
𖼚𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏．
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾐,
𖼀𖽡𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑.”
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏𖼳𖽡 𖼣𖽔𖾑 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 .
4 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐,
𖼑𖽜 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖽝 𖼮𖽑𖽵𖾑.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼇𖽺𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼄𖽑𖽪𖾑 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔𖾐 𖼖𖽡𖽝𖾐．
𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽻𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼉𖽺𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏.
5 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏 𖼘𖽪 ！ 𖼋𖽔𖾏 ，
𖼀𖽡𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
 
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼟𖽡 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏,
𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼸𖽪𖾏 𖼼𖽘𖾏,
𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽔𖾑 𖼳𖽦𖾑 𖼺𖽱𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼎𖽺𖾏 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽐𖼀𖽔𖾏 𖼋𖽡𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 .
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐,
𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼦𖽡𖾐 𖼮𖽔𖾑𖽑 𖼯𖽠,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜𖾏 𖼀𖽪𖾑 𖽐𖼷𖽡𖽝𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖼾𖽱 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼊𖽺𖾐 𖼁𖽠𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏,
𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑.
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽱𖾏 𖼅𖽳𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼷𖽷𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
10 𖼟𖽡 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼡𖽘𖾐 𖼁𖽘𖾐 𖼡𖽘𖾐 𖼷𖽔𖾏, 𖽐𖼚𖽺𖾏 𖼚𖽔𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽙 ,
𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼚𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .
11 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽪𖾏 𖼇𖽻𖾑,
𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼮𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏.
𖼮𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏
12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏,
𖼽𖽔𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾑 𖼚𖽘𖾐 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼾𖽠 𖽃𖽡𖾑,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼮𖽑𖽳𖾑 𖼺𖽡𖽝𖾐 𖼾𖽨𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼦𖽡𖾏 𖼢𖽜𖾏.
13 𖼎𖽺𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏.
14 𖼎𖽺𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽻𖾐 𖼩𖽜𖾏、 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽻𖾐 𖽃𖽔𖾏 𖼷𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐.
15 𖼎𖽺𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼏𖽔𖾏.
16 𖼎𖽺𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽙𖾏, 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.
17 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽪𖾏 𖼇𖽻𖾑,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻．
𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼮𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏.
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼁𖽠𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼀𖽡𖾐 𖼎𖽑𖽔𖾑
18 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼁𖽠𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏.
19 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐, 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼙𖽠𖾏,
𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖽐𖼎𖽙 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐.
20 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 、 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽃𖽡𖾑,
𖼗𖽔𖾏 𖼎𖽻𖾏 𖼮𖽑𖽷𖾑 𖼙𖽡𖽝𖾐 𖽐𖼎𖽙 𖼑𖽘𖾏 𖽐𖼀𖽔𖾑.
21 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼻𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼈𖽳𖾐 𖼡𖽑𖽙𖾑 𖼷𖽔𖾏 𖽐𖼚𖽙 ,
𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖽐𖼎𖽙 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐.
22 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼐𖽱𖾏 𖼖𖽪𖾑 𖽃𖽔 𖽐𖼀𖽱 𖽐𖼚𖽙 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼀𖽘𖾏 𖼊𖽔𖾐,
𖼐𖽱𖾏 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?
