𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 , 𖼷𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐 . 2 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 , 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 ,𖼑𖽔𖾏 𖼐𖽱𖾏 𖼻𖽻𖾏 𖽐𖼎𖽻 , 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼷𖽷𖾐 𖼖𖽑𖽡𖾏. 3 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙𖾏 𖼈𖽺𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼋𖽘𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼊𖽠𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖽐𖼊𖽻𖾏 𖼀𖽑𖽪 𖼷𖽑𖽻𖾐 𖼳𖽡𖽝𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼚𖽪𖾐 𖼊𖽑𖽡𖽝𖾐 𖼷𖽑𖽹𖾏 𖼚𖽜𖾑 𖼘𖽙𖾏 𖼮𖽡𖽝𖾏 , 𖼮𖽑𖽳𖾑 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 , 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼊𖽠𖾐 𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽙𖾏 𖼡𖽳𖾐 𖼮𖽡𖽝𖾏 . 4 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼻𖽷𖾏 𖼳𖽠𖾑 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼑𖽔𖾏 , 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 .
5 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽡 𖼘𖽙 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼛𖽡𖾏 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼐𖽱𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼅𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽔𖾑 𖼊𖽠𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽙𖾏 𖼡𖽳𖾐 𖽐𖼚𖽙 , 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼡𖽑𖽜𖾑 𖼅𖽻𖾏 𖼮𖽑𖽳𖾑 𖼗𖽙𖾏 . 6 𖼐𖽱𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔𖾑 𖼊𖽠𖾐 𖼺𖽘𖾏 , 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 , 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼐𖽡𖽝𖾑 𖼋𖽘𖾏 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼅𖽳𖾏 𖼐𖽱𖾏 , 𖼐𖽑𖽡𖾏 : “𖼷𖽔𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 .” 7 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼻𖽷𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽃𖽔 𖼁𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽡𖾏 𖼋𖽔𖾏 , 𖼑𖽜 𖼟𖽡 𖼊𖽪𖾑 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 , 𖼋𖽡𖽝𖾏 𖼯𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 ?” 8 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽑𖽡𖾏 : “𖼯𖽪𖾏 !” 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼨𖽑𖽪𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼋𖽔𖾏 . 9 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼁𖽔 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖼅𖽻𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼄𖽡𖾏 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼎𖽻𖾏 𖼟𖽡 .” 𖽃𖽔 𖼁𖽙𖾏 𖼁𖽔 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖼅𖽻𖾏 𖼊𖽪𖾑 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏 . 10 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼋𖽘 , 𖽃𖽔 𖼁𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 . 𖼐𖽱𖾏 𖼊𖽡𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼐𖽡𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 (“𖼄𖽙𖾑 𖼳𖽡𖾑”𖼟𖽪𖾏 “𖼮𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏”) , 𖼚𖽠𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 : “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 .”
𖼄𖽙𖾑 𖼳𖽡𖾑 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏
11 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼘𖽑𖽙𖾐 𖼋𖽘𖾏 , 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 , 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾐 𖼐𖽜𖾐 . 12 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 , 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼈𖽺𖾏 𖼎𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐 𖼮𖽡𖽝𖾏 . 13 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼁𖽙𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 , 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼗𖽔 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼊𖽻𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽺𖾑 ?” 14 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼡𖽔 𖼋𖽱𖾏 𖼺𖽹𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼘𖽪 ? 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖼖𖽔𖾑 𖼊𖽔𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖼷𖽷𖾑 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏 ?” 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼋𖽘𖾏 , 𖽐𖼊𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼋𖽘 .” 15 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽙𖾑 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼊𖽔𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼙𖽠𖾏, 𖼀𖽜𖾑 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼚𖽔𖾏 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 .
16 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 , 𖼐𖽱𖾏 𖼼𖽻𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 . 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖽐𖼷𖽑𖽹𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐 ,𖽐𖼊𖽻𖾏 𖼀𖽪𖾑 𖼖𖽪𖾑 𖼚𖽘𖾐 , 𖼽𖽔𖾑 𖼀𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 . 17 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼯𖽦𖾏 , 𖼋𖽔𖾏 𖼖𖽠𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽠𖾏 , 𖼈𖽡𖽝𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼎𖽺𖾏 𖼀𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 . 18 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽙𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼖𖽨 𖽃𖽳𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼖𖽨 𖽃𖽳𖾐 𖼑𖽪 𖼐𖽑𖽡𖾏 : “𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼗𖽙𖾏 𖼇𖽺𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ?” 19 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼛𖽡𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼯𖽦𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼎𖽺𖾏 𖼑𖽜 𖼀𖽡𖾐 𖽐𖼊𖽻𖾏 𖽃𖽻𖾐 𖼗𖽙𖾏 𖼀𖽪𖾐 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 .” 20 𖼐𖽱𖾏 𖽐𖼎𖽙 𖽐𖼷𖽑𖽹𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏 : “𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 ? 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼘𖽪? 𖼅𖽡 𖼅𖽻𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 .” 21 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 , 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽑𖽙𖾑 𖼖𖽔𖾑 𖼎𖽻𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽔𖾏 𖼐𖽤𖾐 . 22 𖼳𖽡𖾑 𖼀𖽑𖽙𖾑 𖼖𖽔𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼞𖽳 𖼳𖽪𖽝𖾏 (“𖼞𖽳 𖼳𖽪𖽝𖾏”𖼟𖽪𖾏 “𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐”) , 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼟𖽪𖾏 : “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 .”
23 𖼮𖽑𖽡𖽻𖾏 𖼗𖽠𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼳𖽡𖽻𖾏 , 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏 . 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 24 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏 , 𖼎𖽺𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 、𖼽𖽡𖾑 𖼺𖽔𖾐 、𖼽𖽔𖾐 𖼞𖽙 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 . 25 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 .
