𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽃𖽡𖾑, 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐, 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼘𖽙 𖼊𖽳 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻, 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼷𖽹𖾏, 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑， 2 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪！ 𖼮𖽑𖽙 𖼅𖽡 𖼚𖽔𖾏 𖼡𖽑𖽜𖾑 𖽐𖼞𖽔𖾏 𖽐𖼷𖽔𖾑 𖼊𖽠𖾏, 𖼀𖽱𖾏 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏, 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐! 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼀𖽱𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏.” 3 𖼘𖽙 𖼊𖽳 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏. 𖼘𖽙 𖼊𖽳 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏, 𖽐𖼊𖽑𖽜𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼋𖽘𖾏. 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼢𖽻 𖼀𖽱𖾏 𖼺𖽷𖾐, 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾑 𖼾𖽠 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼗𖽻𖾏 𖼘𖽑𖽔𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽔𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑. 5 𖼨𖽑𖽪𖾏 𖼘𖽙 𖼊𖽳 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏? 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼯𖽔𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽱𖾏.” 6 𖼘𖽙 𖼊𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼮𖽑𖽢𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼡𖽙𖾏 𖼮𖽡𖽝𖾏, 𖼚𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 7 𖼐𖽑𖽡𖾏: “𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏, 𖼮𖽑𖽙 𖼅𖽡 𖼡𖽑𖽔𖾐 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑 𖼊𖽡𖾐. 8 𖼅𖽡 𖼑𖽔𖾏， 𖼞𖽪𖾑 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼷𖽑𖽹𖾏, 𖼟𖽪𖾏 𖽐𖼷𖽑𖽹𖾏 𖼊𖽡𖾐 𖽐𖼞𖽻𖾐, 𖼯𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼯𖽔𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼐𖽡𖾑, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏.” 9 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼮𖽠𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏! 𖼟𖽡 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼋𖽔𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼖𖽔𖾏! 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼟𖽡𖾏 𖼟𖽡 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.” 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽐𖼮𖽵 𖼐𖽹𖾏 𖼘𖽙 𖼊𖽳, 𖼽𖽔𖾑 𖼊𖽑𖽠𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼢𖽻 𖼺𖽘𖾏.
𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼘𖽙 𖼊𖽳
10 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼄𖽔𖾐 𖼘𖽙 𖼊𖽳 𖼮𖽤𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏, 𖼡𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼮𖽡𖽝𖾏. 11 𖼎𖽺𖾏 𖼟𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼗𖽻𖾏 𖼘𖽑𖽔𖾏, 𖽃𖽔𖾑 𖼅𖽔 𖼞𖽑𖽔𖾏 𖼇𖽔 𖼙𖽻 𖼺𖽘𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼅𖽻𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼗𖽙𖾏, 𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏. 12 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼘𖽙 𖼊𖽳 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼡𖽑𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼺𖽷𖾐 𖼘𖽪? 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼈𖽻𖾑、𖼟𖽪𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏、𖽐𖼎𖽙 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼟𖽡, 𖼟𖽡 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏. 13 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼛𖽠𖾏 𖼑𖽪𖾏 𖽐𖼚𖽙 𖼙𖽻 𖼐𖽡𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽐𖼎𖽙𖾑 𖼸𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼋𖽔𖾏, 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑.” 14 𖼘𖽙 𖼊𖽳 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼅𖽡 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼀𖽜𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑.” 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼅𖽡 𖼸𖽙𖾏 𖼞𖽑𖽡𖽝𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼚𖽙𖾏 𖼊𖽔𖾐.
𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼨𖽑𖽳𖾏 𖼘𖽙 𖼊𖽳 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
15 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏 𖼋𖽘, 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼘𖽙 𖼊𖽳 𖼐𖽑𖽡𖾏: “𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏！ 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼐𖽤𖾐 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼟𖽡 𖼑𖽜 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽻𖾏 𖽐𖼎𖽙 𖼖𖽻𖾐 𖼺𖽘𖾏.” 16 𖼈𖽡𖽝𖾏 𖼘𖽙 𖼊𖽳 𖼊𖽳𖾑𖽑 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏. 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼘𖽙 𖼊𖽳, 𖼎𖽺𖾏 𖼮𖽤𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼷𖽑𖽹𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼡𖽻𖾏 𖼙𖽻； 17 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼀𖽜𖾑 𖼳𖽦𖾑 𖼦𖽡𖾑! 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼡𖽜𖾐 𖼑𖽔𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏, 𖼽𖽔𖾑 𖼏𖽘 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.” 18 𖼘𖽙 𖼊𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪！ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑！ 19 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐, 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑． 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼚𖽡𖾐, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼸𖽙 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼊𖽪𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼋𖽔 𖼺𖽘𖾏. 20 𖼑𖽔𖾏 𖼘𖽪！ 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼳𖽻𖾐 𖽃𖽺𖾐 𖼊𖽡𖾏 𖼈𖽳𖾏, 𖼖𖽡 𖼟𖽻𖾏 𖼀𖽜𖾑 𖼸𖽙, 𖼖𖽪𖾑 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼉𖽺𖾏, 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏.” 21 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏:“ 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼟𖽡, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼖𖽻𖾐 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼐𖽡𖾑. 22 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼉𖽺𖾏， 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽻 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼺𖽙𖾐 𖽃𖽳𖾐 ( “𖼺𖽙𖾐 𖽃𖽳𖾐” 𖼟𖽪𖾏 “𖼳𖽻𖾐 ” ).
