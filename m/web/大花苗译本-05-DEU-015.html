15
“𖼡𖽑𖽙𖾑 𖼸𖽙 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖽃𖽡𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼯𖽔𖾏 . 2 𖼮𖽙𖾏 𖼯𖽔𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼮𖽙𖾏 𖽐𖼮𖽷𖾏 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼎𖽻𖾏 𖼼𖽷𖾏 𖼼𖽙 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙、𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼊𖽜 𖼽𖽔𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼯𖽔𖾏 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏 . 3 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼯𖽪𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼊𖽜 𖼽𖽔𖾑 , 𖼈𖽡𖽝𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼟𖽡 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏 . 4-5 𖼟𖽡 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 , 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼡𖽑𖽔𖾐 𖼟𖽡 , 𖼯𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼋𖽘𖾏 , ( 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼎𖽻𖾏 𖼟𖽡 . ) 6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼏𖽵 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡． 𖼟𖽡 𖼽𖽔𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼎𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 , 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼟𖽡 .
𖽐𖼷𖽷𖾑 𖽐𖼀𖽘𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏
7 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼖𖽪𖾐 𖼙𖽻 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼸𖽺𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏 , 𖽐𖼞𖽻𖾏 𖼋𖽡𖾏 𖼮𖽡𖽝𖾏 , 𖼦𖽡𖾐 𖽐𖼀𖽘𖾏 𖽐𖼷𖽷𖾑 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 . 8 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼷𖽹𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖽐𖼷𖽷𖾑 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 . 9 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 , 𖼐𖽑𖽡𖾏 : 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼯𖽔𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼗𖽙𖾏 𖼸𖽙 𖼋𖽘𖾏 , 𖼟𖽡 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽔𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 , 𖼋𖽪 𖼳𖽷𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼑𖽜 𖼟𖽡 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽔𖾑 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼯𖽡𖾏 𖼟𖽡 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼋𖽘𖾏 . 10 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼐𖽡𖾑 , 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 . 11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽪𖾏 𖼮𖽘𖾑 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 , 𖼽𖽔𖾑 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼋𖽡𖾏 𖼗𖽙𖾏 𖽐𖼀𖽘𖾏 𖼟𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 .
𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼡𖽑𖽜𖾑
（𖼋𖽠𖾏 21：1 - 11）
12 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽃𖽔𖾑 𖼾𖽠 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽃𖽔 𖼁𖽙𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼟𖽡 , 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 , 𖼸𖽙 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 . 13 𖼟𖽡 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼅𖽻𖾏， 14 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏、𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏、𖼖𖽪𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼮𖽡𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼮𖽡𖽻𖾐 𖼮𖽡𖽻𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 , 𖼟𖽡 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 . 15 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼟𖽡, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 . 16 𖼐𖽱𖾏 𖼮𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼇𖽳𖾑 𖼟𖽡 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 , 𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼻𖽻𖾏， 17 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖼮𖽑𖽷𖾏 𖼙𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼎𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼚𖽻𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼮𖽑𖽡𖽙𖾐 𖼊𖽙𖾏 𖼺𖽘𖾏 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼋𖽘𖾏 . 𖼟𖽡 𖼯𖽡𖽙𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 . 18 𖼟𖽡 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽔𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽔𖾏 𖼽𖽨𖾏 𖼟𖽡 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 , 𖼮𖽡𖽻𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼋𖽘𖾏． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 .
19 𖼟𖽡 𖽐𖼀𖽙𖾏 𖼑𖽨𖾐 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽔𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼨𖽑𖽳𖾏 , 𖼖𖽡𖾏 𖼡𖽪𖾑、𖼞𖽡𖾏 𖼙𖽻𖾏、𖽃𖽔𖾑 𖼁𖽱 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖽐𖼀𖽙𖾏 𖼑𖽨𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼨𖽑𖽳𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 , 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼨𖽑𖽳𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽻𖾑 𖼚𖽻𖾐 . 20 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼨𖽑𖽳𖾏 𖼐𖽡𖾑 , 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 , 𖼷𖽑𖽔𖾏 𖼳𖽡𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽹𖾑 𖽃𖽡𖾑 ,𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽻𖾏 . 21 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼨𖽑𖽳𖾏 𖼐𖽡𖾑 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 、𖼊𖽡 𖼛𖽜 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 22 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽐𖼚𖽙 𖼙𖽻 𖼑𖽻𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼖𖽔𖾏 𖼑𖽻𖾏 𖼢𖽻 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏 𖽐𖼎𖽙 𖽐𖼡𖽹𖾏 𖼷𖽹𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 . 23 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 , 𖼽𖽔𖾑 𖽐𖼮𖽢𖾑𖽑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾑𖽑 𖽃𖽻𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 .”
