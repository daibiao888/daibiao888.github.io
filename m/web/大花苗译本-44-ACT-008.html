8
𖼎𖽑𖽻𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼁𖽱𖾏 ， 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼋𖽠𖾏 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼺𖽘𖾏， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽑𖽡𖽝𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑． 2 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼞𖽑𖽡𖽠𖾏 𖼺𖽷𖾑 𖼊𖽑𖽡 𖼇𖽔𖾐 𖼅𖽻𖾏 𖼊𖽱𖾏 𖼺𖽘𖾏，𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼐𖽱𖾏 𖼐𖽡𖽝𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝． 3 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏，𖼁𖽺𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖽐𖼞𖽔𖾏，𖼦𖽡𖾐 𖼞𖽳𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔 𖼾𖽠 𖼖𖽔𖾏 𖼮𖽤𖾐 𖼅𖽻𖾏 𖼡𖽙𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏．
𖼇𖽺𖾑 𖼖𖽡𖾏 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏
4 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏． 5 𖼇𖽺𖾑 𖼖𖽡𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼙𖽻，𖼡𖽑𖽔𖾐 𖼮𖽡𖾑 𖼊𖽪 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 6 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏，𖼎𖽺𖾏 𖼁𖽙𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏． 7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ，𖼀𖽡𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼢𖽔𖾏，𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏；𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏． 8 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝．
𖽃𖽔𖾑 𖼾𖽠𖾏 𖼷𖽷𖾏 𖼅𖽪𖾏 𖽐𖼮𖽜 𖼞𖽘𖾐 𖼞𖽡𖾏
