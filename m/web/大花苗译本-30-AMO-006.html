6
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐,
𖼟𖽡𖾏 𖼏𖽔𖾏 𖽐𖼎𖽜𖾏 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 !
𖼅𖽡 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐,
𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏.
2 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼮𖽢 𖼐𖽡 𖼳𖽹𖾐 𖼑𖽔𖾏,
𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳 ,
𖼅𖽡 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾐 𖼋𖽡𖾏 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼘𖽪𖾏 ?
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼇𖽘𖾑 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼅𖽡 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
3 𖼅𖽡 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼡𖽔𖾏 𖼚𖽡𖾐 𖼺𖽷𖾐,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼗𖽙𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏.
 
4 𖼅𖽡 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼸𖽘𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼷𖽑𖽱𖾏 𖼟𖽻𖾏 𖽃𖽔𖾏, 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼐𖽡𖽙𖾐 𖼸𖽱𖾏 𖼸𖽘𖾏,
𖼑𖽻𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼑𖽡𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏, 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼑𖽨𖾐 𖼑𖽡𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏.
5 𖼅𖽡 𖼳𖽙𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽷𖾑 𖽐𖼡𖽜𖾏 𖽐𖼊𖽻𖾏 𖽐𖼎𖽔𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏, 𖼑𖽜 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼢𖽜𖾏 𖽐𖼎𖽔 ,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑.
6 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑, 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼚𖽜𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽜 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼊𖽳𖾏.
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽪𖾐 𖼊𖽻𖾏 𖼦𖽔𖾏,
𖼗𖽙𖾏 𖼳𖽦𖾑 𖼑𖽻𖾏 𖼦𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘 !
 
8 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼏𖽵 𖼗𖽪𖾏, 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼳𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏, 𖼳𖽪𖾐 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑, 𖽃𖽔𖾏 𖼐𖽡𖾑 ,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼙𖽻 𖽐𖼎𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏, 𖼞𖽑𖽔𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 .
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
9 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽔𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼮𖽔𖾑 𖼳𖽙𖾏 𖼟𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼋𖽔 . 10 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼑𖽜𖾏 𖼮𖽘𖾏, 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽡𖾑 𖼮𖽑𖽳𖾑 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏, 𖼑𖽪 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽷𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏?” 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼅𖽔𖾏.”𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏.”
11 𖼑𖽔𖾏 𖼘𖽪！ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽠𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑,
𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑,
𖽃𖽔𖾑 𖼯𖽱𖾏 𖽐𖼞𖽔𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 .
𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼇𖽳𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑
12 𖼊𖽪𖾑 𖼑𖽜𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼏𖽘 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼡𖽔𖾏 𖼷𖽔𖾏?
𖼋𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼎𖽑𖽻𖾏 𖼑𖽨𖾏 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖼖𖽺𖾑?
𖼈𖽡𖽝𖾏 𖼅𖽡 𖼟𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼮𖽷𖾐 𖽃𖽦𖾐,
𖼟𖽙𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼄𖽡𖾏 𖼇𖽪𖾑.
13 𖼅𖽡 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑, 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏:
“𖼀𖽡𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏 ?”
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼘𖽪 !
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼅𖽡 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼅𖽡 ,
𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐.”
