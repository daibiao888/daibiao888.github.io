3
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖽐𖼎𖽻𖾏 𖼐𖽤𖾐,
𖼐𖽤𖾐 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼙𖽠𖾏, 𖽃𖽔𖾏 𖼗𖽔 𖼐𖽤𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏,
𖼈𖽻𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼋𖽡𖽝𖾏 𖼡𖽔𖾑 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏,
𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼺𖽘𖾏 𖼖𖽺𖾑?
𖼈𖽡𖽝𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏,
𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼺𖽷𖾐 𖼖𖽔𖾏?
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
 
2 𖼟𖽡 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼮𖽢𖾏,
𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼖𖽺𖾑?
𖼟𖽡 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 𖼋𖽙𖾏,
𖼺𖽱𖾏 𖽃𖽔𖾑 𖼖𖽔𖾑 𖼀𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽐𖼚𖽺𖾏 𖽐𖼎𖽙𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
𖼟𖽡 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖽐𖼎𖽙 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼑𖽘 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽷𖾏 𖼡𖽻𖾐 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏,
𖼯𖽡𖽹𖾏 𖼳𖽙𖾑 𖼑𖽘 𖼦𖽡𖾐 𖼗𖽙𖾏, 𖼟𖽡 𖼅𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼚𖽑𖽪𖾏 𖼯𖽡𖽙𖾏 𖼺𖽷𖾐,
𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
4 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
‘ 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼘𖽪！ 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼻𖽻𖾏 𖼽𖽺 𖼗𖽔.
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽡𖽝𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼸𖽙 𖼡𖽘𖾐 𖼘𖽪𖾏? ’
𖼑𖽔𖾏 𖼘𖽪！ 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼈𖽡𖽝𖾏, 𖼟𖽡 𖼮𖽙𖾏 𖼯𖽔𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼀𖽜𖾏 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼯𖽡𖽙 𖼺𖽷𖾐. ”
𖼽𖽨 𖼊𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝
6 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏, 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼯𖽡𖽙𖾏, 𖼐𖽱𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐, 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾐 𖼄𖽡𖾐 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏. 7 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑, 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑． 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼽𖽨 𖼊𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼖𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏. 8 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏, 𖼞𖽪𖾑 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼄𖽔𖾐 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼐𖽤𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖽐𖼎𖽻𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏. 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼽𖽨 𖼊𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏. 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽔𖾏 𖼳𖽷𖾐 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏, 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼈𖽳𖾐、𖼋𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏. 10 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼈𖽡𖽝𖾏, 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏, 𖼦𖽡𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼺𖽷𖾐, 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼊𖽜𖾑． 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖽐𖼎𖽻𖾏 𖼾𖽔𖾏 𖼚𖽔𖾏 𖼞𖽡𖾑 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑, 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼽𖽨 𖼊𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖽃𖽡𖾑 𖼺𖽷𖾐． 12 𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖼘𖽪 ！ 𖼺𖽷𖾐 𖼗𖽙 𖼦𖽡𖾑!
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼑𖽔𖾏 𖼅𖽡 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏,

𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
13 𖼮𖽔 𖼽𖽔𖾑 𖼯𖽜𖾏 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
