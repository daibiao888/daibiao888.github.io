𖼗𖽪𖾏𖽐𖼮𖽔𖾏𖼟𖽪𖾏𖼐𖽑𖽡𖾏𖼖𖽪𖾑𖼊𖽠𖾏𖼎𖽻𖾐
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽜𖾏 𖼮𖽘𖾏
（𖼊𖽑𖽹𖾏 12：46-50；𖼞𖽑𖽙𖾐 3：31-35）
19 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼋𖽔𖾏 𖼋𖽘，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 ，𖼊𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼦𖽡𖾐 𖼋𖽻𖾏． 20 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼳𖽠𖾑 𖼉𖽺𖾏 𖼻𖽻𖾑 ，𖼽𖽔𖾑 𖼁𖽙𖾏 𖼟𖽡．” 21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏，𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐， 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏．”
𖼮𖽢𖾏 𖽐𖼎𖽙 𖼳𖽡𖽠𖾑 𖼊𖽪𖾏 𖼑𖽻 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 8：23-27；𖼞𖽑𖽙𖾐 4：35-41）
22 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 ，𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼀𖽘𖾑𖽑 𖼀𖽡 𖼋𖽡𖾏．” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼞𖽱𖾏 𖽐𖼞𖽙𖾏． 23 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏．𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼷𖽑𖽔𖾐 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝，𖽃𖽻𖾐 𖼖𖽔𖾏 𖼖𖽔𖾑 𖼀𖽪𖾑 𖽐𖼞𖽙𖾏 𖼦𖽡𖾐 𖼚𖽑𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝． 24 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼯𖽷𖾏 ，𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 ！ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏！𖼀𖽡𖾐 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼳𖽦𖾑 𖼋𖽘𖾏！” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼯𖽷𖾏 𖼗𖽙𖾏 ， 𖼖𖽔𖾏 𖼮𖽢𖾏 𖽐𖼎𖽙 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏， 𖼮𖽢𖾏 𖽐𖼎𖽙 𖽃𖽻𖾐 𖼎𖽺𖾏 𖽐𖼮𖽦 𖼋𖽘𖾏，𖼊𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽘𖾏． 25 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼘𖽪？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼎𖽺𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 ，𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼘𖽪 ？𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼮𖽢𖾏 𖽐𖼎𖽙 𖽃𖽻𖾐，𖼮𖽢𖾏 𖽐𖼎𖽙 𖽃𖽻𖾐 𖼊𖽪𖾏 𖼑𖽻 𖼐𖽱𖾏．”
𖼽𖽦𖾑𖼺𖽪𖾑𖼖𖽔𖾏𖼮𖽢𖾏𖽐𖼎𖽙𖽃𖽻𖾐
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐
（𖼊𖽑𖽹𖾏 8：28-34；𖼞𖽑𖽙𖾐 5：1-20）
26 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽳 𖼖𖽔𖾑 𖼺𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏， 𖼟𖽪𖾏 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 ． 27 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪，𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏 ， 𖼋𖽔𖾏 𖼸𖽱𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑．𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖼋𖽘，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏， 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏． 28 𖼐𖽱𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 ，𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏：“𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏？𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽻𖾏 𖽃𖽦𖾐 ！” 29 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏，𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏．𖼋𖽡𖾏 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼐𖽡𖾑 𖼙𖽔 𖼯𖽪 𖼯𖽪𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽻𖾏 𖽐𖼎𖽙 𖼟𖽻𖾏 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽑𖽡𖾏 𖼊𖽠𖾏 𖽐𖼊𖽠𖾏 𖼎𖽪𖾑 𖼐𖽱𖾏 ，𖼎𖽺𖾏 𖼻𖽙𖾑 𖼐𖽱𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼀𖽱𖾏 𖼊𖽪𖾏 𖼎𖽪𖾑 𖽐𖼎𖽙 𖼖𖽑𖽔𖾏 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽘𖾏，𖼀𖽡𖾑 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏． 30 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖽐𖼷𖽷𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ？” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 ‘ 𖽐𖼀𖽙𖾏’．” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼯𖽵𖾏 𖼐𖽱𖾏． 31 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽘𖾐．
