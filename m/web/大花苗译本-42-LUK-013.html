13
𖼐𖽡𖽙𖾐 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 ， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼟𖽪𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼞𖽑𖽡𖽠𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖼳𖽡𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 . 2 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “ 𖼅𖽡 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑， 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽔𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼘𖽪𖾏？ 3 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼦𖽡𖾐 𖼾𖽙！ 𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻，𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼛𖽙𖾏 𖼋𖽔！ 4 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼳𖽡𖾑 𖼘𖽙 𖼽𖽔𖾏 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼢𖽻 𖼙𖽠𖾏，𖼸𖽺𖾏 𖼋𖽔 𖼟𖽻𖾏 𖼾𖽡𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼅𖽡 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽔𖾏 ？ 5 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼦𖽡𖾐 𖼾𖽙！𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻，𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼛𖽙𖾏 𖼋𖽔！”
𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽑𖽡𖾏： “ 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐， 𖼯𖽡𖽙 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔𖾏 𖼅𖽔 𖼑𖽜𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，𖼐𖽱𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏． 7 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼐𖽑𖽡𖾏：‘𖼟𖽡 𖼑𖽔𖾏 ，𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼐𖽡𖾑，𖽐𖼎𖽑𖽦𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏．𖽐𖼊𖽙𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏，𖼦𖽡𖾐 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼚𖽠𖾐 𖽐𖼮𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏 ！’ 8 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：‘𖼮𖽵𖾑 𖼘𖽪 ！𖼳𖽡𖽻𖾏 𖼐𖽔𖾑 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏，𖼋𖽙𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽠𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖽃𖽔 𖼗𖽔𖾏，𖼎𖽺𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖾑； 9 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼷𖽷𖾏 𖼎𖽺𖾏 𖽐𖼊𖽙𖾑 𖼺𖽘𖾏．’”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼚𖽔𖾑
10 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏, 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 11 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼏𖽻 𖼖𖽡𖽾 𖼯𖽵𖾏 𖼐𖽱𖾏 ，𖼄𖽙𖾐 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏，𖼐𖽱𖾏 𖼞𖽑𖽻𖾑 𖼚𖽔𖾑 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼮𖽤𖾐． 12 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏，𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔 𖼁𖽙𖾏，𖼟𖽡 𖼇𖽳𖾑 𖼛𖽡𖾏 𖼊𖽪𖾑 𖼄𖽙𖾐 𖼐𖽡𖾑 𖼋𖽘𖾏！” 13 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼀𖽔𖾐 𖼎𖽻𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏，𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼦𖽡𖾐 𖼞𖽑𖽻𖾑 𖼚𖽔𖾑 𖼋𖽘𖾏， 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 14 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽳𖾑 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽡𖾑，𖼁𖽙𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼞𖽑𖽙𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐， 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ， 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏；𖼐𖽡𖽙𖾐 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖽐𖼚𖽙 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼋𖽔𖾏 𖽐𖼊𖽜 𖼟𖽪𖾏 𖼞𖽑𖽙𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐， 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑．” 15 𖼮𖽵𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏，𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼦𖽡𖾐 𖼚𖽠𖾏 𖼑𖽨𖾏 𖽐𖼎𖽙 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼚𖽘𖾐，𖼮𖽤𖾐 𖼅𖽻𖾏 𖼦𖽻𖾏 𖽃𖽻𖾐 𖼘𖽪？ 16 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖾑， 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏，𖼑𖽔𖾏 𖼘𖽪！𖼐𖽱𖾏 𖼊𖽻𖾏 𖼺𖽔𖾐 𖼊𖽔 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 𖼋𖽘𖾏，𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼚𖽠𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼘𖽪？” 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼋𖽘𖾏；𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏， 𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝 .
𖼐𖽡𖽙𖾐𖼐𖽑𖽪𖾐𖼳𖽙𖾏𖼞𖽑𖽙𖾏𖼄𖽙𖾐𖽐𖼷𖽘𖾐
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼮𖽘𖾑 𖼻𖽻𖾐 𖼛𖽙𖾐 𖽃𖽔 𖼛𖽡𖾏
（𖼊𖽑𖽹𖾏13：31-33；𖼞𖽑𖽙𖾐4：30-32）
18 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪？ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼗𖽙𖾏 𖽐𖼮𖽔𖾏 𖼖𖽺𖾑 ？ 19 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼛𖽡𖾐 𖼮𖽘𖾑 𖼻𖽻𖾐 𖼛𖽙𖾐 𖽃𖽔 𖼛𖽡𖾏， 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼀𖽳𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏，𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼘𖽑𖽙𖾐 𖼊𖽡𖽝𖾑 𖽃𖽡𖾐 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏，𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏，𖼋𖽔𖾏 𖼳𖽻𖾏 𖼉𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡．”
𖽐𖼊𖽪𖾏𖼞𖽡𖾏𖼄𖽱𖾏𖼞𖽪𖽙𖼺𖽱𖾏𖽃𖽡𖾐𖼚𖽡𖾐𖽃𖽔𖼚𖽡𖾏𖼻𖽻𖾐𖼛𖽙𖾐
20 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏： “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼗𖽙𖾏 𖽐𖼮𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼘𖽪？ 21 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏，𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽔𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼊𖽱𖾏 𖽃𖽡𖾐 𖼳𖽜𖾐 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖽐𖼚𖽙，𖼋𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽪𖾏 𖼗𖽙𖾏．”
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏
（𖼊𖽑𖽹𖾏 7：13-14，21-23）
22 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ，𖼙𖽻 𖽐𖼎𖽙 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼮𖽑𖽡𖽻𖾏 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐． 23 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼋𖽡𖽝𖾏 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼯𖽠 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼘𖽪𖾏？” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏： 24 “ 𖼅𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏．𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼮𖽑𖽠𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏． 25 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼡𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏，𖼅𖽡 𖼳𖽠𖾑 𖽐𖼎𖽙𖾏 𖼻𖽻𖾑 𖽐𖼀𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼐𖽑𖽡𖾏：‘𖼮𖽵𖾑 𖼘𖽪 ！𖼑𖽜 𖼀𖽡𖾐 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏．’ 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：‘ 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼅𖽡 ，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼅𖽡 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏！’ 26 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼅𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏： ‘ 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼙𖽔 𖼑𖽻𖾏 𖼙𖽔 𖼦𖽻𖾏，𖼟𖽡 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐．’ 27 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼅𖽡 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏，𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑，𖼀𖽜𖾑 𖼞𖽪𖾑 𖼙𖽠𖾏！’ 28 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐、𖼽𖽡𖾑 𖼺𖽔𖾐、 𖼽𖽔𖾐 𖼞𖽙，𖽐𖼎𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼙𖽠𖾏， 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼟𖽡𖾏 𖼋𖽙𖾏 𖼐𖽡𖽝𖾑𖽑 𖼷𖽷𖾏 𖼞𖽹𖾑． 29 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐、 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏、𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑、𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼋𖽔𖾏，𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏． 30 𖼈𖽡𖽝𖾏 𖼑𖽔𖾏 𖼘𖽪！ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽑𖽔𖾏， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐．”
𖼑𖽜 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼗𖽙𖾏 𖼞𖽑𖽡𖾏
（𖼊𖽑𖽹𖾏 23：37-39）
31 𖼐𖽡𖽙𖾐 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑， 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼺𖽘𖾏 𖼅𖽻𖾏 𖼙𖽠𖾏 𖼦𖽡𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼖𖽡 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼟𖽡．” 32 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽪𖾑 𖼞𖽡𖾏 𖼋𖽱𖾏 𖽃𖽡𖾑：‘ 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼀𖽡 𖼟𖽡 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼞𖽑𖽙𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐，𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏 𖼋𖽘𖾏．’ 33 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏，𖼄𖽑𖽙𖾏 𖼐𖽔𖾏、 𖼀𖽡 𖼟𖽡、 𖽃𖽔 𖼑𖽘 𖼟𖽡𖾏，𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼅𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼗𖽔𖾏 𖼳𖽦𖾑 ． 34 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！𖼟𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐，𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼋𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑，𖼞𖽪𖾑 𖼀𖽡𖾑 𖼯𖽠 𖼟𖽻𖾐 𖽐𖼮𖽑𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼡𖽹𖾐 𖼦𖽡 𖼸𖽱 𖼐𖽱𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼎𖽻𖾏 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 ． 35 𖼑𖽔𖾏 𖼘𖽪！𖼅𖽡 𖽐𖼞𖽔𖾏 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼎𖽻𖾏 𖼅𖽡．𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼎𖽑𖽻𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑，𖼽𖽔𖾑 𖼋𖽙𖾏 𖼸𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏 ： ‘𖼷𖽷𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼮𖽵𖾑 𖽐𖼷𖽷𖾏 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏．’ ”
