𖼎𖽑𖽻𖾏 𖼀𖽔𖾑 𖼀𖽡𖽝 𖼊𖽑𖽔 𖼗𖽪𖾏 𖼚𖽑𖽡𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏
11
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼊𖽔𖾐 𖼗𖽪𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼗𖽪𖾏. 2 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽹𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼳𖽷𖾏 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽷𖾐 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽔𖾏! 𖼀𖽡𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽔𖾐 𖼑𖽡 𖼋𖽺𖾏, 𖼞𖽑𖽡𖽠𖾏 𖼑𖽡 𖼋𖽺𖾏 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼳𖽦𖾑. 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼑𖽡 𖼋𖽺𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐, 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼛𖽙𖾏 𖽃𖽔𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼞𖽙𖾏.” 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽔𖾏! 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼊𖽑𖽘𖾐, 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼺𖽡𖽝𖾐 𖼊𖽠𖾏 𖽐𖼊𖽪𖾏, 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼀𖽡𖾐 𖽐𖼷𖽷𖾏, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼙𖽠𖾏.” 5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏, 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼊𖽑𖽡𖽝𖾐 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏. 6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪！ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼊𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽜𖾑 𖼗𖽪𖾏, 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼗𖽙𖾏, 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽡𖽝𖾑. 7 𖼋𖽔𖾏 𖼘𖽪 ！ 𖼀𖽡𖾐 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼺𖽘𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼅𖽻𖾏 𖼗𖽙𖾏.” 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽙𖾏 𖼼𖽙 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼋𖽘𖾏. 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏, 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼀𖽔𖾑 𖼀𖽡𖽝 (“ 𖼀𖽔𖾑 𖼀𖽡𖽝”𖼟𖽪𖾏 “𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 ”).
𖼳𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 1：24 -27）
10 𖼳𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐: 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏, 𖼳𖽔𖾐 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼇𖽔 𖼺𖽔𖾐. 11 𖼳𖽔𖾐 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼇𖽔 𖼺𖽔𖾐 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼀𖽜𖾐 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑.
12 𖼽𖽔𖾏 𖼇𖽔 𖼺𖽔𖾐 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖼅𖽔𖾏 𖼳𖽔𖾑 𖼖𖽔𖾑． 13 𖼽𖽔𖾏 𖼇𖽔 𖼺𖽔𖾐 𖼅𖽔𖾏 𖼳𖽔𖾑 𖼖𖽔𖾑 𖽃𖽔𖾑 𖼡𖽜𖾑, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼖𖽡 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑.
14 𖼳𖽔𖾑 𖼖𖽔𖾑 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼀𖽳. 15 𖼳𖽔𖾑 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼀𖽳 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼖𖽡 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑.
16 𖼳𖽡𖾑 𖼀𖽳 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼚𖽻𖾐 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼇𖽔 𖼖𖽳. 17 𖼳𖽡𖾑 𖼀𖽳 𖼅𖽔𖾏 𖼇𖽔 𖼖𖽳 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 18 𖼇𖽔 𖼖𖽳 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼖𖽔𖾑 𖽂𖽪. 19 𖼇𖽔 𖼖𖽳 𖼅𖽔𖾏 𖼖𖽔𖾑 𖽂𖽪 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼖𖽡 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑.
20 𖼖𖽔𖾑 𖽂𖽪 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼖𖽪. 21 𖼖𖽔𖾑 𖽂𖽪 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼖𖽪 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼖𖽡 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑.
22 𖼳𖽡𖾑 𖼖𖽪 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼐𖽔 𖼦𖽳𖾏. 23 𖼳𖽡𖾑 𖼖𖽪 𖼅𖽔𖾏 𖼐𖽔 𖼦𖽳𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑.
24 𖼐𖽔 𖼦𖽳𖾏 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽔𖾑. 25 𖼐𖽔 𖼦𖽳𖾏 𖼅𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽔𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼟𖽻𖾏 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑. 26 𖼊𖽑𖽔𖾑 𖼖𖽔𖾑 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏, 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔、𖼐𖽔 𖼦𖽳𖾏、𖼦𖽔𖾐 𖼖𖽔 𖼋𖽘𖾏.
𖼊𖽑𖽔𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏
27 𖼊𖽑𖽔𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐: 𖼊𖽑𖽔𖾑 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔、𖼐𖽔 𖼦𖽳𖾏、𖼦𖽔𖾐 𖼖𖽔； 𖼦𖽔𖾐 𖼖𖽔 𖼅𖽔𖾏 𖼘𖽙 𖼊𖽳. 28 𖼦𖽔𖾐 𖼖𖽔 𖼋𖽔 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖽃𖽡𖾑 𖽂𖽪 𖽃𖽳𖾐 𖼺𖽘𖾏, 𖼋𖽔 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏. 29 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔、𖼐𖽔 𖼦𖽳𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼷𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼐𖽤𖾐； 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼺𖽔𖾐 𖼖𖽹； 𖼐𖽔 𖼦𖽳𖾏 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼄𖽡 𖼮𖽢𖾑, 𖼟𖽪𖾏 𖼦𖽔𖾐 𖼖𖽔 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏； 𖼦𖽔𖾐 𖼖𖽔 𖼟𖽪𖾏 𖼄𖽡 𖼮𖽢𖾑 𖽐𖼎𖽙 𖼽𖽡 𖼮𖽢𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐. 30 𖼺𖽔𖾐 𖼖𖽹 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏, 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼣𖽔𖾑 𖼾𖽻𖾏.
31 𖼊𖽑𖽔𖾑 𖼖𖽔𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼷𖽔𖾑 𖼞𖽱𖾏 𖼞𖽱𖾑 𖼦𖽔𖾐 𖼖𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼘𖽙 𖼊𖽳, 𖼊𖽑𖽡𖽝𖾐 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼺𖽔𖾐 𖼖𖽹, 𖼇𖽳𖾑 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖽃𖽡𖾑 𖽂𖽪 𖽃𖽳𖾐 𖼺𖽘𖾏, 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼦𖽔𖾐 𖼖𖽔 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 32 𖼊𖽑𖽔𖾑 𖼖𖽔𖾑 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼖𖽡 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏, 𖼋𖽔 𖼎𖽻𖾏 𖼦𖽔𖾐 𖼖𖽔 𖼺𖽘𖾏.
