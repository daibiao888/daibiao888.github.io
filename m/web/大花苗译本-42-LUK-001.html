69 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼾𖽡𖾏，

70 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼗𖽙𖾏 ，𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，
71 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑，
𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼀𖽡𖾐 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏．
72 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐，
𖼎𖽺𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏．
73 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐𖼏𖽵 𖼗𖽪𖾏，
74 𖼄𖽔𖾐 𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽨𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，
75 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽜𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐， 𖼎𖽺𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼞𖽡𖾑 𖽐𖼮𖽑𖽹𖾏，
𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏．
76 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼘𖽪！𖼟𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐，

78 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏，
