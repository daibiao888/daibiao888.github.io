𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼸𖽙 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐，𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽻． 2 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ，𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼾𖽘𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼁𖽘𖾏 𖽃𖽻𖾐，𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼗𖽪𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼀𖽡 𖼺𖽷𖾏 𖼊𖽔𖾏，𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼖𖽪𖾐 𖼡𖽘𖾐 𖽃𖽔𖾏 𖽐𖼮𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏． 3 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼀𖽱𖾏 𖼊𖽪𖾐 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼅𖽔𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐、𖼅𖽔𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑、𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼡𖽑𖽔𖾑 𖼀𖽡𖾏 𖼮𖽑𖽤𖾏． 4 （𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏 𖼩𖽪𖾏 𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽻𖾐 𖽃𖽡𖾑，𖽃𖽻𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼡𖽔 𖼋𖽱𖾏 𖼺𖽪𖾐 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏，𖼦𖽡𖾐 𖼞𖽳𖾑 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼎𖽺𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏．） 5 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼄𖽙𖾐 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏． 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏，𖼀𖽻𖾐 𖼐𖽱𖾏 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼋𖽘𖾏，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏？” 7 𖼊𖽪𖾐 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽙𖾏 𖼞𖽪𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽻𖾐，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏，𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽪𖾐 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼊𖽡𖾏 𖼘𖽑𖽻𖾑 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏．” 8 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏，𖼯𖽡𖽙𖾏 𖼟𖽡 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐 𖼅𖽻𖾏 𖼦𖽡𖾑！” 9 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏．
