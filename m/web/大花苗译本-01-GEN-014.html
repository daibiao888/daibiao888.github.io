𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼮𖽨𖾏 𖼘𖽙 𖼊𖽳
13 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼗𖽙𖾏, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔； 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾏 𖼖𖽡𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑．𖼄𖽔𖾏 𖼖𖽡𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼞𖽙 𖼊𖽑𖽡𖽝𖾐 𖼽𖽔𖾏 𖼐𖽹𖾐 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏, 𖼙𖽔 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏. 14 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼊𖽱 ( “𖼊𖽪𖾐 𖽐𖼊𖽱”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼞𖽱𖾑 𖼋𖽡𖾏”) 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏, 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽪𖾑 𖼾𖽨 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖽃𖽡𖾑 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼟𖽻𖾏 𖼾𖽡𖾏 𖼗𖽜𖾏, 𖼖𖽠𖾑 𖼟𖽪𖾏 𖼸𖽙 𖼊𖽗𖾏. 15 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼋𖽘 𖼄𖽑𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 𖼊𖽔𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖼎𖽺𖾏 𖼖𖽠𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖽃𖽡𖾑 𖼦𖽙 𖼀𖽔𖾑, 16 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼊𖽱 𖼘𖽙 𖼊𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐, 𖼎𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔 𖼁𖽙𖾏、𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼞𖽑𖽔𖾏 𖼦𖽔𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏.
𖼄𖽳 𖼮𖽡𖾑 𖼳𖽡𖾐 𖼊𖽳 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔
17 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼊𖽔𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼮𖽡𖾑 𖼊𖽔𖾏 𖼘𖽻𖾐 𖼄𖽔𖾐, 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼳𖽔𖾑 𖽂𖽺𖾑 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏； 𖼳𖽔𖾑 𖽂𖽺𖾑 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽘𖾑. 18 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽳 𖼮𖽡𖾑 𖼳𖽡𖾐 𖼊𖽳 𖼯𖽡𖽙𖾏 𖼳𖽡𖽝𖾐 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑, 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏； 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏. 19 𖼐𖽱𖾏 𖼑𖽜 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼐𖽑𖽡𖾏:
“𖼊𖽻𖾏 𖼅𖽪𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽵𖾑、 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐,
𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔！
20 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼄𖽔𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏！”
𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼗𖽙𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼄𖽳 𖼮𖽡𖾑 𖼳𖽡𖾐 𖼊𖽳. 21 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼞𖽑𖽡𖽠𖾏 𖼅𖽻𖾏 𖼦𖽡𖾑!” 22 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖽐𖼎𖽙 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼙𖽔 𖽐𖼎𖽙 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽵𖾑、 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼏𖽵 𖼗𖽪𖾏： 23 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼁𖽡𖽝 𖽃𖽡𖾑, 𖽃𖽡𖾐 𖼷𖽙𖾏 𖼺𖽙𖾑, 𖽃𖽡𖾐 𖼸𖽙𖾐 𖼖𖽑𖽔𖾏 𖼞𖽑𖽻𖾏, 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼟𖽡 𖼐𖽑𖽡𖾏: ‘𖼞𖽪𖾑 𖼟𖽙𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼅𖽔𖾏 𖼊𖽻𖾏！’ 24 𖼯𖽔 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼽𖽔𖾑 𖼑𖽻𖾏 𖽃𖽡𖾑, 𖼊𖽑𖽡𖽝𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼐𖽹𖾐、𖼽𖽡𖾑 𖼳𖽷𖾏 𖼞𖽙、𖼄𖽔𖾏 𖼖𖽡𖾏 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏, 𖼟𖽙𖾏 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼅𖽻𖾏.”
