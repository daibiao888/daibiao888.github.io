（𖼋𖽠𖾏 37：25 - 28）
30
“𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 . 2 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼞𖽪𖾐 , 𖽐𖼊𖽡𖾑 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼇𖽘𖾑 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼺𖽡𖽝𖾐 𖽃𖽔𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 . 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽻𖾐 𖼋𖽪 𖼞𖽪𖾑 𖼞𖽪𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽘 𖼦𖽡𖾏 𖼊𖽡𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾐 . 3 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑 𖼡𖽑𖽠𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡 𖼙𖽻, 𖼊𖽑𖽡𖽝𖾐 𖼚𖽻𖾐 𖼋𖽪 𖼞𖽪𖾑 𖼞𖽪𖾐 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼎𖽑𖽹𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼙𖽻𖾏. 4 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽔 𖼉𖽙𖾏 𖼞𖽪𖾐 𖼊𖽡𖾏 𖼎𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼎𖽑𖽹𖾏 𖼙𖽻𖾏 𖽃𖽡𖾑 𖼉𖽺𖾏 𖼡𖽘𖾐 , 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾐 𖼺𖽘𖾐 , 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼙𖽡 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 , 𖼟𖽙𖾏 𖼻𖽻𖾏 𖼞𖽱𖾑 𖽃𖽔𖾑 𖼏𖽘 . 5 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽱𖾑 , 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏 . 6 𖼽𖽔𖾑 𖼮𖽑𖽳𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼀𖽑𖽙𖾑 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽺𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 , 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 , 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼦𖽡𖾏 𖼁𖽙𖾏 . 7 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼽𖽔𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖼷𖽑𖽔𖾏 𖼋𖽡𖾏 𖼟𖽡𖾏 𖼐𖽱𖾏 𖼞𖽑𖽙𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼐𖽡𖾑 . 8 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼢𖽹𖾐 𖼮𖽡𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼐𖽡𖾑 , 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼳𖽤𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼷𖽑𖽷𖾑 . 9 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽑𖽷𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼳𖽤𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑. 10 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼽𖽔𖾑 𖼑𖽜 𖽃𖽔𖾑 𖼏𖽘 𖼞𖽪𖾑 𖼞𖽪𖾐 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 , 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑 , 𖼑𖽜 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽠𖾏 𖼾𖽔𖾏 , 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼏𖽵 , 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .”
𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏
