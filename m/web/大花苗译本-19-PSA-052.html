𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏
𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽙𖾑 𖼽𖽡𖾏 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼺𖽻𖾐 𖼘𖽙, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼄𖽡𖾐 𖼖𖽳 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽡𖾑 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
（𖼺𖽔𖾐 𖽃𖽡𖾐 22：6 - 23）
52
𖼊𖽪𖾐 𖼋𖽻 𖼘𖽪！ 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏？
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖼯𖽡𖽙 .
2 𖼟𖽡 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼐𖽡𖾑 𖼘𖽪！
𖼟𖽡 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼚𖽺𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼷𖽑𖽪𖾐, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽑𖽪𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼮𖽳𖾏 .
3 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖽐𖼮𖽑𖽠𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 . （𖼳𖽡𖾏 𖼖𖽔𖾑）
4 𖼊𖽪𖾑 𖽃𖽔𖾏 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼘𖽪！
𖼟𖽡 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐．
 
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼟𖽡 𖼺𖽘𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼟𖽡, 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼄𖽔𖾐 𖼟𖽡 𖼚𖽙𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 . （𖼳𖽡𖾏 𖼖𖽔𖾑）
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼚𖽙𖾏 𖼐𖽱𖾏 ,
7 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪！ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑,
𖼯𖽔 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼊𖽻𖾏, 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾐 𖼖𖽻𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽙𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼾𖽨𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.”
 
8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖽐𖼚𖽙 𖽃𖽡𖾑 𖼞𖽗𖾐 𖼖𖽗 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼷𖽔𖾐,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏.
9 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑．
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼋𖽡𖽝 .
