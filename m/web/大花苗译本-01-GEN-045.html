𖼽𖽙 𖼺𖽳 𖽐𖼷𖽜𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼞𖽔𖾏 𖼅𖽔, 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼞𖽪𖾑 𖼙𖽠𖾏！” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼯𖽜𖾐 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐. 2 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏, 𖼇𖽔 𖼘𖽻𖾐 𖼾𖽡𖾏 𖼖𖽔𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑. 3 𖼽𖽙 𖼺𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼋𖽡𖽝𖾏 𖼡𖽑𖽔𖾑 𖼐𖽡𖽙𖾐?” 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝.
4 𖼽𖽙 𖼺𖽳 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼋𖽔𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑.”𖼸𖽻𖾏 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼽𖽙 𖼺𖽳, 𖼟𖽪𖾏 𖼷𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽺 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼙𖽠𖾏 𖽃𖽡𖾑. 5 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼋𖽺 𖼞𖽪𖾑 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼙𖽠𖾏 𖼾𖽨𖾏 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐、𖼾𖽨𖾏 𖼳𖽪𖾐 𖼾𖽨𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖽐𖼊𖽡𖾏 𖼋𖽔𖾏, 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑. 6 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙 𖼡𖽻𖾐 𖼦𖽡𖾐 𖼷𖽷𖾐, 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼡𖽻𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼺𖽷𖾐. 7 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖽐𖼊𖽡𖾏 𖼋𖽔𖾏, 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼅𖽡 𖼮𖽘𖾑 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽑𖽜𖾐 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼮𖽨𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑. 8 𖼑𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼗𖽙𖾏, 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼅𖽡, 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽱𖾏 𖼉𖽺𖾏, 𖽃𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼮𖽵𖾑, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏.
𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼷𖽹𖾏 𖼽𖽔𖾐 𖼞𖽙
9 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼇𖽺𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: ‘ 𖼟𖽡 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽵𖾑, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖽐𖼡𖽜𖾏 𖼋𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽳𖾑𖽑. 10 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾏 𖼞𖽱𖾑, 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽙𖾑 𖼳𖽔𖾑 𖼊𖽡𖾐, 𖼟𖽙𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑. 11 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼊𖽪𖾑 𖼯𖽡𖽝𖾏 𖼟𖽡, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼺𖽷𖾐, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏.’ 12 𖼑𖽔𖾏 𖼘𖽪! 𖼅𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐𖽃𖽔𖾑 𖼅𖽔 𖼖𖽔𖾏 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 ，𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼗𖽪𖾏. 13 𖼅𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽻𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽙𖾏, 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼈𖽹𖾐, 𖼎𖽺𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼀𖽜𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼇𖽺𖾏.” 14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽙 𖼺𖽳 𖼁𖽔 𖼐𖽱𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼦𖽡𖾑 𖼚𖽘𖾐 𖼮𖽡𖽝𖾏 𖼐𖽡𖽝𖾑, 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼖𖽔𖾏 𖼁𖽔 𖼐𖽱𖾏 𖼦𖽡𖾑 𖼚𖽘𖾐 𖼮𖽡𖽝𖾏 𖼐𖽡𖽝𖾑. 15 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼡𖽑𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼁𖽔 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽡𖽝𖾑, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏.
16 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼟𖽻, 𖼐𖽑𖽡𖾏: “𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼋𖽔𖾏 𖼋𖽘𖾏.” 𖼇𖽔 𖼘𖽻𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽻 𖼷𖽻𖾏 𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝. 17 𖼇𖽔 𖼘𖽻𖾐 𖽐𖼎𖽙 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼡𖽑𖽔𖾏 𖼇𖽺𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏: ‘ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼮𖽑𖽳𖾑 𖽐𖼎𖽘𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼯𖽦𖾏 𖼮𖽹𖾑, 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.’ 18 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼎𖽙 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼛𖽙 𖼻𖽻𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼅𖽡, 𖼟𖽙𖾏 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏. 19 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼖𖽠𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐, 𖼅𖽻𖾏 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼣𖽔𖾑 𖼽𖽻𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽤𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼅𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼞𖽑𖽔𖾏 𖼨𖽑𖽳𖾏 𖼋𖽔𖾏. 20 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽡𖽝.”
21 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏, 𖼽𖽙 𖼺𖽳 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽔𖾐 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼊𖽑𖽡𖽝𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼑𖽻𖾏 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏. 22 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼷𖽔𖾏 𖼞𖽡𖾑 𖼛𖽔𖾏 𖼮𖽑𖽙𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼙𖽡 𖼑𖽡𖽝𖾏, 𖼀𖽜𖾐 𖼷𖽔𖾏 𖼞𖽡𖾑 𖼛𖽔𖾏 𖼮𖽑𖽙𖾏. 23 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼋𖽪𖾏 𖼀𖽡𖾏 𖼊𖽪𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑, 𖼮𖽹𖾑 𖼣𖽹 𖼮𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼟𖽻𖾏 𖼋𖽪𖾏, 𖼮𖽹𖾑 𖼡𖽻𖾐 𖼡𖽜𖾏 、𖼳𖽡𖽝𖾐 𖽐𖼎𖽙 𖼻𖽻𖾐, 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼋𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽡𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑. 24 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽙 𖼺𖽳 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼺𖽷𖾐 𖼙𖽠𖾏, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽡𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.” 25 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏, 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 26 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽙 𖼺𖽳 𖼐𖽡𖽙𖾐 𖼺𖽷𖾐, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼉𖽻 𖼷𖽻𖾏.”𖼽𖽔𖾐 𖼞𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼸𖽡𖽝𖾏 𖼯𖽔, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏. 27 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙,𖼎𖽺𖾏 𖼁𖽙𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼷𖽷𖾐 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏,𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼎𖽺𖾏 𖼳𖽙𖾑. 28 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽑𖽡𖾏: “𖼻𖽻𖾏 𖼋𖽘𖾏! 𖼻𖽻𖾏 𖼋𖽘𖾏! 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼽𖽙 𖼺𖽳 𖼐𖽡𖽙𖾐 𖼺𖽷𖾐, 𖼷𖽪𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽔 𖼺𖽷𖾐, 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 𖽃𖽡𖾐 𖼡𖽔 𖼅𖽔.”
