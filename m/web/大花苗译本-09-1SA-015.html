𖼺𖽻𖾐 𖼘𖽙 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑
4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼺𖽻𖾐 𖼘𖽙 𖼨𖽑𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼊𖽑𖽡 𖼖𖽔𖾑 𖼽𖽡𖽾𖾑, 𖼮𖽔𖾑 𖼾𖽱 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼏𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽠𖾏 𖼅𖽻𖾏 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼉𖽻𖾏, 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼉𖽻𖾏. 5 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽻, 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽃𖽔𖾐 𖼟𖽱𖾐 𖼦𖽘𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼏𖽙 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏. 6 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼐𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼇𖽳𖾑 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼺𖽘𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼦𖽡𖾑, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼄𖽔𖾐 𖼅𖽡 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼸𖽻 𖼊𖽔𖾏 𖼺𖽘𖾏. 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼅𖽡 𖼙𖽔 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽡𖾑 𖼐𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾑 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 7 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼊𖽻𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼎𖽑𖽻𖾏 𖼦𖽔𖾐 𖼇𖽺𖾑 𖼖𖽔𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖽃𖽡𖾑 𖼳𖽪𖾑 𖽃𖽳𖾐， 8 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼮𖽢 𖼮𖽡𖽝𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼞𖽑𖽔𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 . 9 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼽𖽔𖾏 𖼮𖽢 , 𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼑𖽨𖾏、𖼮𖽑𖽷𖾏、𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏、𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏, 𖼈𖽡𖽝𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼾𖽔 𖽃𖽡𖾑, 𖼞𖽑𖽔𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏.
