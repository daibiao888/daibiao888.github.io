𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼷𖽻𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏
19
𖼺𖽻𖾐 𖼘𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽑𖽡𖾏: 𖼽𖽔𖾑 𖼊𖽔𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼺𖽘𖾏, 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏. 2 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼉𖽺𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽔𖾏 𖼟𖽡, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼀𖽡 𖼟𖽡 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐, 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼈𖽺𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑. 3 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼚𖽺𖾏 𖽃𖽡𖾑, 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼺𖽘𖾑 𖼖𖽡𖾏 𖼊𖽘𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏. 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡.” 4 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼑𖽜 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏 𖼻𖽻𖾏 𖼗𖽪𖾏, 𖼐𖽑𖽡𖾏: “𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡, 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏. 5 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼅𖽻𖾏 𖼊𖽔𖾏 𖼸𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼊𖽔𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑, 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼾𖽨𖾏 𖼾𖽔𖾏 𖼖𖽺𖾑?” 6 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽻 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼺𖽘𖾏 , 𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽔𖾏 𖼐𖽱𖾏.” 7 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼨𖽑𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼗𖽙𖾏, 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼺𖽻𖾐 𖼘𖽙, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
