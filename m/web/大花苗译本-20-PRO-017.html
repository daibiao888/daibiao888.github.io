17
𖼅𖽔𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖽐𖼞𖽡 𖼊𖽔𖾐 𖼈𖽡𖽝𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼺𖽻𖾑 𖽐𖼮𖽳，
𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼀𖽪𖾑 𖽐𖼞𖽔𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼦𖽡𖾏 𖼊𖽔𖾏.
2 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽵𖾏 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼎𖽺𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏.
3 𖼖𖽪𖾑 𖼡𖽑𖽠𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼗𖽙 𖼮𖽡𖽺𖾏 𖼑𖽡𖽝𖾏，𖽃𖽔𖾑 𖼷𖽙𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼮𖽡𖽺𖾏 𖼞𖽪𖾐；
𖼯𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽙𖾑 𖽐𖼮𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
4 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼳𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼑𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑, 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼗𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏, 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼐𖽱𖾏 ；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼗𖽔 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼀𖽜𖾑 𖼦𖽡𖾐 𖼛𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐.
6 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼊𖽪𖾐 𖼘𖽑𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑,
𖼉𖽺𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼐𖽑𖽡𖾏 𖼻𖽻𖾏 𖼗𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼦𖽡 𖽐𖼷𖽷,
𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼯𖽡𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏 𖼯𖽡𖽙𖾏．
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏, 𖼑𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏.
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏, 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽪 𖼯𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼗𖽔 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽠𖾏， 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼽𖽺 𖼗𖽔 𖼷𖽑𖽡𖽝𖾏 𖼇𖽳𖾑 𖼷𖽑𖽡𖽝𖾏.
10 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼊𖽻𖾏 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼟𖽻𖾏.
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼮𖽔𖾑 𖼸𖽠𖾏 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 .
12 𖼁𖽻𖾏 𖼏𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼚𖽹𖾏 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼛𖽙𖾏 𖼙𖽠𖾏,
𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼮𖽷𖾐 𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼷𖽷𖾏 𖼖𖽘𖾑 .
13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽪𖾏 𖼁𖽻𖾏 𖼻𖽻𖾏,
𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏.
14 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼺𖽱𖾏 𖼮𖽙𖾏 𖽃𖽻𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐；
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼺𖽪𖾐 𖼽𖽔𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
15 𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖽃𖽔𖾐 𖼼𖽘𖾐 𖼞𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐.
16 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼑𖽡𖽝𖾏 𖼅𖽻𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼖𖽺𖾑?
17 𖼽𖽺 𖼗𖽔 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏,
𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼅𖽔𖾏 𖼯𖽡𖽙 .
18 𖼐𖽡𖽙𖾐 𖼼𖽷𖾏 𖼼𖽙 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼀𖽔𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾑
𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 .
19 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏, 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼾𖽔𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐, 𖼾𖽨𖾏 𖼽𖽔𖾑 𖼦𖽡 𖼋𖽙 .
20 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽝𖾏 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐, 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼁𖽪 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽑𖽡𖾏, 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽐𖼚𖽙 .
21 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 , 𖼾𖽨𖾏 𖼎𖽺𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼙𖽔 𖼚𖽙𖾏.
22 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏；
𖼖𖽡𖽾 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 .
23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 ,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼡𖽘𖾐 𖼎𖽙𖾏 𖼇𖽻𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐.
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼑𖽡𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏.
25 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼟𖽙𖾏 𖼉𖽺𖾏 𖼊𖽳𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼑𖽻𖾏 𖽃𖽦𖾐.
26 𖼨𖽑𖽢𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏；
𖽐𖼊𖽻𖾏 𖼖𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 .
27 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 ， 𖼾𖽨𖾏 𖼯𖽵𖾏 𖼾𖽨𖾏 𖼗𖽪𖾏 ；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽡𖾑 𖼙𖽻 𖼻𖽻𖾏.
28 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼗𖽪𖾏 , 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ；
𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐.
