𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐
21
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼊𖽪𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼊𖽡𖾐 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼖𖽔𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘． 2 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼘𖽻𖾑𖽑 𖼗𖽙𖾏，𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏，𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼼𖽘𖾏 𖼷𖽑𖽷𖾏 𖼋𖽙𖾏 𖼾𖽨𖾏 𖼈𖽻𖾑． 3 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼐𖽑𖽡𖾏：
“ 𖼑𖽔𖾏 𖼘𖽪！𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，
𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐，
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏．
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐，𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
4 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽔𖾏 𖼅𖽔 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼺𖽘𖾏；
𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼋𖽔 𖼋𖽘𖾏，
𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐、𖼐𖽡𖽝𖾑 𖼼𖽷𖾏、𖽃𖽦𖾐 𖽐𖼀𖽜𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼑𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏．”
5 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼐𖽑𖽡𖾏：“ 𖼑𖽔𖾏 𖼘𖽪！𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼞𖽑𖽔𖾏 𖼚𖽑𖽡𖾏 𖼮𖽑𖽦𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏．” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼽𖽔𖾑 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼮𖽡𖽝𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻， 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏．” 6 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼊𖽡𖽝𖾑 𖼋𖽘𖾏！𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼖𖽔𖾑 𖼇𖽔 ，𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽙 𖼄𖽞 𖼞𖽔；𖼞𖽪𖾑 𖼟𖽪𖾏 𖼁𖽱𖾏，𖼞𖽪𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏．𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼷𖽑𖽠𖾑，𖼚𖽠𖾐 𖼚𖽠𖾐 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼦𖽻𖾏． 7 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏；𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼐𖽱𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐． 8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜、𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏、𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽠𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼖𖽨 𖼦𖽫 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼯𖽡 𖽃𖽡𖾑； 𖼟𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼟𖽪𖾏 𖼋𖽔．”
