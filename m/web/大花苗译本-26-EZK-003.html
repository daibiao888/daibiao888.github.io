3
𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼊𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼐𖽡𖾑 , 𖼑𖽻𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼐𖽡𖾑 𖼺𖽘𖾏, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏.” 2 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽻𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼐𖽡𖾑. 3 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽙𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 , 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼟𖽡 𖽃𖽔𖾑 𖼚𖽘𖾐.” 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏.
𖼎𖽺𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼅𖽻𖾏
4 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽔𖾑 、 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼊𖽔𖾐. 6 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽔𖾑 、 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐. 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼷𖽷𖾐 𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼟𖽡 . 7 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼟𖽡, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼇𖽻𖾏 𖼐𖽑𖽡𖽝𖾑 𖼏𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏. 8 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼼𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽑𖽪𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽡𖽝𖾑𖽑 𖼼𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼐𖽡𖽝𖾑𖽑. 9 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼇𖽻𖾏 𖼐𖽡𖽝𖾑𖽑 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼮𖽡𖽾𖾑 𖽐𖼞𖽘𖾑, 𖼼𖽔𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖼈𖽡𖽝𖾏, 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏.” 10 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖼽𖽔𖾑 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐. 11 𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽻 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼦𖽡𖾐 𖼑𖽻 , 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽡𖽾 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼮𖽑𖽳𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏, 𖼐𖽑𖽡𖾏: “𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏!” 13 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽙𖾑 𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽑𖽱𖾏, 𖽐𖼎𖽙 𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼀𖽡𖾑 𖼘𖽙𖾐 𖼦𖽡 𖽐𖼚𖽙 𖼳𖽘𖾐 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏. 14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽡𖽾 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼮𖽑𖽳𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐, 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼙𖽠𖾏, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼖𖽡𖽾 𖽐𖼚𖽙 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽦𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 (“𖼷𖽑𖽹𖾏 𖼋𖽡𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼛𖽘𖾏 𖼼𖽙𖾏”) 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐. 15 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼸𖽙 𖼊𖽑𖽡 𖼖𖽳 · 𖼽𖽔𖾏 𖼀𖽡𖾐 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼀𖽔𖾑 𖼖𖽪𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐, 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐.
𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽡𖾑 𖼳𖽡𖾑 𖼮𖽡𖽝 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑
（𖼮𖽡𖽝 33：1 -9）
16 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 17 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼗𖽪𖾏, 𖽐𖼀𖽘𖾏 𖼘𖽑𖽪𖾑 𖼞𖽪𖾑 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏. 18 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔, 𖼟𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏, 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽙 , 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼾𖽔𖾏( “𖼾𖽔𖾏 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“ 𖽐𖼮𖽑𖽘𖾑 ” ) 𖼟𖽪𖾏 𖼗𖽔𖾏 𖼳𖽦𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡 . 19 𖼟𖽡 𖼮𖽔𖾑 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼐𖽱𖾏 𖼮𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽙, 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼎𖽺𖾏 𖼮𖽨𖾏 𖼾𖽨𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼾𖽔𖾏. 20 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽠𖾏 𖼋𖽠𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽙, 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼾𖽔𖾏 ( “𖼾𖽔𖾏 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“ 𖽐𖼮𖽑𖽘𖾑 ” )𖼟𖽪𖾏 𖼗𖽔𖾏 𖼳𖽦𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡 . 21 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏, 𖼐𖽱𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼑𖽻 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖼮𖽨𖾏 𖼾𖽨𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼾𖽔𖾏.
𖼽𖽔𖾑 𖼸𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏
