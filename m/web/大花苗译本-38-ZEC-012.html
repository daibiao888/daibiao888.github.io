𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐，𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝．
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼊𖽑𖽔𖾑 𖼀𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼑𖽡𖾏 𖼖𖽡𖽾 𖽃𖽡𖾑 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: 2 “𖼑𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽻𖾏 𖼢𖽻 𖼙𖽠𖾏, 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽨 𖼊𖽔𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏. 3 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽑𖽤𖾑, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼽𖽔𖾑 𖼮𖽑𖽳𖾑 𖼮𖽡𖽙𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑, 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 . 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽜𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏, 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼩𖽻𖾏 𖼺𖽘𖾏. 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼾𖽡𖾏, 𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽜𖾏 𖼊𖽡 𖼛𖽜 𖼺𖽘𖾏. 5 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏.
6 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑, 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼋𖽺𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼷𖽻𖾏 𖼡𖽻𖾐 𖽐𖼚𖽙 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 .
