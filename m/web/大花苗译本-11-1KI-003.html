𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽃𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼈𖽻𖾑, 𖼷𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 𖽐𖼷𖽑𖽹𖾏, 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻, 𖼋𖽙𖾏 𖼸𖽙 𖼷𖽑𖽙𖾏 𖼷𖽱𖾏 𖼊𖽡𖽝𖾑 𖼾𖽨𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼟𖽻𖾏 𖼯𖽪𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻. 2 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 3 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽐𖼮𖽑𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼺𖽷𖾐.
𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽐𖼊𖽜 𖼀𖽻𖾐 𖼟𖽳
4 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖽝𖾏 𖼅𖽻𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽃𖽔 𖼏𖽘 𖽃𖽡𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏 𖼯𖽦𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑. 5 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼀𖽡𖽝𖾏, 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼢𖽳 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼊𖽜 𖼋𖽪 𖼳𖽷𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡.” 6 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝、 𖼖𖽠𖾑 𖼞𖽡𖾑、 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼟𖽡 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐. 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼊𖽔𖾐, 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼁𖽺𖾏 𖼗𖽙𖾏． 8 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽹𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼾𖽱 𖼋𖽘 𖼦𖽡𖾐 𖼋𖽻𖾏． 9 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼮𖽷𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼨𖽑𖽢𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼸𖽠𖾏 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼻𖽻𖾏 𖼮𖽷𖾑, 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼨𖽑𖽢𖾐 𖽐𖼎𖽑𖽹𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑?”
10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽐𖼊𖽜 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑, 𖼊𖽻𖾏 𖼮𖽵𖾑 𖽃𖽺𖾐 𖼙𖽔. 11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽜 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑, 𖼦𖽡𖾐 𖼑𖽜 𖼾𖽨𖾏 𖽐𖼊𖽜 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏, 𖼦𖽡𖾐 𖽐𖼊𖽜 𖼅𖽔𖾏 𖼊𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼊𖽜 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖼮𖽔𖾑 𖽐𖼊𖽜 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽙𖾏 𖼸𖽠𖾏 𖼑𖽻 𖼻𖽻𖾏 𖼮𖽷𖾑. 12 𖼞𖽪𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽐𖼊𖽜 𖽃𖽡𖾑, 𖼺𖽘𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼻𖽻𖾏 𖼮𖽷𖾑. 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼟𖽡, 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼟𖽡. 13 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼊𖽜 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡, 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏、 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽡. 14 𖼟𖽡 𖼮𖽔𖾑 𖼾𖽻 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑, 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽑𖽘𖾑 𖼢𖽔𖾏 𖽐𖼎𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏.”
