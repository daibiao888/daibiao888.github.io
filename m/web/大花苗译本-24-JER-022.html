“𖼞𖽪𖾑 𖼑𖽔𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 ,

11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽔𖾑 𖼖𖽪𖽾 (𖼳𖽔𖾑 𖼖𖽪𖽾𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 23: 30 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽙 𖼦𖽔𖾐 𖼺𖽷𖾑), 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑. 12 𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖼋𖽔 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 .
