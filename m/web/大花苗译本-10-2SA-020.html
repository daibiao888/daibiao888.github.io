𖼳𖽷𖾏 𖼀𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽜𖾑
20
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽔𖾑 𖼯𖽱𖾏 𖼸𖽜, 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑, 𖼟𖽪𖾏 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐, 𖼐𖽱𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐 𖼐𖽑𖽡𖾏:
“𖼀𖽡𖾐 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼀𖽠,
𖽐𖼎𖽙 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏．
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！ 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡𖾑 ！”
2 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽔𖾏 𖼇𖽳𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼺𖽘𖾏, 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑 , 𖼈𖽡𖽝𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽔𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽡𖾏 𖼏𖽔𖾏 𖽐𖼎𖽜𖾏. 3 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼟𖽻𖾏 𖼗𖽜𖾏 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼟𖽙𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽃𖽡𖾑 𖼡𖽙𖾏 𖼎𖽪𖾑 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼏𖽙 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼮𖽡𖽝𖾏， 𖼊𖽪𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽝𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽱𖾏． 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼡𖽙𖾏 𖼎𖽪𖾑 𖼮𖽡𖽝𖾏, 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼺𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏, 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏.
𖼽𖽙 𖼽𖽔 𖼊𖽔𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐
4 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖽐𖼚𖽙 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑.” 5 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽳𖾑𖽑 𖼮𖽡𖽝𖾏 𖼟𖽙𖾏 𖼮𖽑𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑. 6 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼳𖽺𖾑 𖼐𖽑𖽡𖾏: “𖼐𖽔𖾑 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽷𖾏 𖼀𖽔𖾑 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼦𖽹𖾏 𖼀𖽡𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾. 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼅𖽻𖾏 𖼖𖽠𖾑 𖼐𖽱𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼀𖽡𖾐 𖼙𖽠𖾏.” 7 𖼽𖽙 𖼽𖽔 𖼑𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼊𖽑𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼊𖽑𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼟𖽪𖾏 𖼋𖽻𖾏, 𖼖𖽔𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼳𖽺𖾑, 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼖𖽠𖾑 𖼀𖽡𖾐 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑. 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖽝𖾏 𖼮𖽡𖽙𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 𖼗𖽙𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽙 𖼽𖽔 𖼐𖽑𖽘𖾑 𖼮𖽡𖽝𖾏 𖼏𖽙 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏, 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 𖼟𖽪𖾏 𖼎𖽑𖽹𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏, 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼐𖽘𖾐, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼽𖽔 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖽃𖽡𖾑, 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼎𖽑𖽻𖾏 𖼐𖽘𖾐 𖼘𖽑𖽻𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏. 9 𖼽𖽙 𖼽𖽔 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼇𖽳𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼗𖽙𖾏, 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 𖼐𖽑𖽡𖾏: “ 𖼞𖽪𖾑 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼘𖽪！ 𖼟𖽡 𖼻𖽻𖾏！” 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼊𖽔𖾑 𖼮𖽡𖽝𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 𖼚𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼡𖽑𖽻𖾐. 10 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 𖼦𖽡𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼽𖽔 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏, 𖼽𖽙 𖼽𖽔 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼮𖽑𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼐𖽑𖽨𖾑 𖼘𖽑𖽻𖾑 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐, 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏. 𖼽𖽙 𖼽𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼳𖽺𖾑, 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼖𖽠𖾑 𖼀𖽡𖾐 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑.
