35
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖽃𖽡𖾑, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏！
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼁𖽔𖾏 𖼏𖽙, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽔𖾏 𖼏𖽙！
2 𖼯𖽡𖽙𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼮𖽡𖽝𖾏,
𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑.
3 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼄𖽑𖽪𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼄𖽑𖽪𖾑 𖼟𖽪𖾏 𖽐𖼊𖽡𖾑 𖼗𖽙𖾏 𖽐𖼎𖽳𖾏 𖼮𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽪𖾑 𖽃𖽡𖾑,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼟𖽡 𖽃𖽡𖾑.”
 
4 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽡𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼙𖽠𖾏！
5 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼦𖽡𖾑 𖼁𖽘𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏,
𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏.
6 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑, 𖽃𖽺𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽡𖽝,
𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏.
 
7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼮𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖼊𖽑𖽔𖾑 𖼸𖽙𖾏,
𖼖𖽡𖾏 𖼅𖽪𖾏 𖽐𖼮𖽡𖽠𖾏 𖼡𖽑𖽙𖾑, 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑.
8 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼷𖽑𖽪𖾐 𖼖𖽻𖾐 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼚𖽺𖾏 𖼊𖽑𖽔𖾑 𖽃𖽡𖾑 𖽐𖼊𖽠𖾏 𖼾𖽨𖾏 𖼮𖽡𖽝𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽑𖽪𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏.
 
9 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼻𖽡𖽝𖾏 𖼐𖽱𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖽃𖽺𖾐 𖼙𖽔.
10 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼺𖽱𖾏 𖼟𖽡,
𖼮𖽨𖾏 𖼞𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏 𖽃𖽡𖾑,
𖼮𖽨𖾏 𖼞𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼐𖽱𖾏.”
 
11 𖼊𖽪𖾐 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏,
𖼳𖽹𖾐 𖼑𖽪 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐.
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽪𖾏 𖼁𖽻𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼷𖽔𖾏.
13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐, 𖼊𖽪𖾏 𖼡𖽜𖾑, 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐.
14 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽺 𖼗𖽔, 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐.
 
15 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼢𖽻 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡 𖼸𖽱．
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼦𖽡 𖼸𖽱 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼦𖽡𖾐 𖼳𖽙𖾏 𖽐𖼮𖽑𖽷𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑.
 
16 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼗𖽱𖾏 𖽃𖽔𖾏 𖼗𖽱𖾏 𖼚𖽙𖾏,
𖼋𖽙𖾏 𖼐𖽑𖽡𖽝𖾑 𖼷𖽷𖾏 𖼞𖽹𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑.
17 𖼮𖽵𖾑 𖼘𖽪！ 𖼟𖽡 𖼑𖽔𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼖𖽔𖾑 𖼋𖽙𖾏 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏！
𖼮𖽨𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 ( “𖼯𖽡𖽝𖾏 𖼳𖽦𖾑” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐”) 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏！
18 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼽𖽔𖾑 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡,
𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡.
 
19 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽡𖾑 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑 𖽃𖽡𖾑 𖽐𖼊𖽑𖽜𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼷𖽷𖾏 𖼖𖽪𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑！
20 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪,
𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼟𖽡𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏.
21 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑, 𖼐𖽑𖽡𖾏:
“𖽃𖽔𖾏 𖼦𖽔𖾏! 𖽃𖽔𖾏 𖼦𖽔𖾏! 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏.”
 
22 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏．
𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐．
23 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼯𖽷𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼨𖽑𖽢𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏,
𖼨𖽑𖽢𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏！
24 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼟𖽡 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖼞𖽪𖾑,
𖼦𖽡𖾐 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑！
25 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾏 𖼦𖽔𖾏, 𖼸𖽙𖾏 𖼀𖽡𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏！”
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖽐𖼡𖽙𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏！”
 
26 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑 𖽐𖼀𖽔𖾏 𖼐𖽑𖽘𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼊𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑！
27 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 (“𖼞𖽪𖾑 𖼑𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼞𖽪𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑”) 𖽃𖽡𖾑 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐！
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪.”
28 𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼽𖽔𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽑𖽡𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑,
𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡.
