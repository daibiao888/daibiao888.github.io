𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼘𖽪! 𖼡𖽑𖽜𖾐 𖼟𖽡 𖽃𖽔 𖼛𖽻𖾏,
𖼯𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼡𖽑𖽙𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏.
2 𖼊𖽑𖽪𖾑 𖼘𖽪! 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖼢𖽻 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼀𖽔𖾑 𖼳𖽔𖾑 𖼑𖽡𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼘𖽪! 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽑𖽻𖾑 𖼻𖽻𖾑 𖼢𖽻 𖼙𖽠𖾏 𖼋𖽘𖾏.
3 𖼑𖽻 𖼘𖽪! 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽘𖾏 𖽐𖼡𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏. 𖼅𖽔𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖼙𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽡𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏.
𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: “𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼾𖽨 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼐𖽡𖾑 . 5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼞𖽑𖽡𖽠𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏. 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽺 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:‘ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏.’ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼾𖽨 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏. 6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 . 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽷𖾏 𖼼𖽙 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽨𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼀𖽜𖾐 𖼛𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.”
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼾𖽨 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼐𖽡𖾑 , 𖼟𖽪𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾏 𖽐𖼊𖽻𖾏 𖽃𖽡𖾑. 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾐 𖼋𖽪 𖼮𖽡𖾑 𖼀𖽜𖾏, 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏”, 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “𖼦𖽡𖾏 𖼚𖽑𖽙𖾏”, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼾𖽨 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼋𖽘. 8 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏 𖽐𖼚𖽙, 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑. 9 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼦𖽡𖾐 𖼊𖽪𖾑 𖼾𖽨 𖼅𖽡 . 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔 , 𖼡𖽑𖽙𖾑 𖼋𖽻𖾏 𖼋𖽔 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼳𖽦𖾑 , 𖼡𖽑𖽙𖾑 𖼋𖽻𖾏 𖼛𖽙𖾏 𖼳𖽦𖾑 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 , 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽡𖽝𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼷𖽑𖽡𖽝𖾏.”
𖼖𖽙𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼮𖽡𖾑 𖼀𖽜𖾏 𖼺𖽘𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏
10 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏” 𖽃𖽡𖾑 𖼖𖽙𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏, 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑. 11 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 . 12 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼮𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽳𖾑 𖼋𖽘𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼙𖽡 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖽃𖽔𖾏 𖼞𖽪𖾑 𖽐𖼡𖽜𖾏 𖼼𖽙 . 13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼀𖽔𖾐 𖽃𖽔 𖼗𖽔𖾏 .” 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼙𖽡 𖼑𖽡𖽝𖾏 𖼐𖽡𖾑 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏, 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼀𖽔𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏. 14 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “𖼦𖽡𖾏 𖼚𖽑𖽙𖾏” 𖽃𖽡𖾑, 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑.
