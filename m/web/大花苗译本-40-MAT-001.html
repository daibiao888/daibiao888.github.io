𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖽃𖽡𖾐 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏
（𖼖𖽪𖾏 3：23-38）
1
𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏，𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖽃𖽡𖾐 𖽐𖼞𖽔𖾐 𖼾𖽡𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑：
2 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼅𖽔𖾏𖼽𖽡𖾑 𖼺𖽔𖾐 ， 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼅𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙， 𖼽𖽔𖾐 𖼞𖽙 𖼅𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼽𖽨 𖼊𖽔𖾏𖼞𖽱𖾑 𖼋𖽡𖾏． 3 𖼽𖽨 𖼊𖽔𖾏𖽐𖼎𖽙𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖼅𖽔𖾏𖼇𖽔 𖼖𖽳 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝𖾐 𖼳𖽡𖽝𖾏 𖼖𖽔𖾑， 𖼇𖽔 𖼖𖽳 𖼺𖽷𖾑 𖼅𖽔𖾏𖼳𖽡𖾑 𖼺𖽷𖾑 𖼖𖽪𖽝， 𖼳𖽡𖾑 𖼺𖽷𖾑 𖼖𖽪𖽝 𖼅𖽔𖾏𖼽𖽔𖾏 𖼖𖽔， 4 𖼽𖽔𖾏 𖼖𖽔 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼐𖽔 𖼊𖽔，𖼽𖽔𖾏 𖼄𖽡𖾐 𖼐𖽔 𖼊𖽔𖼅𖽔𖾏 𖼐𖽔 𖼳𖽪𖽝𖾏，𖼐𖽔 𖼳𖽪𖽝𖾏𖼅𖽔𖾏 𖼺𖽔𖾐 𖼄𖽞， 5 𖼺𖽔𖾐 𖼄𖽞 𖽐𖼎𖽙 𖼖𖽔 𖼦𖽙 𖼅𖽔𖾏𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑， 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖽐𖼎𖽙 𖼖𖽪𖾏 𖼊𖽳 𖼅𖽔𖾏𖽃𖽙 𖼀𖽡𖾐 𖼊𖽳，𖽃𖽙 𖼀𖽡𖾐 𖼊𖽳 𖼅𖽔𖾏𖼽𖽦𖾑 𖼳𖽡𖾑， 6 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼅𖽔𖾏𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙𖽂𖽪𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼅𖽔𖾏𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞． 7 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼅𖽔𖾏𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑，𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼅𖽔𖾏𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽔𖾐， 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽔𖾐 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐， 8 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼅𖽔𖾏𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔，𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼅𖽔𖾏𖼽𖽙 𖼖𖽗，𖼽𖽙 𖼖𖽗 𖼅𖽔𖾏𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐， 9 𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 𖼅𖽔𖾏𖼽𖽙 𖼊𖽑𖽗𖾐，𖼽𖽙 𖼊𖽑𖽗𖾐 𖼅𖽔𖾏𖼽𖽔𖾏 𖼦𖽗𖾐 𖼺𖽷𖾑，𖼽𖽔𖾏 𖼦𖽗𖾐 𖼺𖽷𖾑 𖼅𖽔𖾏𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑． 10 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼅𖽔𖾏𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 ， 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼅𖽔𖾏𖼽𖽔𖾏 𖼄𖽞， 𖼽𖽔𖾏 𖼄𖽞 𖼅𖽔𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏． 11 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏 𖽃𖽡𖾑， 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼅𖽔𖾏𖼽𖽦𖾑 𖼞𖽙𖾑 𖼐𖽡 𖼽𖽔𖾐 𖼊𖽑𖽡𖽝𖾐 𖼽𖽦𖾑 𖼞𖽙𖾑 𖼐𖽡 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡．
12 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼽𖽦𖾑 𖼞𖽙𖾑 𖼐𖽡 𖼽𖽔𖾐 𖼅𖽔𖾏𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽑𖽡𖽝，𖼺𖽔𖾐 𖼖𖽔𖾑 𖼊𖽑𖽡𖽝 𖼅𖽔𖾏𖼺𖽙𖾐 𖼘𖽙 𖼀𖽔𖾑 𖼀𖽳， 13 𖼺𖽙𖾐 𖼘𖽙 𖼀𖽔𖾑 𖼀𖽳 𖼅𖽔𖾏𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽱𖾏， 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽱𖾏 𖼅𖽔𖾏𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼮𖽡𖾏 ，𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼮𖽡𖾏 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼺𖽙𖾐, 14 𖼽𖽔𖾏 𖼺𖽙𖾐 𖼅𖽔𖾏𖼺𖽔𖾐 𖼊𖽪，𖼺𖽔𖾐 𖼊𖽪 𖼅𖽔𖾏𖼽𖽔𖾏 𖼮𖽡𖾑，𖼽𖽔𖾏 𖼮𖽡𖾑 𖼅𖽔𖾏𖼽𖽡𖾑 𖼖𖽡， 15 𖼽𖽡𖾐 𖼖𖽡𖼅𖽔𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐， 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼅𖽔𖾏𖼄𖽔𖾐 𖼊𖽗𖾏， 𖼄𖽔𖾐 𖼊𖽗𖾏 𖼅𖽔𖾏𖼽𖽔𖾐 𖼞𖽙． 16 𖼽𖽔𖾐 𖼞𖽙 𖼅𖽔𖾏𖼽𖽙 𖼺𖽳， 𖼽𖽙 𖼺𖽳 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼈𖽻𖾑， 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼅𖽔𖾏 𖼯𖽡𖽙．
17 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼸𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽷𖾏， 𖼎𖽑𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽷𖾏，𖼎𖽑𖽻𖾏 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏 𖼸𖽙 𖼮𖽡𖾑 𖼊𖽪 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽷𖾏．
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼖𖽪𖾏 2：1-7）
18 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ： 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼏𖽵 𖼎𖽻𖾏 𖼽𖽙 𖼺𖽳， 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼷𖽙𖾏 𖼺𖽷𖾐， 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼅𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑． 19 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼽𖽙 𖼺𖽳 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ， 𖼦𖽡𖾐 𖼯𖽪𖾏 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽻𖾐，𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖽐𖼎𖽻𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏． 20 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼊𖽱𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼅𖽔𖾏 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽡𖽙𖾐 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼽𖽙 𖼺𖽳， 𖼟𖽡 𖼦𖽡𖾐 𖼅𖽪𖾏 𖽐𖼮𖽑𖽹𖾏， 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽙𖾏 𖼟𖽡 𖼐𖽤𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼗𖽙𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼅𖽜， 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏． 21 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐，𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽨𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏．” 22 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼋𖽘𖾏，𖼟𖽪𖾏 𖼽𖽔𖾑 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏：
