29
“𖼖𖽪𖾑 𖼳𖽤𖾏 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 , 𖼅𖽡 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 , 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐 . 2 𖼅𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 , 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏 , 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 3 𖼡𖽻𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼛𖽙𖾏 , 𖼑𖽜 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖼷𖽷𖾐 𖽐𖼀𖽠, 𖼑𖽜 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽔𖾐 𖽐𖼀𖽠； 4 𖼑𖽜 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 ,𖼷𖽑𖽔𖾏 𖼋𖽪𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠. 5 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 , 𖼑𖽜 𖼅𖽡 𖼚𖽠𖾏 𖼾𖽔𖾏 . 6 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖼑𖽡𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 , 𖼊𖽑𖽡𖽝𖾐 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 , 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑
（𖼖𖽡𖾏 23：26 - 32）
7 “𖼖𖽪𖾑 𖼳𖽤𖾏 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼟𖽻𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑， 𖼽𖽔𖾑 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 , 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 . 8 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 , 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 , 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏 , 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 9 𖼡𖽻𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼖𖽑𖽘𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼛𖽙𖾏 , 𖼑𖽜 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖼷𖽷𖾐 𖽐𖼀𖽠, 𖼑𖽜 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽔𖾐 𖽐𖼀𖽠； 10 𖼑𖽜 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖽃𖽡𖾑 𖼷𖽑𖽔𖾏 𖼋𖽪 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼽𖽡𖾑 𖼇𖽔 𖼟𖽻𖾏 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 . 11 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏． 𖼊𖽪𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 , 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 , 𖽐𖼎𖽙 𖼡𖽻𖾐 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼖𖽑𖽘𖾑 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 𖼉𖽺𖾏 𖼢𖽻𖾏 .
