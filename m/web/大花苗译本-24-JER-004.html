𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪！
𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼺𖽷𖾐 𖼗𖽙, 𖼯𖽡𖾏 𖼺𖽷𖾐 𖼗𖽙 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑 .
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖼮𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐,
𖼟𖽡 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽻 𖼋𖽘𖾏,
2 𖼟𖽡 𖼮𖽔𖾑 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝、𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵,
𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼽𖽔𖾑 𖼑𖽜𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏.
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：
𖼽𖽔𖾑 𖼞𖽑𖽳𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼀𖽳𖾐 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
4 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼋𖽠𖾏 𖼺𖽘𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼞𖽪𖾑 𖼑𖽜 𖼅𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼅𖽡 ,
𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼯𖽡 ,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼇𖽳𖾏！
 
5 𖼅𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 ,
𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼐𖽑𖽡𖾏:
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐, 𖼮𖽘𖾏 𖼺𖽘𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱！
𖼀𖽡𖾐 𖼻𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼏𖽔𖾏！
6 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼷𖽑𖽙𖾏 𖼀𖽑𖽳𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽳𖾑𖽑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽐𖼎𖽙 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻.
7 𖼅𖽔𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼎𖽑𖽻𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏,
𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙．
𖼐𖽱𖾏 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼇𖽳𖾑 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼋𖽘 ,
𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼎𖽘𖾏 𖼯𖽢𖾏,
𖼟𖽙𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼙𖽻 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐.
8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐, 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼇𖽳𖾑 𖼀𖽡𖾐.
 
9 “𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼽𖽔𖾑 𖼀𖽑𖽪𖾑 𖼺𖽡𖽝𖾐 𖼛𖽙𖾏 𖼚𖽔𖾐, 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏,𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽔𖾑 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏.” 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 10 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖽃𖽺𖾐! 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐, 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ！ 𖼟𖽡 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼐𖽑𖽡𖾏: 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪． 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼦𖽡𖾑 𖼚𖽘𖾐 𖼋𖽘 .
11 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼊𖽑𖽡𖽝𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼐𖽑𖽡𖾏: 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼁𖽘 𖼮𖽢𖾏 𖼳𖽙𖾑 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼮𖽢𖾏, 𖼷𖽑𖽔𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 (“𖼖𖽡𖾑 𖼉𖽪𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏”), 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽜 𖼷𖽻𖾑 𖼨𖽑𖽢𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽜 𖼷𖽻𖾑 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑． 12 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼁𖽘 𖼮𖽢𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼑𖽜 𖼞𖽪𖾑 𖼷𖽑𖽔𖾐 𖼋𖽔𖾏, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏.
 
13 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼺𖽱𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏,
𖼐𖽱𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼺𖽱𖾏 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 ,
𖼐𖽱𖾏 𖼑𖽜𖾏 𖼇𖽺𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼚𖽘𖾑, 𖼀𖽡𖾐 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！
𖼀𖽡𖾐 𖼦𖽡 𖼋𖽙 𖼙𖽠𖾏 𖼋𖽘𖾏！
14 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！ 𖽐𖼷𖽔𖾑 𖼺𖽘𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼟𖽙𖾏 𖼟𖽡 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏．
𖽐𖼚𖽪𖾏 𖼸𖽳 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐, 𖼖𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
