3
“𖼑𖽔𖾏 𖼘𖽪 ！ 𖼸𖽙 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 2 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼖𖽪𖾑 𖼦𖽘𖾑 (“𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 ”𖼟𖽪𖾏 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏” ), 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 , 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼎𖽺𖾏 𖼇𖽺𖾐 𖼞𖽪𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 3 𖼎𖽺𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 , 𖼄𖽔𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼊𖽪𖾐 𖼘𖽪𖾑𖽑 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏, 𖼋𖽺 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 𖼅𖽔𖾏 𖼮𖽡𖽠𖾑 𖼦𖽻𖾏.
4 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 、 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖽐𖼎𖽙 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪! 𖼅𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏? 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼅𖽡 𖼖𖽡𖾏 𖼇𖽻𖾏. 5 𖼅𖽡 𖼦𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼯𖽡𖽙𖾏 𖼚𖽔𖾏 𖼅𖽡 𖽐𖼞𖽔𖾏 𖼁𖽠 𖽐𖼚𖽙, 6 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽡𖽝𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽺 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 (“ 𖼳𖽡𖾑 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 ” 𖼟𖽪𖾏 “ 𖼽𖽔𖾐 𖽂𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐”), 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼚𖽡𖾐. 7 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼡𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽺 𖼅𖽻𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼅𖽡 𖼖𖽡𖾏 𖼇𖽻𖾏. 8 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼅𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽺 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑 𖼞𖽪𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
 
9 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
𖼅𖽡 𖼽𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏, 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼊𖽪𖾐 𖼋𖽻 ,
𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 .
10 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽺𖾏 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏,
𖼄𖽔𖾐 𖽃𖽔𖾑 𖼖𖽡𖽝𖾐 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖼄𖽑𖽪𖾑.
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼋𖽻 .
11 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼘𖽪 !
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼋𖽔𖾏, 𖽃𖽡𖾐 𖼸𖽻 𖼦𖽡 𖼸𖽱 .
 
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 !
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼟𖽡 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼋𖽻 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 .
 
12 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏,
𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼖𖽪𖾑 𖼦𖽘𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑,
𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐.
13 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼖𖽡𖽝𖾐 𖼯𖽡𖽙𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼳𖽦𖾑 𖼋𖽘𖾏！
𖼋𖽔𖾏 𖼯𖽪𖾏 ! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼮𖽡𖽠𖾑 𖼀𖽪𖾑 𖼋𖽘𖾏．
𖼖𖽪𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼸𖽹𖾏 𖼮𖽡𖽠𖾑 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏 𖼋𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
 
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾐 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐.
15 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼷𖽻𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏 ,
𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼮𖽡𖾏 .
 
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏,
𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐,
𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 .
𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 .
 
17 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏.
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏.”
