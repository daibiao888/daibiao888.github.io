𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 20：12 - 19）
39
𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼄𖽡𖾐 𖼘𖽙 𖼊𖽔 · 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼊𖽗𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼻𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼺𖽘𖾏 𖽐𖼊𖽠𖾑 𖼄𖽻𖾏 𖽐𖼎𖽙 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏. 2 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽃𖽺𖾐 𖼙𖽔 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼼𖽪 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖼞𖽪𖾐、 𖼑𖽡𖽝𖾐、 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏、 𖼛𖽙𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼎𖽙 𖼏𖽙 𖽐𖼞𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼞𖽑𖽔𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 , 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜, 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 .
3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏? 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡?” 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼟𖽪𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑.” 4 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖽐𖼚𖽙 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏?” 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏, 𖼞𖽪𖾑 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏.”
𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏
5 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 . 6 𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏, 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼡𖽑𖽙𖾑 𖼮𖽷𖾏 𖼮𖽡𖽝𖾏 𖼗𖽙𖾏 𖼸𖽙 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏, 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 7 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾑, 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏, 𖼐𖽡𖽙𖾐 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽃𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏．”
8 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼗𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏:“𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖽐𖼎𖽙 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.”
