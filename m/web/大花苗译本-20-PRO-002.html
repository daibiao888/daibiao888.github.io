5 𖼟𖽡 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼎𖽺𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,

7 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐；
8 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 .
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳、𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑、
𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 .
10 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼀𖽻𖾐 𖼟𖽳 𖼟𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
11 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 ,
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼟𖽡 ；
12 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼟𖽡 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 (“𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑”),
𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼦𖽡𖾐 𖼙𖽠𖾏 𖼼𖽘𖾏.
13 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏
𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏.
14 𖽃𖽺𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 ,

15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾐 𖽐𖼞𖽑𖽡𖾐 𖼖𖽡𖾏 𖽐𖼞𖽑𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏.
16 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼟𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏,
𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼡𖽘𖾐 𖼄𖽪𖾐 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑 𖼟𖽡𖾏 𖽐𖼡𖽳𖾏 𖽃𖽳𖾑．
17 𖼐𖽱𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼈𖽻𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼡𖽔𖾏 𖼎𖽻𖾏 𖽃𖽡𖾑,
