𖼽𖽔𖾐 𖼞𖽙 𖼘𖽪 ! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼟𖽡 ,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪！ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼊𖽡𖽝𖾑 𖼟𖽡 𖽃𖽡𖾑,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:
“𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼟𖽡 𖼋𖽘𖾏,
𖼞𖽪𖾑 𖼊𖽔𖾑 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡 𖼋𖽘𖾏, 𖼟𖽡 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝.
2 𖼟𖽡 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽱𖾏 𖼮𖽑𖽡𖽻𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐,
𖼟𖽡 𖼋𖽡𖾏 𖼮𖽑𖽡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏,𖽃𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼟𖽡 ,
𖼟𖽡 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏,𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼷𖽑𖽷𖾑,
𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽡 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 .
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼮𖽵𖾑 ,
𖼞𖽪𖾑 𖼟𖽙𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽔𖾏 𖼟𖽡 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼎𖽙𖾑,
𖼟𖽙𖾏 𖼞𖽪𖾐 𖼳𖽷 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼀𖽔𖾑 𖼘𖽪𖾑𖽑 𖼟𖽡 .
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏,
𖼎𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 ,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪𖾑𖽑 𖼟𖽡,
𖼟𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼘𖽪𖾑𖽑 𖼟𖽡 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑.
5 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼋𖽔𖾏,
𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽡 .
6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼐𖽑𖽡𖾏: 𖼄𖽔𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏！
𖼎𖽺𖾏 𖽐𖼎𖽙 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼐𖽑𖽡𖾏: 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽱𖾏 𖼮𖽡𖽝𖾏！
𖼄𖽔𖾐 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼨𖽑𖽳𖾏 𖼋𖽔𖾏,
𖼄𖽔𖾐 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏 𖼨𖽑𖽳𖾏 𖼺𖽷𖾐 𖼗𖽙，
7 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼾𖽨𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑, 𖼡𖽑𖽙𖾑 𖼷𖽷𖾏 𖼊𖽡𖽝𖾑 𖽃𖽡𖾑.”
 
8 𖼟𖽡 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼅𖽔𖾏 𖼈𖽡𖽝𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐,
𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼈𖽡𖽝𖾏 𖽃𖽔𖾐 𖼗𖽘𖾐 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏！
9 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽡𖾐 𖼸𖽻𖾏 𖼦𖽡 𖼸𖽱 ,
𖼯𖽔𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏．
𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑,
𖼎𖽺𖾏 𖼄𖽔𖾐 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼑𖽻 𖼖𖽺𖾑?
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼋𖽔𖾏, 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼾𖽙,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏.”
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏,
𖼟𖽪𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼺𖽹𖾑．
𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼸𖽙𖾏 𖼞𖽪𖾑,
𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 “𖼷𖽷𖾑 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏”,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾏,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏.
11 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼋𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽨𖾏 𖼮𖽵𖾑 .
12 𖼞𖽪𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏, 𖼞𖽪𖾑 𖼙𖽔 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏, 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺．
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
𖼅𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏,
𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
13 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏, 𖼞𖽪𖾑 𖼟𖽪𖾏 “𖼷𖽷𖾑 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏”,
𖼡𖽔 𖼋𖽱𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼖𖽺𖾑?”
𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼽𖽔𖾑 𖼼𖽠𖾏 𖼢𖽻
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼑𖽜 𖼅𖽡 𖼑𖽪𖾏,
𖼞𖽪𖾑 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖽐𖼡𖽜𖾏 𖼋𖽔𖾏.
15 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑,
𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽷𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼯𖽡𖽙𖾏, 𖼟𖽪𖾏 𖼅𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏.”
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼡𖽘𖾐 𖼳𖽡𖽠𖾏 𖼡𖽑𖽜𖾐 𖼞𖽡𖾑 ,
𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽑𖽜𖾐 𖼞𖽡𖾑 ,
17 𖼟𖽙𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐、 𖼑𖽜𖾏、 𖼊𖽪𖾐 𖼏𖽙𖾏, 𖼊𖽪𖾐 𖼋𖽻 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖽃𖽡𖾐 𖼸𖽻 𖼢𖽻 𖼙𖽠𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏, 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼐𖽑𖽨𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏.
𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽦𖾐 𖼮𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏
18 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:
“𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼑𖽪𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼑𖽪𖾏 .
19 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏, 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼡𖽑𖽜𖾐 𖼞𖽡𖾑,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽔𖾏 𖼡𖽑𖽜𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏.
20 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼞𖽪𖾑,
𖼚𖽡𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼐𖽻 𖼞𖽡𖾏 𖼎𖽻𖾑 𖼊𖽪𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼅𖽔𖾏 𖽃𖽻𖾐,
𖼟𖽙𖾏 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽔𖾏 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏,
𖼻𖽻𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑 𖽃𖽡𖾑 𖼦𖽻𖾏.
21 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖼯𖽡𖽙𖾏,
𖼟𖽙𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼑𖽪𖾏 .
 
22 𖼽𖽔𖾐 𖼞𖽙 𖼘𖽪 ! 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼞𖽪𖾑,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ! 𖼟𖽡 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼳𖽪𖾐 𖼞𖽪𖾑.
23 𖼟𖽡 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼄𖽔𖾐 𖼟𖽡 𖼾𖽘𖾏 𖼯𖽡𖽙𖾏 𖼋𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼞𖽪𖾑,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼑𖽜 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾐 𖼮𖽘𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼑𖽻𖾏 𖽃𖽦𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼑𖽜 𖼵𖽪𖾐 𖼳𖽤𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼳𖽪𖾐 𖼞𖽪𖾑.
24 𖼟𖽡 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼻𖽻𖾐 𖼸𖽷 𖼁𖽠𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼛𖽙𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼎𖽙𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼳𖽪𖾐 𖼟𖽡 .
𖼅𖽳𖾏 𖼟𖽪𖾏 𖼯𖽔𖾏 𖼾𖽔𖾏
25 “ 𖼞𖽪𖾑 , 𖼯𖽔 𖼅𖽔𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼐𖽔𖾑 𖼇𖽳𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼾𖽔𖾏,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.
26 𖼟𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼯𖽷𖾏 𖼞𖽪𖾑, 𖼟𖽡 𖼞𖽪𖾑 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼮𖽑𖽵𖾑,
𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼟𖽡 𖼞𖽡𖾑 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽹𖾏, 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼾𖽙 .
27 𖼟𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖼟𖽡 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼳𖽠𖾑 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑.
28 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 ,
𖼟𖽙𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼊𖽻𖾏 𖼖𖽻𖾐 ,
𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 .”
