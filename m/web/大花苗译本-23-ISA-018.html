18
𖽃𖽺𖾐! 𖼞𖽪𖾐 𖼳𖽷 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼀𖽘𖾑𖽑 𖽃𖽡𖾑, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼷𖽷 𖼟𖽻𖾏 𖽐𖼷𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏.
2 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽡𖾏 𖼳𖽡𖽠𖾑,
𖼐𖽡𖽙𖾐 𖼀𖽑𖽪 𖼷𖽑𖽻𖾐 𖽐𖼞𖽙𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖽃𖽻𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑． 𖼅𖽡 𖼅𖽻𖾏 !
𖼅𖽡 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼇𖽺𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐,
𖼊𖽠𖾏 𖼋𖽡𖾏 𖼅𖽙𖾏 𖼦𖽡𖾏 𖼑𖽡𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼚𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼈𖽳𖾏 𖼖𖽔𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼯𖽵𖾏 𖼗𖽔 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑,
𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼇𖽺𖾐 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.
 
3 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼘𖽪！
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼷𖽑𖽙𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐 𖽃𖽡𖾑, 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 .
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼑𖽔𖾏 ,
𖼺𖽱𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼮𖽡𖾏 𖼑𖽡𖾏 𖼀𖽘𖾏 𖼳𖽙𖾑,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼙𖽱 𖼑𖽡𖾏 𖼦𖽘𖾑 𖼀𖽙𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏.”
5 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼁𖽘𖾏 𖼋𖽠𖾏 𖼎𖽺𖾏 𖽐𖼀𖽑𖽙𖾐,
𖼁𖽘𖾏 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼳𖽦𖾑,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔𖾐 𖼖𖽡𖽝𖾐 𖼮𖽑𖽘𖾐 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾏,
𖼎𖽺𖾏 𖼮𖽑𖽘𖾐 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼚𖽻𖾑 𖽐𖼊𖽪𖾐 𖼖𖽠𖾏 .
6 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼚𖽘𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾐 𖼦𖽘𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏.
𖼐𖽻 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽱𖾏 𖼮𖽑𖽡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏,
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽱𖾏 𖼮𖽑𖽡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐.
 
7 𖼸𖽙 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐, 𖼊𖽠𖾏 𖼋𖽡𖾏 𖼅𖽙𖾏 𖼦𖽡𖾏 𖼑𖽡𖽻𖾏,
𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼐𖽡𖽙𖾐 𖼈𖽳𖾏 𖼊𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑,
𖼟𖽪𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼯𖽵𖾏 𖼗𖽔 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑,
𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼇𖽺𖾐 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐,
𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼎𖽻𖾏 .
