𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼎𖽺𖾏 𖼖𖽔𖾏 𖼽𖽙 𖼀𖽳 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼻𖽻𖾏
（𖼀𖽳 15：1 - 21：34）
15
𖼊𖽑𖽡 𖼄𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 :
2 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼯𖽡𖽙𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 𖼗𖽙𖾏 𖼊𖽡𖾐 𖼖𖽺𖾑 ?
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼮𖽢𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼘𖽪 ?
3 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ,
𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼗𖽙𖾏 𖼊𖽑𖽔𖾑 𖼮𖽔𖾑𖽑 𖼖𖽺𖾑 ?
4 𖼟𖽡 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 ,
𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 .
5 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼡𖽑𖽔𖾐 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 ,
𖼟𖽡 𖼺𖽹𖾑 𖼞𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼸𖽠𖾏 𖼮𖽑𖽵𖾑 𖼑𖽡𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏.
6 𖼟𖽡 𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼊𖽡𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 ,
𖼟𖽡 𖼾𖽨𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼟𖽡 .
7 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼊𖽑𖽪 𖼇𖽻𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼘𖽪 ?
𖼟𖽡 𖼊𖽻𖾏 𖼷𖽷𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼘𖽪𖾏 ?
8 𖼟𖽡 𖼙𖽔 𖼐𖽙𖾑𖽑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼘𖽪𖾏 ?
𖼟𖽡 𖼾𖽨𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼘𖽪𖾏 ?
9 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼦𖽔𖾏 𖼀𖽡𖾐 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼖𖽺𖾑 ?
𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐 𖼦𖽔𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐 𖼖𖽺𖾑 ?
10 𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼚𖽠𖾐 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 ,
𖼗𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼺𖽷𖾐 .
11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼟𖽡 ,
𖼟𖽡 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖼘𖽪𖾏 ?
12 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽃𖽔𖾏 𖼊𖽔𖾑 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼙𖽠𖾏 ?
𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 ,
13 𖼟𖽙𖾏 𖼟𖽡 𖼖𖽡𖽾 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐 ,
𖼎𖽺𖾏 𖼯𖽔𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 ?
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼋𖽡𖽝𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼖𖽺𖾑 ?
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼾𖽨 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼋𖽡𖽝 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼘𖽪 ?
15 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼻𖽡𖽝𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼨𖽑𖽪𖾑 ,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔, 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏，
16 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼐𖽑𖽡𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 ,
𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼋𖽘 !
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼀𖽜𖾑 𖼦𖽡𖾐 𖼛𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
17 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 , 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾑 ,
18 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼡𖽑𖽙𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 ,
𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼈𖽺𖾏 𖽃𖽡𖾑 .
19 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼮𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾐 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 .
20 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽠𖾏 𖼊𖽠𖾑 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼏𖽠𖾏 𖼚𖽑𖽔𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 .
21 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼞𖽙𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 ,
𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼎𖽺𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 .
22 𖼐𖽱𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 ,
𖽃𖽔𖾑 𖼏𖽦𖾏 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼐𖽱𖾏 .
23 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 , 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 𖽐𖼊𖽜 𖼡𖽜𖾑 𖼑𖽻𖾏 , 𖼐𖽑𖽡𖾏 : 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼅𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼖𖽺𖾑 ?
𖼐𖽱𖾏 𖼀𖽻𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼊𖽘𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 .
24 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 ,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 , 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐．
