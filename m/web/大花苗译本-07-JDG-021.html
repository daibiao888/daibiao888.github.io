8 𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼳𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼄𖽡𖾐 𖼺𖽷𖾑 𖼀𖽔𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽺𖾑?” 𖼑𖽔𖾏 𖼘𖽪 ! 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖼋𖽠𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 ‧ 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼚𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽻𖾏 𖼾𖽱 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐𖼮𖽡𖾑 𖼖𖽡𖽝 ‧𖼽𖽔𖾐 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑. 10 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼏𖽙 𖼟𖽪𖾏 𖼋𖽻𖾏, 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼅𖽻𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼄𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽡𖽝 ‧𖼽𖽔𖾐 𖼀𖽡𖾐 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼾𖽠、𖽃𖽔 𖼁𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼞𖽑𖽔𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏. 11 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼾𖽠、𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽃𖽔𖾑 𖼾𖽠 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.” 12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽡𖽝． 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼙𖽔 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼺𖽷𖾐, 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼳𖽷𖾏 𖼘𖽙 𖼖𖽪𖾑 𖼾𖽡 𖽐𖼚𖽙.
