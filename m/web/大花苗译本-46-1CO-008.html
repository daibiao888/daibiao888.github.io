4 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼑𖽻𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼙𖽔 𖼖𖽑𖽘𖾑 𖼁𖽠𖾏，𖼀𖽡𖾐 𖼀𖽻𖾐 𖼊𖽪𖾑 𖼁𖽠𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼦𖽡𖾐 𖼺𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏，𖼀𖽡𖾐 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼋𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 5 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼮𖽵𖾑 𖼳𖽺， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼮𖽵𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐， 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽔𖾐， 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏， 𖼯𖽢 𖼋𖽡𖾏 𖼟𖽻𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼁𖽡𖽝，𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽵𖾑 𖼊𖽔𖾐， 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪，𖼯𖽢 𖼋𖽡𖾏 𖼟𖽻𖾏 𖼮𖽘𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼯𖽡𖽙， 𖼀𖽡𖾐 𖼖𖽔𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼯𖽡𖽙𖾏．
7 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽪𖾏 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽡𖾑． 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽑𖽘𖾑 𖼁𖽠𖾏 𖼋𖽪𖾏 𖼺𖽘𖾏， 𖼸𖽙𖾏 𖼞𖽑𖽡𖽝𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼖𖽑𖽘𖾑 𖼁𖽠𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽡𖽝𖾐 𖼚𖽠𖾐 𖼟𖽪𖾏 𖼯𖽻 𖼛𖽘𖾏 𖼯𖽻 𖼼𖽙 ，𖼎𖽺𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏． 8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏，𖼦𖽡𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼮𖽑𖽔𖾑 𖼯𖽠 𖼡𖽔𖾑 𖼳𖽷𖾏，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏．
