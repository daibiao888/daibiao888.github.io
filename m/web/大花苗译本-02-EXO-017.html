𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼊𖽻𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽙𖾏 𖼋𖽠𖾏 𖽃𖽻𖾐
（𖼉𖽪𖾏 20：1 -13）
17
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽪𖾐 𖼾𖽡 , 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼎𖽑𖽻𖾏 𖼳𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 , 𖼐𖽡𖽙𖾐 𖼖𖽡𖾏 𖼇𖽺𖾑 𖼊𖽡𖽾𖾑 𖼮𖽪𖾐 𖼾𖽡． 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽻𖾐 𖼦𖽻𖾏 , 2 𖽃𖽔𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 , 𖼐𖽑𖽡𖾏 : “𖼄𖽔𖾐 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼦𖽻𖾏 !” 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 ? 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽺𖾑 ?” 3 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼽𖽔𖾑 𖽃𖽻𖾐 𖼦𖽻𖾏 , 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 , 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼨𖽑𖽳𖾏 𖼀𖽡𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼊𖽑𖽡𖽝𖾐 𖼯𖽦𖾏 𖼖𖽔𖾏 𖽐𖼡𖽑𖽜𖾏 𖽃𖽻𖾐 𖼋𖽔 𖼺𖽘𖾏 𖼖𖽺𖾑 ?” 4 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑 ? 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 𖼋𖽔 𖼺𖽘𖾏 .” 5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽐𖼊𖽻𖾏 𖼐𖽡 𖼘𖽙 𖼛𖽡𖾏 𖽃𖽻𖾐 𖽃𖽡𖾑 , 𖼨𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼗𖽻 , 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 . 6 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼦𖽙 𖼖𖽡𖽝 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐． 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 , 𖼎𖽑𖽻𖾏 𖽐𖼚𖽙 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖽃𖽻𖾐 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽻𖾏 𖼦𖽻𖾏 .” 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼗𖽻 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏 . 7 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖼺𖽔𖾐 ( “𖼄𖽔𖾐 𖼺𖽔𖾐 ”𖼟𖽪𖾏 “ 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 ” ) , 𖼎𖽺𖾏 𖽐𖼷𖽷𖾏 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼀𖽔𖾑 (“𖼄𖽡𖾐 𖼖𖽡 𖼀𖽔𖾑” 𖼟𖽪𖾏 “𖼦𖽡𖾏 𖽐𖼊𖽜𖾏”) , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖽐𖼊𖽜𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 ?”
𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐
8 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼐𖽡𖽙𖾐 𖼖𖽡𖾏 𖼇𖽺𖾑 𖼊𖽡𖽾𖾑 , 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 9 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼑𖽜 𖼀𖽡𖾐 𖼺𖽹𖾑 𖼋𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽙𖾏 , 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 𖼀𖽡 𖼟𖽡 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 , 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 .” 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽙𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏 , 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼊𖽑𖽡𖽝𖾐 𖼦𖽪𖾏 𖽃𖽳𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 . 11 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼮𖽘𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏 , 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼚𖽻𖾑 𖼙𖽠𖾏 , 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏 . 12 𖼈𖽡𖽝𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽻 𖼋𖽘𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽹 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼗𖽙𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡 𖼯𖽡𖾏 , 𖼐𖽱𖾏 𖼻𖽻𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 . 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼦𖽪𖾏 𖽃𖽳𖾐 𖼊𖽠𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼐𖽡𖾑 , 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼉𖽺𖾏 , 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼎𖽺𖾏 𖼞𖽑𖽙𖾑 𖼮𖽡𖽝𖾏 , 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 . 13 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 .
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖽐𖼷𖽷𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽑𖽔𖾏 𖼇𖽳𖾏 𖼺𖽘𖾏 , 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼺𖽻𖾏 𖼎𖽻𖾏 𖽐𖼊𖽠𖾑 𖼮𖽡𖽝𖾏 𖽃𖽔𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 , 𖼎𖽺𖾏 𖼟𖽡𖽠 𖼎𖽻𖾏 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼑𖽻 .” 15 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼮𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼏𖽘 , 𖼊𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡 𖼳𖽡𖾑 ” (“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡 𖼳𖽡𖾑 ” 𖼟𖽪𖾏 “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 ”) . 16 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼮𖽘𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 .”
