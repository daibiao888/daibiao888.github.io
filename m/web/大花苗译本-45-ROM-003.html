𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐
3
𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼗𖽙𖾏， 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽺𖾑 ？ 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪 ？ 2 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 ． 𖼼𖽘𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑， 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ． 3 𖼮𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜，𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼖𖽺𖾑 ？𖼋𖽡𖽝𖾏 𖼾𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼯𖽡𖾏 𖼸𖽠𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼘𖽪𖾏？ 4 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼾𖽙！ 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 ：
“ 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖽃𖽡𖾑 ， 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ，
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼳𖽠𖾏 ．”
5 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏， 𖼀𖽡𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼐𖽡𖾑，𖼮𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖼀𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑 ？ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼋𖽡𖽝𖾏 𖼾𖽙 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼘𖽪 ？ 6 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼾𖽙 ！ 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼨𖽑𖽢𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼾𖽔𖾏 𖼖𖽺𖾑 ？ 7 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼑𖽜 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏，𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼖𖽺𖾑 ？ 8 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼀𖽡𖾐 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼎𖽺𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 ”？𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼀𖽡𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼀𖽡𖾐 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑．𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼯𖽡𖽙𖾏．
9 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑 ？𖼀𖽡𖾐 𖼋𖽡𖽝𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽺𖾑 ？𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼾𖽙！𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽹𖾏， 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖽙𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐． 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 ：
“ 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏．
11 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 ，
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
12 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽑𖽔𖾏 𖼇𖽳𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖼺𖽘𖾏，𖽃𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙，
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏．
13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼘𖽑𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖼟𖽪𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏，
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 ，
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽙 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼋𖽙𖾏 ．
14 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽦𖾐 𖼀𖽪𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏．
15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼏𖽘 𖼇𖽺𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑．
16 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏， 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖽐𖼎𖽙 𖼳𖽻𖾏 𖽐𖼚𖽔．
17 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐．
