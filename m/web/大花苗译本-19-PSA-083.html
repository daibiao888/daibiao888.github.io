𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏！
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐！
2 𖼑𖽔𖾏 𖼘𖽪！ 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼟𖽡 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼗𖽙𖾏.
3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼦𖽡 𖽐𖼷𖽷 𖼦𖽡𖾏 𖼊𖽳𖾏, 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼟𖽡 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏.
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽔𖾏 𖼘𖽪！ 𖼀𖽡𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽡𖽝𖾑 𖼞𖽪𖽙！
𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏！”
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡 𖽐𖼷𖽷 𖼦𖽡𖾏 𖼊𖽳𖾏,
𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼽𖽔𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼟𖽡，
6 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐,
𖼄𖽙𖾑 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼳𖽢𖾏 𖼮𖽢 𖼊𖽜𖾑 𖼐𖽜𖾐，
7 𖼮𖽢𖾑 𖼀𖽔𖾑 𖼖𖽳 、𖼽𖽔𖾏 𖼄𖽞 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 、
𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 .
8 𖼽𖽔𖾏 𖼳𖽪 𖼖𖽔𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼀𖽘𖾏 𖼘𖽙 𖼊𖽳 𖼊𖽪𖾐 𖼞𖽡𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
9 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖼯𖽡𖽙𖾏 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏,
𖼺𖽱𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼳𖽪𖽝𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼯𖽡𖽙𖾏 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼀𖽡𖽾𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖽾𖾐 · 𖼊𖽙𖾑 𖽃𖽳𖾐 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽘𖾏.
11 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏, 𖼺𖽱𖾏 𖽂𖽙 𖼖𖽡 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼽𖽡𖾑 𖼀𖽳 ,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽱𖾏 𖼳𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽙 𖼐𖽔 .
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐,
𖽃𖽔𖾏 𖼀𖽡𖾐 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏.
 
13 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽱𖾏 𖽐𖼡𖽜𖾐 𖽐𖼞𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽑𖽙𖾐 𖼷𖽑𖽔𖾐,
𖼺𖽱𖾏 𖼮𖽢𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼦𖽡𖾑 𖼁𖽘𖾏 𖼡𖽻𖾐.
14 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼷𖽑𖽷𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏,
𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼷𖽑𖽷𖾑 𖼯𖽡 𖼎𖽻𖾐 𖼩𖽜𖾏,
15 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼎𖽑𖽻𖾏 𖼮𖽢𖾏 𖼘𖽑𖽙𖾐 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼎𖽑𖽻𖾏 𖼑𖽘 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼮𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼻𖽻𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡 𖽐𖼷𖽷𖾏！
17 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏！
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖽐𖼚𖽙 𖼛𖽙𖾏 𖼋𖽔！
18 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 : 𖼯𖽔 𖼅𖽔𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽡𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐！
