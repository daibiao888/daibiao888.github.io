𖼽𖽙 𖼦𖽔𖾐 𖼺𖽷𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 23： 30 - 35）
36
𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽘𖾏 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼦𖽔𖾐 𖼺𖽷𖾑 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 2 𖼽𖽙 𖼦𖽔𖾐 𖼺𖽷𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼖𖽑𖽡𖾏 . 3 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽪𖽙 𖼋𖽠𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 𖼑𖽡𖽝𖾏 , 𖽃𖽡𖾐 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 𖼞𖽪𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 . 4 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼞𖽙𖾑 𖼮𖽘𖾏 𖼽𖽙 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼮𖽡𖾏 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼚𖽠𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 , 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼽𖽙 𖼦𖽔𖾐 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼙𖽠𖾏 𖼋𖽘𖾏 .
𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 23：36 - 24：7）
5 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 , 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 — 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 . 6 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 , 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼋𖽻𖾏 𖼡𖽑𖽹𖾐 𖼎𖽪𖾑 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏 , 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏 . 7 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼑𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 , 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼞𖽔𖾏 𖼁𖽠 𖼮𖽡𖽝𖾏 (“𖼮𖽵𖾑 𖼳𖽺 𖽐𖼞𖽔𖾏 𖼁𖽠 ”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼾𖽨𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻”) . 8 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖼑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖼳𖽪𖾐 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼸𖽪𖾏 𖼼𖽘𖾏 , 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏《 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏 》𖼮𖽡𖽝𖾏. 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 24： 8 - 17）
9 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 (“𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 ” 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 24:8 𖽃𖽔𖾏 “𖼟𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏” ) , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼖𖽑𖽡𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽻𖾏 𖼐𖽑𖽪𖾐 ，𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 . 10 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 , 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼯𖽡𖽙𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏 , 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼾𖽡𖾏 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 , 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 .
𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 24：18 -20； 𖼽𖽦𖾑 52：1- 3）
11 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏， 12 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 . 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼞𖽑𖽻𖾑 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼳𖽻𖾐 𖼢𖽜 .

23 “𖼀𖽙𖾑 𖼺𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖾐 𖼖𖽡𖽝 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼞𖽑𖽡𖽠𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 , 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼑𖽜 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 . 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏. 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 .”
