𖼽𖽔𖾐 𖼞𖽙 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼣𖽹 𖼮𖽡 𖼅𖽔𖾏 𖼡𖽻𖾐, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼅𖽡 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖽐𖼊𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑, 𖼷𖽑𖽡𖽝𖾏 𖼑𖽔𖾏 𖼷𖽑𖽡𖽝𖾏 𖽐𖼎𖽙𖾏 𖼘𖽪? 2 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼣𖽹 𖼮𖽡 𖼅𖽔𖾏 𖼡𖽻𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼑𖽜 𖼀𖽡𖾐 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼗𖽙, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼻𖽻𖾏 𖼊𖽪𖾑 𖼳𖽦𖾑, 𖼦𖽡𖾐 𖼸𖽙 𖼮𖽑𖽡𖽠𖾏 𖼋𖽔.” 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼟𖽻𖾏 𖼗𖽜𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼅𖽻𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼙𖽠𖾏. 4 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐, 𖼽𖽔𖾐 𖼞𖽙 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔. 5 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼅𖽡 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼐𖽔𖾑.
6 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼺𖽳 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼋𖽺 𖼡𖽻𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼾𖽙 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏. 𖼽𖽙 𖼺𖽳 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡 𖼋𖽔𖾏 𖼋𖽘, 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏. 7 𖼽𖽙 𖼺𖽳 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡, 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼼𖽔𖾏, 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖼋𖽔𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐.” 8 𖼽𖽙 𖼺𖽳 𖼀𖽻𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼅𖽡, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏. 9 𖼽𖽙 𖼺𖽳 𖽐𖼊𖽱𖾏 𖼸𖽙 𖽃𖽔𖾐 𖼋𖽪 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏, 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼟𖽪𖾏 𖼸𖽜 𖼟𖽪𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏, 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼋𖽔𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼮𖽔𖾑𖽑 𖽐𖼮𖽠𖾏.” 10 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪! 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼸𖽜 𖼟𖽪𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏, 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼊𖽜𖾑 𖼯𖽡𖽙𖾏. 11 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼊𖽜𖾑, 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼸𖽜 𖼟𖽪𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼋𖽡𖽝.” 12 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼾𖽙 , 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽔𖾑𖽑 𖽐𖼮𖽠𖾏 𖼊𖽜𖾑 𖼯𖽡𖽙𖾏.” 13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽡𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼊𖽔𖾐, 𖼀𖽡𖾐 𖼡𖽑𖽔𖾐 𖼁𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖽐𖼞𖽔𖾏, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼋𖽘.”

14 𖼽𖽙 𖼺𖽳 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼑𖽡𖽝 𖼐𖽑𖽡𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼸𖽜 𖼟𖽪𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏, 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏. 15 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼮𖽔𖾏 𖼅𖽡: 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼏𖽵 𖼗𖽪𖾏, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼡𖽔𖾏 𖽐𖼷𖽠𖾏 𖼦𖽡𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼉𖽺𖾏 𖼐𖽡𖾑 , 𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼋𖽠𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼺𖽷𖾐 𖼅𖽻. 16 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽷𖾐 𖼅𖽻, 𖼄𖽔𖾐 𖼅𖽡 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼨𖽑𖽳𖾏 𖼋𖽔𖾏. 𖼅𖽡 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽙𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼮𖽡𖽝𖾏. 𖼻𖽻𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝. 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼋𖽡𖽝, 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼑𖽡𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼏𖽵 𖼗𖽪𖾏, 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼸𖽜 𖼟𖽪𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼯𖽡𖽙𖾏.” 17 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽙 𖼺𖽳 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼮𖽡𖽝𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐.
