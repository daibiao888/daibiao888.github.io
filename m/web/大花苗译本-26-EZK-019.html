𖼞𖽑𖽡𖽠𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽃𖽔𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏
19
𖼟𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐， 2 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?
𖼟𖽪𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐,
𖼼𖽻𖾏 𖼯𖽠𖾏 𖼎𖽻𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼊𖽪𖾑 𖼾𖽨 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐.
3 𖼐𖽱𖾏 𖼊𖽪𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽙𖾏 𖼘𖽑𖽙𖾐,
𖼘𖽑𖽙𖾐 𖼊𖽡𖽝𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏,
𖼾𖽻 𖼸𖽠𖾏 𖽐𖼎𖽑𖽦𖾑 𖼡𖽜𖾑 𖼑𖽻𖾏, 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
4 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼯𖽡𖽻𖾏 𖼐𖽱𖾏, 𖽐𖼮𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖼮𖽡𖽝𖾏,
𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖼮𖽤𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼙𖽠𖾏.
5 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼁𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼋𖽙𖾏 𖽃𖽡𖾑 𖼛𖽙𖾏 𖼺𖽘𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏,
𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽪𖾑 𖼊𖽡𖽝𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏.
6 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼀𖽙𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽻𖾏 𖼗𖽙𖾏,
𖼘𖽑𖽙𖾐 𖼊𖽡𖽝𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏,
𖼾𖽻 𖼸𖽠𖾏 𖽐𖼎𖽑𖽦𖾑 𖼡𖽜𖾑 𖼑𖽻𖾏, 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
7 𖼐𖽱𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽻 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼑𖽜 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏,
𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽡𖾑, 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼺𖽘𖾏.
8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏,
𖼄𖽔𖾐 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑,
𖼯𖽡𖽻𖾏 𖼐𖽱𖾏 𖽐𖼮𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖽐𖼚𖽙.
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖽐𖼡𖽜𖾏 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏, 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼚𖽱𖾐 𖽐𖼚𖽙,
𖼨𖽑𖽳𖾏 𖼸𖽙 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
𖼮𖽙𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏,
𖼟𖽙𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏.
𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼊𖽻𖾏 𖼚𖽙𖾐 𖽃𖽔𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏
10 𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 ,
𖼐𖽡𖽙𖾐 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽐𖼚𖽙 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼐𖽡𖽙𖾐 𖼟𖽡 𖽐𖼮𖽑𖽘𖾑 𖽐𖼚𖽙 ” ),
𖼯𖽡𖽙 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽻𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼮𖽡𖽻𖾐,
𖼎𖽺𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼘𖽘𖾑𖽑 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
11 𖼘𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼮𖽤𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏, 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏.
𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼮𖽤𖾏 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼊𖽔𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 ,
𖼘𖽑𖽘𖾑 𖼘𖽑𖽙𖾐 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐,
𖽃𖽔𖾑 𖼯𖽡 𖽐𖼷𖽑𖽻𖾑 𖼻𖽻𖾑, 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼺𖽷𖾐 𖼷𖽹𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏 .
12 𖼈𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼐𖽡𖾑 𖼑𖽜 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼊𖽻𖾏 𖼚𖽙𖾐 𖼗𖽔𖾏 𖽐𖼮𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼮𖽢𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼷𖽑𖽔𖾐 𖽐𖼞𖽡 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐, 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 𖼊𖽻𖾏 𖼖𖽙𖾑 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏,
𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏.
13 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼯𖽡𖽙 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏,
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽻𖾐.
14 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼯𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼏𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏.”
 
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼗𖽙 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼨𖽑𖽪𖾏.
