𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 ，𖼐𖽱𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽨𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽠𖾏 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏． 2 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏． 3 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼺𖽘𖾏， 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼁𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏． 4 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽠𖾏 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏，𖼟𖽡 𖼮𖽔𖾑 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑，𖼟𖽡 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼾𖽨𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏．” 5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼐𖽱𖾏． 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼮𖽡𖽻𖾏 𖼺𖽷𖾐，𖼅𖽡 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼊𖽪𖾏 𖼻𖽻𖾏． 7 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽪𖾐 𖼅𖽡，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼋𖽠𖾏、 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏． 8 𖼅𖽡 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼦𖽡𖾑！𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼟𖽱𖾏 𖼊𖽪𖾑 𖼮𖽦𖾐 𖼐𖽡𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼸𖽙 𖼺𖽷𖾐．” 9 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖽐𖼎𖽙𖾏．
10 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼖𖽔𖾏 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐，𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼅𖽻𖾏，𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼊𖽔𖾐 𖼯𖽷 𖼉𖽺𖾏 𖼅𖽻𖾏 𖼊𖽔𖾐． 11 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐，𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 ，𖼐𖽑𖽡𖾏：“𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏？” 12 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏，𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼐𖽱𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐．”𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼦𖽡𖾐，𖼐𖽱𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐．” 13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏．

32 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼅𖽡 𖽐𖼎𖽙 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼅𖽻𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏． 33 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼺𖽷𖾐，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 34 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑，𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏；𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐，𖼅𖽡 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏．” 35 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼙𖽠𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏？ 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼽𖽨𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾏 𖽃𖽡𖾑，𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽔𖾑 𖼖𖽔𖾏？ 36 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：‘ 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑，𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏，𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐，𖼅𖽡 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏．’ 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ？ ”
