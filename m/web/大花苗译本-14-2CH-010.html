𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼷𖽹𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 12：1 - 20）
10
𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 , 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 2 𖼐𖽡 𖼀𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 . 𖼐𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼎𖽙𖾑 𖼗𖽙𖾏 . 3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼮𖽘𖾏 𖼐𖽱𖾏， 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : 4 “𖼟𖽡 𖼉𖽺𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 , 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 、 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼳𖽷𖾐 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 , 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡.” 5 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼸𖽙 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼅𖽡 𖼎𖽺𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 !” 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽺𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏.
𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼦𖽡𖾐 𖼑𖽻 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑
6 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 , 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 , 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 , 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼅𖽔𖾏 𖼋𖽪 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼊𖽡𖾐 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 .” 7 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽔𖾑 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼙𖽔 ,𖼎𖽑𖽻𖾏 𖼻𖽻𖾏 𖼗𖽪𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 .” 8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼘𖽑𖽙𖾐 , 𖼟𖽪𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼦𖽡𖾏 𖼊𖽳𖾏 , 9 𖼐𖽑𖽡𖾏 : “𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 : 𖼟𖽡 𖼉𖽺𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 , 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼳𖽷𖾐 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 . 𖼅𖽡 𖼅𖽔𖾏 𖼋𖽪 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 .” 10 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : 𖼟𖽡 𖼉𖽺𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 , 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼳𖽷𖾐 𖽃𖽡𖾐 𖽐𖼮𖽷𖾐 𖼣𖽔𖾏. 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼞𖽪𖾑 𖼊𖽔𖾑 𖽃𖽔𖾏 𖼯𖽱𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖽐𖼊𖽪 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼚𖽔𖾑 ； 11 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼞𖽱𖾑 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼐𖽑𖽤𖾑, 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖽐𖼚𖽠𖾏 𖼅𖽡 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽦𖾏 𖼖𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖼅𖽡 .”
