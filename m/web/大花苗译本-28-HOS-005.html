𖼽𖽨 𖼊𖽔𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼢𖽻.
6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽤𖾐 𖼑𖽨𖾏 𖼮𖽑𖽷𖾏 𖼅𖽻𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫

𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼙𖽠𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,

8 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐,

11 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼑𖽜 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑,
𖼎𖽺𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 , 𖼊𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐, 𖼸𖽺𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼋𖽪 𖼺𖽘𖾏.

15 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽠𖾏,
