4

𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼐𖽡𖽙𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼯𖽪𖾏 𖽐𖼀𖽔 𖼞𖽪𖾏 𖽐𖼷𖽻𖾑 𖼚𖽑𖽪𖾏 𖽐𖼚𖽙 𖼺𖽱𖾏 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐．

2 𖼟𖽡 𖼐𖽡𖽝𖾑𖽑 𖼺𖽱𖾏 𖽃𖽡𖾐 𖽐𖼀𖽙𖾐 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼖𖽻𖾑 𖼚𖽻𖾐 𖼺𖽘𖾏, 𖽐𖼷𖽔𖾑 𖼨𖽑𖽪𖾑 𖼺𖽘𖾏 𖽐𖼮𖽡𖾏 𖼖𖽙𖾏,

3 𖼟𖽡 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼷𖽙𖾏 𖼺𖽙𖾑 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐,
𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼑𖽔𖾏,
𖼟𖽡 𖽃𖽔𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖼸𖽷𖾏 𖼐𖽡𖽙𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼯𖽪𖾏 𖽐𖼀𖽔 𖼟𖽪𖾏 𖽐𖼷𖽻𖾑 𖼚𖽑𖽪𖾏 𖽐𖼚𖽙

4 𖼟𖽡 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐 𖼺𖽱𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼳𖽻𖾏 𖼏𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐,
𖼉𖽺𖾏 𖽃𖽡𖾑 𖼮𖽘𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼐𖽱𖾏 𖼛𖽔𖾏 𖽐𖼀𖽙 𖽐𖼎𖽙𖾐.
5 𖼟𖽡 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼄𖽡𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖽐𖼞𖽡𖽠𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔 𖼑𖽡𖽝 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏 𖽐𖼚𖽙 𖼑𖽻𖾏 𖽐𖼡𖽜𖾐,
𖼟𖽪𖾏 𖽃𖽔𖾑 𖽐𖼷𖽹𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏 𖼨𖽑𖽳𖾏.
6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼄𖽪 𖼽𖽙 𖼎𖽻𖾐 𖽐𖼎𖽙 𖼵𖽪𖾐 𖼳𖽤𖾏 𖼎𖽻𖾐,
𖼋𖽙𖾏 𖼟𖽙𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼳𖽠𖾑 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼸𖽡𖽝𖾏,
𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 𖼙𖽠𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑.
7 𖼞𖽪𖾑 𖼖𖽹𖾐 𖼗𖽔 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼘𖽪 ! 𖼟𖽡 𖽃𖽡𖾐 𖼮𖽡𖾏 𖼙𖽱 𖼙𖽱 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼋𖽡𖽝 ,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼋𖽡𖾏 𖼺𖽻𖾏 !
8 𖼞𖽪𖾑 𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 𖼘𖽪 ! 𖼷𖽑𖽜𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 ,
𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 .
𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼐𖽔 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐,
𖼎𖽑𖽻𖾏 𖼳𖽷𖾏 𖼐𖽡 𖽃𖽳𖾐 𖽐𖼎𖽙 𖼦𖽳 𖼄𖽞 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐,

9 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑 , 𖼞𖽪𖾑 𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 𖼘𖽪! 𖼟𖽡 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,

10 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑 , 𖼞𖽪𖾑 𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 𖼘𖽪 !
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏！
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏！
𖼟𖽡 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼚𖽜𖾑 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏！
11 𖼞𖽪𖾑 𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 𖼘𖽪 ! 𖼟𖽡 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼗𖽙𖾏,
