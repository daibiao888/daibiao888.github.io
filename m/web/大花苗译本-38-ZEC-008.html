𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 2 “𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼞𖽪𖾑 𖼑𖽜 𖼳𖽡 𖼣𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽙𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼞𖽪𖾑 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽙𖾑, 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐. 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼳𖽡 𖼣𖽔𖾑 , 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑. 4 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔 𖼾𖽠 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏, 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏, 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏. 5 𖼖𖽪𖾑 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏, 𖼽𖽔𖾑 𖼀𖽪𖾑 𖽐𖼷𖽡𖽝𖾑 𖼊𖽪𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼇𖽜𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑． 6 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼸𖽙 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑, 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼖𖽔𖾏 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼘𖽪𖾏? 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 7 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐, 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼮𖽨𖾏 𖼎𖽙𖾑 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏. 8 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽷𖾐 𖼗𖽙 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖽐𖼎𖽙 𖼖𖽠𖾑 𖼞𖽡𖾑 .

20 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼙𖽻 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏. 21 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 , 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:‘ 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼅𖽻𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼅𖽳𖾏, 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏.’ 22 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏, 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼅𖽳𖾏.” 23 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼗𖽪𖾏, 𖼊𖽔𖾑 𖼮𖽡𖽝𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾑 𖼡𖽘𖾐 𖼮𖽑𖽙𖾏, 𖼐𖽑𖽡𖾏: ‘ 𖼀𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼅𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼋𖽘 . ’
