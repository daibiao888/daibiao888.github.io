22
𖼽𖽔𖾏 𖼖𖽔 𖼞𖽪𖽙 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽪𖽙 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏. 2 𖼸𖽙 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏. 3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼀𖽡𖾐 𖼁𖽡𖽝 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙, 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 𖼖𖽺𖾑?” 4 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽪 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽑𖽡𖾏: “𖼋𖽡𖽝𖾏 𖼟𖽡 𖼯𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙 𖼖𖽺𖾑?” 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼟𖽡 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐, 𖼞𖽪𖾑 𖼑𖽜𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼑𖽜𖾏 𖼖𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐.”
𖼦𖽡 𖼸𖽱 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼋𖽔𖾏 𖼑𖽪
5 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙𖾏 𖼟𖽡 𖼺𖽪𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏.” 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡 𖼸𖽱 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼋𖽔𖾏, 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏, 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏𖼖𖽔𖾑 𖼄𖽙 𖼋𖽡𖽝𖾏 𖼅𖽻𖾏 𖼢𖽻 𖼖𖽔𖾏 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼢𖽻?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼢𖽻 𖼯𖽡𖽙𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.” 7 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽑𖽡𖾏: “𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼡𖽔𖾑 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐, 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼖𖽺𖾑?” 8 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽑𖽡𖾏: “𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽷𖾐, 𖼟𖽪𖾏 𖼽𖽡𖽾𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼄𖽡𖾐 𖼞𖽹𖾑 𖼽𖽔𖾐, 𖼀𖽡𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫． 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼐𖽱𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪, 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼊𖽔𖾐.” 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽑𖽡𖾏: “𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 .” 9 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏 𖼋𖽔𖾏, 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼅𖽻𖾏 𖼨𖽑𖽪𖾏 𖼽𖽡𖽾𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼄𖽡𖾐 𖼞𖽹𖾑 𖼽𖽔𖾐 𖼋𖽔𖾏.” 10 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼐𖽑𖽘𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽻𖾏, 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏. 11 𖼮𖽡𖾑 𖼐𖽔 𖼐𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼋𖽪 𖼞𖽪𖾐 𖼖𖽑𖽻𖾏, 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼞𖽪𖾐 𖼐𖽡𖾑 𖽐𖼎𖽻𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼸𖽙𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.” 12 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼐𖽑𖽡𖾏：“𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙 𖼢𖽻 𖼯𖽡𖽙𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.”

𖼽𖽔𖾏 𖼦𖽔𖾐 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼋𖽔 𖼺𖽘𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 18：28 - 34）
29 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙 𖼙𖽠𖾏 𖼋𖽘𖾏. 30 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼘𖽑𖽪𖾑 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏, 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽑𖽘𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽙𖾏 .” 𖽃𖽔𖾏 𖼐𖽡𖾑， 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼘𖽑𖽪𖾑 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏. 31 𖼺𖽪𖾐 𖼁𖽱 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼋𖽻𖾏, 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼘𖽑𖽙𖾐 𖼳𖽻𖾐, 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼮𖽔𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 𖼊𖽔𖾐.” 32 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔, 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏.” 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏. 33 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼋𖽘𖾏. 34 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼀𖽙𖾑 𖼐𖽜𖾑𖽑, 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼀𖽙𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼷𖽷𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽘𖾐． 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑, 𖼟𖽡 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼗𖽙𖾏, 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼦𖽡𖾑!” 35 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽺𖾐 𖽐𖼊𖽻𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑, 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽠𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐, 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐． 𖼸𖽙 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏，𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖽐𖼮𖽑𖽘𖾑 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖽐𖼚𖽱𖾏 𖼎𖽻𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼮𖽡𖽝𖾏. 36 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖽐𖼷𖽜𖾑 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼏𖽙 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼾𖽨𖾏 𖼙𖽻, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾑 !”
37 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏, 𖼎𖽺𖾏 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑. 38 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐, 𖽐𖼷𖽔𖾑 𖼎𖽻𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼁𖽘𖾏 𖽃𖽻𖾐 ( 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 ), 𖼚𖽡𖾑 𖼋𖽔𖾏 𖼾𖽺𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽘𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏. 39 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼷𖽑𖽱𖾏 𖼟𖽻𖾏 𖽃𖽔𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏, 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏《 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏》𖼮𖽡𖽝𖾏. 40 𖼽𖽔𖾏 𖼦𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖼗𖽠𖾏, 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼳𖽡𖽝𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
