𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼖𖽡𖽾 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏,
𖼄𖽔𖾐 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼘𖽔𖾑𖽑 .
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼘𖽑𖽘𖾑 𖼎𖽻𖾏 𖼞𖽙𖾏 𖽐𖼡𖽜𖾐 𖽐𖼚𖽙 ,

5 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼁𖽡𖽝,
𖼷𖽷𖾑 𖼉𖽺𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼾𖽨𖾏 𖼨𖽑𖽪𖾏 𖼾𖽨𖾏,
𖼎𖽺𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼺𖽻𖾏 𖼟𖽪𖾏 “𖼺𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫”，
𖼾𖽨𖾏 𖼨𖽑𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 .
 
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼞𖽪𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼁𖽱𖾏, 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏,
𖼋𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
7 𖼋𖽳 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼺𖽱𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖽐𖼎𖽑𖽹𖾏, 𖼎𖽺𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼖𖽺𖾑?
𖼮𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏.
8 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼏𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏．
𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼷𖽹𖾏 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼅𖽡 , 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼋𖽘𖾏 𖼘𖽪?
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏, 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼋𖽡𖽝𖾏 𖼡𖽑𖽔𖾑 𖼅𖽔𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼺𖽷𖾐 𖼖𖽺𖾑?
𖼋𖽡𖽝 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏, 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐.”
𖼷𖽑𖽙𖾏 𖼷𖽷𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏

10 𖼡𖽔 𖼋𖽱𖾏 𖼷𖽷𖾏 𖼁𖽠𖾏 𖼼𖽘𖾏, 𖼘𖽡𖾑𖽑 𖼷𖽷𖾏 𖼁𖽠𖾏 𖼼𖽘𖾏, 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.
11 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾐, 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡 𖼸𖽱, 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽠𖾑 𖼏𖽔𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾏 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏, 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
12 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼖𖽑𖽻𖾏 𖼄𖽔𖾐 𖼖𖽑𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼟𖽙𖾏 𖼞𖽪𖾐, 𖼎𖽑𖽻𖾏 𖼮𖽑𖽪𖽺 𖽐𖼊𖽪𖾏 𖼖𖽑𖽻𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐, 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼊𖽻𖾏 𖼟𖽙𖾏 𖼊𖽡𖽝𖾑. 𖼐𖽱𖾏 𖼮𖽑𖽹𖾐 𖼋𖽘 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼦𖽡𖾐 𖼦𖽻𖾏 𖽃𖽻𖾐 𖼎𖽺𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏. 13 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼋𖽠𖾏 𖼮𖽤𖾐 𖼖𖽑𖽔𖾏 𖽐𖼮𖽔𖾏, 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖽐𖼚𖽘𖾏 𖽐𖼚𖽘𖾏 𖼋𖽠𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏， 𖼎𖽑𖽻𖾏 𖼡𖽔𖾑 𖽐𖼚𖽺𖾏 𖼄𖽑𖽘𖾏 𖼻𖽠𖾏 𖼟𖽪𖾏 𖼊𖽡𖽝𖾑 𖼼𖽘𖾏, 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖽐𖼮𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼘𖽙𖾐 𖼇𖽻𖾏 𖼋𖽠𖾏 𖼼𖽘𖾏, 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼼𖽘𖾏, 𖼸𖽻 𖼋𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖼟𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽙 .
14 𖼐𖽱𖾏 𖽐𖼊𖽙𖾑 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏, 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼦𖽡𖾑 𖼑𖽡𖽻𖾏 𖽐𖼎𖽙 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼻𖽻𖾏 𖽐𖼚𖽙 𖼘𖽑𖽙𖾐, 𖼐𖽱𖾏 𖼯𖽡𖽙 𖼊𖽑𖽪𖾑 𖼊𖽻𖾏 𖼑𖽘 𖼗𖽙𖾏 𖽐𖼊𖽙𖾑 𖼾𖽨 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
15 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼐𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼚𖽻𖾑,𖼯𖽡𖽙𖾏 𖼗𖽙 𖼚𖽻𖾑 𖼋𖽠𖾏 𖽐𖼊𖽡𖾏, 𖼎𖽺𖾏 𖼚𖽻𖾑 𖼮𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾐 𖼳𖽡𖽝𖾐, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼁𖽠𖾏 𖼼𖽘𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏, 𖽃𖽔𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼐𖽱𖾏. 16 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖼯𖽘𖾏 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼚𖽻𖾑 𖼷𖽑𖽡𖽝𖾐 𖽐𖼡𖽹𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏, 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽐𖼊𖽡𖾏 𖼋𖽠𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾏 𖼦𖽔𖾏！ 𖼞𖽪𖾑 𖼳𖽙𖾑 𖼋𖽘𖾏, 𖼞𖽪𖾑 𖼁𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼋𖽘𖾏.” 17 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼁𖽠𖾏, 𖼟𖽪𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏, 𖼐𖽱𖾏 𖼊𖽱𖾐 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾑 𖼁𖽠𖾏 𖽃𖽡𖾑 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.”
18 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼮𖽡𖽝𖾏 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏. 19 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼯𖽷𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐, 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼙𖽔 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖼯𖽘𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼮𖽡 𖼋𖽠𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼙𖽔 𖼷𖽑𖽡𖽝𖾐 𖼳𖽡𖽝𖾐, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼙𖽔 𖼷𖽑𖽡𖽝𖾐 𖽐𖼡𖽹𖾏 𖼑𖽻𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼽𖽔𖾑 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼖𖽑𖽘𖾑 𖽃𖽔𖾑 𖼮𖽑𖽱𖾏 𖼋𖽠𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏?”
20 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼏𖽻 𖼮𖽑𖽵𖾑, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐, 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼾𖽨𖾏 𖼮𖽨𖾏 𖼾𖽨𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼘𖽪?”
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏
21 𖼽𖽔𖾐 𖼞𖽙 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 !
𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏, 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼷𖽷𖾏 𖼟𖽡 , 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ! 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼟𖽡 .

𖼞𖽪𖾑 𖼇𖽳𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖼺𖽱𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼽𖽘𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼙𖽠𖾏,

23 𖽐𖼊𖽪𖾏 𖼘𖽪! 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏,

𖼊𖽡𖾐 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐 𖼘𖽪 ! 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏.

𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼘𖽪 !
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼽𖽔𖾐 𖼞𖽙 ,
