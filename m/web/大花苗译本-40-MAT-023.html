37 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！ 𖼟𖽡 𖼊𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐，𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑．𖼞𖽪𖾑 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼟𖽻𖾐 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏，𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼡𖽹𖾐 𖼟𖽙𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼡𖽹𖾐 𖼦𖽡 𖼸𖽱 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏． 38 𖼑𖽔𖾏 𖼘𖽪！𖼅𖽡 𖽐𖼞𖽔𖾏 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼅𖽡． 39 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼋𖽳 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼅𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏，𖼽𖽔𖾑 𖼋𖽙𖾏 𖼸𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏 ： ‘ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼮𖽵𖾑 𖽐𖼷𖽷𖾏 𖼋𖽔𖾏， 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏．’
