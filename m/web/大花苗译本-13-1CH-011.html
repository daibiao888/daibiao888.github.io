11
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 𖼁𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 , 𖼐𖽑𖽡𖾏 : “𖼑𖽔𖾏 𖼘𖽪 ! 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼡𖽹𖾏. 2 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 , 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼏𖽵 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 : 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽪𖾑 𖼾𖽨 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 , 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 .” 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼗𖽻 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼺𖽔𖾐 𖼄𖽪𖾐 𖽃𖽳𖾐 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 . 4 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 , 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 . 5 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 .” 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 , 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻 . 6 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼡𖽔 𖼋𖽱𖾏 𖼺𖽪𖾐 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 .” 𖼳𖽡𖾐 𖼖𖽪𖾐 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼽𖽔 𖼺𖽪𖾐 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 . 7 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽐𖼚𖽙 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻 . 8 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼄𖽡𖾐 𖼘𖽙 𖼳𖽠𖾑 𖼁𖽱𖾏 , 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 , 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼽𖽔 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏. 9 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼟𖽡𖾏 𖼾𖽙 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 .
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼋𖽻
（𖼺𖽔𖾐 𖽃𖽔𖾐 23： 8 - 39）
10 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 , 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼞𖽪𖽙 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 , 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽙𖾏 . 11 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 : 𖼦𖽔𖾐 𖼞𖽳 𖼄𖽙𖾑 𖼐𖽡 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼳𖽙 𖼀𖽔𖾑 , 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 , 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼊𖽔𖾐 𖼮𖽘𖾏 𖼄𖽑𖽪𖾑 𖼐𖽑𖽙𖾏 𖼋𖽔𖾏 𖼺𖽘𖾏 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 .
