𖼮𖽡𖾑 𖼊𖽪 𖼎𖽺𖾏 𖼋𖽔𖾏
6 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏．𖼮𖽵𖾑， 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼖𖽡𖽾 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼷𖽷𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏 𖼋𖽘𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏． 7 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏！𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼑𖽘 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏．”
8 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽙𖾑𖽑、 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽙 𖼦𖽗．𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾑，𖼞𖽪𖾑 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼐𖽱𖾏． 9 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑！𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽔𖾐．𖼟𖽡 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．” 10 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼑𖽘 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 ，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼘𖽙𖾏 𖼮𖽡𖽝𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏． 11 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ；𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏；𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ；𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏．”
12 “ 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏！𖼺𖽘𖾏 𖼅𖽳𖾏、 𖼺𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼯𖽔𖾏 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏． 13 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼖𖽔𖾑 𖼇𖽔，𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽙 𖼄𖽞 𖼞𖽔；𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑， 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 ，𖼞𖽪𖾑 𖼟𖽪𖾏 𖼁𖽱𖾏，𖼞𖽪𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏．”
14 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽑𖽔𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽐𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼉𖽺𖾏 𖽃𖽡𖾑，𖼎𖽺𖾏 𖼸𖽠𖾏 𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼚𖽔𖾏 𖼙𖽻． 15 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖼙𖽻 𖼅𖽔𖾏 ：𖽐𖼀𖽙𖾏 𖼚𖽡𖾑、 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽜𖾐、𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼁𖽠𖾏，𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑．
16 “𖼞𖽪𖾑 — 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼷𖽷𖾐， 𖼑𖽜 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽪𖾑 𖼮𖽡𖽻𖾏 𖼦𖽪𖽺𖾏，𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽹𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡．𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐， 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏，𖼞𖽪𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖾏．”
17 𖼳𖽟𖾏 𖼖𖽡𖽾 𖽐𖼎𖽙 𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 𖼖𖽔𖾏 𖼐𖽑𖽡𖾏：“𖼋𖽔𖾏！” 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽙𖾑𖽑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼋𖽔𖾏！” 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏．𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽪𖾏 𖼋𖽔𖾏 𖼽𖽔𖾑 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼊𖽻𖾏 𖼦𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 ．
18 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽹𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽙𖾑𖽑 𖼑𖽘𖾏 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏：𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼐𖽡𖾑 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼎𖽻𖾏，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼑𖽘 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑． 19 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼑𖽘 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑，𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾏 𖼺𖽘𖾏，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼑𖽘 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼇𖽳𖾏 𖼐𖽱𖾏 𖽐𖼀𖽠 𖼺𖽘𖾏．
20 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼐𖽑𖽡𖾏：“ 𖼾𖽙 𖼯𖽡𖽙𖾏，𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏！” 𖽃𖽔𖾏 𖼄𖽞！𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼘𖽪！𖼞𖽪𖾑 𖼯𖽪𖾏 𖼟𖽡 𖼋𖽔𖾏！
