𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑 𖽐𖼚𖽪𖾏 𖼸𖽳！
2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼙𖽳𖾐 𖼷𖽱𖾏 𖼟𖽙𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏.
3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖽐𖼮𖽳𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖼑𖽡𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏,
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖽐𖼚𖽪𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑 𖽐𖼚𖽪𖾏 𖼸𖽳 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼊𖽑𖽠𖾐 𖼞𖽪𖾑 𖼢𖽻 .
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖽐𖼚𖽺𖾏 𖼊𖽑𖽔𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽔𖾏 𖼮𖽡𖽝𖾏 𖼽𖽔𖾑 𖼮𖽢𖾐 𖼞𖽪𖾑 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽑𖽔𖾑 𖼸𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 𖼮𖽡𖽝𖾏, 𖼽𖽔𖾑 𖼮𖽢𖾐 𖼞𖽪𖾑. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
6 𖼞𖽪𖾑 𖼙𖽔 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.”
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼊𖽜 𖼽𖽔𖾑！
7 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼘𖽪！
𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼁𖽔𖾏 𖼏𖽙 , 𖼟𖽡 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼮𖽡𖽝𖾏.
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖼙𖽱 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼦𖽡𖾑 𖼚𖽘𖾐 𖽐𖼊𖽪𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼞𖽪𖾑 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼟𖽡𖾏 𖼦𖽺𖾏 (“𖼟𖽡𖾏 𖼦𖽺𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼀𖽙 𖼈𖽳𖾑”) 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼊𖽔𖾐！
10 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡 𖼋𖽠𖾏 𖼀𖽻𖾐 𖼷𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼊𖽙𖾐, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 .
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏,
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼽𖽔𖾑 𖼦𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑 𖽐𖼚𖽪𖾏 𖼸𖽳, 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼢𖽻 .
 
12 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼐𖽑𖽡𖾏 𖼞𖽡𖾑 .
13 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖽐𖼷𖽷𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
