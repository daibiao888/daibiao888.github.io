12
𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖼸𖽙 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼳𖽻𖾐 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏．𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼮𖽑𖽹𖾐 𖼋𖽘，𖼎𖽺𖾏 𖼊𖽪𖾏 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐 𖼳𖽻𖾐 𖼗𖽙𖾏 𖼑𖽻𖾏． 2 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 ， 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼑𖽔𖾏，𖼟𖽡 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏．” 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼟𖽡𖽠 𖼺𖽷𖾐，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼐𖽱𖾏 𖼮𖽑𖽹𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？ 4 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏， 𖼑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼊𖽜𖾐 𖼮𖽡𖽝𖾏 𖼖𖽺𖾑？ 𖼋𖽡𖾏 𖼳𖽡𖽝𖾐 𖼐𖽡𖾑 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼑𖽻𖾏，𖼯𖽔 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼋𖽡𖽝𖾏 𖼑𖽻𖾏 𖼢𖽻． 5 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼊𖽠𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏， 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 ， 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼺𖽘𖾏，𖼖𖽔𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏， 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼟𖽡𖽠 𖼖𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑？ 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼉𖽺𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏． 7 ‘ 𖼞𖽪𖾑 𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐，𖼦𖽡𖾐 𖼙𖽔 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏．’ 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏，𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼋𖽘𖾏． 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏
（𖼞𖽑𖽙𖾐 3：1-6；𖼖𖽪𖾏 6：6-11）
9 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏， 𖼁𖽺𖾏 𖼛𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱． 10 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽃𖽡𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 ： “ 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼞𖽑𖽙𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐，𖼋𖽡𖽝𖾏 𖽃𖽔𖾏 𖼢𖽻 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼦𖽡𖾐 𖼢𖽻？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑． 11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼾𖽘𖾏， 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼙𖽠𖾏，𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼊𖽔𖾑 𖼮𖽤𖾐 𖼐𖽱𖾏 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼘𖽪？ 12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼾𖽘𖾏 𖼺𖽷𖾐 ， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼢𖽻 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏．” 13 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 ： “𖼳𖽤𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏．” 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 ，𖼺𖽱𖾏 𖼷𖽑𖽹𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐．
𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
14 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏，𖼦𖽡𖾏 𖼊𖽳𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽑𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 ． 15 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼋𖽘，𖼎𖽺𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏． 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖼐𖽱𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖼻𖽻𖾏 𖼋𖽘𖾏， 16 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐． 17 𖼽𖽔𖾑 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏：
18 “𖼅𖽡 𖼑𖽔𖾏，𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑 𖼊𖽻𖾏，
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏， 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼙𖽔 𖽃𖽡𖾑，
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼎𖽻𖾏 𖼐𖽱𖾏，
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐．
19 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏，𖼐𖽱𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏，
𖼖𖽪𖾑 𖼟𖽡𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏．
20 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼖𖽙𖾑 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼦𖽡𖾐 𖼖𖽙𖾑 𖼊𖽪𖾏 𖼺𖽘𖾏 ，
