𖼗𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖽐𖼚𖽪𖾏
18
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼗𖽔 𖼦𖽡𖾐 𖼷𖽷𖾐 , 𖼾𖽨𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼊𖽔𖾐,
𖼎𖽺𖾏 𖼳𖽪𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼋𖽡𖽝 .
2 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼉𖽳 𖼀𖽻𖾐 𖼊𖽘𖾐,
𖼮𖽔 𖼸𖽠𖾏 𖼾𖽨𖾏 𖼢𖽳 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼊𖽔𖾐.
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼋𖽔𖾏, 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼋𖽔𖾏；
𖼑𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖽐𖼮𖽡𖽻𖾏, 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼚𖽑𖽪𖾏 𖼖𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼮𖽡𖽻𖾏.
4 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽙𖾐,
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡 𖽐𖼚𖽙 𖼀𖽡𖾐 𖼘𖽙𖾐 .
5 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼚𖽑𖽪𖾏 𖼞𖽻𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐,
𖼨𖽑𖽢𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼺𖽘𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙.
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼙𖽳𖾐 𖼷𖽱𖾏 𖼑𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼟𖽙𖾏 𖽐𖼚𖽠𖾏 𖽐𖼊𖽻𖾏.
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼛𖽙𖾏 𖼋𖽔 ,
𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏.
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼺𖽹𖾑 𖼗𖽪𖾏 , 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼺𖽱𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼚𖽘𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼞𖽜𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 ,
𖼟𖽪𖾏 𖼗𖽱𖾏 𖽐𖼚𖽔𖾏 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏.
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼷𖽜𖾏 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼏𖽘 𖼚𖽔𖾏 𖼉𖽺𖾏, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
11 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼑𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐.
12 𖼖𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽪𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏；
𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼅𖽔𖾏 𖼳𖽻𖾐 𖼢𖽜 .
13 𖼑𖽻 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽘𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼗𖽔 𖼐𖽑𖽡𖾏 𖼷𖽹𖾏 𖼺𖽪𖾐 𖼊𖽡𖾐 𖼋𖽘,
𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽔 𖼏𖽔 𖽐𖼎𖽙 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖼸𖽠𖾏 𖽐𖼷𖽜𖾏 𖼋𖽻𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐,
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼎𖽷𖾏 𖼋𖽻𖾏 𖼖𖽺𖾑?
15 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼀𖽻𖾐 𖼟𖽳.
16 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼡𖽑𖽜𖾐 𖼞𖽡𖾑 ,
𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
17 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼐𖽑𖽡𖾏 𖼑𖽪𖾏, 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼞𖽡𖾑 ,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻, 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼋𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼮𖽑𖽵𖾑.
18 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 𖼸𖽠𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼑𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼎𖽺𖾏 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼼𖽔𖾏 𖽃𖽡𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
19 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼷𖽑𖽡𖽝𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼷𖽑𖽡𖽝𖾏, 𖼞𖽑𖽻𖾑 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏, 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼋𖽡𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼺𖽱𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼑𖽡𖾏 𖼎𖽪𖾑 𖽃𖽔 𖼛𖽻𖾏.
20 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼷𖽷𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽔𖾑 𖼚𖽘𖾐；
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏.
21 𖼯𖽡𖽝𖾏 𖽐𖼎𖽙 𖼋𖽔 𖼊𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼡𖽔 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽷𖾏.
22 𖼊𖽻𖾏 𖼐𖽤𖾐 , 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼅𖽳𖾏 𖼋𖽘𖾏.
23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖽐𖼊𖽜 ,
𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽠𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 .
24 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼻𖽻𖾏 𖼮𖽷𖾑 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔, 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼾𖽨𖾏 𖼦𖽡 𖼋𖽙 .
𖼈𖽡𖽝𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽺 𖼗𖽔 , 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 .
