（𖼊𖽑𖽹𖾏 9：1-8；𖼖𖽪𖾏 5：17-26）
2
𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼐𖽑𖽪𖾐， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼀𖽳 𖼐𖽙𖽾 𖼙𖽻．𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽙𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑， 2 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱 ，𖼸𖽙 𖼟𖽪𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽪𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 3 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽱𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼅𖽔𖾏 𖼚𖽻𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽱𖾑 𖼐𖽱𖾏 𖼋𖽔𖾏． 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽹𖾏 𖼐𖽔𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼎𖽑𖽔𖾑 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐， 𖼎𖽑𖽔𖾑 𖼖𖽪𖾑 𖼏𖽪𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑 𖽐𖼎𖽙 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐 𖼟𖽪𖾏 𖼀𖽔𖾏 𖼀𖽱𖾏 𖽃𖽡𖾑 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏． 5 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼐𖽱𖾏， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑 𖼐𖽑𖽡𖾏：“𖼷𖽔𖾑 𖼊𖽪𖾐， 𖼟𖽡 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏 𖼋𖽘𖾏．” 6 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏： 7 “ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑． 𖼋𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏， 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼯𖽔𖾏 𖼾𖽔𖾏 𖼖𖽺𖾑？” 8 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ？ 9 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑 𖼐𖽑𖽡𖾏：‘ 𖼟𖽡 𖼾𖽔𖾏 𖼯𖽔𖾏 𖼋𖽘．’ 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：‘𖼳𖽠𖾑 𖼗𖽙𖾏 𖼯𖽡𖽙𖾏 𖼟𖽡 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐 𖼅𖽻𖾏，’ 𖼼𖽘𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼖𖽡 𖼟𖽻𖾏 𖼖𖽺𖾑？ 10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽔𖾏 𖼾𖽔𖾏．” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼞𖽑𖽻𖾑 𖼐𖽑𖽡𖾏： 11 “ 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 ，𖼳𖽠𖾑 𖼗𖽙𖾏！ 𖼯𖽡𖽙𖾏 𖼟𖽡 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡𖾑．” 12 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 ，𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐， 𖼐𖽡𖽙𖾐 𖼀𖽡𖾏 𖼙𖽡𖽝𖾏 𖽐𖼮𖽢𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 ，𖼷𖽻𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼙𖽔 𖼁𖽙𖾏 𖼺𖽷𖾐．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏
（𖼊𖽑𖽹𖾏 9：9-13；𖼖𖽪𖾏 5：27-32）
13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽪𖾏 𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏． 14 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏， 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑， 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽔𖾏 𖼖𖽳 𖼇𖽺𖾑 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “ 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 ．” 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏． 15 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖽐𖼞𖽔𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏， 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼾𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼖𖽔𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏． 16 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏， 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 ， 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼟𖽻𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏， 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏： “ 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏， 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼟𖽻𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼘𖽪？” 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼏𖽻 𖼊𖽪𖾐 𖼞𖽑𖽙𖾏 𖽐𖼷𖽘𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼽𖽔𖾑 𖼏𖽻 𖼯𖽡𖽙𖾏， 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑
（𖼊𖽑𖽹𖾏 9：14-17；𖼖𖽪𖾏 5：33-39）
18 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑， 𖼽𖽙 𖼦𖽗 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼷𖽷𖾏 𖼖𖽘𖾑． 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼽𖽙 𖼦𖽗 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑， 𖼟𖽡 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼘𖽪？” 19 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼡𖽑𖽔𖾏 𖼺𖽷𖾐， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑？ 20 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏， 𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖼊𖽻𖾏 𖼨𖽑𖽳𖾏 𖼙𖽠𖾏， 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼋𖽘𖾏．
21 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽦𖾐 𖽐𖼷𖽷𖾑 𖼮𖽑𖽙𖾏 𖼡𖽪𖾐， 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽦𖾐 𖼮𖽤𖾐 𖽐𖼮𖽑𖽷𖾏 𖼺𖽘𖾏 𖼮𖽑𖽙𖾏 𖼡𖽪𖾐，𖼚𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼚𖽔 𖼘𖽑𖽙𖾐． 22 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼘𖽑𖽱𖾏 𖼡𖽪𖾐， 𖽐𖼮𖽑𖽹𖾏 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖼷𖽑𖽔𖾐 𖼘𖽑𖽱𖾏 𖼁𖽺𖾏 𖼮𖽑𖽻𖾏 𖼙𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼘𖽑𖽱𖾏 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼺𖽘𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖼽𖽔𖾑 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼘𖽑𖽱𖾏 𖼮𖽑𖽦𖾐．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑
（𖼊𖽑𖽹𖾏 12：1-8；𖼖𖽪𖾏 6：1-5）
23 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐，𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼊𖽡𖾐 𖼳𖽻𖾐， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑 𖼷𖽷𖾏 𖼖𖽘𖾑，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽪𖾏 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐 𖼳𖽻𖾐． 24 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼑𖽔𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼘𖽪？” 25 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼋𖽘𖾏 𖼽𖽔𖾑 𖼑𖽻𖾏， 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑， 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼅𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼟𖽡𖽠 𖼺𖽷𖾐 ？ 26 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖽃𖽔𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏， 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 ， 𖼑𖽻𖾏 𖼺𖽘𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽡𖽙𖾐 𖼳𖽡𖽝𖾐， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼑𖽻𖾏， 𖼋𖽡𖾏 𖼳𖽡𖽝𖾐 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼺𖽘𖾏，𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼑𖽻𖾏．” 27 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏， 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽜 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼷𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． 28 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑．”
