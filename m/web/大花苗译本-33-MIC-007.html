7
𖼞𖽪𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 ! 𖼞𖽪𖾑 𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼳𖽙𖾑 𖼯𖽡𖽝𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼚𖽡𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼡𖽑𖽻𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼑𖽻𖾏 ,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼉𖽳 .
2 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼊𖽪𖾏 𖼮𖽘𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼚𖽺𖾏 𖼮𖽡𖽝𖾏 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑,
𖼞𖽑𖽡𖽠𖾏 𖼸𖽙𖾏 𖼗𖽙 𖼮𖽢𖾐 𖼾𖽨𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏.
𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼚𖽪𖾏
3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽡𖽠𖾏 𖼋𖽡𖾏 𖽃𖽺𖾐 𖽐𖼡𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼦𖽠𖾑 𖼉𖽳 ,
𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼞𖽡𖾑 𖼁𖽙𖾏 𖼺𖽙𖾑 𖼉𖽘𖾏.
 
𖼟𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏, 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽃𖽡𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼷𖽑𖽙𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼦𖽡𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪 .
𖼦𖽡𖾐 𖼻𖽡𖽝𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
5 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼼𖽷𖾏 𖼼𖽙 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽜 𖼻𖽡𖽝𖾏 𖼽𖽺 𖼗𖽔 ,
𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏.
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼉𖽺𖾏,
𖽐𖼷𖽑𖽹𖾏 𖼽𖽔𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐,
𖼐𖽤𖾐 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖽃𖽔𖾑 𖼁𖽙𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐.
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼽𖽔𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 .
 
8 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼘𖽪 ! 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏.
𖼞𖽪𖾑 𖼊𖽻𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽙 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼎𖽻𖾐 𖼮𖽡𖾏 .
9 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏, 𖼋𖽙𖾏 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏, 𖼑𖽜 𖼞𖽪𖾑 𖼼𖽷𖾏 𖼁𖽻𖾏.
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽐𖼚𖽙 ,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
10 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼙𖽔 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?”
𖼐𖽱𖾏 𖽂𖽪𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 , 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼎𖽺𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼐𖽱𖾏 .
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼯𖽪𖾏 𖼖𖽻𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏 𖽃𖽘𖾑 𖽃𖽔 𖼗𖽔𖾏 .
 
11 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪 ! 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏, 𖼟𖽡 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼨𖽑𖽢𖾏 𖼷𖽑𖽙𖾏 ,
𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽡 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼎𖽳𖾏 𖼇𖽘𖾑.
12 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼽𖽔𖾏 𖼳𖽪 ，
𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼑𖽡𖾏 𖼙𖽻，
𖼊𖽑𖽡𖽝𖾐 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼸𖽙 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼐𖽡𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽡𖾑,
𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑,
𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 .
13 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼑𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼑𖽪𖾏，
𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏.
