𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 , 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽐𖼷𖽜𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏. 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼏𖽙、 𖼽𖽔𖾏 𖼖𖽔 𖼏𖽙、 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼏𖽙 , 𖼊𖽑𖽡𖽝𖾐 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽙 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏, 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑. 3 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼏𖽵 𖼮𖽡𖽝𖾏, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼘𖽠𖾑 𖼋𖽠𖾏, 𖼟𖽪𖾏 𖼑𖽜 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖽃𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼾𖽔𖾏 𖼐𖽔𖾑， 4 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 𖽐𖼮𖽑𖽘𖾑, 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼯𖽔𖾏. 5 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖼑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 《𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏》𖼮𖽡𖽝𖾏. 6 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏, 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 7 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼞𖽳𖾑 𖼯𖽵𖾏, 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔𖾐 𖼟𖽱𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽨𖾏 𖼇𖽔 𖼖𖽔𖾑 𖼊𖽡𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼞𖽑𖽔𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏.
𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 36：9 - 10）
8 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼖𖽑𖽡𖾏. 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼐𖽡 𖼦𖽪𖾏 𖼳𖽷𖾑 𖼊𖽑𖽔𖾑, 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼐𖽔 𖼊𖽗𖾑 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏. 9 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼾𖽻 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑.
10 𖼊𖽑𖽻𖾏 𖼐𖽡𖾑 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼊𖽪𖾐 𖼏𖽙𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼙𖽻. 11 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼏𖽙 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼙𖽻 𖽃𖽡𖾑, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼾𖽨𖾏 𖽐𖼎𖽙 𖼋𖽔𖾏 𖼋𖽘 . 12 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐、 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏、 𖼊𖽪𖾐 𖼮𖽡𖽙𖾐 𖼨𖽑𖽳𖾏、𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽠𖾏 𖼙𖽻 , 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏. 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑. 13 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽐𖼚𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼞𖽑𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 𖼞𖽑𖽡𖽠𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼖𖽔𖾏 𖼞𖽑𖽔𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑. 14 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼋𖽻 , 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼋𖽠𖾏, 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼖𖽑𖽻𖾏 𖼞𖽑𖽔𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼞𖽪𖽙 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽡𖽝𖾏. 15 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐、 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼅𖽡、 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏、 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼞𖽑𖽔𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏， 16 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼋𖽻 𖼳𖽤𖾏 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐, 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼋𖽠𖾏 、𖼖𖽡𖾏 𖼞𖽱𖾏 𖼖𖽑𖽻𖾏, 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼋𖽻𖾏 𖽃𖽡𖾑, 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽔𖾏 𖼦𖽔𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼙𖽠𖾏. 17 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼾𖽡𖾏 𖼄𖽔𖾐 𖼊𖽑𖽔𖾏 𖼽𖽔𖾐 𖼘𖽪𖾑𖽑 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼚𖽠𖾏 𖼄𖽔𖾐 𖼊𖽑𖽔𖾏 𖼽𖽔𖾐 𖽐𖼷𖽷𖾏 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑.
𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 36：11-12；𖼽𖽦𖾑 52：1 - 3）
18 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼦𖽔𖾐 𖼄𖽙𖾏 𖼊𖽑𖽔𖾑， 𖼟𖽪𖾏𖼖𖽡 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏. 19 𖼳𖽡𖾑 𖼊𖽡𖾐 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖾏 𖼑𖽡𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏. 20 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼄𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼖𖽠𖾑 𖼋𖽠𖾏.
