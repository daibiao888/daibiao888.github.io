𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽑𖽡 𖼐𖽔 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏, 𖼟𖽪𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏. 2 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼊𖽑𖽡 𖼐𖽔 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏, 𖼟𖽪𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽡 𖼅𖽻𖾏 𖼷𖽙𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼐𖽤𖾐.” 3 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽡𖽙𖾐 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑? 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼷𖽙𖾏 𖼸𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖽃𖽡𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐 𖼖𖽺𖾑?” 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼐𖽤𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼏𖽻 𖼐𖽱𖾏.” 4 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽐𖼊𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐.
𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼊𖽻𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐
5 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽑𖽡 𖼐𖽔, 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽑𖽡 𖼐𖽔 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏, 𖼁𖽙𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏. 6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼺𖽘𖾏, 𖼺𖽱𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐. 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏. 7 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏, 𖼎𖽺𖾏 𖼙𖽔 𖼐𖽱𖾏. 8 𖼮𖽑𖽡𖽻𖾏 𖼗𖽠𖾏 𖽃𖽡𖾐 𖼷𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏, 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼽𖽔𖾑 𖼷𖽙𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼚𖽔𖾏 𖼸𖽙𖾏 𖽃𖽔 𖽐𖼊𖽪 𖼞𖽡𖾑, 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼁𖽙𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼺𖽙𖾐 𖼄𖽪𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼐𖽡𖽙𖾐 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼚𖽙. 9 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼀𖽠𖾏 𖼄𖽪𖾑 𖼯𖽡𖽙𖾏 𖼎𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼑𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼋𖽡𖾏 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽔 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼀𖽠𖾏 𖼗𖽙𖾏.
𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼐𖽑𖽡𖾏 𖼀𖽡𖾑 𖼋𖽻 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔 𖼷𖽠𖾑𖽑
10 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏. 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼗𖽙 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑. 11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼁𖽙𖾏 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑, 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼗𖽔. 12 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼀𖽡𖾑 𖼋𖽻 𖼎𖽻𖾏 𖼅𖽡 𖼷𖽠𖾑𖽑, 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼐𖽡𖾑 𖽐𖼚𖽙, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽠𖾑𖽑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼯𖽡𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼅𖽡 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼖𖽪𖾏 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙, 𖽐𖼎𖽙 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼷𖽔𖾏 𖼞𖽡𖾏 𖼛𖽔𖾏. 13 𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽠𖾑𖽑 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼅𖽡 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼖𖽪𖾏 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙, 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼷𖽔𖾏 𖼞𖽡𖾏 𖼛𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽢𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼊𖽪𖾑 𖼀𖽡 𖼋𖽻 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐.” 14 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻 𖼐𖽑𖽡𖾏:
“𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼑𖽻𖾏 𖼑𖽡 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼋𖽻𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏.”
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽠𖾑𖽑 𖼋𖽠𖾏 𖼀𖽡𖾑 𖼋𖽻 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏.
15 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼐𖽤𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼮𖽑𖽵𖾑 𖼑𖽪 𖼟𖽡 𖼈𖽻𖾑, 𖼮𖽑𖽵𖾑 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼊𖽪𖾑 𖼀𖽡 𖼋𖽻 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐, 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼾𖽡𖾏, 𖼅𖽡 𖼮𖽘𖾏 𖼀𖽡𖾐 𖼋𖽔𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼦𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 ? ” 16 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽡𖾏 𖼐𖽡𖽝𖾑 𖼘𖽑𖽡𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼊𖽔𖾐, 𖼟𖽡 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑, 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼀𖽡𖾑 𖼋𖽻 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽠𖾑𖽑, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼚𖽠𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑.” 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 𖼖𖽺𖾑?” 17 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼐𖽡𖾑 𖽐𖼚𖽙, 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽡𖾏 𖼐𖽡𖽝𖾑 𖼘𖽑𖽡𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼊𖽔𖾑 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼀𖽡𖾑 𖼋𖽻 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼚𖽠𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐, 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐. 18 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼦𖽡𖾐 𖽐𖼷𖽜𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏, 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼐𖽑𖽡𖾏:
“𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼘𖽪𖾏?
𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼘𖽪𖾏?”
𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼊𖽔𖾑 𖽃𖽔𖾏 𖼯𖽱𖾏 𖽃𖽔 𖼑𖽡𖽝 𖼑𖽨𖾏 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐,
𖼯𖽡𖾏 𖼷𖽠𖾑𖽑 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼀𖽡𖾑 𖼋𖽻 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏.”
19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼳𖽷 𖼮𖽡𖾑 𖼖𖽪𖽝 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽙𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽠𖾑𖽑 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼋𖽻 𖽃𖽡𖾑, 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏. 20 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼎𖽺𖾏 𖼡𖽔𖾏 𖼎𖽻𖾏 𖼷𖽑𖽔𖾑 𖼺𖽪𖽺𖾑 𖼑𖽡𖾏 𖼽𖽺 𖼗𖽔 𖼙𖽠𖾏, 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼀𖽡𖾏 𖼮𖽑𖽪𖾏 𖽃𖽡𖾑.
