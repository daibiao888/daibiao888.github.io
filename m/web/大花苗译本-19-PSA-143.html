𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 𖽐𖼊𖽜 𖼽𖽔𖾑 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 .
2 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼟𖽡 𖼡𖽑𖽜𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
 
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼞𖽪𖾑 ,
𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼢𖽻 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼙𖽠𖾏,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼖𖽡𖽾 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼯𖽻 𖼚𖽑𖽔𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏.
 
5 𖼞𖽪𖾑 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏, 𖼮𖽳𖾐 𖽐𖼊𖽱𖾏 𖼟𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 ,
𖽐𖼊𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏.
6 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼟𖽡,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼟𖽡, 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼋𖽙𖾏 𖼑𖽘 𖽃𖽡𖾐 𖼼𖽘𖾐. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼑𖽻𖾏 𖼊𖽡𖾐 𖼞𖽪𖾑！
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼋𖽘𖾏！
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼀𖽙 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐.
8 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑 𖼟𖽡 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 ,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼅𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼟𖽡 .
 
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑，
𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼈𖽺𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
10 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏,

11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐,
