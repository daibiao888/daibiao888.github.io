13
𖼽𖽔𖾏 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼊𖽻𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐, 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 .
2 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼮𖽢𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑,
𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼇𖽜𖾏 𖼇𖽜𖾏 𖼋𖽡𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖼐𖽱𖾏 𖽃𖽔 𖼛𖽻𖾏.
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑,
𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼋𖽻, 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑.
 
4 𖼑𖽻 𖼘𖽪𖾏 ! 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼅𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐. 𖼑𖽻 𖼘𖽪𖾏!
𖼅𖽔𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽔𖾑 𖼾𖽱 𖼸𖽻𖾏 𖼏𖽙, 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏.
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼋𖽔𖾏,
𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏,
𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼽𖽔𖾑 𖼖𖽻𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜.
 
6 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼊𖽡𖾏 𖼈𖽳𖾏 𖼋𖽘𖾏,
𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏.
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼯𖽻 𖼼𖽙 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼾𖽘𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽡𖾏 𖼮𖽑𖽡𖾏 𖽐𖼷𖽑𖽔𖾏,
𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽡𖽻𖾏 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽙𖾐 𖼐𖽑𖽤𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼊𖽪𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏, 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏.
 
9 𖼑𖽔𖾏 𖼘𖽪! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽐𖼮𖽡𖽻𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳、 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐、 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖽐𖼀𖽻𖾏 𖼚𖽠𖾑𖽑,
𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼖𖽻𖾐 𖼺𖽘𖾏 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑.
10 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽔𖾏 𖼟𖽻𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼮𖽡𖾏,
𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽂𖽪𖾏 𖼊𖽠𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼚𖽪𖾐 𖼺𖽘𖾏,
𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼮𖽡𖾏 .
11 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽜 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼨𖽑𖽢𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼑𖽜 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼨𖽑𖽢𖾐 𖼊𖽪𖾐 𖼊𖽠𖾏 𖼊𖽠𖾑.
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼑𖽡𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖽃𖽡𖾑 𖼳𖽻𖾐 𖼢𖽜 .
12 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽠 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽠 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽃𖽳 𖼇𖽺𖾑 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.
13 𖼞𖽪𖾑 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼢𖽳 𖼋𖽠𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼽𖽔𖾑 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 ,
𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 , 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼇𖽳𖾑 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾏 .
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼏𖽻 𖼖𖽠𖾑 ,
𖼺𖽱𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.
15 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼟𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏.
16 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼮𖽠𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼺𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏,
