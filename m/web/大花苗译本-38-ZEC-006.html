𖼁𖽙𖾏 𖼚𖽻𖾐 𖼖𖽪𖾐 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐
6
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼮𖽘𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼖𖽪𖾐 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼎𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏,𖽃𖽔𖾐 𖼖𖽪𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼋𖽻𖾏. 2 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼖𖽪𖾐 𖽃𖽡𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼚𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼑𖽜𖾏 𖼖𖽡𖽝𖾐, 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼖𖽪𖾐 𖽃𖽡𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼚𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼑𖽜 𖼚𖽪𖾐, 3 𖼷𖽷𖾐 𖼖𖽪𖾐 𖼖𖽪𖾐 𖽃𖽡𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼚𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐, 𖼚𖽻𖾐 𖼖𖽪𖾐 𖼖𖽪𖾐 𖽃𖽡𖾑 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼚𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼖𖽪𖾐, 𖼋𖽡𖾏 𖼑𖽜𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏. 4 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼘𖽪! 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼘𖽪?” 5 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼮𖽢𖾏, 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾏 𖼋𖽻𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏.” 6 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖼚𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼑𖽜𖾏 𖼚𖽪𖾐 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏, 𖼑𖽜𖾏 𖼚𖽠𖾐 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖼚𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼖𖽪𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼙𖽠𖾏. 7 𖼋𖽡𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼋𖽔𖾏, 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏. 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼮𖽔𖾑 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏.”𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑. 8 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑, 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼖𖽪𖾑 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼋𖽘𖾏.”
𖼑𖽜 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 10 “𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽑𖽡𖽠𖾏 𖼦𖽳 𖼊𖽹𖾏 、 𖼊𖽙𖾑 𖼀𖽡𖾐 𖼽𖽔𖾐 、 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐, 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼇𖽔𖾑 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏 𖽐𖼞𖽔𖾏, 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏. 11 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼞𖽪𖾐 𖼑𖽡𖽝𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑, 𖽐𖼊𖽻𖾏 𖼎𖽻𖾏 𖼽𖽙 𖼺𖽔𖾐 𖼊𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏. 12 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼑𖽔𖾏 𖼘𖽪! 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 “𖼊𖽪𖾑 𖼟𖽻𖾏” 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼘𖽑𖽘𖾑 𖼘𖽑𖽙𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏. 13 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏, 𖼟𖽙𖾏 𖽃𖽔𖾐 𖼼𖽘𖾐 𖼙𖽻 𖼋𖽪𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐. 14 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼖𖽡𖽝( “𖼳𖽡𖾑 𖼖𖽡𖽝”𖼟𖽪𖾏 “𖼦𖽳 𖼊𖽹𖾏”) , 𖼊𖽙𖾑 𖼀𖽡𖾐 𖼽𖽔𖾐 、 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼽𖽔𖾐 、 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼇𖽔𖾑 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽡𖽝 ( “𖼳𖽡𖽝” 𖼟𖽪𖾏 “𖼽𖽙 𖼳𖽡𖾑 𖼽𖽔𖾏” ) 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖽃𖽔𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏.”
15 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼅𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑.
