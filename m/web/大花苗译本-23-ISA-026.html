𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 𖼐𖽡𖾑:
“𖼀𖽡𖾐 𖼅𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼏𖽵 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼟𖽪𖾏 𖼞𖽻𖾏 𖼙𖽻.
2 𖼅𖽡 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼺𖽘𖾏,
𖼟𖽙𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳、 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏.
3 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑,
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 .
4 𖼅𖽡 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
5 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖽐𖼎𖽙 𖼙𖽻 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼦𖽡 𖼋𖽙 𖼺𖽘𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼙𖽻 𖼎𖽑𖽔𖾑 𖼋𖽡𖽝𖾏 𖼺𖽘𖾏, 𖼢𖽻 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 .
6 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼊𖽠𖾏 𖼯𖽪𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼟𖽪𖾏 𖼏𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼊𖽻𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼯𖽪𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.”
 
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖼮𖽵𖾑 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼘𖽪 !
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼨𖽑𖽢𖾏 𖼋𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 .
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽃𖽡𖾑 𖼋𖽙𖾏 𖼟𖽡 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼯𖽡𖽙𖾏,
𖼟𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖽃𖽡𖾑.
9 𖼋𖽘 𖼄𖽑𖽙𖾏, 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼟𖽡 ,
𖼞𖽪𖾑 𖽐𖼚𖽙 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡 𖼊𖽔𖾐 𖼋𖽡𖽝．
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖽃𖽡𖾑,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼾𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 .
10 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼈𖽡𖽝𖾏,
𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼾𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏,
𖼎𖽺𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏.
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼮𖽘𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏.
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼑𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼢𖽳 𖼋𖽠𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼳𖽙𖾑, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖽐𖼡𖽙𖾏 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏.
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 ，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽜 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼟𖽙𖾏 𖼊𖽡𖽝𖾑.
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼙𖽔 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼀𖽡𖾐,
𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖼟𖽡 𖼊𖽔𖾐, 𖼨𖽑𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼊𖽔𖾐.
14 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼯𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽠𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.
15 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼟𖽡 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼟𖽡 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼋𖽘𖾏,
𖼟𖽡 𖼎𖽺𖾏 𖼎𖽳𖾏 𖼇𖽘𖾑 𖼞𖽪𖽙 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐.
 
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ！𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡 ,
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼷𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 .
17 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑, 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼄𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
