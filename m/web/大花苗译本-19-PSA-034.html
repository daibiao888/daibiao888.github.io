𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼩𖽻𖾏, 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼸𖽙𖾏 𖽐𖼞𖽻𖾏 𖼐𖽡𖾑.
（𖼺𖽔𖾐 𖽃𖽡𖾐 21：10 - 14）
34
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
2 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
3 𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖽃𖽡𖾐 𖼸𖽻 𖼮𖽑𖽳𖾑 𖼺𖽡𖽝𖾐 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏.
 
4 𖼞𖽪𖾑 𖼙𖽔 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑,
𖼮𖽨𖾏 𖼞𖽪𖾑 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏.
5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽑𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
6 𖼞𖽪𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼐𖽡𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼑𖽻 𖼊𖽡𖾐,
𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔.
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼑𖽡𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼮𖽪𖾐 𖼾𖽡 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
8 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖽝𖾏 𖼮𖽵𖾑 𖼅𖽳𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼦𖽘𖾏, 𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏！
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼘𖽪, 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼐𖽱𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠.
10 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼸𖽠𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼮𖽑𖽹𖾐 𖽃𖽔𖾑 𖼚𖽘𖾐,
𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼦𖽡𖾐 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏.
 
11 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼅𖽡 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏！
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡.
12 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼮𖽑𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽡𖾏, 𖽐𖼮𖽑𖽠𖾏 𖼊𖽻𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐,
13 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑.
14 𖼟𖽡 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼽𖽔𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏, 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽠𖾑 𖼸𖽙𖾏.
 
15 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼐𖽱𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽜𖾑 𖼚𖽑𖽪𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽻𖾐 𖼊𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏 𖼺𖽘𖾏.
17 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔.
18 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽡𖽾 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻.
 
19 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼋𖽡𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼐𖽡𖾑，
20 𖼎𖽺𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼮𖽡𖾏 𖼙𖽱 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏,
𖽃𖽡𖾐 𖼋𖽪 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼖𖽙𖾑 𖼺𖽘𖾏.
21 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏.
22 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏.
