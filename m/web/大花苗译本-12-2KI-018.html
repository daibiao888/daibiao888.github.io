18
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽡𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼦𖽙 𖼳𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 2 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐, 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏. 3 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏. 4 𖼐𖽱𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐, 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼁𖽠 , 𖽐𖼊𖽙𖾑 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼳𖽳 𖼖𖽔𖾑, 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼋𖽻𖾐 𖼟𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽙 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼋𖽻𖾐. 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼋𖽻𖾐 𖽃𖽔𖾏 𖼁𖽠𖾏 𖼋𖽻𖾏 (𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 “𖼁𖽠𖾏 𖼋𖽻𖾏” 𖽃𖽔𖾏 “𖼐𖽡 𖼦𖽪𖾏 𖼺𖽷𖾑 𖼊𖽑𖽗𖾐，𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼗𖽪𖾏 ”). 5 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼐𖽱𖾏. 6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽔𖾑 𖼻𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼇𖽳𖾑, 𖼻𖽙𖾑 𖼏𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 . 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐, 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏. 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏. 8 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽢𖾑 𖼺𖽔𖾐 𖼊𖽑𖽡𖽝𖾐 𖼮𖽢𖾑 𖼺𖽔𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐, 𖼎𖽑𖽻𖾏 𖽐𖼊𖽑𖽘𖾐 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏.
9 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽻𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽡𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼦𖽙 𖼳𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼽𖽡𖾑 𖼺𖽳 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏. 10 𖼮𖽑𖽡𖽻𖾏 𖼗𖽠𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼙𖽻, 𖼳𖽡𖾑 𖼳𖽡𖾑 𖼮𖽢𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽙 𖼳𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 11 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼙𖽠𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼦𖽔𖾐 𖼖𖽔 𖽐𖼎𖽙 𖼞𖽙𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖼦𖽔𖾐 𖼀𖽙 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽔𖾐 𖼟𖽱𖾐 𖽃𖽻𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼄𖽔𖾐 𖼊𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼙𖽻， 12 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼗𖽪𖾏, 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼻𖽙𖾑 𖽃𖽡𖾑.
𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐
