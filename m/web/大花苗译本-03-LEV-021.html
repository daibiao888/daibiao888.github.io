𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 : 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽔 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼾𖽨𖾏 . 2 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖼑𖽜𖾏 𖼮𖽘𖾏、𖼑𖽡𖽝𖾏 𖼉𖽺𖾏、𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏、𖼞𖽱𖾑 𖼋𖽡𖾏， 3 𖽐𖼎𖽙 𖼊𖽡𖾑 𖽐𖼞𖽻𖾐 𖼞𖽱𖾑 𖼻𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼡𖽔𖾏 , 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼾𖽨𖾏 . 4 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼊𖽑𖽪 𖼇𖽻𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼾𖽨𖾏 .
5 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽑𖽳𖾏 𖼟𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼚𖽙𖾏 𖼮𖽢𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽑𖽪𖾑 𖼚𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼷𖽪𖾏 𖽃𖽔 𖼮𖽡𖾑 . 6 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 , 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 , 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽑𖽘𖾑 𖼯𖽡𖽙𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .
7 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼈𖽻𖾑 𖽐𖼎𖽻𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼐𖽤𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼖𖽑𖽘𖾑 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 . 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .
