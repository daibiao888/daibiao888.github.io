𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼳𖽔𖾑 𖼖𖽪𖽾 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏.
3
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑．
2 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑:
“𖼐𖽱𖾏 𖼊𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏.” （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
3 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐,
𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼋𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖽃𖽡𖾑.
4 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
5 𖼞𖽪𖾑 𖼢𖽻 𖼀𖽱𖾏, 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼀𖽱𖾏, 𖼞𖽪𖾑 𖼯𖽷𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑.
6 𖼅𖽔𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼋𖽔𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 𖼈𖽡𖽝𖾏,
𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏.
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏！
𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽻𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼀𖽔𖾐 𖽃𖽔𖾏 𖼯𖽹𖾏 ,
𖼮𖽻𖾏 𖽐𖼮𖽻 𖼺𖽘𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼊𖽪𖾑 𖼐𖽑𖽡𖽝𖾑.
8 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼁𖽡𖽝,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
