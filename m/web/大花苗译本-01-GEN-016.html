𖼺𖽔𖾐 𖼖𖽹 𖽐𖼎𖽙 𖼳𖽢𖾏 𖼮𖽢
16
𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼺𖽔𖾐 𖼖𖽹 𖼦𖽡𖾐 𖼑𖽜 𖼐𖽱𖾏 𖼘𖽘𖾑𖽑 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏. 𖼺𖽔𖾐 𖼖𖽹 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼳𖽢𖾏 𖼮𖽢, 𖼟𖽪𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐. 2 𖼺𖽔𖾐 𖼖𖽹 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏, 𖼮𖽑𖽙 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼀𖽱𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼊𖽪𖾐 (“𖼊𖽻𖾏 𖼊𖽪𖾐” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏”).”𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼑𖽻 𖼸𖽙𖾏𖼺𖽔𖾐 𖼖𖽹 𖼗𖽪𖾏. 3 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼑𖽡𖾏 𖼐𖽤𖾐 𖼺𖽔𖾐 𖼖𖽹 𖼄𖽔𖾐 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽢𖾏 𖼮𖽢 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖽃𖽔𖾏 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏； 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼐𖽔 𖼀𖽪𖾑 𖼟𖽻𖾏 𖼳𖽡𖽻𖾏 𖼋𖽘𖾏. 4 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖽐𖼎𖽙 𖼳𖽢𖾏 𖼮𖽢 𖼀𖽱𖾏, 𖼳𖽢𖾏 𖼮𖽢 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏； 𖼐𖽱𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑, 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼮𖽵𖾑. 5 𖼺𖽔𖾐 𖼖𖽹 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼐𖽑𖽡𖾏: “𖼾𖽱 𖼟𖽡 𖼐𖽔𖾑 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼑𖽔𖾏 𖽐𖼮𖽠𖾏, 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼐𖽱𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼎𖽺𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼞𖽪𖾑． 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼞𖽪𖾑 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽪𖾏 𖼞𖽡𖾑.” 6 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼖𖽹 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼯𖽔𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏.” 𖼺𖽔𖾐 𖼖𖽹 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼺𖽔𖾐 𖼖𖽹 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼀𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏.
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼳𖽢𖾏 𖼮𖽢
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼳𖽪𖾑 𖽃𖽳𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼷𖽠𖾑𖽑 𖽃𖽻𖾐 𖽐𖼮𖽷𖾐 𖼐𖽱𖾏， 8 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼺𖽔𖾐 𖼖𖽹 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼳𖽢𖾏 𖼮𖽢, 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏? 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏?” 𖼳𖽢𖾏 𖼮𖽢 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼺𖽔𖾐 𖼖𖽹 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼀𖽜𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏.” 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼟𖽡 𖽃𖽔 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼗𖽪𖾏.” 10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼘𖽘𖾑𖽑 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼟𖽪𖾏 𖼾𖽱 𖼋𖽘𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.”
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼽𖽔𖾑 𖼳𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐,
𖼽𖽔𖾑 𖼊𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏(“𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏” 𖼟𖽪𖾏 “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏”),
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 .
12 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏；
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐.”
13 𖼳𖽢𖾏 𖼮𖽢 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼷𖽷𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏: “𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑．” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼋𖽘𖾏 𖼘𖽪𖾏？” 14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖾑 𖽐𖼷𖽷𖾏 𖼀𖽡𖾐 𖽃𖽳𖾐· 𖼖𖽔𖾑 𖼦𖽹𖾐·𖼖𖽹 (“𖼀𖽡𖾐 𖽃𖽳𖾐· 𖼖𖽔𖾑 𖼦𖽹𖾐·𖼖𖽹” 𖼟𖽪𖾏 “𖼷𖽷𖾑 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐”). 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼊𖽡𖾐 𖼺𖽷𖾑 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
15 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼳𖽢𖾏 𖼮𖽢 𖼑𖽜 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐； 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼊𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏． 16 𖼳𖽢𖾏 𖼮𖽢 𖼑𖽜 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼳𖽻𖾏 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼅𖽔𖾏 𖼾𖽡𖾏 𖼯𖽡𖽻𖾏 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏.
