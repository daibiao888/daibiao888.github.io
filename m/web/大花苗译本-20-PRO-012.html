𖼗𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖽐𖼚𖽪𖾏
12
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏, 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼀𖽻𖾐 𖼟𖽳 ；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 , 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 .
2 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼾𖽔𖾏.
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼻𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙 .
4 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼈𖽻𖾑 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑；
𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼺𖽱𖾏 𖼷𖽑𖽘𖾐 𖼟𖽪𖾏 𖼗𖽱𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖽐𖼚𖽙 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏.
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏,
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼮𖽡𖽝𖾏 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽐𖼀𖽑𖽻𖾏 𖽐𖼀𖽙 𖼢𖽻 𖼙𖽠𖾏 𖼯𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼳𖽠𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽻𖾑,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.
9 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼈𖽡𖽝𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼅𖽔𖾏 𖼡𖽑𖽜𖾑,
𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏, 𖼈𖽡𖽝𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼡𖽜𖾑 𖼑𖽻𖾏 .
10 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽺𖾐 𖼳𖽻𖾑 𖼅𖽳𖾏 𖼾𖽨𖾏 𖼯𖽦𖾏 𖼑𖽡𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 ,
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 .
11 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽻𖾐 𖽐𖼡𖽔𖾏 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑 , 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 .
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼸𖽠𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 .
13 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏、𖼟𖽪𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏,
𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐；
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏.
15 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖽃𖽔𖾏 𖼑𖽪𖾏, 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼯𖽪𖾏 𖼑𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐.
16 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼖𖽡 𖼟𖽻𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽐𖼷𖽜𖾏 𖼋𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑.
17 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝, 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏.
18 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼦𖽡𖾐 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐, 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏, 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼞𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
19 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝, 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏；
𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼊𖽔𖾐.
20 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 , 𖼷𖽷𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏 𖼅𖽔𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼦𖽡𖾐 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 .
22 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼳𖽪𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 .
23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐, 𖼈𖽺𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼮𖽡𖽝𖾏,
