（𖼊𖽑𖽹𖾏 15：32-39）
8
𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 ，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼦𖽡 𖼸𖽱， 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏. 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽪𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： 2 “ 𖼞𖽪𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐， 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼋𖽘． 3 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽜𖾏 𖼮𖽑𖽹𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏， 𖼅𖽻𖾏 𖼉𖽺𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼸𖽠𖾏 𖽐𖼮𖽡𖾏 𖼉𖽪𖾏 𖼢𖽻，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝．” 4 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼐𖽡𖾑，𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼊𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼀𖽻𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖼖𖽺𖾑 ？” 5 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽪 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 ？” 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼮𖽡𖽙𖾐．” 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼼𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐， 𖼞𖽑𖽡𖽠𖾏 𖼳𖽤𖾏 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼋𖽘𖾏， 𖼎𖽺𖾏 𖽐𖼀𖽹𖾏 𖼮𖽡𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏， 𖼟𖽙𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼯𖽡𖽙𖾏 𖼅𖽻𖾏 𖼊𖽜𖾐， 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼎𖽺𖾏 𖼊𖽜𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐． 7 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼀𖽡 𖼯𖽠 𖼋𖽪𖾏 𖼣𖽔𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽐𖼀𖽳𖾏， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼋𖽘𖾏，𖼐𖽑𖽡𖾏：“𖼊𖽜𖾐 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼊𖽑𖽡𖽝𖾐．” 8 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏，𖼎𖽺𖾏 𖼑𖽻𖾏 𖼮𖽻𖾏 𖼋𖽘𖾏， 𖼳𖽻𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼳𖽙𖾏 𖼅𖽔𖾏 𖼳𖽤𖾏 𖽃𖽔𖾐 𖼊𖽠𖾐． 9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏． 10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏， 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼐𖽪𖾏 𖼊𖽑𖽔𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏．
𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 16：1-4）
11 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼊𖽔𖾑 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖼽𖽔𖾑 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑， 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑． 12 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏 𖼐𖽑𖽡𖾏：“ 𖼷𖽑𖽷𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑， 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼎𖽻𖾏 𖼷𖽑𖽷𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼑𖽔𖾏．” 13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼀𖽡 𖼋𖽡𖾏 𖼙𖽠𖾏．
𖽃𖽔𖾑 𖼋𖽪 𖼳𖽻𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏
（𖼊𖽑𖽹𖾏 16：5-12）
14 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼯𖽡𖽙𖾏 𖼳𖽡𖽝𖾐，𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼯𖽔 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼊𖽔𖾐， 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏． 15 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏：“ 𖼅𖽡 𖼽𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐，𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼖𖽡 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏．” 16 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼳𖽡𖽝𖾐 𖼐𖽔𖾑 ？” 17 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼳𖽡𖽝𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽡𖾐 𖼦𖽡𖾏 𖼐𖽑𖽡𖾏？ 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼯𖽷𖾏 𖼺𖽷𖾐 ？ 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐 𖼺𖽷𖾐 ？ 𖼋𖽡𖽝𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼡𖽔 𖼏𖽔 𖽐𖼎𖽙𖾏 𖼺𖽷𖾐 𖼘𖽪𖾏 ？ 18 𖼅𖽡 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼘𖽪𖾏？ 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼦𖽡𖾐 𖼐𖽙𖾑𖽑 𖼘𖽪𖾏？ 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼖𖽺𖾑？ 19 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼀𖽹𖾏 𖼀𖽜𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏， 𖼅𖽡 𖼅𖽻𖾏 𖼳𖽻𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼀𖽪𖾑 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽪𖾐 𖼈𖽘𖾐？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼈𖽘𖾐．” 20 “𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼳𖽤𖾏 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖼎𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼅𖽡 𖼳𖽻𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼀𖽪𖾑 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼊𖽠𖾐 ？”𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼊𖽠𖾐．” 21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼦𖽡𖾐 𖼚𖽠𖾐 𖽐𖼮𖽦𖾐 𖼺𖽷𖾐 𖼘𖽪𖾏？ ”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐
22 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏𖼀𖽳 𖼺𖽹 𖼊𖽔𖾏， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼋𖽔𖾏， 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏． 23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽔𖾑 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼼𖽙𖾏， 𖽐𖼊𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖽻𖾐 𖼚𖽜𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔，𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ， 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼟𖽡𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏？” 24 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼑𖽔𖾏 ，𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 ， 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑．” 25 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔， 𖼐𖽱𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽡𖾐 𖼑𖽔𖾏 ， 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼻𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜，𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑． 26 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 ， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼼𖽙𖾏 𖼐𖽡𖾑， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼼𖽙𖾏．”
𖼀𖽡𖾐 𖼊𖽳 𖼯𖽜𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪
（𖼊𖽑𖽹𖾏 16：13-20；𖼖𖽪𖾏 9：18-21）
27 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 ， 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 · 𖼇𖽺𖾑 𖼖𖽡 𖼀𖽡𖾐 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼼𖽙𖾏， 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖼐𖽱𖾏 𖼑𖽪 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏： “ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 ？” 28 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽪𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗； 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏；𖼎𖽺𖾏 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐．” 29 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 ？ ” 𖼀𖽡𖾐 𖼊𖽳 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏： “ 𖼟𖽡 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪．” 30 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾑，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
（𖼊𖽑𖽹𖾏 16：21-28；𖼖𖽪𖾏 9：22-27）
