12
𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝：𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽐𖼀𖽔𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐，𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏，𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖽐𖼊𖽻𖾏 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐． 2 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼊𖽪𖾐 𖼦𖽡 𖽐𖼀𖽱𖾏 𖼄𖽙𖾐 𖽃𖽡𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏． 3 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼚𖽔𖾐 𖼼𖽘𖾏：𖼑𖽔𖾏 𖼘𖽪！𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐， 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼋𖽪𖾏 𖼞𖽪𖾐； 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖽃𖽡𖾑 𖽐𖼊𖽻𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑． 4 𖼐𖽱𖾏 𖼊𖽪𖾑 𖼡𖽔𖾏 𖽐𖼷𖽘𖾏 𖽐𖼡𖽪𖾏 𖼨𖽑𖽹𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼷𖽷𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠，𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼨𖽑𖽹𖾏 𖼗𖽔𖾏 𖼚𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐．𖼊𖽪𖾑 𖼼𖽘𖾏 𖼐𖽡𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼾𖽨 𖼊𖽪𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼋𖽙𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐． 5 𖽃𖽔 𖼁𖽙𖾏 𖼾𖽨 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐， 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏 𖼖𖽑𖽻𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 ；𖼐𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼊𖽻𖾏 𖼮𖽑𖽳𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏． 6 𖽃𖽔 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼙𖽠𖾏，𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼊𖽪𖾑 𖼾𖽨𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐．
𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏
7 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏． 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼖𖽳 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼼𖽘𖾏 𖽃𖽡𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏，𖼊𖽪𖾑 𖼼𖽘𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼷𖽹𖾏 𖽐𖼮𖽷𖾏． 8 𖼊𖽪𖾑 𖼼𖽘𖾏 𖽐𖼊𖽻𖾏 𖼦𖽡𖾐 𖼳𖽠𖾏，𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼋𖽘． 9 𖼊𖽪𖾑 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼚𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐．𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏，𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐，𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼺𖽔𖾐 𖼊𖽔，𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼚𖽡𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼚𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐，𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼗𖽔𖾏 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏． 10 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽑𖽡𖾏：
“𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏、 𖼛𖽘𖾏 𖼼𖽙𖾏、 𖼞𖽪𖽙，
𖽐𖼎𖽙 𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏，𖼐𖽔𖾑 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡，
𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑，
𖼊𖽻𖾏 𖼗𖽔𖾏 𖽐𖼡𖽜𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏．
11 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼊𖽪𖾑 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼑𖽡𖾏 𖽐𖼮𖽑𖽘𖾑，
𖽐𖼎𖽙 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏，
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙 𖼋𖽔 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼮𖽑𖽔𖾏 𖼾𖽨𖾏 𖼳𖽦𖾑．
12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼘𖽪，
𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑，
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼦𖽡𖾐 𖼗𖽡𖾏 𖼋𖽘𖾏，
