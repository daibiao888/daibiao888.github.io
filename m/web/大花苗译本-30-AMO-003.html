𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！ 𖼅𖽡 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼨𖽑𖽳𖾏 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏 , 𖼽𖽔𖾑 𖼑𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼅𖽡 :
2 “𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼀𖽻𖾐 𖼅𖽡 𖼊𖽔𖾐,
𖽃𖽔𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.
 
3 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼅𖽻𖾏 𖼖𖽺𖾑?
4 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏 𖼖𖽺𖾑?
𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼄𖽔𖾐 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼳𖽷𖾏,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼢𖽔𖾏 𖼟𖽡𖾏 𖼤𖽻𖾏 𖼤𖽙 𖼖𖽺𖾑?
5 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼚𖽡𖾐,
𖼊𖽪𖾑 𖼐𖽻 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖼁𖽻 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼖𖽺𖾑?
𖼖𖽪𖾑 𖼸𖽙𖾏 𖼮𖽔𖾑 𖼮𖽢𖾐 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼊𖽪𖾑 𖼑𖽻 ,
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽜𖾑 𖼖𖽺𖾑?
6 𖽐𖼚𖽙 𖼙𖽻 𖼮𖽔𖾑 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 𖼖𖽺𖾑?
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼮𖽔𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 ,
𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
7 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼐𖽱𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏.
8 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏, 𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖼖𖽺𖾑?
𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼺𖽘𖾏, 𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼖𖽺𖾑?”
𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏
9 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼳𖽷 𖼊𖽑𖽪 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑
𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏:
𖼅𖽡 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐,
𖼑𖽔𖾏 𖽐𖼚𖽙 𖼙𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽑𖽙𖾑 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 .
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐,
𖼷𖽷𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽃𖽡𖾑,
𖼦𖽡𖾐 𖼀𖽻𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑,
𖼟𖽙𖾏 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼯𖽻 𖼼𖽙 ,
𖼦𖽔𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 .”
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: “𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼎𖽑𖽻𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 𖼾𖽘𖾏 𖼑𖽡𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖽃𖽔𖾑 𖼯𖽡𖾏 𖼀𖽔𖾐 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽡𖾐 𖼛𖽺𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼖𖽔𖾏 𖼮𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼯𖽡𖽙𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 𖼸𖽱𖾏 𖼸𖽘𖾏 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽪𖾐 𖼞𖽪𖾐 𖽐𖼎𖽙 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼁𖽘𖾏 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼞𖽪𖾐 𖼊𖽔𖾐.”
𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑
13 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 , 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏.
14 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽢𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽔𖾏 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼏𖽘 ,
𖽃𖽔𖾑 𖼏𖽘 𖼑𖽡𖾏 𖼞𖽪𖾑 𖼞𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽪𖾏 𖼺𖽘𖾏, 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
15 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽱𖾏 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼺𖽘𖾏,
𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼷𖽑𖽱𖾏 𖼟𖽻𖾏 𖽃𖽔𖾏,
𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
