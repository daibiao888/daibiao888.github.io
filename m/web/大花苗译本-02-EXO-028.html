“𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽔 𖼊𖽔 、𖼽𖽔𖾏 𖼀𖽡𖾐 𖼦𖽪𖾏 、𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 、𖼽𖽡𖾑 𖼊𖽑𖽔𖾑 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼟𖽡 , 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑 . 2 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼻𖽻𖾏 𖼑𖽔𖾏 . 3 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 , 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖼖𖽡𖽾 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽡𖾑 , 𖼑𖽜 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼟𖽙𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼙𖽻 𖼋𖽪𖾏 . 4 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼛𖽔𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖽐𖼎𖽙𖾐、𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳、𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 、𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼼𖽘𖾏 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏、𖼮𖽱𖾑 𖼖𖽱𖾐、𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 , 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼐𖽑𖽘𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼐𖽡𖾑 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑 . 5 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼞𖽪𖾐 𖽐𖼎𖽙 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐、𖼊𖽑𖽡𖽝𖾐 𖼺𖽙𖾑 𖼅𖽘𖾏 𖽃𖽔𖾏 .”
6 “𖼅𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼺𖽙𖾑 𖼞𖽪𖾐 𖽐𖼎𖽙 𖼺𖽙𖾐 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 , 𖼊𖽑𖽡𖽝𖾐 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 , 𖼟𖽙𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 . 7 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼸𖽙𖾐 𖼘𖽑𖽘𖾐 𖼟𖽪𖾏 𖼚𖽑𖽙𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏 , 𖼡𖽑𖽹𖾐 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼞𖽑𖽡𖽝𖾐 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 , 𖼟𖽙𖾏 𖼦𖽡𖾏 𖼊𖽡𖾏 𖼮𖽡𖽝𖾏 . 8 𖼘𖽑𖽘𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽐𖼊𖽙𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼞𖽑𖽡𖽠𖾏 𖼺𖽠𖾏 𖼊𖽡𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖼟𖽙𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼊𖽔𖾐, 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼞𖽪𖾐 𖽐𖼎𖽙 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑 、 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 、 𖼊𖽑𖽡𖽝𖾐 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼡𖽠𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 . 9 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼷𖽑𖽷𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼄𖽔𖾐 𖼐𖽻𖾐 𖼖𖽡𖽝𖾐 , 𖼞𖽳𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼷𖽷𖾏 𖼎𖽻𖾏 : 10 𖼚𖽻𖾏 𖼗𖽜𖾏 𖽐𖼷𖽷𖾏 𖼞𖽳𖾏 𖼎𖽻𖾏 𖼮𖽡𖽙𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼐𖽡𖾑 , 𖼚𖽻𖾏 𖼗𖽜𖾏 𖽐𖼷𖽷𖾏 𖼞𖽳𖾏 𖼎𖽻𖾏 𖼮𖽡𖽙𖾑 𖽃𖽔𖾐 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽡𖾑 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼘𖽑𖽙𖾐 𖼸𖽙 𖼳𖽻𖾐 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 . 11 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼞𖽳𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽳𖾏 𖽐𖼊𖽠𖾑 𖼁𖽘𖾏 , 𖼸𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼷𖽷𖾏 , 𖼞𖽳𖾏 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼮𖽡𖽙𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼐𖽡𖾑 , 𖼽𖽔𖾑 𖼘𖽙𖾏 𖼎𖽑𖽹𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼟𖽱𖾏 𖼞𖽪𖾐 . 12 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼮𖽡𖽙𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼐𖽡𖾑 𖼊𖽡𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖽃𖽔𖾐 𖼸𖽙𖾐 𖼘𖽑𖽘𖾐 𖼟𖽪𖾏 𖼚𖽑𖽙𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏 , 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖾏 , 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼽𖽔𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 𖼦𖽡𖾐 𖼁𖽱𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 13 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼸𖽙𖾐 𖼟𖽱𖾏 . 14 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽺𖾏 𖼨𖽑𖽪𖾑, 𖽐𖼀𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼀𖽱𖾏 𖼖𖽑𖽔𖾏 , 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼸𖽙𖾐 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼞𖽪𖾐 , 𖼞𖽑𖽡𖽠𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽐𖼀𖽱𖾏 𖼊𖽡𖽝𖾑 𖼐𖽡𖾑 𖼎𖽑𖽹𖾏 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼸𖽙𖾐 𖼟𖽱𖾏 𖼮𖽡𖽝𖾏 .”

31 “𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖼉𖽺𖾏 𖽐𖼚𖽙 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 , 𖼼𖽘𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼀𖽑𖽜𖾑 𖽐𖼷𖽑𖽔𖾐. 32 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼽𖽔𖾑 𖼷𖽷𖾐 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼮𖽑𖽙𖾏 , 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼮𖽑𖽙𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖼽𖽔𖾑 𖼊𖽡𖾑𖽑 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽑𖽙𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼷𖽷𖾏 𖼑𖽡𖾏 𖼦𖽡𖾑 𖼚𖽘𖾐 𖼮𖽑𖽙𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 𖼙𖽠𖾏. 33 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼐𖽡𖾑 𖼡𖽘𖾐 𖼮𖽑𖽙𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏, 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 . 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼐𖽡𖾑 𖼡𖽘𖾐 𖼮𖽑𖽙𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼖𖽪𖾑 𖼘𖽑𖽠𖾏 𖼞𖽪𖾐 : 34 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼘𖽑𖽠𖾏 𖼞𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 , 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼘𖽑𖽠𖾏 𖼞𖽪𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑, 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼐𖽡𖾑 𖼡𖽘𖾐 𖼮𖽑𖽙𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 . 35 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼅𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼐𖽑𖽘𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼐𖽡𖾑 . 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽑𖽤𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖼐𖽑𖽙𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼐𖽡𖾑 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼋𖽔 𖼺𖽘𖾏 .”
36 “𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼛𖽺𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 , 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽳𖾏 𖽐𖼊𖽠𖾑 𖼁𖽘𖾏 𖼼𖽘𖾏 , 𖼞𖽳𖾏 𖼺𖽻𖾏 : “𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼺𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .” 37 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖼸𖽙𖾐 𖼘𖽑𖽘𖾐 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼡𖽑𖽹𖾐 𖼛𖽔𖾏 𖽃𖽔𖾏 𖼸𖽡𖽝𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼐𖽡𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 . 38 𖼛𖽔𖾏 𖽃𖽔𖾏 𖼸𖽡𖽝𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼖𖽪𖾑 𖼇𖽻𖾏 𖼐𖽑𖽡𖽝𖾑 , 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼽𖽔𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼋𖽡𖾏 𖼀𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼐𖽡𖾑 , 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽡𖾑 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼛𖽔𖾏 𖽃𖽔𖾏 𖼸𖽡𖽝𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼇𖽻𖾏 𖼐𖽑𖽡𖽝𖾑 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼙𖽔 𖼷𖽹𖾏 . 39 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼺𖽙𖾑 𖼅𖽘𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼼𖽘𖾏 𖽐𖼊𖽙𖾏 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 , 𖼎𖽑𖽻𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖽃𖽔𖾏 𖼮𖽱𖾑 𖼖𖽱𖾐 , 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼁𖽘𖾏 𖽐𖼊𖽙𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 .”
