𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑
2
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼞𖽑𖽔𖾏 𖼷𖽷𖾏 𖽐𖼷𖽻𖾏 𖼋𖽘𖾏. 2 𖼸𖽙 𖼋𖽘𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏, 𖼐𖽡𖽙𖾐 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽙 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏. 3 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑, 𖼏𖽵 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽙 𖼟𖽪𖾏 𖼷𖽷𖾏, 𖼎𖽺𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼳𖽙𖾏 𖼋𖽘𖾏.
4 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐. 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑, 5 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼅𖽔𖾏 𖽐𖼡𖽜𖾐 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏, 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾐 𖼟𖽪𖾏 𖼘𖽘𖾑𖽑 𖼘𖽑𖽙𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼟𖽙𖾏 𖼑𖽘 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖼯𖽡𖽙 𖼡𖽻𖾐． 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽔𖾏 𖼦𖽘𖾑 𖼀𖽙𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖽐𖼊𖽪𖾐 𖼐𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 7 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼀𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 (“𖼊𖽜𖾑 𖼐𖽜𖾐” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼽𖽔𖾏 𖼊𖽘𖾑), 𖼄𖽔𖾐 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼀𖽘𖾏 𖼷𖽑𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖽃𖽔 𖽐𖼀𖽱, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼊𖽡𖽝𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼖𖽡𖽾. 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼊𖽘𖾑.
𖼷𖽑𖽙𖾏 𖼉𖽘𖾏 𖼗𖽙 𖼳𖽦𖾑
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼉𖽘𖾏, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽘𖾏 𖽃𖽡𖾑. 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽐𖼊𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼘𖽘𖾑𖽑 𖼘𖽑𖽙𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽺𖾐 𖼙𖽔, 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏, 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏． 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽐𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽐𖼎𖽙 𖽐𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼮𖽷𖾑.
10 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼚𖽻𖾐 𖼁𖽘： 11 𖽃𖽡𖾐 𖼁𖽘 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽡𖾐 𖼳𖽱, 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼦𖽔𖾐 𖼇𖽺𖾑 𖼖𖽔𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼞𖽪𖾐. 12 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑（“𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼛𖽙𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼦𖽘𖾏 ”） 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼄𖽔𖾐 𖼐𖽻𖾐 𖼖𖽡𖽝𖾐. 13 𖽃𖽔𖾐 𖼁𖽘 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼮𖽡𖾑 𖼳𖽱𖾏, 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼞𖽪𖾐 𖼳𖽷 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 14 𖼷𖽷𖾐 𖼁𖽘 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼊𖽡𖾐 𖼞𖽳 𖼖𖽡𖾐 𖼺𖽷𖾑, 𖽐𖼚𖽱𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼳𖽪 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐. 𖼚𖽻𖾐 𖼁𖽘 𖼟𖽪𖾏 𖼽𖽨𖾏 𖼇𖽔 𖼖𖽔𖾑 𖼊𖽡𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏.
15 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼊𖽡𖽝𖾏 𖼉𖽘𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼖𖽪𖾑 𖼉𖽘𖾏. 16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽪𖾑 𖼉𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽐𖼊𖽻𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐, 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼡𖽑𖽙𖾑 𖼑𖽻𖾏, 17 𖼮𖽔 𖼅𖽔𖾏 𖽐𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼮𖽷𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐, 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔！”
𖼑𖽜 𖽃𖽔𖾑 𖼾𖽠 𖼷𖽷𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏
18 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏:“𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼾𖽨𖾏 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼷𖽷𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼗𖽔 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏.” 19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼷𖽷𖾏 𖼊𖽡𖽝𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏,𖼨𖽑𖽳𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏. 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼯𖽦𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼷𖽷𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏. 20 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼊𖽡𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏, 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏． 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼮𖽷𖾐 𖼗𖽔 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏. 21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼀𖽱𖾏, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏, 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖽐𖼊𖽻𖾏 𖽃𖽔𖾑 𖼷𖽑𖽘𖾏 𖼖𖽡𖾏 𖼊𖽘𖾑, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼡𖽹𖾏 𖼘𖽙𖾏 𖼦𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽝𖾏. 22 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼖𖽡𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼷𖽷𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏, 𖼨𖽑𖽳𖾏 𖽐𖼮𖽡𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐. 23 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:
“𖼖𖽜𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏,
𖽐𖼡𖽹𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏,
𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏,

24 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼺𖽘𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏, 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼊𖽔𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑. 25 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑， 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽑𖽡𖽝𖾐 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
