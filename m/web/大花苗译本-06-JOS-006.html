8 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼗𖽪𖾏 , 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼯𖽡𖽙𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼞𖽪𖾐 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑 𖼮𖽑𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐. 9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 , 𖼸𖽻𖾏 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼅𖽻𖾏 , 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼷𖽑𖽙𖾑 𖼷𖽷𖾏 𖼖𖽘𖾑 . 10 𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 , 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼅𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 , 𖼋𖽙𖾏 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 , 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 .” 11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼦𖽡 𖽐𖼮𖽡 𖼙𖽻 , 𖼦𖽡 𖽐𖼮𖽡 𖼙𖽻 𖽃𖽡𖾐 𖼟𖽻𖾐 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼾𖽡 𖽐𖼚𖽙, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼾𖽡 𖽐𖼚𖽙 𖼳𖽙𖾏 𖼀𖽱𖾏 .
