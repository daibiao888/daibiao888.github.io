5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 , 𖼖𖽪𖾑 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏 𖼟𖽻𖾏 𖼚𖽻𖾐 𖼐𖽑𖽪𖾐 𖼢𖽹𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽡𖾑 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏 .
6 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼋𖽔 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏. 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽔𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 . 7 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖼑𖽜 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼋𖽔 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏 , 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 , 𖼟𖽙𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 , 𖼖𖽑𖽘𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼖𖽺𖾑 ?” 8 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼋𖽙𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 .”
9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 10 “𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖽐𖼎𖽙 𖼅𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽜 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼋𖽔 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 , 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏. 11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖽃𖽔𖾐 𖼖𖽑𖽡𖾏 𖼟𖽻𖾏 𖼚𖽻𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼢𖽹𖾐 𖽃𖽡𖾑 , 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 . 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 𖽐𖼎𖽙 𖼻𖽻𖾐 𖽃𖽦𖾐 , 𖽐𖼎𖽙 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑 𖼾𖽘𖾏 𖼁𖽜𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼑𖽻𖾏 . 12 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼸𖽙 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 , 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽙𖾑 𖼺𖽘𖾏 . 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼗𖽙𖾏 𖼻𖽙𖾑 . 13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖽃𖽡𖾑 , 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼾𖽡𖾏 𖼙𖽳𖾏 𖼦𖽡𖾐 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼏𖽵 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼖𖽑𖽘𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 , 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼐𖽱𖾏 𖼾𖽔𖾏 .

15 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼊𖽡𖽝𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼀𖽙 𖼈𖽳𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 , 𖼟𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 . 𖼎𖽑𖽻𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼸𖽙 𖼛𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 , 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 . 16 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼀𖽙 𖼈𖽳𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 , 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 , 17 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼳𖽻𖾏 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼅𖽻𖾏, 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼮𖽪𖾐 𖼾𖽡 . 18 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽻𖾏 , 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼮𖽪𖾐 𖼾𖽡 . 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼳𖽔𖾐 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼀𖽡𖾑 𖼯𖽠 ,𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼀𖽡𖾐 𖼯𖽠 ． 19 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼻𖽙𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼦𖽡𖾐 𖼳𖽠𖾑 𖼅𖽻𖾏 . 20 𖼅𖽔𖾏 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼀𖽡𖾑 𖼯𖽠 𖼐𖽑𖽪𖾐 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼾𖽡 , 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼳𖽠𖾑 𖼅𖽻𖾏 . 21 𖼅𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼎𖽑𖽻𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 , 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼳𖽻𖾏 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼅𖽻𖾏 . 𖼅𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼖𖽔𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 , 𖼳𖽻𖾏 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼅𖽻𖾏. 22 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼮𖽳 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖽝𖾏 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽑𖽪𖾐 , 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽑𖽡𖾏, 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼾𖽡 𖽐𖼚𖽙 𖼦𖽡𖾐 𖼳𖽠𖾑 𖼅𖽻𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼳𖽻𖾏 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼅𖽻𖾏. 23 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼮𖽪𖾐 𖼾𖽡 , 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼳𖽠𖾑 𖼅𖽻𖾏 . 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽙𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 .
