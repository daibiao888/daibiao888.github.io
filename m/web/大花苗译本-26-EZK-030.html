𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 2 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖼐𖽑𖽡𖾏: 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：
𖽃𖽺𖾐! 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑, 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼐𖽡𖽝𖾑 .
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼊𖽡𖾏 𖼈𖽳𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼊𖽡𖾏 𖼈𖽳𖾏,
𖼟𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼚𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑,
𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑.
4 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽔𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏.
𖼣𖽹 𖼮𖽡 𖼑𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪𖾏 𖼷𖽑𖽷𖾐 𖼊𖽻𖾏 𖼦𖽔𖾏,
𖼊𖽠𖾏 𖽐𖼞𖽔𖾏 𖼊𖽻𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏.
5 𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐、 𖼇𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼖𖽪𖽝𖾐 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐”)、 𖼖𖽪𖾏 𖼊𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐、 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾐 𖼀𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼊𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖼣𖽹 𖼮𖽡 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐.” 6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼢𖽻 𖼙𖽠𖾏,
𖼣𖽹 𖼮𖽡 𖼑𖽜 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼐𖽔𖾑 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾏 𖼢𖽜𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼎𖽑𖽻𖾏 𖼄𖽡 𖼊𖽙 𖼸𖽙 𖼺𖽳 𖼇𖽪 𖼐𖽡 , 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
7 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖾐 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖼊𖽡𖾐 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼟𖽙𖾏 𖼣𖽹 𖼮𖽡 𖼙𖽻 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏.
8 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼳𖽠𖾑 𖼯𖽡,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽡𖾑, 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
9 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽡𖽙𖾐 𖽐𖼞𖽙𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏, 𖼟𖽙𖾏𖼞𖽪𖾐 𖼳𖽷 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽳𖾏 𖽃𖽡𖾑 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼺𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾐 𖼼𖽘𖾐. 𖼑𖽔𖾏 𖼘𖽪！ 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼋𖽘𖾏! 10 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏
𖼖𖽻𖾐 𖼺𖽘𖾏 𖼣𖽹 𖼮𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐.
11 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐,
𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼁𖽺𖾏 𖼋𖽔𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽻𖾏 𖼣𖽹 𖼮𖽡 ,
𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼀𖽪𖾑 𖽐𖼷𖽡𖽝𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏.
12 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼋𖽺 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼊𖽠𖾏 𖼊𖽠𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽡𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
 
13 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼁𖽠𖾏 𖼺𖽘𖾏,
𖼎𖽑𖽻𖾏 𖼐𖽙 𖼇𖽪 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼁𖽠𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏.
14 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽔𖾑 𖼊𖽑𖽳 𖼘𖽙 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼣𖽔𖾑 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼯𖽡,
𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼎𖽻𖾏 𖼐𖽙 .
15 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼳𖽱 𖼟𖽪𖾏
𖼣𖽹 𖼮𖽡 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼐𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐.
