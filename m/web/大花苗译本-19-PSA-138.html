𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡 ,
𖼐𖽡𖽙𖾐 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 .
2 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏,
𖼑𖽜 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖽐𖼷𖽷𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼗𖽪𖾏 𖼢𖽳 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼮𖽑𖽡𖽻𖾏 𖼇𖽔𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 .
3 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 , 𖼟𖽡 𖼎𖽺𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑, 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 ,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.
 
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼟𖽡 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐, 𖼈𖽡𖽝𖾏 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼯𖽡𖽙𖾏,
𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼑𖽔𖾏 𖼋𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏.
 
7 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼈𖽡𖽝𖾏, 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐.
𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼞𖽪𖾑 .
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼞𖽪𖾑 𖼑𖽪𖾏 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽷𖾐 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏！
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑.
