𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏
65
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏.
𖼸𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑! 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 !”
2 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼮𖽘𖾏 𖼋𖽡𖾏 𖼨𖽑𖽪𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏.
3 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽙𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼉𖽘𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 ,
𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼑𖽡 𖼋𖽺𖾏 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 .
4 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽔𖾑 𖼮𖽡𖽺𖾏 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 ,
𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 ,
𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖽐𖼀𖽔𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼋𖽺𖾏 𖽐𖼚𖽙 𖼅𖽔𖾏 𖼞𖽔𖾏 𖼞𖽔𖾏 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼳𖽠𖾑 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼙𖽠𖾏 !
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽡 .”
𖼮𖽵𖾑 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖽐𖼀𖽱 𖼑𖽡𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏,
𖼟𖽪𖾏 𖼯𖽘𖾏 𖼋𖽠𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼯𖽡 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑.
6 𖼑𖽔𖾏 𖼘𖽪 ! 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐．
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏,
7 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼅𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼟𖽪𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏 ,
𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼞𖽪𖾑,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑 𖽐𖼮𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐.
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.”
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼡𖽑𖽻𖾏 𖼼𖽡𖽝𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐．
𖼞𖽪𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏,
𖼦𖽡𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏.
9 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽳𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼎𖽑𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑.
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏,
𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑.
10 𖼳𖽔𖾑 𖼖𖽪𖽝 𖼑𖽡𖾏 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼾𖽘𖾐,
𖼽𖽔𖾏 𖼞𖽙 𖼖𖽪𖾑 𖼦𖽘𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙𖾏 𖼑𖽨𖾐 𖼳𖽙𖾏 𖼀𖽱𖾏,
𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏.
11 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑, 𖼑𖽜 𖼮𖽢𖾑 𖼊𖽳 (“𖼮𖽢𖾑 𖼊𖽳” 𖼟𖽪𖾏 “𖼻𖽻𖾏 𖼞𖽡𖾑 𖼙𖽻𖾏”) 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏,
𖼑𖽜 𖼄𖽡𖾐 𖼐𖽡(“𖼄𖽡𖾐 𖼐𖽡”𖼟𖽪𖾏 “𖼳𖽦𖾑 𖼞𖽑𖽡 𖼙𖽻”)𖼘𖽡𖾑𖽑 𖼀𖽪𖾑 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏,
12 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼎𖽙𖾑 𖼚𖽔𖾏 𖽃𖽔𖾏 𖼐𖽑𖽡𖽝𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐,
𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑 𖼊𖽻𖾏 𖼊𖽔𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏, 𖼅𖽡 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼊𖽡𖾐,
𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖼅𖽡 𖼦𖽡𖾐 𖼑𖽻 .
𖼎𖽙𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼺𖽹𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽡𖾑.”
 
13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:
“ 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼑𖽻𖾏, 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽜𖾏 𖼮𖽑𖽹𖾐,
