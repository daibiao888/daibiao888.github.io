𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽻𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏
（𖼀𖽳 22：1 - 27：23）
22
𖼊𖽑𖽡 𖼄𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 :
2 “ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼖𖽺𖾑 ?
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼮𖽔𖾑 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 .
3 𖼟𖽡 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 , 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼖𖽺𖾑?
𖼟𖽡 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼖𖽺𖾑 .
4 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 ,
𖼐𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼟𖽡 ,𖼎𖽺𖾏 𖼨𖽑𖽢𖾐 𖼟𖽡 𖼘𖽪 ?
5 𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼘𖽪 ?
𖼟𖽡 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 .
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼦𖽔𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 ,
𖼚𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼙𖽠𖾏 .
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼚𖽑𖽔𖾏 , 𖼟𖽡 𖼦𖽡𖾐 𖼄𖽔𖾐 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽻𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 , 𖼟𖽡 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 .
8 𖼟𖽡 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼈𖽡𖽝𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 .
9 𖼟𖽡 𖼟𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼙𖽠𖾏 ,
𖼖𖽙𖾑 𖼺𖽘𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 .
10 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼟𖽡 ,
𖼅𖽔𖾏 𖼊𖽳𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 .
11 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼅𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼟𖽡 , 𖼟𖽙𖾏 𖼟𖽡 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 ,
𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼟𖽡 .
 
12 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼘𖽪 ?
𖼟𖽡 𖼑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 , 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 !
13 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼐𖽑𖽡𖾏 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼀𖽻𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 ?
𖼋𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽔𖾏 𖽐𖼎𖽑𖽹𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼖𖽺𖾑 ?
14 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽙𖾏 ,
𖼐𖽱𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼘𖽙𖾏 .
15 𖼟𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼘𖽪𖾏 ?
𖼸𖽙𖾏 𖼞𖽡𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼅𖽻𖾏 𖼊𖽔𖾐 .
16 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼋𖽔 , 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 , 𖼺𖽱𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖽐𖼊𖽱𖾑𖽑 𖼙𖽠𖾏 .
17 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : 𖼀𖽜𖾑 𖼀𖽡𖾐 𖼙𖽠𖾏,
𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼸𖽠𖾏 𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑 ?
18 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 ,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼐𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼡𖽔𖾏 𖼞𖽪𖾑 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .
19 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼙𖽔 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼚𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 ,
20 𖼐𖽑𖽡𖾏 : 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾐 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼳𖽙𖾏 , 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 .
𖼽𖽡𖾑 𖼖𖽡𖾏 𖼇𖽔 𖼞𖽑𖽻𖾑 𖼽𖽙 𖼀𖽳 𖼟𖽙𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
21 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 ,
𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 .
22 𖼟𖽡 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 ,
𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 .
23 𖼟𖽡 𖼮𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 ,
𖼎𖽑𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 .
24 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽘𖾏 ,
