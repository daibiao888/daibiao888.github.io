𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏,
𖼮𖽨𖾏 𖼎𖽙𖾑 𖼽𖽔𖾐 𖼞𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏.
2 𖼟𖽡 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,

3 𖼟𖽡 𖼳𖽻𖾏 𖼎𖽙𖾑 𖼺𖽘𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼦𖽡𖾐 𖼋𖽡𖾐 𖼺𖽘𖾏 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
 
4 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 ,
𖼟𖽙𖾏 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽃𖽡𖾑 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏.
5 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽳𖾑𖽑 𖼸𖽙 𖼉𖽻𖾏 𖼷𖽑𖽷𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
6 𖼟𖽡 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼀𖽡𖾐 𖼮𖽨𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 ,
𖼟𖽙𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼻𖽡𖽝𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼙𖽔 𖼚𖽙𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏,
𖼎𖽺𖾏 𖼄𖽔𖾐 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐.
 
8 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽻 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼏𖽵 𖼄𖽔𖾐 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼺𖽷𖾐 𖼅𖽻 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔 𖼏𖽔 .
9 𖼐𖽱𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 ,
𖼟𖽙𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.
 
10 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼅𖽻𖾏 𖼦𖽡𖾐 𖽐𖼮𖽷𖾐,

11 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼘𖽑𖽘𖾑 𖼋𖽠𖾏,

12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,

13 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
