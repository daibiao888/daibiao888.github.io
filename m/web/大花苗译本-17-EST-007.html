𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽳𖾏 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼋𖽔𖾏 , 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖽃𖽡𖾐 𖼸𖽻 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 . 2 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼐𖽑𖽡𖾏 : “𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼘𖽪 ！ 𖼟𖽡 𖼖𖽔𖾑 𖼽𖽔𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 , 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼊𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼟𖽪𖾏 𖼞𖽪𖽙 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 , 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 .” 3 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽔𖾑 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼽𖽔𖾑 𖽃𖽡𖾑 𖼟𖽪𖾏：𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 ； 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽐𖼊𖽜 𖽃𖽡𖾑 𖼟𖽪𖾏 ： 𖽐𖼊𖽜 𖼟𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 . 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽻𖾏 𖼋𖽺 𖼺𖽘𖾏 𖼋𖽘𖾏 , 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 , 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 , 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽪𖾏 𖼮𖽘𖾑 𖼺𖽘𖾏 . 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼋𖽺 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾑 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼊𖽔𖾐 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽔𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼮𖽑𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 . ” 5 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽪𖽺 𖼖𖽪𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼐𖽑𖽡𖾏 : “𖼷𖽷𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼞𖽔𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 ? 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑 ?” 6 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼐𖽑𖽡𖾏 : “𖼷𖽷𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼸𖽜 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 !” 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽳𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 .
𖼦𖽔𖾐 𖼄𖽔𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏
7 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 , 𖼚𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼙𖽠𖾏 . 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼊𖽑𖽡𖽝𖾐 𖼾𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼮𖽨𖾏 𖼳𖽦𖾑 . 8 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 , 𖼁𖽙𖾏 𖼦𖽔𖾑 𖼄𖽔𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼸𖽱𖾏 𖼸𖽘𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼼𖽻𖾏 𖽃𖽡𖾐 𖽃𖽡𖾑 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “ 𖼐𖽱𖾏 𖼞𖽔𖾐 𖼐𖽡𖽙𖾐 𖼟𖽻 𖽐𖼚𖽙 、 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼘𖽪𖾏 ?” 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖽂𖽪𖾏 𖼋𖽠𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽐𖼀𖽙 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏. 9 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖽐𖼷𖽷𖾏 𖼦𖽔𖾐 𖼀𖽙𖾑 𖼐𖽔 , 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼦𖽔𖾐 𖼄𖽔𖾏 𖼑𖽜 𖼷𖽷𖾑𖼄𖽙 𖼊𖽡𖾐 𖼞𖽹𖾐 𖼟𖽪𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼻𖽻𖾏 𖼗𖽪𖾏 𖼮𖽨𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼋𖽠𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏 , 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼷𖽑𖽙𖾏 𖼎𖽻𖾏 𖼦𖽔𖾐 𖼄𖽔𖾏 𖽐𖼞𖽔𖾏 𖼮𖽡𖽝𖾏 .” 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽑𖽡𖽠𖾏 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼮𖽘𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 .” 10 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼦𖽔𖾐 𖼄𖽔𖾏 𖼮𖽘𖾏 𖼎𖽻𖾏 𖼋𖽠𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽜 𖼄𖽙 𖼊𖽡𖾐 𖼞𖽹𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼻𖽻𖾏 𖼺𖽘𖾏 .
