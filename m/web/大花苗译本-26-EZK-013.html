𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 2 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖽐𖼷𖽜𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏: 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏. 3 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼾𖽨𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏. 4 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪！ 𖼟𖽡 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼊𖽪𖾑 𖼞𖽡𖾏 𖼋𖽱𖾏. 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖼷𖽑𖽠𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼎𖽺𖾏 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻, 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽠𖾏 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼋𖽻. 6 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼮𖽔𖾑 𖼺𖽙𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑. 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼸𖽪𖾏. 7 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼖𖽺𖾑? 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼮𖽔𖾑 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼘𖽪? 𖼅𖽡 𖼐𖽑𖽡𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏, 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏.”
