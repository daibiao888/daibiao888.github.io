𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 ,𖼸𖽙𖾏 𖼡𖽔𖾏 𖼳𖽡𖾏 𖼞𖽑𖽡𖽠𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼋𖽪 𖼮𖽡𖾑 𖼀𖽜𖾏 , 𖼟𖽡 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼷𖽷𖾏 𖼎𖽻𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏， 3 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼺𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼎𖽻𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 ， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼮𖽡𖾑 𖼀𖽜𖾏 . ” 4 𖼟𖽡 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼚𖽙 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 ， 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼁𖽙𖾏 𖼚𖽑𖽪𖾏 . 5 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑 , 𖼐𖽱𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖽃𖽔𖾑 𖽐𖼷𖽔 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼎𖽻𖾏 𖼅𖽡 𖽃𖽡𖾑 𖽐𖼮𖽦 𖼙𖽠𖾏, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 .”
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼮𖽡𖾑 𖼀𖽜𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼡𖽔𖾏 𖼳𖽡𖾏 , 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖽃𖽡𖾐 𖼋𖽪 𖼮𖽡𖾑 𖼀𖽜𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼋𖽪 , 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 . 7 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖾑 𖼀𖽜𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖽐𖼚𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 .
8 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖽐𖼚𖽙 𖼉𖽺𖾏 𖽃𖽡𖾑 , 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 , 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼖𖽑𖽘𖾑 𖽃𖽔𖾑 𖽐𖼷𖽔、𖼋𖽠𖾏 𖼁𖽘𖾏 , 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖼞𖽑𖽡𖽠𖾐 𖼟𖽪𖾏 𖼳𖽦𖾑 𖼋𖽘𖾏 . 9 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽡𖾑 𖼀𖽜𖾏 , 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼾𖽨𖾏 𖼮𖽡𖾑 𖼁𖽜𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 .
10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽑𖽡𖽠𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼯𖽡𖽙𖾏 𖼺𖽷𖾐 𖼅𖽻 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖽐𖼮𖽦 𖼙𖽠𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼛𖽙𖾏 𖼋𖽔 .” 11 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 . 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏 . 12 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼀𖽡𖾐 𖼽𖽔𖾑 𖼋𖽔 𖼋𖽘𖾏 ! 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 𖼋𖽘𖾏 ! 𖼞𖽑𖽔𖾏 𖼛𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼋𖽘 ! 13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔 , 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔 𖼋𖽘𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?”
