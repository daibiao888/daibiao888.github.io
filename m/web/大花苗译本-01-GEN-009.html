𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽙 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐. 2 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏、𖼀𖽡𖾑 𖼞𖽘𖾐 𖼟𖽪𖾏 𖽐𖼞𖽘 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐、𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑, 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼅𖽡, 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖼄𖽔𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼞𖽳𖾑 𖼯𖽵𖾏. 3 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼖𖽔𖾏 𖼯𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼅𖽡 𖼡𖽜𖾑 𖼑𖽻𖾏, 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖾑 𖼞𖽑𖽔𖾏 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼑𖽻𖾏 𖼻𖽻𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐. 4 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽐𖼮𖽑𖽘𖾑, 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽘𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑. 5 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼅𖽡 𖽐𖼮𖽑𖽘𖾑 、𖼦𖽺𖾏 𖼅𖽡 𖼳𖽦𖾑 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼾𖽱 𖼐𖽱𖾏 𖼾𖽔𖾏； 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼾𖽨𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏. 6 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼚𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼚𖽱𖾏. 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏. 7 𖼅𖽡 𖼽𖽔𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼳𖽤𖾏 𖼮𖽘𖾑 、𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.”
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽙 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏

20 𖼐𖽙 𖼽𖽔𖾏 𖼳𖽠𖾑 𖼁𖽱𖾏 𖽃𖽔𖾏 𖼡𖽻𖾐, 𖼐𖽱𖾏 𖼯𖽡𖽙 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏. 21 𖼐𖽱𖾏 𖼦𖽻𖾏 𖼷𖽷𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼮𖽡𖽠𖾑 𖼢𖽻 𖼮𖽡𖽠𖾑 𖼋𖽘𖾏, 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏. 22 𖼮𖽢𖾑 𖼐𖽔 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼦𖽗, 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏, 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏. 23 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏𖼳𖽔𖾐 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼇𖽪 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾑 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏, 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼁𖽺𖾏 𖼅𖽻𖾏，𖽐𖼀𖽙 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼮𖽡𖽝𖾏； 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖼙𖽠𖾏 𖼟𖽙𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏. 24 𖼮𖽡𖽠𖾑 𖼟𖽳𖾏 𖼋𖽘𖾏, 𖼐𖽙 𖼽𖽔𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑. 25 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:
“ 𖼮𖽢𖾑 𖼐𖽔 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏,

26 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽔𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔!

27 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼽𖽔𖾐 𖼇𖽪 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼎𖽳𖾏 𖼇𖽘𖾑,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽔𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼚𖽙,
𖼎𖽺𖾏 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼮𖽢𖾑 𖼐𖽔 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑.”
28 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼐𖽙 𖼽𖽔𖾏 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏. 29 𖼐𖽙 𖼽𖽔𖾏 𖼯𖽡𖽝𖾏 𖼊𖽻𖾏 𖼯𖽢𖾏 𖼀𖽔𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏.
