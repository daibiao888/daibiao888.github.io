42
“𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏,

2 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏, 𖼦𖽡𖾐 𖼮𖽘𖾏 𖼺𖽘𖾐,

3 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖼟𖽪𖾏 𖼖𖽙𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽪𖾏, 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼖𖽙𖾑 𖼊𖽪𖾏 𖼺𖽘𖾏,
𖽃𖽔𖾏 𖼐𖽑𖽨𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼊𖽔𖾐 𖼡𖽔𖾏 𖼮𖽠𖾑 𖼡𖽔𖾏 𖼮𖽠𖾑, 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼇𖽳𖾏 𖼺𖽘𖾏,

4 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼀𖽑𖽪𖾑 𖼺𖽡𖽝𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼛𖽙𖾏 𖼚𖽔𖾐,

𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏.”
 
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 — 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽷𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼊𖽑𖽔𖾑 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼮𖽡𖽝𖾏,
𖼊𖽑𖽔𖾑 𖼮𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽠𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑,
𖼺𖽘𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼀𖽘𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼎𖽺𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼖𖽡𖽾 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑:
“ 6 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼨𖽑𖽪𖾏 𖼟𖽡 ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽤𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 ,
𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖽃𖽔𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼎𖽻𖾐,
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡𖾏 𖼽𖽔𖾑 𖼡𖽑𖽜𖾐 𖼐𖽔𖾐 𖼄𖽱𖾐 𖽃𖽔𖾑 𖼅𖽔 ,
𖼨𖽑𖽳𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾐 𖼞𖽱𖾏 𖼮𖽡𖽝𖾏 𖼋𖽠𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏,
𖼨𖽑𖽳𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏.
8 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼞𖽪𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏.
9 𖼑𖽔𖾏 𖼘𖽪 ！ 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼙𖽔 𖽐𖼊𖽔𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼑𖽪𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏,
𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽠𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼞𖽪𖾑 𖼺𖽪𖾐 𖼐𖽑𖽡𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼑𖽻 .”
𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽱𖾏 𖽐𖼞𖽙𖾏 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼯𖽪𖾏 𖼀𖽪𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑,
𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼘𖽪 !
𖼅𖽡 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼮𖽑𖽦𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 .
11 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽐𖼎𖽙 𖼙𖽻 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏
𖼊𖽑𖽡𖽝𖾐 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐，𖼽𖽔𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼖𖽔𖾑 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏,
𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏.
12 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽡𖽙𖾐 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏.
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽙𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼋𖽙𖾏 𖼮𖽡𖽝𖾏,

14 𖼞𖽪𖾑 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼗𖽡𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏,

15 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖾐 𖼡𖽔𖾏 𖼁𖽙𖾏,

𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼨𖽑𖽹𖾑 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼷𖽑𖽪𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑,

16 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼅𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼅𖽻𖾏,
𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽙𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑,
𖼟𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖽐𖼞𖽑𖽻𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏,
𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏.
17 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼊𖽺𖾐 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏,
𖽐𖼎𖽙 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼘𖽡𖾑𖽑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.”
𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼟𖽪𖾏 𖼘𖽙𖾐 .
𖼖𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜
18 𖼅𖽡 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖽃𖽔𖾐 𖼗𖽘𖾐 𖼐𖽡𖾑，𖼑𖽻 𖼘𖽪！
𖼅𖽡 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼐𖽡𖾑，𖼑𖽔𖾏 𖼘𖽪！
𖼟𖽙𖾏 𖼅𖽡 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏.
19 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼖𖽺𖾑?
𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾐 𖼗𖽘𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼖𖽺𖾑?
𖼡𖽔 𖼋𖽱𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼺𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏 𖽃𖽡𖾑?
𖼡𖽔 𖼋𖽱𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼺𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼖𖽺𖾑?
20 𖼟𖽡 𖼑𖽔𖾏 𖼀𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼈𖽡𖽝𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼐𖽙𖾑𖽑.
 
21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽜 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖽃𖽺𖾐 𖼙𖽔 𖼞𖽑𖽡𖽠𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 (“𖼼𖽘𖾏 𖼻𖽡𖽝𖾑”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 ”) 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏.
22 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼯𖽡𖽙𖾏,
𖼟𖽪𖾏 𖼞𖽱𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼮𖽡𖽝𖾏,𖼈𖽺𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖽐𖼚𖽙 𖼮𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏,
𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 “𖼽𖽔𖾑 𖼎𖽙𖾑 𖼁𖽻𖾏 ” .
23 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼯𖽪𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 𖼋𖽡𖾏 𖼐𖽡𖾑 ,
𖼡𖽔 𖼋𖽱𖾏 𖼯𖽪𖾏 𖽐𖼷𖽜𖾏 𖼺𖽡𖽝𖾐 𖼑𖽻, 𖼟𖽙𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼗𖽙𖾏 𖼖𖽺𖾑?
24 𖼡𖽔 𖼋𖽱𖾏 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏,
𖼄𖽔𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏 𖽃𖽡𖾑 𖼖𖽺𖾑?
𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏.
25 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼄𖽔𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽐𖼎𖽙 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑，
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼳𖽠𖾑 𖼯𖽡,
𖼐𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼺𖽷𖾐, 𖼷𖽑𖽷𖾑 𖼏𖽻 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
