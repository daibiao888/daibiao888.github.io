3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼷𖽔𖾏, 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼾𖽡 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑,

4 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼚𖽺𖾏 𖼊𖽑𖽔𖾑 𖼮𖽡𖽝𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑.
5 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼘𖽪！ 𖼟𖽡 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏.
 
6 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼊𖽹𖾐 𖼁𖽠𖾏,

7 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖼁𖽙𖾏 𖼞𖽪𖾑 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏,
𖼀𖽻𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽔𖾑.
8 𖼟𖽡 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼞𖽪𖾑 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,

9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽜 𖼟𖽪𖾏 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏 𖼐𖽔𖾑 𖼇𖽔 𖼙𖽻 𖼺𖽘𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼗𖽙𖾏 𖼳𖽦𖾑.
10 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽜 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏 𖼐𖽔𖾑 𖼞𖽑𖽔𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏,
𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖽐𖼷𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼾𖽘𖾏 𖼙𖽠𖾏．
𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼾𖽔𖾏 𖼯𖽻 𖼼𖽙,

11 𖼞𖽪𖾑 𖼑𖽜 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼞𖽪𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼼𖽷𖾏 𖼼𖽙 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽂𖽪𖾏 𖽂𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖽃𖽡𖾑 𖼖𖽔𖾏 𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼀𖽜𖾑 𖽐𖼮𖽠𖾏 𖼞𖽪𖾑.
12 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏,
𖼞𖽪𖾑 𖼺𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽔 𖼙𖽠𖾏.
13 𖼞𖽪𖾑 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑,
𖼚𖽻𖾐 𖼺𖽘𖾐 𖼅𖽔𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼦𖽡𖾏 𖼊𖽳𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼞𖽪𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖽐𖼊𖽱𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑．
 
14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！

15 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼞𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,

17 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,

19 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽡, 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡,

𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽪𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝.
20 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼈𖽺𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼈𖽺𖾏 𖼻𖽻𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼊𖽑𖽡𖽝𖾐,
𖼟𖽡 𖼈𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖽐𖼚𖽙,
𖼟𖽙𖾏 𖼦𖽡𖾐 𖽐𖼮𖽷𖾐 𖼏𖽻 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾏 𖽐𖼚𖽺𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼦𖽡𖾏 𖼊𖽔𖾏.
 
21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖽐𖼚𖽙, 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑.
22 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼮𖽡𖽺𖾏 𖼡𖽔𖾏 𖼺𖽘𖾏.”
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏, 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏.
 
23 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐,
𖼈𖽡𖽝𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝𖾏.
24 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼚𖽔𖾐, 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
