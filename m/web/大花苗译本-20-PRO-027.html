4 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼘𖽑𖽙𖾐；
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖽐𖼎𖽳𖾏 𖼋𖽻𖾏 𖼖𖽺𖾑?
5 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 ,
𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐，𖼈𖽡𖽝𖾏 𖼋𖽠𖾏 𖼉𖽺𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝；
𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼟𖽡 𖼯𖽪 𖼯𖽪𖾏 𖼡𖽑𖽻𖾐 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖼮𖽻𖾏 𖼺𖽘𖾏, 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼑𖽘 𖼄𖽪𖾑 𖽐𖼞𖽔𖾏 𖼊𖽪𖾏 𖼳𖽪𖾐；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽹𖾐 𖼋𖽘 , 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖽃𖽦𖾐 𖼖𖽔𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼡𖽘𖾐 𖼄𖽪𖾑.
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼇𖽳𖾑 𖽐𖼞𖽔𖾏 𖼺𖽘𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼳𖽔𖾏 𖽐𖼮𖽡 ,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼖𖽪𖾑 𖼉𖽳𖾏 𖼺𖽘𖾏 𖼡𖽑𖽙𖾑 𖼽𖽘𖾏 𖼡𖽑𖽙𖾑 𖼙𖽠𖾏.
9 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼚𖽜𖾑 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 ,
𖼽𖽺 𖼗𖽔 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽡𖽝 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 .
10 𖼟𖽡 𖼽𖽺 𖼗𖽔 𖽐𖼎𖽙 𖼟𖽡 𖼉𖽺𖾏 𖼑𖽡𖾏 𖼽𖽺 𖼗𖽔 ,
𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼺𖽘𖾏,
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼞𖽔𖾏,
𖼼𖽷𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾏 𖼟𖽡 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐.
11 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼙𖽔 ,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼞𖽪𖾑.
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼁𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼯𖽡𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼙𖽠𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼏𖽘 𖼅𖽻𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
13 𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼑𖽡𖽻𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾑, 𖼯𖽡𖾏 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼮𖽑𖽙𖾏；
𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼊𖽪𖾐 𖼡𖽑𖽠𖾑, 𖼡𖽔 𖼋𖽱𖾏 𖼽𖽔𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏.
14 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼽𖽺 𖼗𖽔 ,
𖼟𖽪𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼐𖽱𖾏 .
15 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼑𖽘 𖼗𖽙𖾏 , 𖼑𖽘 𖼺𖽙𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏；
𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽐𖼡𖽔 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.
16 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼐𖽱𖾏, 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼮𖽢𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼉𖽺𖾏 𖼛𖽙𖾏 𖼞𖽔𖾐 𖼖𖽗.
17 𖼖𖽑𖽻𖾏 𖼨𖽑𖽙𖾑 𖼖𖽑𖽻𖾏, 𖽃𖽺𖾐 𖼨𖽑𖽙𖾑 𖽃𖽺𖾐 𖽐𖼮𖽳𖾏 ,
𖼽𖽺 𖼗𖽔 𖼷𖽑𖽡𖽝𖾏 𖼨𖽑𖽙𖾑 𖼷𖽑𖽡𖽝𖾏 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“ 𖼨𖽑𖽙𖾑 𖼽𖽺 𖼗𖽔 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖾏” )𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.
18 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 ；
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻𖾏 𖽐𖼊𖽻𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 , 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼮𖽵𖾑 𖽐𖼞𖽔𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏.
19 𖼑𖽔𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖼎𖽻𖾏 𖽃𖽻𖾐, 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽑𖽡𖽝𖾐 𖼦𖽡𖾏 𖼺𖽱𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏.
20 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼛𖽙𖾏 𖼋𖽔 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏.
21 𖼖𖽪𖾑 𖼡𖽑𖽠𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼗𖽙 𖼮𖽡𖽺𖾏 𖼑𖽡𖽝𖾏, 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼮𖽡𖽺𖾏 𖼞𖽪𖾐；

22 𖼟𖽡 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽐𖼎𖽙 𖽃𖽔 𖼛𖽡𖾏 𖼳𖽻𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼎𖽑𖽻𖾏 𖼛𖽻𖾏 𖼛𖽻 𖼈𖽡𖽝𖾏,
