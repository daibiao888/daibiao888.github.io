29 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼦𖽺 𖼑𖽡𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽙 𖼊𖽡𖾐 𖼞𖽹𖾐 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼺𖽻𖾏 𖽃𖽔𖾐 𖼛𖽔𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼄𖽻𖾏 𖼐𖽡𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼻𖽙𖾑 “𖼀𖽑𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽪𖾐 ” 𖼐𖽡𖾑 , 30 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼺𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽪𖽺 𖼖𖽪𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽤𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 31 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽙 𖼊𖽡𖾐 𖼞𖽹𖾐 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼸𖽙𖾏 𖼯𖽡𖽹𖾏 𖼻𖽙𖾑 “𖼀𖽑𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽪𖾐 ” 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼾𖽨𖾏 𖽐𖼎𖽙 𖼾𖽨𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖽃𖽡𖾑 , 𖼊𖽪𖾏 𖼡𖽜𖾑 𖽐𖼎𖽙 𖼐𖽡𖽝𖾑 𖽐𖼊𖽜 . 32 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼏𖽵 𖼮𖽡𖽝𖾏 “𖼀𖽑𖽪𖾐 𖽃𖽳𖾐 𖼐𖽑𖽪𖾐 ” 𖼐𖽡𖾑 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 , 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖽐𖼊𖽠𖾑 𖼮𖽡𖽝𖾏 .
