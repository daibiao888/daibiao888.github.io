3 𖼽𖽡𖾑 𖼊𖽑𖽗𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 、 𖼽𖽡𖾑 𖼳𖽷𖾑 𖼄𖽔𖾐 、 𖼽𖽡𖾑 𖼊𖽳 𖼀𖽔𖾑 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾑 𖼞𖽱𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼦𖽔𖾐 𖼳𖽡 𖼖𖽳 𖼀𖽙𖾑 𖼐𖽡 . 4 𖼮𖽡𖾑 𖼊𖽙𖾑 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏𖼀𖽑𖽡𖾐 𖼐𖽪𖾐 𖼽𖽡𖾑 𖼖𖽳 ． 𖼦𖽪𖾏 𖼳𖽔𖾑 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼳𖽡𖽝𖾏 𖽃𖽳𖾐 ． 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼊𖽑𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼦𖽪𖾏 𖽃𖽳𖾐 𖼅𖽔𖾏 𖼯𖽡𖽙𖾏 .
5 𖼊𖽑𖽡 𖼞𖽙𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼽𖽔𖾏 𖼳𖽷𖾑 𖼦𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐 : 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽷𖾏 𖼳𖽡𖾑 𖼖𖽔𖾑 , 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼷𖽷𖾏 𖼐𖽔 𖼖𖽔𖾑 . 6 𖼐𖽔 𖼖𖽔𖾑 𖼑𖽜 𖼽𖽔𖾏 𖼳𖽷𖾑 𖼦𖽪𖾏 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼦𖽪𖾏 𖼺𖽔𖾐 、 𖼳𖽡𖾑 𖼇𖽪 、 𖼊𖽑𖽡 𖼄𖽡𖾐 𖼐𖽡 、 𖼦𖽔𖾐 𖼳𖽢 𖼺𖽷𖾑 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 , 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽔 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 . 7 𖼳𖽡𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽡𖾐 𖼖𖽡𖽝 、 𖼺𖽙𖾐 𖼳𖽢 、 𖼽𖽡𖾑 𖼊𖽑𖽡 𖼐𖽔 . 8 𖼞𖽙𖾑 𖼺𖽷𖾑 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼐𖽡𖽙 、 𖼺𖽙𖾐 𖼀𖽡𖾐 𖼀𖽔𖾑 , 𖽐𖼎𖽙 𖼦𖽔𖾐 𖼖𖽪𖽝 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼦𖽳 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 . 9 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼺𖽷𖾑 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 , 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼊𖽡𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼺𖽷𖾑 , 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ．” 10 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼺𖽷𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 , 𖼎𖽳𖾏 𖼇𖽘𖾑 𖼞𖽪𖾑 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 , 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 , 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 , 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 .” 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 .
