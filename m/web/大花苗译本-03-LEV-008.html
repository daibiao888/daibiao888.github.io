𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏
（𖼋𖽠𖾏 29：1 - 37）
8
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “ 𖼟𖽡 𖼨𖽑𖽳𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽔𖾏 , 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏、𖼛𖽙𖾏 𖼟𖽪𖾏 𖼚𖽜𖾑 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 、𖽃𖽔𖾐 𖼋𖽪 𖼞𖽡𖾑 𖼙𖽻𖾏 𖼾𖽘𖾏 、 𖽃𖽡𖾐 𖼡𖽔𖾐 𖼊𖽠𖾐 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼯𖽡𖽙𖾏 𖼋𖽔𖾐， 3 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽔 𖼛𖽻𖾏 .” 4 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼋𖽘𖾏 . 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽃𖽔 𖼛𖽻𖾏 . 5 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽑𖽡𖾏: “𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 .”
6 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼨𖽑𖽳𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 , 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖽐𖼷𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 . 7 𖼟𖽙𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽑𖽘𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 , 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 , 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 , 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼸𖽙𖾏 𖼘𖽑𖽘𖾐 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽐𖼊𖽙𖾏 𖽃𖽡𖾑 , 𖼞𖽑𖽡𖽠𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖼐𖽑𖽘𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑， 8 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼛𖽔𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖽐𖼎𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽘𖾑 , 𖼞𖽑𖽡𖽠𖾏 𖽂𖽪𖾑 𖼖𖽡𖽾 𖽐𖼎𖽙 𖼊𖽑𖽪𖾐 𖼄𖽡𖽾 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼛𖽔𖾏 𖽐𖼀𖽙 𖽐𖼎𖽙𖾐 𖽐𖼚𖽙， 9 𖼞𖽑𖽡𖽠𖾏 𖼖𖽪𖾑 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼊𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 , 𖼖𖽪𖾑 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼚𖽑𖽙𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼞𖽪𖾐 , 𖼚𖽔𖾏 𖼞𖽪𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼯𖽡𖽙𖾏 .
10 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽑𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 , 𖼟𖽙𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼇𖽺𖾐 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 11 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼛𖽙𖾏 𖽐𖼎𖽷𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼳𖽤𖾏 𖼟𖽻𖾏 , 𖼎𖽺𖾏 𖼚𖽜𖾑 𖽃𖽔𖾑 𖼏𖽘 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼏𖽘 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 , 𖼎𖽺𖾏 𖽐𖼷𖽔𖾑 𖼖𖽪𖾑 𖼋𖽺𖾏 𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽜𖾐 𖼋𖽺𖾏 , 𖼟𖽙𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽺𖾐 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 12 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼛𖽙𖾏 𖼘𖽑𖽡𖾑 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼚𖽜𖾑 𖼐𖽱𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 13 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼨𖽑𖽳𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽘𖾑 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 , 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼗𖽙𖾏 𖼡𖽑𖽠𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼮𖽡𖽝𖾏 . 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼯𖽡𖽙𖾏 .
14 𖼐𖽱𖾏 𖼮𖽤𖾐 𖼊𖽪𖾑 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 𖼋𖽔𖾏 , 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼯𖽔𖾏 𖼾𖽔𖾏 𖽃𖽡𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏， 15 𖼎𖽺𖾏 𖼊𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼚𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 , 𖼚𖽜𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽻𖾐 𖼋𖽪 𖼞𖽪𖾑 𖼞𖽪𖾐 𖼟𖽻𖾏 𖼯𖽪𖾏 , 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼘𖽑𖽡𖾑 𖽐𖼮𖽑𖽘𖾑 𖼎𖽻𖾏 𖼊𖽠𖾏 𖽃𖽔 𖼏𖽘 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼑𖽜 𖽃𖽔𖾑 𖼏𖽘 𖼚𖽠𖾏 𖼾𖽔𖾏 . 16 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽐𖼀𖽱𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖽃𖽔𖾏 𖼐𖽑𖽨𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼛𖽙𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼛𖽙𖾏 𖼀𖽙𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼏𖽻𖾏 𖽐𖼎𖽙 𖼏𖽻𖾏 𖼛𖽙𖾏 , 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 . 17 𖼊𖽪𖾑 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏 𖽐𖼎𖽙 𖼀𖽡𖾑 𖼊𖽠𖾏 𖼊𖽑𖽡𖽝𖾐 𖼡𖽔𖾑 , 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼡𖽻𖾏 𖼾𖽡 𖼙𖽠𖾏. 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼯𖽡𖽙𖾏 .
