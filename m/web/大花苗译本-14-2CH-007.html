𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 8：62- 66）
7
𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏 , 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼘𖽻𖾑𖽑 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 , 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏； 2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 . 3 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼐𖽡𖾑 𖼘𖽻𖾑𖽑 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾐 𖼇𖽔𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽔𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽃𖽡𖾑 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 , 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 :
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼺𖽡𖽝𖾐 ,
𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽷𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 ！”
4 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 . 5 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏 𖼑𖽨𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼉𖽻𖾏 𖼋𖽪𖾏 𖼾𖽘𖾏 𖼗𖽙𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 . 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 . 6 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼙𖽻 𖼋𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 , 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼢𖽜𖾏 𖽐𖼎𖽔 , 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼷𖽷𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 . 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 .
7 𖽃𖽔𖾑 𖼏𖽘 𖼋𖽻𖾏 𖼟𖽪𖾏𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼷𖽷𖾏 𖽃𖽡𖾑 , 𖽐𖼊𖽡𖾏 𖼋𖽘 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽷𖾑 𖼖𖽑𖽘𖾑 、 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼛𖽙𖾏 , 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 𖼑𖽡𖾏 𖼛𖽙𖾏 .
8 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 , 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽜𖾏 𖼚𖽔𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖽃𖽔𖾑 𖼯𖽱𖾏 𖽃𖽔𖾐 𖼟𖽱𖾐 𖽃𖽻𖾐 𖽃𖽡𖾑 , 𖼖𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 , 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 . 9 𖼾𖽡𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 , 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 , 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 . 10 𖼖𖽪𖾑 𖼳𖽤𖾏 𖼖𖽑𖽡𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 , 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼊𖽑𖽡𖽝𖾐 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 .
