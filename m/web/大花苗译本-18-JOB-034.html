10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼞𖽘𖾐 𖼀𖽻𖾐 𖼞𖽡𖾑 , 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 ,

𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 .
11 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 .
12 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 ,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼺𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 .
13 𖼡𖽔 𖼋𖽱𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 ,
𖼡𖽔 𖼋𖽱𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽺𖾑 ?
14 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ,

15 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼛𖽙𖾏 𖼋𖽔𖾏 ,

16 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼞𖽘𖾐 𖼀𖽻𖾐 𖼞𖽡𖾑 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏 ,
