𖼷𖽷𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖽐𖼮𖽡𖽙𖾏
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼉𖽻𖾏 𖼮𖽘𖾑
1
𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐. 2 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖽐𖼀𖽑𖽔𖾐 𖼦𖽡𖾏 𖽐𖼀𖽑𖽻𖾏, 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏； 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼉𖽳𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽻𖾐.
3 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 ”， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼋𖽘𖾏. 4 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼇𖽺𖾐 𖼺𖽘𖾏. 5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 “𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 ”, 𖼨𖽑𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 “𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 ”. 𖼅𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼊𖽑𖽪 𖼇𖽻𖾑.
6 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼞𖽑𖽡𖽠𖾏 𖽃𖽻𖾐 𖼇𖽺𖾐 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐.” 7 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖼋𖽠𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏， 𖼄𖽔𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑 𖽃𖽻𖾐, 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖽃𖽡𖾑 𖽃𖽻𖾐 𖼇𖽺𖾐 𖼺𖽘𖾏. 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏. 8 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 “𖽐𖼊𖽪𖾏”. 𖼅𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑.
9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑 𖽃𖽻𖾐 𖼽𖽔𖾑 𖽐𖼚𖽱𖾏 𖼦𖽡 𖼸𖽱 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏, 𖼟𖽙𖾏 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏.” 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏. 10 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼊𖽡𖾐 𖽐𖼡𖽑𖽔𖾐 𖽃𖽔𖾏 “𖼊𖽡𖾐 ”， 𖼨𖽑𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼦𖽡 𖼸𖽱 𖽃𖽔𖾏 “ 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 ”. 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏. 11 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽡𖾐 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼋𖽠𖾏 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐、𖼻𖽻𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖼛𖽡𖾏, 𖼊𖽑𖽡𖽝𖾐 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏, 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼮𖽡𖽝𖾏.” 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏. 12 𖼊𖽡𖾐 𖼋𖽔𖾏 𖼋𖽠𖾏 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐, 𖽐𖼎𖽙 𖼻𖽻𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖼛𖽡𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏, 𖼊𖽑𖽡𖽝𖾐 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏, 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼮𖽡𖽝𖾏． 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏. 13 𖼅𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑.
14 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼼𖽘𖾏, 𖼟𖽙𖾏 𖼇𖽺𖾐 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼺𖽻𖾏, 𖼊𖽡𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐、𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏、𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏， 15 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼮𖽡𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏， 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐. 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏. 16 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼷𖽷𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼼𖽘𖾏, 𖼖𖽪𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼞𖽳𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐, 𖼖𖽪𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼞𖽳𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼎𖽺𖾏 𖼷𖽷𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐. 17 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼼𖽘𖾏 𖼐𖽡𖾑 𖼇𖽺𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼮𖽡𖽝𖾏, 𖼟𖽙𖾏 𖼮𖽡𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐， 18 𖼞𖽳𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏． 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏. 19 𖼅𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼚𖽻𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑.
20 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼽𖽔𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑； 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼽𖽘𖾏 𖼎𖽻𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏.” 21 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖼋𖽠𖾏 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏, 𖼎𖽺𖾏 𖼷𖽷𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏. 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏. 22 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜， 𖼐𖽑𖽡𖾏: “𖼽𖽔𖾑 𖼳𖽤𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐, 𖼐𖽻 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼳𖽤𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐.” 23 𖼅𖽔𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽜𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑.
24 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽡𖾐 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏, 𖼯𖽦𖾏 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏、𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼞𖽘 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐、 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏． 𖼑𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏.” 25 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖼊𖽡𖽝𖾑 𖼺𖽘𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏； 𖼯𖽦𖾏 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏, 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼞𖽘 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏. 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏.
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼼𖽘𖾏 𖼷𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐
