𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼀𖽻𖾐 𖼘𖽙，𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽹𖾑 𖽐𖼮𖽜 𖼸𖽙𖾏、𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼑𖽡𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝． 2 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏，𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼐𖽡𖽙𖾐 𖼉𖽻𖾏 𖼷𖽑𖽷𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼙𖽔 𖼏𖽵 𖼮𖽡𖽝𖾏． 3 𖼸𖽙 𖼋𖽘𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 ，𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼄𖽻𖾏 𖼻𖽻𖾏 ， 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏； 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼐𖽡𖾑，𖼟𖽪𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼮𖽨𖾏 𖼮𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏． 4 𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼺𖽻𖾏 𖽐𖼊𖽠𖾑 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼊𖽑𖽡 𖼊𖽙𖾑，𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽐𖼮𖽜 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝，𖼊𖽻𖾏 𖼅𖽪𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼋𖽡𖽝𖾏 𖼀𖽡𖾏 𖼋𖽜𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼮𖽨𖾏 𖼮𖽵𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼟𖽡．
