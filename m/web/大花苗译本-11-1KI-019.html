19
𖼽𖽔𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼑𖽪𖾏, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽦𖾑 𖼳𖽡𖾐 𖼀𖽡𖽝. 2 𖼽𖽦𖾑 𖼳𖽡𖾐 𖼀𖽡𖽝 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡 𖼟𖽡 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼟𖽡 𖼳𖽦𖾑 𖼺𖽱𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼳𖽦𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼨𖽑𖽢𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑.”
𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼦𖽙 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐
3 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼀𖽜𖾑 𖼳𖽦𖾑, 𖼸𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼑𖽡𖾏 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑 , 𖼟𖽙𖾏 𖼡𖽑𖽜𖾑 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑. 4 𖼾𖽨𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼞𖽡𖾑, 𖼋𖽔𖾏 𖼸𖽙 𖽃𖽡𖾐 𖼇𖽘𖾐 𖼘𖽙 𖼊𖽑𖽺 𖽐𖼊𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 ( “𖼘𖽙 𖼊𖽑𖽺 ” 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼗𖽪𖾏 ，𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼊𖽑𖽪𖾑 ). 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔 𖼟𖽙𖾏 𖼋𖽔, 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽳𖾑 𖼋𖽘𖾏! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐.” 5 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼘𖽙 𖼊𖽑𖽺 𖽐𖼊𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾏, 𖼊𖽻𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏． 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼀𖽔𖾏 𖼐𖽱𖾏, 𖼐𖽑𖽡𖾏: “𖼳𖽠𖾑 𖼗𖽙𖾏，𖼑𖽻𖾏 𖼦𖽡𖾑 !” 6 𖼐𖽱𖾏 𖼑𖽔𖾏, 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖽃𖽻𖾐 𖽐𖼎𖽙 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽡𖽝𖾐, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽻𖾏 𖼺𖽘𖾏 𖼦𖽻𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏. 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼎𖽺𖾏 𖽃𖽔𖾐 𖼟𖽻𖾐 𖽐𖼀𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼳𖽠𖾑 𖼗𖽙𖾏，𖼑𖽻𖾏 𖼦𖽡𖾑! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼺𖽷𖾐.” 8 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼺𖽘𖾏, 𖼐𖽱𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼋𖽡𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼐𖽡𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼅𖽻𖾏 𖼊𖽘𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼐𖽑𖽪𖾐 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼄𖽑𖽙𖾏, 𖼸𖽙 𖼋𖽘𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼟𖽪𖾏 𖼦𖽙 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐.
