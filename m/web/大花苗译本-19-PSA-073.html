𖼻𖽻𖾏 𖽐𖼎𖽙 𖽐𖼚𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖽐𖼮𖽔
𖼽𖽔𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏.
73
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼚𖽠𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏！
2 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽪𖾑 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖽐𖼚𖽡𖽝 𖼢𖽻,
𖼞𖽪𖾑 𖼮𖽑𖽹𖾏 𖼡𖽔𖾑 𖼏𖽔𖾏 𖼮𖽔𖾑𖽑 𖽐𖼮𖽷𖾑 𖼣𖽔𖾏 𖽐𖼚𖽡𖽝 𖼢𖽻.
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼐𖽔𖾑.
 
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼛𖽙 𖼊𖽔𖾐 𖼋𖽡𖽝𖾏,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼄𖽙𖾐.
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖽃𖽦𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔.
6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼺𖽱𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼐𖽑𖽘𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐,
𖼊𖽠𖾏 𖼊𖽠𖾑 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼛𖽙 𖼐𖽔𖾑 𖽃𖽙𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼚𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼸𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏.
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐.
10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽙𖾑 𖽐𖼮𖽡𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑,
𖼦𖽻𖾏 𖼋𖽘𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝.
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐？
𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼖𖽺𖾑?”
12 𖼑𖽔𖾏 𖼘𖽪！ 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼮𖽡𖽻𖾐.
13 𖼞𖽪𖾑 𖽐𖼮𖽢𖾏 𖼼𖽙 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖽐𖼮𖽢𖾏 𖼼𖽙 𖽐𖼷𖽔𖾑 𖼋𖽡𖾏 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏.
14 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔,
𖼷𖽑𖽔𖾏 𖼊𖽡𖾏 𖼟𖽡𖾏 𖼊𖽻𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏.
 
15 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑.”
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽪𖾏 𖼯𖽡𖽙𖾏 𖼟𖽡 𖽐𖼀𖽙𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙 𖼋𖽘𖾏.
16 𖼞𖽪𖾑 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽔𖾑，
17 𖼸𖽙 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏.
18 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼢𖽻、𖼛𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏.
19 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽔𖾐 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏！
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏.
20 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽱𖾏 𖼯𖽷𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼑𖽔𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑.
𖼮𖽵𖾑 𖼘𖽪！𖼟𖽡 𖼯𖽷𖾏 𖼗𖽙𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏.
 
21 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼡𖽻𖾐 𖽐𖼊𖽙𖾏 𖼗𖽙𖾏,
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖽐𖼀𖽱𖾏 𖼺𖽱𖾏 𖼊𖽻𖾏 𖼮𖽑𖽡𖽙𖾐.
22 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐,
𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼺𖽱𖾏 𖼯𖽦𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
23 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐,
𖼟𖽡 𖼊𖽔𖾑 𖼮𖽤𖾐 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏.
24 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼻𖽻𖾏 𖼗𖽪𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑,
𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼚𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼚𖽙.
25 𖼋𖽠𖾏 𖼟𖽡 𖼺𖽘𖾏, 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑?
𖼋𖽠𖾏 𖼟𖽡 𖼺𖽘𖾏, 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳.
26 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼯𖽻 𖼼𖽙,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼈𖽳𖾐 𖼗𖽘𖾏,
𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼀𖽠, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
