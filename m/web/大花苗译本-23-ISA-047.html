6 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏,

𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
𖼈𖽡𖽝𖾏 𖼟𖽡 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼊𖽪𖾏 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏,

7 𖼟𖽡 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 .”
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,

8 𖼟𖽡 𖼖𖽜𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽔𖾐, 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖽃𖽡𖾑,

9 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼛𖽙𖾏 𖼋𖽔𖾏, 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼐𖽡𖽙𖾐,
𖽃𖽔𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼊𖽔𖾐 𖽃𖽡𖾐 𖽐𖼮𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽔𖾐 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 ,

10 𖼟𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼻𖽡𖽝𖾏 𖼾𖽨𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼐𖽑𖽡𖾏 :

𖼳𖽻𖾏 𖽐𖼚𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼚𖽠𖾏 𖼮𖽨𖾏
