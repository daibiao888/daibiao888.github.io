𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 , 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽠𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 , 𖼽𖽙 𖼽𖽔 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼏𖽙 𖼅𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖽐𖼊𖽻𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 . 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 . 𖼽𖽙 𖼽𖽔 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 , 𖼄𖽔𖾐 𖼙𖽻 𖽐𖼊𖽻𖾏 𖼢𖽻 𖼙𖽠𖾏 . 2 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏( “𖼞𖽡𖾏 𖼄𖽱𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼄𖽔𖾐 𖼖𖽳 𖼞𖽑𖽗𖾑 ”， 𖼄𖽔𖾐 𖼖𖽳 𖼞𖽑𖽗𖾑 𖼟𖽪𖾏𖼄𖽡𖾐 𖼖𖽳 𖼞𖽪𖽾𖾑 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼷𖽷𖾏． )𖼑𖽡𖾏 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼞𖽪𖾐, 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖽃𖽡𖾐 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳, 𖼎𖽺𖾏 𖼎𖽑𖽹𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼐𖽡𖾑 𖽐𖼊𖽻𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 . 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽙 𖼙𖽻 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 . 3 𖼮𖽤𖾐 𖽐𖼚𖽙 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖽃𖽔𖾏 𖼊𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼳𖽻𖾐 、 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼡𖽻𖾐 、 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 , 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼯𖽡𖽙𖾏𖼽𖽔𖾏 𖼄𖽞 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 . 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼞𖽑𖽔𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏 .
𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐
（𖼺𖽔𖾐 𖽃𖽔𖾐 21：15 - 22）
4 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼺𖽳 𖽐𖼎𖽙 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 𖼦𖽪𖾏 𖼳𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼞𖽹𖾑 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽡𖾏 𖼀𖽑𖽺𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽵𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏 . 5 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 . 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼖𖽳 𖼦𖽔𖾐 𖼐𖽔 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽙𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼖𖽔𖾑 𖼦𖽔𖾐 𖼄𖽡𖾐 . 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼊𖽪𖾑 𖽃𖽔𖾑 𖼮𖽤𖾏 𖼄𖽑𖽪𖾑 𖼘𖽑𖽙𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼸𖽱𖾏 𖽐𖼊𖽙𖾏 𖼐𖽱𖾏 𖼊𖽪𖾑 𖼊𖽑𖽡𖾏 𖽐𖼊𖽻𖾐 . 6 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼊𖽑𖽳 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 ,𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 , 𖼊𖽠𖾏 𖼋𖽡𖾏 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼚𖽻𖾐 𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼊𖽠𖾏 𖼋𖽡𖾏 , 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 . 7 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 , 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼄𖽡𖾐 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 . 8 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 , 𖼞𖽑𖽔𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
