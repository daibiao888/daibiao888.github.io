𖼷𖽷𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏
6 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼽𖽘𖾏 𖼎𖽻𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏，𖼅𖽔𖾏 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐，𖼟𖽪𖾏 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐、 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 、𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼗𖽪𖾏 、 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 ． 7 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏：“𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏． 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏、 𖼊𖽡𖾐 、𖼳𖽡𖽠𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐．”
8 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾑 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼷𖽹𖾏 𖼐𖽑𖽡𖾏：“𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼼𖽠𖾏 𖼢𖽻 𖼙𖽠𖾏 𖼋𖽘𖾏！𖼼𖽠𖾏 𖼢𖽻 𖼙𖽠𖾏 𖼋𖽘𖾏！𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼙𖽔 𖼮𖽑𖽵𖾑 𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼖𖽔𖾏 𖼦𖽻𖾏 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼮𖽡𖽠𖾑 𖽐𖼎𖽙 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖼮𖽡𖽠𖾑．
