𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼉𖽺𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 , 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 .” 2 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼸𖽻 𖽃𖽔𖾑 𖼈𖽳𖾐 , 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 . 3 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 𖽃𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼑𖽡𖾏 𖼺𖽙𖾑 𖼖𖽑𖽻𖾏 𖽐𖼎𖽙 𖽐𖼡𖽜𖾏 , 𖼎𖽺𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 , 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼙𖽪𖾏 . 4 𖼎𖽺𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼨𖽑𖽹𖾑 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼋𖽔𖾏 . 5 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼊𖽪𖾐 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼮𖽔𖾑𖽑 𖼯𖽠 𖼀𖽻𖾐 𖼟𖽳 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 , 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 , 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 , 𖼟𖽙𖾏 𖼻𖽻𖾏 𖽐𖼷𖽷𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼡𖽑𖽔𖾐 𖽐𖼷𖽻𖾏 𖼉𖽻𖾏 𖼞𖽪𖽙 , 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 .” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽔 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 , 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 .
