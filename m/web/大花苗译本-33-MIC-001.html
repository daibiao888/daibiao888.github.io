5 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏,

𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 ,

7 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼋𖽪 ,

10 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼎𖽻𖾏 𖼄𖽻𖾏,
