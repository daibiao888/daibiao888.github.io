𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼙𖽔 𖼚𖽙𖾏,
𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽔𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼄𖽞 𖼞𖽪𖽺𖾏 𖼋𖽠𖾏 𖼁𖽘𖾏,
2 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼁𖽘𖾏 𖼟𖽡𖾏 𖼮𖽑𖽦𖾐 𖼞𖽡𖾑𖽑,
𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽑𖽡𖽝𖾐 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏．
𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 ,
𖼮𖽢𖾑 𖼄𖽡𖾐 𖽐𖼎𖽙 𖼳𖽔𖾑 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼻𖽻𖾏 𖼼𖽘𖾏, 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 ,
𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 .
 
3 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼯𖽻 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼇𖽻𖾑 𖼯𖽡𖽻 𖼟𖽪𖾏 𖼯𖽻 𖼼𖽙 𖼊𖽻𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
4 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼀𖽡𖾏 𖼮𖽨𖾑 𖼮𖽙𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼚𖽔𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏．
𖼑𖽔𖾏 𖼘𖽪 ！ 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼗𖽙𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 ,
𖼽𖽔𖾑 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼗𖽙𖾏 𖼮𖽨𖾏 𖼅𖽡 .”
 
5 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼊𖽡𖾑 𖼛𖽜 𖽃𖽔𖾑 𖼅𖽔 𖼽𖽔𖾑 𖼡𖽑𖽜𖾐,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼗𖽘𖾐 𖼽𖽔𖾑 𖼐𖽙𖾑𖽑 𖼗𖽪𖾏 .
6 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼽𖽔𖾑 𖼚𖽑𖽡𖽝𖾏 𖼺𖽱𖾏 𖼷𖽹𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽜 𖼅𖽜 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼙𖽔 𖼨𖽑𖽪𖾏．
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽻𖾐 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼚𖽱𖾏 𖼋𖽠𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐.
7 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼺𖽔𖾏 𖼟𖽪𖾏 𖼳𖽙𖾑 𖼞𖽪𖾐 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼁𖽘𖾏 𖽃𖽻𖾐,
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐.
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼳𖽙𖾏 𖼀𖽱𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼘𖽑𖽘𖾑 𖼋𖽠𖾏 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐、 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐 𖽐𖼎𖽙 𖼀𖽑𖽪 𖼷𖽑𖽻𖾐.
 
8 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼸𖽙𖾐 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑．
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏,
𖼟𖽪𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼊𖽔𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼊𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑.
9 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐,
𖼯𖽦𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼐𖽡𖾑,
𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼮𖽷𖾐 𖼋𖽡𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑 ,
𖼮𖽔𖾑 𖼅𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼊𖽔𖾐.
10 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼚𖽠𖾏 𖼮𖽨𖾏 𖽃𖽡𖾑,
𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼳𖽡 𖼣𖽔𖾑 ,
𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼽𖽔𖾑 𖼎𖽙𖾑 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏 𖽐𖼎𖽙 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏 𖼞𖽑𖽔𖾏 𖼀𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏.
