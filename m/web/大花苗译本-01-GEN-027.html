𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖼋𖽘𖾏, 𖽃𖽔𖾑 𖼅𖽔 𖼇𖽔 𖼙𖽻 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏, 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼋𖽔𖾏, 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐.” 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑.” 2 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼗𖽻𖾏 𖼋𖽘𖾏, 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼋𖽱 𖼖𖽔𖾑 𖼋𖽔. 3 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼚𖽘𖾐 𖼐𖽜𖾐, 𖼟𖽪𖾏 𖼐𖽑𖽘𖾐 𖼈𖽙𖾏 𖽐𖼎𖽙 𖼐𖽜𖾑𖽑, 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖽐𖼡𖽹𖾏 𖼻𖽻𖾏 𖼸𖽔, 4 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼑𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼯𖽡𖽙𖾏 𖼋𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼑𖽻𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽔 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡.” 5 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏， 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏. 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑 𖽐𖼊𖽻𖾏 𖽐𖼡𖽹𖾏 𖼻𖽻𖾏 𖼸𖽔 𖼙𖽠𖾏, 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼡𖽹𖾏 𖼻𖽻𖾑 𖼸𖽔 𖼯𖽡𖽙𖾏 𖼗𖽙. 6 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽑𖽡𖾏: 7 ‘ 𖼟𖽡 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽐𖼡𖽹𖾏 𖼻𖽻𖾑 𖼸𖽔 𖼗𖽙𖾏, 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼑𖽻𖾏, 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼦𖽡𖾐 𖼋𖽔 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡.’ 8 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼟𖽡 𖽃𖽡𖾑, 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏. 9 𖼟𖽡 𖼚𖽔𖾏 𖽐𖼀𖽙𖾏 𖼮𖽑𖽷𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,𖼑𖽜 𖼞𖽪𖾑 𖼮𖽤𖾐 𖽃𖽔𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏 𖼟𖽪𖾏 𖼛𖽙 𖼗𖽙𖾏, 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼑𖽻𖾏 𖽃𖽡𖾑, 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏. 10 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼑𖽻𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾐 𖼋𖽔 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡.” 11 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖽃𖽡𖾐 𖼮𖽡𖾏 𖼙𖽱 𖼅𖽔𖾏 𖼚𖽻𖾐, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼮𖽢𖾏； 12 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼞𖽪𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏, 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼏𖽵 𖼅𖽳𖾏.” 13 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼊𖽪𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑, 𖼟𖽡 𖼯𖽔 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏, 𖼅𖽻𖾏 𖼮𖽤𖾐 𖼮𖽑𖽷𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑.” 14 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼮𖽤𖾐 𖼗𖽙𖾏 𖼋𖽘𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐, 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼑𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏. 15 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽘𖾑， 16 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏 𖼀𖽡𖾑 𖼊𖽠𖾏 𖼡𖽑𖽠𖾑 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼅𖽙𖾏 𖼦𖽡𖾏 𖼑𖽡𖽻𖾏， 17 𖼄𖽔𖾐 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖽐𖼎𖽙 𖼳𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽔𖾐 𖼞𖽙 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
18 𖼽𖽔𖾐 𖼞𖽙 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾑 𖼈𖽹𖾐!” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑．𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏?” 19 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐 𖼽𖽡𖾑 𖼺𖽻𖾐, 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏. 𖼮𖽑𖽙 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏, 𖼑𖽻𖾏 𖼞𖽪𖾑 𖽐𖼡𖽹𖾏 𖼻𖽻𖾏 𖼸𖽔, 𖼻𖽻𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼏𖽵 𖼅𖽳𖾏.” 20 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼇𖽺𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼘𖽪?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼻𖽻𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼯𖽡𖽙𖾏.” 21 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼟𖽡 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽻𖾐?” 22 𖼽𖽔𖾐 𖼞𖽙 𖼎𖽺𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼽𖽡𖾑 𖼺𖽔𖾐. 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼯𖽡𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏, 𖼋𖽡𖾏 𖼯𖽡𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.” 23 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼇𖽺𖾐 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼚𖽻𖾐, 𖼺𖽱𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐. 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼏𖽵 𖼅𖽳𖾏. 24 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼋𖽡𖽝?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼾𖽙 𖼯𖽡𖽙𖾏.” 25 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼑𖽻𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏 𖼻𖽻𖾏 𖼸𖽔, 𖼑𖽜 𖼟𖽡 𖼏𖽵 𖼅𖽳𖾏.” 𖼽𖽔𖾐 𖼞𖽙 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼮𖽡𖽠𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼦𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏. 26 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼡𖽑𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.” 27 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖼡𖽑𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏. 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽃𖽡𖾐 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏, 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼏𖽵 𖼅𖽳𖾏, 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼊𖽪𖾐 𖼑𖽡𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼑𖽡𖾏 𖽐𖼷𖽑𖽡𖽝𖾐 𖼦𖽘𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
28 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾑 𖼙𖽱,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼟𖽪𖾏 𖼛𖽙,
𖼊𖽑𖽡𖽝𖾐 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖼟𖽡.
29 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡,
𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼟𖽡,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑,
𖼟𖽡 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼟𖽡.
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼟𖽡, 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏,

“𖼑𖽔𖾏 𖼘𖽪！𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼟𖽪𖾏 𖼛𖽙 𖼟𖽡 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼀𖽠,
