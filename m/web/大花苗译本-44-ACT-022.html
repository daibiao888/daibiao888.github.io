22
“ 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 ， 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼊𖽑𖽔𖾑 𖼮𖽔𖾑𖽑 𖼞𖽪𖾑 𖼁𖽡𖽝 𖼡𖽑𖽳𖾏 𖼅𖽡．” 2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽙𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼗𖽪𖾏，𖼎𖽺𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏． 3 𖼀𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼘𖽑𖽘𖾑 𖼾𖽨 𖼎𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼮𖽢𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽔𖾏 𖼺𖽪𖾏 𖼙𖽻，𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼘𖽑𖽙𖾐 𖼊𖽡𖽝𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼄𖽔𖾐 𖼖𖽡𖽝 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼟𖽡𖽠 𖽐𖼊𖽠𖾑，𖼷𖽹𖾏 𖼳𖽻𖾏 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐． 4 𖼞𖽪𖾑 𖼙𖽔 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼋𖽡𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼐𖽡𖾑， 𖼟𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼸𖽙 𖼋𖽔，𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽔 𖼾𖽠， 𖼖𖽔𖾏 𖼡𖽑𖽹𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏． 5 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼗𖽻，𖼸𖽠𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏，𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖽐𖼊𖽠𖾑 𖼄𖽻𖾏 ，𖼟𖽪𖾏 𖼺𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 ，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼽𖽔𖾑 𖼡𖽑𖽹𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼋𖽡𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼐𖽡𖾑， 𖼮𖽤𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
6 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑，𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼙𖽻，𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼑𖽻𖾏 𖼳𖽪𖾏，𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏， 𖼅𖽔𖾏 𖼮𖽡𖾏 𖼙𖽪 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼗𖽙𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏． 7 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼢𖽻 𖼚𖽠𖾑 𖽐𖼀𖽑𖽻𖾏 𖽐𖼀𖽙 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐，𖼐𖽑𖽙𖾑 𖼅𖽔𖾏 𖼳𖽘𖾐 𖼗𖽪 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：‘𖼺𖽻𖾐 𖼘𖽙 ！ 𖼺𖽻𖾐 𖼘𖽙 ！𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑 𖼖𖽺𖾑？’ 8 𖼞𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 ：‘𖼮𖽵𖾑 𖼘𖽪 ！𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？’ 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼟𖽪𖾏 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽡𖾏 𖼦𖽺𖾏 𖽃𖽡𖾑．’ 9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏，𖼖𖽔𖾏 𖼁𖽙𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖽃𖽡𖾑，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽑𖽙𖾑 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼳𖽘𖾐 𖼗𖽪 ． 10 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏： ‘𖼮𖽵𖾑 𖼘𖽪 ！𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？’ 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 ：‘𖼳𖽠𖾑 𖼗𖽙𖾏，𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 ，𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 ， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼟𖽡 𖼅𖽻𖾏 𖽃𖽔𖾏．’ 11 𖼞𖽪𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼊𖽪𖾑 𖼮𖽡𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼮𖽡𖾏 𖼐𖽔𖾑，𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏，𖼎𖽺𖾏 𖼮𖽤𖾐 𖼞𖽪𖾑 𖼋𖽡𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼙𖽻．
12 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼐𖽔 𖼐𖽡 𖼽𖽔𖾏 ， 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼷𖽻𖾑 𖼐𖽱𖾏． 13 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑， 𖼳𖽠𖾑 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡𖾏 𖼊𖽘𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼺𖽻𖾐 𖼘𖽙 ，𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼦𖽡𖾑．’ 𖼞𖽪𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼑𖽔𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐，𖼎𖽺𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼋𖽘𖾏． 14 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：‘𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽹𖾑 𖼟𖽡，𖼟𖽙𖾏 𖼟𖽡 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏，𖼎𖽺𖾏 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽡𖾑，𖼎𖽺𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑 𖼐𖽱𖾏 𖼳𖽘𖾐 𖼗𖽪 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏． 15 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼙𖽔𖾏 𖼁𖽙𖾏、𖼡𖽑𖽙𖾑 𖼙𖽔 𖼐𖽑𖽙𖾑 𖽃𖽡𖾑 ，𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐． 16 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼊𖽳𖾑𖽑 𖼮𖽡𖽝𖾏 𖼖𖽺𖾑 ？ 𖼳𖽠𖾑 𖼗𖽙𖾏， 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏， 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽐𖼷𖽔𖾑， 𖽐𖼷𖽔𖾑 𖼺𖽘𖾏 𖼟𖽡 𖼾𖽔𖾏 ．’
