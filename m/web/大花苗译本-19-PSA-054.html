𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽙𖾏 𖼮𖽵𖾑 𖽐𖼀𖽘𖾏
𖼳𖽡𖾑 𖼇𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖽐𖼎𖽙𖼺𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏: 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼚𖽺𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼘𖽪 , 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽡𖾑 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽷𖾑 𖼺𖽙𖾑 .
（𖼺𖽔𖾐 𖽃𖽡𖾐 23：14 - 24）
54
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑,
𖼎𖽑𖽻𖾏 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼑𖽜 𖼞𖽪𖾑 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏.
2 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽻 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏,
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼗𖽪𖾏 .
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼊𖽱𖾑 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 ,
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 𖼳𖽦𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
4 𖼑𖽔𖾏 𖼘𖽪 ! 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 ,
𖼮𖽵𖾑 𖼟𖽪𖾏 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼯𖽡𖽙𖾏.
5 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽡𖾑,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼸𖽙𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏.
 
6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼟𖽡 .
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖽐𖼷𖽷𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.
7 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏.
