𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼊𖽪𖾐 𖼄𖽙𖾐 𖽃𖽻𖾐
14
𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼊𖽔𖾐 𖼡𖽔 𖼅𖽔 𖼑𖽔𖾏 𖼐𖽱𖾏． 2 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼄𖽙𖾐 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽙𖾏． 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼊𖽑𖽡𖽝𖾐 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼞𖽑𖽙𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐，𖼋𖽡𖽝𖾏 𖼞𖽑𖽙𖾏 𖼢𖽻 𖼖𖽔𖾏 𖼞𖽑𖽙𖾏 𖼦𖽡𖾐 𖼢𖽻 ？” 4 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐．𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽔𖾑 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 ，𖼞𖽑𖽙𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼻𖽻𖾏，𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏． 5 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽡𖾐 𖼋𖽪 𖼑𖽨𖾏，𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐，𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼡𖽪𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼖𖽺𖾑 ？” 6 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 ．
7 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽔𖾏 𖼡𖽑𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖼋𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼺𖽹𖾑 𖼇𖽻𖾑 𖼮𖽑𖽡𖽠𖾏，𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： 8 “ 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖼅𖽻𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼮𖽑𖽦𖾐，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼮𖽑𖽡𖽠𖾏，𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏 𖼡𖽑𖽔𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽡 𖼋𖽔𖾏， 9 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏：‘𖼳𖽠𖾑 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐！’𖼟𖽡 𖼎𖽺𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼺𖽷𖾐 𖼗𖽙 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼢𖽜𖾏． 10 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼢𖽜𖾏，𖼟𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏：‘𖼽𖽺 𖼗𖽔， 𖼋𖽔𖾏 𖼐𖽡𖽙𖾐 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼮𖽑𖽡𖽠𖾏．’ 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼅𖽔𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼋𖽘𖾏． 11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼾𖽨𖾏，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼢𖽜𖾏 ，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼦𖽡𖾏 𖼢𖽜𖾏 𖼾𖽨𖾏， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐．”
𖼨𖽑𖽪𖾏 𖼡𖽑𖽔𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏
12 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼊𖽜𖾐 𖼳𖽪𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼊𖽜𖾐 𖼄𖽑𖽙𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼟𖽡 𖼽𖽺 𖼗𖽔 𖼞𖽱𖾑 𖼋𖽡𖾏，𖽐𖼎𖽙 𖼑𖽜𖾏 𖼮𖽘𖾏 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 ，𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼨𖽑𖽪𖾏 𖼟𖽡，𖼟𖽡 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼁𖽻𖾏． 13 𖼟𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏，𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 、 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐、 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖼯𖽠𖾏、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐，𖼟𖽡 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏 ， 14 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼁𖽻𖾏 𖼟𖽡．𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖽃𖽡𖾑，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 ．”
𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼟𖽪𖾏 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏
（𖼊𖽑𖽹𖾏 22：1-10）
15 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖽃𖽡𖾑， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽙𖾑 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑， 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏．” 16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼨𖽑𖽪𖾏 𖼡𖽑𖽔𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐． 17 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏，𖼎𖽺𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：‘𖼷𖽑𖽜𖾑 𖼋𖽔𖾏 𖼦𖽡𖾑，𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏．’ 18 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼾𖽡𖾏 𖼙𖽳𖾏 𖼦𖽡𖾐 𖼋𖽔𖾏．𖼺𖽪𖾐 𖼁𖽱 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽡𖽝𖾏，𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼑𖽔𖾏，𖽐𖼮𖽢𖾏 𖼟𖽡 𖼯𖽔𖾏 𖼞𖽪𖾑．’ 19 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏：‘ 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼀𖽜𖾐 𖽐𖼞𖽡𖽠𖾏 𖼑𖽨𖾏 𖼮𖽡𖽝𖾏，𖼽𖽔𖾑 𖼅𖽻𖾏 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼻𖽻𖾏 𖼯𖽡𖽙𖾏，𖽐𖼮𖽢𖾏 𖼟𖽡 𖼯𖽔𖾏 𖼞𖽪𖾑．’ 20 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽑𖽡𖾏；‘𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼷𖽙𖾏 𖼐𖽤𖾐，𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼋𖽔𖾏．’ 21 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑 𖼺𖽷𖾐 𖼗𖽙，𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑，𖼮𖽵𖾑 𖽐𖼞𖽔𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐， 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏：‘ 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼚𖽙 𖼙𖽻，𖼚𖽔𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐、 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐，𖼨𖽑𖽳𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾏 、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐、 𖼅𖽔𖾏 𖼮𖽑𖽔𖾑 𖽐𖼮𖽠𖾏、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 、𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼡𖽑𖽘𖾑 𖽃𖽡𖾑 𖼋𖽔𖾏．’ 22 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏：‘𖼮𖽵𖾑 𖼘𖽪 ！𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼊𖽘𖾏 𖼋𖽘𖾏，𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼺𖽷𖾐．’ 23 𖼮𖽵𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏：‘ 𖼟𖽡 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑，𖽐𖼎𖽙 𖽃𖽔𖾑 𖼮𖽡𖽺𖾏 𖼉𖽘𖾏，𖽃𖽔𖾏 𖼊𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽪𖾑 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏． 24 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼨𖽑𖽪𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼑𖽻𖾏 𖼞𖽪𖾑 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏．’”
𖽃𖽔𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼽𖽔𖾑 𖽐𖼊𖽱𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏
（𖼊𖽑𖽹𖾏 10：37-38）
25 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻 𖼐𖽑𖽡𖾏： 26 “ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑，𖼮𖽔𖾑 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼮𖽑𖽠𖾏（“𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 ”𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼳𖽪𖾐”）𖼐𖽱𖾏 𖼾𖽨𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏、 𖼐𖽤𖾐 、𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏、𖼞𖽱𖾑 𖼋𖽡𖾏 、𖼞𖽱𖾑 𖼻𖽷𖾑、𖽐𖼎𖽙 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 ，𖼐𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏． 27 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼏𖽷𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼊𖽪𖾑 𖼞𖽑𖽡𖾐 𖼖𖽡𖾐 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏． 28 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼖𖽡𖾏 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼊𖽑𖽘𖾐，𖼟𖽪𖾏 𖼦𖽡𖾐 𖼺𖽪𖾐 𖼼𖽻𖾏 𖼮𖽡𖽝𖾏 𖼮𖽔𖾑 𖼺𖽙𖾏 𖼽𖽔𖾑 𖼀𖽡𖾑 𖼯𖽠 𖼑𖽡𖽝𖾏，𖼋𖽡𖽝𖾏 𖼷𖽑𖽙𖾏 𖼋𖽻𖾏 𖼖𖽔𖾏 𖼷𖽑𖽙𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏？ 29 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼊𖽠𖾏 𖽐𖼞𖽔𖾏 ，𖼈𖽡𖽝𖾏 𖼷𖽑𖽙𖾏 𖼦𖽡𖾐 𖼊𖽡𖽝𖾑， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼚𖽙𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： 30 ‘ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼁𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼋𖽘𖾏 ，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑．’ 31 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼽𖽔𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏， 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼺𖽪𖾐 𖼐𖽡𖽙𖾐 𖽐𖼊𖽱𖾏 𖽃𖽡𖾐 𖽐𖼊𖽱𖾏，𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼏𖽙 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖽃𖽔𖾐 𖼉𖽻𖾏 𖼏𖽙 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼖𖽺𖾑？ 32 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏，𖼽𖽔𖾑 𖼷𖽪𖾑 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼺𖽷𖾐，𖼷𖽷𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼺𖽘𖾑 𖽃𖽡𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏． 33 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼡𖽔 𖼋𖽱𖾏，𖼮𖽔𖾑 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽡 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 ，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏．”
（𖼊𖽑𖽹𖾏 5：13；𖼞𖽑𖽙𖾐 9：50）
