“ 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑，𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼟𖽻𖾏 𖼗𖽜𖾏 𖽐𖼞𖽻𖾏 𖽐𖼮𖽡𖽻，𖼯𖽡𖽙𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼷𖽹𖾏 𖼈𖽻𖾑 𖼮𖽑𖽦𖾐． 2 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐，𖼀𖽜𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐． 3 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼯𖽡𖽙𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 ， 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼛𖽙𖾏． 4 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼯𖽡𖽙𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐，𖼎𖽺𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼛𖽙𖾏 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖼮𖽡𖽝𖾏． 5 𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖼗𖽙𖾏 𖼙𖽡，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼣𖽙𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏． 6 𖼋𖽘 𖼄𖽑𖽙𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：‘ 𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖼗𖽙𖾏 𖼋𖽘𖾏，𖼅𖽡 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏．’ 7 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖽐𖼞𖽻𖾏 𖽐𖼮𖽡𖽻 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼷𖽑𖽱𖾏 𖼷𖽑𖽱𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐． 8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼐𖽑𖽡𖾏：‘ 𖼀𖽡𖾐 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼽𖽔𖾑 𖼋𖽔 𖼋𖽘𖾏 𖼷𖽑𖽜𖾑 𖼅𖽡 𖼡𖽜𖾑 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼛𖽙𖾏 𖼇𖽺𖾐 𖼀𖽡𖾐．’ 9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：‘ 𖽐𖼮𖽑𖽹𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑，𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼾𖽨𖾏 𖼅𖽻𖾏 𖼅𖽔𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽺 𖼛𖽙𖾏．’ 10 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼅𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑，𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖼗𖽙𖾏 𖼋𖽘𖾏．𖽐𖼞𖽻𖾏 𖽐𖼮𖽡𖽻 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼛𖽙𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼡𖽜𖾑 𖼮𖽑𖽦𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏，𖼡𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏． 11 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼸𖽻𖾏 𖽐𖼞𖽻𖾏 𖽐𖼮𖽡𖽻 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏，𖼨𖽑𖽪𖾏：‘ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼮𖽵𖾑 𖼘𖽪 ！𖼡𖽑𖽜𖾐 𖼀𖽡𖾐 𖽃𖽔 𖼛𖽻𖾏！’ 12 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 ： ‘ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼅𖽡．’ 13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ， 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽷𖾏 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽪𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑．”
𖼎𖽑𖽻𖾏 𖼺𖽡𖽝𖾐 𖽃𖽔𖾏 𖼮𖽵𖾑 𖼑𖽪𖾏
（𖼖𖽪𖾏 19：11-27）
14 “𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼗𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼨𖽑𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼋𖽔𖾏，𖼄𖽔𖾐 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 15 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼀𖽻𖾐 𖼸𖽠𖾏，𖼄𖽔𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼎𖽻𖾏
