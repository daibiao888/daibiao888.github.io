𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼯𖽢𖾏 𖼳𖽡𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏, 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑, 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏. 2 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝.” 3 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐； 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 4 “𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼟𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼉𖽺𖾏. 5 𖼎𖽑𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼟𖽡 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔, 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 (“𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 ”𖼟𖽪𖾏 “𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼉𖽺𖾏”), 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽘𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼉𖽺𖾏. 6 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼘𖽘𖾑𖽑 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝； 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏. 7 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽡𖾑 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 8 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑, 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.”
𖼁𖽱𖾏 𖼷𖽑𖽙𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏
9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼽𖽔𖾑 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏. 10 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏, 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡, 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏, 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏. 11 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 (“𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏” 𖼟𖽪𖾏 “𖽐𖼷𖽪𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼮𖽑𖽡𖽠𖾏 𖼉𖽜𖾏 𖼊𖽪𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑” 14、23、24、25 𖼽𖽘𖾏 𖼦𖽡𖾏 𖼺𖽱𖾏), 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏. 12 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼾𖽠, 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼅𖽔𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑, 𖼳𖽻𖾏 𖼗𖽙𖾏 𖼾𖽡𖾏 𖼐𖽑𖽪𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏. 13 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏 𖼳𖽻𖾏 𖼗𖽙𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼅𖽔𖾏 𖼗𖽙𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏, 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏. 14 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼇𖽳𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼷𖽜𖾑 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏.”
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽠𖾏 𖼺𖽔𖾐 𖼖𖽹 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑
15 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼖𖽹, 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽐𖼷𖽷𖾏 𖼺𖽔𖾐 𖼖𖽹 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼺𖽔𖾐 𖼖𖽔𖾑. 16 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐. 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼺𖽘𖾐 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼑𖽡𖾏 𖼑𖽡𖽝𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏.” 17 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼋𖽔𖾏 𖼡𖽔𖾑 𖼚𖽙𖾏, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: “𖽃𖽡𖾐 𖼀𖽔𖾏 𖼳𖽡𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼊𖽪𖾐 𖼺𖽷𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏? 𖼺𖽔𖾐 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼯𖽢𖾏 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏 𖼺𖽷𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏?” 18 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽡𖾑 𖼳𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼯𖽡𖾏 𖼷𖽻𖾏 𖼋𖽘𖾏.”
𖼏𖽵 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐
