𖼦𖽔𖾐 𖼀𖽔𖾑 𖼞𖽪 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
3
𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼦𖽔𖾐 𖼀𖽔𖾑 𖼞𖽪 𖼑𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼳𖽷𖾏 𖼮𖽢𖾑 𖼽𖽙𖽾𖾑 𖼳𖽘𖾐.
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼟𖽡 𖼳𖽘𖾐 𖼗𖽪 , 𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽡𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼟𖽡 𖼑𖽪𖾏 ,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽡𖾑 𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽑𖽡𖽠𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖽃𖽔𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏.
 
3 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼊𖽑𖽡 𖼄𖽔𖾏 𖼋𖽔𖾏,
𖼊𖽪𖾐 𖼨𖽑𖽪𖾑 𖼎𖽑𖽻𖾏 𖼀𖽔𖾑 𖼖𖽔 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏. (𖼳𖽡𖾏 𖼖𖽔𖾑)
𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽐𖼊𖽪𖾏,
𖼷𖽻𖾑 𖼖𖽔 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐.
4 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼮𖽡𖾏,
𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽡𖾏 𖼋𖽠𖾏 𖼐𖽱𖾏 𖼮𖽡𖾏 𖼋𖽔𖾏,
𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.
5 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼞𖽡𖾏 𖽃𖽔𖾏 𖼼𖽙 ,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼡𖽘𖾐 𖼊𖽠𖾏 𖼅𖽔𖾏 𖼳𖽙𖾑 𖼉𖽠𖾏 𖼋𖽙𖾏 𖼋𖽻𖾏 𖼚𖽠𖾏 𖼋𖽠𖾏 𖼋𖽔𖾏.
6 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏, 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 ,
𖼐𖽱𖾏 𖽃𖽡𖾐 𖼑𖽔𖾏, 𖼮𖽡𖽠𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐,
𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖼗𖽡𖾏 𖼋𖽠 𖽐𖼀𖽑𖽔𖾐 𖼺𖽘𖾏 ,
𖽃𖽔𖾑 𖼯𖽱𖾏 𖼎𖽻𖾐 𖼩𖽜𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼗𖽡𖾏 𖼼𖽠𖾏 𖼢𖽻 ,
𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼊𖽑𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
 
7 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼞𖽪𖾐 𖼳𖽔𖾑 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏.
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼯𖽡𖽜𖾏 𖼎𖽻𖾏 𖼑𖽜𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖼦𖽡𖾐 𖼙𖽔 𖼛𖽡𖾏 𖽃𖽻𖾐,
𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐,
𖼳𖽪𖾐 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
9 𖼟𖽡 𖼊𖽪𖾑 𖼐𖽜𖾑𖽑 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼈𖽙𖾏 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 . (𖼳𖽡𖾏 𖼖𖽔𖾑)
𖼟𖽡 𖼎𖽑𖽻𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼇𖽺𖾐 𖼺𖽘𖾏 𖽐𖼚𖽡𖽝𖾐 𖼊𖽡𖾐.
10 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼁𖽙𖾏 𖼟𖽡 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏,
𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏,
𖽃𖽻𖾐 𖼊𖽙𖾐 𖽐𖼷𖽑𖽷𖾏 𖼳𖽱𖾑,
𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼮𖽘𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐”).
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽪𖾑 𖼈𖽙𖾏 𖼀𖽙𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼟𖽡𖾏 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼮𖽑𖽦𖾑 ,
𖼟𖽡 𖼊𖽪𖾑 𖼄𖽑𖽪𖾑 𖼮𖽡𖾏 𖼮𖽑𖽦𖾑 𖼟𖽡𖾏 𖼮𖽡𖾏 𖽐𖼚𖽘𖾏 ,
𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽳 𖽐𖼎𖽙𖾏.
12 𖼟𖽡 𖼋𖽠𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼯𖽪𖾏 𖼖𖽻𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖼋𖽠𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽻𖾏 𖼡𖽑𖽔𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼡𖽻𖾐.
13 𖼟𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼮𖽨𖾏 𖼟𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼚𖽜𖾑,
𖽐𖼊𖽻𖾏 𖽐𖼮𖽻 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽐𖼞𖽔𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼢𖽳 𖼋𖽠 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 (“𖼊𖽠𖾏 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏 ”), 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼦𖽡𖾐 𖼚𖽘𖾐. (𖼳𖽡𖾏 𖼖𖽔𖾑)
14 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽪𖾑 𖼄𖽑𖽪𖾑, 𖼐𖽑𖽙𖾏 𖼊𖽙𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼺𖽱𖾏 𖼮𖽢𖾏 𖼀𖽡𖾐 𖼘𖽙𖾐 , 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼀𖽡𖾐 𖼇𖽺𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 , 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏.
15 𖼟𖽡 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼏𖽘 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑,
𖼟𖽪𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙 .
 
16 𖼞𖽪𖾑 𖼐𖽙𖾑𖽑 𖼊𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽡𖾑 , 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏,
𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏,
𖽃𖽔𖾑 𖼷𖽑𖽘𖾏 𖼗𖽱𖾏 𖽐𖼚𖽔 ,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏,
𖼞𖽪𖾑 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼋𖽙𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖼦𖽔𖾏 𖼀𖽡𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐.
 
17 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼦𖽡𖾐 𖼖𖽑𖽘𖾑 ,
𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼦𖽡𖾐 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 ,
𖼞𖽗𖾐 𖼖𖽗 𖽐𖼊𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 ,
