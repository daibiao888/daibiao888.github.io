𖼑𖽜 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑,𖼐𖽑𖽡𖽝𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼟𖽱𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽑𖽡𖽝𖾐 𖼺𖽙𖾐 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽙 𖼽𖽔𖾏 𖼺𖽷𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼐𖽡𖽙𖾐 𖼦𖽘𖾑 𖽐𖼮𖽳𖾑 𖽐𖼊𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑, 𖼊𖽔𖾏 𖼋𖽔 𖽃𖽡𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼐𖽡𖾑, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼾𖽻, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 “𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏”𖼳𖽘𖾐.
（𖼺𖽔𖾐 𖽃𖽔𖾐 8：13）
60
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼀𖽡𖾐 𖼺𖽘𖾏, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏,
𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐, 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏！
2 𖼟𖽡 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏, 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼋𖽠 𖼞𖽘𖾐 𖽐𖼀𖽑𖽔𖾐 𖼺𖽘𖾏 ,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽠 𖼞𖽘𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙 .
3 𖼟𖽡 𖼟𖽙𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼢𖽻 𖼟𖽡𖾏 𖼖𖽡𖾐 𖼉𖽪𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐.
 
4 𖼟𖽡 𖼄𖽔𖾐 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 ,
𖼟𖽙𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 (𖼡𖽑𖽙𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼺𖽪𖾐 𖼈𖽙𖾏”). （𖼳𖽡𖾏 𖼖𖽔𖾑）
5 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼊𖽡𖾐 𖼀𖽡𖾐, 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐, 𖼻𖽻𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏.
6 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼇𖽺𖾐 𖼺𖽘𖾏 𖼳𖽷𖾏 𖼮𖽡𖽝𖾏 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽔𖾏 𖼺𖽪𖾑 𖼞𖽙 𖼦𖽘𖾑 .
7 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝 ,
𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝．
𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽻𖾏 𖼋𖽻𖾏,
𖼽𖽨 𖼊𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.
8 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼋𖽺𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼞𖽑𖽻𖾏 𖼎𖽻𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 .
𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼘𖽪！ 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼸𖽠𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼖𖽺𖾑?”
 
9 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏？
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑？
10 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼾𖽙 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼀𖽡𖾐 𖼺𖽘𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼸𖽻𖾏 𖼏𖽙 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏 𖼋𖽘𖾏 𖼖𖽔𖾏 ?
11 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐 𖽐𖼊𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽘𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙.
12 𖼀𖽡𖾐 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽪𖾏 𖼖𖽻𖾐 𖼀𖽡𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏.
