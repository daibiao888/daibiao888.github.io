33
𖼽𖽔𖾐 𖼞𖽙 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏, 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼋𖽔𖾏 𖼋𖽘, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼸𖽙𖾏 𖼋𖽔𖾐 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼇𖽺𖾐 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼖𖽡𖾏 𖼽𖽔𖾏, 𖼖𖽔𖾑 𖼮𖽡𖽝 𖽐𖼎𖽙 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑. 2 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼊𖽑𖽡𖽝𖾐 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼅𖽻𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏, 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼅𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼖𖽔𖾑 𖼮𖽡𖽝 𖽐𖼎𖽙 𖼽𖽙 𖼺𖽳 𖼅𖽻𖾏 𖼊𖽔𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐. 3 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏, 𖼅𖽻𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼳𖽤𖾏 𖼟𖽻𖾏, 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏.
4 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼏𖽘 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏, 𖼁𖽔 𖼐𖽱𖾏 𖼮𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼉𖽔𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖼚𖽘𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼡𖽑𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽑𖽡𖽝𖾐 𖼖𖽔𖾏 𖼐𖽡𖽝𖾑. 5 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏, 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼗𖽙𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼘𖽪?” 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼯𖽡𖽙𖾏.” 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼺𖽪𖾐 𖼋𖽔𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑. 7 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼖𖽔𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼋𖽔𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼽𖽙 𖼺𖽳 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼮𖽡𖽝 𖼖𖽔𖾏 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑. 8 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽑𖽡𖾏: “𖼅𖽺𖾐 𖽐𖼀𖽙𖾐 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽷𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼘𖽪?” 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏.” 9 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽑𖽡𖾏: “ 𖼷𖽔𖾑 𖼞𖽱𖾑 𖼘𖽪 ！𖼞𖽪𖾑 𖼁𖽡𖽝 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼋𖽘𖾏, 𖼟𖽡 𖼁𖽡𖽝 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼟𖽡！” 10 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐！ 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏, 𖽃𖽔𖾏 𖼯𖽡𖾏 𖼮𖽑𖽙 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼋𖽡𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼐𖽡𖾑； 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼚𖽑𖽪𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏. 11 𖼮𖽑𖽙 𖼟𖽡 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼗𖽙 𖼎𖽻𖾏 𖼟𖽡, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏.” 𖼽𖽔𖾐 𖼞𖽙 𖼯𖽪 𖼯𖽪𖾏 𖽃𖽔𖾐 𖼊𖽔𖾑 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏.
12 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏.” 13 𖼽𖽔𖾐 𖼞𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼊𖽻𖾏 𖼅𖽔𖾏 𖼺𖽷𖾐, 𖼑𖽨𖾏 𖼾𖽘𖾏 𖼖𖽔𖾏 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼺𖽷𖾐, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼷𖽹𖾑 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐, 𖽐𖼀𖽙𖾏 𖼯𖽦𖾏 𖼖𖽔𖾑 𖼋𖽔 𖼺𖽘𖾏. 14 𖼮𖽑𖽙 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼡𖽑𖽜𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼮𖽔𖾏 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼟𖽪𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑.”
15 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖾏, 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼀𖽡 𖼯𖽠 𖼖𖽜𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼮𖽡𖽝𖾏.” 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼅𖽪𖾏 𖼋𖽘𖾏, 𖼯𖽔 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏.” 16 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼳𖽠𖾑 𖼊𖽠𖾏, 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏. 17 𖼈𖽡𖽝𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼺𖽪𖾑 𖼞𖽙 𖼙𖽠𖾏, 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼾𖽨𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼎𖽺𖾏 𖼑𖽜 𖽐𖼀𖽙𖾏 𖼯𖽦𖾏 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼺𖽪𖾑 𖼞𖽙 (“ 𖼺𖽪𖾑 𖼞𖽙” 𖼟𖽪𖾏 “𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐”).
