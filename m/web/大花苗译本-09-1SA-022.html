𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏, 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼊𖽪𖾏 𖼖𖽔 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼙𖽠𖾏． 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽻𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 2 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏、𖼎𖽷𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾑、𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑, 𖼞𖽑𖽔𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏. 3 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼄𖽡𖾐 𖼺𖽷𖾑 𖼀𖽔𖾑 𖼙𖽠𖾏, 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽢𖾏 𖼟𖽡 𖼯𖽪𖾏 𖼞𖽪𖾑 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼋𖽔𖾏, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼋𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 . 4 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼗𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 , 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼖𖽔𖾏 𖼗𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 . 5 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼮𖽢𖾑 𖼊𖽳 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑, 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽠𖾏.” 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼺𖽘𖾏, 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼦𖽔𖾐 𖼖𖽡𖽝 𖽐𖼚𖽙 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼙𖽠𖾏.

6 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽙𖾑𖽑 𖼟𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽔𖾐, 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼙𖽱 𖼙𖽱 𖼟𖽪𖾏 𖼚𖽻𖾑 𖼖𖽠𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼄𖽑𖽪𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽻 𖼷𖽻𖾏 𖼡𖽑𖽜𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏. 7 𖼺𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼺𖽘𖾑 𖼇𖽳𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏, 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖽐𖼎𖽙 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼘𖽪? 𖼸𖽠𖾏 𖼺𖽹𖾑 𖼮𖽘𖾏 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪? 8 𖼅𖽡 𖼦𖽡𖾐 𖼙𖽳𖾐 𖼦𖽡𖾏 𖼷𖽱𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼦𖽡𖾐 𖼙𖽳𖾐 𖼦𖽡𖾏 𖼷𖽱𖾏 𖼞𖽪𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖼟𖽡𖾏 𖼦𖽹𖾏 𖼞𖽪𖾑, 𖼺𖽱𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽳𖾏.” 9 𖼊𖽑𖽻𖾏 𖼐𖽡𖾑 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽙𖾑 𖼽𖽡𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼮𖽡𖽻𖾏 𖼐𖽙 𖼀𖽳 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼊𖽑𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼄𖽡𖾐 𖼖𖽳 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 10 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼄𖽡𖾐 𖼖𖽳 𖼑𖽜 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽙𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼊𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼏𖽻𖾏 𖼐𖽱𖾏.
