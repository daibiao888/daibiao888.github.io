𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼐𖽫𖾐 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼄𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼞𖽹𖾑 𖼟𖽪𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼐𖽔 𖼀𖽔 𖼐𖽤𖾐 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼑𖽜𖾏 𖼮𖽘𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳 𖼽𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽻 𖼐𖽡𖾑 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐, 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽻 𖼖𖽺𖾑?” 6 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼮𖽡 𖼄𖽔𖾐 𖼳𖽡𖾐 𖼞𖽳 𖼖𖽔𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼳𖽡𖾐 𖼞𖽳 𖼖𖽔𖾑 𖼺𖽙𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽡𖽝 𖼸𖽙𖾏 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏. 7 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼚𖽻𖾐 𖼖𖽑𖽡𖾏. 8 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼮𖽡𖾑 𖼳𖽪 𖼊𖽜𖾑
