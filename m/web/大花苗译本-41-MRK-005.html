（𖼊𖽑𖽹𖾏 8：28-34；𖼖𖽪𖾏 8：26-39）
5
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 𖼺𖽘𖾑 𖼀𖽡 𖼋𖽡𖾏，𖼟𖽪𖾏 𖼞𖽳 𖼖𖽔𖾑 𖼺𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏． 2 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽡𖾐 𖽐𖼡𖽜𖾏 𖽐𖼞𖽙𖾏， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼯𖽵𖾏 𖼐𖽱𖾏 ， 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼸𖽱𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑， 3 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖽃𖽡𖾑， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼡𖽑𖽹𖾐 𖼋𖽻𖾏 𖼐𖽱𖾏， 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽻𖾏 𖼖𖽔𖾏 𖼡𖽑𖽹𖾐 𖼦𖽡𖾐 𖼋𖽻𖾏， 4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽔 𖼎𖽑𖽻𖾏 𖼟𖽻𖾏 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽑𖽡𖾏 𖼊𖽠𖾏 𖽐𖼎𖽙 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽻𖾏 𖼡𖽑𖽹𖾐 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼟𖽻𖾐 𖼋𖽘， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼀𖽱𖾏 𖼊𖽪𖾏 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽻𖾏 𖼺𖽘𖾏， 𖼟𖽻𖾏 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽑𖽡𖾏 𖼊𖽠𖾏 𖼐𖽱𖾏 𖽐𖼀𖽱𖾏 𖼚𖽑𖽘𖾏 𖼺𖽘𖾏， 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼯𖽵𖾏 𖼋𖽻𖾏 𖼐𖽱𖾏． 5 𖼐𖽱𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖽃𖽡𖾑， 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏， 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼾𖽨𖾏 𖼀𖽑𖽔𖾏 𖼾𖽨𖾏． 6 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖼎𖽺𖾏 𖼏𖽘𖾏 𖼅𖽻𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏． 7 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼡𖽔𖾏 𖼳𖽷𖾏 ？ 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔！” 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼙𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “ 𖼟𖽡 𖼷𖽷𖾑 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼐𖽡𖾑，𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏．” 9 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ？” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏： “ 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 ‘𖽐𖼀𖽙𖾏’， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼐𖽔𖾑．” 10 𖼐𖽱𖾏 𖼯𖽪 𖼯𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖼟𖽙𖾏 𖼡𖽑𖽔𖾐 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼊𖽡𖾐．
11 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖽐𖼊𖽘𖾏 𖽃𖽔 𖼼𖽘 ， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼀𖽙𖾐 𖽐𖼀𖽔𖾏 𖼑𖽙 𖼻𖽻𖾐； 12 𖼋𖽡𖾏 𖼀𖽡𖾐 𖼚𖽘𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼊𖽜 𖼐𖽑𖽡𖾏：“ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼀𖽙𖾏 𖽐𖼀𖽔𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑．” 13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏，𖼀𖽡𖾑 𖼚𖽘𖾐 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼀𖽙𖾏 𖽐𖼀𖽔𖾏， 𖽐𖼀𖽙𖾏 𖽐𖼀𖽔𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼚𖽑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼷𖽔𖾏， 𖼀𖽻𖾐 𖼁𖽻 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼷𖽑𖽪𖾐 𖼋𖽔 𖼊𖽘𖾏． 𖽐𖼀𖽔𖾏 𖼖𖽔𖾑 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏． 14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼾𖽨 𖽐𖼀𖽔𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼙𖽠𖾏 ，𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼙𖽻， 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼼𖽙𖾏． 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼑𖽔𖾏 𖼅𖽔𖾏 𖼑𖽪𖾏 𖽃𖽔𖾑 𖼳𖽷𖾏． 15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼉𖽺𖾏 𖽃𖽡𖾑， 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏 𖽃𖽡𖾑，𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼙𖽔 𖼊𖽻𖾏 𖽐𖼀𖽙𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽡𖾑， 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼮𖽡𖽝𖾏，𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏． 16 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑， 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏 𖽃𖽡𖾑 𖼡𖽑𖽙𖾑 𖽐𖼮𖽷𖾐 ， 𖽐𖼎𖽙 𖽐𖼀𖽙𖾏 𖽐𖼀𖽔𖾏 𖽃𖽡𖾑 𖼑𖽪𖾏， 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐． 17 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽐𖼊𖽜 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏． 18 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏， 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼯𖽵𖾏 𖽃𖽡𖾑， 𖼋𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽙𖾏 𖼯𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐． 19 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 ，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏， 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼑𖽜𖾏 𖼮𖽘𖾏， 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏：𖼟𖽪𖾏 𖼮𖽵𖾑 𖼑𖽜 𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐， 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽡 𖼋𖽘𖾏．” 20 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽢𖾑 𖼀𖽙𖾑 𖼖𖽡𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼣𖽹 𖼖𖽪𖾐 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼄𖽙𖾐 𖽐𖼚𖽱𖾏 𖽐𖼮𖽑𖽘𖾑
（𖼊𖽑𖽹𖾏 9：18-26；𖼖𖽪𖾏 8：40-56）
21 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼞𖽙𖾏 𖼎𖽺𖾏 𖼋𖽡𖾏 𖼚𖽔𖾏 𖼀𖽘𖾑𖽑 𖼀𖽡 𖼋𖽡𖾏， 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏， 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑． 22 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽳𖾑 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱， 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼣𖽹 𖼖𖽪𖾐 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑， 𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽠𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐， 23 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼖𖽔𖾑 𖼄𖽡𖽝𖾑 𖽐𖼷𖽑𖽹𖾏 𖼖𖽔𖾑 𖼋𖽔 𖼋𖽘𖾏， 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏．” 24 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼅𖽻𖾏． 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼸𖽙𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼐𖽹𖾏 𖼐𖽱𖾏．
