7 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼅𖽻𖾏 𖼨𖽑𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼗𖽻 𖼋𖽔𖾏 , 𖼄𖽔𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 , 𖼞𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽹𖾏 . 8 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼀𖽡𖾐 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 .” 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼗𖽪𖾏 𖼎𖽙𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼚𖽙 𖼗𖽙𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼸𖽠𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 , 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽐𖼮𖽜 𖼟𖽡 𖼋𖽘𖾏 .” 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼄𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼷𖽻𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼀𖽡 𖼟𖽡 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼾𖽨𖾏 , 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽑𖽔𖾏 𖼮𖽑𖽙𖾏 . 11 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼗𖽙𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 . 12 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼏𖽵 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽐𖼚𖽜𖾏 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼐𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼋𖽔 . 13 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼋𖽔 𖼺𖽘𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼎𖽑𖽻𖾏 𖼈𖽙𖾏 𖼀𖽙𖾑 𖼟𖽙𖾏 𖼋𖽔 , 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼟𖽪𖾏 𖼯𖽦𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 , 𖼸𖽙 𖼟𖽪𖾏 𖼞𖽪𖾐 𖼳𖽘𖾐 𖽃𖽺𖾐 𖼷𖽑𖽙𖾑 𖽐𖼊𖽡𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼋𖽡𖽝𖾏 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 .” 14 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼡𖽜𖾏 𖼎𖽻𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼾𖽨𖾏 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼷𖽑𖽔𖾏 𖼮𖽑𖽙𖾏 . 15 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏 : “𖼸𖽙 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽃𖽔 𖼁𖽙𖾏 .”

16 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 , 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼅𖽔𖾏 𖼺𖽙𖾐 𖼢𖽔𖾐 , 𖼺𖽙𖾐 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽔𖾐 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼞𖽪𖾐 𖼳𖽘𖾐 𖽃𖽺𖾐 𖽐𖼎𖽙𖾑 , 𖼖𖽪𖾑 𖼾𖽡 𖽐𖼚𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏 . 17 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼨𖽑𖽳𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼋𖽠𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼗𖽙𖾏 𖼅𖽻𖾏 𖼸𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼈𖽡𖽝𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼡𖽘𖾐 𖼎𖽻𖾐 𖼊𖽔𖾐 . 18 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖽐𖼚𖽙 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑 , 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼽𖽘𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖼺𖽱𖾏 𖼷𖽑𖽷𖾑 𖼑𖽡 𖼋𖽺𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖽃𖽡𖾐 𖼎𖽻𖾐 𖼙𖽱 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 . 19 𖼞𖽪𖾐 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖽐𖼎𖽙𖾑 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 . 20 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼳𖽡𖾑 𖼐𖽺𖾐 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 , 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 . 21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽵 𖼁𖽺𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽔𖾏 , 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼋𖽔 𖼙𖽠𖾏. 22 𖼎𖽺𖾏 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 , 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼾𖽨𖾏 , 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 .” 23 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼙𖽔 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾐 𖼐𖽑𖽡𖾏 : 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼏𖽵 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 , 𖼟𖽙𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼚𖽑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .” 24 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼦𖽡𖾑! 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽪𖽝 𖽃𖽡𖾐 𖼸𖽻 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽵 𖼁𖽺𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 .” 25 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 .
