𖼞𖽪𖾑 𖼟𖽪𖾏 𖼳𖽔𖾑 𖼖𖽪𖽝 𖽐𖼎𖽘𖾏 𖼋𖽡𖽝𖾏 𖼑𖽡𖾏 𖼁𖽘𖾏 𖼄𖽞 𖼞𖽪𖽺𖾏 (“𖼁𖽘𖾏 𖼄𖽞 𖼞𖽪𖽺𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼁𖽘𖾏 𖼳𖽪𖽺𖾐 𖼳𖽡𖽝𖾏”),
𖼟𖽪𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼑𖽡𖾏 𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏.
 
𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 :
2 𖼞𖽪𖾑 𖼖𖽹𖾐 𖼗𖽔 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼺𖽱𖾏 𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏 𖼐𖽡𖽙𖾐 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖽐𖼚𖽙 .
 
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 :
3 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼾𖽠 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼀𖽑𖽡𖽾 𖽐𖼞𖽙𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽃𖽔𖾏 𖼻𖽻𖾏,
𖼞𖽪𖾑 𖼙𖽔 𖼙𖽔 𖼚𖽙𖾏 𖼚𖽙𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼐𖽑𖽡𖽝𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼷𖽷𖾑 𖽐𖼷𖽑𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑.
4 𖼐𖽱𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏,
𖼎𖽑𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖼀𖽑𖽳𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾏 𖼇𖽔𖾑.
5 𖼷𖽑𖽜𖾑 𖼅𖽡 𖼞𖽑𖽡𖽠𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼳𖽡𖽝𖾐 𖽐𖼷𖽷𖾑 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼄𖽔𖾐 𖼷𖽷𖾑 𖼀𖽑𖽡𖽾 𖽐𖼞𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏.
6 𖼐𖽱𖾏 𖼳𖽤𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼇𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏,
𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼁𖽔 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏.
7 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪！
𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼷𖽹𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼮𖽑𖽙, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼯𖽷𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑, 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽪𖾏”).
𖽃𖽔𖾐 𖼸𖽙𖾐
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 :
8 𖼑𖽻 𖼘𖽪! 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏,
𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽱𖾏 𖽐𖼷𖽜𖾑 𖼎𖽻𖾐 𖽐𖼷𖽜𖾑 𖼩𖽜 𖼋𖽔𖾏.
9 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼺𖽱𖾏 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏, 𖼺𖽱𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼷𖽹𖾏.
𖼑𖽔𖾏 𖼘𖽪 ! 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 𖽃𖽔𖾑 𖼡𖽜𖾐,
𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼯𖽱𖾏 𖼡𖽑𖽙𖾑 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙,
𖼑𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑.
10 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
𖼈𖽻𖾑 𖼮𖽑𖽦𖾐：
“𖼞𖽪𖾑 𖼖𖽹𖾐 𖼗𖽔 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼞𖽪𖾑 𖼖𖽜𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏, 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼦𖽡𖾑 !
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏,
𖽃𖽻𖾐 𖼑𖽘𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏.
12 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼁𖽘𖾏 𖼋𖽠𖾏,
𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼑𖽘𖾏 𖼐𖽻 𖼡𖽔𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼟𖽪𖾏 𖼨𖽑𖽢𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏”),
𖽐𖼡𖽔𖾐 𖼉𖽺𖾏 𖼻𖽻𖾑 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖽐𖼚𖽙 𖼖𖽔𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏.
13 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼳𖽦𖾑 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼋𖽘𖾏,
𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼋𖽠𖾏 𖼁𖽘𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏.
𖼞𖽪𖾑 𖼖𖽹𖾐 𖼗𖽔 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖼞𖽪𖾑 𖼖𖽜𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏, 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼦𖽡𖾑！
14 𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖼘𖽪！ 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽐𖼚𖽙,
𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼷𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼷𖽑𖽘𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏．
𖼷𖽑𖽜𖾑 𖼟𖽡 𖼯𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏,
𖼊𖽻𖾏 𖼐𖽙𖾑𖽑 𖼟𖽡 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼊𖽻𖾏 𖼅𖽔𖾏,
𖼟𖽡 𖼚𖽑𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏.”
15 𖼽𖽔𖾑 𖼑𖽜 𖼀𖽡𖾐 𖼯𖽡𖽻𖾏 𖼞𖽡𖾑 𖼋𖽱𖾏,
𖼟𖽪𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼞𖽡𖾏 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼎𖽑𖽔𖾑 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 .
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼋𖽠𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 .
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐：
16 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼺𖽙𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼖𖽔𖾏 𖼺𖽙𖾏 𖼐𖽱𖾏 ,
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾏 𖼊𖽪𖾑 𖼾𖽨 𖽐𖼀𖽙𖾏 𖼯𖽦𖾐.
17 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼘𖽪!
𖼷𖽑𖽜𖾑 𖼟𖽡 𖼋𖽙𖾏 𖼟𖽙𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼳𖽠𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼸𖽡𖽝𖾏,
𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 𖼙𖽠𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼟𖽡 𖼽𖽔𖾑 𖼺𖽷𖾐 𖼗𖽙𖾏, 𖼺𖽱𖾏 𖼊𖽔𖾑 𖼇𖽔𖾐 𖼷𖽑𖽹𖾏,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼺𖽱𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼷𖽹𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼊𖽑𖽳 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐.
