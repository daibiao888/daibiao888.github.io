10
𖼊𖽪𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽙𖾏 𖼉𖽺𖾏 𖼙𖽔 𖼚𖽙𖾏,

2 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼮𖽔 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏.
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽻𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼡𖽑𖽙𖾑 𖽐𖼀𖽜𖾏 𖼺𖽡𖽝𖾐 𖼉𖽳 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽑𖽠𖾐 𖼺𖽘𖾏.
4 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼞𖽜𖾏 𖼽𖽔𖾑 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼘𖽑𖽻𖾐 𖽐𖼡𖽔𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽻𖾏.
5 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼳𖽙𖾑 𖼳𖽻𖾏 𖽐𖼊𖽡𖾏, 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼡𖽻𖾐 𖼀𖽱𖾏 𖽐𖼚𖽻𖾏 𖼣𖽙𖾏, 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
6 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖽐𖼚𖽪𖾏 𖼸𖽳 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼗𖽱𖾏 𖽐𖼚𖽔𖾏.
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼢𖽻 .
9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳𖾏, 𖼷𖽑𖽔𖾏 𖼏𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 .
10 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏, 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼢𖽻 .
11 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐,
𖽐𖼚𖽪𖾏 𖼸𖽳 𖼈𖽳𖾑 𖽐𖼀𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
12 𖼳𖽪𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖽐𖼮𖽑𖽠𖾏 𖼸𖽠𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽡𖽠𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏.
13 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 𖼊𖽡𖾑 𖼢𖽻𖾏 𖼏𖽻 𖼡𖽠𖾏 𖽐𖼊𖽻𖾏.
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼳𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼀𖽻𖾐 𖼟𖽳 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼖𖽡 𖼟𖽻𖾏 𖼾𖽔𖾏 𖼑𖽪𖾏 .
15 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼑𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏,
𖼊𖽪𖾐 𖼳𖽻𖾏 𖼑𖽡𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼗𖽱𖾏 𖽃𖽔𖾏.
16 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.
17 𖼻𖽙𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏, 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖾏 𖼁𖽜𖾏 𖼙𖽠𖾏.
18 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼮𖽡𖽝𖾏, 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑；
𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏, 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 .
19 𖼗𖽪𖾏 𖼮𖽡𖽻𖾐 𖼸𖽠𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏,
𖼯𖽵𖾏 𖼾𖽨𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
20 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼺𖽱𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼡𖽜𖾏.
21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼑𖽡𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼊𖽪𖾑 𖼾𖽨 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼑𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼛𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏.
22 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼎𖽻𖾏 , 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽔𖾏 𖼊𖽻𖾏,
𖼦𖽡𖾐 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽳𖾏.
23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 , 𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼇𖽜𖾏,
