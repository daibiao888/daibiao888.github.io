𖼐𖽑𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼇𖽺𖾏
14
“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼡𖽑𖽙𖾑 𖼳𖽻𖾏 ,
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 , 𖼮𖽡𖽻𖾐 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 .
2 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼺𖽱𖾏 𖼁𖽘𖾏 , 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼤𖽜𖾏 𖼺𖽘𖾏 ,
𖼺𖽱𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖼽𖽘𖾏 𖼙𖽠𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 .
3 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼡𖽑𖽜𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼘𖽪𖾏?
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼘𖽪𖾏 ?
4 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼖𖽺𖾑 ?
𖼦𖽡𖾐 𖼞𖽳𖾑 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 !
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 ,
𖼐𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 ,
𖼟𖽡 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽐𖼚𖽜𖾏 𖼮𖽡𖽝𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽑𖽡𖽻𖾏 .
6 𖼎𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼋𖽡𖾐 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼐𖽱𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼳𖽙𖾏 ,
𖼋𖽙𖾏 𖼸𖽙 𖼐𖽱𖾏 𖼺𖽱𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽱𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐.
7 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 ,
𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼊𖽙𖾑 𖼺𖽘𖾏 , 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼘𖽘𖾑𖽑 𖽃𖽔𖾑 𖽐𖼷𖽔 𖼺𖽷𖾐 ,
𖽃𖽔𖾑 𖼯𖽡 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼅𖽔𖾏 𖼘𖽑𖽙𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏 .
8 𖽃𖽔𖾑 𖼯𖽤𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼗𖽻𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽔 𖼗𖽔𖾏 𖼈𖽡𖽝𖾏 ,
𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼺𖽘𖾏 𖼖𖽔𖾏 𖼋𖽔 𖼎𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽙 ,
9 𖼋𖽙𖾏 𖼸𖽙 𖽂𖽪𖾏 𖼅𖽔𖾏 𖼀𖽘𖾏 𖽃𖽻𖾐 , 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼘𖽘𖾑𖽑 𖽃𖽔𖾑 𖽐𖼷𖽔 𖼎𖽺𖾏 𖼘𖽘𖾑𖽑 𖽃𖽔𖾑 𖼯𖽡 ,
𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼯𖽡𖽙 𖽃𖽡𖾐 𖼼𖽘𖾐 .
10 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 , 𖼐𖽱𖾏 𖼊𖽪𖾏 𖼀𖽘𖾏 𖼺𖽘𖾏 𖼋𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑 ?
11 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 ,
𖼖𖽪𖾑 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡.
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 , 𖽂𖽪𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 ,
𖼸𖽙 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼯𖽷𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼀𖽱𖾏 𖼨𖽑𖽪𖾏 𖼯𖽷𖾏 .
13 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼈𖽺𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 ,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 , 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼊𖽡𖾏 𖼏𖽵 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 , 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼞𖽪𖾑 .
14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼯𖽡𖽝𖾏 𖼖𖽺𖾑 ?
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼋𖽙𖾏 ,
𖼋𖽙𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽙𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 ( “𖼚𖽠𖾏 𖼮𖽙𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼚𖽑𖽡𖾏 𖼮𖽑𖽦𖾐 ”).
15 𖼟𖽡 𖽂𖽪𖾏 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼊𖽡𖾐 ,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 , 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 .
16 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼮𖽔𖾑 𖼾𖽱 𖼞𖽪𖾑 𖼊𖽠𖾏 𖼐𖽜𖾑 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 .
17 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼊𖽻𖾏 𖼟𖽡 𖼘𖽙𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼐𖽑𖽘𖾐 𖼮𖽡𖽝𖾏 ,
𖼟𖽡 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 .
 
18 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 ,
𖼈𖽳𖾐 𖼗𖽘𖾏 𖼮𖽑𖽹𖾏 𖼮𖽑𖽡𖽠𖾏 𖼙𖽠𖾏 .
19 𖽃𖽻𖾐 𖽐𖼚𖽱𖾏 𖼨𖽑𖽙𖾑 𖼖𖽻𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 ,
𖽃𖽻𖾐 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼖𖽻𖾐 𖽐𖼷𖽔𖾑 𖼺𖽘𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽔 𖼗𖽔𖾏 ,
𖼟𖽡 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 .
20 𖼟𖽡 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 ,
𖼟𖽡 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏 𖼐𖽱𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 , 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼙𖽠𖾏 .
21 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼈𖽡𖽝𖾏 , 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 ,
𖼦𖽡𖾏 𖼢𖽜𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼈𖽡𖽝𖾏, 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 .
22 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼺𖽱𖾑 𖼑𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼄𖽙𖾐 𖼊𖽔𖾐 ,
𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼑𖽜 𖼾𖽨𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 .”
