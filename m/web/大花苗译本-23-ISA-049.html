𖼈𖽺𖾏 𖼞𖽪𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖼟𖽪𖾏 𖽐𖼊𖽡𖾏 𖼺𖽪𖾐 𖼈𖽙𖾏 𖽐𖼚𖽙 .
3 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪! 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑,

4 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 ,
𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽔𖾐 𖽃𖽔𖾏 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼑𖽔𖾏 𖼺𖽔𖾑．
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑.”
𖼮𖽵𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼺𖽘𖾏 𖼸𖽙 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏
5 𖼐𖽔𖾑 𖼐𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐,
𖼐𖽱𖾏 𖼷𖽷𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏,
𖼽𖽔𖾑 𖼟𖽙𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 ,
𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑,
𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏.
6 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏, 𖼟𖽙𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏,
𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽙𖾏 𖼺𖽷𖾐 𖼗𖽙,
𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼊𖽔𖾐, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼎𖽻𖾐,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼡𖽑𖽔𖾐 𖼸𖽙 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏.
 
7 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽺𖾐 𖼳𖽪𖾐,
𖼊𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 ,
𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼛𖽪𖾏 𖼇𖽻𖾑,
𖼟𖽪𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑, 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼟𖽡.”
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏, 𖼞𖽪𖾑 𖼙𖽔 𖼑𖽻 𖼊𖽡𖾐 𖼟𖽡 ,
𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏, 𖼞𖽪𖾑 𖼙𖽔 𖽐𖼀𖽘𖾏 𖼟𖽡 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 ,
𖼽𖽔𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏.”
9 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼡𖽑𖽹𖾐 𖽐𖼊𖽠𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽠𖾏 𖼗𖽙𖾏 𖼦𖽡𖾑 !”
𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼢𖽳 𖼋𖽠𖾏 𖼦𖽡𖾑 !”
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽐𖼎𖽹𖾏 𖼞𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽡𖽠𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼚𖽙𖾏 𖽐𖼊𖽪𖾐 𖼮𖽢𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼡𖽜𖾏.
10 𖼦𖽡𖾐 𖼮𖽑𖽹𖾐 𖼦𖽡𖾐 𖽐𖼡𖽑𖽜𖾏,
𖼳𖽙𖾑 𖼉𖽠𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼦𖽡𖾐 𖼟𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,

𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽻𖾐.
11 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑,
𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾏 𖼷𖽱𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐.

18 𖼟𖽡 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑,

19 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,

“𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽺𖾐 𖽐𖼡𖽹𖾏,

𖼡𖽔 𖼋𖽱𖾏 𖼖𖽔𖾑 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽙𖾏 𖼘𖽑𖽙𖾐 𖼖𖽺𖾑?

𖼟𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
