𖼽𖽙 𖼐𖽔 𖽐𖼊𖽠𖾑
𖼽𖽙 𖼐𖽔 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼚𖽔𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏
1
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼊𖽑𖽹𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼐𖽔 𖼐𖽑𖽡𖾏: 2 “𖼟𖽡 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼉𖽺𖾏 𖽐𖼚𖽙 𖽃𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 . 3 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽙 𖼐𖽔 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 𖼏𖽘 𖼀𖽜𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 𖼐𖽱𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼀𖽑𖽔𖾏 𖼙𖽻 , 𖽐𖼮𖽷𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 . 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽙𖾏, 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏, 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽙𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼚𖽔𖾏 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫．
4 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼳𖽠𖾑 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼳𖽡𖽠𖾑 𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖽐𖼞𖽙𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏. 5 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼾𖽨𖾏 𖼮𖽵𖾑 𖼳𖽺, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽙𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼺𖽘𖾏, 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼳𖽷𖾐 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏. 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽙 𖼐𖽔 𖽂𖽪𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼷𖽜𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽘𖾐 𖽃𖽡𖾑 𖼢𖽻 𖼀𖽱𖾏 𖼗𖽠𖾏 , 𖼻𖽻𖾏 𖼀𖽱𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 . 6 𖽐𖼞𖽙𖾏 𖼮𖽵𖾑 𖼚𖽔𖾏 𖼷𖽜𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽘𖾐 𖼊𖽻𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼀𖽱𖾏 𖼐𖽡𖾑 𖼘𖽪! 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑? 𖼳𖽠𖾑 𖼗𖽙 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 ! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽵𖾑 𖼳𖽺, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼮𖽵𖾑 𖼳𖽺 𖼯𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼀𖽡𖾐, 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼛𖽙𖾏 𖼋𖽔𖾏.”
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽙𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽔𖾏 ! 𖼀𖽡𖾐 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 , 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼑𖽜 𖼡𖽔 𖼋𖽱𖾏.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 , 𖽐𖼡𖽪𖾏 𖼏𖽻 𖼽𖽙 𖼐𖽔 . 8 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐, 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼑𖽜 𖼡𖽔 𖼋𖽱𖾏? 𖼟𖽡 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏? 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏? 𖼟𖽡 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼞𖽪𖽙 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐? 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏?” 9 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾐 𖼐𖽜𖾐. 𖼞𖽪𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼳𖽡𖽠𖾑 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.” 10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼘𖽪?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽘 .
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼘𖽑𖽙𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼎𖽻𖾏 𖼟𖽡, 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼎𖽺𖾏 𖽐𖼮𖽦?” 12 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼮𖽑𖽳𖾑 𖼞𖽪𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼺𖽘𖾏, 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼽𖽔𖾑 𖼎𖽺𖾏 𖽐𖼮𖽦 𖼋𖽘𖾏. 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼅𖽡 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏.” 13 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼞𖽱𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏, 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼈𖽡𖽝𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏. 14 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼀𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 , 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼛𖽙𖾏 𖼋𖽔𖾏, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼻𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐,𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼷𖽷𖾑 𖼻𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽐𖼮𖽑𖽘𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼀𖽡𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽺𖾐 𖼙𖽔 𖽃𖽡𖾑 𖽃𖽔𖾏.” 15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼮𖽑𖽳𖾑 𖼽𖽙 𖼐𖽔 𖼗𖽙𖾏 , 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼺𖽘𖾏, 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼎𖽺𖾏 𖽐𖼮𖽦 𖼋𖽘𖾏. 16 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼄𖽔𖾐 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼖𖽑𖽘𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 .
17 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼡𖽙𖾏 𖼽𖽙 𖼐𖽔 𖼺𖽘𖾏, 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼀𖽳𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼷𖽷𖾐 𖼄𖽑𖽙𖾏.
