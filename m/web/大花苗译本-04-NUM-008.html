𖼽𖽔𖾏 𖼖𖽪𖽝 𖼮𖽡𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐
8
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽑𖽡𖾏 : 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽡𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 , 𖼳𖽤𖾏 𖼗𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼽𖽔𖾑 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 .” 3 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 . 𖼐𖽱𖾏 𖼮𖽡𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 , 𖼟𖽙𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏 𖽐𖼊𖽪𖾐 𖼙𖽪 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 . 4 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼷𖽜𖾏 𖼎𖽻𖾐 𖼐𖽡𖾑 𖼼𖽘𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 , 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖽐𖼎𖽙 𖽃𖽔𖾏 𖽐𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼁𖽘𖾏 𖼖𖽔𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖽐𖼊𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽃𖽔𖾏 𖼖𖽔𖾑 𖼷𖽜𖾏 𖼎𖽻𖾐 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 .
𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏
