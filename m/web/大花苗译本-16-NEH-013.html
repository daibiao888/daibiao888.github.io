𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽡𖽠 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼊𖽠𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽻 , 𖼺𖽱𖾑 𖼑𖽱𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼐𖽑𖽡𖾏 : “𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 . 2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖽐𖼎𖽙 𖽃𖽻𖾐 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼎𖽻𖾏 𖼀𖽔𖾑 𖼖𖽔 𖼟𖽙𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏.” 3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼋𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼐𖽡𖾑 , 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼇𖽺𖾐 𖼡𖽔𖾏.
𖼗𖽔𖾏 𖼊𖽙𖾑 𖼀𖽡𖾐 𖼽𖽔𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼻𖽻𖾑
