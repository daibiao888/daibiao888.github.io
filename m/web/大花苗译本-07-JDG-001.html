1
𖼽𖽙 𖼳𖽪𖾑 𖼽𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼺𖽪𖾐 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏?” 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼽𖽨 𖼊𖽔𖾏 𖼺𖽪𖾐 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽘𖾏.” 3 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 𖼊𖽻𖾏 𖽃𖽡𖾑, 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠 𖼊𖽻𖾏 𖽃𖽡𖾑． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 4 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼳𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼺𖽳 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐. 5 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼮𖽷𖾐 𖼽𖽔𖾏 𖼊𖽙𖾑 𖼐𖽡 ·𖼀𖽡𖾐 𖼺𖽳, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼳𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐. 6 𖼽𖽔𖾏 𖼊𖽙𖾑 𖼐𖽡 ·𖼀𖽡𖾐 𖼺𖽳 𖼏𖽘 𖼀𖽜𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽠𖾑 𖼯𖽢， 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼀𖽑𖽔𖾏 𖼊𖽪𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼑𖽡𖽝 𖼋𖽡𖾏 𖼷𖽑𖽹𖾏 𖼑𖽡𖽝 𖼊𖽠𖾏. 7 𖼽𖽔𖾏 𖼊𖽙𖾑 𖼐𖽡 ·𖼀𖽡𖾐 𖼺𖽳 𖼐𖽑𖽡𖾏: “ 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼋𖽡𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼊𖽠𖾏 𖼏𖽻 𖼞𖽪𖾑 𖼀𖽑𖽔𖾏 𖼊𖽪𖾏 𖼺𖽘𖾏, 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼞𖽡 𖼏𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽨𖾑 𖽃𖽔𖾑 𖼁𖽱 𖼡𖽜𖾑 𖼑𖽻𖾏, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼋𖽘𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼽𖽔𖾏 𖼊𖽙𖾑 𖼐𖽡 ·𖼀𖽡𖾐 𖼺𖽳 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐, 𖼐𖽱𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏.
8 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼙𖽻, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼐𖽑𖽙𖾏 𖽐𖼚𖽙 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼎𖽺𖾏 𖼮𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼙𖽻 𖼺𖽘𖾏. 9 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏, 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐、 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼊𖽡𖾐、 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏. 10 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼮𖽢𖾑 𖼐𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝, 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼳𖽷𖾏 𖼳𖽹𖾑 、 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼄𖽔𖾏 、 𖼊𖽔 𖼄𖽹𖾐. 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽪𖽝 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 ·𖼽𖽔𖾏 𖼀𖽔𖾑.
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐𖼊𖽡𖾐 𖼀𖽡 . 𖼊𖽡𖾐 𖼀𖽡 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 · 𖼳𖽡𖾑 𖼇𖽪. 12 𖼮𖽢𖾑 𖼖𖽳 𖼐𖽑𖽡𖾏: “𖼡𖽔 𖼋𖽱𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 · 𖼳𖽡𖾑 𖼇𖽪 𖼙𖽻 𖼺𖽘𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼄𖽔𖾐 𖼞𖽪𖾑 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐.” 13 𖼮𖽢𖾑 𖼖𖽳 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼮𖽡𖾑 𖼐𖽔 𖼺𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽂𖽙 𖼊𖽑𖽙 𖼐𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑, 𖼮𖽢𖾑 𖼖𖽳 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼷𖽑𖽹𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐. 14 𖼡𖽔𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼮𖽢𖾑 𖼖𖽳 𖽐𖼊𖽜 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼙𖽡𖽝𖾏, 𖼽𖽔𖾏 𖼺𖽔𖾐 𖽃𖽡𖾐 𖽐𖼡𖽜𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼗𖽙𖾏, 𖼮𖽢𖾑 𖼖𖽳 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏？” 15 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼖𖽳 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽜𖾑 𖼟𖽡 𖼺𖽘𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼟𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼊𖽡𖾐, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼊𖽑𖽡𖽝𖾐.” 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼸𖽷 𖼀𖽡 𖼋𖽻 𖼎𖽻𖾏 𖼐𖽱𖾏.
