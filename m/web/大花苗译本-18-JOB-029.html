𖼽𖽙 𖼀𖽳 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 , 𖼐𖽑𖽡𖾏 :
2 “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 ,
𖼺𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 .
3 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼊𖽠𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 ,
𖼞𖽪𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼎𖽻𖾐 𖼮𖽡𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 .
4 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 ,
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 , 𖽃𖽺𖾐 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 .
5 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 ,
𖼞𖽪𖾑 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑 .
6 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖽐𖼷𖽔𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼎𖽻𖾏 𖼞𖽔𖾏 𖼄𖽡𖾏 𖽐𖼚𖽙 ,
𖼈𖽳𖾐 𖼗𖽘𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼛𖽙𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 .
7 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼸𖽙 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 ,
𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 .
8 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 ,
𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 .
9 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼞𖽑𖽔𖾏 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼀𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 .
10 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼻𖽡𖽝𖾏 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖽐𖼎𖽪𖾏 .
11 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼑𖽻 𖼞𖽪𖾑 , 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ,
𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼞𖽪𖾑 , 𖼎𖽺𖾏 𖼷𖽻𖾑 𖼞𖽪𖾑 .
12 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼐𖽡𖽝𖾑 𖽐𖼊𖽜 𖽃𖽡𖾑 ,
𖽐𖼎𖽙 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽘𖾏 .
13 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼟𖽙𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼚𖽙𖾏 .
14 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 ,
𖼞𖽑𖽡𖽠𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 𖽐𖼎𖽙 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 .
15 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼅𖽔,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 .
16 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼑𖽡𖾏 𖼉𖽺𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼀𖽻𖾐 , 𖼞𖽪𖾑 𖼳𖽹𖾐 𖼑𖽔𖾏 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 .
17 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼐𖽑𖽡𖽝𖾑 𖼀𖽔𖾐 ,
𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼐𖽡𖽝𖾑𖽑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼦𖽔𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 .
18 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐( “𖽐𖼚𖽙 𖽐𖼞𖽔𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼚𖽙 𖼉𖽳𖾏”) ,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 , 𖼮𖽡𖽻𖾐 𖼺𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐 .
19 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼗𖽔 𖼸𖽙 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽻𖾐 ,
𖼀𖽡𖾑 𖼙𖽱 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏 𖽐𖼊𖽪𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼯𖽡 𖽐𖼊𖽻𖾏 .
20 𖼞𖽪𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽪𖾐 𖼮𖽑𖽦𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽑𖽪𖾐,
𖼞𖽪𖾑 𖼊𖽪𖾑 𖼐𖽜𖾑𖽑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼻𖽻𖾏 𖼯𖽡𖽙𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼐𖽑𖽪𖾐.
 
21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 , 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼋𖽙𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 .
22 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 ,
𖼞𖽪𖾑 𖼗𖽪𖾏 𖼺𖽱𖾏 𖼀𖽡𖾑 𖼙𖽱 𖽐𖼊𖽪𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .
23 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼑𖽘 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼎𖽺𖾏 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼑𖽘 .
24 𖼞𖽪𖾑 𖼚𖽙𖾏 𖽃𖽔𖾏 𖽐𖼷𖽑𖽷𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼚𖽑𖽡𖾏 𖼺𖽘𖾏 .
25 𖼞𖽪𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽹𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏, 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼇𖽻𖾑 𖼮𖽑𖽡𖽠𖾏 .
𖼞𖽪𖾑 𖼺𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼏𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 ,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 . ”
