𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼨𖽑𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏, 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏, 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖼛𖽙𖾏 𖼐𖽡𖾑, 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙. 2 𖼸𖽙 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽡 𖼳𖽷𖾏 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑, 𖼽𖽙 𖼳𖽔𖾑 𖼇𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼦𖽪𖾏. 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏, 3 𖼄𖽔𖾐 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏,𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼞𖽪𖾑 𖼚𖽜𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏. 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼗𖽪𖾏 𖼽𖽔𖾑 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼏𖽘 𖼀𖽜𖾑, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽔𖾑 𖼊𖽳𖾑𖽑.” 4 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼄𖽙 𖼙𖽠𖾏. 5 𖼸𖽙 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼼𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼘𖽪！ 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏.” 𖼽𖽦𖾑 𖼦𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,𖼟𖽡 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼡𖽔 𖼋𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏?” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼖𖽡𖾏 𖼇𖽻𖾏 𖼏𖽙 𖼘𖽪 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏.” 6 𖼽𖽦𖾑 𖼦𖽪𖾏 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏, 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼛𖽙𖾏 𖼘𖽡𖾑𖽑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏,𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼚𖽜𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏. 7 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼟𖽡 𖼮𖽵𖾑 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 , 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽜𖾑 𖽐𖼮𖽑𖽘𖾑 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼳𖽡𖾐 𖼀𖽡𖽝 𖽃𖽔𖾏 𖼮𖽡𖾑. 8 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼋𖽔𖾏, 𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽙𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐, 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼺𖽘𖾏. 9 𖼟𖽙𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼾𖽡𖾏 𖼺𖽱𖾏 𖼐𖽡 𖼀𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼾𖽡𖾏, 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼾𖽡𖾏. 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼳𖽡𖾐 𖼀𖽡𖽝, 𖼚𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼑𖽻𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼊𖽱𖾏.” 𖼐𖽱𖾏 𖽂𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼗𖽪𖾏, 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏. 11 𖼽𖽦𖾑 𖼦𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽡𖽝𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽪?
