（𖼞𖽑𖽙𖾐 4：1-9；𖼖𖽪𖾏 8：4-8）
13
𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑． 2 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 ， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑． 3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖼐𖽑𖽡𖾏：“𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼮𖽘𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼀𖽳𖾐 𖼮𖽘𖾑． 4 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖽃𖽡𖾑，𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 𖼙𖽠𖾏，𖼑𖽘𖾏 𖼐𖽻 𖼋𖽔𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏． 5 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼗𖽔𖾏 𖽐𖼚𖽡𖽝𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼦𖽡𖾐 𖼊𖽙𖾐，𖼘𖽑𖽘𖾑 𖼟𖽻𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝， 6 𖼈𖽡𖽝𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼮𖽑𖽘𖾑 𖼻𖽡𖽝𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐，𖼎𖽺𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 𖼊𖽘𖾏． 7 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏，𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼘𖽑𖽙𖾐 𖼗𖽙，𖼎𖽺𖾏 𖽐𖼷𖽻𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏． 8 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼻𖽻𖾏 𖼊𖽡𖾐，𖼎𖽺𖾏 𖼷𖽠𖾑𖽑 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖽐𖼀𖽠， 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖽐𖼀𖽠，𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽠． 9 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻．”
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏
（𖼞𖽑𖽙𖾐 4：10-12；𖼖𖽪𖾏 8：9-10）
10 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼖𖽺𖾑？” 11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏， 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼊𖽔𖾐，𖼦𖽡𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐． 12 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏， 𖼽𖽔𖾑 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏， 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼊𖽪𖾏 𖼽𖽔𖾑 𖼦𖽔𖾏 𖼙𖽠𖾏． 13 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽔𖾏 𖼖𖽔𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 ，
𖼑𖽻 𖼖𖽔𖾏 𖼑𖽻 𖼦𖽡𖾐 𖼊𖽻𖾏，𖼎𖽺𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏．
14 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑， 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏：
‘ 𖼅𖽡 𖼐𖽙𖾑𖽑 𖼯𖽡𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏， 𖼈𖽡𖽝𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏，
𖼑𖽔𖾏 𖼯𖽡𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼯𖽡𖽙𖾏 ，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐．
15 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼊𖽻𖾏 𖼛𖽙𖾏 𖼡𖽑𖽠𖾑 𖼮𖽡𖽝𖾏，
𖽐𖼮𖽠𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽻，
𖼡𖽜𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏，
𖽐𖼮𖽑𖽹𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼁𖽙𖾏，
𖽃𖽔 𖽐𖼀𖽳𖾏 𖼐𖽙𖾑𖽑，
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼯𖽷𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙，
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼞𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．’
16 𖼈𖽡𖽝𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏 ，𖼅𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼖𖽔𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏． 17 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖽃𖽔 𖼊𖽑𖽻𖾏 𖼖𖽡𖾏 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ，𖼽𖽔𖾑 𖼑𖽔𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾑 ， 𖼈𖽡𖽝𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏，𖼽𖽔𖾑 𖼑𖽻 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑 𖽃𖽡𖾑 ， 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑 ．”
𖼚𖽠𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏
（𖼞𖽑𖽙𖾐 4：13-20；𖼖𖽪𖾏 8：11-15）
18 “ 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼮𖽘𖾑 𖼐𖽡𖾑． 19 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽙𖾑𖽑 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 ，𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏， 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽱𖾑 𖼦𖽔𖾏 𖼙𖽠𖾏，𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 ． 20 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐，𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻 𖼞𖽘𖾐 𖼞𖽡𖾏， 𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏， 21 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 ， 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼊𖽔𖾐， 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼑𖽜 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔，𖼡𖽑𖽙𖾑 𖼯𖽢 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼢𖽻 𖼙𖽠𖾏 𖼋𖽘𖾏． 22 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼎𖽻𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏，𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼙𖽔 𖼑𖽻 𖼞𖽘𖾐 𖼞𖽡𖾏 ， 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽘𖾏 𖼮𖽡𖽝𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏，𖽐𖼮𖽷𖾐 𖼏𖽻 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖼮𖽑𖽵𖾑 𖼚𖽡𖾐 𖼐𖽱𖾏，𖼐𖽹𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼺𖽘𖾏 𖼷𖽷𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐． 23 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼎𖽻𖾏 𖼻𖽻𖾏 𖼊𖽡𖾐，𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼺𖽘𖾏，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼷𖽠𖾑𖽑 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖽐𖼀𖽠， 𖼅𖽔𖾏 𖼚𖽻𖾏 𖼯𖽡𖽻𖾏 𖽐𖼀𖽠，𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽠．”
𖼳𖽻𖾐 𖽐𖼎𖽙 𖼀𖽡𖾏 𖼷𖽑𖽻𖾑 𖼚𖽘𖾐
24 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼐𖽑𖽡𖾏：“𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼻𖽻𖾏 𖼮𖽘𖾑 𖼡𖽻𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐． 25 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽱𖾏 𖼙𖽠𖾏 ，𖼐𖽱𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼯𖽡𖽙𖾏 𖼀𖽡𖾏 𖼷𖽑𖽻𖾑 𖼚𖽘𖾐 𖼀𖽳𖾐 𖼳𖽡𖽠𖾏 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼳𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽡𖽝𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏． 26 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽨𖾐 𖼘𖽑𖽙𖾐 𖼷𖽠𖾑𖽑 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐，𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼀𖽡𖾏 𖼷𖽑𖽻𖾑 𖼚𖽘𖾐 𖼖𖽔𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏． 27 𖽐𖼞𖽔𖾏 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏：‘ 𖼮𖽵𖾑 𖼘𖽪 ！𖼟𖽡𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼀𖽳𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼘𖽪？𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼀𖽡𖾏 𖼷𖽑𖽻𖾑 𖼚𖽘𖾐 𖼐𖽡𖾑？’ 28 𖽐𖼞𖽔𖾏 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏：‘ 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽔𖾏．’ 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏：‘ 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼚𖽙𖾐 𖼺𖽘𖾏 𖼘𖽪 ？’ 29 𖽐𖼞𖽔𖾏 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏：‘ 𖼦𖽡𖾐 𖼅𖽪𖾏，𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼚𖽙𖾐 𖼀𖽡𖾏 𖼷𖽑𖽻𖾑 𖼚𖽘𖾐 ，𖼳𖽻𖾐 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼚𖽙𖾐 𖼊𖽑𖽡𖽝𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼙𖽠𖾏． 30 𖼯𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼮𖽘𖾏 𖼞𖽑𖽡𖽝𖾐 𖼘𖽑𖽙𖾐， 𖼷𖽷𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏． 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼋𖽘𖾏，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 ，𖼺𖽪𖾐 𖼄𖽔𖾐 𖼀𖽡𖾏 𖼷𖽑𖽻𖾑 𖼚𖽘𖾐 𖼚𖽙𖾐 𖼺𖽘𖾏， 𖽐𖼊𖽠𖾏 𖼮𖽡𖽝𖾏 ，𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼷𖽑𖽷𖾑 ，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼳𖽻𖾏 𖼳𖽻𖾐 𖼗𖽙𖾏 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼼𖽪 𖼮𖽡𖽝𖾏．’ ”
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼮𖽘𖾑 𖼻𖽻𖾐 𖼛𖽙𖾐 𖽃𖽔 𖼛𖽡𖾏
（𖼞𖽑𖽙𖾐 4：30-32；𖼖𖽪𖾏 13：18-21）
31 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼗𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼻𖽻𖾐 𖼛𖽙𖾐 𖽃𖽔 𖼛𖽡𖾏，𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽳𖾐 𖼎𖽻𖾏 𖼾𖽨𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐． 32 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼮𖽘𖾑 𖽃𖽺𖾐 𖼳𖽻𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔 𖼛𖽡𖾏 𖼊𖽔𖾐，𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽘，𖼘𖽑𖽙𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼻𖽻𖾐， 𖼘𖽑𖽙𖾐 𖼊𖽡𖽝𖾑 𖽐𖼊𖽻𖾏 𖼋𖽘𖾏， 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼋𖽔𖾏 𖼳𖽻𖾏 𖼉𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡．”
33 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼗𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 ：“ 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼳𖽪𖾏，𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾐 𖼊𖽱𖾏 𖽃𖽡𖾐 𖼳𖽜𖾐 𖽃𖽔𖾑 𖼋𖽪 𖼳𖽻𖾐 𖽐𖼚𖽙，𖼋𖽙𖾏 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼳𖽻𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽪𖾏 𖼋𖽘𖾏．”
（𖼞𖽑𖽙𖾐 4：33-34）
34 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏， 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏． 35 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏 ：
“ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏，
𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼁𖽱𖾏 𖼷𖽷𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼗𖽙𖾏 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏．”
𖼚𖽠𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏
36 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏，𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏，𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽑𖽡𖾏：“𖼄𖽔𖾐 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼟𖽪𖾏 𖼀𖽡𖾏 𖼷𖽑𖽻𖾑 𖼚𖽘𖾐 𖼚𖽠𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐．” 37 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼷𖽷𖾑 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼻𖽻𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐， 38 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐，𖼻𖽻𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼊𖽪𖾐，𖼀𖽡𖾏 𖼷𖽑𖽻𖾑 𖼚𖽘𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼊𖽪𖾐， 39 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼀𖽳𖾐 𖼀𖽡𖾏 𖼷𖽑𖽻𖾑 𖼚𖽘𖾐，𖼐𖽱𖾏 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐，𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐． 40 𖼄𖽔𖾐 𖼀𖽡𖾏 𖼷𖽑𖽻𖾑 𖼚𖽘𖾐 𖼳𖽻𖾏 𖼗𖽙𖾏，𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏，𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 41 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐，𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳，𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏， 42 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏，𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼟𖽡𖾏 𖼋𖽙𖾏 𖼐𖽡𖽝𖾑𖽑 𖼷𖽷𖾏 𖼞𖽹𖾑． 43 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 ，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼯𖽡𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼉𖽺𖾏 𖼞𖽪𖽙 𖽐𖼚𖽙 ， 𖼽𖽔𖾑 𖼮𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼮𖽡𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐．𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼑𖽻，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙
44 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼈𖽺𖾏 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼮𖽡𖽝𖾏，𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏， 𖼐𖽱𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼅𖽻𖾏 𖼋𖽺 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼺𖽘𖾏， 𖼅𖽔𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑．
45 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏， 𖽐𖼎𖽑𖽦𖾑 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑， 46 𖼐𖽱𖾏 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽃𖽡𖾑，𖼎𖽺𖾏 𖼅𖽻𖾏 𖼋𖽺 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼺𖽘𖾏，𖼅𖽔𖾏 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽃𖽡𖾑．
47 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼸𖽙𖾏 𖼟𖽪𖾏 𖼾𖽠𖾏 𖽐𖼀𖽳𖾏，𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼎𖽹𖾏 𖼳𖽡𖽠𖾑，𖼟𖽪𖾏 𖼳𖽻𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽻𖾐 𖼗𖽙， 48 𖼖𖽪𖾑 𖼸𖽙𖾏 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼋𖽘𖾏， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼡𖽪𖾏 𖽐𖼮𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪，𖼎𖽺𖾏 𖼼𖽻𖾏 𖼮𖽡𖽝，𖼺𖽹𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼗𖽙𖾏 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖼊𖽑𖽻𖾐 𖼮𖽡𖽝𖾏，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼯𖽡𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏． 49 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 ， 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼽𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，𖼺𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽠𖾏 𖼊𖽠𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏， 50 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼺𖽘𖾏，𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼟𖽡𖾏 𖼋𖽙𖾏 𖼐𖽡𖽝𖾑𖽑 𖼷𖽷𖾏 𖼞𖽹𖾑 ．”

53 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑，𖼎𖽺𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼺𖽘𖾏， 54 𖼺𖽷𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 ， 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 ， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼋𖽡𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾏 𖼟𖽳 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏 𖼐𖽔𖾑？ 55 𖼐𖽱𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼷𖽷𖾑 𖼄𖽪 𖼷𖽔𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼘𖽪𖾏？ 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼘𖽪𖾏？ 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽔𖾐 𖼞𖽙、𖼽𖽙 𖼳𖽡𖾑、𖼳𖽡𖾑 𖼄𖽞、 𖼽𖽨 𖼊𖽔𖾏 𖼘𖽪𖾏 ？ 56 𖼐𖽱𖾏 𖼅𖽔 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼘𖽪𖾏？ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼋𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏？” 57 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼢𖽻（𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼳𖽪𖾐 𖼑𖽡𖾏 ”）．𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐，𖼋𖽠𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼺𖽘𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏．” 58 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼮𖽡𖽻𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼋𖽘𖾏．
