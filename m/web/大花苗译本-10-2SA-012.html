𖼖𖽔𖾏 𖼡𖽑𖽳𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼾𖽔𖾏

15 𖼐𖽔 𖼊𖽗𖾑 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏. 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽂𖽪𖾑 𖼖𖽡 𖼽𖽔𖾏 𖼐𖽤𖾐 𖼑𖽜 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼳𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑. 16 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽜 𖼊𖽪𖾐 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑, 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏, 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼙𖽱 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐. 17 𖼐𖽱𖾏 𖼾𖽡𖾏 𖼷𖽷𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼗𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼐𖽱𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪, 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽘𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏. 18 𖼸𖽙 𖼋𖽘𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼣𖽔𖾑 𖼾𖽻𖾏 𖼋𖽔 𖼗𖽠𖾏 𖼋𖽘𖾏. 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼯𖽡𖽝𖾏 𖼺𖽷𖾐, 𖼀𖽡𖾐 𖼞𖽑𖽻𖾑 𖼐𖽱𖾏 ，𖼐𖽱𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼑𖽻 𖼀𖽡𖾐 𖼗𖽪𖾏, 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏， 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼐𖽱𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼖𖽺𖾑?” 19 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏, 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼑𖽪 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖼐𖽑𖽡𖾏: “𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏 𖼐𖽔𖾑 𖼖𖽔𖾏?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏.” 20 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼚𖽜𖾑 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏, 𖼘𖽑𖽪𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼐𖽑𖽘𖾑 𖼺𖽘𖾏, 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏, 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽜𖾐 𖼉𖽔𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼋𖽘𖾏 𖼦𖽻𖾏 𖼋𖽘𖾏. 21 𖼉𖽻 𖼷𖽻𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼯𖽡𖽝𖾏 𖼺𖽷𖾐, 𖼟𖽡 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏, 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼟𖽡 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏.” 22 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼯𖽡𖽝𖾏 𖼺𖽷𖾐, 𖼞𖽪𖾑 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽱𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼦𖽡𖾐 𖼋𖽔. 23 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼖𖽔𖾑 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼺𖽷𖾐, 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼗𖽙 𖼖𖽺𖾑? 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽙𖾑 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏.”

𖼦𖽔𖾏 𖼊𖽻𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼙𖽻
（𖼷𖽑𖽷𖾏 𖽃𖽡𖾐 20：1 - 3）
26 𖼽𖽙 𖼽𖽔 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑. 27 𖼽𖽙 𖼽𖽔 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “ 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑, 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖽃𖽻𖾐 𖼙𖽻. 28 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼏𖽙 𖼋𖽔𖾏, 𖼮𖽪𖾐 𖼾𖽡 𖼟𖽻𖾏 𖼙𖽻𖾏 𖽐𖼊𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼨𖽑𖽪𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑.” 29 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼙𖽻, 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑. 30 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏( “𖼞𖽡𖾏 𖼄𖽱𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼄𖽔𖾐 𖼖𖽳 𖼞𖽑𖽗𖾑 ”，𖼄𖽔𖾐 𖼖𖽳 𖼞𖽑𖽗𖾑 𖼟𖽪𖾏 𖼄𖽡𖾐 𖼖𖽳 𖼞𖽪𖽾𖾑𖼯𖽡𖽙𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖽐𖼷𖽷𖾏 𖼄𖽙𖾑 𖼘𖽙， 𖼟𖽪𖾏𖼽𖽔𖾏 𖼄𖽞𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼮𖽵𖾑 𖼳𖽺 𖽐𖼷𖽷𖾏 ) 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖼞𖽪𖾐, 𖼞𖽪𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖽃𖽡𖾐 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳, 𖼎𖽺𖾏 𖼎𖽑𖽹𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑． 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑 𖽐𖼊𖽻𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏． 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽙 𖼙𖽻 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐. 31 𖼄𖽔𖾐 𖽐𖼚𖽙 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖽃𖽔𖾏 𖼊𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼳𖽻𖾐、 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼖𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼡𖽻𖾐、𖽐𖼎𖽙 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼊𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼑𖽡 𖼋𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏， 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼞𖽑𖽔𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏.
