8
𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏, 𖼦𖽡𖾐 𖼨𖽑𖽪𖾏 𖼀𖽡𖾐 𖽐𖼎𖽙 𖼅𖽻𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 𖼀𖽡𖾐 𖼖𖽺𖾑?” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐. 2 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽪𖾏 𖼟𖽪𖾏 𖼀𖽘𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼋𖽡𖽝𖾏 𖼺𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼘𖽪? 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼳𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼽𖽡𖾑 𖼳𖽡𖽝𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼚𖽡𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼘𖽪? 3 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖽂𖽙 𖼖𖽡 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼽𖽡𖾑 𖼀𖽳 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽳𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽺𖾑?” 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑, 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼾𖽘𖾏 𖼺𖽘𖾏.
4 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖽐𖼎𖽙 𖼷𖽷𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽡𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼋𖽡𖾏 𖼛𖽡𖾏, 𖼦𖽪𖽺𖾑 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼋𖽘𖾏 𖼈𖽡𖽝𖾏, 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼺𖽷𖾐. 5 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖽐𖼎𖽙 𖼺𖽪𖾑 𖼞𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽜𖾑 𖼅𖽡 𖼄𖽔𖾐 𖼳𖽡𖽝𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼑𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼋𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽙 𖼐𖽔 𖼐𖽔𖾑. ” 6 𖼺𖽪𖾑 𖼞𖽙 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼳𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽙 𖼐𖽔 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖽐𖼚𖽡𖽝𖾏 𖼋𖽡 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽘𖾏 𖼖𖽔𖾏? 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼄𖽔𖾐 𖼳𖽡𖽝𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼸𖽻𖾏 𖼏𖽙 𖼐𖽡𖾑 𖼑𖽻𖾏 𖼘𖽪𖾏?” 7 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼐𖽑𖽡𖾏: “𖼻𖽻𖾏 𖼯𖽡𖽙𖾏 ! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼳𖽡𖾑 𖼀𖽔𖾑𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽙 𖼐𖽔 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽐𖼎𖽙 𖼡𖽠𖾏 𖼞𖽡𖾏 𖼁𖽙𖾏 𖽐𖼚𖽠𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼄𖽙𖾐.” 8 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼀𖽑𖽡𖾐 𖼐𖽪𖾐 𖼽𖽡𖾑 𖼖𖽳, 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 , 𖼀𖽑𖽡𖾐 𖼐𖽪𖾐 𖼽𖽡𖾑 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼺𖽱𖾏 𖼺𖽪𖾑 𖼞𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼊𖽡𖾐 𖼐𖽱𖾏. 9 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼀𖽑𖽡𖾐 𖼐𖽪𖾐 𖼽𖽡𖾑 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽪 𖼺𖽷𖾐 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼐𖽡𖾑.”
10 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼳𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽙 𖼐𖽔 𖼊𖽑𖽡𖽝𖾐 𖼏𖽙 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽢𖾑 𖼞𖽙, 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼟𖽪𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼊𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼉𖽻𖾏. 11 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼎𖽑𖽻𖾏 𖼐𖽙 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼽𖽙 𖼀𖽡𖾐 𖼦𖽔𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐, 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼖𖽪𖾑 𖼾𖽡 𖼸𖽻𖾏 𖼏𖽙 𖼸𖽙𖾏 𖼞𖽑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽘𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼏𖽙. 12 𖼳𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽙 𖼐𖽔 𖼏𖽘 𖼀𖽜𖾑, 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼄𖽡𖾐 𖼊𖽡𖽝𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽙 𖼐𖽔, 𖼮𖽡𖽠𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼏𖽙 𖼺𖽘𖾏.
13 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖼎𖽑𖽻𖾏 𖼳𖽡𖾑 𖼖𖽡𖽝 𖼺𖽷𖾑 𖼖𖽪𖾑 𖽃𖽔 𖼼𖽘 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼺𖽷𖾐 𖼗𖽙, 14 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼺𖽪𖾑 𖼞𖽙 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼑𖽪𖾏 𖼐𖽱𖾏: “𖼺𖽪𖾑 𖼞𖽙 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏、𖼊𖽪𖾐 𖼗𖽻 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏?” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏、𖼊𖽪𖾐 𖼗𖽻 𖼳𖽤𖾏 𖼯𖽡𖽻𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖽐𖼷𖽷𖾏 𖼺𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏. 15 𖼮𖽡𖾑 𖼊𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼺𖽪𖾑 𖼞𖽙 𖽐𖼎𖽙 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼚𖽙𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼳𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽙 𖼐𖽔 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖽐𖼚𖽡𖽝𖾏 𖼋𖽡 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽘𖾏 𖼖𖽔𖾏? 𖼟𖽡 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼄𖽔𖾐 𖼳𖽡𖽝𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖽃𖽡𖾑 𖼑𖽻𖾏 𖼘𖽪𖾏? 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼳𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽙 𖼐𖽔 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑.” 16 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼚𖽙 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼗𖽻 𖼮𖽡𖽝𖾏, 𖼎𖽑𖽻𖾏 𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽐𖼎𖽙 𖼡𖽠𖾏 𖼞𖽡𖾏 𖼁𖽙𖾏 𖽐𖼚𖽠𖾏 (“𖽐𖼚𖽠𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼖𖽔𖾏 𖼡𖽑𖽔𖾐”) 𖼺𖽪𖾑 𖼞𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐. 17 𖼎𖽺𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖼀𖽑𖽡𖾐 𖼐𖽪𖾐 𖼽𖽡𖾑 𖼖𖽳 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑, 𖼊𖽔𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽘𖾏.
𖼳𖽡𖾑 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽙 𖼐𖽔 𖼊𖽻𖾏 𖼊𖽔𖾏
