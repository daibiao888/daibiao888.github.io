𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑
（𖼊𖽑𖽹𖾏 24：1-2；𖼖𖽪𖾏 21：5-6）
13
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 ，𖼟𖽡 𖼑𖽔𖾏， 𖼋𖽡𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽡𖽝 ！ 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽡𖽝！” 2 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽡𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖾑 𖼘𖽪？ 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 ，𖼷𖽑𖽔𖾏 𖼮𖽡𖽙𖾐 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏．”
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏
（𖼊𖽑𖽹𖾏 24：3-14；𖼖𖽪𖾏 21：7-19）
3 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽡𖽙𖾐 𖼞𖽗𖾐 𖼖𖽗 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽱𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 ， 𖼀𖽡𖾐 𖼊𖽳、 𖼽𖽔𖾐 𖼞𖽙、𖼽𖽙 𖼦𖽗、 𖼣𖽔𖾑 𖼊𖽳 𖼖𖽡𖽝， 𖼑𖽪 𖼐𖽱𖾏 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏： 4 “ 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐， 𖼊𖽑𖽻𖾑 𖼋𖽱 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 ？ 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑， 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏？” 5 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏， 𖽐𖼮𖽑𖽹𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽵𖾑 𖼅𖽡． 6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼽𖽔𖾑 𖼞𖽑𖽳 𖼅𖽳𖾏 𖼎𖽷𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼐𖽑𖽡𖾏：‘ 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪 ’，𖼎𖽺𖾏 𖼽𖽔𖾑 𖼮𖽑𖽵𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐． 7 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏， 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏， 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽐𖼮𖽡𖽻𖾏 𖼺𖽷𖾐． 8 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼉𖽺𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏， 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖽙 𖼉𖽺𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏， 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼅𖽔𖾏 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏，𖼅𖽔𖾏 𖼮𖽷𖾑 𖼑𖽡𖽝𖾏 𖼮𖽷𖾏 𖼳𖽡𖽻𖾏，𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼳𖽠𖾑 𖼁𖽱𖾏． 9 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼅𖽡 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼑𖽪𖾏， 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼅𖽡， 𖼅𖽡 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐， 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏． 10 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏，𖼄𖽻𖾏 𖼻𖽻𖾏 𖼺𖽪𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐． 11 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼅𖽡 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼉𖽻 𖼷𖽻𖾏， 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼊𖽳𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼅𖽡， 𖼅𖽡 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼅𖽡， 𖼟𖽪𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾． 12 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏 、 𖼉𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼊𖽪𖾐 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽔， 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐，𖼟𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼋𖽔． 13 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼅𖽡， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽜𖾏 𖼸𖽙 𖼡𖽘𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽨𖾏．”
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽑𖽪𖾐 𖼺𖽻𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼘𖽑𖽙𖾐
（𖼊𖽑𖽹𖾏 24：15-28；𖼖𖽪𖾏 21：20-24）
14 “𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽡 𖼳𖽪𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 、𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖽃𖽡𖾑， 𖼳𖽠𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼳𖽠𖾑（ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡𖽠 𖼋𖽡𖾏 𖽐𖼊𖽠𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼀𖽻𖾐 ），𖼊𖽑𖽻𖾏 𖽃𖽡𖾑， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼽𖽔𖾑 𖼏𖽘 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼙𖽠𖾏； 15 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖼏𖽪𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏， 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏． 16 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖼮𖽑𖽙𖾏． 17 𖼸𖽙 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 ，𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖽃𖽡𖾑 ， 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 ． 18 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽙𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼯𖽻𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏． 19 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔， 𖼟𖽪𖾏 𖼋𖽳 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼗𖽙𖾏 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑， 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼗𖽙 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏． 20 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼦𖽡𖾐 𖼊𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑， 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼺𖽹𖾑 𖼊𖽻𖾏， 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑． 21 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏： ‘ 𖼑𖽔𖾏 𖼘𖽪！ 𖼮𖽡𖾑 𖼊𖽪 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑！ 𖼑𖽔𖾏 𖼘𖽪！𖼮𖽡𖾑 𖼊𖽪 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 ！’ 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽜． 22 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 、𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑，𖼽𖽔𖾑 𖼋𖽔𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏， 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼮𖽑𖽵𖾑，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼺𖽹𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽑𖽵𖾑 ． 23 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏， 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼺𖽪𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏．”
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏
（𖼊𖽑𖽹𖾏 24：29-31；𖼖𖽪𖾏 21：25-28）
24 “ 𖼸𖽙 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾐， 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，
𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼊𖽘𖾏， 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼮𖽡𖾏，
25 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼖𖽔𖾑 𖼀𖽻𖾐 𖼗𖽙，
𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖽃𖽔𖾏 𖼼𖽙．
26 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 ， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏， 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖽐𖼚𖽻 𖼮𖽡𖾏 ． 27 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏， 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽐𖼊𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼊𖽡𖾐， 𖼨𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼺𖽹𖾑 𖽃𖽡𖾑 𖼦𖽡 𖼸𖽱 𖼋𖽔𖾏．
𖼽𖽔𖾑 𖼯𖽷𖾏 𖼻𖽙𖾑
（𖼊𖽑𖽹𖾏 24：32-35；𖼖𖽪𖾏 21：29-33）
28 “𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼾𖽻 𖼊𖽪𖾑 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑： 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼘𖽑𖽘𖾑 𖽃𖽔𖾑 𖽐𖼷𖽔 𖼘𖽑𖽘𖾑 𖽃𖽔 𖽐𖼚𖽻𖾏， 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖽐𖼊𖽪𖾏 𖼳𖽙𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏． 29 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏， 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼁𖽙𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼗𖽙𖾏 𖼐𖽡𖽙𖾐 𖼢𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼋𖽘． 30 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡， 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼮𖽑𖽡𖽻𖾏 𖼺𖽷𖾐， 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏． 31 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼊𖽡𖾐 𖼛𖽙𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏 ， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼛𖽙𖾏 ．”
𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐
（𖼊𖽑𖽹𖾏 24：36-44）
32 “𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑， 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐， 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼊𖽪𖾐， 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐， 𖼯𖽔 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖼀𖽻𖾐 𖼊𖽔𖾐． 33 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 ，𖼽𖽔𖾑 𖼯𖽷𖾏 𖼻𖽙𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼊𖽑𖽻𖾑 𖼋𖽱 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏． 34 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾑 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼺𖽘𖾏， 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼙𖽠𖾏， 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑， 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏， 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽪𖾐 𖼻𖽙𖾑 𖽃𖽔 𖼛𖽻𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽷𖾏 𖼻𖽙𖾑． 35 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽷𖾏 𖼻𖽙𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽐𖼞𖽔𖾏 𖼑𖽡𖾏 𖼮𖽵𖾑 𖼊𖽑𖽻𖾑 𖼋𖽱 𖼖𖽔𖾑 𖼗𖽙𖾏， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼋𖽘 𖼄𖽑𖽙𖾏， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼡𖽹𖾐 𖼡𖽔𖾏， 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑． 36 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏，𖼁𖽙𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏． 37 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏，𖼟𖽪𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏， 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽷𖾏 𖼻𖽙𖾑．”
