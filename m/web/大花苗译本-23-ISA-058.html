𖼊𖽪𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝
58
𖼟𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏,
𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐．
𖽐𖼎𖽙 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏,
𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.
2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 ,
𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
𖼺𖽱𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼦𖽡𖾐 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 ,
𖽐𖼎𖽙 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖽃𖽺𖾐 𖼙𖽔 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼊𖽪𖾏 𖼡𖽜𖾑, 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼖𖽺𖾑?
𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼽𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼟𖽡 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼖𖽺𖾑?”
𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽪𖾏 𖼡𖽜𖾑 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽃𖽺𖾐 𖼙𖽔 ,
𖽃𖽔𖾏 𖼊𖽔𖾑 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼑𖽜 𖼅𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐.
4 𖼑𖽔𖾏 𖼘𖽪! 𖼅𖽡 𖼊𖽪𖾏 𖼡𖽜𖾑, 𖼈𖽡𖽝𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖽐𖼎𖽙 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏,
𖼯𖽡𖽙𖾏 𖽃𖽔𖾏 𖼷𖽑𖽱𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖽐𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽪𖾏 𖼡𖽜𖾑,
𖼦𖽡𖾐 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖽐𖼎𖽙𖾑 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐.
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽪𖾏 𖼡𖽜𖾑, 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑 𖼟𖽙𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼯𖽡𖽙𖾏 𖼘𖽪?
𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽻𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼺𖽱𖾏 𖽐𖼊𖽻𖾏 𖼡𖽳𖾐, 𖼎𖽑𖽻𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼷𖽙𖾏 𖼮𖽑𖽻𖾑 𖼀𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼘𖽪?
𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼙𖽔 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼘𖽪?
 
6 𖼑𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼽𖽔𖾑 𖼚𖽠𖾏 𖼺𖽘𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏,
𖼚𖽠𖾏 𖼺𖽘𖾏 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏,
𖼟𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏,
𖼖𖽙𖾑 𖼺𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
7 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼟𖽡 𖼳𖽡𖽝𖾐 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏,
𖼷𖽹𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼟𖽪𖾏 𖼳𖽔𖾏 𖽐𖼮𖽡 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏,
𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖾑 𖼮𖽢𖾏 𖼄𖽔𖾐 𖼮𖽑𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽐𖼀𖽙 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼾𖽨𖾏 𖼑𖽜𖾏 𖼮𖽘𖾏 𖼦𖽡𖾐 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼙𖽠𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼎𖽻𖾐 𖼮𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽡𖾏 𖼺𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼑𖽡𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 ,
𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼄𖽙𖾐 𖼽𖽔𖾑 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏,
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽡 𖼊𖽡𖾑 𖼢𖽻𖾏.
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐,
𖼟𖽡 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 .”
 
𖼟𖽡 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼖𖽡𖾑 𖼡𖽔𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽐𖼎𖽙 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,𖼊𖽑𖽡𖽝𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳,
10 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽔𖾑 𖼊𖽱𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼊𖽻𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼟𖽡 𖼎𖽻𖾐 𖼮𖽡𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾏 𖽐𖼊𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼋𖽘 𖼐𖽑𖽪𖾐.
11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 ,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡, 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖽃𖽔𖾑 𖼷𖽑𖽘𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏．
𖼟𖽡 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼟𖽪𖾏 𖽐𖼊𖽙𖾑 𖽃𖽻𖾐,
