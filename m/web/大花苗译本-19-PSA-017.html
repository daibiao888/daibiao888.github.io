17
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑,

𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼋𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑.

3 𖼟𖽡 𖼙𖽔 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐,
𖼟𖽡 𖼐𖽡𖽙𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑．
𖼟𖽡 𖼨𖽑𖽙𖾑 𖽐𖼮𖽔𖾏 𖼞𖽪𖾑, 𖼈𖽡𖽝𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏,
𖼞𖽪𖾑 𖼖𖽪𖾐 𖼺𖽡𖽝𖾐 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏.
4 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽱𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏,
𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼾𖽨𖾏 𖼻𖽙𖾑 𖼾𖽨𖾏,
𖼦𖽡𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼊𖽠𖾏 𖼊𖽠𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑.
5 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐 𖼏𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼊𖽠𖾏 𖼐𖽜𖾑,
𖼞𖽪𖾑 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼚𖽡𖽝 𖼢𖽻.
 
6 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽔𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼊𖽱𖾐 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑, 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏.
7 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼗𖽙𖾏,
𖼟𖽡 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡, 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
 
8 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖽃𖽔 𖼅𖽔 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼤𖽔𖾏 𖼤𖽔𖾏 𖽃𖽔 𖼅𖽔,
𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼈𖽺𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
9 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼞𖽪𖾑,
𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼟𖽪𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑 𖽃𖽡𖾑.
 
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏.
11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼀𖽡𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼮𖽡𖽝𖾏, 𖽃𖽔𖾑 𖼅𖽔 𖼷𖽷𖾏 𖼖𖽪𖾑,
𖼖𖽔𖾑 𖼊𖽑𖽠𖾐 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼢𖽻 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼙𖽠𖾏.
12 𖼐𖽱𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽐𖼮𖽑𖽷𖾏 𖽐𖼚𖽔 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼄𖽔𖾐 𖼗𖽙,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖼸𖽺𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏.
 
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼸𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼢𖽻 𖼙𖽠𖾏, 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼮𖽨𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽪𖾐 𖼊𖽠𖾏 𖼊𖽠𖾑.
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼀𖽜𖾑 𖼛𖽡𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽔𖾑 𖼐𖽡𖽙𖾐 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽔𖾐,
𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼚𖽘𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏.
 
15 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼁𖽙𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏,
𖼞𖽪𖾑 𖼯𖽷𖾏 𖼗𖽙𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼼𖽘𖾏 𖼼𖽔𖾏 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼋𖽘𖾏.
