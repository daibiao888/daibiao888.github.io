𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼙𖽻 (“𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖼙𖽻”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼙𖽻 𖼑𖽡𖾏 𖼖𖽑𖽔𖾏 𖼺𖽻𖾑”), 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼮𖽡𖽝𖾏. 2 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖽐𖼮𖽔𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖽐𖼮𖽔𖾏 𖽃𖽡𖾑, 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏, 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼮𖽔𖾏 𖼊𖽔𖾐 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏. 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼊𖽪𖾏 𖼳𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏.
𖽐𖼊𖽻𖾏 𖼳𖽠𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐
