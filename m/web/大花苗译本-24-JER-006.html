𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！
𖼅𖽡 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼋𖽠𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ,
𖼐𖽡𖽙𖾐 𖼊𖽑𖽡 𖼞𖽙𖾑 𖼽𖽔𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐,
𖼐𖽡𖽙𖾐 𖼀𖽳 · 𖼦𖽔𖾐 𖼮𖽡𖾑 𖼖𖽡𖽾 𖼷𖽑𖽙𖾏 𖼀𖽑𖽳𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽐𖼎𖽙 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼋𖽔𖾏 𖼋𖽘 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼮𖽤𖾏 𖼑𖽔𖾏”).
2 𖼳𖽡 𖼣𖽔𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 (“𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖼖𖽡𖾑 𖼉𖽪𖾏”) 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼊𖽻𖾏 𖼄𖽔𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 .
3 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐 𖼚𖽔𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼉𖽺𖾏 𖽃𖽡𖾑,
𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 ,
𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼾𖽨 𖼯𖽦𖾏 .
4 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏．
𖼳𖽠𖾑 𖼗𖽙𖾏 𖼦𖽡𖾑！ 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼷𖽪𖾑 𖼟𖽪𖾏 𖼋𖽘 𖼐𖽑𖽪𖾐 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏．
𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏！ 𖼐𖽑𖽪𖾐 𖼢𖽹𖾐 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼋𖽘𖾏,
𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾑 𖽐𖼷𖽻𖾏 𖼮𖽤𖾐 𖽐𖼊𖽡𖾑 𖼋𖽘𖾏.
5 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼦𖽡𖾑！ 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾐 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏,
𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼟𖽻 .
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：
𖼅𖽡 𖼽𖽔𖾑 𖽐𖼊𖽙𖾑 𖽐𖼊𖽻𖾏,
𖼮𖽪𖾐 𖼡𖽔𖾏 𖼚𖽻𖾏 𖽃𖽘𖾑 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐,
𖽐𖼚𖽙 𖼙𖽻 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 .
7 𖼡𖽑𖽙𖾑 𖽃𖽻𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 𖽃𖽻𖾐 𖼗𖽙 ,
𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽐𖼀𖽻𖾏 𖼋𖽠𖾏 𖽐𖼚𖽪𖾏 𖼗𖽙 .
𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽙𖾑𖽑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽔𖾏,
𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖽐𖼎𖽙 𖼮𖽑𖽡𖽠𖾏 𖼀𖽡𖾏 𖼡𖽔𖾏 𖼄𖽙𖾐 𖼖𖽔𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
8 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 ,
𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼑𖽡𖽻𖾏,
𖽐𖼮𖽑𖽹𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐.
 
9 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑,
𖼞𖽡𖾑 𖼀𖽔𖾑 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼺𖽷𖾐 𖽃𖽡𖾑 𖼦𖽔𖾏 𖼙𖽠𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼚𖽡𖾏 𖼺𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐.
𖼟𖽡 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼯𖽡 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼚𖽡𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼚𖽡𖾏.
10 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼡𖽔 𖼋𖽱 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻 𖼖𖽺𖾑?
𖼑𖽔𖾏 𖼘𖽪！𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽙𖾑𖽑.
𖼑𖽔𖾏 𖼘𖽪! 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼙𖽔.
11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑, 𖽐𖼷𖽜𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.
𖼟𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽢𖾑𖽑 𖼎𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏，
𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼸𖽻𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑,
𖼎𖽑𖽻𖾏 𖼈𖽻𖾑 𖼸𖽙 𖼐𖽤𖾐,
𖼎𖽑𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼗𖽻𖾏 𖼸𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼄𖽔𖾐.
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖽐𖼎𖽙 𖼐𖽤𖾐,
𖼽𖽔𖾑 𖼎𖽙𖾑 𖼺𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 ．
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
13 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼳𖽻𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐, 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐 𖼦𖽠𖾑 𖼉𖽳 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑,
𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐.
14 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼞𖽑𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐,
𖼐𖽑𖽡𖾏:𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽘𖾏! 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽘𖾏!
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 .
15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼳𖽪𖾐, 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼢𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽻𖾏 𖼢𖽻 .
𖼞𖽪𖾑 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼢𖽻 𖼙𖽠𖾏.”
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼅𖽡 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼳𖽹𖾐 𖼑𖽔𖾏 ,
𖼑𖽪 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏,
𖼸𖽙𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐.
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼅𖽡 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼳𖽙𖾏,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
17 𖼞𖽪𖾑 𖼺𖽹𖾑 𖼮𖽘𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼅𖽡 , 𖼐𖽑𖽡𖾏 :
𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾐 𖼳𖽘𖾐．
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼑𖽻 .
18 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼘𖽪! 𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻,
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼘𖽪! 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽐𖼮𖽷𖾐 𖽃𖽡𖾑.
19 𖼊𖽡𖾐 𖼘𖽪！ 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻，
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑,
𖼟𖽪𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽳𖾐 𖽐𖼊𖽱𖾏 𖼷𖽷𖾏 𖼯𖽡𖽙𖾏.
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼳𖽪𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏.
20 𖼵𖽪𖾐 𖼳𖽤𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑,
𖽃𖽔𖾏 𖽐𖼊𖽻𖾏 𖼳𖽷𖾑 𖼊𖽑𖽔𖾏（𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖼻𖽻𖾐 𖼸𖽷 𖼁𖽠𖾏 ”） 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐,
𖼺𖽘𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪?
𖼅𖽡 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏,
𖼅𖽡 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 𖼖𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 .”
21 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽠𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 ,
𖼉𖽺𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐, 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑,
𖼼𖽷𖾏 𖼼𖽙 𖽐𖼎𖽙 𖼽𖽺 𖼗𖽔 𖼖𖽔𖾏 𖼛𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏.”
22 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼑𖽔𖾏 𖼘𖽪! 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼋𖽔𖾏,
𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽻𖾏 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑 ,
𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻.
23 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽙𖾏 𖼐𖽜𖾑𖽑 𖽐𖼎𖽙 𖼄𖽑𖽪𖾑,
𖼀𖽡𖾑 𖼙𖽻 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏, 𖼺𖽱𖾏 𖼳𖽡𖽠𖾑 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏.
𖼳𖽡 𖼣𖽔𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪 !
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 ,
𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾐, 𖼳𖽠𖾑 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼼𖽘𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼟𖽡 .”
24 𖼀𖽡𖾐 𖼐𖽙𖾑𖽑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽘𖾐 𖼄𖽻𖾏, 𖼋𖽡𖾏 𖼯𖽻 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼊𖽔𖾑 𖼀𖽡𖾐 𖼮𖽡𖽝𖾏,
𖼄𖽙𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼊𖽪𖾐 𖼷𖽷𖾏 𖼖𖽘𖾑.
25 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼎𖽹𖾏 𖼙𖽡𖽝𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏,
𖼚𖽻𖾐 𖼺𖽘𖾐 𖼊𖽪𖾏 𖼅𖽔𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏.
26 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪 (“𖼖𖽡𖾑 𖼉𖽪𖾏 “𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏”)！ 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐, 𖼦𖽡 𖽐𖼚𖽙 𖼎𖽻𖾏 𖼮𖽑𖽻𖾑 𖽃𖽔 𖼗𖽔𖾏．
𖼟𖽡 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼊𖽪𖾐 𖼐𖽡𖽝𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼳𖽦𖾑 𖼽𖽔𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼀𖽡𖾐.
 
27 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑, 𖼟𖽙𖾏 𖼟𖽡 𖼀𖽻𖾐，𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏.
28 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼅𖽻𖾏 𖼗𖽙𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
