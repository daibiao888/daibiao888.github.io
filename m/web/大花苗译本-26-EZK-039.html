𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼞𖽙𖾑 𖼞𖽳 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
39
“𖼟𖽡𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽙𖾑 𖼞𖽳 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖼐𖽑𖽡𖾏: 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑： 𖼑𖽔𖾏 𖼘𖽪, 𖼘𖽙 𖼳𖽷𖾑、 𖼄𖽡𖾐 𖼳𖽳 、 𖼊𖽑𖽪𖾐 𖼀𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽙𖾑 𖼞𖽳 𖼘𖽪！ 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐. 2 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼟𖽡, 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏, 𖼨𖽑𖽳𖾏 𖼟𖽡 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐. 3 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽡 𖼊𖽪𖾑 𖼐𖽑𖽜𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼇𖽳𖾏, 𖽐𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽡 𖼊𖽪𖾑 𖼈𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏. 4 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼸𖽻𖾏 𖼏𖽙, 𖼊𖽑𖽡𖽝𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡, 𖼽𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼢𖽻 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼄𖽔𖾐 𖼟𖽡 𖼎𖽻𖾏 𖼚𖽘𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏. 5 𖼟𖽡 𖼽𖽔𖾑 𖼢𖽻 𖼎𖽻𖾏 𖽐𖼎𖽘𖾏 𖼯𖽢𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏, 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼄𖽔𖾐 𖼞𖽙 , 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼐𖽡𖽙𖾐 𖼏𖽻𖾏 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫. 7 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏. 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑.” 8 𖼑𖽔𖾏 𖼘𖽪！ 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖼊𖽡𖾏 𖼈𖽳𖾏 𖼋𖽘𖾏, 𖼑𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑, 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏. 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏.
