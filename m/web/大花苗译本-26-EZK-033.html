𖼽𖽡𖾑 𖼳𖽡𖾑 𖼮𖽡𖽝𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑
（𖼮𖽡𖽝 3：16 - 21）
33
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 2 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼾𖽨𖾏 𖼞𖽪𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼟𖽙𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼗𖽙𖾏 𖼷𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼞𖽪𖽙 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖼖𖽪𖾑 𖼞𖽪𖽙 𖽃𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 . 3 𖼐𖽱𖾏 𖼁𖽙𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏, 4 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽙𖾑𖽑 𖼞𖽪𖾐 𖼳𖽘𖾐 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖽃𖽡𖾑, 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼖𖽻𖾐 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼾𖽔𖾏 ( “𖼾𖽔𖾏” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼮𖽑𖽘𖾑 ” ) 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏. 5 𖼐𖽱𖾏 𖼐𖽙𖾑𖽑 𖼞𖽪𖾐 𖼳𖽘𖾐 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐, 𖼐𖽱𖾏 𖼾𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 . 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑. 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼁𖽙𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼗𖽙𖾏 𖼸𖽙, 𖼦𖽡𖾐 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐, 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐, 𖼸𖽙 𖼟𖽪𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 , 𖼐𖽱𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽙 𖼈𖽡𖽝𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼳𖽦𖾑 ( “𖼾𖽔𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼮𖽑𖽘𖾑”)．
7 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼮𖽘𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼻𖽙𖾑, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏. 8 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼐𖽑𖽡𖾏: 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼘𖽪, 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔！ 𖼟𖽡 𖼽𖽡𖾑 𖼳𖽡𖾑 𖼮𖽡𖽝 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼇𖽳𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑 𖼋𖽔 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽙, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏． 9 𖼮𖽔𖾑 𖼟𖽡 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽻𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼺𖽷𖾑 𖼡𖽜𖾐, 𖼐𖽱𖾏 𖼋𖽔 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽙, 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼮𖽨𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼾𖽔𖾏.
