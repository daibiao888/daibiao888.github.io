𖼀𖽻𖾐 𖼘𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐
12
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼞𖽑𖽻𖾑 𖼅𖽡，𖼄𖽔𖾐 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼙𖽠𖾏 ，𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔，𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝． 2 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼾𖽻 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼸𖽪𖾏 𖼼𖽘𖾏，𖼅𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼚𖽑𖽡𖾏 𖼮𖽑𖽦𖾐，𖼟𖽙𖾏 𖼅𖽡 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏，𖼟𖽙𖾏 𖼅𖽡 𖼳𖽹𖾐 𖼑𖽔𖾏 𖽐𖼎𖽑𖽹𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏， 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏、 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 、𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏．
