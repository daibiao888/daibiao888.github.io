𖼽𖽔𖾐 𖼞𖽙 𖽐𖼮𖽑𖽹𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐
32
𖼽𖽔𖾐 𖼞𖽙 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼮𖽷𖾐 𖼐𖽱𖾏. 2 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖼦𖽔𖾐 𖼐𖽡𖽝𖾏 (“𖼄𖽔𖾐 𖼦𖽔𖾐 𖼐𖽡𖽝𖾏”𖼟𖽪𖾏 “𖽃𖽔𖾐 𖼸𖽻 𖼊𖽪𖾐 𖼏𖽙𖾏”).
3 𖼽𖽔𖾐 𖼞𖽙 𖼺𖽪𖾐 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽔𖾏 𖼳𖽡𖾑 𖽃𖽳𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼟𖽪𖾏 𖼽𖽡𖾐 𖼊𖽪𖽾𖾑 𖼊𖽡𖾐, 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐. 4 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼐𖽑𖽡𖾏: ‘𖼟𖽡 𖼡𖽑𖽜𖾑 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏. 5 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼑𖽨𖾏、𖼑𖽜𖾏 𖼖𖽔𖾑、𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏、𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏、𖼐𖽔𖾑 𖼐𖽡𖾑 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽙𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑, 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼅𖽳𖾏. ’ ”
6 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼅𖽻𖾏 𖼺𖽷𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼨𖽑𖽳𖾏 𖼚𖽻𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼋𖽔𖾏 𖼋𖽙𖾏 𖼟𖽡.” 7 𖼽𖽔𖾐 𖼞𖽙 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽳𖾏, 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽑𖽡𖽝𖾐 𖽐𖼀𖽙𖾏 𖼯𖽦𖾏、𖽐𖼀𖽙𖾏 𖼑𖽨𖾏、𖼘𖽙 𖼊𖽑𖽙𖾐 𖼇𖽺𖾐 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼸𖽻. 8 𖼐𖽑𖽡𖾏: “𖼽𖽡𖾑 𖼺𖽻𖾐 𖼮𖽔𖾑 𖼋𖽔𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼐𖽡𖾑, 𖽃𖽡𖾐 𖼸𖽻 𖼟𖽪𖾏 𖼳𖽙𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼏𖽘 𖼀𖽜𖾑. ” 9 𖼽𖽔𖾐 𖼞𖽙 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪,𖼟𖽡 𖼙𖽔 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:‘ 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼟𖽡 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼎𖽙 𖼾𖽨𖾏 𖼑𖽜𖾏 𖼮𖽘𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼻𖽻𖾏.’ 10 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑, 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼖𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏. 𖼞𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼋𖽡𖾏 𖼖𖽪𖾑 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼊𖽔𖾐, 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽪𖾑 𖼊𖽡𖽝𖾑 𖽃𖽔𖾐 𖼸𖽻 𖼋𖽘𖾏. 11 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼊𖽔𖾏 𖼞𖽪𖾑, 𖼐𖽤𖾐 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼖𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽔𖾏 𖼺𖽘𖾏. 12 𖼟𖽡 𖼙𖽔 𖼐𖽑𖽡𖾏: ‘𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼳𖽡𖽠𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐, 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖼾𖽱 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏. ’ ”
13 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾐 𖼞𖽙 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖽃𖽔𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏. 14 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼮𖽑𖽷𖾏,𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏, 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼾𖽘𖾏, 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼋𖽪𖾏 𖼞𖽡𖾑 𖼙𖽻𖾏 𖼾𖽘𖾏, 15 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼘𖽙 𖼊𖽑𖽙𖾐 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏, 𖼊𖽑𖽡𖽝𖾐 𖼣𖽔𖾑 𖼣𖽔𖾏, 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽨𖾏,𖼟𖽻𖾏 𖼋𖽪𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏, 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽜𖾏 𖼖𖽔𖾑 , 𖼟𖽻𖾏 𖼋𖽪𖾏 𖼡𖽔𖾑 𖼯𖽱𖾏 𖼀𖽡𖾏 𖼊𖽪𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑. 16 𖼞𖽑𖽡𖽠𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖽐𖼀𖽙𖾐, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼡𖽑𖽜𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼋𖽡𖾏 𖼅𖽻𖾏, 𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖽐𖼀𖽙𖾏 𖼦𖽡𖾏 𖼡𖽔𖾏 𖼮𖽡𖽝𖾏, 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽔𖾏 𖼮𖽡𖽝𖾏.” 17 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏，𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽻𖾏 𖼽𖽡𖾑 𖼺𖽻𖾐 𖽐𖼮𖽷𖾐 𖼟𖽡 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼑𖽪 𖼟𖽡 𖼐𖽑𖽡𖾏: ‘ 𖼟𖽡 𖼟𖽪𖾏 𖼾𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐? 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏? 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖽐𖼊𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼁𖽡𖽝 𖼘𖽪? ’ 18 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏: ‘ 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼽𖽔𖾐 𖼞𖽙， 𖼟𖽪𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼽𖽡𖾑 𖼺𖽻𖾐, 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐. ’ ” 19 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽔𖾐 𖼸𖽻、𖼷𖽷𖾐 𖼸𖽻 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼀𖽙𖾏 𖼯𖽦𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼅𖽻𖾏 𖽐𖼮𖽷𖾐 𖼽𖽡𖾑 𖼺𖽻𖾐 𖽃𖽡𖾑, 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏, 20 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏: ‘ 𖼟𖽡 𖼡𖽑𖽜𖾑 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐. ’ ” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏: “𖼞𖽪𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼊𖽡𖾏 𖼚𖽠𖾏 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐， 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑.” 21 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼺𖽪𖾐 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼙𖽠𖾏. 𖼄𖽑𖽙𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾐 𖼞𖽙 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏 𖼖𖽪𖾑 𖼾𖽡 𖽐𖼚𖽙.

22 𖼋𖽘 𖼄𖽑𖽙𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐, 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑, 𖼊𖽑𖽡𖽝𖾐 𖼟𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼋𖽡𖾏 𖼽𖽔𖾐 𖼀𖽙 𖽃𖽔𖾑 𖼟𖽱𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 . 23 𖼺𖽪𖾐 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽡𖾏 𖽃𖽻𖾐, 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼋𖽡𖾏 𖼅𖽻𖾏. 24 𖼛𖽡𖾏 𖼽𖽔𖾐 𖼞𖽙 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼋𖽘𖾏, 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖼡𖽑𖽻𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼁𖽙𖾏 𖼞𖽡𖾑. 25 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼳𖽠𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼎𖽺𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼀𖽔𖾐 𖽐𖼮𖽡𖽻𖾏 ．𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼀𖽔𖾐 𖽐𖼮𖽡𖽻𖾏 𖼎𖽠𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏. 26 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼮𖽡𖾏 𖼡𖽘𖾐 𖽐𖼊𖽪𖾐 𖼋𖽘, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼦𖽡𖾑.” 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾐 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼅𖽻𖾏.” 27 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼷𖽷𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏? “𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼽𖽔𖾐 𖼞𖽙.” 28 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽺𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼋𖽘𖾏, 𖼽𖽔𖾑 𖽐𖼷𖽷𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼞𖽑𖽔𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏.” 29 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼮𖽑𖽙 𖼄𖽔𖾐 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑.” 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼅𖽪𖾏 𖼑𖽪 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼏𖽵 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙. 30 𖼽𖽔𖾐 𖼞𖽙 𖼟𖽙𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼀𖽑𖽡𖾐 𖼐𖽪𖾐 𖼽𖽡𖾑 𖼖𖽳 (“𖼀𖽑𖽡𖾐 𖼐𖽪𖾐 𖼽𖽡𖾑 𖼖𖽳” 𖼟𖽪𖾏 “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 ”)， 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼚𖽑𖽪𖾏 𖼊𖽱𖾐 𖼚𖽑𖽪𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼊𖽻𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑.” 31 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼖𖽔𖾑 𖼋𖽠𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾐 𖼞𖽙 𖼇𖽳𖾑 𖼀𖽑𖽡𖾐 𖼐𖽪𖾐 𖼽𖽡𖾑 𖼖𖽳, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼡𖽘𖾑𖽑 𖼅𖽻𖾏 𖼞𖽡𖾑. 32 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼑𖽻𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼀𖽔𖾐 𖽐𖼮𖽡𖽻𖾏 𖼑𖽡𖾏 𖼳𖽠𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼀𖽔𖾐 𖽐𖼮𖽡𖽻𖾏 𖼐𖽔𖾑.
