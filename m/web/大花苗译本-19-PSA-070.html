𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽨𖾏 𖽐𖼀𖽘𖾏 𖼟𖽙𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
（𖽐𖼞𖽻𖾏 40：13 - 17）
70
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑；
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑．
2 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 𖼳𖽦𖾑 𖽃𖽡𖾑, 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑, 𖼑𖽡𖾏 𖼡𖽜𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
3 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾏 𖼦𖽔𖾏、𖽃𖽔𖾏 𖼦𖽔𖾏” 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼐𖽔𖾑 𖼑𖽡𖾏 𖼡𖽜𖾐.
 
4 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼟𖽡 𖽃𖽡𖾑, 𖼑𖽜 𖼟𖽡 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼙𖽔 𖼚𖽙𖾏,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼉𖽳 𖼟𖽡 𖼮𖽨𖾏 𖼅𖽳𖾏 𖽃𖽡𖾑, 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽑𖽡𖾏: “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐！”
5 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏.
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 ,
𖼟𖽡 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑, 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽳𖾑𖽑 𖼮𖽡𖽝𖾏！
