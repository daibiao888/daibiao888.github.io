13
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏.
𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐
2 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼺𖽙𖾐 𖼖𖽔𖾑 𖼙𖽻 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼟𖽪𖾏 𖼊𖽗𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏, 𖼐𖽱𖾏 𖼐𖽤𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽤𖾑, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏. 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽤𖾑, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼽𖽔𖾑 𖼐𖽑𖽤𖾑 𖼽𖽔𖾑 𖼳𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐. 4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐, 𖼮𖽡𖽠𖾑 𖼸𖽺𖾏 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽻𖾏, 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏. 5 𖼑𖽔𖾏 𖼘𖽪 ! 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽤𖾑 𖼽𖽔𖾑 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽪𖾑 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼇𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖽃𖽡𖾐 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖽃𖽔𖾏 𖼐𖽔 𖼳𖽡𖾏 𖽃𖽳𖾐 （“𖼐𖽔 𖼳𖽡𖾏 𖽃𖽳𖾐”𖼟𖽪𖾏 “𖼇𖽺𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫”）𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽱𖾏 𖼮𖽨𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.” 6 𖽃𖽔 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼐𖽑𖽡𖾏: “𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼚𖽑𖽪𖾏, 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼑𖽡 𖽃𖽺𖾐 𖽐𖼮𖽑𖽹𖾏, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽪 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑. 7 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽤𖾑 𖼽𖽔𖾑 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽡𖽠𖾑 𖼸𖽺𖾏 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽻𖾏, 𖼡𖽜𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖽂𖽪𖾏 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽻𖾏 𖼋𖽔 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼐𖽔 𖼳𖽡𖾏 𖽃𖽳𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐.”
8 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖼎𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼮𖽵𖾑 𖼘𖽪 ! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼷𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼻𖽻𖾏 𖼻𖽻𖾏 𖼡𖽑𖽔𖾐 𖼀𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼯𖽡𖽙𖾏 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖼐𖽡𖾑.” 9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖼗𖽪𖾏, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖽃𖽡𖾑, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼎𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑. 10 𖽃𖽔 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼏𖽘 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐, 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑.” 11 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖼘𖽪?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏.” 12 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽔𖾑 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑, 𖼷𖽔𖾑 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏? 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?” 13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽑𖽡𖾏: 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏. 14 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼎𖽑𖽻𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏, 𖼮𖽡𖽠𖾑 𖼸𖽺𖾏 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽻𖾏, 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼻𖽙𖾑.”
15 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼯𖽪𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼐𖽡𖽙𖾐, 𖼻𖽻𖾏 𖼑𖽜 𖼟𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏.” 16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼞𖽑𖽻𖾑 𖼞𖽪𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼟𖽡 𖼡𖽜𖾑 𖼑𖽻𖾏, 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼁𖽱 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐. 17 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽢𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑, 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑, 𖼀𖽡𖾐 𖼻𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡.” 18 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼑𖽪 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏, 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼯𖽡𖽙𖾏.” 19 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼮𖽑𖽷𖾏 𖽐𖼎𖽙 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 , 𖼐𖽡𖽙𖾐 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑, 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼖𖽔𖾏 𖼑𖽔𖾏， 20 𖼁𖽙𖾏 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼯𖽡 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼏𖽘 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼀𖽡𖾑 𖼗𖽺𖾏 𖼋𖽠𖾏 𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼙𖽠𖾏. 𖼄𖽔𖾐 𖼐𖽙 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
