𖼦𖽔𖾐 𖼀𖽔𖾑 𖼞𖽪 𖽐𖼊𖽠𖾑
𖼦𖽔𖾐 𖼀𖽔𖾑 𖼞𖽪 𖼑𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
1
𖼅𖽜 𖼡𖽑𖽜𖾐 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼦𖽔𖾐 𖼀𖽔𖾑 𖼞𖽪 𖼡𖽑𖽙𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽡𖾑.
2 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪𖾏! 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡 ,
𖼟𖽡 𖼦𖽡𖾐 𖼑𖽻 𖼊𖽡𖾐, 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑 𖽐𖼚𖽪𖾏 𖼐𖽡𖽝𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 ,
𖼟𖽡 𖼦𖽡𖾐 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼺𖽷𖾐.
3 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏?
𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼁𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼚𖽪𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼖𖽺𖾑?
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼊𖽠𖾑 𖽐𖼚𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼇𖽺𖾐 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼯𖽪 𖼯𖽪𖾏 𖼅𖽔𖾏.
4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼮𖽙𖾏 𖼛𖽔𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑,
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼢𖽳 𖼋𖽠．
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 𖼟𖽡𖾏 𖼎𖽙𖾑 𖼡𖽘𖾐 𖼎𖽙𖾏 𖼇𖽻𖾑.
 
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖼅𖽡 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙, 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏,
𖼟𖽙𖾏 𖼅𖽡 𖼏𖽘𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼏𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏,
𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏, 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽜 .
6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼟𖽪𖾏 𖼸𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽠𖾏 𖼊𖽠𖾑 𖽐𖼚𖽪𖾏 𖼸𖽳 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼷𖽹𖾑 𖽃𖽡𖾑, 𖼅𖽻𖾏 𖽐𖼷𖽻𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑,
𖼦𖽔𖾏 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖽐𖼎𖽙 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏.
8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜𖾏 𖼇𖽺𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾑 𖼇𖽜𖾏,
𖽐𖼚𖽪𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽳𖾑 𖼘𖽳𖾐,
𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼦𖽡𖾏 𖼨𖽑𖽹𖾏 𖼏𖽘 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼋𖽔𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽘 𖼇𖽺𖾏 𖼺𖽱𖾏 𖼚𖽘𖾑 𖼽𖽘𖾏 𖽐𖼎𖽑𖽦𖾑 𖼡𖽜𖾑 𖼑𖽻𖾏,
9 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼑𖽜 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼋𖽔𖾏,
𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼎𖽻𖾏 𖼊𖽑𖽔𖾏,
𖼳𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑 𖼦𖽡 𖼸𖽱 , 𖼮𖽡𖽻𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐.
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖽃𖽔𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼚𖽙𖾏,
𖼚𖽙𖾏 𖼙𖽻𖾏 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 ,
𖼮𖽪𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖽐𖼊𖽻𖾏 𖼦𖽔𖾏.
11 𖼞𖽑𖽡𖽠𖾏 𖼾𖽨𖾏 𖼚𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽔𖾏 𖼮𖽵𖾑 𖼳𖽺, 𖼺𖽱𖾏 𖼮𖽢𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼷𖽑𖽔𖾐 𖼮𖽑𖽡𖽻𖾏,
𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏.”
𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼞𖽪𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖽃𖽡𖾑 𖼘𖽪 !
𖼟𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖽃𖽔 𖽐𖼷𖽔 𖼷𖽹𖾏 𖼅𖽔𖾏 𖼋𖽘 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼀𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽻 𖼋𖽔 .
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼟𖽡 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ;
𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼘𖽪! 𖼟𖽡 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
13 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼦𖽡𖾐 𖼑𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼚𖽪𖾐,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼑𖽔𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼖𖽺𖾑?
𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖽐𖼡𖽙𖾏 𖼖𖽻𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼾𖽨𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏,
𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼖𖽺𖾑?
14 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼊𖽪𖾑 𖽐𖼀𖽳𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼞𖽘 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼞𖽳𖾑 𖽃𖽡𖾑 𖼖𖽺𖾑?
 
15 𖼮𖽢𖾑 𖼖𖽳 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼊𖽪𖾑 𖽐𖼡𖽜𖾏 𖽐𖼡𖽜𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽝𖾏,
𖼊𖽑𖽔𖾑 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼾𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼮𖽑𖽡 𖼙𖽔 𖼸𖽙𖾏 𖼳𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑.
16 𖼎𖽺𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏,
