10
𖼽𖽔𖾏 𖼀𖽡𖾐 𖼄𖽡𖾐 𖼖𖽳 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽙 𖼊𖽙𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼀𖽑𖽪𖾐 𖽂𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼊𖽑𖽙 𖼖𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼮𖽨𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼳𖽔𖾑 𖼄𖽡. 2 𖼊𖽑𖽙 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼋𖽻𖾏 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼊𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽱𖾏 𖼋𖽔 𖼺𖽘𖾏 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼳𖽔𖾑 𖼄𖽡.
𖼽𖽔𖾏 𖽃𖽳𖾐 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏
3 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽔𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏, 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽪𖾐 𖼋𖽻𖾏 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼊𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏. 4 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐, 𖼯𖽡𖽜𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼋𖽪𖾏 𖼡𖽔𖾑 𖼯𖽱𖾏 𖼀𖽡𖾏 𖼊𖽪𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼗𖽪𖾏 𖼙𖽻. 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼦𖽔𖾐 𖽂𖽙 𖼊𖽑𖽳 · 𖼽𖽔𖾏 𖽃𖽳𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏. 5 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼋𖽔 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼮𖽢𖾑 𖼄𖽞.
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼇𖽳𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
6 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼅𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼀𖽔𖾑 𖼖𖽡 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼺𖽷𖾑 𖼊𖽑𖽔𖾑 𖼖𖽪, 𖼊𖽑𖽡𖽝𖾐 𖼽𖽔𖾏 𖼖𖽔 𖼮𖽵𖾑 𖼳𖽺、𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼮𖽵𖾑 𖼳𖽺、𖼄𖽙𖾑 𖼽𖽔𖾏 𖼮𖽵𖾑 𖼳𖽺、𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽵𖾑 𖼳𖽺、𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽵𖾑 𖼳𖽺, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼇𖽳𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏. 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐,𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏. 8 𖼎𖽑𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼁𖽱𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼊𖽡𖾐,𖼊𖽻𖾏 𖼟𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏. 9 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖾏 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼋𖽔𖾏 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽑𖽡𖽝𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼾𖽡𖾏, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼎𖽺𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝.
10 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽝𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,𖼅𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼀𖽔𖾑 𖼖𖽡 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.” 11 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾑 𖼾𖽙 𖼙𖽔 𖼮𖽨𖾏 𖼅𖽡 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽔𖾏 𖼄𖽙𖾑 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽺𖾑? 12 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼖𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼄𖽔𖾐 𖼽𖽱𖽾 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼅𖽡 , 𖼅𖽡 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼅𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏. 13 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼅𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼮𖽨𖾏 𖼅𖽡 𖼋𖽘𖾏. 14 𖼅𖽡 𖼅𖽻𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼅𖽡 𖼺𖽹𖾑 𖼊𖽻𖾏 ! 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼮𖽨𖾏 𖼅𖽡 !” 15 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖽃𖽔𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖼯𖽔𖾏 𖼟𖽡 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼯𖽡𖽙𖾏 𖼀𖽡𖾐, 𖼯𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐.” 16 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺, 𖼎𖽺𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖽃𖽦𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼊𖽳𖾏.
17 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱, 𖼮𖽪𖾐 𖼾𖽡 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽡𖽝, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼮𖽪𖾐 𖼾𖽡 𖼐𖽡𖽙𖾐 𖼄𖽡𖾐 𖼺𖽷𖾑 𖼀𖽔𖾑. 18 𖼮𖽡𖾑 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼮𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼦𖽡 𖽐𖼷𖽷 𖼦𖽡𖾏 𖼊𖽳𖾏 𖼐𖽑𖽡𖾏: “𖼡𖽔 𖼋𖽱𖾏 𖼺𖽪𖾐 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼽𖽔𖾏 𖼄𖽞 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼡𖽔 𖼋𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽡𖽝 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏.”
