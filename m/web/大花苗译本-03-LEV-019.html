𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢 𖽐𖼎𖽙 𖼖𖽠𖾑 𖼞𖽡𖾑
19
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “ 𖼟𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 3 𖼅𖽡 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏 . 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 4 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼚𖽔𖾏 𖼊𖽪𖾑 𖼁𖽠𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽨𖾏 𖼘𖽑𖽡𖾑 𖼁𖽠𖾏 𖼼𖽘𖾏 . 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
5 𖼅𖽡 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼟𖽙𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼷𖽹𖾏 𖼳𖽻𖾏 . 6 𖼋𖽡𖾏 𖼀𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼑𖽻𖾏 . 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏 𖼮𖽡𖽝𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 . 7 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼮𖽔𖾑 𖼎𖽺𖾏 𖼑𖽻𖾏 , 𖼋𖽡𖾏 𖽐𖼡𖽹𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼙𖽔 𖼷𖽹𖾏 𖼳𖽻𖾏 . 8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼑𖽻𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 . 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼇𖽳𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 .
9 𖼐𖽡𖽙𖾐 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼳𖽻𖾏 𖼤𖽜𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼤𖽜𖾏 𖼋𖽘𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽪𖾐 𖼙𖽡𖽝𖾐 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼉𖽜 𖼀𖽻𖾐 𖼙𖽠𖾏. 10 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼚𖽡𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔𖾏 𖼅𖽔 𖼑𖽜𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼋𖽘𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼉𖽜 𖼀𖽻𖾐 𖼙𖽠𖾏, 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑． 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
11 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼺𖽘𖾐 𖽃𖽔𖾏 𖼸𖽜 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽑𖽡𖽝𖾏 𖽐𖼎𖽙 𖼷𖽑𖽡𖽝𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑 . 12 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 , 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 . 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
13 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽔𖾏 𖼐𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐． 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼟𖽪𖾏 𖼞𖽻𖾏 𖼑𖽡𖾏 𖽐𖼡𖽜𖾏 𖼼𖽙 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 . 14 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼗𖽘𖾐 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼮𖽙𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽠𖾏 𖼎𖽻𖾏 𖼊𖽡𖾑 𖼛𖽜 𖼐𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼮𖽔𖾑 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
15 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼞𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼋𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 , 𖼮𖽔 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼨𖽑𖽢𖾐 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙 . 16 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽻𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼼𖽷𖾏 𖼼𖽙 𖽃𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 , 𖼸𖽙 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽔 (“𖼮𖽑𖽡𖽠𖾏 𖼋𖽔”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖽐𖼚𖽱𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽘𖾑 ”) 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
17 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 , 𖼮𖽔𖾑 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼑𖽜 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼾𖽔𖾏 . 18 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼼𖽷𖾏 𖼁𖽻𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼳𖽪𖾐 𖼟𖽡 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼅𖽡 𖼼𖽷𖾏 𖼼𖽙 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼾𖽨𖾏 . 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
19 𖼅𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 . 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾐 𖼮𖽘𖾏 𖼯𖽦𖾏 𖼦𖽡𖾏 𖼡𖽠𖾏 𖼳𖽤𖾏 𖼮𖽘𖾑 , 𖼦𖽡𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾐 𖼼𖽘𖾐 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏 𖼯𖽡𖽙 𖼎𖽻𖾏 𖼟𖽡 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼼𖽘𖾐 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 .

32 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼇𖽻𖾏 𖼋𖽠𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 , 𖼟𖽡 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼁𖽜 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
33 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼐𖽱𖾏 . 34 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐 , 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼑𖽔𖾏 𖼺𖽱𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼺𖽱𖾏 𖼾𖽨𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼙𖽔 𖽃𖽔𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 . 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
35 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 , 𖼐𖽡𖽙𖾐 𖼊𖽪𖾑 𖽐𖼮𖽔𖾏、𖼸𖽱𖾏 𖼞𖽡𖾏、𖼖𖽪𖾑 𖼊𖽱𖾑、𖼖𖽪𖾑 𖼳𖽜𖾐 , 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏 . 36 𖼅𖽡 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼸𖽱𖾏 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳、𖽃𖽔𖾑 𖼖𖽪𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼮𖽤𖾐 𖽐𖼞𖽳、𖼳𖽜𖾐 𖼊𖽱𖾏 𖼟𖽪𖾏 𖼮𖽤𖾐 𖽐𖼞𖽳、𖼖𖽔𖾑 𖼊𖽳𖾑 𖼟𖽪𖾏 𖼮𖽤𖾐 𖽐𖼞𖽳 . 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼙𖽔 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼗𖽙𖾏 . 37 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 、 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 . 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .”
