10
𖽃𖽺𖾐 ! 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑, 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖽃𖽡𖾑,
2 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼊𖽻𖾏,
𖼦𖽔𖾏 𖼙𖽠𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼞𖽑𖽡𖽠𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏,
𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽔𖾏 𖼗𖽙𖾏 .
3 𖼸𖽙 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐, 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙,
𖼯𖽡𖽹𖾏 𖽃𖽡𖾑, 𖼅𖽡 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑?
𖼅𖽡 𖼽𖽔𖾑 𖼏𖽘 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼖𖽺𖾑?
𖼅𖽡 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼮𖽡𖽝𖾏 𖼖𖽺𖾑?
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽔𖾑 𖼊𖽻𖾏 𖼞𖽡𖾑 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼦𖽡 𖼯𖽡𖾏,
𖼢𖽻 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼈𖽡𖽝𖾏, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼳𖽻𖾏 𖼎𖽙𖾑 ,
𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼳𖽤𖾐 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖽐𖼞𖽻𖾏.
𖼽𖽔𖾏 𖼳𖽪 𖽃𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼮𖽡𖾑 𖼀𖽜𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐
5 𖽃𖽺𖾐! 𖼽𖽔𖾏 𖼳𖽪 𖼘𖽪𖾏 , 𖼞𖽪𖾑 𖼊𖽪𖾑 𖼡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏.
6 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏,
𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽡𖾑 ,
𖼦𖽔𖾏 𖼙𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏, 𖼦𖽔𖾏 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏,
𖼯𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖼺𖽱𖾏 𖼯𖽪𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏 𖽃𖽘𖾑 𖽃𖽔 𖼗𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐.
7 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼖𖽻𖾐 𖼺𖽘𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽺𖾑?
9 𖼮𖽢𖾑 𖼖𖽳 𖼐𖽙 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼮𖽢𖾑 𖼮𖽡𖾑 𖼄𖽡𖾐 𖼳𖽷𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼦𖽔𖾐 𖼄𖽔𖾐 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼀𖽔 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
10 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼳𖽤𖾐 𖼸𖽙 𖼋𖽡𖾏 𖼞𖽪𖽙 𖼟𖽪𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼁𖽠𖾏 𖽃𖽡𖾑 𖼋𖽘𖾏.
𖼋𖽡𖾏 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏 𖼮𖽡𖽻𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼁𖽠𖾏.
11 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽐𖼎𖽙 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽡𖾏 𖼯𖽤𖾏,
𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼯𖽡𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼎𖽙 𖼁𖽠𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽺𖾑?”
12 𖼮𖽵𖾑 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼨𖽑𖽢𖾐 𖼽𖽔𖾏 𖼳𖽪 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐 𖼾𖽨𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏.” 13 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏:
“𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑, 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼑𖽡𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐,
𖼞𖽪𖾑 𖼮𖽑𖽹𖾏 𖼺𖽘𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼑𖽡𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐,
𖼦𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼮𖽷𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽪𖾐 𖼋𖽻, 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼦𖽡𖾏 𖼢𖽜𖾏 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐.
14 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼳𖽤𖾐 𖼸𖽙 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾐 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼉𖽺𖾏 𖽃𖽡𖾑,
𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖾏 𖼳𖽤𖾐 𖼸𖽙 𖼉𖽳𖾏 𖼐𖽻,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 ,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼡𖽳𖾏 𖼐𖽻 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼺𖽘𖾏,
