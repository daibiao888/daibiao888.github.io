𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐
8 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾏 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖼾𖽡𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼚𖽻𖾏 𖼖𖽑𖽡𖾏. 9 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏, 𖼾𖽻 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏, 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡 𖼀𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽙 𖽃𖽡𖾑. 10 𖼽𖽔𖾐 𖼀𖽡𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽔𖾑 𖼖𖽪𖽾 𖽐𖼷𖽜𖾑 𖼐𖽱𖾏, 𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽔𖾏 𖼐𖽱𖾏, 𖼎𖽺𖾏 𖼦𖽔𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐. 11 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏《 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏》 𖼮𖽡𖽝𖾏. 12 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽪𖾏 𖼏𖽵 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽪𖽙 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼸𖽙 𖼚𖽻𖾐 𖼷𖽑𖽷𖾏.” 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼋𖽘𖾏.

32 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽡𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽡𖾐 𖼮𖽢𖾑𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖼊𖽑𖽗𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 33 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼀𖽜𖾐 𖼳𖽡𖽻𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼟𖽻𖾏 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏. 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼳𖽔𖾑 𖼟𖽪𖾏 𖼺𖽔𖾐 𖼊𖽪 𖼑𖽡𖾏 𖽐𖼷𖽑𖽹𖾏. 34 𖼽𖽙 𖼊𖽑𖽗𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼾𖽻 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽂𖽪𖾑 𖼳𖽡𖾑 𖼽𖽔𖾐 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑. 35 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏. 𖼽𖽙 𖼊𖽑𖽗𖾐 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼺𖽘𖾑 𖼡𖽔 𖼸𖽷 𖼖𖽪𖾑 𖽃𖽔 𖼛𖽻𖾏 ( 𖼟𖽪𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 ). 36 𖼽𖽙 𖼊𖽑𖽗𖾐 𖼑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏, 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏《𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽡𖽙𖾏》 𖼮𖽡𖽝𖾏. 37 𖼐𖽡𖽙𖾐 𖼋𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼁𖽱𖾏 𖼟𖽙𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽡𖾏 𖼳𖽱 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼀𖽡𖾐 𖼮𖽢𖾑 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏. 38 𖼽𖽙 𖼊𖽑𖽗𖾐 𖼇𖽳𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏, 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼊𖽡𖾐 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏. 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽷𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
