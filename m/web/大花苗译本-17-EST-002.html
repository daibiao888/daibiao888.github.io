𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼺𖽪𖽺 𖼖𖽪𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼻𖽻𖾏 𖼋𖽘𖾏 , 𖼎𖽺𖾏 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖽂𖽔𖾐 𖼳𖽷 𖼊𖽑𖽡 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖽃𖽡𖾑 , 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽐𖼡𖽜𖾏 𖽐𖼊𖽠𖾑 𖽐𖼎𖽻𖾏 𖼐𖽱𖾏 . 2 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼦𖽡𖾐 𖼋𖽳𖾏 𖼑𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽑𖽦𖾑 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 . 3 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼷𖽑𖽔𖾏 𖼚𖽻𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼚𖽔𖾏 𖼳𖽪𖾑 𖼳𖽔𖾑 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖽙𖾐 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼷𖽪𖾏 𖼺𖽠𖾏 𖼺𖽘𖾏 𖼳𖽡𖾑 𖼞𖽹𖾑 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 , 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼯𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖼚𖽜𖾑 𖼚𖽑𖽪𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 . 4 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼑𖽔𖾏 𖼏𖽻 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼺𖽹𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 , 𖼘𖽪𖾑𖽑 𖽂𖽔𖾐 𖼳𖽷 𖼊𖽑𖽡 .” 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 , 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 .
5 𖼳𖽪𖾑 𖼳𖽔𖾑 𖼙𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖽐𖼷𖽷𖾏 𖼄𖽙 𖼊𖽡𖾐 𖼞𖽹𖾐 , 𖼟𖽪𖾏 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖾑 𖼳𖽷𖾏 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼘𖽔𖾑𖽑 ， 𖼳𖽷𖾏 𖼄𖽺𖾐 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑 , 𖼽𖽔𖾏 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 . 6 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼀𖽔𖾑 𖼀𖽡𖾐 𖼖𖽪𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡 𖼀𖽪𖾏 𖼮𖽢 𖼐𖽡 𖼺𖽔𖾐 𖼞𖽑𖽡𖽠𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽦𖾑 𖼞𖽙𖾑 𖼐𖽡 𖼽𖽔𖾐 (“ 𖼽𖽦𖾑 𖼞𖽙𖾑 𖼐𖽡 𖼽𖽔𖾐 ” 𖼎𖽺𖾏 𖽐𖼷𖽷𖾏 “𖼽𖽙 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑 ”) 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼦𖽔𖾏 𖼗𖽙𖾏 , 𖼮𖽡𖾑 𖼺𖽷𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 . 7 𖼄𖽙 𖼊𖽡𖾐 𖼞𖽹𖾐 𖼊𖽪𖾑 𖼾𖽨 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼾𖽡𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼦𖽔𖾐 𖼊𖽔𖾏 𖼳𖽔𖾑 (𖽃𖽔𖾑 𖼡𖽜𖾐 𖼚𖽠𖾏 𖽐𖼷𖽷𖾏 “𖼽𖽡𖾑 𖼺𖽷𖾑 𖼊𖽑𖽡𖽝 ” )， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 . 𖼖𖽜𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽡𖾑 𖼻𖽻𖾏 𖼼𖽘𖾏 𖼯𖽡𖾏 𖼎𖽺𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 , 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 𖼋𖽔 𖼺𖽘𖾏 , 𖼄𖽙 𖼊𖽡𖾐 𖼞𖽹𖾐 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼳𖽻𖾏 𖽃𖽔𖾏 𖼾𖽨𖾏 𖽐𖼷𖽑𖽹𖾏 .
