6 𖼘𖽪𖾏 𖼊𖽳 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏, 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼁𖽙𖾐 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏. 7 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼑𖽻𖾏 𖼦𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼚𖽻𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼅𖽙 𖽃𖽔𖾑 𖽐𖼊𖽪. 𖼘𖽪𖾏 𖼊𖽳 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼋𖽔𖾏 𖼡𖽑𖽜𖾐 𖼐𖽱𖾏 𖼀𖽑𖽙𖾑 𖽃𖽔𖾏 𖼘𖽑𖽱𖾏 𖼺𖽘𖾏 𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑. 8 𖼸𖽙 𖼚𖽔𖾏 𖼋𖽘 𖼄𖽑𖽙𖾏 𖼋𖽘𖾏, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼯𖽷𖾏 𖼗𖽙𖾏 𖼦𖽡𖾏 𖽐𖼷𖽜𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑, 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙 𖼅𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏. 9 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “ 𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏?” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼘𖽪𖾏 𖼊𖽳 ， 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼾𖽡𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼑𖽜𖾏 𖼮𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽡𖾏 𖼞𖽪𖾑, 𖼟𖽡 𖼅𖽔𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾐 𖼁𖽙𖾐, 𖼷𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐.” 10 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪! 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼅𖽳𖾏 . 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼢𖽳 𖼋𖽠𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼦𖽡𖾏 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏、 𖼳𖽻𖾏, 𖼟𖽡 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼊𖽱𖾏 𖼡𖽔𖾏 𖼎𖽻𖾏. 11 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪! 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏, 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼙𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼀𖽻𖾐 𖼟𖽡 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽡 𖼙𖽔𖾏 𖼀𖽡 𖼙𖽻. 12 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼑𖽜𖾏 𖼮𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽡𖾏 𖼋𖽡𖽝, 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼊𖽡𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑 𖼺𖽷𖾐. 13 𖼟𖽡 𖼄𖽑𖽙𖾏 𖼐𖽔𖾑 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼀𖽡 𖼟𖽡 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼙𖽻 𖼋𖽪𖾏 𖽃𖽔𖾏, 𖼎𖽺𖾏 𖼯𖽔𖾏 𖼐𖽱𖾏! 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏, 𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼏𖽵 𖼗𖽪𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼋𖽘𖾏 𖼞𖽪𖾑 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼎𖽙𖾑 𖼟𖽡, 𖼟𖽡 𖼮𖽔 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼀𖽱𖾏 𖼸𖽙 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏.”
14 𖼘𖽪𖾏 𖼊𖽳 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽡𖽝𖾏 𖼯𖽜𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼷𖽑𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼋𖽘𖾏. 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏.” 15 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼘𖽪𖾏 𖼊𖽳 𖼐𖽑𖽡𖾏: “𖼚𖽠𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏 𖼟𖽪𖾏 𖽐𖼀𖽔𖾏 𖼗𖽙𖾏.” 𖼐𖽱𖾏 𖼚𖽠𖾏 𖼗𖽙𖾏 𖼊𖽑𖽔𖾑 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏, 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼉𖽺𖾏 𖼊𖽻𖾏 𖼚𖽻𖾏 𖼈𖽘𖾐 (“𖼚𖽻𖾏 𖼈𖽘𖾐” 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼮𖽡𖽻𖾐 𖼞𖽡𖾏 )𖽃𖽔𖾑 𖼖𖽪𖾐 𖼅𖽙 𖼮𖽑𖽳𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏. 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻 𖼙𖽠𖾏 𖼋𖽘𖾏. 16 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼘𖽪𖾏 𖼊𖽳 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼁𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼁𖽙𖾐 𖼐𖽑𖽡𖾏: “ 𖼖𖽔𖾐 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪! 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏?” 𖼘𖽪𖾏 𖼊𖽳 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼁𖽙𖾐 𖽃𖽡𖾐 𖼎𖽙𖾏. 17 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼄𖽔𖾐 𖼚𖽻𖾏 𖼈𖽘𖾐 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼅𖽙 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼟𖽡 𖽃𖽔𖾐 𖼁𖽙𖾐.” 18 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼁𖽙𖾐 𖼐𖽑𖽡𖾏: “𖼖𖽔𖾐 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪! 𖼟𖽡 𖼮𖽔 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼋𖽙𖾏, 𖼑𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼖𖽡𖾏 𖼯𖽤𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖽃𖽔𖾏 𖼦𖽡𖾐 𖼊𖽡𖽝𖾑 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼳𖽙𖾏 𖼮𖽡𖽝𖾏.”
