𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐
13
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼷𖽷𖾏 𖼅𖽔𖾏． 2 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼷𖽹𖾏 𖼡𖽑𖽔𖾏 ，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏；𖽃𖽡𖾑 𖼟𖽪𖾏 𖼙𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽹𖾏 𖼡𖽑𖽔𖾏 ， 𖼦𖽡𖾐 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼎𖽺𖾏 𖼷𖽹𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼋𖽘． 3 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼊𖽠𖾏 𖼎𖽪𖾑 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖽐𖼊𖽠𖾏 𖼎𖽪𖾑 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐，𖼅𖽡 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏 𖼟𖽪𖾏 𖽃𖽦𖾐． 4 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐 𖽃𖽔𖾏 𖼈𖽻𖾑， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼐𖽤𖾐 𖼈𖽻𖾏 𖼸𖽘𖾏 𖼮𖽑𖽡𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼑𖽪𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼉𖽳 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏． 5 𖼅𖽡 𖼟𖽱𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼦𖽠𖾑 𖼉𖽳 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐，𖼽𖽔𖾑 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼟𖽡 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼗𖽔𖾏 𖼟𖽡 𖼺𖽘𖾏．” 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼮𖽙𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼚𖽔𖾐 𖼐𖽑𖽡𖾏：
“ 𖼮𖽵𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏，
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏，
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？”
