4 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 5 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫， 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽔𖾏 , 𖼟𖽪𖾏 𖼞𖽪𖾐、𖼑𖽡𖽝𖾏、𖼋𖽻𖾏、 6 𖼺𖽙𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑、𖼺𖽙𖾑 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼮𖽑𖽦𖾐、𖼺𖽙𖾑 𖼅𖽘𖾏、𖼚𖽻𖾐 𖼮𖽑𖽷𖾏、 7 𖼞𖽡𖾑 𖼙𖽻𖾏 𖼾𖽘𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖼟𖽪𖾏 𖼚𖽻𖾏 𖼖𖽡𖽝𖾐、𖼳𖽡𖽠𖾑 𖼚𖽡𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏、𖼷𖽻𖾏 𖼮𖽢 𖼋𖽠𖾏、 8 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐 , 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼚𖽜𖾑 𖽐𖼎𖽙 𖼳𖽤𖾏 ， 9 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼄𖽔𖾐 𖼐𖽻𖾐 𖼖𖽡𖽝𖾐 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 , 𖼟𖽪𖾏 𖼘𖽙𖾏 𖼎𖽑𖽹𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼊𖽳 𖽐𖼎𖽙 𖼛𖽔𖾏 𖽐𖼀𖽙 𖽐𖼎𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 . ”
（𖼋𖽠𖾏 39：32 - 43）
