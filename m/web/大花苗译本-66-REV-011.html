14 𖽃𖽔𖾐 𖼼𖽘𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖽃𖽡𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏；𖼑𖽔𖾏 𖼘𖽪！ 𖼷𖽷𖾐 𖼼𖽘𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐 𖽃𖽡𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏．
𖼷𖽑𖽙𖾑 𖼳𖽤𖾏 𖼋𖽪𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾑 𖼊𖽳𖾏
15 𖼳𖽤𖾏 𖼗𖽜𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼷𖽑𖽙𖾑 𖼞𖽡𖾑 𖼊𖽳𖾏，𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽑𖽡𖾏：
“𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽪𖽙 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾑 𖼞𖽪𖽙 𖼙𖽠𖾏 𖼋𖽘𖾏，
𖼐𖽱𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙𖾏 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏．”
16 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼚𖽻𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼗𖽻 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐，𖼟𖽪𖾏 𖼼𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑，𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 17 𖼐𖽑𖽡𖾏：
“ 𖼮𖽵𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 ，𖼟𖽪𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐、 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼘𖽪！
𖼀𖽡𖾐 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼟𖽡！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽘𖾏．
18 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐，
𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏．
𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼖𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏；
𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑 、𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐、𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑，
𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼘𖽑𖽙𖾐 𖼳𖽻𖾐 𖼊𖽡𖾏 𖼣𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼖𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏，
𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼖𖽻𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽻𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼋𖽘．”
19 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏， 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙，𖼐𖽱𖾏 𖼖𖽔𖾑 𖼼𖽪𖾏 𖼦𖽡𖾏 𖼊𖽑𖽹𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏；𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽔𖾏 𖼺𖽙𖾐 𖼀𖽡𖾏 𖼗𖽺𖾏 𖽐𖼊𖽘𖾏、𖼅𖽔𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑、𖼅𖽔𖾏 𖼺𖽙𖾐 𖽐𖼊𖽑𖽡𖾏、𖼅𖽔𖾏 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏、 𖼅𖽔𖾏 𖼑𖽘 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐．
