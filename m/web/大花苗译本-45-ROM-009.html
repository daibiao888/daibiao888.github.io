9
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼊𖽪 𖽐𖼚𖽙 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝，𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑；𖼅𖽔𖾏 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖼚𖽠𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼷𖽱𖾏 𖼖𖽑𖽠𖾑，𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏． 2 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼊𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼦𖽡𖾐 𖼳𖽙𖾏 𖼅𖽔𖾏 𖼊𖽳𖾏 𖼄𖽙𖾐． 3 𖼑𖽜 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡，𖼞𖽪𖾑 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖼑𖽜𖾏 𖼮𖽘𖾏， 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 ， 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼺𖽘𖾏，𖼞𖽪𖾑 𖼖𖽔𖾏 𖼯𖽪𖾏 𖼯𖽡𖽙． 4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼊𖽪𖾐 𖼑𖽡𖾏 𖼙𖽻 𖼋𖽪𖾏 、𖽐𖼚𖽻 𖼮𖽡𖾏、𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 、𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 、𖼡𖽑𖽘𖾑 𖼢𖽔𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 ， 𖽐𖼎𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽡𖽝． 5 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 ， 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼐𖽑𖽡𖾏，𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ． 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼯𖽢 𖼋𖽡𖾏 𖼟𖽻𖾏 𖼮𖽘𖾑 𖼦𖽡𖾏 𖼇𖽔𖾑，𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 ． 𖽃𖽔𖾏 𖼄𖽞！
6 𖼈𖽡𖽝𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏： 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 𖼺𖽘𖾏 ． 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 ， 𖼦𖽡𖾐 𖼾𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾐 𖼐𖽜𖾐； 7 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏，𖼯𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏； 𖼮𖽔 𖼅𖽔𖾏 𖼟𖽪𖾏 “ 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 ，𖼋𖽡𖽝𖾏 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏．” 8 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼾𖽨 𖼗𖽙𖾏， 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 ， 𖼮𖽔 𖼅𖽔𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼏𖽵 𖼎𖽻𖾏 𖽃𖽡𖾑， 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼺𖽙𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏． 9 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼏𖽵 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑：“𖼸𖽙 𖼦𖽡𖾏 𖼑𖽡𖽻𖾏 𖼐𖽔𖾑 𖼯𖽡𖽹𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼋𖽔𖾏 ， 𖼺𖽔𖾐 𖼖𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐．” 10 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾐， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑，𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼽𖽡𖾑 𖼺𖽔𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑 𖼋𖽘𖾏． 11 （𖼊𖽪𖾐 𖽃𖽔𖾐 𖽐𖼷𖽹𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼘𖽑𖽘𖾑 𖼅𖽔𖾏 𖼺𖽷𖾐，𖼻𖽻𖾏 𖼮𖽷𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼺𖽷𖾐，𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼺𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪𖾏，𖼟𖽪𖾏 𖼑𖽔𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐）， 12 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼖𖽡𖾏 𖼀𖽳 𖼮𖽢𖾑 𖼐𖽑𖽡𖾏：“𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼏𖽷𖾏．” 13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏： “𖼽𖽔𖾐 𖼞𖽙 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 ， 𖼽𖽡𖾑 𖼺𖽻𖾐 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼳𖽪𖾐． ”
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐
14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ， 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑 ？ 𖼋𖽡𖽝𖾏 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳？𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼾𖽙！ 15 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏：
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼡𖽔 𖼋𖽱𖾏，𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼡𖽔 𖼋𖽱𖾏，
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏， 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 ．”
16 𖼎𖽑𖽻𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼑𖽔𖾏 𖼗𖽙𖾏，𖼖𖽡𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼼𖽙， 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼊𖽔𖾐． 17 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼅𖽔𖾏 𖼗𖽪𖾏 𖽐𖼎𖽙 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏： “𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏，𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏，𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏 𖼡𖽑𖽔𖾐 𖽐𖼷𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼚𖽡𖽝𖾏 𖼋𖽻𖾏．” 18 𖼑𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼗𖽙𖾏，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼡𖽔 𖼋𖽱𖾏，𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼡𖽔 𖼋𖽱𖾏，𖽐𖼮𖽑𖽠𖾏 𖼟𖽙𖾏 𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏，𖼡𖽔 𖼋𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼼𖽔𖾏．
𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜
19 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽷𖾐 𖼖𖽺𖾑 ？ 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼐𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼺𖽷𖾐 𖼖𖽺𖾑？” 20 𖼽𖽡𖾏！ 𖼟𖽡 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑，𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 ， 𖼞𖽔𖾐 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼊𖽱𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼘𖽪𖾏 ？ 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏，𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼷𖽷𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑？” 21 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏，𖼎𖽑𖽻𖾏 𖽃𖽡𖾐 𖼊𖽑𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼀𖽔𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑，𖼎𖽺𖾏 𖼀𖽔𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼳𖽷𖾐 𖼘𖽪𖾏 ？ 22 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 ，𖼢𖽳 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏， 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 ， 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼽𖽔𖾑 𖼖𖽻𖾐 𖽃𖽡𖾑， 23 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼢𖽳 𖼋𖽠𖾏 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼎𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐， 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽙𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽃𖽡𖾑． 24 𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏，𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼎𖽑𖽻𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽔𖾐 ，𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼊𖽑𖽡𖽝𖾐 ． 25 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 《𖼦𖽙 𖼳𖽡𖾑 𖽃𖽔𖾐》 𖽐𖼊𖽠𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏：
“ 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏，

“ 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼉𖽻𖾏 𖼏𖽙 𖼮𖽵𖾑 𖼑𖽜 𖼀𖽡𖾐 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏 ，
𖼀𖽡𖾐 𖽐𖼷𖽙𖾑 𖽐𖼷𖽙𖾑 𖼺𖽱𖾏 𖼺𖽙𖾐 𖼊𖽙𖾐 𖼄𖽔𖾐、𖽃𖽳 𖼄𖽙 𖼖𖽔𖾑𖽃𖽡𖾐 𖼼𖽘𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏．”
30 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ，𖼀𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑 ？ 𖼸𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼅𖽡 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ，𖼎𖽺𖾏 𖼎𖽙𖾑 𖼊𖽻𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳， 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼮𖽜 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳． 31 𖼈𖽡𖽝𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼑𖽡𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔𖾏 𖼦𖽡𖾐 𖼸𖽙 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼑𖽡𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳． 32 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼐𖽔𖾑 𖼖𖽺𖾑 ？ 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼎𖽑𖽦𖾑，𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖽐𖼎𖽑𖽦𖾑， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 𖼎𖽻𖾏 𖼮𖽡𖽙𖾑 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼊𖽠𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏． 33 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：
