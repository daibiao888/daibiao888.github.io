89
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼨𖽑𖽪𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼄𖽔𖾐 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼡𖽑𖽔𖾐 𖼸𖽙 𖼉𖽻𖾏 𖼷𖽑𖽷𖾏.
2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼽𖽔𖾑 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏.
3 𖼟𖽡 𖼙𖽔 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼺𖽹𖾑 𖼮𖽡𖽝𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖽐𖼎𖽙 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼏𖽵 𖼗𖽪𖾏 , 𖼐𖽑𖽡𖾏:
4 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽡𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼉𖽻𖾏 𖼷𖽑𖽷𖾏.” （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑,
𖼐𖽡𖽙𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 .
6 𖼐𖽡𖽙𖾐 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡 𖽐𖼮𖽔 𖼖𖽺𖾑?
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼺𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽺𖾑?
7 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼷𖽻𖾏 𖽐𖼮𖽑𖽹𖾏,
𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐.
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼉𖽻𖾏 𖼏𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼺𖽱𖾏 𖼟𖽡?
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐.
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼚𖽻𖾐 𖼺𖽘𖾐.
9 𖼟𖽡 𖼞𖽳𖾑 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼦𖽡 𖽐𖼚𖽙 𖽃𖽡𖾑, 𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 .
10 𖼟𖽡 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼖𖽔𖾑 𖼦𖽔𖾐 𖼀𖽳, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼟𖽡 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼖𖽠𖾑 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏.
11 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼁𖽡𖽝 , 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽔𖾏 𖼟𖽡 𖼁𖽡𖽝 ,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽃𖽡𖾑, 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏.
12 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖽐𖼎𖽙 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼮𖽑𖽘𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏,
𖼊𖽑𖽔𖾑 𖼀𖽳 𖽐𖼎𖽙 𖼦𖽳 𖼄𖽞 𖼑𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼙𖽔 𖼚𖽙𖾏.
13 𖼟𖽡 𖼅𖽔𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 , 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐.
14 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼎𖽙 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏,
𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
 
15 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼎𖽺𖾏 𖼙𖽔 𖼨𖽑𖽪𖾏 𖼟𖽡 , 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏！
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽐𖼚𖽙 .
16 𖼐𖽱𖾏 𖼸𖽻 𖼑𖽜 𖼟𖽡 𖽐𖼷𖽷𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏,
𖼑𖽜 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐.
17 𖼟𖽡 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼙𖽔 𖼀𖽡𖾐, 𖼀𖽡𖾐 𖼊𖽪𖾑 𖼞𖽪𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐.
18 𖼀𖽡𖾐 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐, 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼀𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑.
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼊𖽔 𖽂𖽺𖾏 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏
19 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑, 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏, 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽃𖽡𖾑.
20 𖼞𖽪𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 ,
𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼛𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼚𖽜𖾑 𖼐𖽱𖾏 .
21 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏,
𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 .

𖼦𖽡𖾐 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑，
