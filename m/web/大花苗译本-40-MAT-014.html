𖼳𖽡𖾑 𖼖𖽡 𖼮𖽑𖽘𖾐 𖼽𖽙 𖼦𖽗 𖼖𖽡𖾑 𖼇𖽻𖾏
（𖼞𖽑𖽙𖾐 𖾏𖾏𖾏 6：14-29；𖼖𖽪𖾏 9：7-9）
14
𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 ，𖼳𖽡𖾑 𖼖𖽡 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼺𖽹𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽙𖾑𖽑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼊𖽙𖾏 𖽐𖼷𖽷𖾏， 2 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗 𖼎𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏 𖼗𖽙 𖼋𖽘𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼋𖽠𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼐𖽡𖾑 ．” 3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼳𖽡𖾑 𖼖𖽡 𖼑𖽜 𖼐𖽱𖾏 𖼞𖽱𖾑 𖼇𖽺𖾑 𖼖𖽡 𖼐𖽤𖾐𖼳𖽡𖾑 𖼘𖽙 𖼊𖽡𖾐 𖼑𖽪𖾏， 𖼄𖽔𖾐 𖼽𖽙 𖼦𖽗 𖼞𖽱𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼺𖽘𖾏． 4 𖽃𖽡𖾑 𖼟𖽪𖾏𖼽𖽙 𖼦𖽗 𖼙𖽔 𖽐𖼎𖽙 𖼳𖽡𖾑 𖼖𖽡 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼦𖽔𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 ．” 5 𖼳𖽡𖾑 𖼖𖽡 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽐𖼮𖽑𖽹𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼽𖽙 𖼦𖽗 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐． 6 𖼸𖽙 𖼳𖽡𖾑 𖼖𖽡 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼾𖽨𖾏，𖼳𖽡𖾑 𖼘𖽙 𖼊𖽡𖾐 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏，𖼳𖽡𖾑 𖼖𖽡 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝． 7 𖼳𖽡𖾑 𖼖𖽡𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼏𖽵， 𖼐𖽑𖽡𖾏：𖼦𖽡𖾐 𖼞𖽳𖾑 𖼐𖽱𖾏 𖽐𖼊𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏． 8 𖽐𖼷𖽑𖽹𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼄𖽔𖾐 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑𖼽𖽙 𖼦𖽗 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑．” 9 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽳𖾏 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼺𖽪𖾐 𖼏𖽵 𖼗𖽪𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏，𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼎𖽙𖾑 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽑𖽡𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏． 10 𖼐𖽱𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼀𖽔𖾏 𖼮𖽑𖽘𖾐𖼽𖽙 𖼦𖽗 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏， 11 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼋𖽺𖾏 𖽐𖼚𖽡𖽝𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏，𖼖𖽔𖾑 𖽐𖼷𖽑𖽹𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐． 12 𖼽𖽙 𖼦𖽗 𖼑𖽡𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏，𖼮𖽑𖽳𖾑 𖼖𖽪𖾑 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼅𖽻𖾏 𖼊𖽱𖾏 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑．
𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐 𖼮𖽡𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖼮𖽻𖾏
（𖼞𖽑𖽙𖾐 6：30-44；𖼖𖽪𖾏 9：10-17；𖼽𖽙 6：1-14）
13 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏，𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏，𖼾𖽨𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼙𖽠𖾏 ． 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐． 14 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐． 15 𖼖𖽔𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼋𖽘，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖽐𖼎𖽙𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼉𖽺𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏，𖼖𖽔𖾑 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼋𖽘，𖼟𖽙𖾏 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏，𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼼𖽙𖾏 𖼅𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏．” 16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼦𖽡𖾐 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 ， 𖼅𖽡 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏！” 17 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼯𖽔 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐， 𖽃𖽔𖾐 𖼋𖽪 𖽐𖼀𖽳𖾏 𖼊𖽔𖾐．” 18 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼯𖽡𖽙𖾏 𖼋𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑．” 19 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼼𖽻𖾏 𖼎𖽻𖾏 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖽐𖼡𖽜𖾐 ， 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼀𖽜𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖽃𖽔𖾐 𖼋𖽪 𖽐𖼀𖽳𖾏， 𖼑𖽔𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 ， 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼺𖽘𖾏， 𖼎𖽺𖾏 𖽐𖼀𖽹𖾏 𖼳𖽡𖽝𖾐 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐． 20 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏 𖼮𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼳𖽻𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼳𖽙𖾏，𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼮𖽡𖽺𖾑． 21 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏，𖼋𖽠𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼺𖽘𖾏，𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐．
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑
（𖼞𖽑𖽙𖾐 6：45-52；𖼽𖽙 6：15-21）
22 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼖𖽠𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏，𖼟𖽙𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼺𖽪𖾐 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼀𖽡 𖼋𖽡𖾏 ， 𖼐𖽱𖾏 𖼾𖽨𖾏 𖼐𖽡𖽙𖾐 𖽐𖼎𖽙𖾏 ，𖼮𖽙𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 ． 23 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼋𖽘𖾏，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼾𖽨𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏．𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼋𖽘，𖼯𖽔 𖼅𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 24 𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 ，𖽐𖼞𖽙𖾏 𖼡𖽔𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼋𖽘，𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼮𖽢𖾏 𖽐𖼀𖽪𖾐， 𖽃𖽻𖾐 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏，𖽐𖼞𖽙𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏． 25 𖼸𖽙 𖼋𖽘 𖼄𖽑𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑，𖼽𖽔𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 26 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼅𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼳𖽡𖽠𖾑，𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ， 𖼐𖽑𖽡𖾏 ： “𖼟𖽪𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼀𖽡𖾑 𖼚𖽘𖾐！” 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏． 27 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼮𖽙𖾏 𖼺𖽡𖽝𖾐， 𖼟𖽪𖾏 𖼞𖽪𖾑 ，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏．” 28 𖼀𖽡𖾐 𖼊𖽳𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡，𖼟𖽙𖾏 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽻𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑．” 29 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼋𖽔𖾏！” 𖼀𖽡𖾐 𖼊𖽳 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖽐𖼞𖽙𖾏，𖼅𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽻𖾐 𖼽𖽔𖾑 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 30 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼁𖽙𖾏 𖼮𖽢𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏，𖼖𖽔𖾑 𖼦𖽡 𖼋𖽙 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼮𖽵𖾑 𖼘𖽪 ！𖼮𖽨𖾏 𖼞𖽪𖾑！” 31 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼊𖽔𖾑 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ： “𖼟𖽡 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼯𖽠 𖼐𖽡𖾑，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼷𖽑𖽠𖾑 𖽐𖼊𖽱𖾏？” 32 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽂𖽪𖾏 𖽐𖼮𖽡𖾏 𖽐𖼞𖽙𖾏，𖼮𖽢𖾏 𖽐𖼮𖽦 𖼙𖽠𖾏 𖼋𖽘𖾏． 33 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽙𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 ： “𖼟𖽡 𖼋𖽡𖽝 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐．”
𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼞𖽳 𖼐𖽡 𖼺𖽔𖾐 𖼖𖽳 𖼊𖽪𖾐 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐
（𖼞𖽑𖽙𖾐 6：53-56）
34 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽡𖾏 𖼳𖽡𖽠𖾑，𖼗𖽙𖾏 𖼸𖽙 𖼞𖽳 𖼐𖽡 𖼺𖽔𖾐 𖼖𖽳 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏． 35 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 𖽃𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 ， 𖼨𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 36 𖼯𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼐𖽱𖾏 𖼮𖽑𖽙𖾏 𖼘𖽑𖽘𖾐 𖼳𖽷𖾏 ， 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽔𖾏 𖼺𖽔𖾏 𖼊𖽻𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏．
