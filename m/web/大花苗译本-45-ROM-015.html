𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾐 𖽐𖼀𖽘𖾐
15
𖼀𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼇𖽺𖾐 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑 𖼞𖽡𖾑 𖼯𖽻 𖼚𖽑𖽔𖾏 ，𖼦𖽡𖾐 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼙𖽔 ． 2 𖼀𖽡𖾐 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼙𖽔 ，𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖽐𖼀𖽘𖾏． 3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼙𖽔 ，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏，𖼞𖽑𖽔𖾏 𖼀𖽻𖾐 𖼷𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 ．” 4 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼖𖽡𖾏 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏，𖼟𖽪𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾐 𖼐𖽔𖾑 𖼺𖽻𖾏 𖼯𖽡𖽙𖾏，𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼑𖽡𖾏 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏，𖼟𖽙𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏． 5 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼸𖽺𖾏 𖽐𖼷𖽜𖾏 𖽐𖼎𖽙 𖼮𖽑𖽙𖾑 𖼎𖽳𖾏 𖼎𖽻𖾏 𖽃𖽡𖾑，𖼟𖽙𖾏 𖼅𖽡 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐，𖼾𖽻𖾏 𖼮𖽡𖾑 𖼊𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑， 6 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏，𖼷𖽻𖾑 𖼖𖽔 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 ，𖼟𖽪𖾏 𖼀𖽡𖾐 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼑𖽡𖾏 𖼉𖽺𖾏． 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼅𖽡 𖽃𖽡𖾐 𖼼𖽘𖾐， 𖼟𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏． 8 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏，𖼮𖽡𖾑 𖼊𖽪 𖼟𖽪𖾏 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝，𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽪𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽵𖾐 𖼑𖽪𖾏，𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖽐𖼎𖽑𖽹𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼏𖽵 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐， 9 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐，𖼑𖽜 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：
“𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡，
𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖽐𖼷𖽷𖾏．”
10 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
“𖼅𖽡 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐，𖼽𖽔𖾑 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾐 𖼸𖽻 𖽃𖽺𖾐 𖼙𖽔．”
11 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：
“𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼘𖽪！𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼮𖽵𖾑！
𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 ．”
12 𖼎𖽺𖾏 𖼅𖽔𖾏𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏：
“𖽐𖼎𖽪𖾏 𖼡𖽜𖾐 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼽𖽦𖾑 𖼳𖽡𖾑，
𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏 𖼗𖽙𖾏 𖼯𖽵𖾏 𖼞𖽳𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖽃𖽡𖾑，
𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 ．”

𖼀𖽻𖾐 𖼘𖽙 𖼡𖽑𖽔𖾐 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐
