𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏
31
“𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼾𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏.” 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏. 2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏:
“𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏 𖽃𖽡𖾑,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼊𖽻𖾏 𖼅𖽳𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼳𖽙𖾏.
3 𖽃𖽔 𖼊𖽑𖽻𖾏 (𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐”) 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 (𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖼞𖽪𖾑 ” ) 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽡𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 ,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼷𖽹𖾏 𖼖𖽑𖽱𖾏 𖼟𖽡 .
4 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪 !
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼟𖽡, 𖼟𖽡 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏.
𖼟𖽡 𖼽𖽔𖾑 𖼎𖽺𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼯𖽡𖽙𖾏 𖼮𖽡𖽝𖾏 𖽐𖼎𖽔𖾏 𖼟𖽪𖾏 𖽐𖼀𖽔𖾏,
𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏.
5 𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼇𖽻𖾑 𖼎𖽻𖾐, 𖼯𖽡𖽙 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 ,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙 𖼯𖽡𖽙 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏, 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏 𖼽𖽔𖾑 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽡𖾑.
6 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏, 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑, 𖼖𖽔𖾑 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏:
𖼳𖽠𖾑 𖼗𖽙𖾏 ! 𖼀𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼳𖽡 𖼣𖽔𖾑,
𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑.”
 
7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽔𖾐 𖼞𖽙 𖼙𖽔 𖼚𖽙𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏,
𖼑𖽜𖾏 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽑𖽪 𖼇𖽻𖾑 𖽃𖽡𖾑 𖼙𖽔 𖼨𖽑𖽪𖾏,
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐、𖼷𖽻𖾑 𖼖𖽔 , 𖼐𖽑𖽡𖾏:
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 !
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑.
8 𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼺𖽘𖾑 𖼮𖽢𖾏 𖼁𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽳𖾏 𖼗𖽙𖾏,
𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾏 𖼦𖽡 𖼸𖽱 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑,
𖼅𖽔𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐、𖼖𖽡𖾏 𖼡𖽘𖾑𖽑、𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼐𖽑𖽤𖾑、𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼾𖽨 𖼊𖽪𖾐,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼺𖽷𖾐 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 .
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏 𖼗𖽙𖾏 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼦𖽡𖾏 𖼛𖽠𖾏 𖼢𖽻 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼉𖽺𖾏,
𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖽃𖽔𖾐 𖼘𖽑𖽙𖾐.
 
10 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼘𖽪 ! 𖼽𖽔𖾑 𖼑𖽻 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 ,
𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐, 𖼐𖽑𖽡𖾏:
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼐𖽱𖾏,
𖼎𖽺𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼐𖽱𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐.
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼋𖽘𖾏,
𖼚𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼛𖽡𖾏 𖼋𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
12 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽘𖾏 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼟𖽪𖾏 𖼅𖽔𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏、𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖽐𖼎𖽙 𖼛𖽙𖾏、𖼾𖽘𖾏 𖼁𖽜𖾏、𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼟𖽪𖾏 𖽐𖼊𖽙𖾑 𖽃𖽻𖾐 𖼎𖽻𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼟𖽪𖾏 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏.
13 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼺𖽱𖾑 𖼽𖽔𖾑 𖼙𖽔 𖼚𖽙𖾏 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖽃𖽡𖾑, 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼙𖽔 𖼚𖽙𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼙𖽔 𖼚𖽙𖾏,
