15
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐?
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑?
 
2 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼼𖽘𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑,
𖼎𖽑𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝.
3 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼦𖽡𖾐 𖼎𖽑𖽻𖾏 𖽐𖼚𖽪𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖼽𖽺 𖼗𖽔,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼗𖽔 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼼𖽷𖾏 𖼼𖽙.
4 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼈𖽡𖽝𖾏 𖼎𖽺𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽱𖾏 𖼏𖽵 𖼗𖽪𖾏 𖼺𖽘𖾏, 𖼾𖽨𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼈𖽡𖽝𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏.
5 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼑𖽡𖽝𖾏 𖼮𖽙𖾏 𖽐𖼮𖽷𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏,
𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼅𖽻𖾏 𖼟𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏.
