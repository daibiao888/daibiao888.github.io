𖼽𖽙 𖽃𖽳𖾐 𖽐𖼊𖽠𖾑
𖼽𖽙 𖽃𖽳𖾐 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
1
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏𖼀𖽡𖾐 𖼊𖽑𖽪𖾐 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽙 𖽃𖽳𖾐 .
2 𖽃𖽔𖾑 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼘𖽪！ 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏！
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖽐𖼚𖽙 𖼐𖽡𖾑 𖼘𖽪！ 𖼽𖽔𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼅𖽡 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼑𖽡𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏,
𖼙𖽔 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼘𖽪𖾏?
3 𖼅𖽡 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾐 𖼊𖽪𖾐,
𖼊𖽪𖾐 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾐 𖼞𖽱𖾑 ,
𖼞𖽱𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏.
 
4 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼷𖽷 𖼟𖽻𖾏 𖼑𖽻𖾏 𖼳𖽙𖾏, 𖼙𖽘𖾑 𖼙𖽘𖾏 𖼷𖽷 𖼷𖽷 𖼋𖽔𖾏 𖼑𖽻𖾏 ,
𖼙𖽘𖾑 𖼙𖽘𖾏 𖼷𖽷 𖼷𖽷 𖼑𖽻𖾏 𖼳𖽙𖾏, 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼷𖽷 𖼷𖽷 𖼋𖽔𖾏 𖼑𖽻𖾏 ,
𖼣𖽔𖾑 𖼣𖽔𖾏 𖼷𖽷 𖼷𖽷 𖼑𖽻𖾏 𖼳𖽙𖾏, 𖼙𖽘𖾑 𖼙𖽘𖾏 𖼷𖽷 𖼟𖽻𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 .
 
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼢𖽻 𖼮𖽡𖽠𖾑 𖼘𖽪! 𖼽𖽔𖾑 𖼯𖽷𖾏 𖼨𖽑𖽪𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏.
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼉𖽳 𖼮𖽡𖽠𖾑 𖼘𖽪 ! 𖼽𖽔𖾑 𖼑𖽜 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽡𖽠𖾑 𖼎𖽑𖽻𖾏 𖼅𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼊𖽪𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
6 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼷𖽷𖾑 𖼷𖽷 (“𖽃𖽡𖾐 𖼸𖽻 𖼷𖽷𖾑 𖼷𖽷 ” 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖽃𖽡𖾐 𖼮𖽘𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 ” ) 𖼎𖽺𖾏 𖼘𖽑𖽙𖾐 𖼎𖽺𖾏 𖼋𖽻𖾏, 𖼟𖽪𖾏 𖼾𖽱 𖼋𖽘𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏,
𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾑 𖼐𖽑𖽡𖽝𖾑 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼐𖽑𖽡𖽝𖾑,
𖼊𖽪𖾑 𖼙𖽘𖾏 𖼙𖽘𖾏 𖼐𖽑𖽡𖽝𖾑 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼙𖽘𖾏 𖼙𖽘𖾏 𖼐𖽑𖽡𖽝𖾑.
7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 ,
𖼎𖽺𖾏 𖼖𖽠𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖽃𖽡𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏,
𖼖𖽠𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖼎𖽺𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼺𖽘𖾏,
𖼟𖽙𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼢𖽳 𖼋𖽠𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼗𖽙.
 
8 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪 ! 𖼟𖽡 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼨𖽑𖽪𖾏, 𖼺𖽱𖾏 𖼊𖽡𖾑 𖽐𖼞𖽻𖾐 𖽐𖼷𖽑𖽹𖾏 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐,
𖼑𖽜 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽻𖾏 𖼳𖽷𖾐 𖽃𖽡𖾑 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏.
9 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 , 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼊𖽪𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼖𖽔𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏.
10 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏, 𖼊𖽡𖾐 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼊𖽻𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑,
𖼮𖽡𖽠𖾑 𖼮𖽑𖽦𖾐 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏,
𖼛𖽙𖾏 𖽃𖽔𖾐 𖼮𖽑𖽦𖾐 𖼖𖽔𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠.
 
11 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏；
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾏 𖼨𖽳𖾑𖽑 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼘𖽪!
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼅𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼊𖽪𖾏 𖼮𖽘𖾑 𖼺𖽘𖾏 𖼋𖽘𖾏.
12 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 ,
𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼊𖽻𖾏 𖼄𖽙𖾐,
𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑、 𖽐𖼊𖽻𖾏 𖼺𖽱𖾐 𖼺𖽱𖾐、 𖽐𖼊𖽻𖾏 𖼀𖽑𖽡𖽾 𖽐𖼞𖽙𖾐 𖼊𖽑𖽡𖽝𖾐 𖽐𖼎𖽘𖾏 𖼯𖽢𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼞𖽑𖽔𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 ,
𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼞𖽑𖽔𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
13 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼘𖽪 ! 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼚𖽔𖾑 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼐𖽡𖽝𖾑 𖼨𖽑𖽪𖾏 ．
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽡𖾑 𖼘𖽪! 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽝𖾑 𖽐𖼷𖽘𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖼘𖽪 ! 𖼅𖽡 𖼽𖽔𖾑 𖼋𖽔𖾏 𖽐𖼀𖽔𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼟𖽱𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽻𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖽐𖼎𖽙 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 , 𖼎𖽑𖽻𖾏 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼚𖽙 𖼊𖽪𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
 
14 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏 𖼦𖽡 𖼸𖽱 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑,
𖼨𖽑𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼊𖽪𖾐 𖼗𖽻 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖽐𖼚𖽙 ,
𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏,
𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
