𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼞𖽪𖾑,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽑𖽔𖾐 𖼯𖽵𖾏 𖼞𖽪𖾑！
2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽪𖾑 𖼈𖽙𖾏 𖼀𖽙𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑,
𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼸𖽺𖾏 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏.
 
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼞𖽪𖾑 𖽐𖼡𖽹𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽐𖼷𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼑𖽜 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏, 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐.
4 𖼞𖽪𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖽐𖼎𖽘𖾏 𖼐𖽑𖽤𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.
 
5 𖼑𖽜 𖼞𖽪𖾑 𖼑𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼊𖽔𖾐,
𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐 𖼮𖽵𖾏 𖼗𖽱𖾏 𖽐𖼚𖽱𖾏 𖼁𖽻.
6 𖼞𖽪𖾑 𖼄𖽙𖾐 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼚𖽔𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑 𖼺𖽘𖾏,
𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼟𖽡𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏 𖽐𖼮𖽡 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑.
7 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼚𖽔𖾑 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼞𖽪𖾐,
𖼞𖽪𖾑 𖽐𖼡𖽹𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽐𖼷𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
8 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏, 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖼑𖽜 𖼟𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏.
 
9 𖼮𖽵𖾑 𖼘𖽪！ 𖼞𖽪𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼞𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼞𖽪𖾑 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏 𖼦𖽡𖾐 𖼈𖽺𖾏 𖼟𖽡 𖼮𖽡𖽝𖾏.
10 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼚𖽑𖽡𖽝𖾏 𖼚𖽜𖾑𖽑, 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼯𖽻 𖼼𖽙,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼋𖽘𖾏.
11 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼽𖽺 𖼗𖽔 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼽𖽺 𖼗𖽔, 𖼑𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽔𖾑 𖼳𖽠𖾑 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼺𖽘𖾐 𖼙𖽠𖾏,
𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼳𖽠𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙𖾏.
 
12 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼊𖽑𖽔𖾑 𖼸𖽙𖾏 𖼮𖽡𖽝𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖼽𖽔𖾑 𖼦𖽹𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏.
13 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼗𖽘𖾐 𖼦𖽡𖾐 𖼑𖽻,
𖼺𖽱𖾏 𖼡𖽔𖾑 𖼏𖽔 𖼦𖽡𖾐 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
14 𖼞𖽪𖾑 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽑𖽙𖾑,
𖼦𖽡𖾐 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽪𖾏.
 
15 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼟𖽡！
𖼮𖽵𖾑 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑！
16 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏,
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐 𖼛𖽔𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏.”
 
17 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽻𖾏 𖼢𖽻,
𖼞𖽪𖾑 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
18 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼯𖽜𖾏 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼾𖽨𖾏 𖼾𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽳𖾏.
19 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽺𖾐 𖼘𖽑𖽻𖾐 𖽐𖼡𖽔𖾏, 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼮𖽑𖽨𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝.
20 𖼞𖽪𖾑 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼈𖽡𖽝𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼚𖽪𖾏 𖼁𖽻𖾏 𖼻𖽻𖾏 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼊𖽡𖾑 𖽐𖼊𖽪𖾐.
 
21 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏；
𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐．
22 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼘𖽪！
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑．
