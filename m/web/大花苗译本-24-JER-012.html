𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,

𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼟𖽡 ,

𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽤𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼺𖽱𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽔𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽙𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏.
4 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼐𖽡𖽝𖾑 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡 , 𖼖𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑?
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼯𖽦𖾏 𖽐𖼎𖽙 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖼊𖽪𖾏 𖼮𖽘𖾑 𖼺𖽘𖾏．
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏.
𖼽𖽦𖾑 𖼖𖽡𖾏 𖼄𖽡𖾐 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼾𖽨𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼮𖽔𖾑 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽠𖾏 𖼅𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼏𖽘,
𖼟𖽡 𖼺𖽱𖾑 𖼑𖽱𖾏 𖼟𖽪𖾏 𖼚𖽑𖽔𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖽐𖼎𖽙 𖼑𖽜𖾏 𖼦𖽡𖾏 𖼨𖽑𖽹𖾏 𖼏𖽘 𖼖𖽺𖾑?
𖼟𖽡 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 , 𖼖𖽔𖾏 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼢𖽻,
𖼐𖽡𖽙𖾐 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑?
6 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼞𖽔𖾏, 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼯𖽡𖽙𖾏 𖼟𖽡 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏.
𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼻𖽻𖾏 𖼗𖽪𖾏 𖼈𖽡𖽝𖾏, 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏.
 
7 𖼞𖽪𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏,

𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑, 𖼮𖽙𖾏 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
8 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼚𖽙 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐,
𖼋𖽠𖾏 𖼺𖽘𖾐 𖼊𖽱𖾑 𖼞𖽪𖾑 ,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼳𖽪𖾐 𖼐𖽱𖾏 .
9 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼚𖽘𖾑 𖽐𖼚𖽻 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼖𖽺𖾑?
𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼚𖽘𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼖𖽺𖾑 ,
𖼅𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼦𖽡 𖼸𖽱 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏,
𖼨𖽑𖽳𖾏 𖼋𖽔𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼦𖽡𖾑！
10 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 ,
𖼯𖽪𖾏 𖼞𖽪𖾑 𖽐𖼀𖽠 𖼺𖽘𖾏,

11 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼎𖽺𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏, 𖼐𖽑𖽡𖾏 𖼞𖽡𖾑 𖼊𖽪𖾐 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 ,
𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
