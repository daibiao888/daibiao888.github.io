65
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼐𖽡𖽙𖾐 𖼳𖽡 𖼣𖽔𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽙𖾏 𖼮𖽡𖽝𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡,
𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼙𖽔 𖼏𖽵, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡.
2 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼘𖽪！
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽐𖼮𖽑𖽘𖾑 𖼀𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼟𖽡.
3 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼳𖽠𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏, 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼯𖽔𖾏 𖼺𖽘𖾏.
4 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼺𖽹𖾑 𖼞𖽑𖽡𖽠𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼟𖽡, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏,
𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏!
𖼀𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐、𖼟𖽡 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼑𖽡𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼋𖽘𖾏.
 
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽨𖾏 𖼀𖽡𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼀𖽡𖾐,
𖼟𖽡 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼯𖽠 𖼊𖽡𖾐 𖼺𖽻𖾏 𖽐𖼎𖽙 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼚𖽡𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼻𖽡𖽝𖾏.
6 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏,
𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼇𖽻𖾑 𖼎𖽻𖾐，
7 𖼟𖽙𖾏 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼐𖽱𖾏 𖼳𖽘𖾐 𖽐𖼎𖽙 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏,
𖼊𖽑𖽡𖽝𖾐 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏 𖼋𖽘𖾏.
8 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼺𖽻𖾏, 𖼎𖽑𖽻𖾏 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖽐𖼮𖽑𖽹𖾏 𖼮𖽑𖽜𖾏 𖼮𖽑𖽜𖾏,
𖼟𖽡 𖼟𖽙𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖽐𖼎𖽙 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼙𖽔 𖼚𖽙𖾏.
 
9 𖼟𖽡 𖼑𖽔𖾏 𖼞𖽻𖾏 𖼊𖽡𖾐,
𖼎𖽺𖾏 𖼟𖽙𖾏 𖼑𖽘 𖼗𖽙𖾏 𖽐𖼊𖽪𖾐 𖼎𖽻𖾏,
𖼟𖽙𖾏 𖽐𖼡𖽹𖾏 𖼊𖽡𖾐 𖽃𖽺𖾐 𖼛𖽙 𖼻𖽻𖾏．
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼎𖽺𖾏 𖼀𖽪𖾑 𖽐𖼚𖽔𖾏,
𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖽃𖽻𖾐 𖽐𖼊𖽙𖾑 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏,
𖼟𖽙𖾏 𖼻𖽻𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏.
10 𖼟𖽡 𖽐𖼊𖽙𖾑 𖼐𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼸𖽙𖾏 𖼟𖽱𖾏 𖼖𖽡𖾏 𖼉𖽻𖾏,
𖽐𖼊𖽪𖾐 𖼐𖽻𖾏 𖼩𖽜 𖼖𖽡𖾏 𖼉𖽻𖾏 𖼊𖽑𖽡𖽝𖾐,
𖼺𖽘𖾏 𖼀𖽡𖾑 𖼙𖽱 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼊𖽻𖾏 𖼄𖽔𖾏,
𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼘𖽑𖽙𖾐, 𖼊𖽻𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼅𖽳𖾏.
11 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼞𖽻𖾏 𖽐𖼷𖽑𖽔𖾑,
𖼟𖽡 𖼸𖽙𖾏 𖼞𖽡𖾑 𖽐𖼚𖽱𖾏 𖼛𖽙𖾏，
12 𖽐𖼚𖽱𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼖𖽪𖾑 𖽐𖼎𖽘𖾏 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐,
𖼖𖽔𖾑 𖼣𖽔𖾏 𖼎𖽻𖾐 𖼞𖽑𖽡𖽠𖾏 𖼙𖽔 𖼚𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽔𖾏 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏.
13 𖼖𖽪𖾑 𖽐𖼎𖽘𖾏 𖽐𖼡𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 𖽃𖽔𖾏 𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑,
𖼖𖽪𖾑 𖽐𖼚𖽙 𖼦𖽘𖾑 𖼞𖽑𖽔𖾏 𖼋𖽔𖾏 𖼀𖽪𖾑 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼡𖽻𖾐 𖼖𖽻𖾏,
𖼋𖽡𖾏 𖼐𖽡𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼙𖽔 𖼚𖽙𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏.
