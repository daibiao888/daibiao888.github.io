𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 :
5
𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑 , 𖼞𖽪𖾑 𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 𖼘𖽪 !
𖼞𖽪𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖽐𖼚𖽙 𖼉𖽘𖾏 𖼋𖽘 ,
𖼞𖽪𖾑 𖼚𖽡𖾏 𖼄𖽪 𖼽𖽙 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼋𖽘𖾏,
𖼞𖽪𖾑 𖼑𖽻𖾏 𖽃𖽔𖾑 𖼑𖽘 𖼄𖽪𖾑 𖽐𖼎𖽙 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏,
𖼦𖽻𖾏 𖼞𖽪𖾑 𖼮𖽡𖽠𖾑 𖽐𖼎𖽙 𖼄𖽡𖾏 𖼋𖽘𖾏.
 
𖼨𖽑𖽪𖾑 𖼁𖽜：
𖼞𖽪𖾑 𖼽𖽺 𖼗𖽔 𖼅𖽡 , 𖼷𖽑𖽜𖾑 𖼑𖽻𖾏 !
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑, 𖼷𖽑𖽜𖾑 𖼦𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 !
𖼚𖽻𖾐 𖼸𖽙𖾐
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 :
2 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼀𖽱𖾏 𖽐𖼎𖽙𖾏, 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼯𖽷𖾏 𖼮𖽡𖽝𖾏,
𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏, 𖼐𖽱𖾏 𖽐𖼀𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼐𖽑𖽡𖾏:
 
𖼈𖽻𖾑 𖼮𖽑𖽦𖾐：
“𖼞𖽪𖾑 𖼖𖽔𖾑 𖼞𖽱𖾑, 𖼞𖽪𖾑 𖼖𖽹𖾐 𖼗𖽔 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐, 𖼞𖽪𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏,
𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼇𖽻𖾏 𖼀𖽪𖾑 𖼀𖽡𖾑 𖼙𖽱 ,
𖼞𖽪𖾑 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼊𖽻𖾏 𖼋𖽘 𖼄𖽑𖽙𖾏 𖼀𖽡𖾑 𖼙𖽱 𖽐𖼊𖽪𖾐.”
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐：
3 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏: 𖼞𖽪𖾑 𖼚𖽠𖾏 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼎𖽺𖾏 𖼐𖽑𖽘𖾑 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖽐𖼷𖽔𖾑 𖼊𖽠𖾏 𖼺𖽘𖾏, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼖𖽔𖾑 𖼎𖽺𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼘𖽪?
4 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼁𖽺𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔 𖼛𖽻𖾏 𖼋𖽔𖾏,
𖼞𖽪𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼳𖽠𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
5 𖼞𖽪𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼽𖽔𖾑 𖼑𖽜 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏．
𖼞𖽪𖾑 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖽐𖼚𖽱𖾏 𖼄𖽪 𖼽𖽙 ,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼅𖽔𖾏 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼄𖽪 𖼽𖽙 𖽐𖼚𖽙 𖼚𖽔𖾏 𖼊𖽪𖾑 𖼎𖽪𖾑 𖽃𖽔 𖼛𖽻𖾏.
6 𖼞𖽪𖾑 𖼑𖽜 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖼡𖽑𖽜𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼺𖽘𖾏,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼺𖽷𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏.
𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖽐𖼮𖽑𖽡𖾏 𖼉𖽪𖾏 𖼢𖽻 𖼗𖽠𖾏 ,
𖼞𖽪𖾑 𖽐𖼎𖽑𖽦𖾑 𖼐𖽱𖾏, 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏,
𖼞𖽪𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐.
7 𖼖𖽪𖾑 𖼙𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡 𖽐𖼮𖽡 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽃𖽡𖾑 𖽐𖼮𖽷𖾐 𖼞𖽪𖾑 ,
𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖽃𖽡𖾑 𖼚𖽠𖾏 𖼞𖽪𖾑 𖼮𖽑𖽙𖾏 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼙𖽠𖾏.
8 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 ,
𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏,
𖼅𖽡 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 , 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽪𖾏 𖽐𖼊𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏.
 
𖼨𖽑𖽪𖾑 𖼁𖽜 :
9 𖼟𖽡 𖼖𖽜𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼐𖽡𖾑 𖼘𖽪 !
𖼟𖽡 𖼈𖽻𖾑 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼈𖽻𖾑 𖼘𖽪?
𖼟𖽡 𖼈𖽻𖾑 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼟𖽪𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼈𖽻𖾑 ?
𖼟𖽡 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑 ?
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 𖼷𖽻𖾑 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼮𖽑𖽦𖾐 𖼻𖽻𖾏
𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 :
10 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖽃𖽡𖾑 𖽃𖽺𖾐 𖼚𖽠𖾐， 𖼁𖽠𖾏 𖼚𖽑𖽪𖾏 𖼞𖽡𖾏 𖽐𖼊𖽔𖾏,
𖼻𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖽃𖽡𖾐 𖼉𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐.
11 𖼐𖽱𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼺𖽱𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼐𖽱𖾏 𖼚𖽻𖾐 𖼇𖽻𖾏 𖼀𖽡𖾏 𖼊𖽑𖽪, 𖼚𖽪𖾐 𖼺𖽱𖾏 𖼖𖽡𖾑 𖽃𖽔𖾐.
12 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼺𖽱𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐,
𖼎𖽑𖽻𖾏 𖼞𖽔𖾏 𖼄𖽡𖾏 𖽐𖼷𖽔𖾑 𖼨𖽑𖽪𖾑, 𖼘𖽙𖾏 𖼎𖽑𖽹𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.
13 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖼸𖽷𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖽐𖼊𖽙𖾏 𖼁𖽘𖾏 𖼟𖽪𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏,
𖼺𖽱𖾏 𖽃𖽔𖾑 𖼷𖽜𖾏 𖼟𖽪𖾏 𖼯𖽡𖽙 𖽐𖼡𖽜𖾐 𖼮𖽵𖾏 𖼦𖽘𖾏,
𖼐𖽱𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼺𖽱𖾏 𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏,
𖼎𖽺𖾏 𖽐𖼚𖽱𖾏 𖼄𖽪 𖼽𖽙 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼗𖽙𖾏.
14 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼺𖽱𖾏 𖼡𖽔𖾏 𖼽𖽘𖾑 𖼞𖽪𖾐,
𖼘𖽙𖾏 𖼎𖽑𖽹𖾏 𖼳𖽪𖽺𖾐 𖼷𖽑𖽘𖾑 𖼽𖽱𖾏 𖼎𖽻𖾏,
𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼺𖽱𖾏 𖼊𖽪𖾑 𖽐𖼷𖽑𖽱𖾏 𖼟𖽻𖾏 𖼟𖽪𖾏 𖼸𖽻 𖼞𖽳𖾏,
𖼟𖽻𖾏 𖼯𖽪𖾏 𖼘𖽙𖾏 𖼎𖽑𖽹𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑 𖽐𖼷𖽔𖾐.
15 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼀𖽔𖾐 𖼺𖽱𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼽𖽱𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐 ,
𖼊𖽜𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽜𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖽃𖽔𖾏,
𖼐𖽱𖾏 𖼼𖽘𖾏 𖼺𖽱𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 ,
𖼎𖽺𖾏 𖼻𖽻𖾏 𖼺𖽱𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏.
16 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼡𖽘𖾐 𖼄𖽪𖾑,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼷𖽻𖾏 𖼑𖽡 𖽐𖼮𖽑𖽠𖾏．
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼘𖽪 !
𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼻𖽻𖾏 𖼽𖽺 𖼗𖽔 .
