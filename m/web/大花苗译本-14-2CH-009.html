（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 10：1 - 13）
9
𖼳𖽷𖾏 𖼀𖽔𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 , 𖼎𖽺𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 , 𖼽𖽔𖾑 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼊𖽔𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼮𖽔𖾏 𖼑𖽪 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 . 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼘𖽙 𖼊𖽑𖽙𖾐 𖼮𖽹𖾑 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 、 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 . 𖼐𖽱𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑 , 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 . 2 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼑𖽪 𖽃𖽡𖾑 𖼞𖽑𖽔𖾏 𖼊𖽡𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 , 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽡𖾐 . 3 𖼳𖽷𖾏 𖼀𖽔𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽙𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽡𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏, 4 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾐 𖼎𖽻𖾏 𖼸𖽱𖾏 𖼞𖽡 𖼏𖽻𖾏 , 𖼉𖽻 𖼷𖽻𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 , 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼷𖽑𖽷𖾏 , 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼮𖽡𖽠𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼯𖽵𖾏 𖼮𖽡𖽠𖾑 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼼𖽘𖾏 , 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼊𖽑𖽹𖾑 𖽃𖽔𖾏 𖼷𖽜𖾏, 𖼎𖽺𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼗𖽙𖾏 .
5 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼑𖽪𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼡𖽑𖽙𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏 ！ 6 𖼞𖽪𖾑 𖼺𖽪𖾐 𖼁𖽱 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼋𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑 , 𖼸𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽔𖾏 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 𖼋𖽘𖾏 , 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼟𖽡 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖽃𖽡𖾑 , 𖼦𖽡𖾐 𖼸𖽙 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 𖼺𖽷𖾐 , 𖼟𖽡 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖽃𖽡𖾑. 7 𖼟𖽡 𖼉𖽻 𖼷𖽻𖾏 、𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 , 𖼑𖽻 𖼟𖽡 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏 . 8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼯𖽡𖽙𖾏 ! 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼙𖽔 𖼟𖽡 , 𖼟𖽙𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 , 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼽𖽔𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼟𖽙𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳.
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼳𖽷𖾏 𖼀𖽔𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 𖼞𖽪𖾐 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼯𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 , 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 , 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 . 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼮𖽵𖾏 𖼦𖽘𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼋𖽘𖾏 . 10 𖼳𖽡𖾑 𖼖𖽔 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖽐𖼎𖽙 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 , 𖼎𖽑𖽻𖾏 𖽃𖽳 𖼇𖽺𖾑 𖼨𖽑𖽹𖾑 𖼞𖽪𖾐 𖼗𖽙𖾏 , 𖼎𖽺𖾏 𖼨𖽑𖽹𖾑 𖼊𖽑𖽔 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼗𖽙𖾏 . 11 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼊𖽑𖽔 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽃𖽔𖾏 𖼊𖽑𖽹𖾑 , 𖼎𖽺𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖽃𖽔𖾏 𖼢𖽜𖾏 𖽐𖼎𖽙 𖼢𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼺𖽙𖾑 , 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼎𖽑𖽻𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼗𖽙𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 .
12 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼀𖽔𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼯𖽡𖽙𖾏 𖼋𖽔𖾐 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 , 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼽𖽔𖾑 𖼡𖽑𖽙𖾑 𖽐𖼊𖽜 𖽃𖽡𖾑 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 . 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼾𖽨𖾏 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼙𖽠𖾏 𖼋𖽘𖾏 .
𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽡𖾏 𖼅𖽔𖾏 𖼊𖽻𖾏

29 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼑𖽪𖾏 , 𖼎𖽑𖽻𖾏 𖼁𖽱𖾏 𖼸𖽙 𖼺𖽻𖾏 , 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽔 𖼊𖽗𖾑 𖼑𖽡𖾏 𖽐𖼊𖽠𖾑 𖽐𖼎𖽙 𖼳𖽷𖾏 𖼘𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼽𖽔𖾐《𖽐𖼊𖽠𖾑 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 》𖼮𖽡𖽝𖾏 , 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼁𖽙𖾏 𖼽𖽡𖾏 𖼊𖽙𖾑 𖼑𖽡𖾏《 𖽐𖼊𖽠𖾑 𖼅𖽜 𖼡𖽑𖽜𖾐》 𖼟𖽪𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽡 𖼀𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼮𖽡𖽝𖾏 𖼖𖽺𖾑 ? 30 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 . 31 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏 𖼙𖽠𖾏 , 𖼊𖽱𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼖𖽪𖾑 𖼙𖽻 . 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .
