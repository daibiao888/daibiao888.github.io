𖼽𖽔𖾏 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏.
82
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽵𖾑 𖼳𖽺 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐.
2 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳,
𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏, 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼖𖽺𖾑? （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
3 𖼅𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼑𖽜 𖼊𖽪𖾐 𖼳𖽻𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼊𖽡𖾏 𖽐𖼷𖽔𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖼋𖽡𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼳𖽻𖾏 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 .”
4 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼡𖽑𖽠𖾑 𖼞𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼊𖽻𖾏,
𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏.
 
5 𖼅𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐, 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏,
𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽙 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏 ,
𖼊𖽡𖾐 𖼑𖽡𖾏 𖼊𖽪𖾑 𖽃𖽔𖾐 𖼯𖽤𖾐 𖼊𖽠𖾏 𖼞𖽑𖽔𖾏 𖽃𖽔𖾏 𖼼𖽙 𖼋𖽘𖾏.
 
6 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐.
7 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼋𖽔 , 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼅𖽡 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼋𖽔 𖼙𖽠𖾏.”
 
8 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼨𖽑𖽢𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼉𖽻𖾏 𖼞𖽪𖽙 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏.
