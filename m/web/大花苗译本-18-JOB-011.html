𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼋𖽔𖾏 𖼊𖽔𖾐 𖼡𖽔𖾏 𖼚𖽙𖾏 , 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼟𖽡 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼘𖽪 ?
4 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 ,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 .
5 𖼊𖽻𖾏 𖼅𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 ,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽐𖼊𖽻𖾏 𖼟𖽡 .
6 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼟𖽡 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼺𖽘𖾐 .
𖼟𖽡 𖼽𖽔𖾑 𖼀𖽻𖾐 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼟𖽡 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼋𖽘 .
7 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏 𖼖𖽺𖾑 ?
𖼋𖽡𖽝𖾏 𖼟𖽡 𖼷𖽠𖾏 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖼖𖽺𖾑 ?
8 𖼐𖽱𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 , 𖼟𖽡 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑 ?
𖼐𖽱𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼊𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 , 𖼟𖽡 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼀𖽻𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 ?
9 𖼐𖽱𖾏 𖽐𖼊𖽡𖾑 𖼇𖽘𖾑 , 𖽐𖼊𖽡𖾑 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 ,
𖼇𖽘𖾑 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 .
10 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏, 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽱𖾏 𖼮𖽡𖽝𖾏 ,
𖼦𖽡 𖼸𖽱 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼨𖽑𖽢𖾐 , 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖽐𖼎𖽳𖾏 𖼋𖽻𖾏 𖼐𖽱𖾏 𖼘𖽪 ?
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 .
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 , 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽪𖾏?
12 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼟𖽡𖾏 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳,
𖼑𖽜𖾏 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼣𖽔𖾏 𖼣𖽔𖾏 𖼀𖽡𖾏 𖼊𖽪𖾏 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼖𖽔𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽘 .
𖼞𖽑𖽻𖾑 𖼟𖽙𖾏 𖼀𖽜𖾑 𖼾𖽔𖾏 𖼟𖽙𖾏 𖼚𖽡𖾐
13 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼐𖽱𖾏 ,
𖼮𖽘𖾏 𖼋𖽡𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 ,
14 𖼟𖽡 𖼮𖽔𖾑 𖼀𖽜𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 ,
𖼦𖽡𖾐 𖼟𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 .
15 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼖𖽪𖾑 𖼚𖽑𖽪𖾏 𖼗𖽙𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼀𖽡𖾏 𖼡𖽔𖾏 ,
𖼟𖽡 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 .
16 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼟𖽡 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 ,
𖼟𖽪𖾏 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖼗𖽙𖾏 𖼖𖽔𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖽐𖼚𖽱𖾏 𖼮𖽑𖽡𖽻𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 .
17 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 , 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼇𖽔𖾑 𖼊𖽪𖾑 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼋𖽘 𖼐𖽑𖽪𖾐 ,

18 𖼟𖽡 𖼑𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽔𖾏 , 𖼎𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 ,

19 𖼟𖽡 𖼢𖽻 𖼀𖽱𖾏 𖼙𖽠𖾏 , 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽑𖽙 𖼟𖽡 ,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼊𖽜 𖼅𖽳𖾏 .
