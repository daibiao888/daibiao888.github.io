10
𖼐𖽡𖽙𖾐 𖼞𖽹𖾑 𖼺𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼞𖽙𖾑 𖼐𖽡 𖼖𖽨 ，𖼐𖽱𖾏 𖼟𖽪𖾏 𖼽𖽡𖾏 𖼊𖽔𖾏 𖼖𖽡𖾏 𖼏𖽙 𖼖𖽪𖾑 𖼾𖽡 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐． 2 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐， 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖼐𖽱𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼷𖽷𖾑 𖽐𖼀𖽘𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏，𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 3 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐，𖼋𖽘 𖼐𖽑𖽪𖾐 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼯𖽡𖽹𖾐， 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼚𖽔𖾐 𖽃𖽔 𖼅𖽔 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼁𖽺𖾏 𖼋𖽔𖾏，𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽙𖾑 𖼐𖽡 𖼖𖽨 ．” 4 𖼞𖽙𖾑 𖼐𖽡 𖼖𖽨 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼑𖽔𖾏，𖼎𖽺𖾏 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑 ？” 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾑 𖽐𖼀𖽘𖾏，𖼖𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 ，𖼐𖽱𖾏 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼋𖽘𖾏． 5 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼀𖽑𖽔𖾏 ，𖼨𖽑𖽪𖾏 𖼳𖽡𖾑 𖼄𖽞 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼀𖽡𖾐 𖼊𖽳 𖼋𖽔𖾏． 6 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼖𖽡𖾏 𖼞𖽱𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏， 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼳𖽡𖾑 𖼄𖽞 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 ，𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑．” 7 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼞𖽙𖾑 𖼐𖽡 𖼖𖽨 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑，𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼏𖽙 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼺𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏． 8 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼎𖽺𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼀𖽑𖽔𖾏 𖼙𖽠𖾏．
𖼀𖽡𖾐 𖼊𖽳 𖼁𖽙𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏
9 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼮𖽳，𖼀𖽡𖾐 𖼊𖽳 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼖𖽪𖾑 𖼏𖽪𖾏 𖼅𖽻𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏． 10 𖼐𖽱𖾏 𖼮𖽑𖽹𖾐 𖽃𖽔𖾑 𖼚𖽘𖾐 𖽐𖼮𖽑𖽠𖾏 𖼑𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏，𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼉𖽔𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑，𖼀𖽡𖾐 𖼊𖽳 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼚𖽔𖾐 𖽃𖽔 𖼅𖽔 𖼁𖽙𖾏． 11 𖼐𖽱𖾏 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏，𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼡𖽑𖽹𖾐 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼞𖽪𖾐，𖼮𖽙𖾏 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐． 12 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼅𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼋𖽡𖾏，𖽐𖼎𖽙 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼞𖽘 𖼀𖽡𖾑 𖼊𖽡𖾐，𖼊𖽑𖽡𖽝𖾐 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏． 13 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖼗𖽪 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼀𖽡𖾐 𖼊𖽳 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼊𖽔𖾏 𖼑𖽻𖾏．” 14 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼦𖽡𖾐 𖼢𖽻， 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾑，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼑𖽻𖾏 𖼺𖽷𖾐．” 15 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑，𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏．” 16 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼟𖽻𖾐，𖼋𖽡𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖽃𖽔𖾐 𖼳𖽔𖾐 𖼙𖽠𖾏．
