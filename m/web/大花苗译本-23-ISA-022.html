𖼑𖽜 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏
22
𖼦𖽔𖾑 𖼗𖽔𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼼𖽘𖾏 𖼼𖽔𖾏 𖼑𖽡𖾏 𖼅𖽜 𖼡𖽑𖽜𖾐 :
 
𖼅𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼙𖽻 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾏 𖼎𖽪𖾏 𖽐𖼞𖽔𖾏 𖼙𖽠𖾏 𖼖𖽺𖾑?
2 𖼟𖽡 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖾑,
𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼘𖽪！
𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼐𖽑𖽙𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼊𖽔𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼑𖽜 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏.
3 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽻 𖼷𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼏𖽘 𖼀𖽜𖾑 ,
𖼏𖽻 𖼊𖽪𖾐 𖼀𖽙𖾑 𖼐𖽜𖾑𖽑 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐．
𖼅𖽡 𖼮𖽔𖾑 𖼏𖽘 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼙𖽠𖾏 𖼖𖽔𖾏 ,
𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽻𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐．
4 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏!
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏(“𖼖𖽡𖾑 𖼉𖽪𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 ”) 𖼊𖽻𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏,
𖼎𖽑𖽻𖾏 𖼼𖽙 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼞𖽪𖾑 .”
 
5 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 — 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼚𖽔𖾐 𖼼𖽘𖾏
𖼅𖽔𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼀𖽑𖽔𖾐 𖼦𖽡𖾏 𖽐𖼀𖽑𖽻𖾏, 𖼊𖽻𖾏 𖼯𖽪𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐, 𖽐𖼊𖽱𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽙𖾏 :
𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏,
𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏 𖽐𖼎𖽙𖾑 𖼸𖽙 𖼖𖽪𖾑 𖼎𖽻𖾐.
6 𖼽𖽡𖾑 𖼖𖽗 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖼈𖽙𖾏,
𖼎𖽺𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 , 𖼏𖽙 𖼊𖽑𖽡𖽝𖾐 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 ,
𖼮𖽡 𖽃𖽳𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏.
7 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼳𖽠𖾑 𖼀𖽪𖾑 𖼟𖽡 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 ,
𖼎𖽺𖾏 𖼅𖽔𖾏 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼳𖽠𖾑 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼼𖽘𖾏 𖼮𖽡𖽝𖾏.
8 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼛𖽔𖾏 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼽𖽨 𖼊𖽔𖾏 𖽃𖽡𖾑 𖼺𖽘𖾏,
𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼟𖽡 𖼎𖽺𖾏 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼏𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 “𖽐𖼞𖽔𖾏 𖼡𖽔𖾏 𖼻𖽻𖾏”.
9 𖼅𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼙𖽻 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐, 𖼎𖽺𖾏 𖼊𖽜𖾑 𖼖𖽪𖾑 𖼁𖽘𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼋𖽻 𖼑𖽡𖾏 𖽃𖽻𖾐.
10 𖼎𖽺𖾏 𖼾𖽱 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏, 𖼎𖽑𖽔𖾑 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽘𖾏 , 𖼯𖽡𖽙𖾏 𖼗𖽙 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻. 11 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼮𖽡𖽠𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼁𖽘𖾏 𖽃𖽻𖾐, 𖼯𖽡𖽙𖾏 𖼗𖽙 𖽐𖼊𖽡𖾏 𖼁𖽘𖾏 𖽃𖽻𖾐 𖽃𖽔𖾏 𖼗𖽻𖾏 𖼑𖽡𖾏 𖽃𖽻𖾐. 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼏𖽵 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 .
 
12 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼮𖽵𖾑— 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏 𖼟𖽡𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏,
𖼖𖽡𖾏 𖼇𖽻𖾏 𖼚𖽙𖾏 𖼮𖽢𖾏, 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼀𖽔𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐.
13 𖼑𖽔𖾏 𖼘𖽪! 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽙𖾑 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼊𖽔𖾏 𖼑𖽨𖾏 𖼊𖽔𖾏 𖼮𖽑𖽷𖾏, 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑, 𖼐𖽑𖽡𖾏:
“𖼀𖽡𖾐 𖼑𖽻𖾏 𖼦𖽻𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡 𖼟𖽡 𖼽𖽔𖾑 𖼋𖽔 𖼋𖽘𖾏.”
14 𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
“𖼋𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖾑 𖼸𖽙 𖼟𖽪𖾏 𖼅𖽡 𖼋𖽔, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼯𖽔𖾏,
