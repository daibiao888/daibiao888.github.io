𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
（𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 24： 1 - 26）
12
𖼽𖽦𖾑 𖼦𖽪𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖽃𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏. 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏, 𖼟𖽪𖾏 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑𖼊𖽜𖾑 𖼐𖽜𖾐. 2 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼐𖽡𖽙𖾐 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏. 3 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏, 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼷𖽑𖽷𖾑 𖼳𖽤𖾏. 4 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽡𖾑, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼎𖽙𖾑, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖽐𖼡𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼎𖽙𖾑, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏, 5 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼀𖽻𖾐 𖼳𖽻𖾏 𖼋𖽔𖾏, 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔.” 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼸𖽙 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼷𖽷𖾐 𖼳𖽡𖽻𖾏, 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼦𖽡𖾐 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔. 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼋𖽔𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔 𖼖𖽺𖾑? 𖼎𖽑𖽻𖾏 𖼄𖽑𖽙𖾏 𖼑𖽔𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼑𖽡𖽝𖾏, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼞𖽑𖽡𖽠𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔.” 8 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼳𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖽝𖾏 𖼋𖽘, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽔.
