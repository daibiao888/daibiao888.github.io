𖼽𖽙 𖼦𖽗 𖼄𖽻𖾏 𖼻𖽻𖾏
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼎𖽻𖾐 𖼮𖽡𖾏
1
𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏，𖼗𖽪𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐，𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽙𖾏． 2 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖼷𖽷𖾑 𖼐𖽡𖾑 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐． 3 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼊𖽻𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏． 4 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽐𖼚𖽙，𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾑 𖼎𖽻𖾐 𖼮𖽡𖾏． 5 𖼊𖽪𖾑 𖼎𖽻𖾐 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏， 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼳𖽠𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼊𖽪𖾑 𖼎𖽻𖾐．
6 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼽𖽙 𖼦𖽗 ，𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼋𖽔𖾏． 7 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼑𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏，𖼟𖽪𖾏 𖼑𖽜 𖼊𖽪𖾑 𖼎𖽻𖾐 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏，𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼮𖽜． 8 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽪𖾑 𖼎𖽻𖾐，𖼟𖽪𖾏 𖼑𖽜 𖼊𖽪𖾑 𖼎𖽻𖾐 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏． 9 𖼊𖽪𖾑 𖼎𖽻𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼎𖽻𖾐 𖼋𖽡𖽝，𖼐𖽱𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐，𖼮𖽡𖾏 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐． 10 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐，𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏，𖼈𖽡𖽝𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏． 11 𖼐𖽱𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼾𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼷𖽹𖾏 𖼐𖽱𖾏． 12 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏，𖼟𖽪𖾏 𖽐𖼮𖽜 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏． 13 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼎𖽑𖽻𖾏 𖽐𖼮𖽑𖽘𖾑 𖼾𖽨，𖼦𖽡𖾐 𖼾𖽙 𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼀𖽜𖾏 𖼺𖽡𖽝𖾐 𖼾𖽨，𖼦𖽡𖾐 𖼾𖽙 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼾𖽨，𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼾𖽨．
14 𖼗𖽪𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐，𖼀𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏， 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼊𖽪𖾐 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏，𖼐𖽱𖾏 𖼅𖽔𖾏 𖼅𖽳𖾏 𖽐𖼎𖽙 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏． 15 𖼽𖽙 𖼦𖽗 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏，𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏：‘ 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼊𖽑𖽔𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏．’ ” 16 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏 𖽐𖼚𖽙，𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼅𖽳𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼅𖽳𖾏 𖼊𖽑𖽡𖽝𖾐 𖼅𖽳𖾏． 17 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐，𖼅𖽳𖾏 𖽐𖼎𖽙 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼉𖽺𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪． 18 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼙𖽔 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼯𖽔 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼉𖽺𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼊𖽔𖾐．
𖼽𖽙 𖼦𖽗 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 3：1-12；𖼞𖽑𖽙𖾐 1：1-8；𖼖𖽪𖾏 3：1-18）
19 𖼽𖽙 𖼦𖽗 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑： 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽷𖾐 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼅𖽻， 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 ？” 20 𖼽𖽙 𖼦𖽗 𖼯𖽜𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼦𖽡𖾐 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏，𖼐𖽱𖾏 𖼯𖽜𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼮𖽡𖾑 𖼊𖽪．” 21 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏：“𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？𖼟𖽡 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾏？” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙．” “𖼟𖽡 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼷𖽷𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑？” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼦𖽡𖾐 𖼾𖽙．” 22 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？𖼟𖽙𖾏 𖼀𖽡𖾐 𖼻𖽻𖾏 𖼎𖽙𖾑 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼀𖽡𖾐 𖼋𖽔𖾏．𖼟𖽡 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏？” 23 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：
“𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：
‘𖼨𖽑𖽢𖾏 𖼮𖽵𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 ’ 𖽃𖽡𖾑．
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏．”
