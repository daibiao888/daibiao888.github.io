𖼊𖽗𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽑𖽦𖾑 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐
18
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏. 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼊𖽗𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼺𖽷𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐 𖼺𖽷𖾐. 2 𖼊𖽗𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽑𖽻𖾏 𖼺𖽙𖾐 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼳𖽷 𖼊𖽑𖽻 𖼷𖽷𖾐 𖼾𖽨𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼋𖽻 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼑𖽔𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑, 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼅𖽻𖾏 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑.” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼸𖽙 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐, 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏, 𖼎𖽺𖾏 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑. 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏, 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏, 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼑𖽪 𖼐𖽱𖾏, 𖼐𖽑𖽡𖾏: “𖼡𖽔 𖼋𖽱𖾏 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 ? 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏? 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼊𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏?” 4 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼄𖽡𖾐 𖼮𖽢𖾑 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼄𖽔𖾐 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏.” 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖽐𖼮𖽢𖾏 𖼟𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽙𖾏 𖼀𖽡𖾐 𖼀𖽻𖾐 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼅𖽻𖾏 𖼊𖽙𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼊𖽙𖾏.” 6 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼯𖽪𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼋𖽡𖽝𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽪 𖼅𖽻𖾏, 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.”
7 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼸𖽙 𖼚𖽔𖾏 𖼖𖽔𖾑 𖼽𖽡, 𖼁𖽙𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽳𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐． 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼮𖽑𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽔𖾏 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼚𖽡𖾐, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏. 8 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼸𖽙 𖼺𖽙𖾐 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼳𖽷 𖼊𖽑𖽻 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏, 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻 𖼐𖽑𖽡𖾏: “𖼋𖽡𖽝𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏?” 9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼳𖽠𖾑 𖼗𖽙𖾏 𖼀𖽡𖾐 𖽐𖼮𖽡𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾑！ 𖼀𖽡𖾐 𖼙𖽔 𖼑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝. 𖼅𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼼𖽙 𖼖𖽺𖾑? 𖼽𖽔𖾑 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼅𖽻𖾏 𖼊𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽃𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼊𖽡𖾐, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼊𖽔𖾑 𖼊𖽳𖾑𖽑. 10 𖼅𖽡 𖼸𖽙 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑, 𖼽𖽔𖾑 𖼁𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽳𖾏, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼖𖽔𖾏 𖼇𖽘𖾑. 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼋𖽘𖾏, 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 ，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠.”
11 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼊𖽗𖾏 𖼳𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼚𖽻𖾏 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼯𖽡𖽙𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏, 𖼎𖽑𖽻𖾏 𖼺𖽙𖾐 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼳𖽷 𖼊𖽑𖽻 𖼳𖽠𖾑 𖼊𖽠𖾏， 12 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼑𖽡𖾏 𖼮𖽡𖾑 𖼖𖽡𖽝 ‧ 𖼽𖽦𖾑 𖼖𖽡, 𖼐𖽡𖽙𖾐 𖼮𖽡𖾑 𖼖𖽡𖽝 · 𖼽𖽦𖾑 𖼖𖽡 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾏 𖼮𖽪𖾐 𖼾𖽡． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼦𖽔𖾐 𖼐𖽡 · 𖼊𖽗𖾏 (“𖼄𖽔𖾐 𖼦𖽔𖾐 𖼐𖽡 · 𖼊𖽗𖾏 ” 𖼟𖽪𖾏 “𖼊𖽗𖾏 𖼖𖽪𖾑 𖼾𖽡”) 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏. 13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽡𖾐 𖼮𖽢𖾑 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏.
