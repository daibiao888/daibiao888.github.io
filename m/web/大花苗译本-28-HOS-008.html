𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼦𖽡𖾐 𖼊𖽑𖽹𖾏 𖽐𖼎𖽙 𖼇𖽳𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏
8
𖼟𖽡 𖼎𖽑𖽻𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐 !
𖼞𖽡𖾑 𖼀𖽔𖾑 𖼺𖽱𖾏 𖼚𖽘𖾑 𖼽𖽘𖾏 𖼋𖽔𖾏 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼾𖽡𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼞𖽪𖾑 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 .
2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼀𖽡𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽡 𖼋𖽘𖾏.”
3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 (“𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ”),
𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼐𖽱𖾏 .
4 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼉𖽺𖾏 𖼞𖽪𖾑 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐, 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼾𖽨𖾏 𖼞𖽪𖾐 𖼑𖽡𖽝𖾐 𖼑𖽜 𖼾𖽨𖾏 𖼷𖽷𖾏 𖼁𖽠𖾏 𖼐𖽔𖾑 𖼊𖽻𖾏 𖼮𖽷𖾐 𖼊𖽪𖾏 𖼺𖽘𖾏.
5 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼘𖽪！ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼟𖽡 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏,
𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼑𖽨𖾏 𖼁𖽠 .
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼸𖽙 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼖𖽺𖾑?
6 𖼊𖽪𖾑 𖼑𖽨𖾏 𖼁𖽠 𖼐𖽡𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 ,
𖼟𖽪𖾏 𖼊𖽪𖾐 𖼖𖽡𖾏 𖼞𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖽝𖾏,
𖼦𖽡𖾐 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽔𖾑 𖼣𖽔𖾏 𖼑𖽨𖾏 𖼁𖽠 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼟𖽙𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏.
7 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖼯𖽡𖽙 𖼟𖽪𖾏 𖼮𖽢𖾏,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼳𖽻𖾏 𖼟𖽪𖾏 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼡𖽻𖾐 𖼖𖽻𖾏 𖼘𖽑𖽘𖾑 𖼋𖽠 𖼦𖽡𖾐 𖼷𖽠𖾑𖽑 𖽃𖽔𖾑 𖼐𖽑𖽘𖾐,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽑𖽻𖾑 𖼢𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼡𖽻𖾐.
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼢𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼋𖽪 ,
𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 .
8 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽻𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 ,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼺𖽱𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼙𖽔 .
9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼻𖽡𖽝𖾏𖼽𖽔𖾏 𖼳𖽪 , 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼅𖽻𖾏 𖼾𖽨𖾏.
𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽑𖽠𖾏.
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼈𖽡𖽝𖾏,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼦𖽡 𖼸𖽱 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖽐𖼎𖽘𖾏 𖼐𖽑𖽤𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼮𖽑𖽨𖾏 𖼎𖽻𖾏, 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼦𖽡𖾐 𖼮𖽑𖽜 .
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼺𖽘𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏
11 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼏𖽘 𖽃𖽔𖾏 𖼾𖽔𖾏,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖽃𖽔𖾑 𖼏𖽘 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏.
12 𖼞𖽪𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼉𖽻𖾏 𖼋𖽪𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 ,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽳𖾑 𖽐𖼮𖽷𖾏 𖼞𖽔𖾐.
13 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼡𖽹𖾏 𖼑𖽻𖾏 ,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏,
𖼇𖽔𖾏 𖼳𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 .
14 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼐𖽱𖾏, 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 ,
𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽙𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼏𖽔𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 𖼙𖽻 𖼼𖽙𖾏,
𖼷𖽑𖽷𖾑 𖼺𖽘𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 .”
