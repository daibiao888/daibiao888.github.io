𖽐𖼊𖽠𖾑 𖽐𖼞𖽻𖾏 𖽃𖽔𖾐 𖼑𖽘
𖼐𖽡𖽙𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼊𖽻𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼀𖽘𖾏
𖼞𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
42
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖼟𖽡,
𖼺𖽱𖾏 𖼊𖽪𖾑 𖼷𖽹𖾏 𖽐𖼮𖽑𖽠𖾏 𖼦𖽻𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐.
2 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼡𖽑𖽔𖾐 𖽐𖼡𖽑𖽜𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖼞𖽪𖾑 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼋𖽡𖽝𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼊𖽻𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼚𖽑𖽪𖾏 𖼖𖽺𖾑?
3 𖼞𖽪𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽔𖾏 𖼅𖽔 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?”
 
4 𖼞𖽪𖾑 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼅𖽻𖾏 𖼗𖽙𖾏,
𖼯𖽡𖽙𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖽐𖼎𖽙 𖼷𖽻𖾑 𖼖𖽔,
𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼻𖽙𖾑 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐．
𖼞𖽪𖾑 𖼎𖽙𖾑 𖽐𖼊𖽱𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝.
5 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽪！ 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐?
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏?
𖼟𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 𖼺𖽷𖾐, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑.
6 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼽𖽙 𖼊𖽗𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏、
𖼦𖽳 𖼄𖽞 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏、𖼎𖽑𖽻𖾏 𖼄𖽡𖾐 𖼺𖽔𖾐 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏 𖼟𖽡.
7 𖼟𖽡 𖽃𖽻𖾐 𖽐𖼮𖽔𖾏 𖽐𖼎𖽙𖾑 , 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼊𖽙𖾐 𖽐𖼎𖽙 𖼳𖽡𖽠𖾑 𖼟𖽪𖾏 𖼊𖽙𖾐 𖽐𖼎𖽙𖾑 𖽃𖽔𖾏 𖽐𖼮𖽑𖽤𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐,
𖼟𖽡 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏, 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼞𖽑𖽔𖾏 𖼈𖽳𖾑 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 .
8 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽘𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑；
𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑.
9 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼷𖽔𖾏 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼑𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼖𖽺𖾑?”
10 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼞𖽪𖾑, 𖼺𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼋𖽪 𖼺𖽘𖾏,
𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏,
𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
 
11 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽪！ 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐?
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏?
𖼟𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 𖼺𖽷𖾐, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
