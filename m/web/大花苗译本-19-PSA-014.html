𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏.
（𖽐𖼞𖽻𖾏 53）
14
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏: 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼗𖽱𖾏 𖽐𖼚𖽔, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼳𖽪𖾐,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏.
 
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼖𖽔𖾑 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏,
𖼅𖽔𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏.
 
3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽔𖾏 𖼇𖽳𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖼺𖽘𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏,
𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏．
 
4 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼖𖽔𖾏 𖼯𖽤𖾏?
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫.
 
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
6 𖼅𖽡 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔.
 
7 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼎𖽑𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽨𖾏 𖼎𖽙𖾑 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼗𖽠𖾏, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼽𖽔𖾐 𖼞𖽙 𖼽𖽔𖾑 𖼗𖽙𖾏 𖼳𖽦𖾑, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼽𖽔𖾑 𖼙𖽔 𖼚𖽙𖾏.
