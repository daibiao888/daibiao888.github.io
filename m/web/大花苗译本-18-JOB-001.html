𖼽𖽙 𖼀𖽳 𖽐𖼮𖽡𖽙𖾏
𖼽𖽙 𖼀𖽳 𖼑𖽡𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏 𖽐𖼎𖽙 𖼅𖽔𖾏 𖼊𖽻𖾏
1
𖼐𖽡𖽙𖾐 𖽂𖽪𖾑 𖼺𖽷𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽷𖾏 𖼽𖽙 𖼀𖽳 . 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏 , 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼀𖽜𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 . 2 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼗𖽜𖾏 𖼊𖽪𖾐 , 𖼷𖽷𖾐 𖼖𖽜𖾐 𖽐𖼷𖽑𖽹𖾏 . 3 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏 𖼾𖽘𖾏 𖼮𖽑𖽷𖾏 , 𖼷𖽷𖾐 𖼷𖽑𖽡𖽝𖾐 𖼋𖽪𖾏 𖼘𖽙 𖼊𖽑𖽙𖾐 , 𖼀𖽜𖾐 𖼀𖽔𖾏 𖽐𖼞𖽡𖽠𖾏 𖼑𖽨𖾏 , 𖼀𖽜𖾐 𖼀𖽔𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽜𖾏 𖼖𖽔𖾑， 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 . 4 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼅𖽡 𖼸𖽙𖾏 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 , 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼅𖽔 𖼋𖽔𖾏 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼑𖽻𖾏 𖼦𖽻𖾏 . 5 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏 , 𖼽𖽙 𖼀𖽳 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼾𖽨𖾏 , 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠𖾏 𖼖𖽑𖽘𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “ 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼅𖽡 𖽃𖽔𖾏 𖼾𖽔𖾏 , 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 .” 𖼽𖽙 𖼀𖽳 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 .
6 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 , 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼀𖽙𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏 , 𖼺𖽔𖾐 𖼊𖽔 𖼖𖽔𖾏 𖼋𖽔𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 . 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽪 𖼺𖽔𖾐 𖼊𖽔 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏 ？” 𖼺𖽔𖾐 𖼊𖽔 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏 .” 8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽪 𖼺𖽔𖾐 𖼊𖽔 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖽝𖾏 𖼟𖽡 𖼙𖽔 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽹𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽜𖾑 𖼽𖽙 𖼀𖽳 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼙𖽔 ? 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼺𖽱𖾏 𖼐𖽱𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏、𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏、 𖼀𖽜𖾑 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 .” 9 𖼺𖽔𖾐 𖼊𖽔 𖼊𖽡𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 : “𖼽𖽙 𖼀𖽳 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼯𖽡𖽙 𖼖𖽺𖾑 ? 10 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼟𖽡 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼺𖽙𖾑 𖼉𖽘𖾏 𖼟𖽻𖾏 𖼙𖽻𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏 𖼊𖽑𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼘𖽪 ? 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽻𖾏 𖼟𖽡 𖼺𖽘𖾏 𖼅𖽳𖾏 , 𖼐𖽱𖾏 𖽐𖼀𖽙𖾏 𖼯𖽦𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽔𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 . 11 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼮𖽔𖾑 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 , 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼟𖽡 𖼺𖽘𖾏 .” 12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼊𖽔 𖼐𖽑𖽡𖾏 : “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼮𖽔 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼳𖽤𖾐 𖼋𖽡𖾏 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼐𖽱𖾏 .” 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼺𖽔𖾐 𖼊𖽔 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼑𖽡𖾏 𖼙𖽠𖾏 .
