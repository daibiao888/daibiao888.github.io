𖼑𖽜 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽜𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.
16
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡.
2 𖼞𖽪𖾑 𖼙𖽔 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏, 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑,
𖼞𖽪𖾑 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼉𖽺𖾏 𖼢𖽻𖾏.
 
3 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏,
𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼙𖽔.
4 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼖𖽔𖾑 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
𖽐𖼮𖽑𖽘𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼘𖽑𖽡𖾑 𖽐𖼊𖽙𖾑 𖽃𖽡𖾑 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏,
𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏.
 
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼀𖽠 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖽐𖼡𖽪𖾏 𖼊𖽻𖾏, 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑.
6 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼖𖽑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖽐𖼮𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼋𖽡𖽝 𖼯𖽡𖽙𖾏.
7 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐 𖼞𖽪𖾑 𖽃𖽡𖾑,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑.
8 𖼞𖽪𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼉𖽔𖾏 𖼾𖽨𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼺𖽷𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼼𖽙.
 
9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼙𖽔 𖼚𖽙𖾏, 𖼞𖽪𖾑 𖼖𖽡𖽾 (“𖼖𖽡𖽾”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“𖽐𖼚𖽻 𖼮𖽡𖾏”) 𖼗𖽙𖾏 𖼳𖽦𖾑,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼡𖽹𖾏 𖼖𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐.
10 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼟𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼊𖽻𖾏 𖼁𖽙𖾏 𖼗𖽱𖾏 𖽐𖼚𖽔𖾏 .
 
11 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖽃𖽡𖾑 𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑．
𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼅𖽔𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼐𖽡𖽙𖾐 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼖𖽡 𖼺𖽷𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 .
