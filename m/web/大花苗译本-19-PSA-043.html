43
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏,
𖼑𖽜 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏,
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏.
2 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑, 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏 𖼖𖽺𖾑?
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼑𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐 𖼖𖽺𖾑?
 
3 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖼁𖽙𖾏 𖼞𖽡𖾑 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼻𖽻𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑,
𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏, 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐！
4 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼙𖽔 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑．
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！
𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡．
 
5 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽪！ 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐?
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏?
𖼟𖽡 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽐𖼮𖽦𖾏 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏 𖼺𖽷𖾐, 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 , 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
