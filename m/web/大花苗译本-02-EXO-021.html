𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼡𖽑𖽜𖾑
（𖼎𖽺𖾏 15：12 - 18）
21
“𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼡𖽑𖽙𖾑 𖼖𖽔𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
2 𖼟𖽡 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼟𖽡 𖼚𖽻𖾏 𖼳𖽡𖽻𖾏 , 𖼸𖽙 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 , 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 . 3 𖼐𖽱𖾏 𖼮𖽔𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼋𖽔𖾏 𖼊𖽔𖾐 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼗𖽜𖾐 𖼈𖽻𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼐𖽤𖾐 𖼋𖽘 , 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 . 4 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼮𖽔𖾑 𖼄𖽔𖾐 𖼐𖽤𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖼐𖽤𖾐 𖼮𖽔𖾑 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙 𖽐𖼷𖽑𖽹𖾏 , 𖼐𖽤𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼽𖽔𖾑 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼁𖽡𖽝 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼾𖽨𖾏 𖼋𖽠𖾏 𖼾𖽨𖾏 𖼅𖽻𖾏 . 5 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼐𖽑𖽡𖾏 : 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽤𖾐 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 , 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 . 6 𖼐𖽱𖾏 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼚𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 ( “𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏”) 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽙 𖽃𖽔 𖼛𖽻𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼀𖽔𖾐 𖽃𖽔 𖼛𖽻𖾏 , 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼀𖽔𖾐 𖽃𖽔 𖼛𖽻𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖼮𖽑𖽷𖾏 𖼙𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 .”
7 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼋𖽺 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 , 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼾𖽠 𖼡𖽑𖽜𖾑 𖼖𖽡𖾏 𖽃𖽡𖾑 𖼋𖽠𖾏 𖼙𖽠𖾏. 8 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼺𖽹𖾑 𖼏𖽵 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 , 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽱𖾏 , 𖼽𖽔𖾑 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼘𖽑𖽪𖾑 𖼎𖽙𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 . 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼋𖽺 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 . 9 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼮𖽔𖾑 𖼺𖽹𖾑 𖼏𖽵 𖼐𖽱𖾏 𖼎𖽻𖾏 𖼾𖽨𖾏 𖼊𖽪𖾐 , 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼷𖽑𖽹𖾏 . 10 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼷𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 , 𖼖𖽔𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽡𖾑 𖼡𖽜𖾑 𖼑𖽻𖾏、𖼮𖽑𖽙𖾏 𖼐𖽑𖽘𖾑、𖼊𖽑𖽡𖽝𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙𖾏 𖼐𖽱𖾏 𖼀𖽱𖾏 , 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽠 𖼺𖽘𖾏 . 11 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽷𖾐 𖼼𖽘𖾐 𖼐𖽡𖾑 , 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼎𖽑𖽻𖾏 𖼑𖽡𖽝𖾏 𖼅𖽔𖾏 𖼘𖽑𖽪𖾑 𖼎𖽙𖾑 , 𖼚𖽠𖾐 𖼚𖽠𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 . ”
𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏
“ 12 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼋𖽔 𖼺𖽘𖾏 . 13 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼷𖽪𖾑 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼈𖽡𖽝𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 , 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼏𖽘 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 . 14 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼯𖽡𖽙𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼼𖽷𖾏 𖼼𖽙 𖼺𖽘𖾏 , 𖼐𖽱𖾏 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼏𖽘 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼙𖽠𖾏 𖼖𖽔𖾏 , 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼯𖽡𖽻𖾏 𖼐𖽱𖾏 𖼗𖽙𖾏 𖼐𖽑𖽙𖾏 𖼟𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏 . ”
15 “𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏 . ”
16 “𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼸𖽳𖾐 𖼸𖽳𖾐 𖼮𖽑𖽵𖾑 𖼨𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽺 𖼺𖽘𖾏 , 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼮𖽡𖽝𖾏 𖼺𖽷𖾐 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖽐𖼊𖽻𖾏 𖼋𖽔 𖼺𖽘𖾏 . ”
