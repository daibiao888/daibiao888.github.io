19
𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼞𖽪𖾑 𖼐𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏：
“𖼦𖽔𖾏 𖼖𖽡𖾏 𖼘𖽪𖾏 𖼽𖽔！
𖼮𖽨𖾏 𖼅𖽳𖾏、𖽐𖼚𖽻 𖼮𖽡𖾏、 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼁𖽡𖽝．
2 𖼐𖽱𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼊𖽔𖾐 𖼋𖽡𖽝、 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳；
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽢𖾐 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽃𖽡𖾑 𖼋𖽘𖾏，
𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽃𖽡𖾑 𖼎𖽑𖽻𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼟𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼗𖽱𖾏 𖽐𖼚𖽔 𖼺𖽘𖾏．
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼅𖽡 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 ，
𖼎𖽺𖾏 𖼇𖽔𖾏 𖼳𖽙𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼐𖽡𖾑 𖼾𖽔𖾏 𖼟𖽪𖾏 𖽐𖼚𖽱𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼅𖽡 𖽐𖼮𖽑𖽘𖾑．”
3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖽃𖽡𖾐 𖼟𖽻𖾐 𖼐𖽑𖽡𖾏：
“𖼦𖽔𖾏 𖼖𖽡𖾏 𖼘𖽪𖾏 𖼽𖽔！
𖼀𖽘𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼐𖽡𖾑 𖽐𖼷𖽑𖽙𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐， 𖼸𖽙𖾏 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾏 𖼋𖽘𖾏．”
4 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼚𖽻𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼗𖽻 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼚𖽻𖾐 𖼋𖽪 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑，𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑，𖼐𖽑𖽡𖾏：
“𖽃𖽔𖾏 𖼄𖽞！𖼦𖽔𖾏 𖼖𖽡𖾏 𖼘𖽪𖾏 𖼽𖽔！”
𖼾𖽘𖾏 𖼁𖽜𖾏 𖼷𖽙𖾏 𖼐𖽤𖾐
5 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 ，𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼎𖽑𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼐𖽑𖽡𖾏：
“𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼅𖽡 𖼘𖽪！
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏，
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼘𖽑𖽙𖾐 𖼳𖽻𖾐，
𖼅𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏！”
6 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽙𖾑𖽑 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏，𖼺𖽱𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖽐𖼷𖽑𖽷𖾏 𖼳𖽱𖾑，𖼺𖽱𖾏 𖼺𖽙𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖼳𖽘𖾐， 𖼐𖽑𖽡𖾏：
“𖼦𖽔𖾏 𖼖𖽡𖾏 𖼘𖽪𖾏 𖼽𖽔！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽵𖾑 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏、
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏，𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽘𖾏．
7 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼙𖽔 𖼚𖽙𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 ，
𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼷𖽙𖾏 𖼐𖽤𖾐 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏，
𖼐𖽱𖾏 𖼐𖽤𖾐 𖼮𖽑𖽦𖾐 𖼖𖽔𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏．
8 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼮𖽑𖽙𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖼟𖽪𖾏 𖼚𖽠𖾐 𖼮𖽡𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽘𖾑，
𖼋𖽡𖾏 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳．”
9 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏：𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼋𖽔𖾏 𖼑𖽻𖾏 𖼾𖽘𖾏 𖼁𖽜𖾏 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖼷𖽙𖾏 𖼐𖽤𖾐 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼋𖽘𖾏．” 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 ：“ 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝．” 10 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼷𖽑𖽙𖾏 𖼯𖽡𖽻 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼊𖽠𖾏 𖼦𖽡𖾐 𖽐𖼚𖽡𖽝𖾐 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖼐𖽱𖾏．𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑！𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼸𖽻𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖽃𖽡𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼊𖽔𖾐． 𖼟𖽡 𖼽𖽔𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．”
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖽾 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼡𖽑𖽔𖾐 𖼁𖽱𖾏 𖽐𖼊𖽔𖾏 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏．
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖼚𖽠𖾐
