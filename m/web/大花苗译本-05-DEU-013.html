“𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 , 𖼢𖽳 𖼋𖽠𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼎𖽻𖾏 𖼅𖽡 𖼑𖽔𖾏 , 2 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏 : 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 , 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 , 𖼚𖽔𖾐 𖼼𖽘𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼢𖽳 𖼋𖽠𖾏 , 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼯𖽡𖽙𖾏 𖼈𖽡𖽝𖾏 . 3 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽻 𖼸𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖽃𖽡𖾑 𖼗𖽪𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼅𖽡 , 𖼽𖽔𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼅𖽡 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼮𖽑𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽙𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏． 4 𖼅𖽡 𖼽𖽔𖾑 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 , 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏 , 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼗𖽪𖾏 , 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽱𖾏 , 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼊𖽔𖾐 . 5 𖼸𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽐𖼀𖽪𖾏 𖼺𖽔𖾑 𖽃𖽡𖾑 , 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼗𖽪𖾏 𖽐𖼷𖽜𖾑 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼟𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼾𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽑𖽜𖾑 , 𖼽𖽔𖾑 𖼮𖽑𖽵𖾑 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼅𖽻𖾏 𖼎𖽻𖾏 𖽃𖽡𖾑 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼋𖽔 𖼺𖽘𖾏． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽑𖽻𖾏 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼋𖽠𖾏 𖼺𖽘𖾏 𖼊𖽪𖾑 𖽐𖼚𖽪𖾏 𖼸𖽳 𖽃𖽡𖾑 .
