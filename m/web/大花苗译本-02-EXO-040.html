34 𖼊𖽑𖽻𖾏 𖼐𖽡𖾑 , 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼀𖽙 𖼈𖽳𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 . 35 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖽐𖼮𖽳 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼚𖽻 𖼮𖽡𖾏 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 . 36 𖼷𖽑𖽔𖾏 𖼟𖽻𖾏 𖼟𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼳𖽻𖾏 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏； 37 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼳𖽻𖾏 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼳𖽠𖾑 𖼊𖽠𖾏 , 𖼋𖽙𖾏 𖼸𖽙 𖼟𖽪𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼳𖽻𖾏 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏. 38 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 , 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐．𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 .
