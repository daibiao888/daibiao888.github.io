81
𖼅𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼙𖽔 𖼨𖽑𖽪𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏,
𖼋𖽠𖾏 𖼳𖽘𖾐 𖼙𖽔 𖼚𖽙𖾏 𖼎𖽻𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏！
2 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏, 𖽐𖼊𖽻𖾏 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼯𖽱𖾏 𖽐𖼎𖽔𖾏,
𖽐𖼎𖽷𖾑 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.
3 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽑𖽔𖾏 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐、
𖼖𖽑𖽡𖾏 𖽐𖼎𖽘𖾏 𖼊𖽑𖽡𖽝𖾐 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼟𖽱𖾏 𖽃𖽡𖾑 𖼷𖽑𖽙𖾑 𖼞𖽪𖾐，
4 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽡𖾏 𖼏𖽵 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏,
𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏.
5 𖼐𖽱𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼐𖽡𖽙𖾐 𖼽𖽙 𖼺𖽳 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼷𖽑𖽙𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏,
 
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐.
6 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏 𖼘𖽑𖽱𖾑 𖼺𖽘𖾏 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑,
𖼟𖽙𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼮𖽙𖾏 𖼮𖽡𖽝𖾏 𖼖𖽪𖾑 𖼮𖽡𖽺𖾑.
7 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜, 𖼞𖽪𖾑 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼟𖽡 𖼋𖽘𖾏,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽙𖾐 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾑 𖼊𖽡𖾐 𖼟𖽡,
𖼐𖽡𖽙𖾐 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼀𖽔𖾑 𖽃𖽻𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼮𖽔𖾏 𖼑𖽔𖾏 𖼟𖽡 . （𖼳𖽡𖾏 𖼖𖽔𖾑）
8 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼟𖽡 .
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼘𖽪！ 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 .
9 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼮𖽵𖾑 𖼳𖽺,
𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽵𖾑 𖼳𖽺 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏.
10 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼙𖽔 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏,
𖼟𖽡 𖼽𖽔𖾑 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼀𖽪𖾑 𖼷𖽻𖾏.
 
11 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼞𖽪𖾑 .
12 𖼞𖽪𖾑 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏,
𖼸𖽙𖾏 𖽐𖼎𖽙𖾏 𖼾𖽨𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽃𖽔𖾏.
13 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼯𖽪𖾏 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼯𖽪𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼸𖽙𖾏 𖼞𖽡𖾑，
14 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼯𖽵𖾏 𖼮𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 ,
𖼎𖽙𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼗𖽙𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 .
15 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼦𖽡 𖼯𖽡𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼨𖽑𖽢𖾐 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏.
16 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼳𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽻𖾏 ,
𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏.”
