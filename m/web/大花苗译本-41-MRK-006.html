30 𖼊𖽪𖾐 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼅𖽡 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼄𖽔𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏， 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼞𖽑𖽔𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼋𖽘𖾏． 31 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼳𖽙𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏．” 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖼗𖽙𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼊𖽪𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏． 32 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽙𖾏，𖼐𖽡𖽙𖾐 𖼊𖽔𖾐 𖼯𖽷𖾏 𖼉𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏． 33 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽠𖾏， 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼁𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 𖼎𖽑𖽻𖾏 𖼊𖽠𖾏 𖼅𖽻𖾏，𖽃𖽔𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼏𖽘 ，𖼺𖽪𖾐 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑． 34 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽡𖾏 𖼋𖽠𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪， 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼎𖽺𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼾𖽨 ，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏．
35 𖽃𖽔𖾏 𖼊𖽘𖾑 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏， 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏， 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼋𖽘， 36 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏， 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼼𖽙𖾏， 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼾𖽨𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏．” 37 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“𖼅𖽡 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏．” 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼛𖽡𖾏 𖼑𖽡𖽝𖾏 𖼅𖽻𖾏 𖼅𖽔𖾏 𖼳𖽡𖽝𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖼖𖽺𖾑 ？” 38 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏： “ 𖼅𖽡 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼳𖽡𖽝𖾐？ 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼑𖽔𖾏 ．” 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼀𖽻𖾐 𖼋𖽘 ，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖽃𖽔𖾐 𖼋𖽪 𖽐𖼀𖽳𖾏 𖼊𖽔𖾐．” 39 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏， 𖼅𖽻𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼼𖽘𖾏 𖽃𖽡𖾐 𖼼𖽘𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑 𖽐𖼡𖽜𖾐 𖽐𖼷𖽔𖾐． 40 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼼𖽘𖾏， 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽜𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾏， 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖼗𖽜𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾏． 41 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼀𖽜𖾑 𖼮𖽡𖽙𖾐 𖼳𖽡𖽝𖾐 𖽃𖽔𖾐 𖼋𖽪 𖽐𖼀𖽳𖾏 𖽃𖽡𖾑，𖼑𖽔𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼮𖽢𖾏 𖼙𖽔 𖼋𖽘𖾏， 𖼎𖽺𖾏 𖽐𖼀𖽹𖾏 𖼳𖽡𖽝𖾐 𖼄𖽔𖾐 𖼮𖽡𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽜𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐， 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽐𖼀𖽳𖾏 𖼇𖽺𖾐 𖼎𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽻𖾏． 42 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽻𖾏 𖼑𖽻𖾏 ， 𖼎𖽺𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼮𖽻𖾏． 43 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼳𖽻𖾏 𖽃𖽔𖾑 𖼁𖽱 𖼳𖽡𖽝𖾐 𖽃𖽔𖾑 𖼁𖽱 𖽐𖼀𖽳𖾏 𖼟𖽪𖾏 𖼳𖽙𖾏， 𖽐𖼊𖽡𖾏 𖼀𖽪𖾑 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼈𖽘𖾐 ． 44 𖽃𖽔𖾑 𖼾𖽠 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼳𖽡𖽝𖾐， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼀𖽜𖾐 𖼷𖽑𖽡𖽝𖾐 ．
