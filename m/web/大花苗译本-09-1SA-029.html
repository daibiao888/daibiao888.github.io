𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽙 𖼦𖽡 𖼸𖽱 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼇𖽪, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 𖼮𖽑𖽡𖽠𖾏 𖼷𖽠𖾑𖽑 𖽃𖽻𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼮𖽪𖾐 𖼾𖽡. 2 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼨𖽑𖽳𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼏𖽙, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼀𖽔𖾏, 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼷𖽑𖽡𖽝𖾐, 𖼦𖽡𖾐 𖼸𖽙𖾐 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏, 𖼸𖽙𖾏 𖼽𖽔𖾏 𖼮𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏. 3 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼅𖽡 𖼐𖽑𖽡𖾏: “𖼸𖽻𖾏 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?” 𖼽𖽔𖾏 𖼮𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼖𖽺𖾑? 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼳𖽡𖽻𖾏 𖼋𖽘𖾏, 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼦𖽡 𖼯𖽡𖾏 𖼞𖽪𖾑, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏.”
𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼦𖽡𖾐 𖼮𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏𖼊𖽔𖾏 𖽂𖽺𖾏
4 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼅𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼎𖽻𖾏 𖼽𖽔𖾏 𖼮𖽡, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽡𖾑 𖼙𖽠𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼮𖽷𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼎𖽙𖾑 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼞𖽡𖾑 𖼀𖽔𖾑. 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼘𖽪? 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼎𖽑𖽻𖾏 𖼀𖽡𖾐 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼘𖽪𖾏？ 5 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖽃𖽔 𖼁𖽙𖾏 𖼅𖽡 𖼨𖽑𖽪𖾏 𖽐𖼞𖽻𖾏 𖼦𖽡𖾏 𖼋𖽠𖾏 𖼚𖽑𖽡𖽝𖾏 𖼐𖽑𖽡𖾏:
“𖼺𖽻𖾐 𖼘𖽙 𖼊𖽔𖾏 𖼋𖽔 𖼷𖽑𖽡𖽝𖾐 𖼷𖽑𖽡𖽝𖾐,
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽔𖾏 𖼋𖽔 𖼉𖽻𖾏 𖼉𖽻𖾏,
𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼷𖽷𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖾑 𖼘𖽪?”
6 𖼽𖽔𖾏 𖼮𖽡 𖼨𖽑𖽪𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼋𖽔𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼏𖽵 𖼗𖽪𖾏, 𖼟𖽡 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼟𖽡 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼏𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽻𖾏 𖼗𖽙𖾏, 𖼞𖽪𖾑 𖼁𖽙𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏. 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼏𖽘 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼸𖽙 𖼐𖽔𖾑 𖼐𖽡𖾑, 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼟𖽡 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼦𖽡𖾐 𖼙𖽔 𖼟𖽡. 7 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽪 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼅𖽻𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼦𖽡𖾐 𖼙𖽔 𖼟𖽡.” 8 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼮𖽡 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼋𖽘𖾏 𖼐𖽔𖾑 𖼘𖽪? 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖼸𖽙 𖼚𖽔𖾏 𖼄𖽑𖽙𖾏 𖼐𖽔𖾏, 𖼟𖽡 𖽐𖼎𖽑𖽦𖾑 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼖𖽺𖾑?” 9 𖼽𖽔𖾏 𖼮𖽡 𖼊𖽡𖾐 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼐𖽑𖽡𖾏: “𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏. 10 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼟𖽡, 𖼟𖽪𖾏 𖼟𖽡 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏, 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼀𖽡 𖼟𖽡 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼺𖽘𖾐 𖽐𖼊𖽪𖾏 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼅𖽻𖾏 𖼦𖽡𖾑！” 11 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼎𖽙𖾑 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼙𖽠𖾏. 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 𖼙𖽠𖾏 𖼋𖽘𖾏.
