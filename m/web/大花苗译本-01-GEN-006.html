𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳
6
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼁𖽱𖾏 𖼮𖽡𖽻𖾐 、 𖼎𖽺𖾏 𖼘𖽘𖾑𖽑 𖼅𖽔𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 2 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏, 𖼎𖽺𖾏 𖼾𖽨𖾏 𖼯𖽔𖾏 𖼾𖽨𖾏 𖼺𖽹𖾑 𖼞𖽑𖽡𖽠𖾏, 𖼨𖽑𖽳𖾏 𖼗𖽙𖾏 𖽃𖽔𖾏 𖼐𖽤𖾐. 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽠𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼋𖽔 , 𖼞𖽪𖾑 𖼖𖽡𖽾 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙, 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼸𖽙 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽡𖽻𖾏 𖼺𖽷𖾐.”
4 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐； 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼅𖽡 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼅𖽡 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖼡𖽠𖾏 𖼅𖽔𖾏 𖼊𖽪𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖽃𖽡𖾑.
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐, 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖽐𖼊𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏. 6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼷𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐. 7 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏. 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、𖼟𖽪𖾏 𖼯𖽦𖾏、𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼞𖽘 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐、𖼊𖽑𖽡𖽝𖾐 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼽𖽔𖾑 𖼞𖽑𖽔𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽷𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏.” 8 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼐𖽙 𖼽𖽔𖾏 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼊𖽻𖾏 𖼅𖽳𖾏.
𖼐𖽙 𖼽𖽔𖾏 𖼷𖽷𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏
9 𖼐𖽙 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐. 𖼐𖽙 𖼽𖽔𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳, 𖼐𖽡𖽙𖾐 𖼊𖽪𖾑 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏. 𖼐𖽙 𖼽𖽔𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏. 10 𖼐𖽙 𖼽𖽔𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼟𖽪𖾏 𖼳𖽔𖾐、𖼦𖽗、𖼽𖽔𖾐 𖼇𖽪.
11 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾏 𖼗𖽱𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏, 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖽐𖼚𖽪𖾏 𖼸𖽳. 12 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖼁𖽙𖾏 𖼋𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼗𖽱𖾏 𖽐𖼚𖽔； 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼀𖽡𖾑 𖼙𖽻 𖼸𖽪𖾏 𖼼𖽘𖾏 𖼗𖽱𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏. 13 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽙 𖼽𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖼋𖽔𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼋𖽘； 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 . 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽡𖾐 𖼸𖽻 𖼖𖽻𖾐 𖼺𖽘𖾏. 14 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽙𖾑 𖼇𖽺𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽙𖾏, 𖼇𖽺𖾐 𖽃𖽡𖾐 𖼮𖽑𖽤𖾏 𖽃𖽡𖾐 𖼮𖽑𖽤𖾏 𖼷𖽷𖾏, 𖽐𖼚𖽙 𖼢𖽻𖾏 𖼘𖽙𖾏 𖼛𖽙𖾏 𖼊𖽑𖽪𖾏 𖼮𖽡𖽝𖾏. 15 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼐𖽡𖾑 𖼼𖽘𖾏 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑: 𖼽𖽔𖾑 𖽐𖼊𖽡𖾑 𖼷𖽷𖾐 𖼀𖽔𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼇𖽘𖾑 𖼀𖽜𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼺𖽡𖽝𖾐 𖼷𖽷𖾐 𖼯𖽡𖽻𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏. 16 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼯𖽪𖾏 𖼽𖽔𖾑 𖼅𖽔𖾏 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏, 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏． 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼑𖽡𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼽𖽔𖾑 𖼡𖽑𖽜𖾐 𖼎𖽻𖾏 𖼺𖽘𖾑 𖼖𖽡𖾏 𖼊𖽘𖾑； 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼽𖽔𖾑 𖼇𖽺𖾐 𖼡𖽔𖾑 𖼳𖽔𖾐、𖼀𖽡𖾑 𖽐𖼎𖽘𖾐、 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼷𖽷𖾐 𖼷𖽜𖾏. 17 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖽃𖽻𖾐 𖼘𖽑𖽙𖾐 𖼷𖽷𖾑 𖽐𖼚𖽻 𖼅𖽔 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐, 𖼷𖽑𖽪𖾐 𖼖𖽻𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏, 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼳𖽦𖾑 𖼀𖽘𖾏 𖽃𖽡𖾑, 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏. 18 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏； 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼊𖽪𖾐、𖼟𖽡 𖼐𖽤𖾐、𖼊𖽪𖾐 𖼑𖽡𖾏 𖼐𖽤𖾐, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏. 19 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽐𖼡𖽹𖾏 𖽐𖼮𖽑𖽘𖾏, 𖽃𖽡𖾐 𖼮𖽘𖾏 𖽃𖽔𖾐 𖼋𖽪, 𖽃𖽡𖾐 𖼋𖽪 𖼡𖽔𖾏 𖼷𖽷𖾑, 𖽃𖽡𖾐 𖼋𖽪 𖼡𖽔𖾑 𖼑𖽡𖽝, 𖼟𖽡 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏, 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑. 20 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼑𖽘𖾏 𖼐𖽻 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏、 𖼯𖽦𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏, 𖽐𖼎𖽹𖾏 𖼊𖽡𖾐 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼞𖽘 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼸𖽙𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼼𖽘𖾏, 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖽃𖽔𖾐 𖼋𖽪, 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼻𖽻𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑. 21 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼡𖽻𖾐 𖼡𖽜𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽃𖽔𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏.” 22 𖼐𖽙 𖼽𖽔𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑． 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼞𖽑𖽔𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼋𖽘𖾏.
