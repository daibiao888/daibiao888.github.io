𖼄𖽔𖾐 𖼮𖽑𖽡 𖼊𖽪𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾏 𖼀𖽻𖾐 𖼘𖽙
6 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼽𖽔𖾏 𖼳𖽡𖾏 𖼽𖽔𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 ，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖼇𖽪 𖼖𖽱𖾐 𖼮𖽢𖾑 𖽐𖼎𖽙 𖼮𖽢𖾑 𖼖𖽔𖾑 𖼊𖽑𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏． 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼄𖽺𖾐 𖼳𖽡𖾑 𖼽𖽔𖾏 𖽐𖼚𖽜𖾏 𖼊𖽡𖾐，𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽠𖾏 𖼚𖽔𖾏 𖼀𖽡𖾐 𖼊𖽑𖽪𖽺𖾑 𖼐𖽡 ，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽡𖾏 𖼖𖽡𖽾 𖼦𖽡𖾐 𖼯𖽪𖾏． 8 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖼄𖽺𖾐 𖼳𖽡𖾑 𖼽𖽔𖾏 ，𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽑𖽳 𖼘𖽙 𖼽𖽔𖾏 ． 9 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼀𖽻𖾐 𖼘𖽙 ，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼄𖽔𖾐 𖼮𖽑𖽡 𖼊𖽪𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽐𖼊𖽜 𖼐𖽑𖽡𖾏：“ 𖽐𖼮𖽢𖾏 𖼟𖽡 𖼋𖽡𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼄𖽔𖾐 𖼮𖽑𖽡 𖼊𖽪𖽝𖾏 𖽐𖼀𖽘𖾏 𖼀𖽡𖾐．” 10 𖼀𖽻𖾐 𖼘𖽙 𖼁𖽙𖾏 𖼋𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏 𖼐𖽡𖾑，𖼀𖽡𖾐 𖼎𖽺𖾏 𖼞𖽑𖽜 𖼑𖽜 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼽𖽔𖾑 𖼋𖽡𖾏 𖼚𖽔𖾏 𖼄𖽔𖾐 𖼮𖽑𖽡 𖼊𖽪𖽝𖾏 ，𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽔𖾑 𖼀𖽡𖾐 𖼡𖽑𖽔𖾐 𖼄𖽻𖾏 𖼻𖽻𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐．
