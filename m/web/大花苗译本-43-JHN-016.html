𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼊𖽻𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼋𖽔𖾏
16
“𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼋𖽘𖾏，𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼢𖽻． 2 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼖𖽠𖾑 𖼅𖽡 𖼋𖽠𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖼙𖽠𖾏，𖼎𖽺𖾏 𖼟𖽪𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼅𖽡 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼉𖽺𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼞𖽪𖾑． 4 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼋𖽘𖾏，𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼅𖽡 𖼎𖽺𖾏 𖽐𖼮𖽡𖽙𖾏 𖼸𖽙 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼋𖽘𖾏．𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽳𖾏 𖼅𖽡，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽡𖽙𖾐．
5 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾐 𖼞𖽪𖾑 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼑𖽪 𖼞𖽪𖾑：‘ 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏？’ 6 𖼈𖽡𖽝𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑， 𖼊𖽳𖾏 𖼎𖽺𖾏 𖼀𖽪𖾑 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐． 7 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼞𖽪𖾑 𖼅𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼅𖽡 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏．𖼞𖽪𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼅𖽻𖾏，𖼊𖽪𖾐 𖼮𖽑𖽙𖾏 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑，𖼞𖽪𖾑 𖼮𖽔𖾑 𖼅𖽻𖾏，𖼎𖽺𖾏 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼅𖽡 𖼉𖽺𖾏 𖼐𖽡𖾑． 8 𖼐𖽱𖾏 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏， 𖼽𖽔𖾑 𖼑𖽜 𖼾𖽔𖾏、𖼑𖽜 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳、𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏、𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐． 9 𖼑𖽜 𖼾𖽔𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖽐𖼮𖽜 𖼞𖽪𖾑． 10 𖼑𖽜 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼙𖽠𖾏，𖼅𖽡 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑． 11 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼖𖽪𖾑 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼏𖽻 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼋𖽘𖾏．
12 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼺𖽷𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼐𖽑𖽡𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏． 13 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏，𖼐𖽱𖾏 𖼽𖽔𖾑 𖼨𖽑𖽳𖾏 𖼅𖽡 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝．𖼐𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏，𖼟𖽪𖾏 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼊𖽪𖾏 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡． 14 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼞𖽪𖾑， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼷𖽹𖾏 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡． 15 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼉𖽺𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼁𖽡𖽝，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏，𖼐𖽱𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼷𖽹𖾏 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡．”

29 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼑𖽔𖾏 𖼘𖽪！𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽑𖽡𖾏． 30 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼀𖽻𖾐 𖼟𖽡 𖼀𖽻𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽪 𖼟𖽡，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼀𖽡𖾐 𖽐𖼮𖽜 𖼟𖽡 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼗𖽙𖾏．” 31 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼅𖽡 𖽐𖼮𖽜 𖼖𖽺𖾑？ 32 𖼑𖽔𖾏 𖼘𖽪！ 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏，𖼐𖽔𖾑 𖼐𖽡𖾑 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏，𖼅𖽡 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏 𖼙𖽠𖾏 𖼡𖽔𖾏 𖼑𖽡𖽝𖾏，𖼇𖽳𖾑 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽡𖽝𖾏 𖼊𖽔𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐． 33 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼽𖽔𖾑 𖼟𖽙𖾏 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪．𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼅𖽡 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔，𖼈𖽡𖽝𖾏 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼚𖽔𖾐，𖼞𖽪𖾑 𖼳𖽠𖾏 𖼺𖽘𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼋𖽘．”
