𖼷𖽻𖾑 𖼖𖽔 𖼮𖽵𖾑 𖼑𖽻 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔 𖼑𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏

𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼟𖽪𖾏 𖽐𖼊𖽜 𖼽𖽔𖾑 .
2 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖽃𖽡𖾐 𖼷𖽑𖽷𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼐𖽱𖾏 .
3 𖼋𖽔 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐 𖼞𖽪𖾑 ,
𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼄𖽔𖾐 𖼊𖽻𖾏 𖼞𖽪𖾑 ,
𖼞𖽪𖾑 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏.
4 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼷𖽷𖾏,
𖼐𖽑𖽡𖾏: “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑！”
 
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼅𖽔𖾏 𖼅𖽳𖾏 𖽐𖼮𖽑𖽠𖾏 , 𖼅𖽔𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏.
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼋𖽱𖾏 𖼅𖽳𖾏,
𖼞𖽪𖾑 𖼀𖽻𖾐 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼗𖽔 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑.
7 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼺𖽻𖾑 𖽐𖼮𖽳𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼯𖽡𖽙𖾏 𖼟𖽡.
 
8 𖼮𖽵𖾑 𖼘𖽪！ 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼳𖽦𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏,
𖼮𖽨𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽔𖾏 𖼅𖽔 ,
𖼮𖽨𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽻𖾏 𖼢𖽻 .
9 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐
𖼅𖽻𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.
10 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏: “ 𖼞𖽪𖾑 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 .”
𖼈𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼞𖽪𖾑 𖽐𖼮𖽜 𖼋𖽘𖾏.
11 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼙𖽔 𖼐𖽑𖽡𖾏:
“𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼊𖽔𖾐！”
 
12 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫？
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 .
13 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐,
𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏.
14 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐
𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼏𖽵 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .
 
15 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 ,
𖼑𖽔𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼑𖽡𖾏 𖼋𖽔 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑.
16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 , 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽜𖾑,
𖼟𖽪𖾏 𖼟𖽡 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐, 𖼟𖽡 𖼙𖽔 𖼚𖽠𖾏 𖼺𖽘𖾏 𖼞𖽪𖾑 𖽐𖼊𖽠𖾏 𖼡𖽑𖽹𖾐.
17 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼮𖽑𖽔𖾏 𖼙𖽔 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼟𖽡 ,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏.
18-19 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼞𖽔𖾏 𖽐𖼚𖽙 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏,
𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼏𖽵 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫．
 
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫!
