𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 “ 𖼄𖽔 𖼦𖽔𖾐 𖼖𖽔𖾑” 𖼳𖽘𖾐.
（𖽐𖼞𖽻𖾏 14）
53
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.”
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼗𖽱𖾏 𖽐𖼚𖽔, 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼳𖽪𖾐,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏.
 
2 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼽𖽔𖾑 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏,
𖼋𖽡𖽝𖾏 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏.
 
3 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼞𖽑𖽔𖾏 𖼇𖽳𖾑 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖼺𖽘𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏,
𖽃𖽡𖾐 𖼖𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏．
 
4 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼘𖽪?
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼺𖽱𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
 
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽪𖾐 𖼾𖽡 𖽐𖼊𖽻𖾏 𖼟𖽡 𖽃𖽡𖾑 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾏 𖼏𖽔𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏．
𖼟𖽡 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼺𖽘𖾏 𖼋𖽘𖾏.
6 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼎𖽑𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏．
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼮𖽨𖾏 𖼎𖽙𖾑 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏, 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑,
𖼽𖽔𖾐 𖼞𖽙 𖼽𖽔𖾑 𖼗𖽙𖾏 𖼳𖽦𖾑 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼽𖽔𖾑 𖼙𖽔 𖼚𖽙𖾏 .
