3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼽𖽔𖾑 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏𖼳𖽡 𖼣𖽔𖾑

4 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！ 𖼽𖽔𖾑 𖼳𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼑𖽻 𖼞𖽪𖾑；
𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼑𖽡𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼘𖽪！ 𖼽𖽔𖾑 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾐 𖽐𖼎𖽑𖽦𖾐 𖼑𖽻 𖼞𖽪𖾑；
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼞𖽪𖾑 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖼊𖽪𖾑 𖼎𖽻𖾐.
5 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽡𖾏 𖼈𖽳𖾏,
𖼞𖽪𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼢𖽳 𖼋𖽠𖾏．
𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼉𖽻𖾏 𖼉𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖽠𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼞𖽪𖾑, 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼁𖽱𖾏.
6 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏,
𖼎𖽺𖾏 𖼑𖽔𖾏 𖼎𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼦𖽘𖾑 𖼀𖽙𖾏 𖼾𖽘𖾏 𖼙𖽠𖾏,
𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖽐𖼀𖽑𖽙𖾐 𖼋𖽘 ,

𖼮𖽔 𖼅𖽔𖾏 𖼞𖽪𖾑 𖼮𖽨𖾏 𖼅𖽳𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼷𖽷𖾏 𖼐𖽡𖽙𖾐,
𖼞𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼗𖽔𖾏 𖼺𖽘𖾏.
 
7 𖼅𖽡 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼘𖽪 ! 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 𖼘𖽪 !
𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼗𖽪𖾏, 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏.
8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼞𖽘𖾐 𖼊𖽑𖽙𖾏 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼮𖽷𖾐 𖼖𖽪𖾑 𖼮𖽑𖽙𖾏,
𖼀𖽡𖾑 𖼞𖽘𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼋𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼮𖽷𖾐 𖼚𖽻𖾐 𖼾𖽘𖾐．
𖼮𖽔𖾑 𖼅𖽔𖾏 𖼞𖽪𖾑 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼷𖽷𖾐 𖼸𖽙 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,

9 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖽐𖼀𖽘𖾐 𖼘𖽪, 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏！𖼳𖽠𖾑 𖽐𖼷𖽱𖾏！

10 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼎𖽙 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼊𖽙𖾐 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏,

𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼖𖽡𖾏 𖼅𖽙𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼋𖽔 𖽃𖽡𖾑,

13 𖼟𖽡 𖼎𖽺𖾏 𖼖𖽡𖾏 𖼅𖽙𖾏 𖼺𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼊𖽑𖽔𖾑 𖼀𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜、
𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏、 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼟𖽡 𖽃𖽡𖾑,
𖼎𖽺𖾏 𖼑𖽜 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐 𖼢𖽳 𖼋𖽠𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
𖼟𖽡 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼊𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼖𖽔𖾏?
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼖𖽺𖾑?
14 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼇𖽺𖾏 𖼋𖽘𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽙𖾏,
𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼋𖽔 𖼺𖽘𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑,
𖼐𖽱𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼮𖽔𖾑𖽑 𖼯𖽠.
15 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼩𖽪𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑, 𖼟𖽙𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼟𖽡𖾏 𖽐𖼎𖽙𖾑 𖽐𖼎𖽜𖾏,
𖼉𖽻𖾏 𖼏𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼷𖽷𖾏.
16 𖼞𖽪𖾑 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼗𖽪𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏,
𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼟𖽡 ,
𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼯𖽡𖽙 𖼏𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜, 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼊𖽠𖾏,
𖼎𖽺𖾏 𖽐𖼎𖽙 𖼳𖽡 𖼣𖽔𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏.”
 
17 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼘𖽪！ 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏！ 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏！ 𖼳𖽠𖾑 𖼗𖽙𖾏！
𖼟𖽡 𖼎𖽑𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽻𖾏 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖽐𖼀𖽻𖾏 𖼚𖽠𖾑𖽑,
𖼦𖽻𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼢𖽻 𖼟𖽡𖾏 𖼖𖽡𖾐 𖼉𖽪𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐 𖽃𖽡𖾑, 𖼦𖽻𖾏 𖼟𖽪𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
18 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽻𖾏 𖽃𖽡𖾑, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏,
𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽻𖾏 𖼘𖽑𖽙𖾐, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼮𖽤𖾐 𖽐𖼀𖽘𖾏 𖼐𖽱𖾏.
19 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏 𖽐𖼎𖽙 𖼎𖽑𖽔𖾑 𖼖𖽻𖾐、 𖼮𖽑𖽹𖾐 𖽐𖼡𖽑𖽜𖾏 𖽐𖼎𖽙 𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏,
𖽃𖽔𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼟𖽡,
𖼡𖽔 𖼋𖽱𖾏 𖼑𖽜 𖼟𖽡 𖼄𖽙𖾐 𖼺𖽡𖽝𖾐 𖼐𖽡𖽝𖾑?
𖼞𖽪𖾑 𖼖𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖽐𖼚𖽡𖾏 𖼎𖽳𖾏 𖼟𖽡 𖼖𖽺𖾑?
20 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖽐𖼮𖽑𖽡𖾏 𖼉𖽪𖾏 𖼢𖽻 𖼗𖽠𖾏 ,
𖼢𖽻 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼷𖽑𖽔𖾏 𖼯𖽢𖾏 𖼟𖽡𖾏,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼖𖽡𖾏 𖼞𖽻𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼟𖽪𖾏 𖼟𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏.
 
21 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼟𖽡 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼦𖽻𖾏 𖼮𖽡𖽠𖾑 𖼢𖽻 𖼐𖽡𖾑 ,
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑．
22 𖼟𖽡 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼟𖽪𖾏 𖼟𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
𖼟𖽪𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽑𖽔𖾑 𖼞𖽡𖾑 𖽃𖽡𖾑, 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
“𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼙𖽔 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼢𖽻 𖼟𖽡𖾏 𖼖𖽡𖾐 𖼉𖽪𖾐 𖼖𖽡𖾐 𖼉𖽳𖾐,
𖼟𖽪𖾏 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼎𖽑𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼷𖽹𖾏 𖼙𖽠𖾏,
𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼦𖽻𖾏.
23 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼖𖽪𖾑 𖼞𖽑𖽙𖾐 𖼐𖽡𖾑 𖼮𖽡𖾑 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼙𖽔 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽑𖽡𖾏:
𖼟𖽡 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼯𖽔𖾏 𖼀𖽡𖾐 𖼯𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏！
𖼟𖽡 𖼎𖽑𖽻𖾏 𖼊𖽡𖾑 𖼢𖽻𖾏 𖽃𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏 𖼯𖽔𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏.”
