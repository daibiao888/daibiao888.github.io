𖼄𖽡𖾐 𖼖𖽡𖾏 𖼣𖽔𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏
10 𖼚𖽻𖾏 𖽃𖽻𖾐 𖼎𖽑𖽻𖾏 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼳𖽻𖾏 𖽐𖼮𖽡𖾏 𖼙𖽠𖾏 , 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼣𖽔𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼳𖽠𖾑 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏 𖼋𖽘𖾏 , 𖼚𖽠𖾐 𖼺𖽱𖾏 𖽐𖼀𖽪𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 ,𖼽𖽔𖾏 𖼖𖽪𖽝 𖽂𖽪𖾏 𖼁𖽙𖾏 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼣𖽔𖾏 𖼐𖽻𖾐 𖼺𖽘𖾏 𖼋𖽘𖾏 . 11 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼀𖽡𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏 , 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼾𖽔𖾏 𖼐𖽡𖾑 𖼮𖽑𖽨𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 . 12 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼤𖽔𖾑 𖼤𖽔𖾏 𖼟𖽪𖾏 𖼋𖽔 𖼙𖽠𖾏 ，𖽂𖽪𖾏 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼷𖽹𖾏 𖽐𖼡𖽹𖾏 𖼗𖽱𖾏 𖼺𖽘𖾏 𖽃𖽡𖾐 𖼀𖽡𖾐 𖽐𖼎𖽘𖾐 .” 13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽡𖽝𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 : “𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪 ！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼐𖽱𖾏 ！” 14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼐𖽱𖾏 𖼉𖽺𖾏 𖼮𖽔𖾑 𖽐𖼊𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏 , 𖼐𖽱𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼘𖽪 ? 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼡𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 , 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼁𖽺𖾏 𖼗𖽙𖾏 .” 15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼣𖽔𖾏 𖼊𖽻𖾏 𖼡𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼾𖽡 𖼉𖽺𖾏 𖼢𖽻𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐． 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡𖾐 𖼳𖽠𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏 , 𖼸𖽙 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼣𖽔𖾏 𖼁𖽺𖾏 𖼗𖽙𖾏 . 16 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼎𖽑𖽻𖾏 𖼦𖽔𖾐 𖼳𖽡𖾐 𖼖𖽪 𖼳𖽠𖾑 𖼊𖽠𖾏 , 𖼐𖽡𖽙𖾐 𖼀𖽔𖾑 𖼖𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼮𖽪𖾐 𖼾𖽡 .
