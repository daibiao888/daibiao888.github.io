𖼷𖽷𖾑 𖼷𖽷 𖼑𖽡𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏
7
𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏： 𖼑𖽜 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼤𖽜𖾏 𖼻𖽻𖾐 ( “𖼻𖽻𖾐 ” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖽐𖼡𖽜𖾐 ” ) 𖼺𖽘𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐, 𖼑𖽔𖾏 𖼘𖽪！ 𖼻𖽻𖾐 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼘𖽘𖾑𖽑 , 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼘𖽘𖾑𖽑 𖽃𖽡𖾑, 𖼑𖽔𖾏 𖼘𖽪 ! 𖼮𖽵𖾑 𖼷𖽷𖾏 𖼷𖽷𖾑 𖼷𖽷 . 2 𖼷𖽷𖾑 𖼷𖽷 𖼑𖽻𖾏 𖼋𖽘 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖽃𖽡𖾑 𖼋𖽡𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾐, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:
“𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪 ! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼯𖽔𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼯𖽻 𖼼𖽙 ,
𖼐𖽱𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼳𖽠𖾑 𖼋𖽻𖾏 𖼖𖽺𖾑?”
3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽑𖽡𖾏:
“ 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽡𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼋𖽘𖾏 .”
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽻 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼯𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔
4 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏： 𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽱𖾏 𖼟𖽙𖾏 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐, 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼷𖽑𖽷𖾑 𖼖𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐, 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼺𖽘𖾏. 5 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏:
“𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼮𖽳 𖼮𖽡𖽝𖾏 𖼦𖽡𖾑 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼯𖽻 𖼼𖽙 ,
𖼐𖽱𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼸𖽠𖾏 𖼳𖽠𖾑 𖼏𖽔𖾏 𖼋𖽻 𖼖𖽺𖾑!”
6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽑𖽡𖾏:
“𖼊𖽪𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼋𖽘 .”
𖼎𖽑𖽻𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖽐𖼮𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼯𖽔𖾏
7 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏, 𖼑𖽔𖾏 𖼘𖽪 ! 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼮𖽑𖽤 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖼞𖽡𖾏 𖼖𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼷𖽷𖾏 𖼮𖽡𖽝𖾏, 𖼮𖽵𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖼞𖽡𖾏 𖼖𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼮𖽑𖽤 𖽃𖽔𖾑 𖽐𖼊𖽪 . 8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾑 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼘𖽪! 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖽃𖽔𖾑 𖼳𖽷𖾏?” 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼁𖽙𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖼞𖽡𖾏 𖼖𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏.” 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏:
“𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽤𖾐 𖼖𖽑𖽔𖾏 𖼞𖽡𖾏 𖼖𖽪𖾑 𖼟𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽺𖾏 𖼯𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
9 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼄𖽪𖾐 𖼺𖽳𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼚𖽑𖽡𖾏 𖼡𖽔𖾑 𖼁𖽙𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽠𖾑 𖽐𖼷𖽱𖾏, 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽻𖾏 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼑𖽡𖾏 𖼾𖽡𖾏.”
𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼑𖽪𖾏 𖽃𖽔𖾑 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏
10 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖽃𖽔𖾑 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽐𖼷𖽜𖾑 𖼟𖽡 , 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑, 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼐𖽡𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏. 11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
‘𖼽𖽦𖾑 𖼘𖽙 𖼀𖽙𖾑 𖼣𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼊𖽔𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏
𖼇𖽳𖾑 𖼾𖽨𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏.’ ”
12 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖽃𖽔𖾑 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼷𖽷𖾑 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼁𖽙𖾏 𖼐𖽡𖾑 𖼘𖽪 , 𖼅𖽻𖾏 𖼙𖽠𖾏, 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼙𖽠𖾏, 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽪𖾑 𖼟𖽡 𖽃𖽔𖾑 𖼚𖽘𖾐, 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏. 13 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼀𖽳 𖼊𖽑𖽳 𖼖𖽡𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑, 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 .”
𖽃𖽔𖾑 𖼄𖽙𖾑 𖼺𖽷𖾑 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐
14 𖽃𖽔𖾑 𖼄𖽙𖾑 𖼺𖽷𖾑 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼄𖽔𖾐 𖼳𖽡𖽝𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼺𖽪𖾐 𖼁𖽱 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 (“𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 ”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 “ 𖼊𖽪𖾐 ”), 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏, 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼮𖽑𖽪𖾑 𖼨𖽑𖽢𖾏 𖽐𖼊𖽻𖾏 𖼷𖽷𖾑 𖼷𖽑𖽷𖾏 𖼼𖽔𖾏. 15 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼺𖽹𖾑 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼸𖽙𖾏 𖽐𖼎𖽙 𖽐𖼀𖽙𖾏 𖼾𖽘𖾐, 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: ‘ 𖼟𖽡 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 .’
