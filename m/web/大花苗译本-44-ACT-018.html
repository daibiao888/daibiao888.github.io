5 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼖𖽔𖾑 𖽐𖼎𖽙 𖼊𖽑𖽡 𖼄𖽙𖾑 𖼊𖽑𖽹𖾏 𖼎𖽑𖽻𖾏 𖼄𖽔𖾐 𖼮𖽑𖽡 𖼊𖽪𖽝𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏， 𖼀𖽻𖾐 𖼘𖽙 𖼞𖽑𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 ，𖽐𖼎𖽙 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪． 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 、𖼎𖽺𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑，𖼀𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖽐𖼮𖽑𖽡𖽙𖾏 𖼡𖽘𖾐 𖼮𖽑𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽐𖼮𖽑𖽘𖾑 𖼚𖽔𖾏 𖼅𖽡 𖼾𖽨𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏， 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽳𖾑（𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “ 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 ”）．𖼋𖽳 𖼟𖽻𖾏 𖼐𖽔𖾑 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽘． 7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽔𖾏，𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽐𖼷𖽷𖾏 𖼊𖽑𖽡 𖼊𖽙𖾑 · 𖼽𖽨 𖼺𖽷𖾑 𖼊𖽪𖾑 ，𖼐𖽱𖾏 𖼟𖽪𖾏 𖼖𖽑𖽘𖾑 𖼸𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼐𖽱𖾏 𖽐𖼞𖽔𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱． 8 𖼮𖽡𖾑 𖼖𖽡𖾏 𖼺𖽷𖾑 𖼀𖽪𖾏 𖼟𖽪𖾏 𖼞𖽳𖾑 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼎𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼾𖽡𖾏 𖼞𖽑𖽔𖾏 𖽐𖼮𖽜 𖼮𖽵𖾑 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼅𖽔𖾏 𖼞𖽙𖾑 𖼖𖽡 𖽐𖼊𖽙𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏，𖼎𖽺𖾏 𖽐𖼮𖽜 𖼊𖽻𖾏 𖽐𖼷𖽔𖾑． 9 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏， 𖼮𖽵𖾑 𖼐𖽡𖽙𖾐 𖼚𖽔𖾐 𖼼𖽘𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏：“ 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏，𖼯𖽔 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼡𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏， 10 𖼅𖽔𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼐𖽡𖽙𖾐，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼸𖽠𖾏 𖽐𖼡𖽜𖾏 𖼋𖽡𖾏 𖼟𖽡𖾏 𖼟𖽡，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑，𖼞𖽪𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖽃𖽔𖾏 𖼁𖽠𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏．” 11 𖼀𖽻𖾐 𖼘𖽙 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 𖼖𖽡 𖼚𖽻𖾏 𖼖𖽑𖽡𖾏，𖼡𖽑𖽔𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏．
𖼷𖽔𖾑 𖼑𖽡𖽝 𖼉𖽻 𖼷𖽻𖾏 𖼮𖽢𖾑 𖼖𖽨 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼑𖽪𖾏
