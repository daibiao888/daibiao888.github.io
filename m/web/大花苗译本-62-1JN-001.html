𖼦𖽔𖾑 𖼗𖽔𖾏 𖼸𖽙 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏 𖼷𖽹𖾏 𖼅𖽔𖾏 𖼋𖽘 𖽃𖽡𖾑， 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼙𖽔 𖼐𖽑𖽙𖾑、 𖼀𖽡𖾐 𖼾𖽨𖾏 𖼙𖽔 𖼁𖽙𖾏、 𖼀𖽡𖾐 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼮𖽦𖾏 𖽐𖼮𖽻𖾑 𖼁𖽙𖾏 、 𖼾𖽨𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼙𖽔 𖼺𖽔𖾏 𖼺𖽔𖾏， 2 𖼊𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼐𖽡𖾑 𖼙𖽔 𖼢𖽳 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼋𖽘𖾏， 𖼀𖽡𖾐 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏，𖼐𖽔𖾑 𖼐𖽡𖾑 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏， 𖼄𖽔𖾐 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼁𖽱 𖽐𖼎𖽙 𖼉𖽺𖾏 𖼐𖽡𖽙𖾐， 𖼎𖽺𖾏 𖼙𖽔 𖼢𖽳 𖼋𖽠𖾏 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼐𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡． 3 𖼀𖽡𖾐 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 、 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼙𖽔 𖼐𖽑𖽙𖾑 𖽃𖽡𖾑 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡， 𖼟𖽙𖾏 𖼅𖽡 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏， 𖼀𖽡𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼉𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖾑 𖼊𖽪 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏． 4 𖼀𖽡𖾐 𖼺𖽻𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼎𖽻𖾏 𖼅𖽡， 𖼟𖽙𖾏 𖼅𖽡 （“𖼅𖽡”𖼅𖽔𖾏 𖽐𖼊𖽠𖾑 𖽃𖽔 𖼗𖽻𖾏 𖽃𖽔𖾏“𖼀𖽡𖾐”）𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽻𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏．
5 𖼎𖽺𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑， 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼣𖽔𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏． 𖼋𖽡𖾏 𖼻𖽻𖾏 𖼄𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼎𖽑𖽻𖾏 𖼮𖽵𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡． 6 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏， 𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼅𖽻𖾏 𖼞𖽡𖾑， 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 ，𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝． 7 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑， 𖼀𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽡𖾏 𖼚𖽑𖽙𖾏， 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼮𖽑𖽘𖾑 𖼎𖽺𖾏 𖽐𖼷𖽔𖾑 𖼨𖽑𖽪𖾑 𖼀𖽡𖾐 𖼾𖽔𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜． 8 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼀𖽡𖾐 𖼾𖽨𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏， 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼾𖽨𖾏 𖼮𖽑𖽵𖾑 𖼾𖽨𖾏， 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼋𖽘． 9 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼯𖽜𖾏 𖼀𖽡𖾐 𖼾𖽨𖾏 𖼾𖽔𖾏， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 ， 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼯𖽔𖾏 𖼀𖽡𖾐 𖼾𖽔𖾏， 𖽐𖼷𖽔𖾑 𖼨𖽑𖽪𖾑 𖼀𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳． 10 𖼀𖽡𖾐 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼙𖽔 𖽃𖽔𖾏 𖼾𖽔𖾏， 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖽃𖽔𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑， 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽐𖼚𖽙 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽘．
