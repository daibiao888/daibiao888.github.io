𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼮𖽡𖾑 𖼳𖽷𖾏, 𖼟𖽪𖾏 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾏 𖼇𖽺𖾑 𖼽𖽔𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏, 𖼀𖽡𖾐 𖼞𖽙𖾑 𖼖𖽔𖾑 𖼑𖽡𖾏 𖼞𖽱𖾑 𖼘𖽔𖾑𖽑, 𖼳𖽡𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼞𖽱𖾏 𖼞𖽱𖾑, 𖼽𖽔𖾏 𖼀𖽡𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐, 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽑𖽜 ( “𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽑𖽜 ”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “ 𖼙𖽘𖾑 𖼙𖽘𖾏 𖼊𖽪𖾐 𖼅𖽔𖾐 ”). 2 𖼐𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼺𖽻𖾐 𖼘𖽙, 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼡𖽔𖾑 𖼾𖽠 𖼎𖽺𖾏 𖼻𖽻𖾏 𖽐𖼎𖽻, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼋𖽳𖾏 𖼐𖽱𖾏, 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏. 3 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼳𖽷𖾏 𖼾𖽨 𖼀𖽡𖾑 𖼯𖽠 𖼋𖽪𖾏 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼛𖽙𖾏 𖼙𖽠𖾏 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑 𖼅𖽻𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽜𖾏 𖼖𖽔𖾑.” 4 𖼺𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼮𖽑𖽡𖽻𖾏𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖼳𖽔𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼊𖽡𖾐, 𖼖𖽔𖾏 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏, 𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖼳𖽔𖾑 𖼖𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑, 𖼎𖽺𖾏 𖼮𖽑𖽡𖽻𖾏 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏.
5 𖽐𖼮𖽡𖽻𖾏 𖼋𖽘𖾏 𖼺𖽪𖾑 𖼇𖽪 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏, 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼺𖽷𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏, 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼦𖽡𖾐 𖼑𖽜 𖽃𖽔𖾑 𖼑𖽡𖽝 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼎𖽙𖾑 𖼑𖽜 𖼀𖽡𖾐 𖼊𖽳𖾏.” 6 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏: “𖼖𖽪𖾑 𖼙𖽻 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐, 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏, 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼸𖽪𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏, 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼐𖽱𖾏 𖼸𖽠𖾏 𖼄𖽔𖾐 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼅𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐.” 7 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼮𖽔𖾑 𖼅𖽻𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼖𖽺𖾑? 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖽃𖽡𖾑 𖼖𖽔𖾏 𖼑𖽻𖾏 𖼋𖽘 𖼋𖽘𖾏， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖽐𖼮𖽷𖾏 𖼩𖽻𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑, 𖼋𖽡𖽝𖾏 𖼀𖽡𖾐 𖼡𖽑𖽔𖾑 𖼅𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏?” 8 𖼡𖽑𖽜𖾑 𖼊𖽡𖾐 𖼺𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼑𖽡𖾏 𖼚𖽻𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼑𖽡𖽝𖾏, 𖼯𖽡𖽙𖾏 𖼅𖽻𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼷𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼺𖽷𖾐, 𖼮𖽙𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼅𖽻𖾏 𖽃𖽡𖾑 𖼡𖽑𖽳𖾏 𖼀𖽡𖾐.” 9 ( 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖼮𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼑𖽪 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼯𖽡𖾏 𖼐𖽑𖽡𖾏: 𖼅𖽻𖾏 𖼑𖽪 𖼊𖽪𖾐 𖼺𖽪𖾐 𖼁𖽙𖾏！ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐, 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼺𖽪𖾐 𖼁𖽙𖾏). 10 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽙, 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼽𖽔𖾐.” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼙𖽻 𖼟𖽪𖾏 𖼷𖽷𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼙𖽠𖾏 𖼋𖽘𖾏.
