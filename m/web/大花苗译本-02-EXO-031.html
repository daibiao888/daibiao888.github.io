𖼽𖽔𖾑 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏
12 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 13 “𖼟𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 : 𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼞𖽪𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽑𖽳𖾐 𖼷𖽑𖽷𖾏 𖼑𖽡𖾏 𖼺𖽻𖾏 , 𖼟𖽙𖾏 𖼅𖽡 𖼀𖽻𖾐 𖼞𖽪𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 . 14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 , 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼷𖽪𖾑 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏 , 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼐𖽑𖽪𖾐 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽻𖾐 𖼺𖽘𖾏 . 15 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 , 𖼈𖽡𖽝𖾏 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼳𖽙𖾏 𖼼𖽙 , 𖼟𖽪𖾏 𖼑𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼻𖽙𖾑 𖽃𖽔𖾏 𖼐𖽑𖽪𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏．𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽃𖽡𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼟𖽙𖾏 𖼋𖽔 𖼺𖽘𖾏 . 16 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼊𖽑𖽳𖾐 𖼷𖽑𖽷𖾏 𖼻𖽙𖾑 𖼐𖽑𖽪𖾐 𖼳𖽙𖾏 𖽃𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏 . 17 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼑𖽡𖾏 𖼺𖽻𖾏 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼚𖽻𖾏 𖼐𖽑𖽪𖾐 𖽐𖼚𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼷𖽷𖾏 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 , 𖼳𖽤𖾏 𖼐𖽑𖽪𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼳𖽙𖾏 𖼼𖽙 𖼟𖽡𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 .”
