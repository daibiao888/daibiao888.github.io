𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽻 𖼚𖽑𖽔𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽃𖽡𖾑
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏 ．𖼯𖽡𖽙𖾏 “𖼮𖽑𖽻 𖼖𖽪𖾏”𖼳𖽘𖾐 （“𖼮𖽑𖽻 𖼖𖽪𖾏” 𖼟𖽪𖾏 “𖽃𖽔𖾑 𖼑𖽡𖽝 𖼷𖽹𖾏”）.
22
𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏! 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏! 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼇𖽳𖾑 𖼞𖽪𖾑 𖼺𖽘𖾏?
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼀𖽜𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼮𖽨𖾏 𖼞𖽪𖾑？ 𖼦𖽡𖾐 𖼑𖽻 𖼞𖽪𖾑 𖼺𖽘𖾐 𖼗𖽪 𖼟𖽪𖾏 𖽐𖼷𖽘𖾏？
2 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜,
𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽔𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 , 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼑𖽻 𖼊𖽡𖾐.
3 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼷𖽻𖾑 𖼖𖽔 𖽃𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑.
4 𖼀𖽡𖾐 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡, 𖼟𖽡 𖼎𖽺𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏.
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽝𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏.
 
6 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼞𖽘𖾐, 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑, 𖼊𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.
7 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖼚𖽙𖾏 𖼞𖽪𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼇𖽜𖾏 𖼇𖽜𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖼐𖽑𖽡𖾏:
8 “𖼐𖽱𖾏 𖼇𖽳𖾑 𖼮𖽙𖾏 𖼾𖽨𖾏 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼮𖽨𖾏 𖼐𖽱𖾏 !
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽱𖾏, 𖼟𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼐𖽱𖾏 !”
9 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼑𖽡𖽝𖾏 𖼖𖽪𖾑 𖽐𖼎𖽙𖾐, 𖼟𖽡 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼟𖽡.
10 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼊𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽂𖽪𖾏 𖼘𖽑𖽘𖾑 𖼾𖽨 𖼞𖽪𖾑, 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏.
11 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼋𖽘𖾏, 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑.
 
12 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼞𖽪𖾑,
𖼀𖽔𖾑 𖼳𖽔𖾑 𖼊𖽪𖾑 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖼙𖽻𖾏 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏.
13 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽔𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽪𖾑,
𖼺𖽱𖾏 𖼊𖽪𖾑 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼉𖽺𖾏 𖼋𖽡𖾏 𖼉𖽺𖾏 𖼊𖽠𖾏 𖼢𖽔𖾏 𖽐𖼎𖽻𖾏 𖽃𖽔𖾏 𖼙𖽠𖾏.
 
14 𖼞𖽪𖾑 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼊𖽻𖾏 𖽐𖼮𖽢𖾑𖽑 𖼋𖽠𖾏 𖼗𖽙𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼽𖽘𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏 𖼞𖽑𖽔𖾏 𖼚𖽑𖽘𖾏 𖼺𖽘𖾏,
𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖼺𖽱𖾏 𖼮𖽡𖽝𖾐 𖼟𖽪𖾏 𖼷𖽑𖽷𖾑 𖼾𖽘𖾏.
15 𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡, 𖼺𖽱𖾏 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼷𖽑𖽹𖾏 𖼉𖽔𖾏,
𖼞𖽪𖾑 𖼊𖽔𖾑 𖽐𖼚𖽺𖾏 𖼻𖽡𖽝𖾏 𖽐𖼎𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖽐𖼎𖽪𖾏,
𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼋𖽔 𖼑𖽡𖾏 𖽃𖽘𖾑 𖽃𖽔 𖼗𖽔𖾏 𖼮𖽡𖽝𖾏.
 
16 𖽐𖼀𖽙𖾏 𖼚𖽡𖾑 𖼟𖽻𖾏 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏, 𖼸𖽻𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽙𖾐 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏, 𖼞𖽪𖾑 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏.
17 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼷𖽑𖽘𖾏, 𖼞𖽪𖾑 𖼞𖽑𖽔𖾏 𖼾𖽱 𖽐𖼷𖽻𖾏 𖼋𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼷𖽷𖾏 𖼖𖽪𖾑 𖼑𖽔𖾏 𖼞𖽪𖾑.
18 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼇𖽺𖾐 𖼞𖽪𖾑 𖼮𖽑𖽙𖾏 𖼀𖽡𖾏 𖼘𖽑𖽠𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽜 𖼞𖽪𖾑 𖼮𖽑𖽙𖾏 𖼄𖽡𖾐 𖼋𖽱𖾐 𖽐𖼡𖽪𖾏 𖽃𖽔𖾑 𖼷𖽑𖽷𖾐 𖼏𖽠.
 
19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐！
𖼞𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼋𖽔𖾏 𖽐𖼀𖽘𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼇𖽺𖾏!
20 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽃𖽔𖾏 𖼐𖽑𖽡𖽝𖾑 𖽃𖽔𖾏 𖼏𖽦𖾏,
𖼮𖽨𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽪𖾑 𖼚𖽡𖾑 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏,
21 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏,
𖼟𖽡 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑 𖼋𖽘𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼑𖽨𖾏 𖼡𖽔𖾏 𖼻𖽻𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐.
22 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏,
𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡.
23 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！𖼅𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏；
𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼘𖽪！ 𖼽𖽔𖾑 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽱𖾏；
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼘𖽪！𖼽𖽔𖾑 𖽐𖼮𖽑𖽹𖾏 𖼐𖽱𖾏．
