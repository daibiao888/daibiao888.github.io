𖼽𖽔𖾏 𖼞𖽪𖾐 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽻𖾑 𖼡𖽑𖽔𖾐
30
𖼽𖽔𖾐 𖼮𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼽𖽔𖾏 𖼞𖽪𖾐 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼗𖽪𖾏, 𖼟𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝. 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼊𖽑𖽡𖽝 𖼊𖽑𖽡𖽝𖾐 𖽂𖽪𖾑 𖼮𖽢 𖼐𖽑𖽡𖾏 :
2 “ 𖼞𖽪𖾑 𖼡𖽔 𖼏𖽔 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐.
3 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼻𖽻𖾏 𖼾𖽻 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖼎𖽺𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼑𖽡𖾏 𖼀𖽻𖾐 𖼟𖽳 .
4 𖼡𖽔 𖼋𖽱𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖽐𖼡𖽜𖾏 𖼗𖽙𖾏 𖼖𖽺𖾑?
𖼡𖽔 𖼋𖽱𖾏 𖼳𖽻𖾏 𖼮𖽢𖾏 𖼎𖽻𖾏 𖼷𖽑𖽹𖾏 𖽐𖼚𖽡𖽝𖾏 𖼋𖽡 𖼮𖽡𖽝𖾏?
𖼡𖽔 𖼋𖽱𖾏 𖼡𖽑𖽠𖾑 𖽃𖽻𖾐 𖼎𖽻𖾏 𖽐𖼚𖽙 𖼮𖽑𖽙𖾏 𖼮𖽡𖽝𖾏?
𖼡𖽔 𖼋𖽱𖾏 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 𖼊𖽡𖾐 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏?
𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏?
𖼐𖽱𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 ?
𖼋𖽡𖽝𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼖𖽺𖾑?
 
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼷𖽑𖽔𖾏 𖼖𖽙𖾏 𖼙𖽔 𖼮𖽡𖽺𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼋𖽘𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 , 𖼐𖽱𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐.
6 𖼐𖽱𖾏 𖼗𖽪𖾏 , 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐,
𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼐𖽱𖾏 𖼖𖽔𖾏 𖼟𖽡 , 𖼎𖽺𖾏 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑.
 
7 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽃𖽔𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 ,
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼋𖽔 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 .
8 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖽐𖼎𖽙 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼡𖽔𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏, 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼊𖽻𖾏,
𖼺𖽘𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼏𖽻 𖼎𖽻𖾏 𖼞𖽪𖾑 .
9 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽻𖾏 𖼮𖽻𖾏 𖼺𖽘𖾏 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼟𖽡 , 𖼐𖽑𖽡𖾏 :
‘𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼖𖽺𖾑? ’
𖼎𖽺𖾏 𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼞𖽪𖾑 𖼮𖽑𖽹𖾐 𖼳𖽻𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼑𖽡𖽝 𖼗𖽔 ,
𖼸𖽙 𖼟𖽪𖾏 𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏.
 
10 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑,
𖽐𖼮𖽑𖽹𖾏 𖼞𖽑𖽡𖽝𖾑 𖼐𖽱𖾏 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼟𖽡, 𖼟𖽡 𖼎𖽺𖾏 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏.
 
11 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 (“𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏”𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏“ 𖼷𖽑𖽷𖾏”) 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖼉𖽺𖾏,
𖼦𖽡𖾐 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐.
12 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼾𖽨𖾏 𖼑𖽔𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖽐𖼷𖽔𖾑 𖼾𖽨𖾏 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼺𖽘𖾏.
13 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖽃𖽔𖾑 𖼅𖽔 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼀𖽡𖾑 𖼊𖽠𖾏 𖽃𖽔 𖼅𖽔 𖼖𖽔𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐.
14 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼷𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽝𖾑𖽑 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖽐𖼊𖽘𖾏, 𖼐𖽑𖽡𖽝𖾑 𖼀𖽔𖾐 𖼺𖽱𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏,
𖼽𖽔𖾑 𖽐𖼡𖽙𖾏 𖼑𖽻𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐, 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏.
 
15 𖼊𖽪𖾑 𖽐𖼚𖽔𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖽐𖼷𖽑𖽹𖾏 :
‘ 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑！ 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑！’
𖼅𖽔𖾏 𖼷𖽷𖾐 𖼼𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏,
𖼊𖽑𖽡𖽝𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 ‘ 𖼷𖽻𖾏 𖼋𖽘𖾏 ’，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼼𖽘𖾐：
16 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 、 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼺𖽜𖾏 𖼊𖽪𖾐、
𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽻𖾐 𖼮𖽻𖾏 𖼦𖽡𖾐 𖼷𖽻𖾏、 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼐𖽑𖽡𖾏 ‘ 𖼷𖽻𖾏 𖼋𖽘𖾏 ’.
 
17 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼚𖽙𖾏 𖼉𖽺𖾏, 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼊𖽪𖾑 𖼖𖽡𖾐 𖽃𖽔𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼞𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏， 𖼊𖽻𖾏 𖼣𖽔𖾑 𖼣𖽔𖾏 𖼚𖽘𖾑 𖼑𖽻𖾏 𖼺𖽘𖾏.
18 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼼𖽘𖾐,
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼀𖽻𖾐 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼼𖽘𖾐,
19 𖼟𖽪𖾏： 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼚𖽘𖾑 𖼐𖽡𖽙𖾐 𖽐𖼊𖽘𖾏 𖽐𖼊𖽪𖾏 𖼽𖽘𖾏,
𖼞𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼐𖽡𖽙𖾐 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽐𖼊𖽻𖾏,
𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽙𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑,
𖼞𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼾𖽠 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼀𖽱𖾏.
 
20 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏,
𖼐𖽱𖾏 𖼑𖽻𖾏 𖼺𖽘𖾏 𖼎𖽺𖾏 𖼳𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏,
𖼐𖽑𖽡𖾏 : ‘ 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏.’
 
21 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼊𖽡𖾐 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼼𖽘𖾐,
𖼊𖽑𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼊𖽡𖾐 𖼞𖽱𖾑 𖼏𖽷𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼼𖽘𖾐：
22 𖼟𖽪𖾏 𖼡𖽑𖽜𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼊𖽪𖾐 𖼡𖽔 𖼏𖽔 𖼑𖽻𖾏 𖼮𖽻𖾏 ,
23 𖽐𖼷𖽑𖽹𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽪𖾐 𖼡𖽔𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏,
𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾑 𖼷𖽹𖾏 𖽃𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼮𖽵𖾑 .
 
24 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼯𖽦𖾏 ,
𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝：
25 𖼞𖽡𖾑 𖽐𖼷𖽻 𖼟𖽪𖾏 𖽃𖽡𖾑 𖼮𖽘𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼼𖽙 ,
𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼯𖽡𖽹𖾏 𖼳𖽙𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 .
26 𖼖𖽔𖾑 𖼡𖽔𖾏 𖼷𖽔𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼋𖽻𖾏 𖽃𖽺𖾐 𖼮𖽑𖽜,
𖼈𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼞𖽘𖾐 𖼈𖽳𖾐 𖼗𖽘𖾏 𖽐𖼚𖽙 𖽃𖽔𖾏 𖽐𖼞𖽔𖾏.
27 𖼷𖽷𖾑 𖼷𖽷 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
𖼈𖽡𖽝𖾏 𖼇𖽺𖾐 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏.
28 𖼊𖽑𖽪𖾑 𖼅𖽔𖾏 𖼟𖽳𖾏 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼯𖽡𖽻𖾏 𖼊𖽻𖾏 𖼋𖽻𖾏 𖼈𖽡𖽝𖾏,
𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 .
 
29 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽠𖾏 𖼅𖽻𖾏 𖼟𖽡𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼅𖽔𖾏 𖼷𖽷𖾐 𖼼𖽘𖾐,
𖼊𖽑𖽡𖽝𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼟𖽡𖾏 𖼅𖽔𖾏 𖼚𖽔𖾐 𖼅𖽔𖾏 𖼼𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖼚𖽻𖾐 𖼼𖽘𖾐 :
30 𖼡𖽔𖾐 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼯𖽦𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽪𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽪𖾏,
𖼦𖽡𖾐 𖼞𖽳𖾑 𖽐𖼮𖽷𖾐 𖼡𖽔𖾑 𖼳𖽷𖾏, 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼑𖽡𖾏 𖼡𖽜𖾐,
31 𖼚𖽡𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖽐𖼡𖽹𖾏（𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏“𖼡𖽔𖾏 𖼘𖽻𖾏 𖼡𖽹𖾐 ”）, 𖽃𖽔𖾑 𖼁𖽱 𖼮𖽑𖽷𖾏,
𖼊𖽑𖽡𖽝𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼋𖽻𖾏.
 
32 𖼟𖽡 𖼮𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 , 𖼾𖽨𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼾𖽨𖾏,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼋𖽘 , 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼋𖽡𖾏 𖼀𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏.
33 𖼷𖽙𖾏 𖼑𖽨𖾏 𖼄𖽡𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼛𖽙𖾏 𖼄𖽡𖾏,
𖼐𖽹𖾏 𖽃𖽔𖾑 𖽐𖼀𖽱 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼁𖽺𖾏 𖽐𖼮𖽑𖽘𖾑,
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏, 𖽐𖼷𖽑𖽙𖾑 𖼑𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼳𖽠𖾑 𖼑𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
