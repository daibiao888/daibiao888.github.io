19
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑, 𖼅𖽔𖾏 𖽃𖽡𖾑 𖼖𖽜𖾐 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼊𖽡𖾐 𖼺𖽘𖾑 𖽃𖽡𖾑, 𖼷𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼽𖽨 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽔𖾏 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏. 2 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼇𖽳𖾑 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼺𖽘𖾏, 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼀𖽳 𖼖𖽡𖾏 𖼦𖽟, 𖼸𖽙 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼈𖽹𖾐 𖽐𖼞𖽔𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼚𖽻𖾐 𖼖𖽑𖽡𖾏. 3 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑 𖽐𖼎𖽙 𖽃𖽔𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼖𖽔𖾑 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏, 𖼯𖽡𖽙𖾏 𖼻𖽻𖾏 𖼗𖽪𖾏 𖼞𖽑𖽻𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼺𖽷𖾐 𖼗𖽙. 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼎𖽺𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼈𖽻𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏． 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼷𖽹𖾏 𖼐𖽱𖾏. 4 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼟𖽪𖾏 𖼖𖽜𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼉𖽺𖾏, 𖼞𖽑𖽻𖾑 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼊𖽻𖾏 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐. 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼑𖽻𖾏 𖼦𖽻𖾏, 𖼀𖽱𖾏 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑. 5 𖼸𖽙 𖼚𖽔𖾏 𖼚𖽻𖾐 𖼐𖽑𖽪𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖼷𖽷𖾑 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼽𖽔𖾑 𖼳𖽠𖾑 𖼅𖽻𖾏, 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼉𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼈𖽻𖾑 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼉𖽔𖾏, 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼅𖽡 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏.” 6 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼼𖽻𖾏 𖼑𖽻𖾏 𖼦𖽻𖾏, 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽜𖾑 𖼟𖽡 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼦𖽔𖾑.” 7 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼞𖽑𖽻𖾑 𖼐𖽱𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼀𖽱𖾏 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏. 8 𖼸𖽙 𖼚𖽔𖾏 𖼀𖽜𖾐 𖼐𖽑𖽪𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏, 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼷𖽑𖽜𖾑 𖼟𖽡 𖼑𖽻𖾏 𖽃𖽡𖾐 𖽐𖼮𖽷𖾏 𖼉𖽔𖾏, 𖼊𖽑𖽡𖽝𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼛𖽘𖾏 𖼼𖽙𖾏, 𖼋𖽙𖾏 𖼟𖽙𖾏 𖼐𖽑𖽪𖾐 𖼢𖽹𖾐 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼅𖽻𖾏.” 𖽃𖽔𖾏 𖼐𖽡𖾑 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏. 9 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖼊𖽑𖽡𖽝𖾐 𖼡𖽑𖽜𖾑 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼟𖽪𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽱𖾏 𖼉𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼑𖽔𖾏 𖼘𖽪! 𖼐𖽑𖽪𖾐 𖼖𖽔𖾑 𖽐𖼷𖽜𖾑 𖼋𖽘𖾏, 𖼷𖽑𖽜𖾑 𖼟𖽡 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼋𖽘, 𖼟𖽡 𖼀𖽱𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼟𖽙𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽡𖽝𖾏 𖼋𖽪, 𖼀𖽡 𖼟𖽡 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏.
10 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼀𖽱𖾏 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼋𖽘𖾏, 𖼎𖽺𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽃𖽔𖾐 𖼋𖽪 𖼑𖽜𖾏 𖼖𖽔𖾑, 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼑𖽡𖽝𖾏 𖼳𖽠𖾑 𖼊𖽠𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼗𖽙𖾏 𖼸𖽙 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼀𖽘𖾑𖽑 𖼀𖽡 𖼋𖽡𖾏, 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐. 11 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽙𖾏 𖼖𖽔𖾑 𖼸𖽙 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼖𖽔𖾑 𖽐𖼷𖽜𖾑 𖼋𖽘𖾏, 𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽐𖼎𖽙 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏: “𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼀𖽪𖾏 𖼺𖽷𖾑 𖼙𖽻 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼞𖽔𖾏 𖼀𖽱𖾏 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏.” 12 𖼷𖽷𖾑 𖼮𖽵𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼀𖽡𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼙𖽻 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐, 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼀𖽡𖾐 𖼋𖽡𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼙𖽠𖾏.” 13 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼡𖽑𖽜𖾑 𖼐𖽑𖽡𖾏: “𖼋𖽔𖾏 𖼀𖽡𖾐 𖼅𖽻𖾏 𖼀𖽱𖾏 𖼚𖽔𖾏 𖼋𖽡𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖾑 𖽃𖽡𖾑 𖼡𖽑𖽙𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼄𖽔𖾐.” 14 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼊𖽑𖽔𖾏, 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼸𖽙 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼑𖽡𖾏 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏, 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖽐𖼷𖽜𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏. 15 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼀𖽱𖾏 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏, 𖼎𖽺𖾏 𖼼𖽻𖾏 𖽐𖼚𖽙 𖼙𖽻 𖽃𖽡𖾑 𖼖𖽪𖾑 𖼟𖽡𖾏, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽱𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏.
