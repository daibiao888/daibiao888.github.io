𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 2 “𖼟𖽡 , 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼘𖽪！ 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖽃𖽔𖾏 𖽐𖼞𖽻𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐, 3 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼳𖽡𖽠𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼞𖽙𖾏 𖼗𖽙𖾏 𖼷𖽠𖾏, 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏, 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼘𖽪! 𖼟𖽡 𖼙𖽔 𖼐𖽑𖽡𖾏:
𖼞𖽪𖾑 𖼟𖽪𖾏 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏.
4 𖼟𖽡 𖽐𖼚𖽜𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖼟𖽡 𖼟𖽙𖾏 𖼟𖽡 𖼻𖽻𖾏 𖽐𖼞𖽻𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝.
5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼳𖽷𖾏 𖼐𖽡 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽑𖽪𖾑 𖽃𖽔𖾏 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏,
𖼎𖽑𖽻𖾏 𖼖𖽡 𖼀𖽔𖾑 𖼐𖽺𖾏 𖼑𖽡𖾏 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖼊𖽪𖾑 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖼮𖽘𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐.
6 𖼎𖽑𖽻𖾏 𖼀𖽔𖾑 𖼳𖽔𖾑 𖼑𖽡𖾏 𖼳𖽤𖾏 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 𖼟𖽡 𖼊𖽪𖾑 𖼚𖽔𖾑 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽐𖼞𖽙𖾏,
𖼎𖽑𖽻𖾏 𖽐𖼷𖽑𖽱𖾏 𖼟𖽻𖾏 𖼗𖽙𖾏 𖼎𖽑𖽹𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏( “𖽃𖽔𖾑 𖼸𖽡𖽝𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏”𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼡𖽘𖾐 𖽐𖼞𖽙𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼸𖽡𖽝𖾏” ) 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼺𖽹𖾏 𖼀𖽑𖽪𖾐 𖼖𖽪𖾏 𖼺𖽷𖾑 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖼑𖽡𖾏 𖽐𖼊𖽻𖾏 𖽃𖽔𖾏 .
7 𖼟𖽡 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼣𖽹 𖼮𖽡 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖼅𖽙 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼁𖽘𖾏 𖼎𖽻𖾏 𖽃𖽔𖾏,
𖽃𖽔𖾏 𖼟𖽡 𖼛𖽔𖾏 𖼀𖽑𖽳𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐,
𖼟𖽡 𖼛𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼳𖽡𖽠𖾑 𖼎𖽻𖾐 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼀𖽑𖽜𖾑 𖽐𖼷𖽔𖾐 𖽐𖼎𖽙 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼖𖽡𖽝𖾐 𖼀𖽑𖽜𖾑 𖽃𖽔𖾏.
8 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼊𖽪𖽝𖾏 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼇𖽔 𖽃𖽔𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼞𖽱𖾏 𖽐𖼞𖽙𖾏,
𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼘𖽪! 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽃𖽔𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼯𖽵𖾏 𖽐𖼞𖽙𖾏.
9 𖼮𖽢𖾑 𖼀𖽔𖾑 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼗𖽻𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳
𖼐𖽡𖽙𖾐 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼷𖽷𖾑 𖼺𖽠𖾏,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼞𖽙𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖽐𖼎𖽙 𖼖𖽪𖾑 𖽐𖼞𖽙𖾏 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼼𖽙,
𖼐𖽡𖽙𖾐 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏.
10 𖼀𖽙𖾑 𖼺𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐、 𖼖𖽪𖾏 𖼊𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐、 𖼇𖽪 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼏𖽙 𖼖𖽪𖾑 𖼾𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽘𖾏 𖼮𖽡𖽝𖾏 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼊𖽑𖽡𖽝𖾐 𖼖𖽪𖾑 𖼮𖽱𖾐 𖼖𖽱𖾐 𖼋𖽻𖾏, 𖼢𖽳 𖼋𖽠𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏. 11 𖼽𖽔𖾏 𖼇𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼟𖽡 𖼸𖽻𖾏 𖼏𖽙 𖼻𖽙𖾑 𖼚𖽻𖾐 𖼺𖽘𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻, 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼖𖽪𖾑 𖽐𖼊𖽑𖽘𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼞𖽳 𖼄𖽔𖾐 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽘𖾏 𖼮𖽡𖽝𖾏 𖼛𖽔𖾏 𖼟𖽪𖾏 𖽐𖼎𖽳𖾏,𖼟𖽙𖾏 𖼟𖽡 𖼻𖽻𖾏 𖼋𖽘𖾏 𖽐𖼮𖽢𖾏. 12 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖼟𖽡 𖼅𖽔𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼟𖽡 𖼊𖽪𖾐 𖼉𖽻 𖼙𖽔𖾏, 𖼯𖽡𖽙𖾏 𖼑𖽡𖽝𖾏、 𖼖𖽑𖽻𖾏、 𖼳𖽡𖾏、 𖼷𖽑𖽔𖾏, 𖼋𖽔𖾏 𖼘𖽑𖽪𖾑 𖼟𖽡 𖼚𖽘𖾐 𖼐𖽜𖾐. 13 𖼽𖽔𖾐 𖽂𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐、 𖼊𖽑𖽪𖾐 𖼀𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐、 𖼄𖽡𖾐 𖼳𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏,𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙
