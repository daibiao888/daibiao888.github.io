𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 , 𖼻𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑 . 2 𖼟𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 , 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖼟𖽡𖾏 𖽐𖼮𖽔𖾏 𖼟𖽡 𖼚𖽻𖾐 𖼺𖽘𖾐 . 3 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼽𖽔𖾑 𖼘𖽘𖾑𖽑 𖼋𖽠𖾏 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 , 𖼋𖽡𖾏 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖽐𖼎𖽙 𖼟𖽡 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼀𖽱𖾏 , 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼸𖽘𖾏 , 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼉𖽻 𖼷𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 , 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼷𖽙𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼖𖽪𖾑 𖼋𖽺𖾏 𖼟𖽪𖾏 𖼄𖽑𖽻𖾏 𖼳𖽡𖽝𖾐 , 4 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽻 𖼷𖽻𖾏 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 .”
5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽑𖽡𖾏 : 𖼞𖽑𖽡𖽠𖾏 𖼟𖽡 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏、𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐、𖼁𖽘𖾏 𖽃𖽻𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖼟𖽙𖾏 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖽐𖼮𖽡𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼺𖽘𖾏 .” 6 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽻𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑 , 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖼎𖽺𖾏 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 . 7 𖼊𖽪𖾐 𖽃𖽔𖾏 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖼚𖽘𖾐 𖼸𖽷𖾏 𖼅𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 , 𖼟𖽙𖾏 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 .
8 𖼇𖽔 𖼘𖽻𖾐 𖼨𖽑𖽪𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼋𖽔𖾏 𖼐𖽑𖽡𖾏 : “𖼮𖽑𖽙 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖼐𖽡𖾑 𖼇𖽳𖾑 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼺𖽘𖾏 , 𖼞𖽪𖾑 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 𖼖𖽑𖽘𖾑 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .” 9 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼇𖽔 𖼘𖽻𖾐 𖼐𖽑𖽡𖾏 : “𖼯𖽔𖾏 𖼟𖽡 ! 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼊𖽻𖾑𖽑 𖼋𖽱 𖼑𖽜 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼉𖽻 𖼷𖽻𖾏 𖼊𖽑𖽡𖽝𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼋𖽠𖾏 𖼺𖽘𖾏 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖼟𖽙𖾏 𖼇𖽳𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 , 𖼮𖽔𖾑 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼮𖽡𖽝𖾏 𖼊𖽔𖾐 𖼖𖽺𖾑 ?” 10 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼀𖽡 𖼟𖽡 .” 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼸𖽙𖾏 𖼟𖽡 𖼗𖽪𖾏 ! 𖼟𖽙𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼺𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼀𖽡𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼯𖽡𖽙𖾏 . 11 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽻 𖼊𖽑𖽡𖽝𖾐 𖼟𖽡 𖼉𖽻 𖼷𖽻𖾏 𖽐𖼎𖽙 𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 , 𖼮𖽔 𖼷𖽷𖾐 𖼎𖽻𖾏 𖼐𖽡 𖼘𖽙 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼮𖽡𖽝𖾏 𖼊𖽔𖾐 .” 12 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼇𖽳𖾑 𖼇𖽔 𖼘𖽻𖾐 𖼺𖽘𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏. 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼑𖽜 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖼟𖽪𖾏 𖼟𖽡𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 . 13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖼸𖽙𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽔𖾏 , 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏、𖼖𖽪𖾑 𖼊𖽡𖾏 𖼯𖽡𖽳𖾏、𖽐𖼎𖽹𖾏 𖼙𖽡𖽝𖾏 𖼞𖽑𖽔𖾏 𖼋𖽔 𖼺𖽘𖾏 . 14 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾏 𖽐𖼎𖽦𖾑 𖼡𖽔 𖼺𖽱𖾏 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼚𖽻𖾏 𖼮𖽡𖽝𖾏 , 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼖𖽔𖾏 𖼮𖽵𖾏 𖼗𖽱𖾏 . 15 𖼈𖽡𖽝𖾏 𖼇𖽔 𖼘𖽻𖾐 𖼁𖽙𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼳𖽷𖾐 𖼋𖽘 , 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼎𖽺𖾏 𖼼𖽔𖾏 𖼦𖽡𖾐 𖼐𖽑𖽡𖽻𖾏 𖼑𖽻 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏 .
