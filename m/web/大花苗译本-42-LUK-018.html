𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼐𖽑𖽡𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏，𖼟𖽪𖾏 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼀𖽑𖽪𖾑 𖼺𖽡𖽝𖾐． 2 𖼐𖽑𖽡𖾏： “ 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏，𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽑𖽡𖽝𖾐． 3 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 ，𖼯𖽪 𖼯𖽪𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑，𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽜 𖼞𖽪𖾑 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏．’ 4 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 ，𖽃𖽔𖾑 𖼡𖽜𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽑𖽡𖾏：‘𖼞𖽪𖾑 𖼦𖽡𖾐 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐， 5 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖽐𖼷𖽔𖾏 𖼐𖽡𖾑 𖼮𖽑𖽙 𖼞𖽪𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝，𖼞𖽪𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏，𖼦𖽡𖾐 𖽐𖼮𖽑𖽠𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼋𖽔𖾏 𖼮𖽑𖽙 𖼞𖽪𖾑．’ ” 6 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼐𖽑𖽙𖾑 𖼷𖽷𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼐𖽡𖾑 ，𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏． 7 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽹𖾑 𖼊𖽻𖾏，𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏，𖼋𖽡𖽝𖾏 𖼾𖽙 𖼐𖽱𖾏 𖼊𖽑𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏，𖼦𖽡𖾐 𖼞𖽳𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽪𖾏，𖼦𖽡𖾐 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼘𖽪？ 8 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼐𖽱𖾏 𖼽𖽔𖾑 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏．𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼎𖽺𖾏 𖼋𖽔𖾏，𖼋𖽡𖽝𖾏 𖼡𖽑𖽔𖾑 𖼁𖽙𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼘𖽪？”
𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
9 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼡𖽑𖽳𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳，𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼊𖽪𖾑 𖼗𖽪𖾏 𖽐𖼮𖽔𖾏 𖼐𖽡𖾑 ， 10 𖼐𖽑𖽡𖾏 ：“ 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖽃𖽡𖾐 𖼖𖽜𖾐 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏． 11 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏，𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼗𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽑𖽡𖾏：‘𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！𖼞𖽪𖾑 𖼮𖽑𖽔𖾏 𖼙𖽔 𖼟𖽡，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼟𖽪𖾏 𖼼𖽠𖾏 𖼀𖽑𖽡𖽝𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐、 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏、 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼷𖽷𖾑 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼐𖽡𖾑． 12 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼞𖽑𖽱𖾏 𖼊𖽪𖾏 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼡𖽜𖾑，𖼞𖽪𖾑 𖼊𖽻𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼟𖽻𖾏 𖽐𖼀𖽠 𖼺𖽘𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．’ 13 𖼷𖽷𖾑 𖼊𖽪𖾐 𖼳𖽻𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖽃𖽡𖾑 𖼳𖽠𖾑 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝， 𖼦𖽡𖾐 𖼞𖽔𖾐 𖼮𖽘𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖽐𖼊𖽪𖾏，𖼮𖽔𖾑 𖽐𖼀𖽔𖾏 𖽐𖼎𖽙𖾐 𖼐𖽑𖽡𖾏：‘𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！𖼡𖽑𖽜𖾐 𖼅𖽳𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑 𖼷𖽷𖾑 𖼊𖽪𖾐 𖼾𖽔𖾏 𖼐𖽡𖾑．’ 14 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏，𖼺𖽙𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑．𖽃𖽡𖾑 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐 𖼾𖽨𖾏，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼢𖽜𖾏，𖼷𖽷𖾑 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼦𖽡𖾏 𖼢𖽜𖾏 𖼾𖽨𖾏 ，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐．”
𖼗𖽪𖾏𖽐𖼮𖽔𖾏𖼟𖽪𖾏𖼐𖽑𖽡𖾏𖼇𖽔𖼖𖽡𖼺𖽹𖾏𖽐𖼎𖽙𖼊𖽪𖾐𖼳𖽻𖾏𖼳𖽳𖾏
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼑𖽜 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼏𖽵 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏
（𖼊𖽑𖽹𖾏 19：13-15；𖼞𖽑𖽙𖾐 10：13-16）
15 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽔 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼽𖽔𖾑 𖼐𖽱𖾏 𖼚𖽑𖽱𖾏 𖼚𖽑𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼁𖽙𖾏 𖼎𖽺𖾏 𖼖𖽔𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑． 16 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏， 𖼐𖽑𖽡𖾏：“ 𖼯𖽔𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙，𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏． 17 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖽃𖽡𖾑，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼺𖽱𖾏 𖼸𖽻𖾏 𖼣𖽔𖾏 𖼽𖽻𖾏 𖼐𖽡𖾑，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏．”
𖼉𖽻 𖼷𖽻𖾏 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏
（𖼊𖽑𖽹𖾏 19：16-30；𖼞𖽑𖽙𖾐 10：17-31）
18 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼉𖽻 𖼷𖽻𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏，𖼞𖽪𖾑 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏，𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼸𖽠𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏？” 19 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼖𖽺𖾑 ？ 𖼋𖽠𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏，𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏． 20 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼀𖽻𖾐 𖼯𖽡𖽙：‘𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏，𖼦𖽡𖾐 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ，𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼺𖽘𖾐 𖽃𖽔𖾏 𖼸𖽜，𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 ，𖼽𖽔𖾑 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏．’ ” 21 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏： “ 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼞𖽑𖽔𖾏 𖼻𖽙𖾑 𖼋𖽘𖾏．” 22 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽙𖾑 𖼋𖽘𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼯𖽠 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼺𖽷𖾐，𖼽𖽔𖾑 𖼋𖽺 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏 𖼺𖽘𖾏，𖼇𖽺𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼳𖽻𖾏 ， 𖼟𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏，𖼟𖽡 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼋𖽔𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐．” 23 𖼐𖽱𖾏 𖼐𖽑𖽙𖾑 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼊𖽳𖾏 𖼊𖽔𖾐 𖼋𖽡𖽝，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼅𖽔𖾏 𖼊𖽻𖾏．
24 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙，𖼟𖽪𖾏 𖼁𖽺𖾏 𖼊𖽔𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝！ 25 𖼘𖽙 𖼊𖽑𖽙𖾐 𖼁𖽺𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼞𖽻𖾐，𖼖𖽡 𖼟𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙．” 26 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽙𖾑 𖼐𖽑𖽡𖾏：“ 𖼮𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼮𖽨𖾏 𖼘𖽪？” 27 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 ．” 28 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼑𖽔𖾏，𖼀𖽡𖾐 𖼇𖽳𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼅𖽔𖾏 𖼸𖽙𖾏 𖼟𖽡 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼋𖽘．” 29 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏： “ 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼞𖽪𖽙 𖼇𖽳𖾑 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏、𖼇𖽳𖾑 𖼐𖽤𖾐、𖼇𖽳𖾑 𖼞𖽱𖾑 𖼋𖽡𖾏、𖼇𖽳𖾑 𖼑𖽡𖽝𖾏 𖼉𖽺𖾏、𖼇𖽳𖾑 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏， 30 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼷𖽑𖽷𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼟𖽪𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼮𖽡𖽻𖾐，𖼐𖽡𖽙𖾐 𖼷𖽑𖽷𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼗𖽙𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏．”
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼷𖽷𖾐 𖼟𖽻𖾐 𖼟𖽻𖾐 𖽃𖽡𖾑 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼋𖽔 𖼈𖽡𖽝𖾏 𖼽𖽔𖾑 𖼎𖽺𖾏 𖼯𖽡𖽝𖾏
（𖼊𖽑𖽹𖾏 20：17-19；𖼞𖽑𖽙𖾐 10：32-34）
31 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼨𖽑𖽳𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼑𖽔𖾏 𖼘𖽪！ 𖼀𖽡𖾐 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 ，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼺𖽻𖾏 𖼮𖽡𖽝𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼅𖽔𖾏 𖼞𖽳𖾑 𖼖𖽔𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑． 32 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 ，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼚𖽙𖾏 𖼐𖽱𖾏、𖼊𖽱𖾑 𖼷𖽑𖽷𖾑 𖼐𖽱𖾏 、𖽐𖼊𖽙𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖽻𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼚𖽑𖽪𖾏， 33 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖽐𖼚𖽠𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏， 𖼐𖽑𖽙𖾏 𖼐𖽱𖾏 𖼺𖽘𖾏，𖼸𖽙 𖼷𖽷𖾐 𖼐𖽑𖽪𖾐 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼯𖽡𖽝𖾏 𖼗𖽙．” 34 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽪𖾏 𖼀𖽻𖾐 𖼦𖽡𖾐 𖽐𖼎𖽑𖽹𖾏 ，𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼮𖽡𖽝𖾏，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏．
𖼽𖽦𖾑𖼺𖽪𖾑𖽐𖼊𖽔𖾏𖼐𖽑𖽡𖾏𖼐𖽱𖾏𖼋𖽔𖼎𖽺𖾏𖼯𖽡𖽝𖾏
𖼽𖽦𖾑 𖼺𖽪𖾑 𖼞𖽑𖽙𖾏 𖼐𖽔𖾐 𖼄𖽱𖾐
（𖼊𖽑𖽹𖾏 20：29-34；𖼞𖽑𖽙𖾐 10：46-52）
35 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼽𖽔𖾑 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽡𖾏 𖼞𖽙𖾑 𖼙𖽻 𖽃𖽡𖾑， 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼼𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 𖼳𖽻𖾏 𖼡𖽜𖾑． 36 𖼐𖽑𖽙𖾑 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏，𖼎𖽺𖾏 𖼑𖽪 𖼟𖽪𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏． 37 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏，𖼟𖽪𖾏 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽑𖽡𖽻𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑． 38 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏： “ 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼘𖽪！𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑 ．” 39 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼖𖽔𖾏 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼐𖽱𖾏 𖼨𖽑𖽪𖾏，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏， 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐 𖼞𖽪𖾑 ！” 40 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼳𖽠𖾑 𖽐𖼎𖽙𖾏，𖼡𖽑𖽔𖾏 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏，𖼐𖽱𖾏 𖼋𖽔𖾏 𖼸𖽙 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 ， 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： 41 “ 𖼟𖽡 𖼽𖽔𖾑 𖼞𖽪𖾑 𖼑𖽜 𖼟𖽡 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏： “ 𖼮𖽵𖾑 𖼘𖽪 ！𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼊𖽔𖾐．” 42 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏！𖼟𖽡 𖽐𖼮𖽜 𖼮𖽨𖾏 𖼟𖽡 𖼋𖽘𖾏．” 43 𖼐𖽔𖾐 𖼄𖽱𖾐 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼋𖽘𖾏，𖼎𖽺𖾏 𖼸𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐，𖼄𖽔𖾐 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏． 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼁𖽙𖾏 𖼊𖽪𖾑 𖼑𖽪𖾏 𖼐𖽡𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼷𖽻𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．
𖼽𖽦𖾑𖼺𖽪𖾑𖼞𖽑𖽙𖾏𖼐𖽔𖾐𖼄𖽱𖾐
