𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼞𖽗𖾐 𖼖𖽗 𖼇𖽻𖾑 𖼎𖽻𖾐． 2 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏．𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏． 3 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏 𖽐𖼎𖽙 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼊𖽻𖾏 𖼄𖽔𖾐 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼟𖽙𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐， 4 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼡𖽑𖽔𖾐 𖽐𖼊𖽠𖾏，𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖾑 𖼸𖽪𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 𖼷𖽷𖾏 𖼖𖽘𖾑 𖼊𖽻𖾏 𖼄𖽔𖾐 𖼯𖽡𖽙． 5 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼀𖽡𖾐，𖼮𖽘𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼟𖽙𖾏 𖼋𖽔．𖼟𖽡 𖼐𖽑𖽡𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？” 6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼽𖽔𖾑 𖽐𖼮𖽔𖾏 𖼮𖽑𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑，𖽐𖼎𖽑𖽦𖾑 𖼞𖽡𖾑 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼐𖽱𖾏．𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑，𖼎𖽑𖽻𖾏 𖽃𖽔𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏 𖼺𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼊𖽠𖾑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐． 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼊𖽪𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼁𖽜𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 ： “ 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼡𖽔 𖼋𖽱𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼺𖽪𖾐 𖼗𖽔𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏．” 8 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼎𖽻𖾑 𖼺𖽻𖾏 𖽃𖽔𖾑 𖼅𖽔 𖽐𖼊𖽠𖾑 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐． 9 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏，𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼗𖽻𖾏 𖽃𖽡𖾑 𖼁𖽱𖾏， 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼋𖽠𖾏 𖼙𖽠𖾏； 𖼯𖽔 𖼷𖽷𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽃𖽡𖾑 𖼮𖽡𖽝𖾏 𖼊𖽔𖾐． 10 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼁𖽜𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔 𖼁𖽙𖾏，𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏？ 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼊𖽡𖾏 𖼟𖽡 𖼾𖽔𖾏 𖼖𖽔𖾏？” 11 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼘𖽪 ！𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼊𖽡𖾏 𖼟𖽡 𖼾𖽔𖾏．𖼟𖽡 𖼅𖽻𖾏 𖼦𖽡𖾑！𖼋𖽳 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼾𖽔𖾏．”
