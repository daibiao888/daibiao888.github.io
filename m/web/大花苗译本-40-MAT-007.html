（𖼖𖽪𖾏 6：37- 38；41- 42）
7
“ 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼗𖽔 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼨𖽑𖽢𖾐 𖼅𖽡 ． 2 𖼅𖽡 𖼨𖽑𖽢𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼅𖽡 𖼖𖽡𖾏 𖼯𖽤𖾏． 𖼅𖽡 𖼎𖽑𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼙𖽪𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼙𖽪𖾏 𖼎𖽻𖾏 𖼅𖽡． 3 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼁𖽙𖾏 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼐𖽡𖽙𖾐， 𖼈𖽡𖽝𖾏 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙 𖼊𖽪𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼖𖽺𖾑 ？ 4 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼅𖽔𖾏 𖼊𖽪𖾑 𖽐𖼊𖽻𖾏，𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼎𖽙 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼐𖽑𖽡𖾏 ：‘ 𖼯𖽔𖾏 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼋𖽠𖾏 𖼗𖽙𖾏 𖼖𖽺𖾑 ？’ 5 𖼟𖽡 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽑𖽳𖾏 𖼅𖽳𖾏 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼐𖽡𖾑 ，𖼟𖽡 𖼺𖽪𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽑𖽡𖽠𖾏 𖼊𖽪𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼾𖽨𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼗𖽔𖾏 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼸𖽠𖾏 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑，𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼅𖽻𖾏 𖼞𖽑𖽡𖽠𖾏 𖽃𖽔𖾑 𖼸𖽔𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼞𖽱𖾑 𖼋𖽡𖾏 𖽃𖽔𖾑 𖼅𖽔 𖼗𖽔𖾏 𖼺𖽘𖾏． 6 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼎𖽻𖾏 𖼊𖽪𖾑 𖼚𖽡𖾑，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼗𖽔𖾏 𖼅𖽡 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼎𖽻𖾏 𖼊𖽪𖾑 𖽐𖼀𖽔𖾏，𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽪𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼺𖽘𖾏，𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼗𖽙 𖼋𖽙𖾏 𖼅𖽡．”
𖼑𖽪𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏
（𖼖𖽪𖾏 11：9-13）
7 “ 𖼅𖽡 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏， 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼅𖽡， 𖼅𖽡 𖽐𖼎𖽑𖽦𖾑 ，𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏，𖼅𖽡 𖽐𖼀𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏，𖼎𖽺𖾏 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼅𖽡 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏． 8 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽃𖽡𖾑，𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏，𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽡𖾑，𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼀𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏 𖽃𖽡𖾑， 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏 ． 9 𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼷𖽔𖾑 𖼊𖽪𖾐 𖼽𖽔𖾑 𖼳𖽡𖽝𖾐 𖼑𖽻𖾏， 𖼎𖽺𖾏 𖼄𖽔𖾐 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼖𖽺𖾑 ？ 10 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼊𖽔𖾑 𖽐𖼀𖽳𖾏， 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼐𖽘𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼘𖽪 ？ 11 𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏，𖼈𖽡𖽝𖾏 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼎𖽻𖾏 𖼅𖽡 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏，𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼅𖽡 𖼉𖽺𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏，𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼄𖽔𖾐 𖼻𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼐𖽱𖾏 𖼘𖽪 ？ 12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼅𖽡 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏，𖼅𖽡 𖽐𖼮𖽑𖽠𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽙𖾏 𖼅𖽡 𖼖𖽡𖾏 𖼯𖽤𖾏，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽡𖾏 𖼯𖽤𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼎𖽙 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏．”
𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏
（𖼖𖽪𖾏 13：24）
13 “ 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼁𖽺𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏．𖽃𖽡𖾑 𖼟𖽪𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼇𖽘𖾑 ， 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐． 14 𖼈𖽡𖽝𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑， 𖽃𖽔 𖼛𖽻𖾏 𖼟𖽪𖾏 𖽐𖼡𖽹𖾏，𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼘𖽑𖽙𖾐 ，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼯𖽠 ． ”
𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑
（𖼖𖽪𖾏 6：43-44）
15 “ 𖼅𖽡 𖼽𖽔𖾑 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑．𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼅𖽡， 𖼳𖽪𖾏 𖼡𖽻𖾏 𖼐𖽑𖽘𖾑 𖼾𖽘𖾏 𖼮𖽑𖽙𖾏，𖼉𖽺𖾏 𖽐𖼚𖽙 𖼟𖽪𖾏 𖼄𖽑𖽘𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏． 16 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽙．𖼞𖽡𖾑 𖼁𖽙𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼚𖽡𖾏 𖼊𖽻𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼘𖽪？𖼞𖽑𖽻𖾐 𖼁𖽙𖾏 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼚𖽡𖾏 𖼊𖽻𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼘𖽪？ 17 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 ， 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼷𖽑𖽔𖾏 𖼇𖽘𖾐 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏，𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏． 18 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏．𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏， 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏． 19 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼷𖽷𖾏 𖼻𖽻𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽡𖾑，𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽙𖾑 𖼗𖽙𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼯𖽘𖾏 𖼋𖽠． 20 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐，𖼎𖽺𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏．”
𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏
（𖼖𖽪𖾏 13：25-27）
21 “ 𖼦𖽡𖾐 𖼾𖽙 𖼷𖽑𖽔𖾏 𖼗𖽜𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾏 𖼞𖽪𖾑 ‘ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼮𖽵𖾑 𖼘𖽪！’ 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖽐𖼊𖽪𖾏 𖼞𖽪𖽙，𖼮𖽔 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖽃𖽔𖾏 ，𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼊𖽻𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏． 22 𖼸𖽙 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏：‘ 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼮𖽵𖾑 𖼘𖽪！ 𖼋𖽡𖽝𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏，𖼋𖽡𖽝𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖼖𖽠𖾑 𖼀𖽡𖾑 𖼚𖽘𖾐，𖼋𖽡𖽝𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽡𖽝𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏？’ 23 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼟𖽡𖾏 𖼚𖽑𖽔𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：‘ 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖼀𖽻𖾐 𖼅𖽡，𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼐𖽡𖾑， 𖼀𖽜𖾑 𖼞𖽪𖾑 𖼙𖽠𖾏！’ ”
𖽃𖽔𖾐 𖼮𖽘𖾏 𖼊𖽠𖾏 𖽐𖼞𖽔𖾏
（𖼖𖽪𖾏 6：47-49 ）
24 “ 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽙𖾑𖽑 𖼞𖽪𖾑 𖼗𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼅𖽻𖾏 𖽃𖽔𖾏，𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐，𖼅𖽻𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑． 25 𖼑𖽘 𖼗𖽙𖾏，𖽃𖽻𖾐 𖼷𖽑𖽪𖾐， 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖽐𖼊𖽱𖾑𖽑 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑，𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑 𖼦𖽡𖾐 𖼼𖽠𖾏 𖼦𖽡𖾐 𖼢𖽻 𖼙𖽠𖾏，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽠𖾏 𖽐𖼞𖽔𖾏 𖼷𖽑𖽙𖾏 𖼎𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖽃𖽔𖾑 𖼈𖽳𖾐 𖼐𖽔𖾑． 26 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼐𖽙𖾑𖽑 𖼞𖽪𖾑 𖼗𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏 𖽃𖽔𖾏 ， 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔， 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼷𖽑𖽙𖾏 𖽐𖼞𖽔𖾏 𖼎𖽻𖾏 𖽃𖽔 𖼗𖽔𖾏 𖼄𖽡𖾐 𖼄𖽡𖾐 𖼦𖽡𖾏 𖼇𖽔𖾑． 27 𖼑𖽘 𖼗𖽙𖾏，𖽃𖽻𖾐 𖼷𖽑𖽪𖾐 ， 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖽐𖼊𖽱𖾑𖽑 𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑，𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑 𖼎𖽺𖾏 𖼼𖽠𖾏 𖼢𖽻 𖼙𖽠𖾏，𖼎𖽺𖾏 𖼼𖽠𖾏 𖼢𖽻 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝．” 28 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽙𖾑𖽑 𖼑𖽡𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼎𖽺𖾏 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏， 29 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏， 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏，𖼦𖽡𖾐 𖼺𖽱𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏．
