𖼞𖽑𖽻𖾑 𖼉𖽻𖾏 𖼮𖽘𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫

𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 !
 
𖼅𖽡 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏！
2 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏．
𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽐𖼊𖽪𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏．
 
3 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏；
𖼐𖽑𖽪𖾐 𖼡𖽪𖾐 𖼟𖽪𖾏 𖼮𖽡𖾏 𖼁𖽙𖾏 𖼞𖽡𖾑 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏．
4 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖽐𖼎𖽙 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼑𖽡𖾏 𖽃𖽻𖾐 𖼘𖽪！
𖼅𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏．
 
5 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽂𖽪𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 , 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼞𖽑𖽔𖾏 𖼷𖽷𖾏 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏.
6 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼷𖽑𖽙𖾏 𖼏𖽔𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼦𖽡𖾐 𖼋𖽘𖾏,
𖼐𖽱𖾏 𖼏𖽵 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼡𖽘𖾑𖽑 𖼢𖽔𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼎𖽑𖽔𖾑 𖼺𖽘𖾏 ( “𖼎𖽑𖽔𖾑 𖼺𖽘𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼮𖽑𖽡𖽻𖾏 𖼙𖽠𖾏” ).
 
7 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽪𖾏 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼖𖽪𖾑 𖼳𖽡𖽠𖾑 𖼊𖽪𖾑 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽡𖽠𖾑 𖽃𖽻𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐,
8 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽐𖼎𖽙 𖼑𖽘 𖽃𖽔𖾐 𖼖𖽪𖾐 , 𖽐𖼀𖽪𖾏 𖽐𖼎𖽙 𖼦𖽘𖾑 𖼀𖽙𖾏,
𖼮𖽢𖾏 𖼘𖽑𖽙𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 ,
9 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼖𖽔𖾑 𖼣𖽔𖾏 𖼎𖽻𖾐,
𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽤𖾏 𖼀𖽳 𖽐𖼊𖽻𖾏 ,
10 𖼯𖽦𖾏 𖼞𖽙𖾏 𖽃𖽔𖾏 𖼸𖽔𖾏 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖼉𖽺𖾏 𖽐𖼞𖽔𖾏,
𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖽐𖼞𖽘 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼐𖽻 𖼟𖽪𖾏 𖼽𖽘𖾏 ,
11 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏,
𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼑𖽪𖾏 ,
12 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽐𖼎𖽙 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏,
𖽃𖽔𖾏 𖼗𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼣𖽔𖾑 𖼽𖽻𖾏, 𖼅𖽡 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫！
13 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏！
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼮𖽔 𖼅𖽔𖾏 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼺𖽡𖽝𖾐, 𖼐𖽱𖾏 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼊𖽡𖾐 𖽃𖽔𖾑 𖼳𖽔𖾐.
14 𖼐𖽱𖾏 𖼞𖽑𖽡𖽠𖾏 𖼐𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼮𖽘𖾏 𖼟𖽙𖾏 𖼺𖽡𖽝𖾐.
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼐𖽱𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐, 𖼟𖽪𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼐𖽱𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼐𖽱𖾏．
 
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 !
