𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼺𖽱𖾏 𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏
9
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖼮𖽑𖽡𖽠𖾏,
𖼸𖽻 𖼊𖽡𖽝𖾑 𖼳𖽤𖾏 𖼋𖽪𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐.
2 𖼊𖽔𖾏 𖼯𖽦𖾏, 𖼳𖽡𖽠𖾏 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼊𖽜𖾐 𖽐𖼡𖽹𖾏 𖼡𖽜𖾏 𖼮𖽡𖽝𖾏.
3 𖼷𖽷𖾐 𖼟𖽙𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽜𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏,
𖼾𖽨𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼙𖽻 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏,
4 𖼐𖽑𖽡𖾏 : “𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽙𖾑 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑！”
𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 𖼐𖽑𖽡𖾏 :
5 “𖼅𖽡 𖼋𖽔𖾏, 𖼑𖽻𖾏 𖼞𖽪𖾑 𖼳𖽡𖽝𖾐, 𖼦𖽻𖾏 𖼞𖽪𖾑 𖼮𖽡𖽠𖾑 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼳𖽡𖽠𖾏.”
6 𖼅𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 , 𖼽𖽔𖾑 𖼗𖽔𖾏 𖼺𖽘𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔, 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏,
𖼎𖽺𖾏 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐.”
 
7 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼡𖽑𖽳𖾏 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼗𖽔 𖽃𖽡𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼾𖽨𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽪𖾐 𖽐𖼚𖽪𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏.
8 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼗𖽔 , 𖽐𖼮𖽑𖽹𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼳𖽪𖾐 𖼟𖽡,
𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼮𖽑𖽠𖾏 𖼟𖽡 .
9 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼐𖽱𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 ,
𖼡𖽑𖽔𖾐 𖼡𖽑𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼀𖽻𖾐 𖼟𖽳 .
10 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏,
𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐.
11 𖼟𖽡 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑 , 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐,
𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽑𖽡𖽝𖾐 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
12 𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 , 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼾𖽨𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏；
𖼟𖽡 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼗𖽔, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼾𖽨𖾏 𖼽𖽔𖾑 𖼞𖽱𖾑 𖼎𖽷𖾏.
𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽻𖾐
13 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼗𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐,
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔，𖼦𖽡𖾐 𖼀𖽻𖾐 𖼞𖽔𖾐 𖼡𖽔𖾏 𖼳𖽷𖾏.
14 𖼐𖽱𖾏 𖼼𖽻𖾏 𖼾𖽨𖾏 𖽐𖼞𖽔𖾏 𖽃𖽔 𖼛𖽻𖾏,
𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼙𖽻 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐，
15 𖼨𖽑𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽙𖾏 𖼞𖽡𖾑 ,
𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖽃𖽡𖾑,
16 𖼐𖽑𖽡𖾏 : “𖼡𖽔 𖼋𖽱𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽔 𖼏𖽔 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖼐𖽡𖾑！”
𖼎𖽺𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 𖼐𖽑𖽡𖾏 :
17 “𖽃𖽻𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝 𖼗𖽙𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑,
𖼳𖽡𖽝𖾐 𖼟𖽪𖾏 𖼈𖽺𖾏 𖼑𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼡𖽘𖾐．”
18 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼊𖽪𖾐 𖼋𖽔𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐,
𖼐𖽱𖾏 𖼡𖽑𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐.
