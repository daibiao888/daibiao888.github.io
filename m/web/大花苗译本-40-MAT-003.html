𖼽𖽙 𖼦𖽗 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏
（𖼞𖽑𖽙𖾐 1：1-8；𖼖𖽪𖾏 3：1-18；𖼽𖽙 1：19-28）
3
𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼅𖽔𖾏 𖼄𖽔𖾐 𖽐𖼷𖽔𖾑 𖼽𖽙 𖼦𖽗 𖼋𖽔𖾏，𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼡𖽑𖽔𖾐 𖼞𖽘𖾐 𖼞𖽡𖾏，𖼐𖽑𖽡𖾏： 2 “𖽐𖼊𖽪𖾏 𖼞𖽪𖽙 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏，𖼅𖽡 𖼽𖽔𖾑 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻．” 3 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽹 𖼽𖽔𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼯𖽡𖽙𖾏，𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：
“ 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼅𖽔𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏，
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼮𖽵𖾑 𖼞𖽡𖾑，
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾏 𖼐𖽱𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼞𖽳𖾏．”
4 𖼷𖽷𖾑 𖼽𖽙 𖼦𖽗 𖼐𖽡𖾑 𖼐𖽑𖽘𖾑 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖼚𖽻𖾐 𖼘𖽙 𖼊𖽑𖽙𖾐 𖽐𖼊𖽙𖾏， 𖼞𖽑𖽙𖾏 𖼳𖽦𖾏 𖼖𖽑𖽔𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏， 𖼑𖽻𖾏 𖼷𖽷𖾑 𖼷𖽷 𖽐𖼎𖽙 𖼄𖽪𖾑 𖼞𖽔𖾏 𖼞𖽔𖾏 𖼉𖽺𖾏 𖼻𖽻𖾑． 5 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑， 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽐𖼎𖽙 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 ， 𖽐𖼎𖽙 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽔𖾐 𖼀𖽑𖽘𖾐 𖼞𖽑𖽡𖽝𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼦𖽗 𖼉𖽺𖾏 𖽃𖽡𖾑， 6 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽜𖾏 𖼾𖽔𖾏 𖼺𖽘𖾏， 𖼐𖽡𖽙𖾐 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽃𖽻𖾐 𖼊𖽻𖾏 𖼽𖽙 𖼦𖽗 𖽐𖼷𖽔𖾑． 7 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼁𖽙𖾏 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼺𖽔𖾐 𖼊𖽪𖾑 𖼞𖽹𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽻𖾏 𖽐𖼷𖽔𖾑， 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼘𖽪！ 𖼡𖽔 𖼋𖽱𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼅𖽡 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏． 8 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻． 9 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼐𖽑𖽡𖾏：‘ 𖼀𖽡𖾐 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖽃𖽔𖾏 𖼀𖽡𖾐 𖼉𖽺𖾏 ．’ 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 ，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼋𖽡𖾏 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼐𖽡𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼳𖽠𖾑 𖼋𖽠𖾏𖼽𖽔𖾏 𖼀𖽳 𖼖𖽔𖾐 𖼦𖽗𖾐 𖼊𖽪𖾐 𖼞𖽡𖾏． 10 𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼊𖽻𖾏 𖼮𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏 𖽐𖼊𖽻𖾏 𖼋𖽘𖾏， 𖼷𖽑𖽔𖾏 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼻𖽻𖾏， 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼊𖽙𖾑 𖼗𖽙𖾏 ， 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼯𖽘𖾏 𖼋𖽠． 11 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖽐𖼷𖽔𖾑 𖼅𖽡 𖼟𖽙𖾏 𖼅𖽡 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻， 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼗𖽙𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐， 𖼟𖽪𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼘𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑， 𖼞𖽪𖾑 𖼑𖽜 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼞𖽑𖽻𖾏 𖼞𖽪𖾑 𖼊𖽪𖾏 𖼦𖽡𖾐 𖽐𖼷𖽷𖾏 ， 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖽐𖼎𖽙 𖽃𖽔𖾏 𖼖𖽑𖽘𖾑 𖼋𖽠𖾏 𖽐𖼷𖽔𖾑 𖼅𖽡． 12 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼯𖽡𖽙𖾏 𖼈𖽘𖾐 𖼢𖽻𖾏， 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼳𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽡𖾏 𖼟𖽡𖾏 𖼟𖽙𖾏 𖼨𖽑𖽪𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼺𖽘𖾏， 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼳𖽻𖾐 𖼗𖽙𖾏 𖽐𖼊𖽡𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼼𖽪 𖼮𖽡𖽝𖾏，𖼷𖽑𖽷𖾑 𖼦𖽡𖾑 𖼁𖽘𖾏 𖼎𖽻𖾏 𖼯𖽘𖾏 𖼋𖽠 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼋𖽔 ．”
𖼽𖽙 𖼦𖽗 𖽐𖼷𖽔𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼞𖽑𖽙𖾐 1：9-11；𖼖𖽪𖾏 3：21-22）
13 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑， 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼋𖽔𖾏 𖼸𖽙 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼚𖽔𖾏 𖼽𖽙 𖼦𖽗𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼽𖽙 𖼦𖽗 𖽐𖼷𖽔𖾑 𖼐𖽱𖾏． 14 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽙 𖼦𖽗 𖽐𖼎𖽳𖾏 𖼐𖽱𖾏，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“𖼞𖽪𖾑 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼊𖽻𖾏 𖼟𖽡 𖽐𖼷𖽔𖾑 𖼞𖽪𖾑， 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼖𖽺𖾑 ？” 15 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼯𖽪𖾏 𖼯𖽡𖽙𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼋𖽘𖾏 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 （“𖼞𖽘𖾐 𖼞𖽡𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑” 𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 “𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳”）．” 𖼽𖽙 𖼦𖽗 𖼎𖽺𖾏 𖼯𖽪𖾏 𖼐𖽱𖾏． 16 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽻𖾏 𖽐𖼷𖽔𖾑 𖼋𖽘𖾏， 𖼎𖽑𖽻𖾏 𖽃𖽻𖾐 𖽐𖼮𖽡𖾏 𖼗𖽙𖾏， 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖽐𖼊𖽪𖾏 𖼎𖽺𖾏 𖼡𖽑𖽜𖾐 𖼺𖽘𖾏，𖼐𖽱𖾏 𖼎𖽺𖾏 𖼁𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾，𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏，𖼷𖽠𖾏 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑． 17 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏 𖼋𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼷𖽷𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼮𖽑𖽠𖾏，𖼞𖽪𖾑 𖽃𖽺𖾐 𖼙𖽔 𖼐𖽱𖾏．”
