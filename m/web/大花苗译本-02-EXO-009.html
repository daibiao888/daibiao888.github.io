9
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼁𖽺𖾏 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼇𖽔 𖼘𖽻𖾐 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽡𖾑 𖼀𖽳 𖼖𖽹 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 : 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 , 𖼻𖽻𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖾑 . 2 𖼟𖽡 𖼮𖽔𖾑 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼅𖽻𖾏 , 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖽃𖽔𖾐 𖼊𖽔𖾑 𖼷𖽷𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽡𖽝𖾏 , 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼮𖽑𖽨𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐 𖼟𖽪𖾏 𖼐𖽑𖽤𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼎𖽻𖾏 𖼟𖽡 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼟𖽡 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐, 𖼟𖽪𖾏 𖼑𖽜𖾏、𖼑𖽜𖾏 𖼖𖽔𖾑、𖼘𖽙 𖼊𖽑𖽙𖾐、𖽐𖼀𖽙𖾏 𖼑𖽨𖾏、𖽐𖼀𖽙𖾏 𖼾𖽘𖾏 𖽐𖼮𖽑𖽷𖾏 . 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼇𖽺𖾐 𖼡𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼯𖽦𖾏 𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 𖼑𖽡𖾏 𖼯𖽦𖾏 , 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼺𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑 , 𖽃𖽡𖾑 𖼋𖽪 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽔 .” 5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼊𖽡𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 , 𖼐𖽑𖽡𖾏 : “𖼀𖽡 𖼟𖽡 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼐𖽡𖾑 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 .” 6 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 , 𖼣𖽹 𖼮𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼯𖽦𖾏 𖼋𖽔 𖼖𖽔𖾑 𖼋𖽘𖾏 𖼋𖽘𖾏 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼯𖽦𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽔 . 7 𖼇𖽔 𖼘𖽻𖾐 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼑𖽔𖾏 , 𖼡𖽔 𖼋𖽱𖾏 𖼀𖽻𖾐 , 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽦𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼋𖽔 . 𖼈𖽡𖽝𖾏 𖼇𖽔 𖼘𖽻𖾐 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖼼𖽔𖾏 , 𖼦𖽡𖾐 𖼯𖽪𖾏 𖼟𖽙𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 .
𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼷𖽑𖽘𖾐
8 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼉𖽺𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼞𖽑𖽡𖽠𖾏 𖼡𖽑𖽙𖾑 𖼑𖽡 𖼋𖽺𖾏 𖽐𖼞𖽑𖽡𖽠𖾐 , 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼇𖽔 𖼘𖽻𖾐 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼨𖽑𖽢𖾏 𖼎𖽻𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 . 9 𖼋𖽡𖾏 𖽐𖼞𖽑𖽡𖽠𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼣𖽹 𖼮𖽡 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏 , 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼎𖽙 𖼯𖽦𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖽃𖽔𖾏 𖼷𖽑𖽘𖾐 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼀𖽡𖾏 𖼘𖽑𖽱𖾑.”
