𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼮𖽵𖾑 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐
𖽃𖽡𖾐 𖼸𖽙𖾐 𖽐𖼞𖽻𖾏, 𖼞𖽑𖽙𖾐 𖼖𖽔𖾑 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏, 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽷𖾑 𖼖𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼳𖽡𖾑 𖼄𖽔𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 “𖼄𖽔 𖼦𖽔𖾐 𖼖𖽔𖾏 𖼖𖽡𖾏 𖼣𖽗𖾏 𖽃𖽳” 𖼳𖽘𖾐 (“𖼄𖽔 𖼦𖽔𖾐 𖼖𖽔𖾏 𖼖𖽡𖾏 𖼣𖽗𖾏 𖽃𖽳”𖼟𖽪𖾏 “𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐 𖼑𖽡𖾏 𖼄𖽙𖾐 𖽐𖼷𖽘𖾐” )
88
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫、𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼘𖽪！
𖼞𖽪𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼐𖽡𖽝𖾑 𖽐𖼊𖽜．
2 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼷𖽑𖽜𖾑 𖼟𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜！
 
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏.
4 𖼞𖽪𖾑 𖼺𖽙𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽻𖾑 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼼𖽘𖾐,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 ( “𖼦𖽡𖾐 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖽐𖼀𖽘𖾏”).
5 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼗𖽔𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼺𖽱𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼊𖽔𖾏, 𖼀𖽱𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖽐𖼚𖽙,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼟𖽡 𖼮𖽳𖾐 𖽐𖼮𖽡𖽙𖾏,
𖽐𖼎𖽙 𖼟𖽡 𖼇𖽺𖾐 𖼡𖽔𖾏 𖼺𖽘𖾏.
6 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼡𖽑𖽙𖾑 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐 𖽐𖼚𖽙 ,
𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼊𖽙𖾐.
7 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼸𖽺𖾏 𖼐𖽑𖽤𖾑 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑,
𖼟𖽡 𖼎𖽑𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼚𖽘𖾏 𖼯𖽡𖽙𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖽃𖽦𖾐. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
8 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼀𖽻𖾐, 𖼮𖽡𖽺𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼮𖽡𖽝𖾏,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽪𖾐.
𖼞𖽪𖾑 𖼊𖽻𖾏 𖼞𖽱𖾏 𖼮𖽡𖽝𖾏, 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏.
9 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽜 𖼟𖽪𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏 𖼐𖽔𖾑 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡𖾏 𖼺𖽘𖾏.
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 , 𖼮𖽘𖾏 𖼋𖽡𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 .
10 𖼟𖽡 𖼽𖽔𖾑 𖼑𖽜 𖼊𖽪𖾐 𖼋𖽔𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑 𖼺𖽷𖾐 𖼘𖽪𖾏 ?
𖼊𖽪𖾐 𖼋𖽔𖾏 𖼑𖽡𖾏 𖽃𖽔𖾑 𖼛𖽡𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼷𖽷𖾏 𖼞𖽡 𖼑𖽡𖽻 𖼸𖽠𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖼺𖽷𖾐 𖼘𖽪𖾏 ? （𖼳𖽡𖾏 𖼖𖽔𖾑）
11 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼦𖽡𖾏 𖽐𖼷𖽘𖾏 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼟𖽡 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖼺𖽷𖾐 𖼘𖽪𖾏 ?
𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼟𖽡 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼺𖽷𖾐 𖼘𖽪𖾏 ?
12 𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑, 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼖𖽺𖾑?
𖼟𖽡 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑, 𖼋𖽡𖽝𖾏 𖼸𖽠𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖽐𖼮𖽡𖽙𖾏 𖼺𖽘𖾏 𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽻𖾐 𖼖𖽺𖾑?
 
13 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡 ,
𖼞𖽪𖾑 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼸𖽙 𖼟𖽡 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐.
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪！ 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼗𖽔𖾏 𖼞𖽪𖾑 𖼺𖽘𖾏？
𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼀𖽙 𖼟𖽡 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑？
15 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼳𖽻𖾐 𖼷𖽹𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐, 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼋𖽔 ,
𖼞𖽪𖾑 𖼊𖽻𖾏 𖼟𖽡 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏, 𖼎𖽺𖾏 𖼏𖽘𖾏 𖼡𖽑𖽙𖾐 𖼡𖽑𖽙𖾐 .
16 𖼟𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼷𖽑𖽪𖾐 𖼖𖽻𖾐 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼟𖽡 𖼮𖽡𖽠𖾏 𖼏𖽘𖾏 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼖𖽻𖾐 𖼺𖽘𖾏.
17 𖼋𖽡𖾏 𖼐𖽡𖾑 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼺𖽱𖾏 𖽃𖽻𖾐 𖼞𖽑𖽡 𖼙𖽻 𖼞𖽪𖾑 ,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽡𖾐 𖼸𖽻 𖼋𖽔𖾏 𖼟𖽻𖾏 𖼙𖽻𖾐 𖼞𖽪𖾑 .
18 𖼟𖽡 𖼞𖽑𖽡𖽠𖾏 𖼞𖽪𖾑 𖼽𖽺 𖼗𖽔 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏 𖼮𖽡𖽺𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼮𖽡𖽝𖾏 ,
𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽐𖼚𖽙 .
