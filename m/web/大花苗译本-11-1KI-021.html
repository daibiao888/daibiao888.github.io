𖼽𖽦𖾑 𖼳𖽡𖾐 𖼀𖽡𖽝 𖽐𖼊𖽻𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖼦𖽔𖾏 𖼐𖽔 𖼀𖽳 𖼖𖽪𖾑 𖼉𖽘𖾏
5 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽦𖾑 𖼳𖽡𖾐 𖼀𖽡𖽝 𖼋𖽔𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽜 𖼡𖽔𖾑 𖼳𖽷𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼮𖽷𖾑 𖼐𖽡𖽙𖾐 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼦𖽡𖾐 𖼑𖽻𖾏 𖼉𖽔𖾏 𖼘𖽪?” 6 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “𖼑𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽔 𖼀𖽳 𖼐𖽑𖽡𖾏: 𖼟𖽡 𖼄𖽔𖾐 𖼟𖽡 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼄𖽔𖾐 𖽐𖼡𖽜𖾏 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼟𖽡, 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽪𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼉𖽘𖾏 𖼘𖽑𖽪𖾑 𖼎𖽻𖾏 𖼟𖽡, 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼎𖽻𖾏 𖼟𖽡.” 7 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽤𖾐 𖼽𖽦𖾑 𖼳𖽡𖾐 𖼀𖽡𖽝 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼯𖽵𖾏 𖼞𖽳𖾑𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽪𖽙 𖼘𖽪? 𖼮𖽔𖾑 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏, 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼖𖽙𖾏 𖼳𖽦𖾑 𖼑𖽻𖾏 𖼉𖽔𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼺𖽷𖾑 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽔 𖼀𖽳 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼷𖽷𖾑 𖽃𖽔 𖼅𖽔 𖼑𖽨𖾏 𖽃𖽔 𖼅𖽔 𖼑𖽜𖾏 𖼎𖽻𖾏 𖼟𖽡.” 8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼽𖽔𖾏 𖼦𖽔𖾐 𖼑𖽡𖾏 𖽐𖼷𖽷𖾏 𖼺𖽻𖾏 𖼄𖽻𖾏, 𖼎𖽑𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽪𖾑 𖽐𖼷𖽷𖾏 𖼼𖽘𖾏 𖼸𖽺𖾏 𖽐𖼷𖽷𖾏 𖼮𖽡𖽝𖾏, 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼗𖽻 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽔 𖼀𖽳 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑. 9 𖼄𖽻𖾏 𖽐𖼚𖽙 𖼺𖽻𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽡 𖼽𖽔𖾑 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼊𖽪𖾏 𖼡𖽜𖾑, 𖼟𖽙𖾏 𖼐𖽔 𖼀𖽳 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐， 10 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼡𖽔𖾑 𖼯𖽱𖾏 𖼸𖽜 𖼋𖽔𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽔 𖼀𖽳 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐, 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼛𖽠𖾏 𖼑𖽪𖾏 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏, 𖼟𖽡 𖽐𖼮𖽱𖾏 𖼩𖽺𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽘𖾏, 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼎𖽺𖾏 𖼮𖽤𖾐 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼊𖽻𖾏 𖼋𖽔 𖼺𖽘𖾏.
