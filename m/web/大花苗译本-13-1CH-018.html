18
𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖼇𖽺𖾑 𖼖𖽡𖾏 𖼺𖽷𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼯𖽵𖾏 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳 𖽐𖼎𖽙 𖼼𖽙𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼮𖽢𖾑 𖼊𖽑𖽳 . 2 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼄𖽙𖾑 𖼽𖽔𖾏 , 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼊𖽔𖾏 𖽂𖽺𖾏 , 𖼊𖽪𖾏 𖼳𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 . 3 𖼺𖽙𖾐 𖼀𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽔𖾐 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼳𖽡𖽝𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽨𖾏 𖼇𖽔 𖼖𖽔𖾑 𖼊𖽡𖾐 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖼙𖽠𖾏 , 𖼽𖽔𖾑 𖼞𖽑𖽙𖾑 𖼏𖽔𖾏 𖼾𖽨𖾏 𖼞𖽪𖽙 𖼛𖽘𖾏 𖼼𖽙𖾏 ,𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 , 𖼸𖽙 𖼚𖽔𖾏 𖼦𖽔𖾐 𖼄𖽔𖾐， 4 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖼐𖽱𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽪𖾐, 𖼏𖽙 𖼟𖽪𖾏 𖼯𖽡𖽜𖾏 𖼑𖽜𖾏 𖼳𖽤𖾏 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐, 𖼏𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼊𖽠𖾏 𖽃𖽔𖾐 𖼉𖽻𖾏 𖼗𖽜𖾏, 𖼀𖽑𖽔𖾏 𖼑𖽜𖾏 𖼟𖽪𖾏 𖽐𖼡𖽪𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖽃𖽡𖾑 𖼞𖽡𖾏 𖼎𖽻𖾏 𖼳𖽠𖾑 𖼟𖽙𖾏 𖼊𖽪𖾏 𖼺𖽘𖾏 , 𖼈𖽡𖽝𖾏 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼀𖽔𖾏 𖼗𖽪𖾏 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼑𖽡𖾏 𖼑𖽜𖾏 𖼮𖽡𖽝𖾏 . 5 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼑𖽡𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖽐𖼀𖽘𖾏 𖼺𖽙𖾐 𖼀𖽔𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽔𖾐 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼳𖽡𖽝𖾏 , 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼊𖽔𖾏 𖼺𖽘𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾐 𖼉𖽻𖾏 𖽃𖽔𖾐 𖼷𖽑𖽡𖽝𖾐 . 6 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼑𖽡𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 , 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 , 𖼊𖽪𖾏 𖼳𖽳𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 . 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼊𖽻𖾏 𖼳𖽠𖾏 . 7 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼦𖽔𖾐 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼳𖽡𖽝𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖼛𖽔𖾏 𖽐𖼎𖽳𖾏 𖼺𖽪𖾐 𖼈𖽙𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼞𖽪𖾐 𖽃𖽔𖾏 , 𖼯𖽡𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 . 8 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼊𖽑𖽡 𖼀𖽔𖾑 ( “𖼊𖽑𖽡 𖼀𖽔𖾑 ” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 “𖼀𖽡𖾐 𖼊𖽑𖽔𖾑 ” ) 𖽐𖼎𖽙 𖼮𖽱𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼦𖽔𖾐 𖼊𖽔𖾏 𖼊𖽡𖾐 𖼳𖽡𖽝𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖼙𖽻 𖼐𖽡𖾑 , 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼋𖽻𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐 , 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼗𖽙 𖼺𖽙𖾐 𖼘𖽙 𖼄𖽞 𖼯𖽡𖽙𖾏 𖼋𖽡𖾏 𖼋𖽻𖾏 𖼐𖽡𖾑 𖼷𖽷𖾏 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏、 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼋𖽻𖾏 、 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼋𖽻𖾏 .
