𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼡𖽑𖽔𖾐 𖼨𖽑𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐
8
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼦𖽡𖾐 𖼢𖽔𖾏 𖼨𖽑𖽪𖾏?
𖼀𖽻𖾐 𖼊𖽘𖾐 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼦𖽡𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼘𖽪?
2 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪 𖼞𖽡𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼺𖽡𖽝𖾐,
𖼳𖽠𖾑 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼳𖽔𖾏,
3 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻 𖽃𖽔𖾑 𖽐𖼊𖽪, 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼙𖽻,
𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼙𖽻, 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼐𖽑𖽡𖾏 :
4 “𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪！ 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖼅𖽡,
𖼞𖽪𖾑 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼳𖽘𖾐.
5 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼘𖽪！ 𖼅𖽡 𖼽𖽔𖾑 𖼚𖽠𖾐 𖽐𖼮𖽑𖽦𖾐；
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼘𖽪！ 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼋𖽡𖾏 𖼞𖽡𖾑 𖼀𖽻𖾐 𖽐𖼎𖽑𖽹𖾏.
6 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 , 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼡𖽜𖾏,
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
7 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼋𖽠𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏,
𖼞𖽪𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
8 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼖𖽡𖾐 𖼉𖽳𖾐 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏.
9 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼑𖽔𖾏 𖼞𖽪𖾑 𖼗𖽪𖾏， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼾𖽙 𖼯𖽡𖽙𖾏,
𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 .
10 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏, 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐,
𖼯𖽪𖾏 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳, 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏.
 
11 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 (“ 𖼖𖽔𖾑 𖼘𖽑𖽠𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑” 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖽃𖽔𖾑 𖼈𖽳𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽡𖽝𖾐”).
𖼋𖽡𖾏 𖼟𖽪𖾏 𖼟𖽡 𖽃𖽺𖾐 𖼙𖽔， 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼋𖽳𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 .
12 𖼞𖽪𖾑 — 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼚𖽠𖾐 𖽐𖼮𖽑𖽦𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐,
𖼎𖽺𖾏 𖽐𖼎𖽑𖽦𖾑 𖼊𖽻𖾏 𖼀𖽻𖾐 𖼟𖽳 𖽐𖼎𖽙 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑.
13 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏；
𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏、𖼙𖽙𖾏 𖽐𖼊𖽪𖾏 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽃𖽔𖾏、 𖼞𖽡𖾑 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 、 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼳𖽪𖾐.
14 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼞𖽑𖽻𖾏 𖼞𖽑𖽻𖾑 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼟𖽳 𖼟𖽪𖾏 𖼋𖽡𖽝 ,
𖼞𖽪𖾑 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐, 𖼞𖽪𖾑 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.
15 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼞𖽳𖾑 𖼯𖽵𖾏 𖼞𖽪𖽙 ,
𖼉𖽻 𖼷𖽻𖾏 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼊𖽡𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳.
16 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼅𖽔𖾐 𖼊𖽪𖾐 𖼋𖽠𖾏 𖽐𖼷𖽷𖾏, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼨𖽑𖽪𖾑 𖼁𖽜 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.
17 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 ,
𖼷𖽷𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙 𖼖𖽠𖾑 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼁𖽙𖾏 𖼞𖽪𖾑 .
18 𖼅𖽔𖾏 𖼊𖽻𖾏 𖽐𖼎𖽙 𖽐𖼚𖽻 𖼮𖽡𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 ,
𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾐 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 .
19 𖽃𖽔𖾑 𖼖𖽪𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼷𖽷𖾏 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾐, 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼻𖽻𖾏,
𖼞𖽪𖾑 𖼡𖽻𖾐 𖼖𖽻𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼡𖽜𖾏 𖼊𖽔𖾑.
20 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 ,
𖼅𖽻𖾏 𖼎𖽻𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 ,
21 𖼟𖽙𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼊𖽻𖾏 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑,
𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼼𖽪 .
𖼦𖽡𖾐 𖼑𖽡𖽝 𖼅𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼅𖽔𖾏 𖼀𖽻𖾐 𖼟𖽳 𖼋𖽘𖾏
22 𖼐𖽡𖽙𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼳𖽠𖾑 𖼁𖽱𖾏 𖼷𖽷𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 ,
𖼐𖽡𖽙𖾐 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼺𖽪𖾐 𖼁𖽱 𖼷𖽷𖾏 𖼷𖽑𖽔𖾏 𖼮𖽘𖾑 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏 𖼅𖽔𖾏 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏,
23 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼗𖽙𖾏, 𖼎𖽑𖽻𖾏 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼁𖽱𖾏,
𖼦𖽡𖾐 𖼑𖽡𖽝 𖼅𖽔𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼮𖽡𖽝𖾏 𖼋𖽘𖾏；
24 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖼊𖽙𖾐,
𖼦𖽡𖾐 𖼅𖽔𖾏 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼞𖽙𖾏, 𖼅𖽔𖾏 𖼞𖽪𖾑 𖼋𖽘 .
25 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼊𖽻𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏,
