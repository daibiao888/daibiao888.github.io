𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 𖼊𖽡𖾐 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖽃𖽔𖾐 𖼳𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼖𖽪𖾑 𖽃𖽔𖾐 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 , 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 , 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼮𖽑𖽡𖽠𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼚𖽙 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖼐𖽑𖽡𖾏 : 2 “𖼅𖽡 𖼽𖽔𖾑 𖼮𖽔𖾑 𖼾𖽱 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱、𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏、𖼄𖽑𖽻𖾐 𖼳𖽡𖾏、𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽷𖾏 𖼑𖽡𖾏 𖼮𖽡𖽻𖾐 𖼯𖽠 , 𖼮𖽔𖾑 𖼾𖽱 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾑 𖼾𖽠 . 3 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖼎𖽑𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽡𖽻𖾏 𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏 𖼋𖽻𖾏 𖽃𖽡𖾑, 𖼟𖽡 𖽐𖼎𖽙 𖼽𖽔𖾏 𖼖𖽪𖽝 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽔𖾑 𖼸𖽻𖾏 𖼏𖽙 𖼮𖽔𖾑 𖼾𖽱 . 4 𖼷𖽑𖽔𖾏 𖼳𖽡𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔𖾏 𖼾𖽨𖾏 𖼄𖽑𖽻𖾐 𖼳𖽡𖾏 𖼐𖽡𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 , 𖽐𖼀𖽘𖾏 𖼅𖽡 . 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼷𖽷𖾏 𖼟𖽪𖾏 :
𖼖𖽨 𖼀𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼳𖽷𖾏 𖼊𖽨𖾑 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽱 ；
6 𖼳𖽡𖾑 𖼄𖽡𖽝𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼺𖽪𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼊𖽹𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼳𖽷𖾏 𖼖𖽪𖾏 𖼄𖽡𖽝 ；
7 𖼽𖽨 𖼊𖽔𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼐𖽔 𖼊𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽔 𖼳𖽪𖽝𖾏 ；
8 𖼽𖽡𖾑 𖼺𖽔𖾐 𖼮𖽢𖾑 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼺𖽪𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽔 𖼊𖽑𖽗𖾐 𖼐𖽡𖽝 ；
9 𖼳𖽡𖾑 𖼀𖽪𖾏 𖼖𖽪𖽝 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼳𖽡𖾑 𖼖𖽪𖽝 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔 ；
10 𖼽𖽙 𖼺𖽳 𖼊𖽪𖾐 𖼞𖽡𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼦𖽪 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼄𖽔𖾐 ； 𖼄𖽔𖾐 𖼐𖽔 𖼳𖽡𖾑 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼀𖽡𖾐 𖼊𖽔𖾏 𖼳𖽱 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼄𖽔𖾐 𖼖𖽡𖽝 ；
11 𖼀𖽡𖽝 𖼽𖽔𖾐 𖼄𖽡𖾐 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼮𖽡𖾑 𖼊𖽙𖾑 𖼐𖽡 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽡𖾐 𖼊𖽗𖾏 ；
12 𖼊𖽗𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼳𖽔𖾑 𖼊𖽹𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼽𖽡𖾑 𖼳𖽡𖽝𖾏 ；
13 𖼽𖽔𖾏 𖼳𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖽃𖽳 𖼖𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼀𖽑𖽔 𖼮𖽡𖽝 ；
14 𖼮𖽢𖾑 𖼊𖽳 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼊𖽨𖾑 𖽃𖽳𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼺𖽔𖾐 ；
15 𖼐𖽔 𖼇𖽪 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼳𖽡𖾏 𖽃𖽡𖾑 , 𖼅𖽔𖾏 𖼽𖽡𖾑 𖼐𖽔 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼳𖽡𖾑 𖼖𖽔𖾑 .
16 𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼺𖽹𖾑 𖼗𖽙𖾏 , 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖽃𖽔𖾑 𖼾𖽠𖾏 𖽃𖽔𖾐 𖼀𖽑𖽱𖾐 𖼳𖽡𖾏 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 , 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 .”
17 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼄𖽙𖾑 𖼳𖽡𖾑 、𖼽𖽔𖾏 𖼖𖽪𖽝 𖼨𖽑𖽳𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼊𖽔𖾑 𖽐𖼷𖽷𖾏 𖼺𖽹𖾑 𖼮𖽡𖽝𖾏 𖼐𖽡𖾑 , 18 𖼸𖽙 𖼖𖽪𖾑 𖽃𖽔𖾐 𖼖𖽑𖽡𖾏 𖼺𖽡𖽝𖾐 𖽃𖽡𖾐 𖼨𖽑𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼦𖽡 𖼸𖽱 . 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼸𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼞𖽔𖾏 𖼾𖽡𖾏、𖼄𖽑𖽻𖾐 𖼳𖽡𖾏、𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼷𖽷𖾏 𖼑𖽡𖾏 𖼮𖽡𖽻𖾐 𖼯𖽠 , 𖼎𖽑𖽻𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼳𖽡𖽻𖾏 𖽐𖼮𖽡𖾏 𖽃𖽔𖾑 𖼳𖽔𖾐 , 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼡𖽔 𖼗𖽜𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽻𖾏 𖽐𖼮𖽡𖽙𖾏 𖼮𖽡𖽝𖾏 . 19 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽡𖽙𖾐 𖼳𖽡𖾑 𖼐𖽺𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏 𖼮𖽔𖾑 𖼺𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 .
