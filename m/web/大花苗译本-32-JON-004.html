𖼊𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼽𖽙 𖼐𖽔 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼦𖽡𖾐 𖼙𖽔 , 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 . 2 𖼎𖽺𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “ 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼞𖽪𖽙 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑? 𖼞𖽪𖾑 𖼀𖽻𖾐 𖼟𖽡 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼅𖽳𖾏, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽑𖽪𖾐, 𖼦𖽡𖾐 𖼖𖽡 𖼟𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐, 𖼅𖽔𖾏 𖼘𖽑𖽪𖾑 𖽐𖼮𖽑𖽠𖾏 𖼟𖽪𖾏 𖼀𖽪𖾑 𖼷𖽻𖾏, 𖼎𖽺𖾏 𖼎𖽙𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼟𖽪𖾏 𖼙𖽔 𖼐𖽑𖽡𖾏 𖽃𖽡𖾑. 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼊𖽑𖽔𖾑 𖼳𖽷𖾑 𖼙𖽠𖾏. 3 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼳𖽻𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼙𖽠𖾏 ! 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼋𖽔 𖼙𖽠𖾏 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐.” 4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼞𖽡𖾑 𖼘𖽪?”
5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼽𖽙 𖼐𖽔 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼙𖽻 , 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼙𖽻 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐. 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖼑𖽜 𖼾𖽨𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽐𖼞𖽔𖾏 𖼮𖽱𖾐 𖼖𖽱𖾐 , 𖼼𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼚𖽺𖾏 𖽃𖽔𖾏 𖽐𖼷𖽻𖾏, 𖼽𖽔𖾑 𖼑𖽔𖾏 𖼋𖽡𖽝𖾏 𖼖𖽪𖾑 𖼙𖽻 𖽃𖽡𖾑 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏． 6 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖽃𖽡𖾐 𖼇𖽘𖾐 𖽐𖼊𖽻𖾏 𖼷𖽷𖾏 𖼞𖽑𖽡𖽝𖾑 𖼑𖽨𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼘𖽘𖾑𖽑 𖼺𖽡𖽝𖾐 𖼮𖽑𖽡𖽻𖾏 𖼽𖽙 𖼐𖽔 , 𖽃𖽔𖾑 𖽐𖼷𖽻𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼐𖽱𖾏 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏, 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼮𖽑𖽘𖾑 𖼻𖽡𖽝𖾐 𖼑𖽡𖾏 𖽃𖽦𖾐 𖽐𖼀𖽜𖾏. 𖼽𖽙 𖼐𖽔 𖼑𖽜 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾏 𖼞𖽑𖽡𖽝𖾑 𖼑𖽨𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝. 7 𖼐𖽑𖽪𖾐 𖽃𖽔𖾐 𖼡𖽜𖾐 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼀𖽡𖾑 𖼞𖽘𖾐 𖼮𖽷𖾐 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾏 𖼞𖽑𖽡𖽝𖾑 𖼑𖽨𖾏, 𖼟𖽙𖾏 𖼡𖽑𖽔𖾑 𖼺𖽘𖾏. 8 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼋𖽠𖾏 𖼋𖽔𖾏 𖽃𖽺𖾐 𖼐𖽡𖽙𖾐 𖽃𖽺𖾐 𖼳𖽙𖾑 𖽃𖽡𖾑, 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽙𖾏 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 𖼮𖽢𖾏 𖼷𖽑𖽔𖾐 𖼳𖽙𖾑 𖼊𖽔𖾐 𖼋𖽡𖽝, 𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼻𖽡𖽝𖾐 𖼽𖽙 𖼐𖽔 𖼖𖽪𖾑 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 , 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼑𖽜 𖼾𖽨𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼋𖽔 , 𖼐𖽑𖽡𖾏: “ 𖼞𖽪𖾑 𖼋𖽔 𖼺𖽘𖾏 𖼖𖽔𖾑 𖼻𖽻𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏 .” 9 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼽𖽙 𖼐𖽔 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼑𖽜 𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾏 𖼞𖽑𖽡𖽝𖾑 𖼑𖽨𖾏 𖼐𖽡𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼞𖽡𖾑 𖼘𖽪?” 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖼸𖽙 𖼮𖽑𖽡𖽠𖾏 𖼋𖽔 .” 10 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼇𖽘𖾑 𖽐𖼊𖽻𖾏 𖼷𖽷𖾏 𖼞𖽑𖽡𖽝𖾑 𖼑𖽨𖾏 𖼐𖽡𖾑 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼟𖽡 𖼯𖽡𖽙 , 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽡 𖼮𖽑𖽪𖾑 𖼨𖽑𖽢𖾏, 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼊𖽔𖾐 𖼐𖽱𖾏 𖼘𖽘𖾑𖽑 𖼋𖽠𖾏 𖼋𖽔𖾏, 𖽃𖽡𖾐 𖼄𖽑𖽙𖾏 𖼊𖽔𖾐 𖼐𖽱𖾏 𖼡𖽑𖽔𖾑 𖼋𖽔 𖼺𖽘𖾏, 𖼟𖽡 𖼖𖽔𖾏 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼐𖽱𖾏, 11 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽡 𖼐𖽡 𖽂𖽺𖾏 𖼖𖽪𖾑 𖼙𖽻 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖾑 , 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼉𖽻𖾏 𖼮𖽡𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼡𖽔 𖼋𖽱𖾏 𖼇𖽺𖾐 𖽐𖼎𖽑𖽹𖾏 𖼻𖽻𖾏 𖽐𖼎𖽙 𖼮𖽷𖾑 , 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐, 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼮𖽑𖽔𖾏 𖼮𖽑𖽔𖾏 𖼘𖽪?”
