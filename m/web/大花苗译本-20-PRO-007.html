7
𖼞𖽪𖾑 𖼊𖽪𖾐, 𖼟𖽡 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼗𖽪𖾏 ,
𖼄𖽔𖾐 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖽐𖼮𖽡𖽙𖾏 𖼎𖽻𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐.
2 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼷𖽹𖾑 𖼟𖽡 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼯𖽡𖽝𖾏 ；
𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏， 𖼺𖽱𖾏 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽡𖾏 𖼖𖽔𖾑 𖼤𖽔𖾏 𖼤𖽔𖾏 𖽃𖽔 𖼅𖽔 .
3 𖼡𖽑𖽹𖾐 𖼎𖽻𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖽐𖼷𖽷𖾑 𖼋𖽡𖾏,
𖼞𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐.
4 𖽐𖼎𖽙 𖼀𖽻𖾐 𖼊𖽘𖾐 𖼀𖽻𖾐 𖼟𖽳 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼞𖽱𖾑 𖼻𖽷𖾑 .”
𖼨𖽑𖽪𖾏 𖼀𖽻𖾐 𖼊𖽘𖾐 𖽃𖽔𖾏 𖼑𖽜𖾏 𖼮𖽘𖾏 .
5 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼟𖽡 𖼀𖽜𖾑 𖼛𖽡𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏,
𖼀𖽜𖾑 𖼟𖽙𖾏 𖼚𖽡𖾐 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼑𖽡𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑 .
 
6 𖼞𖽪𖾑 𖼙𖽔 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽐𖼞𖽔𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏 ,
𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑 𖼁𖽺𖾏 𖼑𖽔𖾏 𖼑𖽡𖾏 𖼋𖽡𖾏 𖽃𖽔𖾏 𖼯𖽱𖾏 𖼡𖽑𖽙𖾑 𖼑𖽔𖾏 𖼎𖽻𖾏 𖼻𖽻𖾑 .
7 𖼁𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐, 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼺𖽱𖾑 𖼑𖽱𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼏𖽪𖾐 𖽃𖽔 𖼏𖽔 .
8 𖽐𖼮𖽡𖽻𖾏 𖼖𖽪𖾑 𖼟𖽡𖾏, 𖼅𖽻𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽃𖽔𖾑 𖼮𖽡𖽺𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼀𖽜𖾏 𖼚𖽔𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏,
𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼐𖽱𖾏 𖽐𖼞𖽔𖾏.
9 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏,
𖼟𖽪𖾏 𖼋𖽘 𖼄𖽑𖽙𖾏, 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
 
10 𖼑𖽔𖾏 𖼘𖽪！ 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼐𖽱𖾏 ,
𖽐𖼊𖽱𖾏 𖼼𖽘𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏, 𖼅𖽔𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏.
11 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 , 𖼦𖽡𖾐 𖼾𖽨𖾏 𖼯𖽵𖾏 𖼾𖽨𖾏,
𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼷𖽷𖾐.
12 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼟𖽡𖾏, 𖼅𖽔𖾏 𖼯𖽡𖽹𖾏 𖼑𖽡𖽻𖾏 𖼐𖽡𖽙𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼇𖽘𖾑,
𖼡𖽑𖽙𖾑 𖼯𖽢 𖼼𖽻𖾏 𖼯𖽠𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾑 𖼮𖽡𖽺𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼀𖽜𖾏.
13 𖼐𖽱𖾏 𖼊𖽔𖾑 𖼷𖽷𖾑 𖽃𖽔𖾏 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖽃𖽡𖾑,
𖽐𖼎𖽙 𖼐𖽱𖾏 𖼡𖽑𖽻𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏, 𖼟𖽡𖾏 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼚𖽑𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 :
14 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼖𖽑𖽘𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑 ,
𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼞𖽪𖾑 𖼡𖽑𖽔𖾐 𖼑𖽡𖽻 𖼊𖽻𖾏 𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼏𖽵 .
15 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼷𖽹𖾏 𖼟𖽡 ,
𖽃𖽺𖾐 𖽐𖼮𖽑𖽠𖾏 𖼁𖽙𖾏 𖼟𖽡 𖼚𖽑𖽪𖾏, 𖼻𖽻𖾏 𖼞𖽡𖾑 𖼙𖽻 𖽐𖼮𖽷𖾐 𖼟𖽡 𖼋𖽘𖾏.
16 𖼞𖽪𖾑 𖼞𖽑𖽡𖽠𖾏 𖼀𖽑𖽙𖾑 𖼮𖽜𖾐 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼁𖽘𖾏,
𖽐𖼎𖽙 𖼣𖽹 𖼮𖽡 𖽐𖼊𖽻𖾐 𖼅𖽘𖾐 𖽐𖼚𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼁𖽘𖾏 𖼀𖽔𖾏 𖼞𖽪𖾑 𖼸𖽱𖾏 𖼸𖽘𖾏.
17 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼞𖽑𖽡𖽠𖾏 𖼄𖽪 𖼽𖽙、 𖼮𖽑𖽹 𖼳𖽤𖾏、 𖼀𖽡𖾏 𖼚𖽻𖾏 𖼞𖽪𖽺𖾏 𖽐𖼊𖽻𖾏,
𖼚𖽜𖾑 𖼞𖽪𖾑 𖼸𖽱𖾏 𖼸𖽘𖾏.
18 𖼟𖽡 𖼋𖽔𖾏 𖼀𖽡𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖼸𖽙 𖼚𖽔𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑,
𖼀𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼮𖽑𖽠𖾏 𖼙𖽔 𖼗𖽙𖾏 𖼳𖽦𖾑.
19 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼈𖽻𖾑 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏,
𖼐𖽱𖾏 𖼋𖽠𖾏 𖽐𖼎𖽘𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐 𖼙𖽠𖾏.
20 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼖𖽪𖾑 𖼐𖽑𖽘𖾐 𖼑𖽡𖽝𖾏 𖼙𖽠𖾏,
𖼸𖽙 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼖𖽑𖽡𖾏 𖽐𖼎𖽘𖾏 𖼊𖽔𖾑 𖼙𖽡𖽝𖾏 𖼗𖽙𖾏 𖼚𖽔𖾏 𖽐𖼞𖽔𖾏.
 
21 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼯𖽡𖽙𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖽐𖼡𖽳𖾏 𖽃𖽳𖾑 𖼟𖽡𖾏 𖽃𖽔𖾏 𖼳𖽡𖽝𖾏 𖼐𖽘𖾑 𖼮𖽑𖽵𖾑 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼸𖽙𖾏 𖽐𖼎𖽙𖾏,
𖼎𖽑𖽻𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼟𖽪𖾏 𖼡𖽘𖾐 𖼄𖽪𖾑 𖼡𖽘𖾐 𖽐𖼮𖽳𖾑 𖽃𖽔𖾏 𖼊𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼅𖽻𖾏.
22 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼞𖽡𖾏 𖽐𖼮𖽑𖽷𖾏 𖼸𖽙𖾏 𖼐𖽱𖾏 , 𖼺𖽱𖾏 𖼑𖽨𖾏 𖼮𖽤𖾐 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖼊𖽔𖾏,
𖼎𖽺𖾏 𖼺𖽱𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽔 𖼏𖽔 𖼐𖽑𖽘𖾑 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼖𖽑𖽻𖾏 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐,
23 𖼋𖽙𖾏 𖼸𖽙 𖼈𖽙𖾏 𖼮𖽑𖽡𖽙𖾐 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐, 𖼺𖽱𖾏 𖼊𖽪𖾑 𖼐𖽻 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼟𖽪𖾏 𖼇𖽺𖾏,
𖼈𖽡𖽝𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼽𖽔𖾑 𖼛𖽙𖾏 𖼳𖽦𖾑.
 
24 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼘𖽪！ 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼑𖽻 𖼸𖽙𖾏 𖼞𖽪𖾑 ,
𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼑𖽻 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼋𖽠𖾏 𖼞𖽪𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏.
25 𖼟𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼖𖽡𖾐 𖼈𖽳𖾐 𖼎𖽻𖾏 𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼑𖽡𖾏 𖼸𖽙𖾏 𖼞𖽡𖾑 ,
