5 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼐𖽱𖾏 𖼸𖽙𖾏 𖼖𖽑𖽔𖾏 𖼳𖽦𖾏,

6 𖼊𖽳𖾑 𖼘𖽳𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼾𖽘𖾏 𖼁𖽜𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐,

𖼊𖽔𖾑 𖼣𖽔𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 𖽐𖼎𖽙 𖼮𖽙𖾑 𖼖𖽡𖽝𖾐 𖼟𖽪𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼳𖽷𖾐 𖽃𖽺𖾐 𖼋𖽻𖾏 𖽐𖼎𖽙 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼛𖽙 𖽃𖽔𖾏 𖽃𖽡𖾐 𖽐𖼀𖽙𖾐,

7 𖼊𖽪𖾑 𖽃𖽔 𖼑𖽡𖽝 𖼑𖽨𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼊𖽪𖾑 𖽃𖽔 𖼑𖽡𖽝 𖼚𖽹𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙 𖼑𖽻𖾏 𖽐𖼡𖽜𖾐,

8 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼑𖽻𖾏 𖼄𖽡𖾏 𖼽𖽔𖾑 𖽃𖽔𖾏 𖼇𖽜𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽡𖾏 𖽃𖽔 𖼛𖽻𖾏 𖼡𖽑𖽙𖾑,
𖼣𖽔𖾑 𖼽𖽻𖾏 𖼟𖽪𖾏 𖼇𖽳𖾑 𖼄𖽡𖾏 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼐𖽘𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼋𖽙𖾏 𖼑𖽡𖾏 𖼖𖽪𖾑 𖼡𖽑𖽙𖾑.
9 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼐𖽡𖾑 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑,
