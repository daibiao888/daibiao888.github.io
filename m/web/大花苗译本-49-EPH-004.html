“𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 ， 𖼦𖽔𖾏 𖼊𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼊𖽻𖾏 𖼦𖽔𖾏 𖼙𖽠𖾏 𖽃𖽡𖾑，

9 “ 𖼐𖽱𖾏 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼡𖽻𖾏 𖽐𖼊𖽪𖾏 ” 𖼗𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼖𖽺𖾑？𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖼙𖽔 𖼺𖽪𖾐 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼊𖽙𖾐 𖼘𖽪？ 10 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏 𖽃𖽡𖾑， 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖽐𖼮𖽡𖾏 𖼮𖽑𖽡𖽻𖾏 𖼦𖽡𖾏 𖼇𖽔𖾑 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽐𖼊𖽪𖾏， 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼅𖽔𖾏．
