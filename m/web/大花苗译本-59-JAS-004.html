𖼅𖽡 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏 𖼖𖽺𖾑？ 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏 𖼋𖽔𖾏 𖼖𖽺𖾑？ 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼅𖽡 𖽃𖽔𖾏 𖼮𖽡𖾑 𖽐𖼀𖽜𖾏 𖼺𖽡𖽝𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏 𖽃𖽡𖾑 𖼋𖽔𖾏 𖼘𖽪？ 2 𖼅𖽡 𖼦𖽠𖾑 𖼉𖽳， 𖼈𖽡𖽝𖾏 𖼊𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏． 𖼅𖽡 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼅𖽡 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 ，𖼖𖽔𖾏 𖼊𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏， 𖼅𖽡 𖼎𖽺𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼦𖽡𖾏 𖽐𖼊𖽻𖾏． 𖼅𖽡 𖼊𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼊𖽜 𖼐𖽔𖾑； 3 𖼅𖽡 𖽐𖼊𖽜 𖼖𖽔𖾏 𖼊𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼖𖽡𖾏 𖼅𖽪𖾏 𖽐𖼊𖽜 ， 𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖽐𖼊𖽜 𖼗𖽙𖾏 𖽐𖼞𖽡𖾏 𖼎𖽻𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼇𖽜𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼊𖽔𖾐． 4 𖼅𖽡 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼚𖽡𖾑 𖽐𖼀𖽔𖾏 （𖼳𖽡𖾑 𖼖𖽔 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼟𖽪𖾏 𖽃𖽔𖾏“𖽃𖽔 𖼁𖽙𖾏 𖼚𖽡𖾏 𖽐𖼀𖽔𖾏 𖼘𖽪 ”）𖼐𖽡𖾑 𖼘𖽪！ 𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔， 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼘𖽪？ 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔， 𖼟𖽪𖾏 𖽐𖼎𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑． 5 𖽐𖼊𖽪𖾏 𖽐𖼊𖽠𖾑 𖼐𖽑𖽡𖾏：“ 𖼖𖽡𖽾 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼀𖽡𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖽐𖼚𖽙，𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽑𖽠𖾏 𖼸𖽙 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼳𖽪𖾐 ”， 𖼅𖽡 𖽐𖼊𖽱𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽐𖼮𖽢𖾏 𖼼𖽙 𖼘𖽪𖾏？ 6 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼅𖽳𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼄𖽔𖾐 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽡𖽻𖾐，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼗𖽪𖾏 𖼐𖽑𖽡𖾏：
“ 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏，
𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼳𖽻𖾐 𖼢𖽜 ．”
7 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼅𖽡 𖼽𖽔𖾑 𖼑𖽻 𖼸𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖼽𖽔𖾑 𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼚𖽘𖾐， 𖼊𖽪𖾑 𖼀𖽡𖾐 𖼚𖽘𖾐 𖼎𖽺𖾏 𖼇𖽳𖾑 𖼅𖽡 𖼀𖽜𖾑 𖼙𖽠𖾏． 8 𖼅𖽡 𖼽𖽔𖾑 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏， 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼅𖽡． 𖼅𖽡 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼘𖽪！ 𖼅𖽡 𖼷𖽑𖽹𖾏 𖼋𖽡𖾏 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏； 𖼅𖽡 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼘𖽪！ 𖼅𖽡 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏． 9 𖼅𖽡 𖼽𖽔𖾑 𖼮𖽑𖽡𖾏 𖼊𖽳𖾏、 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐、 𖼐𖽡𖽝𖾑 𖼼𖽷𖾏； 𖼞𖽑𖽡𖽠𖾏 𖼅𖽡 𖼙𖽔 𖼚𖽙𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾏 𖼺𖽡𖽝𖾐，𖼞𖽑𖽡𖽠𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼮𖽑𖽡𖾏 𖽐𖼷𖽑𖽔𖾏． 10 𖼅𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼾𖽨𖾏 𖼳𖽻𖾐 𖼢𖽜 𖼾𖽨𖾏，𖼮𖽵𖾑 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼅𖽡 𖼳𖽠𖾑 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐．
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽠𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏
11 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼅𖽡， 𖼅𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽠𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾏 ． 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽠𖾏 𖼞𖽱𖾑 𖼋𖽡𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼨𖽑𖽢𖾐 𖼞𖽱𖾑 𖼋𖽡𖾏， 𖼺𖽙𖾏 𖼟𖽪𖾏 𖼮𖽑𖽡𖽠𖾏 𖼊𖽑𖽠𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼎𖽙 𖼨𖽑𖽢𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 ； 𖼟𖽡 𖼮𖽔𖾑 𖼨𖽑𖽢𖾐 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 ， 𖼟𖽡 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼻𖽙𖾑 𖼸𖽙𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑， 𖼟𖽪𖾏 𖼊𖽪𖾐 𖼨𖽑𖽢𖾐 𖼾𖽔𖾏 𖼋𖽘𖾏． 12 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼷𖽑𖽙𖾏 𖼷𖽱𖾏 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖽐𖼎𖽙 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼨𖽑𖽢𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐， 𖼮𖽔 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽔𖾐， 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼸𖽠𖾏 𖼮𖽨𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 ， 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼖𖽻𖾐 𖼺𖽘𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽡𖾑． 𖼟𖽡 𖼟𖽪𖾏 𖼡𖽔 𖼋𖽱𖾏，𖼟𖽡 𖼞𖽔𖾐 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼟𖽡 𖼼𖽷𖾏 𖼼𖽙 ！
13 𖽃𖽺𖾐！ 𖼅𖽡 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼄𖽑𖽙𖾏 𖼐𖽔𖾏 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼀𖽡 𖼟𖽡， 𖼀𖽡𖾐 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼙𖽻， 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽃𖽡𖾐 𖼳𖽡𖽻𖾏 ，𖽃𖽔𖾏 𖼉𖽻 𖼙𖽔𖾏 𖽐𖼎𖽑𖽦𖾑 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏．” 14 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼀𖽡 𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼅𖽡 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼀𖽻𖾐． 𖼅𖽡 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？ 𖼅𖽡 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼚𖽙𖾏 𖼦𖽘𖾑 𖼀𖽙𖾏 𖼊𖽔𖾐 ， 𖼁𖽙𖾏 𖽃𖽡𖾐 𖽐𖼀𖽠𖾏 𖼣𖽔𖾏 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖼋𖽘𖾏． 15 𖼅𖽡 𖼦𖽡𖾐 𖼋𖽳𖾏 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏：“ 𖼮𖽵𖾑 𖼮𖽔𖾑 𖼯𖽪𖾏，𖼀𖽡𖾐 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼯𖽡𖽝𖾏 𖼮𖽡𖽝𖾏，𖼎𖽺𖾏 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼡𖽑𖽙𖾑 𖼯𖽢 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼖𖽡 𖼉𖽺𖾏．” 16 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽔𖾑 𖼐𖽡𖾑 𖼅𖽡 𖼙𖽙𖾏 𖽐𖼊𖽪𖾏 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏， 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼾𖽨𖾏 𖼷𖽻𖾑 𖼾𖽨𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳． 17 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼊𖽜𖾑 𖼐𖽜𖾐 𖼮𖽔𖾑 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼋𖽡𖾏 𖼞𖽡𖾑 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽪𖾏， 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽻𖾏 𖽃𖽔𖾏 ，𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼾𖽔𖾏 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏．
