𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼷𖽑𖽜𖾑 𖼟𖽡 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼀𖽡𖾏 𖽐𖼎𖽑𖽦𖾏 𖼑𖽻 𖼞𖽪𖾑 𖽐𖼊𖽜 𖼽𖽔𖾑 ,
𖼦𖽡𖾐 𖼽𖽔𖾑 𖽐𖼚𖽺𖾏 𖼙𖽠𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏！
2 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏, 𖼑𖽻 𖼊𖽡𖾐 𖼞𖽪𖾑．
𖼞𖽪𖾑 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏 𖼦𖽡𖾐 𖼻𖽻𖾏 𖼐𖽡𖽙𖾐, 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼟𖽡𖾏 𖽐𖼷𖽘𖾏 𖼙𖽠𖾏，
3 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼑𖽡𖾏 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏, 𖼊𖽪𖾐 𖽐𖼚𖽪𖾐 𖼑𖽡𖾏 𖼸𖽺𖾏 𖼯𖽵𖾏 ,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼄𖽔𖾐 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼊𖽑𖽡𖽝𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 , 𖼎𖽑𖽻𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼟𖽡𖾏 𖼦𖽺𖾏 𖼞𖽪𖾑 .
 
4 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖽐𖼚𖽙 𖽃𖽺𖾐 𖼄𖽙𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝 ,
𖼋𖽔 𖼑𖽡𖾏 𖼊𖽳𖾏 𖽐𖼮𖽑𖽹𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 .
5 𖼊𖽳𖾏 𖽐𖼮𖽑𖽹𖾏 𖼊𖽑𖽡𖽝𖾐 𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑 ,
𖼏𖽘𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼀𖽙 𖼈𖽳𖾑 𖼞𖽪𖾑 𖼮𖽡𖽝𖾏.
6 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼅𖽔𖾏 𖽃𖽔𖾑 𖼊𖽡𖾏 𖼺𖽱𖾏 𖼊𖽔𖾑 𖽐𖼡𖽔𖾐,
𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽘𖾏 𖼅𖽻𖾏 𖼊𖽻𖾏 𖼗𖽙𖾏 𖼳𖽦𖾑 𖼋𖽡𖽝𖾏 𖼋𖽪 .
7 𖼑𖽔𖾏 𖼘𖽪！ 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼚𖽡𖾐,
𖼀𖽱𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼯𖽢𖾏. （𖼳𖽡𖾏 𖼖𖽔𖾑）
8 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖼀𖽜𖾑 𖼚𖽔𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏,
𖼀𖽜𖾑 𖼛𖽡𖾏 𖼑𖽘 𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.”
 
9 𖼮𖽵𖾑 𖼘𖽪 ！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖽐𖼡𖽙𖾏 𖼖𖽻𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼗𖽪𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼙𖽻 𖼁𖽙𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏.
10 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼋𖽜𖾏 𖼙𖽻 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏 𖼞𖽑𖽡 𖼙𖽻 𖼅𖽻𖾏 𖼗𖽙𖾏,
𖼐𖽡𖽙𖾐 𖽐𖼚𖽙 𖼙𖽻 𖼅𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼟𖽡𖾏 𖼦𖽺𖾏.
11 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼷𖽑𖽪𖾐 𖼖𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐,
𖼸𖽺𖾏 𖼯𖽵𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼮𖽑𖽵𖾑 𖼀𖽑𖽡𖽝𖾏 𖼦𖽡𖾐 𖼇𖽳𖾑 𖼖𖽪𖾑 𖼟𖽡𖾏.
 
12 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽐𖼮𖽱𖾏 𖼖𖽔𖾏 𖼞𖽪𖾑 ,
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖽐𖼷𖽜𖾏 𖼋𖽻𖾏,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑 𖼘𖽑𖽙𖾐 𖽐𖼊𖽪𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑,
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼳𖽪𖾐 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼸𖽠𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼐𖽱𖾏 .
13 𖽐𖼊𖽱𖾏 𖼦𖽡𖾐 𖼸𖽙 𖼟𖽪𖾏 𖼟𖽡 , 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼸𖽷𖾐,
𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽺 𖼗𖽔 , 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼽𖽺 𖼗𖽔 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼀𖽻𖾐 𖼞𖽪𖾑！
14 𖼀𖽡𖾐 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼦𖽔𖾑 𖼗𖽔𖾏 , 𖼺𖽱𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼡𖽘𖾐 𖼄𖽪𖾑,
𖼀𖽡𖾐 𖽐𖼎𖽙 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼐𖽡𖽙𖾐 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼅𖽻𖾏.
15 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼛𖽙𖾏 𖼋𖽔𖾏 𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼷𖽠𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏，
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼯𖽡𖽝𖾏 𖼙𖽠𖾏 𖽐𖼡𖽜𖾏 𖼚𖽔𖾏 𖼊𖽡𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏，
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼐𖽡𖽙𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐, 𖽐𖼚𖽪𖾏 𖼸𖽳 𖼊𖽔𖾐 𖼋𖽡𖽝 .
 
16 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏,
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼀𖽘𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑.
17 𖼦𖽡𖾐 𖼞𖽳𖾑 𖼟𖽪𖾏 𖼄𖽑𖽙𖾏 𖽐𖼊𖽪𖾏、𖼳𖽠𖾑 𖽐𖼷𖽙𖾑、𖼋𖽘 𖼐𖽑𖽪𖾐, 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼳𖽘𖾐 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼀𖽪𖾑 𖼷𖽻𖾏 𖼟𖽡𖾏 𖼖𖽑𖽙𖾏 𖼞𖽑𖽡𖾏,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏.
18 𖼐𖽱𖾏 𖼚𖽠𖾏 𖼮𖽨𖾏 𖼞𖽪𖾑 𖼯𖽡𖽝𖾏 𖼳𖽦𖾑 𖼟𖽙𖾏 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑 ,
𖼟𖽙𖾏 𖼞𖽪𖾑 𖼊𖽻𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼊𖽜𖾏 𖼮𖽡𖽻𖾐 𖼊𖽔𖾐 𖼋𖽡𖽝.
19 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼚𖽠𖾏 𖼚𖽑𖽡𖾏, 𖼦𖽡𖾐 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖽐𖼮𖽑𖽹𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖽃𖽔 𖼊𖽑𖽻𖾏 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖼊𖽻𖾏 𖼐𖽑𖽙𖾑, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖽃𖽦𖾐. （𖼳𖽡𖾏 𖼖𖽔𖾑）
 
20 𖼐𖽱𖾏 𖽐𖼷𖽜𖾑 𖼺𖽘𖾏 𖼾𖽨𖾏 𖼦𖽡𖾑 𖼊𖽑𖽹𖾏,
𖼳𖽤𖾐 𖼋𖽡𖾏 𖽐𖼊𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼻𖽻𖾏.
21 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼺𖽱𖾏 𖼛𖽙𖾏 𖼄𖽡𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖽐𖼚𖽡𖽝 ,
𖼐𖽱𖾏 𖽐𖼚𖽙 𖽃𖽔𖾐 𖼺𖽡𖽝𖾐 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼁𖽔𖾏 𖼏𖽙 ,
𖼐𖽱𖾏 𖼗𖽪𖾏 𖼊𖽻𖾏 𖼅𖽔𖾏 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼛𖽙𖾏,
𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼟𖽪𖾏 𖽐𖼡𖽪𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 .
