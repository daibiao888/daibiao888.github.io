𖼀𖽔𖾑 𖼖𖽔 𖼺𖽪𖾐 𖼁𖽱 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏
23
𖼀𖽔𖾑 𖼖𖽔 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼖𖽳 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼮𖽪𖾐 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖼐𖽡𖾑 𖼑𖽜 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖼳𖽤𖾏 𖼋𖽪𖾏 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 .” 2 𖼀𖽔𖾑 𖼖𖽳 𖼸𖽙𖾏 𖼀𖽔𖾑 𖼖𖽔 𖼑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼋𖽘𖾏 . 𖼀𖽔𖾑 𖼖𖽳 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼖𖽔 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖽃𖽡𖾑 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 . 3 𖼀𖽔𖾑 𖼖𖽔 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼖𖽳 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼟𖽡 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼞𖽪𖾑 𖼨𖽑𖽳𖾏 𖽐𖼊𖽡𖾏 𖼅𖽻𖾏, 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽔𖾏 𖼸𖽱𖾏 𖼞𖽪𖾑 , 𖼑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼞𖽪𖾑 𖼡𖽔𖾑 𖼳𖽷𖾏 , 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡 .” 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼀𖽔𖾑 𖼖𖽔 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 ．
4 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼋𖽔𖾏 𖼸𖽱𖾏 𖼀𖽔𖾑 𖼖𖽔 , 𖼀𖽔𖾑 𖼖𖽔 𖼐𖽑𖽡𖾏 : “𖼞𖽪𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 .” 5 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼄𖽔𖾐 𖼗𖽪𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼀𖽔𖾑 𖼖𖽔 , 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼖𖽳 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑 .” 6 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼖𖽳 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼁𖽙𖾏 𖼐𖽱𖾏 𖽐𖼎𖽙 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽃𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 . 7 𖼀𖽔𖾑 𖼖𖽔 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 , 𖼐𖽑𖽡𖾏 :
“𖼀𖽔𖾑 𖼖𖽳 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏𖼽𖽔𖾏 𖼖𖽔，
𖼄𖽙𖾑 𖼽𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖼋𖽠𖾏 𖼖𖽪𖾑 𖼎𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼺𖽘𖾑 𖼐𖽑𖽪𖾐 𖼋𖽔𖾐 , 𖼐𖽑𖽡𖾏 :
𖼋𖽔𖾏 𖼘𖽪 ! 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼽𖽔𖾐 𖼞𖽙 ,
𖼋𖽔𖾏 𖼘𖽪! 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 .
8 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼁𖽙𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 ,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼽𖽔𖾑 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 ,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 𖽃𖽡𖾑 ,
𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼽𖽔𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼖𖽔𖾏 ?
9 𖼞𖽪𖾑 𖼎𖽑𖽻𖾏 𖼖𖽔𖾑 𖽐𖼮𖽷𖾏 𖼎𖽻𖾐 𖽃𖽔𖾐 𖼈𖽳𖾐 𖼑𖽔𖾏 𖼐𖽱𖾏 ,
𖼎𖽑𖽻𖾏 𖼖𖽪𖾑 𖽃𖽔𖾏 𖼯𖽱𖾏 𖼎𖽻𖾐 𖼑𖽔𖾏 𖼐𖽱𖾏 ,
𖼸𖽻𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼾𖽨𖾏 𖼐𖽡𖽙𖾐 𖼾𖽨𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼉𖽻𖾏 𖼞𖽪𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 .
10 𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼖𖽠𖾑 𖼾𖽱 𖼽𖽔𖾐 𖼞𖽙 𖼑𖽡𖾏 𖼮𖽑𖽻𖾑 𖼦𖽡𖾏 𖼚𖽘𖾏？
𖼡𖽔 𖼋𖽱𖾏 𖼸𖽠𖾏 𖼮𖽔𖾑 𖼺𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼚𖽻𖾐 𖽐𖼀𖽠 𖽃𖽡𖾐 𖽐𖼀𖽠 ?
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼋𖽔 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼋𖽔；
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖼮𖽑𖽡𖽠𖾏 𖼺𖽻𖾏 .”
11 𖼀𖽔𖾑 𖼖𖽳 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼖𖽔 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼎𖽻𖾏 𖼞𖽪𖾑 𖼘𖽪 ? 𖼞𖽪𖾑 𖼨𖽑𖽳𖾏 𖼟𖽡 𖼋𖽔𖾏 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼞𖽪𖾑 𖼞𖽡𖾑 𖼀𖽔𖾑 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼟𖽡 𖼎𖽺𖾏 𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼏𖽵 𖼅𖽳𖾏 .” 12 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑 , 𖼞𖽪𖾑 𖼸𖽠𖾏 𖼦𖽡𖾐 𖼺𖽡𖽝𖾐 𖼳𖽻𖾐 𖼡𖽑𖽔𖾐 𖼐𖽑𖽡𖾏 𖼖𖽔𖾏 𖼯𖽤𖾏 ?” 13 𖼀𖽔𖾑 𖼖𖽳 𖼐𖽑𖽡𖾏 : “𖼮𖽑𖽙 𖼟𖽡 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖽃𖽡𖾐 𖼡𖽑𖽙𖾑 , 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏； 𖼟𖽡 𖼦𖽡𖾐 𖼞𖽑𖽔𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 , 𖼮𖽔𖾑 𖼸𖽠𖾏 𖼑𖽔𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖽐𖼊𖽪 𖽃𖽡𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽔𖾐 , 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼮𖽱𖾏 𖽐𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 .”
𖼀𖽔𖾑 𖼖𖽔 𖽃𖽔𖾐 𖼟𖽻𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏
14 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽳𖾏 𖼀𖽔𖾑 𖼖𖽔 𖼸𖽙 𖼚𖽔𖾏 𖼺𖽙𖾐 𖼇𖽺𖾑 𖼮𖽡𖽙𖾑 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 , 𖽐𖼮𖽡𖾏 𖼚𖽔𖾏 𖼀𖽑𖽡𖾐 𖼺𖽷𖾑 𖼮𖽢𖾑 𖼖𖽔𖾑 𖽐𖼮𖽷𖾐 𖼎𖽻𖾐 , 𖼮𖽪𖾐 𖼳𖽤𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼏𖽘 , 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖽃𖽔𖾑 𖼏𖽘 𖼖𖽑𖽘𖾑 𖽃𖽡𖾐 𖼋𖽪 𖼖𖽡𖾏 𖼡𖽪𖾑 𖼑𖽨𖾏 , 𖽃𖽡𖾐 𖼋𖽪 𖼞𖽡𖾏 𖼙𖽻𖾏 𖼾𖽘𖾏 . 15 𖼀𖽔𖾑 𖼖𖽔 𖽐𖼎𖽙 𖼀𖽔𖾑 𖼖𖽳 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼋𖽡𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖼐𖽡𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼋𖽙𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼺𖽘𖾑 𖼉𖽺𖾏 𖼅𖽻𖾏 𖼸𖽱𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 .” 16 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼋𖽔𖾏 𖼸𖽱𖾏 𖼀𖽔𖾑 𖼖𖽔 , 𖼄𖽔𖾐 𖼗𖽪𖾏 𖼡𖽑𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 : “𖼟𖽡 𖼺𖽷𖾐 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼖𖽳 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼽𖽔𖾑 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑 .” 17 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼺𖽷𖾐 𖼗𖽙𖾏 𖼚𖽔𖾏 𖼀𖽔𖾑 𖼖𖽳 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 , 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼯𖽦𖾏 𖼟𖽪𖾏 𖼞𖽑𖽔𖾏 𖼷𖽑𖽷𖾑 𖼖𖽑𖽘𖾑 𖽃𖽔𖾑 𖽐𖼊𖽪 , 𖼄𖽙𖾑 𖼽𖽔𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼖𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖽃𖽡𖾐 𖼡𖽑𖽙𖾏 . 𖼀𖽔𖾑 𖼖𖽳 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 : “𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼘𖽪 ?” 18 𖼀𖽔𖾑 𖼖𖽔 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 , 𖼐𖽑𖽡𖾏 :
“𖼀𖽔𖾑 𖼖𖽳 , 𖼟𖽡 𖼳𖽠𖾑 𖼗𖽙𖾏 𖼑𖽻 ,
𖼳𖽡𖾑 𖼀𖽙 𖼑𖽡𖾏 𖼊𖽪𖾐 , 𖼟𖽡 𖼑𖽻 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 .
19 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖼮𖽑𖽵𖾑 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼎𖽙𖾑 𖼁𖽔𖾏 ,
𖼐𖽱𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽡𖾐 𖼸𖽙𖾏 𖽃𖽔𖾏 𖼖𖽺𖾑 ?
𖼐𖽱𖾏 𖼋𖽠𖾏 𖼗𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼊𖽡𖽝𖾑 𖼘𖽪 ?
20 𖼞𖽪𖾑 𖼻𖽡𖽝𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼏𖽵 𖼅𖽳𖾏 ,
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏 𖼙𖽔 𖼺𖽘𖾏 𖼅𖽳𖾏 , 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼷𖽜𖾑 𖼦𖽡𖾐 𖼋𖽡𖾐 .
21 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼽𖽔𖾐 𖼞𖽙 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼙𖽔 𖼁𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼅𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖽐𖼚𖽪𖾏 .
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽡𖽙𖾐 ,

22 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼋𖽠𖾏 𖼣𖽹 𖼮𖽡 ,

23 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼸𖽷𖾑 𖼅𖽪𖾏 𖼟𖽪𖾏 𖼦𖽺𖾏 𖼽𖽔𖾐 𖼞𖽙 ,
𖼎𖽺𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼦𖽺𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 .
𖼐𖽔𖾑 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼽𖽔𖾐 𖼞𖽙 , 𖼟𖽪𖾏 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼐𖽑𖽡𖾏 :
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽜 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼘𖽑𖽙𖾐！
