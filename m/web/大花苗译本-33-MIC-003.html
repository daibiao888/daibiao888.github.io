𖼡𖽑𖽔𖾐 𖼖𖽔𖾏 𖼊𖽪𖾐 𖼾𖽨𖾏 𖼾𖽘𖾏 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏
3
𖼞𖽪𖾑 𖼐𖽑𖽡𖾏: 𖼽𖽔𖾐 𖼞𖽙 𖼊𖽪𖾐 𖼮𖽡𖽙𖾐 𖼨𖽑𖽳𖾏 𖼅𖽡, 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖼑𖽡𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼅𖽡 𖼘𖽪 !
𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻,
𖼋𖽡𖽝𖾏 𖼅𖽡 𖼦𖽡𖾐 𖼀𖽻𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽠𖾑 𖼞𖽡𖾑 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
2 𖼅𖽡 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏, 𖼙𖽔 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼖𖽠𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼀𖽡𖾏 𖼊𖽠𖾏,
𖼞𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼷𖽑𖽘𖾏 𖼑𖽡𖾏 𖽐𖼡𖽹𖾏.
3 𖼅𖽡 𖼑𖽻𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖽐𖼡𖽹𖾏,
𖼖𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾏 𖼊𖽠𖾏,
𖽐𖼊𖽻𖾏 𖼖𖽙𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼷𖽑𖽘𖾏,
𖼷𖽑𖽻𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼷𖽑𖽹𖾏 𖼺𖽱𖾏 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼮𖽙𖾏 𖼎𖽻𖾏 𖼖𖽪𖾑 𖼀𖽑𖽻𖾏,
𖼺𖽱𖾏 𖽃𖽔𖾑 𖼷𖽑𖽹𖾏 𖽐𖼡𖽹𖾏 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼀𖽑𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐.
4 𖼸𖽙 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖽐𖼮𖽷𖾐 𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽡𖾑, 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫,
𖼉𖽡𖽝𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼑𖽻 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏.
𖼸𖽙 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑, 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖽐𖼀𖽙 𖼚𖽑𖽪𖾏 𖼮𖽡𖽝𖾏 𖼦𖽡𖾐 𖼑𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏.
𖼖𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼐𖽑𖽡𖾏 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼟𖽪𖾏 𖼮𖽑𖽵𖾑
5 𖼐𖽑𖽡𖾏 𖼸𖽙 𖼸𖽻𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼟𖽪𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼅𖽻𖾏 𖼮𖽑𖽢𖾏 𖼾𖽔𖾏 𖼞𖽡𖾑 𖼙𖽠𖾏 𖽃𖽡𖾑,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾑 𖼐𖽑𖽡𖽝𖾑 𖼅𖽔𖾏 𖼟𖽪𖾏 𖼼𖽙𖾏 𖼺𖽷𖾐, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏: ‘ 𖼋𖽡𖽝𖾏 𖼋𖽪 𖼋𖽘𖾏 !’
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼑𖽻𖾏 𖽃𖽡𖾑, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼊𖽻𖾏 𖼐𖽱𖾏 ( 𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖽃𖽔𖾏 “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼁𖽔𖾏 𖼏𖽙”).
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 :
6 “𖼅𖽡 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼁𖽙𖾏 𖼚𖽔𖾐 𖼼𖽘𖾏,
𖼽𖽔𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖼟𖽪𖾏 𖼚𖽪𖾐 𖼖𖽡𖾐 𖽐𖼮𖽡𖽝𖾐, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼮𖽔𖾑 𖼺𖽙𖾏.
𖼖𖽪𖾑 𖼐𖽑𖽪𖾐 𖼽𖽔𖾑 𖼑𖽜 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼅𖽡 𖽐𖼷𖽜𖾑 𖼙𖽠𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖽐𖼎𖽘𖾑 𖼐𖽑𖽪𖾐 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖼸𖽻𖾏 𖽐𖼊𖽪𖾏.
7 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼺𖽪𖾐 𖼁𖽙𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼚𖽑𖽪𖾏,
𖼸𖽻𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼮𖽔𖾑 𖼺𖽙𖾏 𖼖𖽔𖾏 𖼽𖽔𖾑 𖼊𖽑𖽱𖾏 𖼸𖽘𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼽𖽔𖾑 𖼀𖽙𖾏 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖼮𖽡𖽝𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏.”
 
8 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼞𖽪𖾑, 𖼞𖽪𖾑 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼖𖽡𖽾 𖽐𖼮𖽪𖾏 𖼀𖽪𖾑 𖼛𖽘𖾏 𖼼𖽙𖾏、
𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳 𖽐𖼎𖽙 𖼅𖽔𖾏 𖼚𖽔𖾐、 𖼀𖽻𖾐 𖼟𖽳 ,
𖼸𖽠𖾏 𖽐𖼎𖽙 𖼽𖽔𖾐 𖼞𖽙 𖼐𖽑𖽡𖾏 𖽐𖼎𖽑𖽹𖾏 𖼐𖽱𖾏 𖼮𖽑𖽡𖽠𖾏 𖼾𖽔𖾏,
𖼡𖽑𖽳𖾏 𖼋𖽠𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼑𖽡𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏.
𖼸𖽻𖾏 𖼟𖽪𖾏 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖽃𖽔𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼼𖽷𖾏 𖼁𖽻𖾏
9 𖼽𖽔𖾐 𖼞𖽙 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼊𖽪𖾐 𖼮𖽡𖽙𖾐 𖼨𖽑𖽳𖾏,
𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼾𖽡𖾏 𖽃𖽡𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼘𖽪!
𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 , 𖼅𖽡 𖼳𖽪𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼋𖽡𖽝𖾏 𖼦𖽡 𖼋𖽳,
𖼐𖽡𖽙𖾐 𖼷𖽑𖽔𖾏 𖼼𖽘𖾏 𖼑𖽪𖾏 𖼷𖽑𖽪𖾑 𖼷𖽡𖽝𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 𖽃𖽡𖾑.
10 𖼎𖽑𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼮𖽑𖽘𖾑 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼳𖽡 𖼣𖽔𖾑 ,
𖼎𖽑𖽻𖾏 𖼾𖽔𖾏 𖽐𖼚𖽪𖾏 𖼮𖽘𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 .
11 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖽐𖼮𖽠𖾏 𖼨𖽑𖽢𖾐 𖼇𖽺𖾏 𖼑𖽪𖾏,
𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼑𖽜 𖽐𖼡𖽜𖾏 𖼼𖽙 𖼐𖽑𖽡𖾏 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏,
𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼑𖽜 𖼑𖽡𖽝𖾏 𖼋𖽪 𖼷𖽑𖽷𖾑 𖽃𖽔𖾏 𖼚𖽘𖾐 𖼮𖽔𖾑 𖼺𖽙𖾏,
𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫, 𖼐𖽑𖽡𖾏:
“𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽡𖽙𖾐 𖼀𖽡𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼖𖽔𖾏 𖼯𖽤𖾏 ?
𖼳𖽻𖾏 𖽐𖼚𖽔 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽠𖾏 𖼎𖽑𖽻𖾏 𖼀𖽡𖾐 .”
12 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼑𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽃𖽔𖾏,
𖼳𖽡 𖼣𖽔𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼯𖽡𖽙𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼮𖽡𖽙𖾐 𖼊𖽡𖾐,
𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼊𖽻𖾏 𖼚𖽑𖽡𖾏 𖽃𖽔𖾏 𖽃𖽔𖾑 𖼁𖽠𖾏 𖽃𖽔 𖼗𖽔𖾏,
𖼖𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼐𖽡𖾑 𖼇𖽻𖾑 𖼎𖽻𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖽃𖽔𖾏 𖼻𖽻𖾏 𖼑𖽡𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐.
