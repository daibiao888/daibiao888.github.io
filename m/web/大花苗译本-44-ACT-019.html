𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽔𖾏 𖼀𖽙𖾑 𖼘𖽙 𖼐𖽡𖽙𖾐 𖼞𖽙𖾑 𖼖𖽡 𖽐𖼊𖽙𖾑 𖽃𖽡𖾑，𖼀𖽻𖾐 𖼘𖽙 𖽐𖼮𖽡𖽻𖾏 𖼮𖽑𖽡𖽻𖾏 𖽃𖽔𖾑 𖼸𖽷 𖽃𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏，𖼎𖽺𖾏 𖼗𖽙𖾏 𖽐𖼮𖽡𖽻𖾏 𖼽𖽡𖾑 𖼇𖽪 𖼺𖽙𖾐 ，𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼮𖽷𖾐 𖼏𖽻𖾏 𖼀𖽡𖾑 𖼯𖽠 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏． 2 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼮𖽜 𖼋𖽡𖽝𖾏 𖼙𖽔 𖼊𖽻𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾 𖼋𖽘𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼑𖽡𖽝 𖼺𖽷𖾐？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏： “𖼦𖽡𖾐 𖼑𖽡𖽝 ，𖼀𖽡𖾐 𖼦𖽡𖾐 𖼙𖽔 𖼐𖽑𖽙𖾑 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼳𖽟𖾏 𖼖𖽡𖽾． ” 3 𖼀𖽻𖾐 𖼘𖽙 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔𖾏 𖼯𖽡𖾏，𖽐𖼷𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖼊𖽻𖾏 𖼟𖽪𖾏 𖽐𖼷𖽔𖾑 𖽃𖽔𖾏 𖼳𖽷𖾏？” 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽪𖾏 𖼽𖽙 𖼦𖽗 𖼑𖽡𖾏 𖽐𖼷𖽔𖾑．” 4 𖼀𖽻𖾐 𖼘𖽙 𖼐𖽑𖽡𖾏：“ 𖼽𖽙 𖼦𖽗 𖼑𖽡𖾏 𖽐𖼷𖽔𖾑 𖼟𖽪𖾏 𖽐𖼎𖽻𖾏 𖼀𖽡𖾑 𖼙𖽻 𖼑𖽡𖾏 𖽐𖼷𖽔𖾑，𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼖𖽡𖾑 𖼉𖽪𖾏，𖼽𖽔𖾑 𖽐𖼮𖽜 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼋𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼡𖽜𖾐 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑．” 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽙𖾑 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼎𖽺𖾏 𖼻𖽡𖽝𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼷𖽷𖾏 𖼊𖽻𖾏 𖽐𖼷𖽔𖾑． 6 𖼀𖽻𖾐 𖼘𖽙 𖼮𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏，𖼳𖽟𖾏 𖼖𖽡𖽾 𖼎𖽺𖾏 𖼘𖽑𖽻𖾑 𖼗𖽙𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑 ，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼗𖽪𖾏 ，𖼎𖽺𖾏 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽐𖼊𖽔𖾏 𖼊𖽑𖽡𖽝𖾐． 7 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼖𖽔𖾑 𖼅𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 ．
