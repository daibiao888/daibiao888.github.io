𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽪 𖼐𖽑𖽡𖾏: “𖼺𖽻𖾐 𖼘𖽙 𖼾𖽡𖾏 𖼋𖽡𖽝𖾏 𖼡𖽑𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼯𖽡𖽝𖾏 𖼐𖽡𖽙𖾐 𖼺𖽷𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏? 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼑𖽜 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏.” 2 𖼺𖽻𖾐 𖼘𖽙 𖼾𖽡𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑, 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏 𖼳𖽡𖾐 𖼀𖽔𖾑, 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼨𖽑𖽪𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “ 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼳𖽡𖾐 𖼀𖽔𖾑 𖼘𖽪?” 𖼐𖽱𖾏 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏: “ 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼟𖽪𖾏 𖼾𖽙 𖼯𖽡𖽙𖾏.” 3 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼺𖽻𖾐 𖼘𖽙 𖼾𖽡𖾏 𖼋𖽡𖽝𖾏 𖼡𖽑𖽔𖾑 𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼋𖽘? 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼺𖽡𖽝𖾐 𖽐𖼮𖽑𖽠𖾏 𖼎𖽑𖽻𖾏 𖼅𖽳𖾏 𖼯𖽡𖽙𖾏 𖼐𖽱𖾏.” 𖼳𖽡𖾐 𖼀𖽔𖾑 𖼊𖽡𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼅𖽔𖾏 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.” 4 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖼋𖽱𖾏?” 𖼳𖽡𖾐 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼘𖽙 · 𖼊𖽡𖾐 𖼀𖽔𖾑 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼄𖽔𖾐 𖼮𖽡 𖽐𖼞𖽔𖾏.” 5 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑, 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼷𖽷𖾐 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼎𖽑𖽻𖾏 𖼘𖽙 · 𖼊𖽡𖾐 𖼀𖽔𖾑 𖼽𖽔𖾏 𖼄𖽡𖾐 𖼖𖽡𖾏 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼄𖽔𖾐 𖼮𖽡 𖽐𖼞𖽔𖾏 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏. 6 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼷𖽔𖾑 𖼞𖽱𖾏 𖼞𖽱𖾑 𖼟𖽪𖾏 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼊𖽪𖾐 𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼊𖽔𖾏 𖽂𖽺𖾏, 𖼎𖽺𖾏 𖼞𖽑𖽡𖾐 𖼏𖽵𖾐 𖼛𖽪𖾏 𖼇𖽻𖾑. 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳!”𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳 𖼐𖽑𖽡𖾏: “𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼐𖽡𖽙𖾐 𖼐𖽡𖾑.” 7 𖼊𖽔𖾏 𖽂𖽺𖾏 𖼐𖽑𖽡𖾏: “ 𖼟𖽡 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖽐𖼮𖽑𖽹𖾏, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼑𖽜 𖼟𖽡 𖼉𖽺𖾏 𖼽𖽙 𖼐𖽔 𖼊𖽗𖾑 𖼺𖽘𖾏 𖼅𖽳𖾏 𖼎𖽻𖾏 𖼟𖽡, 𖼄𖽔𖾐 𖼟𖽡 𖽃𖽔𖾏 𖼾𖽠𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽡𖾐 𖼙𖽡𖽝𖾐 𖼞𖽑𖽔𖾏 𖼎𖽙𖾑 𖼁𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡, 𖼟𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖽃𖽡𖾐 𖼸𖽻 𖼑𖽻𖾏 𖼉𖽔𖾏.” 8 𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳𖾏 𖼎𖽺𖾏 𖼛𖽪𖾏 𖼇𖽻𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼺𖽙𖾏 𖼋𖽪 𖼡𖽔𖾏 𖼳𖽷𖾏, 𖼟𖽪𖾏 𖼺𖽱𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼚𖽡𖾑 𖼟𖽪𖾏 𖼋𖽔 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼊𖽔𖾐. 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼊𖽻𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼑𖽔𖾏 𖼞𖽻𖾏!”
9 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼟𖽙𖾏 𖼺𖽻𖾐 𖼘𖽙 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑 𖼳𖽡𖾐 𖼀𖽔𖾑 𖼋𖽔𖾏, 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼺𖽻𖾐 𖼘𖽙 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼾𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼞𖽙𖾏 , 𖼞𖽑𖽔𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼷𖽔𖾑 𖼞𖽱𖾏 𖼞𖽱𖾑 𖼋𖽘𖾏. 10 𖼟𖽡 𖽐𖼎𖽙 𖼟𖽡 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐, 𖼡𖽑𖽜𖾑 𖼅𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼑𖽜 𖼟𖽡 𖼮𖽵𖾑 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳𖾏 𖼯𖽡𖽙𖾏 𖼊𖽡𖾐 𖽃𖽔𖾏 𖼡𖽻𖾐, 𖼄𖽔𖾐 𖼋𖽡𖾏 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼳𖽻𖾏 𖼊𖽻𖾏 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏. 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐 𖼑𖽻𖾏 𖼉𖽔𖾏.” 𖼳𖽡𖾐 𖼀𖽔𖾑 𖼅𖽔𖾏 𖼟𖽻𖾏 𖼀𖽜𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐, 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 𖼗𖽜𖾏 𖼡𖽑𖽜𖾑. 11 𖼳𖽡𖾐 𖼀𖽔𖾑 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼞𖽪𖾑 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼡𖽑𖽙𖾑 𖼡𖽑𖽔𖾏 𖼇𖽺𖾏 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑, 𖼡𖽑𖽜𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼻𖽙𖾑 𖼸𖽙𖾏 𖽃𖽔𖾏.” 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽺𖾏 𖼐𖽑𖽡𖾏: “𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼐𖽡𖽙𖾐 𖼑𖽻𖾏 𖼉𖽔𖾏, 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖽃𖽡𖾐 𖼼𖽘𖾐.” 12 𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳𖾏 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖽃𖽔𖾑 𖼯𖽱𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖽐𖼷𖽷𖾏𖼄𖽡𖾐 𖼮𖽢𖾑, 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼳𖽡𖾐 𖼀𖽔𖾑 𖽐𖼞𖽔𖾏 𖽃𖽡𖾑, 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽔𖾏 𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳𖾏 𖼑𖽡𖾏 𖼡𖽑𖽜𖾑. 13 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼄𖽡𖾐 𖼇𖽺𖾑 𖼀𖽙𖾑 𖼳𖽳𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐, 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖽐𖼎𖽙 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽡𖽙𖾐 𖼦𖽡𖾏 𖽐𖼎𖽙𖾏 𖼑𖽻𖾏 𖼉𖽔𖾏, 𖼐𖽱𖾏 𖽃𖽔𖾐 𖼷𖽑𖽹𖾏 𖽃𖽔𖾑 𖼯𖽡 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼺𖽘𖾏 𖼅𖽻𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏.
