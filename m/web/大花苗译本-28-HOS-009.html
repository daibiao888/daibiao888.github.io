𖼮𖽔𖾑 𖼟𖽪𖾏 𖼘𖽡𖾑𖽑 𖽐𖼊𖽙𖾑 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏, 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼷𖽹𖾏 𖼳𖽻𖾏.
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽡𖾑 𖼷𖽑𖽡𖽝𖾐 𖼀𖽡𖾏 𖼮𖽘𖾑 , 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼺𖽱𖾏 𖼡𖽜𖾑 𖼑𖽻𖾏 𖼟𖽪𖾏 𖼺𖽘𖾏 𖼊𖽪𖾐 𖼋𖽔𖾏,

5 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 𖼘𖽑𖽙𖾐 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐,
𖼅𖽡 𖼸𖽠𖾏 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼖𖽺𖾑?
6 𖼑𖽔𖾏 𖼘𖽪！ 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽜𖾑 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼣𖽹 𖼮𖽡 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽻𖾑 𖽃𖽔𖾏 𖼮𖽡𖾑,

𖼈𖽡𖽝𖾏 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼛𖽙𖾏 𖼺𖽘𖾏 𖼊𖽪𖾐 𖼇𖽳𖾏, 𖼦𖽡𖾐 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼮𖽡𖽝𖾏.
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼇𖽳𖾑 𖼺𖽘𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏, 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼅𖽔𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 𖼋𖽘𖾏.”
13 𖼞𖽪𖾑 𖼑𖽔𖾏𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼺𖽱𖾏𖼊𖽑𖽪𖽺𖾑 𖼘𖽙 𖼯𖽡𖽙 𖼎𖽻𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼻𖽻𖾏,
𖼈𖽡𖽝𖾏 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼽𖽔𖾑 𖼄𖽔𖾐 𖼾𖽨𖾏 𖼊𖽪𖾐 𖽐𖼷𖽑𖽹𖾏 𖼨𖽑𖽳𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 ,
𖼮𖽙𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏.
14 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼘𖽪! 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼞𖽡 𖼮𖽑𖽨𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼮𖽑𖽨𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼘𖽪?
𖼽𖽔𖾑 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖼮𖽑𖽡𖽻𖾐 𖼀𖽡𖾏 𖼏𖽔𖾏、
𖼷𖽑𖽹𖾏 𖼄𖽡𖾏 𖼡𖽑𖽔𖾑 𖼖𖽻𖾏 .
 
15 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼐𖽑𖽡𖾏: “𖼐𖽱𖾏 𖼸𖽻𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖽐𖼚𖽪𖾏 𖼖𖽔𖾏 𖼐𖽡𖽙𖾐 𖼮𖽡 𖼮𖽢 ,
𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼳𖽪𖾐 𖼐𖽱𖾏 𖼸𖽻𖾏.
𖼑𖽜 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏,
𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼞𖽪𖾑 𖽐𖼞𖽔𖾏 𖼖𖽠𖾑 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏.
𖼞𖽪𖾑 𖼦𖽡𖾐 𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼟𖽪𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏.
16 𖼽𖽡𖾑 𖼇𖽔 𖼖𖽡𖽝 𖼊𖽻𖾏 𖽐𖼊𖽻𖾏,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽔𖾑 𖼯𖽤𖾐 𖼡𖽑𖽔𖾑 𖽐𖼞𖽡𖾏, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼦𖽡𖾐 𖼸𖽠𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖼖𖽪𖾐,
𖼮𖽔𖾑 𖼟𖽪𖾏 𖼳𖽻𖾏 𖼊𖽪𖾐, 𖼞𖽪𖾑 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼐𖽑𖽙𖾏 𖼊𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽐𖼮𖽑𖽠𖾏.”
 
17 𖼞𖽪𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼇𖽳𖾑 𖼗𖽔𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏,
𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼑𖽻 𖼸𖽙𖾏 𖼐𖽱𖾏 ,
𖼐𖽱𖾏 𖼸𖽻𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖼳𖽔𖾏 𖽐𖼮𖽡 𖼚𖽔𖾏 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 .
