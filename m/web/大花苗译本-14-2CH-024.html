𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏
（𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽔𖾐 12： 1 - 21）
24
𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽡𖾑 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼳𖽤𖾏 𖼳𖽡𖽻𖾏 , 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖽃𖽔𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽻𖾏 𖼚𖽻𖾐 𖼯𖽡𖽻𖾏 𖼳𖽡𖽻𖾏 . 𖼐𖽱𖾏 𖽃𖽔𖾑 𖼐𖽡𖽝𖾐 𖽐𖼷𖽷𖾏 𖼳𖽡𖾑 𖼀𖽡𖾐 𖼽𖽔𖾏 , 𖼟𖽪𖾏 𖼀𖽡𖽝 𖼳𖽷𖾏 𖼀𖽔𖾑 𖼊𖽜𖾑 𖼐𖽜𖾐 . 2 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼐𖽡𖽙𖾐 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 , 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽔𖾑 𖼅𖽔 𖼑𖽔𖾏 𖼟𖽪𖾏 𖽐𖼮𖽤𖾐 𖽐𖼞𖽳 . 3 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼷𖽙𖾏 𖽃𖽔𖾐 𖼖𖽜𖾐 𖼐𖽤𖾐 𖼎𖽻𖾏 𖼐𖽱𖾏 , 𖼎𖽺𖾏 𖼟𖽪𖾏 𖼘𖽑𖽘𖾑 𖼊𖽪𖾐 𖼘𖽑𖽘𖾏 𖼇𖽳𖾑 .
𖼎𖽺𖾏 𖼨𖽑𖽢𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏
4 𖽃𖽔𖾑 𖼡𖽜𖾐 , 𖼽𖽙 𖽃𖽔𖾐 𖼳𖽷𖾑 𖼅𖽔𖾏 𖼺𖽡𖽝𖾐 𖽐𖼊𖽱𖾏 𖼟𖽪𖾏 𖼎𖽺𖾏 𖼨𖽑𖽢𖾏 𖼷𖽑𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 , 5 𖼎𖽺𖾏 𖼦𖽡 𖼸𖽱 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖽐𖼎𖽙 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 , 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 : “𖼅𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼽𖽨 𖼊𖽔𖾏 𖼷𖽑𖽔𖾏 𖼗𖽪𖾏 𖼙𖽻 , 𖼟𖽙𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽠𖾏 𖼑𖽡𖽝𖾏 , 𖼷𖽑𖽔𖾏 𖼳𖽡𖽻𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾏 𖽐𖼷𖽷𖾑 𖼅𖽡 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 , 𖼅𖽡 𖼽𖽔𖾑 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 𖽃𖽔𖾏 𖼋𖽡𖾏 𖼑𖽪𖾏 𖼐𖽡𖾑 .” 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼞𖽡 𖽐𖼮𖽑𖽷𖾏 . 6 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼨𖽑𖽪𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼮𖽵𖾑 𖼽𖽦𖾑 𖼦𖽙 𖼽𖽦𖾑 𖼊𖽔𖾏 𖼋𖽔𖾏 , 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼦𖽡𖾐 𖼟𖽙𖾏 𖼖𖽡𖾏 𖽂𖽺𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼄𖽔𖾐 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼡𖽑𖽜𖾏 𖼄𖽙𖾑 𖼳𖽡𖾑 𖽐𖼎𖽙 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼟𖽪𖾏 𖼦𖽡 𖼸𖽱 , 𖼑𖽜 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖼟𖽪𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼊𖽡𖾏 𖼏𖽵 𖼮𖽡𖽝𖾏 𖽃𖽡𖾑 , 𖼎𖽑𖽻𖾏 𖼽𖽨 𖼊𖽔𖾏 𖽐𖼎𖽙 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐 𖼳𖽻𖾏 𖼯𖽡𖽙𖾏 𖼋𖽔𖾐 𖼖𖽺𖾑 ?” 7 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼖𖽜𖾑 𖽃𖽔 𖼁𖽙𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏 𖼽𖽔𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖾏 𖼽𖽔𖾐 𖼑𖽡𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽪𖾐 𖼙𖽔 𖼀𖽡𖾏 𖼎𖽑𖽔𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 , 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼞𖽔𖾏 𖽐𖼚𖽙 𖼚𖽘𖾐 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼖𖽑𖽘𖾑 𖼎𖽻𖾏 𖼀𖽔𖾑 𖼖𖽡 .
