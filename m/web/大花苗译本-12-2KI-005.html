5
𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼏𖽙𖾏 𖼐𖽹𖾐 𖼄𖽔𖾏 , 𖼐𖽡𖽙𖾐 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐 𖼊𖽻𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐, 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼙𖽔 𖼡𖽜𖾑 𖼷𖽹𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽻𖾏 𖼳𖽠𖾏. 𖼐𖽱𖾏 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼋𖽻 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼮𖽑𖽜 , 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼊𖽻𖾏 𖼄𖽙𖾐 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏 𖼺𖽘𖾏. 2 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼽𖽔𖾏 𖼖𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽃𖽔𖾏 𖼡𖽔𖾏 𖼸𖽻𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏, 𖼎𖽑𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽪𖽙 𖼦𖽔𖾏 𖼊𖽻𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼣𖽔𖾏 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏, 𖼖𖽔𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼉𖽔𖾏 𖼾𖽨𖾏 𖼐𖽹𖾐 𖼄𖽔𖾏 𖼑𖽡𖾏 𖼐𖽤𖾐. 3 𖼐𖽱𖾏 𖽐𖼎𖽙 𖽃𖽔 𖼁𖽙𖾏 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏: “𖼊𖽻𖾏 𖼅𖽪𖾏 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼺𖽔𖾐 𖼄𖽔𖾐 𖼖𖽡𖾏 𖼽𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐, 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼸𖽠𖾏 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼐𖽱𖾏 𖼄𖽙𖾐 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐.” 4 𖼐𖽹𖾐 𖼄𖽔𖾏 𖼁𖽺𖾏 𖼅𖽻𖾏, 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼐𖽱𖾏 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏: 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼖𖽔𖾑 𖽐𖼞𖽻𖾏 𖽐𖼷𖽑𖽹𖾏 𖼐𖽑𖽡𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑. 5 𖼽𖽔𖾏 𖼖𖽔 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼅𖽻𖾏, 𖼞𖽪𖾑 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽻𖾏 𖼄𖽻𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏.” 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽹𖾐 𖼄𖽔𖾏 𖼙𖽠𖾏 𖼋𖽘𖾏, 𖼐𖽱𖾏 𖼯𖽡𖽙𖾏 𖼟𖽻𖾏 𖼊𖽑𖽔𖾑 𖼖𖽡𖽝 𖼊𖽳 𖼑𖽡𖽝𖾏、 𖼚𖽻𖾏 𖼷𖽑𖽡𖽝𖾐 𖼳𖽳𖾐 𖼞𖽑𖽳 𖼖𖽳 𖼞𖽪𖾐、 𖼟𖽻𖾏 𖼷𖽔𖾏 𖼞𖽡𖾑 𖼛𖽔𖾏. 6 𖼎𖽺𖾏 𖼯𖽡𖽙𖾏 𖼄𖽻𖾏 𖼅𖽻𖾏 𖼎𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼄𖽻𖾏 𖽐𖼚𖽙 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼉𖽻 𖼷𖽻𖾏 𖼐𖽹𖾐 𖼄𖽔𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼟𖽡, 𖼟𖽡 𖼷𖽹𖾏 𖼊𖽻𖾏 𖼛𖽔𖾏 𖼄𖽻𖾏 𖼐𖽡𖾑, 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼐𖽱𖾏 𖼄𖽙𖾐 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏.” 7 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽔𖾏 𖼄𖽻𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖽐𖼮𖽑𖽷𖾏 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼾𖽙 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏, 𖼸𖽠𖾏 𖼟𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼯𖽡𖽝𖾏 𖼖𖽺𖾑? 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼋𖽔𖾏 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼐𖽱𖾏 𖼄𖽙𖾐 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏, 𖼅𖽡 𖼑𖽔𖾏 𖽃𖽡𖾐 𖼑𖽔𖾏, 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼽𖽔𖾑 𖽐𖼎𖽑𖽦𖾑 𖼀𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼮𖽔𖾑𖽑 𖽐𖼮𖽠𖾏 𖽐𖼊𖽻𖾏 𖼀𖽡𖾐 𖼋𖽘.
8 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼐𖽙𖾑𖽑 𖼊𖽻𖾏 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼮𖽑𖽷𖾏 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏, 𖼎𖽺𖾏 𖼷𖽷𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼅𖽻𖾏 𖼁𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼮𖽑𖽷𖾏 𖼮𖽑𖽙𖾏 𖼺𖽘𖾏 𖼖𖽺𖾑? 𖼟𖽙𖾏 𖼐𖽱𖾏 𖼋𖽔𖾏 𖼚𖽔𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼐𖽡𖾑, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼅𖽔𖾏 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼋𖽘.” 9 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽹𖾐 𖼄𖽔𖾏 𖼐𖽡𖽙𖾐 𖼀𖽑𖽪𖾑 𖼼𖽙𖾐 𖼑𖽜 𖼸𖽙 𖼚𖽔𖾏 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖽐𖼞𖽔𖾏, 𖼳𖽠𖾑 𖽐𖼎𖽙𖾏 𖽃𖽔 𖼛𖽻𖾏. 10 𖼽𖽡𖾑 𖼖𖽡𖾏 𖼳𖽔𖾑 𖼷𖽷𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖽐𖼎𖽙 𖼐𖽹𖾐 𖼄𖽔𖾏 𖼐𖽑𖽡𖾏: “𖼟𖽡 𖼅𖽻𖾏 𖼐𖽡𖽙𖾐 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽐𖼷𖽔𖾑 𖼳𖽤𖾐 𖼟𖽻𖾏, 𖼟𖽡 𖽐𖼡𖽹𖾏 𖼎𖽺𖾏 𖼽𖽔𖾑 𖼎𖽙𖾑 𖼻𖽻𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏, 𖼎𖽺𖾏 𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏.” 11 𖼐𖽹𖾐 𖼄𖽔𖾏 𖼎𖽺𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖼗𖽠𖾏 𖼋𖽘𖾏, 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖽐𖼊𖽱𖾏 𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼋𖽠𖾏 𖼋𖽔𖾏 𖼁𖽙𖾏 𖼞𖽪𖾑, 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼑𖽡𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽐𖼷𖽷𖾏, 𖼮𖽡𖽙𖾏 𖼮𖽡𖽙𖾏 𖼋𖽡𖾏 𖼎𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼄𖽙𖾐, 𖼞𖽑𖽙𖾏 𖼻𖽻𖾏 𖼊𖽪𖾑 𖼐𖽻𖾐 𖽐𖼮𖽢𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼐𖽡𖾑. 12 𖼊𖽔𖾏 𖼄𖽔𖾐 𖼺𖽳 𖼖𖽪𖾑 𖽃𖽔𖾐 𖼟𖽱𖾐 𖼽𖽔𖾏 𖼀𖽔𖾏 𖼐𖽔 𖽐𖼎𖽙 𖼇𖽔 𖽃𖽳𖾐 𖼇𖽔, 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼻𖽻𖾏 𖼖𖽔𖾑 𖼊𖽪𖾑 𖼽𖽡𖾑 𖼺𖽳 𖼖𖽡𖽝 𖼨𖽑𖽪𖾑 𖼁𖽜 𖽃𖽻𖾐 𖼘𖽪! 𖼞𖽪𖾑 𖼐𖽡𖽙𖾐 𖼉𖽺𖾏 𖽃𖽡𖾑 𖽐𖼷𖽔𖾑 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼘𖽪? 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐 𖽐𖼷𖽑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽡𖾐 𖽃𖽔𖾏 𖼮𖽡𖾑 𖼙𖽠𖾏 𖼋𖽘𖾏. 13 𖼐𖽱𖾏 𖼡𖽑𖽜𖾑 𖼋𖽔𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏: “𖼞𖽪𖾑 𖼉𖽺𖾏 𖼘𖽪！ 𖼊𖽪𖾑 𖼺𖽪𖾐 𖼀𖽻𖾐 𖼮𖽔𖾑 𖼡𖽑𖽔𖾐 𖼇𖽺𖾏 𖼟𖽡 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼙𖽘𖾑 𖼙𖽘𖾏 𖼑𖽪𖾏, 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼘𖽪？ 𖼦𖽡𖾐 𖼅𖽪𖾏 𖼐𖽑𖽡𖾏 𖼐𖽱𖾏 𖼟𖽙𖾏 𖼟𖽡 𖼅𖽻𖾏 𖽐𖼷𖽔𖾑 𖼊𖽻𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼊𖽔𖾐 ?” 14 𖽃𖽔𖾑 𖼙𖽡𖽝𖾏 𖼐𖽹𖾐 𖼄𖽔𖾏 𖽐𖼡𖽜𖾏 𖼅𖽻𖾏, 𖼸𖽙𖾏 𖼖𖽡𖾏 𖼟𖽪𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽱𖾏 𖼗𖽪𖾏, 𖼐𖽡𖽙𖾐 𖼽𖽙 𖼊𖽗𖾏 𖽐𖼊𖽪𖾏 𖼛𖽡𖾏 𖽐𖼷𖽔𖾑 𖼳𖽤𖾐 𖼟𖽻𖾏, 𖼐𖽱𖾏 𖽐𖼡𖽹𖾏 𖼎𖽺𖾏 𖼎𖽙𖾑 𖼻𖽻𖾏, 𖼺𖽱𖾏 𖼣𖽔𖾑 𖼽𖽻𖾏 𖼐𖽱𖾏 𖽐𖼡𖽹𖾏, 𖼐𖽱𖾏 𖼎𖽺𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼋𖽘𖾏.
