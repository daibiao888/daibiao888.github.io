𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼋𖽠𖾏 𖼙𖽠𖾏，𖼮𖽑𖽡𖽻𖾏 𖼚𖽔𖾏 𖼮𖽡 𖼖𖽪𖽝 𖼖𖽪𖾑 𖼦𖽘𖾑 𖼀𖽑𖽘𖾑 𖼀𖽡 𖼋𖽡𖾏．𖼐𖽡𖽙𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽪𖾐 𖼉𖽘𖾏，𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑． 2 𖼽𖽨 𖼊𖽔𖾏 𖼟𖽪𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖽃𖽡𖾑，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼦𖽡 𖼸𖽱 𖼎𖽻𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑． 3 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑，𖼽𖽨 𖼊𖽔𖾏 𖼨𖽑𖽳𖾏 𖽃𖽡𖾐 𖼸𖽻 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼊𖽑𖽡𖽝𖾐 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼅𖽡 𖽐𖼎𖽙 𖼇𖽔 𖼖𖽡𖾏 𖼺𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑，𖼯𖽡𖽙𖾏 𖼊𖽠𖾏 𖼎𖽻𖾐、𖼖𖽑𖽘𖾑 𖼎𖽻𖾐、𖽐𖼊𖽘𖾏 𖼄𖽑𖽪𖾏 𖽃𖽔𖾏 𖼏𖽦𖾏 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖼚𖽔𖾏 𖼉𖽺𖾏 𖽃𖽡𖾑． 4 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼀𖽻𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼑𖽪𖾏 𖼟𖽪𖾏 𖼖𖽔𖾑 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼾𖽨𖾏 𖽃𖽔𖾏 𖼮𖽡𖾑，𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽐𖼎𖽑𖽦𖾑 𖼡𖽔 𖼋𖽱𖾏？” 5 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽪𖾏 𖽐𖼎𖽑𖽦𖾑 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑．” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏．” 𖼽𖽨 𖼊𖽔𖾏 𖼟𖽪𖾏 𖼋𖽺 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼳𖽠𖾑 𖼎𖽻𖾏 𖼈𖽺𖾏 𖽃𖽡𖾑． 6 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏 “𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏”，𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼑𖽡𖾏 𖼡𖽜𖾐 𖼀𖽑𖽻𖾏 𖽐𖼀𖽙 𖼢𖽻 𖼎𖽻𖾏 𖼀𖽡𖾑 𖼊𖽡𖾐． 7 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖽐𖼎𖽑𖽦𖾑 𖼡𖽔 𖼋𖽱𖾏？”𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽔 𖼺𖽔𖾐 𖼖𖽳 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑．” 8 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼅𖽡 𖼋𖽘𖾏， 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼯𖽡𖽙𖾏．𖼮𖽔𖾑 𖼟𖽪𖾏 𖼅𖽡 𖽐𖼎𖽑𖽦𖾑 𖼞𖽪𖾑，𖼮𖽙𖾏 𖼟𖽙𖾏 𖼸𖽻𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼙𖽠𖾏 𖼦𖽡𖾑．” 9 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏：“𖼸𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼛𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼺𖽘𖾏．” 10 𖼳𖽡𖾑 𖼄𖽞 · 𖼀𖽡𖾐 𖼊𖽳 𖼯𖽡𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖽃𖽔𖾑 𖼏𖽦𖾏， 𖼎𖽺𖾏 𖼎𖽑𖽙𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖼀𖽑𖽔𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑，𖼀𖽑𖽔𖾏 𖼊𖽪𖾏 𖼐𖽱𖾏 𖼷𖽑𖽹𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼖𖽡 𖼺𖽷𖾏，𖼷𖽷𖾑 𖼡𖽑𖽜𖾑 𖽃𖽡𖾑 𖽐𖼷𖽷𖾏 𖼄𖽔𖾐 𖼖𖽳 𖼞𖽪𖾐． 11 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼊𖽳 𖼐𖽑𖽡𖾏：“𖼳𖽻𖾏 𖽃𖽔𖾑 𖼏𖽦𖾏 𖼎𖽻𖾏 𖼐𖽘𖾐！𖼞𖽑𖽙𖾑 𖼈𖽳𖾐 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼉𖽺𖾏 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼞𖽪𖾑，𖼞𖽪𖾑 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼦𖽡𖾐 𖼦𖽻𖾏 𖼘𖽪？” 12 𖼸𖽻𖾏 𖼏𖽙𖾏 𖽐𖼎𖽙 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼯𖽵𖾏 𖽃𖽡𖾐 𖼷𖽑𖽡𖽝𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐，𖽐𖼎𖽙 𖼽𖽨𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼎𖽺𖾏 𖼄𖽔𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 ，𖼡𖽑𖽹𖾐 𖼐𖽱𖾏． 13 𖼺𖽪𖾐 𖼨𖽑𖽳𖾏 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼽𖽔𖾏 𖼐𖽔 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐， 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽡𖽻𖾏 𖽃𖽡𖾑 𖼞𖽹𖾑 𖼽𖽔𖾏 𖼇𖽔 𖽃𖽔𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 ， 𖼽𖽔𖾏 𖼐𖽔 𖼟𖽪𖾏 𖼞𖽹𖾑 𖼽𖽔𖾏 𖼇𖽔 𖼑𖽡𖾏 𖽃𖽔𖾏 𖼾𖽠𖾏． 14 𖼞𖽹𖾑 𖼽𖽔𖾏 𖼇𖽔 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼙𖽔 𖼞𖽑𖽻𖾑 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：‘𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽑𖽪𖾑 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼋𖽔 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼻𖽻𖾏 𖼮𖽑𖽡𖽠𖾏 ’ 𖽃𖽡𖾑 𖼯𖽡𖽙𖾏 𖼋𖽘𖾏．
𖼀𖽡𖾐 𖼊𖽳 𖼦𖽡𖾐 𖼯𖽜𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑

𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼳𖽹𖾐 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 26：59-66；𖼞𖽑𖽙𖾐 14：55-64；𖼖𖽪𖾏 22：66-71）
19 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼎𖽺𖾏 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼑𖽪 𖼐𖽱𖾏 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖽐𖼎𖽙 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐． 20 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖽐𖼞𖽳𖾏 𖽐𖼞𖽳𖾏 𖽐𖼎𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏，𖼞𖽪𖾑 𖼷𖽑𖽔𖾏 𖼯𖽡𖽹𖾏 𖼐𖽡𖽙𖾐 𖽐𖼞𖽔𖾏 𖼦𖽡 𖼸𖽱 𖽐𖼎𖽙 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏，𖼟𖽪𖾏 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼽𖽨 𖼊𖽑𖽹𖾏𖼊𖽜𖾑 𖼐𖽜𖾐 𖼦𖽡 𖼸𖽱 𖽃𖽡𖾑 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐，𖼞𖽪𖾑 𖼦𖽡𖾐 𖼙𖽔 𖽐𖼚𖽺𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏． 21 𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖼑𖽪 𖼞𖽪𖾑？ 𖼟𖽡 𖼽𖽔𖾑 𖼅𖽻𖾏 𖼑𖽪 𖼸𖽻𖾏 𖼟𖽪𖾏 𖼙𖽔 𖼐𖽙𖾑𖽑 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼗𖽪𖾏 𖽃𖽡𖾑，𖼞𖽪𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏，𖼋𖽡𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼡𖽑𖽙𖾑 𖼐𖽑𖽡𖾏，𖼑𖽔𖾏 𖼘𖽪！𖼐𖽱𖾏 𖼸𖽻𖾏 𖼀𖽻𖾐 𖼯𖽡𖽙．” 22 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏 𖼋𖽡𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑，𖼅𖽔𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼳𖽠𖾑 𖽃𖽡𖾐 𖼺𖽘𖾐，𖼎𖽑𖽻𖾏 𖼐𖽱𖾏 𖽐𖼚𖽡𖽝𖾏 𖼋𖽡 𖽐𖼀𖽔𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼊𖽡𖾐 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼘𖽪𖾏？” 23 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼮𖽔𖾑 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙，𖼟𖽡 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏 𖼋𖽠𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼦𖽡𖾐 𖼾𖽙，𖼮𖽔𖾑 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽙 𖼯𖽡𖽙𖾏，𖼟𖽡 𖽃𖽔𖾏 𖼯𖽤𖾏 𖼐𖽔𖾑 𖽐𖼊𖽻𖾏 𖼞𖽪𖾑？” 24 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼽𖽔𖾏 𖼐𖽔 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖽙𖾏 𖼡𖽑𖽹𖾐 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼮𖽡𖽝𖾏 ，𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼚𖽔𖾏 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼞𖽹𖾑 𖼽𖽔𖾏 𖼇𖽔 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑．
（𖼊𖽑𖽹𖾏 26：71-75；𖼞𖽑𖽙𖾐 14：69-72；𖼖𖽪𖾏 22：58-62）
25 𖼳𖽡𖾑 𖼄𖽞 · 𖼀𖽡𖾐 𖼊𖽳 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 𖽐𖼎𖽙 𖽐𖼊𖽡𖾏 𖼋𖽠𖾏，𖼅𖽔𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼐𖽱𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽪𖾐 𖽐𖼎𖽙𖾏 𖼘𖽪𖾏？” 𖼐𖽱𖾏 𖼦𖽡𖾐 𖼯𖽜𖾏 ，𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼦𖽡𖾐 𖼾𖽙．” 26 𖼷𖽔𖾑 𖼑𖽡𖽝 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼑𖽡𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼡𖽑𖽜𖾑，𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼀𖽡𖾐 𖼊𖽳 𖼀𖽑𖽔𖾏 𖼊𖽪𖾏 𖼐𖽱𖾏 𖽃𖽔 𖽐𖼀𖽳𖾏 𖼺𖽘𖾏 𖽃𖽡𖾑 𖼑𖽡𖾏 𖼑𖽜𖾏 𖼮𖽘𖾏，𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼋𖽡𖽝𖾏 𖼦𖽡𖾐 𖼾𖽙 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖼁𖽙𖾏 𖼟𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼖𖽪𖾑 𖼉𖽘𖾏 𖼘𖽪𖾏？” 27 𖼀𖽡𖾐 𖼊𖽳 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼯𖽜𖾏，𖼊𖽪𖾏 𖼮𖽑𖽢𖾏 𖼡𖽹𖾐 𖼡𖽔𖾏 𖼋𖽘𖾏．
𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼳𖽹𖾐 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 27：1-2，11-14；𖼞𖽑𖽙𖾐 15：1-5；𖼖𖽪𖾏 23：1-5）
28 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼞𖽹𖾑 𖼽𖽔𖾏 𖼇𖽔 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖼨𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽪𖾑 𖼟𖽻，𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼺𖽷𖾐．𖼐𖽱𖾏 𖼸𖽻𖾏 𖼦𖽡𖾐 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽪𖾑 𖼟𖽻， 𖽐𖼮𖽑𖽹𖾏 𖼊𖽻𖾏 𖼘𖽙𖾏 𖼦𖽡𖾏 𖼡𖽳𖾏 𖼺𖽘𖾏，𖼑𖽜 𖼟𖽪𖾏 𖼽𖽔𖾑 𖼑𖽻𖾏 𖼡𖽜𖾑 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼟𖽱𖾏 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏． 29 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼅𖽻𖾏 𖽐𖼮𖽡𖽻𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼅𖽡 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼐𖽱𖾏 𖼑𖽪𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏 𖼐𖽔𖾑 𖼖𖽺𖾑？” 30 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽡𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼮𖽔𖾑 𖼦𖽡𖾐 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼚𖽪𖾏，𖼀𖽡𖾐 𖼎𖽺𖾏 𖼦𖽡𖾐 𖼄𖽔𖾐 𖼋𖽔𖾏 𖼎𖽻𖾏 𖼟𖽡．” 31 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼅𖽡 𖼨𖽑𖽳𖾏 𖼐𖽱𖾏 𖼅𖽻𖾏 𖼎𖽑𖽻𖾏 𖼅𖽡 𖼼𖽘𖾏 𖼻𖽡𖽝𖾑 𖼨𖽑𖽢𖾐 𖼐𖽱𖾏 𖼾𖽔𖾏．” 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼅𖽔𖾏 𖼙𖽻 𖼋𖽪𖾏 𖼟𖽪𖾏 𖼐𖽑𖽙𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．” 32 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼸𖽪𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼙𖽔 𖼐𖽑𖽡𖾏，𖼾𖽨𖾏 𖼽𖽔𖾑 𖼋𖽔 𖼖𖽡𖾏 𖼯𖽤𖾏．
33 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼎𖽺𖾏 𖼁𖽺𖾏 𖼚𖽔𖾏 𖼉𖽻 𖼷𖽻𖾏 𖼟𖽪𖾏 𖼘𖽑𖽙𖾐 𖼖𖽪𖾑 𖼟𖽻，𖼨𖽑𖽪𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼋𖽔𖾏 ，𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼑𖽡𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏？” 34 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼟𖽡 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏，𖼡𖽑𖽙𖾑 𖼯𖽢 𖼟𖽪𖾏 𖼀𖽑𖽔𖾏 𖼽𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼊𖽑𖽻𖾏 𖽃𖽡𖾑 𖼐𖽑𖽡𖾏 𖼡𖽑𖽳𖾏 𖼟𖽡？” 35 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼋𖽡𖽝𖾏 𖼾𖽙 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪？𖼟𖽡 𖼾𖽨𖾏 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼅𖽡 𖼄𖽔𖾐 𖼟𖽡 𖼎𖽻𖾏 𖼞𖽪𖾑．𖼟𖽡 𖽃𖽔𖾏 𖼑𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏 𖼺𖽘𖾏 𖼐𖽔𖾑？” 36 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼐𖽡𖾑 𖼁𖽡𖽝，𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼮𖽔𖾑 𖼟𖽪𖾏 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼁𖽡𖽝，𖼞𖽪𖾑 𖼊𖽪𖾐 𖼷𖽷𖾐 𖼟𖽪𖾏 𖼑𖽔𖾏 𖼻𖽙𖾑 𖼽𖽔𖾑 𖼑𖽜 𖼞𖽪𖾑 𖽐𖼊𖽻𖾏 𖽐𖼮𖽷𖾏，𖼦𖽡𖾐 𖼄𖽔𖾐 𖼞𖽪𖾑 𖼎𖽻𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐．𖼐𖽔𖾑 𖼐𖽡𖾑 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼞𖽪𖽙 𖼦𖽡𖾐 𖼾𖽙 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐 𖼁𖽡𖽝．” 37 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼯𖽡𖾏，𖼟𖽡 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼖𖽔𖾏？” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼞𖽪𖾑 𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．𖼞𖽪𖾑 𖼑𖽜 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼅𖽔𖾏 𖼗𖽙，𖼞𖽪𖾑 𖼑𖽜 𖼋𖽡𖾏 𖼐𖽡𖾑 𖼋𖽔𖾏 𖽐𖼮𖽡𖽻 𖽐𖼚𖽡𖽝𖾏 𖼊𖽡𖾐， 𖼟𖽪𖾏 𖼑𖽜 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝 𖽃𖽔𖾏 𖼷𖽠𖾏 𖽐𖼊𖽜𖾏．𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼺𖽙𖾏 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝，𖼽𖽔𖾑 𖼑𖽻 𖼞𖽪𖾑 𖼳𖽘𖾐 𖽃𖽔𖾏 𖼡𖽜𖾏．” 38 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽘𖾐 𖼞𖽡𖾏 𖼋𖽡𖽝 𖼟𖽪𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏？”
（𖼊𖽑𖽹𖾏 27：15-31；𖼞𖽑𖽙𖾐 15：6-20；𖼖𖽪𖾏 23：13-25）
𖼖𖽙𖾏 𖼗𖽪𖾏 𖼐𖽡𖾑 𖼐𖽑𖽡𖾏 𖼋𖽘𖾏， 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼎𖽺𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏，𖽐𖼮𖽡𖽻𖾏 𖼛𖽔𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼐𖽑𖽡𖾏：“ 𖼞𖽪𖾑 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼡𖽔𖾏 𖼳𖽷𖾏． 39 𖼈𖽡𖽝𖾏 𖼅𖽡 𖼊𖽑𖽳𖾐 𖼸𖽳𖾐 𖼖𖽔𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑， 𖼟𖽪𖾏 𖼸𖽙 𖼯𖽡𖽹𖾏 𖼮𖽦𖾐 𖼟𖽪𖾏 𖼮𖽑𖽡𖽻𖾏 𖼅𖽻𖾏，𖼅𖽡 𖼽𖽔𖾑 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼮𖽙𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽻𖾏 𖼅𖽡，𖼋𖽡𖽝𖾏 𖼅𖽡 𖽐𖼮𖽑𖽠𖾏 𖼞𖽪𖾑 𖼮𖽙𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼎𖽻𖾏 𖼅𖽡 𖼯𖽡𖽙𖾏？” 40 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼎𖽺𖾏 𖼮𖽘𖾏 𖼺𖽘𖾐 𖼨𖽑𖽪𖾏 𖼐𖽑𖽡𖾏：“𖼦𖽡𖾐 𖼽𖽔𖾑 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑， 𖼽𖽔𖾑 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑．” 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼷𖽷𖾑 𖼀𖽔𖾑 𖼖𖽔𖾑 𖼀𖽔𖾑 𖼐𖽡𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼸𖽜．
