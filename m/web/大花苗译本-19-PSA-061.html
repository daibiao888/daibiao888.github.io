𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼊𖽻𖾏 𖼅𖽳𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏 , 𖼄𖽔𖾐 𖼎𖽻𖾏 𖼊𖽪𖾐 𖼡𖽑𖽔𖾐 𖽐𖼞𖽻𖾏． 𖼯𖽡𖽙𖾏 𖼢𖽜𖾏 𖼟𖽪𖾏 𖽐𖼎𖽷𖾑 𖼺𖽙𖾑 .
61
𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼑𖽻 𖼞𖽪𖾑 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 ,
𖼑𖽻 𖼊𖽔𖾐 𖽐𖼎𖽜𖾏 𖼞𖽪𖾑 𖼞𖽑𖽜𖾑 𖼏𖽔𖾏！
2 𖼊𖽑𖽻𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖽃𖽔𖾑 𖼺𖽡𖽝𖾐 𖼷𖽻𖾏 𖽐𖼊𖽪𖾏 𖽃𖽡𖾑, 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼎𖽑𖽻𖾏 𖼊𖽡𖾐 𖼺𖽻𖾏 𖼨𖽑𖽪𖾏 𖽐𖼊𖽜 𖼟𖽡．
𖼞𖽑𖽜𖾑 𖼏𖽔𖾏 𖼟𖽡 𖼨𖽑𖽳𖾏 𖼞𖽪𖾑 𖽐𖼮𖽡𖽻𖾏 𖼚𖽔𖾏 𖼖𖽪𖾑 𖼈𖽳𖾐 𖼗𖽘𖾏 𖼟𖽪𖾏 𖼺𖽡𖽝𖾐 𖼇𖽔𖾑 𖼊𖽪𖾑 𖼞𖽪𖾑.
 
3 𖽃𖽡𖾑 𖼟𖽪𖾏 𖼟𖽡 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼮𖽑𖽡𖽠𖾏 𖽐𖼚𖽺𖾏 𖼳𖽻𖾏 𖽐𖼚𖽔 ,
𖼟𖽪𖾏 𖼞𖽪𖾑 𖼖𖽪𖾑 𖼷𖽜𖾏 𖼟𖽪𖾏 𖽃𖽺𖾐 𖼏𖽔𖾏, 𖼟𖽙𖾏 𖼞𖽪𖾑 𖼀𖽜𖾑 𖼛𖽡𖾏 𖼞𖽡𖾑 𖼀𖽔𖾑 .
4 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖼟𖽡 𖽐𖼞𖽔𖾏 𖽐𖼊𖽻𖾐 𖽐𖼚𖽙！
𖼞𖽪𖾑 𖼽𖽔𖾑 𖼻𖽡𖽝𖾏 𖽐𖼮𖽡𖽻𖾏 𖼎𖽻𖾏 𖼟𖽡 𖼡𖽘𖾐 𖽃𖽔𖾏 𖼊𖽡𖾏 𖼀𖽡𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖽐𖼚𖽺𖾏！ （𖼳𖽡𖾏 𖼖𖽔𖾑）
5 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪！ 𖼟𖽡 𖼟𖽪𖾏 𖼷𖽷𖾑 𖼟𖽪𖾏 𖼑𖽻 𖼊𖽡𖾐 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼏𖽵,
𖼟𖽡 𖼄𖽔𖾐 𖽃𖽔𖾏 𖼞𖽙𖾏 𖼺𖽘𖾏 𖼎𖽻𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼟𖽪𖾏 𖼐𖽑𖽨𖾑 𖼙𖽪𖾏 𖼟𖽡 𖽐𖼷𖽷𖾏.
 
6 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼮𖽑𖽨𖾏 𖼊𖽑𖽡𖽝𖾐 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏,
𖼟𖽙𖾏 𖼐𖽱𖾏 𖼑𖽡𖽝𖾏 𖼳𖽡𖽻𖾏 𖼐𖽡𖽙𖾐 𖼸𖽙 𖼷𖽑𖽔𖾏 𖼷𖽑𖽷𖾏.
7 𖼊𖽻𖾏 𖼅𖽪𖾏 𖼐𖽱𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖼐𖽡𖽙𖾐 𖼎𖽻𖾏 𖽐𖼊𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼦𖽡𖾑 𖽐𖼚𖽡𖽝𖾐,
𖼊𖽻𖾏 𖼅𖽪𖾏 𖼟𖽡 𖼞𖽡𖾏 𖼷𖽡𖽝𖾏 𖼘𖽑𖽪𖾐 𖽐𖼮𖽑𖽠𖾏 𖽐𖼎𖽙 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽜 𖼢𖽻 𖼊𖽔𖾐 𖼋𖽡𖽝 𖼡𖽑𖽠𖾑 𖼻𖽙𖾑 𖼐𖽱𖾏！
8 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼞𖽪𖾑 𖼽𖽔𖾑 𖼷𖽻𖾑 𖼖𖽔 𖼟𖽡 𖽐𖼷𖽷𖾏, 𖼸𖽙 𖼟𖽪𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏,
𖼎𖽺𖾏 𖼻𖽻𖾏 𖼷𖽑𖽔𖾏 𖼐𖽑𖽪𖾐 𖼁𖽻𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼞𖽪𖾑 𖼙𖽔 𖽐𖼎𖽙 𖼟𖽡 𖼏𖽵.
