15 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼋𖽻𖾏, 𖽃𖽡𖾐 𖼋𖽪 𖼺𖽡𖽝𖾐 𖼟𖽻𖾏 𖼾𖽡𖾏 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖽃𖽡𖾐 𖼞𖽑𖽡𖾏 𖼉𖽜𖾏 𖼟𖽪𖾏 𖼟𖽻𖾏 𖽃𖽔𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏； 16 𖼎𖽺𖾏 𖼎𖽑𖽻𖾏 𖼋𖽻𖾏 𖼘𖽑𖽡𖾑 𖽃𖽔𖾏 𖽃𖽔𖾐 𖼖𖽪𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼊𖽜𖾐 𖼎𖽻𖾏 𖼦𖽡𖾐 𖼇𖽔𖾑, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼺𖽡𖽝𖾐 𖼀𖽜𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏. 17 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼅𖽔𖾏 𖼸𖽙𖾏 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼼𖽘𖾏 𖽐𖼎𖽙 𖼖𖽑𖽔𖾏 𖽃𖽔 𖼉𖽙𖾏 𖼟𖽪𖾏 𖼦𖽡𖾏 𖽐𖼡𖽜𖾏, 𖽃𖽡𖾐 𖼋𖽪 𖼳𖽤𖾏 𖼗𖽪𖾏. 18 𖼖𖽪𖾑 𖼸𖽙𖾏 𖼟𖽻𖾏 𖼯𖽪𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼼𖽘𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖽐𖼀𖽙 𖼈𖽳𖾑 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐, 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼯𖽡𖽙𖾏. 19 𖼡𖽘𖾐 𖽃𖽔𖾏 𖽐𖼮𖽔𖾏 𖽐𖼞𖽔𖾏 𖼟𖽪𖾏 𖼅𖽻𖾏 𖼗𖽙𖾏 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼅𖽔𖾏 𖼚𖽻𖾐 𖽐𖼀𖽘𖾐 𖼡𖽳𖾏, 𖼸𖽻 𖼞𖽳𖾏 𖼮𖽡𖽝𖾏 𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏. 20 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖽐𖼮𖽷𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼟𖽪𖾏 𖽃𖽙𖾏 𖼋𖽠𖾏 𖼗𖽙𖾏, 𖼨𖽑𖽻𖾏 𖼊𖽡𖾏 𖼖𖽪𖾑 𖼸𖽙𖾏, 𖼡𖽔𖾑 𖼑𖽡𖽝𖾏 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼼𖽘𖾏 𖼷𖽷𖾑 𖼖𖽡𖾏 𖼀𖽑𖽱𖾑 𖼞𖽑𖽡 𖼙𖽻, 𖽃𖽔𖾐 𖼼𖽘𖾏 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼅𖽔𖾏 𖽃𖽔𖾐 𖼀𖽔𖾏 𖼗𖽪𖾏 . 21 𖼐𖽱𖾏 𖼄𖽔𖾐 𖽃𖽔𖾐 𖼋𖽪 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼷𖽑𖽙𖾏 𖼎𖽻𖾏 𖽐𖼞𖽔𖾏 𖼖𖽑𖽘𖾏 𖼡𖽘𖾐 𖽃𖽔𖾏 𖽐𖼮𖽔𖾏 𖼉𖽺𖾏 𖼊𖽑𖽔𖾏: 𖼺𖽘𖾑 𖼺𖽷𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪, 𖼟𖽙𖾏 𖽐𖼷𖽷𖾏 𖼽𖽔𖾐 𖼮𖽡𖽾𖾑, 𖼺𖽘𖾑 𖼇𖽳𖾏 𖼷𖽑𖽙𖾏 𖽃𖽡𖾐 𖼋𖽪 𖼟𖽙𖾏 𖼐𖽱𖾏 𖽐𖼷𖽷𖾏 𖼀𖽙𖾑 𖽃𖽔𖾑 𖼺𖽷𖾑. 22 𖼐𖽡𖽙𖾐 𖽃𖽔 𖽐𖼮𖽡𖽻𖾏 𖽃𖽔𖾐 𖽐𖼮𖽡𖾐 𖼞𖽳𖾏 𖼅𖽔𖾏 𖼁𖽘𖾏 𖼡𖽙𖾏 𖽐𖼊𖽱𖾏． 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 , 𖼼𖽙 𖼟𖽪𖾏 𖼷𖽷𖾏 𖽃𖽔𖾑 𖽐𖼮𖽡𖾐 𖼊𖽡𖽝𖾑 𖼋𖽘𖾏.
𖼘𖽡𖾑𖽑 𖼳𖽡𖽠𖾑 𖼋𖽻𖾏
(𖼷𖽑𖽷𖾏 𖽃𖽔𖾐 4：2 - 5)
