𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼳𖽹𖾐 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑
（𖼊𖽑𖽹𖾏 27：1-2，11-14；𖼞𖽑𖽙𖾐 15：1-5；𖼽𖽙 18：28-38）
23
𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼎𖽺𖾏 𖼳𖽠𖾑 𖼗𖽙𖾏，𖼨𖽑𖽳𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼚𖽔𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑𖼡𖽑𖽙𖾑 𖽃𖽡𖾑， 2 𖼎𖽺𖾏 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼐𖽱𖾏 𖼐𖽑𖽡𖾏：“ 𖼀𖽡𖾐 𖼁𖽙𖾏 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼡𖽑𖽔𖾐 𖼮𖽑𖽵𖾑 𖼷𖽱𖾏 𖼀𖽡𖾐 𖼷𖽑𖽻𖾏 𖼄𖽑𖽻𖾐，𖼊𖽜𖾑 𖽐𖼎𖽳𖾏 𖼀𖽡𖾐 𖼦𖽡𖾐 𖼽𖽔𖾑 𖼟𖽪𖾏 𖼊𖽪𖾏 𖼳𖽳𖾏 𖼑𖽡𖽝𖾏 𖼎𖽻𖾏 𖼞𖽹𖾑 𖼺𖽔𖾐，𖼎𖽺𖾏 𖼾𖽨𖾏 𖼐𖽑𖽡𖾏 𖼾𖽨𖾏 𖼟𖽪𖾏 𖼮𖽡𖾑 𖼊𖽪，𖼟𖽪𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏．” 3 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼑𖽪 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼟𖽪𖾏 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼘𖽪𖾏 ？” 𖼽𖽦𖾑 𖼺𖽪𖾑 𖼊𖽡𖾐 𖼐𖽑𖽡𖾏：“ 𖼟𖽡 𖼐𖽑𖽡𖾏 𖼟𖽪𖾏 𖼾𖽙 𖼯𖽡𖽙𖾏．” 4 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼎𖽺𖾏 𖽐𖼎𖽙 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖼊𖽑𖽡𖽝𖾐 𖼨𖽑𖽪𖾑 𖼁𖽜 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼐𖽑𖽡𖾏：“ 𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼞𖽪𖾑 𖽐𖼎𖽑𖽦𖾑 𖼦𖽡𖾐 𖼊𖽻𖾏 𖼐𖽱𖾏 𖼅𖽔𖾏 𖼾𖽔𖾏 𖼡𖽔𖾑 𖼳𖽷𖾏．” 5 𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼐𖽱𖾏 𖼸𖽻𖾏 𖽂𖽪𖾏 𖽂𖽪𖾏 𖼎𖽑𖽻𖾏 𖼼𖽙𖾏 𖼐𖽑𖽡𖾏：“ 𖼐𖽱𖾏 𖼡𖽑𖽔𖾐 𖼮𖽑𖽵𖾑 𖼷𖽱𖾏 𖼖𖽡𖾑 𖼉𖽪𖾏，𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼽𖽨 𖼊𖽑𖽹𖾏 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼡𖽑𖽔𖾐 𖼊𖽜𖾑 𖼐𖽜𖾐 ，𖼟𖽪𖾏 𖼎𖽑𖽻𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼳𖽠𖾑 𖽃𖽔𖾏 𖼞𖽙𖾏，𖼸𖽙 𖼚𖽔𖾏 𖼡𖽑𖽙𖾑 𖼐𖽡𖾑．”
6 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼐𖽑𖽙𖾑 𖼖𖽡𖾏 𖼐𖽡𖾑，𖼎𖽺𖾏 𖼑𖽪 ：“𖼷𖽷𖾑 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼐𖽡𖾑 𖼟𖽪𖾏 𖼮𖽢𖾑 𖼖𖽡𖾏 𖼖𖽡𖾏 𖼊𖽜𖾑 𖼐𖽜𖾐 𖼘𖽪𖾏？” 7 𖼎𖽺𖾏 𖼀𖽻𖾐 𖼐𖽱𖾏 𖼐𖽡𖽙𖾐 𖼊𖽡𖾐 𖼮𖽑𖽡𖽠𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼖𖽡 𖼯𖽵𖾏 ，𖼎𖽺𖾏 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼚𖽔𖾏 𖼳𖽡𖾑 𖼖𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑，𖼯𖽡𖽹𖾏 𖽃𖽡𖾑 𖼸𖽪𖾏 𖼟𖽪𖾏 𖼳𖽡𖾑 𖼖𖽡 𖼐𖽡𖽙𖾐 𖼽𖽦𖾑 𖼖𖽪𖾏 𖼺𖽔𖾐 𖼖𖽟𖾐．
𖼨𖽑𖽳𖾏𖼽𖽦𖾑𖼺𖽪𖾑𖼚𖽔𖾏𖼀𖽡𖾐𖼖𖽔𖾑𖼊𖽙𖾑𖼦𖽡𖾑𖽐𖼚𖽡𖽝𖾐
𖼳𖽡𖾑 𖼖𖽡 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼚𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑
8 𖼳𖽡𖾑 𖼖𖽡 𖼁𖽙𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼎𖽺𖾏 𖼙𖽔 𖼊𖽔𖾐 𖼋𖽡𖽝，𖽃𖽡𖾑 𖼟𖽪𖾏 𖼐𖽱𖾏 𖼙𖽔 𖼐𖽑𖽙𖾑 𖼐𖽱𖾏 𖼑𖽪𖾏，𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼁𖽙𖾏 𖼐𖽱𖾏 𖼟𖽪𖾏 𖼗𖽡𖾏 𖼋𖽘， 𖼎𖽺𖾏 𖽐𖼮𖽑𖽠𖾏 𖼑𖽔𖾏 𖼐𖽱𖾏 𖽃𖽔𖾏 𖽃𖽡𖾐 𖼼𖽘𖾐 𖼑𖽪𖾏 𖼟𖽪𖾏 𖽐𖼮𖽹𖾏 𖽐𖼷𖽜𖾑． 9 𖽃𖽔𖾏 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼎𖽺𖾏 𖼑𖽪 𖼐𖽱𖾏 𖼗𖽪𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐，𖽃𖽡𖾑 𖼈𖽡𖽝𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑 𖽃𖽡𖾐 𖼖𖽙𖾏 𖼖𖽔𖾏 𖼦𖽡𖾐 𖼊𖽡𖾐． 10 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏 𖼊𖽑𖽪𖾏 𖼇𖽻𖾑 𖽐𖼎𖽙 𖼊𖽪𖾐 𖼀𖽻𖾐 𖽐𖼊𖽠𖾏，𖼨𖽑𖽪𖾑 𖼁𖽜 𖼳𖽠𖾑 𖼮𖽡𖽝𖾏 ，𖼎𖽑𖽻𖾏 𖼼𖽙 𖼛𖽠𖾏 𖼑𖽪𖾐 𖼐𖽱𖾏． 11 𖼳𖽡𖾑 𖼖𖽡 𖽐𖼎𖽙 𖼐𖽱𖾏 𖼊𖽪𖾐 𖼏𖽙𖾏 𖼑𖽔𖾏 𖼦𖽡𖾐 𖼋𖽻𖾏 𖼽𖽦𖾑 𖼺𖽪𖾑，𖼎𖽺𖾏 𖼷𖽻𖾑 𖼊𖽑𖽔𖾏 𖼚𖽙𖾏 𖼐𖽱𖾏， 𖼄𖽔𖾐 𖼮𖽑𖽙𖾏 𖼟𖽪𖾏 𖽐𖼚𖽻 𖼀𖽡𖾏 𖼏𖽜𖾏 𖼎𖽻𖾏 𖼐𖽱𖾏 𖼐𖽑𖽘𖾑， 𖼄𖽔𖾐 𖼐𖽱𖾏 𖼺𖽘𖾏 𖼺𖽷𖾑 𖼡𖽜𖾐 𖼚𖽔𖾏 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑 ． 12 𖼊𖽪𖾏 𖽐𖼊𖽡𖾏 𖼳𖽡𖾑 𖼖𖽡 𖽐𖼎𖽙 𖼀𖽡𖾐 𖼖𖽔𖾑 𖼊𖽙𖾑 𖼟𖽪𖾏 𖽃𖽡𖾐 𖼖𖽜𖾐 𖼳𖽪𖾐 𖽃𖽡𖾐 𖼖𖽜𖾐，𖼎𖽺𖾏 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖽃𖽡𖾑 𖽃𖽔𖾏 𖼽𖽺 𖼗𖽔 𖼟𖽪𖾏 𖼻𖽻𖾏．
𖼨𖽑𖽳𖾏𖼽𖽦𖾑𖼺𖽪𖾑𖼚𖽔𖾏𖼳𖽡𖾑 𖼖𖽡𖼦𖽡𖾑𖽐𖼚𖽡𖽝𖾐
