𖼷𖽻𖾑 𖼖𖽔 𖼮𖽵𖾑 𖼯𖽡𖽙𖾏 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏
𖼊𖽔𖾏 𖽂𖽺𖾏 𖼑𖽡𖾏 𖽐𖼞𖽻𖾏.
110
𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽐𖼎𖽙 𖼞𖽪𖾑 𖼮𖽵𖾑 𖼐𖽑𖽡𖾏:
“𖼟𖽡 𖼼𖽻𖾏 𖼐𖽡𖽙𖾐 𖼞𖽪𖾑 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏,
𖼋𖽙𖾏 𖼞𖽪𖾑 𖼟𖽙𖾏 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖽃𖽔𖾏 𖼡𖽔𖾑 𖼋𖽙𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼷𖽑𖽹𖾏 𖼊𖽠𖾏 𖼷𖽠𖾐.”
 
2 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼟𖽙𖾏 𖼟𖽡 𖼎𖽑𖽻𖾏 𖼳𖽡 𖼣𖽔𖾑 𖼳𖽤𖾐 𖼋𖽠𖾏 𖼊𖽪𖾑 𖼮𖽡𖾏 𖼀𖽜𖾏 𖼟𖽪𖾏 𖼅𖽔𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏 𖼗𖽙𖾏,
𖼟𖽡 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼞𖽡𖾑 𖼀𖽔𖾑 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖼯𖽵𖾏 𖼛𖽘𖾏 𖼼𖽙𖾏.
3 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼟𖽡 𖼡𖽑𖽜𖾐 𖼏𖽙 ,
𖼟𖽡 𖼖𖽡𖾑 𖼉𖽪𖾏 𖼽𖽔𖾑 𖼡𖽘𖾐 𖼺𖽡𖽝𖾐 𖼑𖽻 𖼸𖽙𖾏,
𖼟𖽡 𖽃𖽔𖾑 𖼘𖽑𖽔𖾏 𖼊𖽜𖾐 𖼐𖽜𖾐 𖼞𖽑𖽡𖽠𖾏 𖼨𖽑𖽪𖾑 𖼯𖽢𖾏 𖼐𖽑𖽘𖾑 𖽃𖽔𖾏 𖼼𖽘𖾏,
𖼺𖽱𖾏 𖼳𖽠𖾑 𖽐𖼷𖽙𖾑 𖼀𖽡𖾑 𖼙𖽱 𖽃𖽻𖾐 𖼚𖽔𖾏 𖼟𖽡 𖼡𖽑𖽙𖾑 𖽃𖽡𖾑.
 
4 𖼽𖽦𖾑 𖼦𖽙 𖼦𖽫 𖼏𖽵 𖼗𖽪𖾏, 𖼦𖽡𖾐 𖼸𖽠𖾏 𖽐𖼷𖽜𖾑 𖼚𖽑𖽡𖾏,
𖼐𖽱𖾏 𖼟𖽪𖾏 𖼸𖽙𖾏 𖼄𖽳 𖼮𖽡𖾑 𖼳𖽡𖾐 𖼊𖽳 𖼑𖽡𖾏 𖼼𖽘𖾏 𖼗𖽡𖾏 𖽐𖼊𖽡𖾏 𖽃𖽔𖾏 𖼊𖽪𖾐 𖼖𖽑𖽘𖾏.
 
5 𖼮𖽵𖾑 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖼟𖽡 𖼺𖽘𖾑 𖼖𖽡 𖼺𖽷𖾏, 𖼐𖽡𖽙𖾐 𖼐𖽑𖽪𖾐 𖼑𖽡𖽻𖾏 𖼟𖽪𖾏 𖼐𖽱𖾏 𖽃𖽙𖾏 𖼺𖽡𖽝𖾐,
𖼐𖽱𖾏 𖽃𖽺𖾐 𖼋𖽡𖽝 𖼽𖽔𖾑 𖽐𖼊𖽻𖾏 𖼢𖽻 𖼞𖽡𖾏 𖼄𖽱𖾏 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐.
6 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼨𖽑𖽢𖾐 𖼷𖽑𖽔𖾏 𖼞𖽪𖽙 ,
𖼡𖽻𖾑 𖼋𖽔 𖼀𖽪𖾑 𖼷𖽑𖽔𖾏 𖼡𖽑𖽙𖾑,
𖼐𖽱𖾏 𖼽𖽔𖾑 𖼐𖽡𖽙𖾐 𖼞𖽪𖽙 𖼟𖽪𖾏 𖼮𖽡𖽻𖾐 𖼀𖽡𖾑 𖽐𖼎𖽘𖾐 𖽐𖼊𖽻𖾏 𖼢𖽻 𖼐𖽱𖾏 𖼸𖽻𖾏 𖼊𖽪𖾐 𖼯𖽡𖽙𖾐 𖼨𖽑𖽳𖾏.
7 𖼐𖽱𖾏 𖼽𖽔𖾑 𖼦𖽻𖾏 𖽃𖽔𖾑 𖼟𖽱𖾐 𖽃𖽻𖾐 𖼟𖽪𖾏 𖼐𖽡𖽙𖾐 𖽃𖽔𖾑 𖽐𖼊𖽪𖾏 𖼞𖽡𖾑 ,
𖼑𖽜 𖼖𖽡𖾏 𖼐𖽡𖾑 𖼽𖽔𖾑 𖼮𖽘𖾏 𖼖𖽡𖾏 𖼇𖽻𖾏 𖼗𖽙𖾏 .
